1//! AletheiaDB HTTP Server Binary
2//!
3//! This binary launches the AletheiaDB HTTP server, providing a REST API
4//! for interacting with the database.
5//!
6//! # Usage
7//!
8//! ```bash
9//! # Run with default settings (port 1963, restrictive CORS)
10//! cargo run --bin aletheia-server --features http-server
11//!
12//! # Run with custom port
13//! ALETHEIADB_PORT=3000 cargo run --bin aletheia-server --features http-server
14//!
15//! # Run with permissive CORS (development only!)
16//! ALETHEIADB_CORS_PERMISSIVE=true cargo run --bin aletheia-server --features http-server
17//!
18//! # Health check (auth is required by default: pass a valid API key)
19//! curl -H "x-api-key: $ALETHEIADB_BOOTSTRAP_ADMIN_KEY" http://localhost:1963/status
20//! ```
21//!
22//! # Environment Variables
23//!
24//! - `ALETHEIADB_PORT`: Port to listen on (default: 1963)
25//! - `ALETHEIADB_HOST`: Host to bind to (default: 0.0.0.0)
26//! - `ALETHEIADB_CORS_PERMISSIVE`: Set to "true" to allow any origin (default: false)
27//! - `ALETHEIADB_CORS_ORIGINS`: Comma-separated list of allowed origins
28//! - `ALETHEIADB_CONFIG`: Path to a TOML config file consumed by
29//!   [`AletheiaDBConfig::from_toml_file`]. Takes precedence over
30//!   `ALETHEIADB_DATA_DIR`.
31//! - `ALETHEIADB_DATA_DIR`: Directory for WAL + index persistence. When set,
32//!   the server persists state across restarts. When unset (and `ALETHEIADB_CONFIG`
33//!   is also unset), runs in-memory (ephemeral — useful for demos, useless
34//!   for production).
35//! - `ALETHEIADB_AUTH_MODE`: `required` (default) or `anonymous`. In
36//!   `required` mode every request must present a valid API key
37//!   (`Authorization: Bearer <key>` or `x-api-key`), and the server refuses
38//!   to start with zero credentials. `anonymous` disables authentication
39//!   entirely (explicit opt-in; a prominent warning is logged). Invalid
40//!   values fall back to `required` (fail-closed) with a warning.
41//! - `ALETHEIADB_BOOTSTRAP_ADMIN_KEY`: Plaintext admin key installed at
42//!   startup as principal `bootstrap-admin` (role `admin`). Memory-only —
43//!   re-supply it on every start, or use it once to mint persisted keys via
44//!   `POST /admin/keys` (persisted under `{data_dir}/auth/keys.json` when a
45//!   data dir is configured).
46//!
47//! # Endpoints
48//!
49//! All AletheiaDB endpoints require a credential in `required` mode
50//! (`Authorization: Bearer <key>` or `x-api-key`); see
51//! `docs/guides/access-control-matrix.md` for role classes.
52//!
53//! - `GET /status` - Health check endpoint (metrics class), returns
54//!   `{"status": "healthy"}`
55//! - `POST /query` - JSON query API (`find_node`, `create_node`, bulk ops,
56//!   `execute_query`, ...); read or write class per operation
57//! - `POST /admin/keys` - Mint an API key (admin class)
58//! - `GET /admin/keys` - List principals, masked (admin class)
59//! - `POST /admin/keys/revoke` - Revoke a key by principal id (admin class)
60//!
61//! The autumn-web framework additionally serves unauthenticated
62//! health/metadata routes (`/health`, `/live`, `/ready`, `/startup`,
63//! `/actuator/health|info|metrics|a11y|ui`); its sensitive actuator group is
64//! force-disabled in every profile. See
65//! `docs/guides/security-quickstart.md`.
66//!
67//! # Security Notes
68//!
69//! - By default, CORS is restrictive (only localhost:3000 allowed)
70//! - Set `ALETHEIADB_CORS_ORIGINS` to configure allowed origins for production
71//! - Only use `ALETHEIADB_CORS_PERMISSIVE=true` in development environments
72//!
73//! # Graceful Shutdown
74//!
75//! The server handles SIGTERM and SIGINT signals for graceful shutdown,
76//! allowing in-flight requests to complete before terminating.
77
78use aletheiadb::auth::{SecretString, auth_mode_from_env};
79use aletheiadb::http::{CorsConfig, ServerConfig, run_server};
80use std::env;
81use std::path::PathBuf;
82
83/// Parse port from environment variable with warning on invalid values.
84fn parse_port() -> u16 {
85    match env::var("ALETHEIADB_PORT") {
86        Ok(port_str) => match port_str.parse::<u16>() {
87            Ok(port) => port,
88            Err(e) => {
89                eprintln!(
90                    "WARNING: Invalid ALETHEIADB_PORT '{}': {}. Using default port 1963.",
91                    port_str, e
92                );
93                1963
94            }
95        },
96        Err(_) => 1963,
97    }
98}
99
100/// Parse host from environment variable.
101fn parse_host() -> String {
102    env::var("ALETHEIADB_HOST").unwrap_or_else(|_| "0.0.0.0".to_string())
103}
104
105/// Parse CORS configuration from environment variables.
106fn parse_cors_config() -> CorsConfig {
107    // Check if permissive mode is enabled
108    let is_permissive = env::var("ALETHEIADB_CORS_PERMISSIVE")
109        .map(|v| v.to_lowercase() == "true" || v == "1")
110        .unwrap_or(false);
111
112    if is_permissive {
113        return CorsConfig::permissive();
114    }
115
116    // Parse allowed origins from comma-separated list
117    match env::var("ALETHEIADB_CORS_ORIGINS") {
118        Ok(origins_str) => {
119            let mut config = CorsConfig::restrictive();
120            for origin in origins_str
121                .split(',')
122                .map(str::trim)
123                .filter(|s| !s.is_empty())
124            {
125                config = config.allow_origin(origin);
126            }
127            config
128        }
129        Err(_) => CorsConfig::restrictive(),
130    }
131}
132
133/// Parse the data directory env var into an optional `PathBuf`.
134///
135/// Unset or empty → `None` (in-memory mode). Any non-empty value is taken
136/// as a path literal; `AletheiaDB::with_unified_config` creates the directory
137/// structure on startup. Delegates to the shared helper so every exposed
138/// binary reads the same env var with the same semantics.
139fn parse_data_dir() -> Option<PathBuf> {
140    aletheiadb::config::data_dir_from_env()
141}
142
143/// Parse `ALETHEIADB_BOOTSTRAP_ADMIN_KEY` into a redacted secret, if set.
144///
145/// The value is trimmed: the HTTP extractor trims presented tokens, so an
146/// env value with stray whitespace (e.g. from `echo` without `-n` piped into
147/// the environment) could otherwise never verify.
148fn parse_bootstrap_admin_key() -> Option<SecretString> {
149    match env::var("ALETHEIADB_BOOTSTRAP_ADMIN_KEY") {
150        Ok(raw) if !raw.trim().is_empty() => Some(SecretString::new(raw.trim())),
151        _ => None,
152    }
153}
154
155#[autumn_web::main]
156async fn main() {
157    let port = parse_port();
158    let host = parse_host();
159    let cors = parse_cors_config();
160    let data_dir = parse_data_dir();
161
162    let mut builder = ServerConfig::builder()
163        .port(port)
164        .host(host)
165        .cors(cors)
166        .auth_mode(auth_mode_from_env());
167    if let Some(path) = data_dir {
168        builder = builder.data_dir(path);
169    }
170    if let Some(key) = parse_bootstrap_admin_key() {
171        builder = builder.bootstrap_admin_key(key);
172    }
173    let config = builder.build();
174
175    if let Err(e) = run_server(config).await {
176        eprintln!("aletheia-server failed: {e}");
177        std::process::exit(1);
178    }
179}
