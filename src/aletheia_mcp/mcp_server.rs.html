1//! AletheiaDB MCP Server binary.
2//!
3//! This binary exposes AletheiaDB's bi-temporal graph database capabilities
4//! through the Model Context Protocol (MCP), enabling LLMs like Claude to
5//! interact with the database.
6//!
7//! # Usage
8//!
9//! ```bash
10//! cargo run --bin aletheia-mcp --features mcp-server
11//! ```
12//!
13//! The server communicates over stdio using the MCP protocol.
14//!
15//! # Available Tools
16//!
17//! ## Node Operations
18//! - `get_node`: Get a node by ID
19//! - `create_node`: Create a new node with label and properties
20//! - `update_node`: Update node properties
21//! - `delete_node`: Delete a node
22//! - `list_nodes`: List nodes with optional filtering
23//! - `count_nodes`: Count nodes
24//!
25//! ## Edge Operations
26//! - `get_edge`: Get an edge by ID
27//! - `create_edge`: Create an edge between nodes
28//! - `update_edge`: Update edge properties
29//! - `delete_edge`: Delete an edge
30//! - `list_edges`: List edges with optional filtering
31//! - `count_edges`: Count edges
32//! - `get_outgoing_edges`: Get outgoing edges from a node
33//! - `get_incoming_edges`: Get incoming edges to a node
34//!
35//! ## Graph Traversal
36//! - `traverse`: Traverse the graph from a starting node
37//!
38//! ## Vector Search
39//! - `find_similar`: Find similar nodes by embedding
40//! - `enable_vector_index`: Enable vector indexing on a property
41//! - `list_vector_indexes`: List enabled vector indexes
42//!
43//! ## Temporal Queries
44//! - `get_node_at_time`: Get node state at a specific time
45//! - `get_edge_at_time`: Get edge state at a specific time
46//!
47//! ## Hybrid Queries
48//! - `hybrid_query`: Combined graph + vector + temporal query
49//!
50//! ## Declarative Query
51//! - `query`: Execute a single read-only Cypher/AQL statement (with optional
52//!   `$param` bindings) and get structured rows back. Mutating statements are
53//!   rejected before execution. Cypher requires the `cypher` feature; AQL is
54//!   always available.
55//!
56//! # Authentication (Issue #3350)
57//!
58//! The MCP transport is stdio, so the credential is **session-scoped**:
59//! supplied once at process start and re-verified on every tool call (a
60//! revocation in the shared key store takes effect on the next call).
61//!
62//! Environment variables:
63//!
64//! - `ALETHEIADB_AUTH_MODE`: `required` (default) or `anonymous`. In
65//!   `required` mode every tool call is authorized against the session
66//!   principal's role, and the server refuses to start with zero
67//!   credentials. `anonymous` disables authentication entirely (explicit
68//!   opt-in; a prominent warning is printed to stderr). Invalid values fall
69//!   back to `required` (fail-closed) with a warning.
70//! - `ALETHEIADB_MCP_API_KEY`: The session credential. In `required` mode a
71//!   missing/unknown/revoked key does not stop the server — every tool call
72//!   then returns the uniform `UNAUTHENTICATED` error.
73//! - `ALETHEIADB_BOOTSTRAP_ADMIN_KEY`: Plaintext admin key installed at
74//!   startup as principal `bootstrap-admin` (role `admin`). Memory-only —
75//!   re-supply it on every start, or use it once against the HTTP server's
76//!   `POST /admin/keys` to mint persisted keys.
77//! - Key persistence: when `ALETHEIADB_DATA_DIR` is set, keys are loaded
78//!   from `{data_dir}/auth/keys.json` — the same path the HTTP server uses,
79//!   so keys minted via the HTTP admin endpoints work here when both point
80//!   at the same data directory.
81
82use std::sync::Arc;
83
84use rmcp::{ServiceExt, transport::stdio};
85
86use aletheiadb::AletheiaDB;
87use aletheiadb::auth::{AuthMode, AuthStore, Role, SecretString, auth_mode_from_env};
88use aletheiadb::mcp::{AletheiaMcpServer, McpAuthConfig, validate_mcp_auth_startup};
89
90/// Read a secret-bearing env var into a redacted [`SecretString`], if set
91/// and non-empty.
92///
93/// The value is trimmed: presented tokens are trimmed at the transport
94/// layer, so an env value with stray whitespace could otherwise never
95/// verify.
96fn parse_secret_env(name: &str) -> Option<SecretString> {
97    match std::env::var(name) {
98        Ok(raw) if !raw.trim().is_empty() => Some(SecretString::new(raw.trim())),
99        _ => None,
100    }
101}
102
103/// Build the auth key store: persisted under `{data_dir}/auth/keys.json`
104/// when `ALETHEIADB_DATA_DIR` is set (the same derivation the HTTP server
105/// uses), memory-only otherwise. Installs the bootstrap admin key
106/// (memory-only, never persisted) when configured.
107fn build_auth_store() -> Result<AuthStore, String> {
108    let store = match aletheiadb::config::data_dir_from_env() {
109        Some(dir) => AuthStore::with_persistence(dir.join("auth").join("keys.json"))
110            .map_err(|e| format!("failed to open auth key store: {e}"))?,
111        None => AuthStore::new(),
112    };
113    if let Some(key) = parse_secret_env("ALETHEIADB_BOOTSTRAP_ADMIN_KEY") {
114        store
115            .insert_bootstrap_key("bootstrap-admin", Role::Admin, key.expose())
116            .map_err(|e| format!("failed to install bootstrap admin key: {e}"))?;
117    }
118    Ok(store)
119}
120
121#[tokio::main]
122async fn main() -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
123    // Initialize the database. Persistence is selected by environment:
124    //   ALETHEIADB_CONFIG=/path/to/config.toml   -> full TOML-driven config
125    //   ALETHEIADB_DATA_DIR=/path                -> canonical durable layout
126    //   (neither set)                            -> ephemeral tempdir-backed DB
127    let db = AletheiaDB::open_from_env()?;
128
129    // Resolve authentication (Issue #3350): required by default; the
130    // session credential authenticates every tool call.
131    let auth_mode = auth_mode_from_env();
132    let store = build_auth_store()?;
133    if let Err(msg) = validate_mcp_auth_startup(auth_mode, &store) {
134        eprintln!("aletheia-mcp refusing to start: {msg}");
135        std::process::exit(1);
136    }
137    let store = Arc::new(store);
138
139    let session_key = parse_secret_env("ALETHEIADB_MCP_API_KEY");
140    if auth_mode == AuthMode::Required
141        && session_key
142            .as_ref()
143            .is_none_or(|key| store.verify(key.expose()).is_none())
144    {
145        eprintln!(
146            "WARNING: no valid session credential was supplied \
147             (ALETHEIADB_MCP_API_KEY is unset or not recognized by the key \
148             store); every tool call will return UNAUTHENTICATED until the \
149             server is restarted with a valid key."
150        );
151    }
152
153    let mut auth = McpAuthConfig::new(auth_mode, store);
154    if let Some(key) = session_key {
155        auth = auth.with_credential(key);
156    }
157
158    // Create the MCP server (with_auth prints the prominent anonymous-mode
159    // warning to stderr when applicable — stdout is the protocol channel).
160    let server = AletheiaMcpServer::with_auth(Arc::new(db), auth);
161
162    // Serve over stdio using the MCP protocol
163    let service = server.serve(stdio()).await?;
164
165    // Wait for the server to shut down
166    service.waiting().await?;
167
168    Ok(())
169}
