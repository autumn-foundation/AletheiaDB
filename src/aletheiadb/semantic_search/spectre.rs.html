1//! Spectre: Semantic Perspective Engine.
2//!
3//! "It depends on how you look at it."
4//!
5//! Spectre allows you to view the graph through a specific "Lens" (Concept).
6//! It modifies graph traversals by warping the effective distance between nodes
7//! based on their alignment with the Lens.
8//!
9//! # Concepts
10//! - **Lens**: A vector defining a semantic axis (e.g., "Risk", "Innovation").
11//! - **Focus**: The alignment of a node with the Lens.
12//! - **Subjective Traversal**: Pathfinding where "cost" is determined by alignment.
13//!   A path through "Safety" looks different from a path through "Speed".
14//!
15//! # Example
16//!
17//! ```rust,no_run
18//! use aletheiadb::AletheiaDB;
19//! use aletheiadb::semantic_search::spectre::{Spectre, Lens};
20//! use aletheiadb::core::id::NodeId;
21//!
22//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
23//! let db = AletheiaDB::new()?;
24//! let spectre = Spectre::new(&db);
25//!
26//! // Define Lenses
27//! let safety_lens = Lens::new(&[0.0, 1.0]); // Y-axis is Safety
28//! let speed_lens = Lens::new(&[1.0, 0.0]);  // X-axis is Speed
29//!
30//! # let start = NodeId::new(1).unwrap();
31//! # let end = NodeId::new(5).unwrap();
32//!
33//! // Find the "Safe" path
34//! let safe_path = spectre.traverse(start, end, &safety_lens, "embedding")?;
35//!
36//! // Find the "Fast" path
37//! let fast_path = spectre.traverse(start, end, &speed_lens, "embedding")?;
38//! # Ok(())
39//! # }
40//! ```

49/// A semantic lens defining a perspective.
50#[derive(Debug, Clone)]
51pub struct Lens {
52    /// The normalized direction vector of the perspective.
53    pub vector: Vec<f32>,
54}
55
56impl Lens {
57    /// Create a new Lens from a vector.
58    pub fn new(vector: &[f32]) -> Self {
59        Self {
60            vector: normalize(vector),
61        }
62    }
63}

65/// The Spectre Engine for subjective graph analysis.
66pub struct Spectre<'a> {
67    db: &'a AletheiaDB,
68}
69
70#[derive(Clone, Copy, PartialEq)]
71struct State {
72    cost: f32,
73    node: NodeId,
74}
75
76impl Eq for State {}
77
78impl Ord for State {
79    fn cmp(&self, other: &Self) -> Ordering {
80        // Min-heap priority queue
81        other
82            .cost
83            .partial_cmp(&self.cost)
84            .unwrap_or(Ordering::Equal)
85    }
86}
87
88impl PartialOrd for State {
89    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
90        Some(self.cmp(other))
91    }
92}

94impl<'a> Spectre<'a> {
95    /// Create a new Spectre instance.
96    pub fn new(db: &'a AletheiaDB) -> Self {
97        Self { db }
98    }
99
100    /// Calculate the alignment of a node with the lens (Focus Score).
101    ///
102    /// # Returns
103    /// A score between -1.0 (Opposed) and 1.0 (Aligned).
104    pub fn focus(&self, node_id: NodeId, lens: &Lens, property: &str) -> Result<f32> {
105        let node = self.db.get_node(node_id)?;
106        let vec = node
107            .properties
108            .get(property)
109            .and_then(|v| v.as_vector())
110            .ok_or_else(|| {
111                Error::other(format!(
112                    "Node {} missing vector property '{}'",
113                    node_id, property
114                ))
115            })?;
116
117        // Cosine similarity
118        crate::core::vector::cosine_similarity(vec, &lens.vector)
119    }
120
121    /// Find a path from `start` to `end` that maximizes alignment with the Lens.
122    ///
123    /// This performs Dijkstra's algorithm where the cost of visiting a node
124    /// is inversely proportional to its alignment with the Lens.
125    ///
126    /// Cost function: `1.0 - focus(node)`
127    /// - Perfect Alignment (1.0) -> Cost 0.0
128    /// - Orthogonal (0.0) -> Cost 1.0
129    /// - Opposite (-1.0) -> Cost 2.0
130    ///
131    /// Nodes missing the vector property are penalized with Cost 2.0.
132    pub fn traverse(
133        &self,
134        start: NodeId,
135        end: NodeId,
136        lens: &Lens,
137        vector_prop: &str,
138    ) -> Result<Vec<NodeId>> {
139        if start == end {
140            return Ok(vec![start]);
141        }
142
143        let mut pq = BinaryHeap::new();
144        pq.push(State {
145            cost: 0.0,
146            node: start,
147        });
148
149        let mut dist = HashMap::new();
150        dist.insert(start, 0.0);
151
152        let mut came_from = HashMap::new();
153
154        while let Some(State { cost, node }) = pq.pop() {
155            if node == end {
156                return Ok(self.reconstruct_path(came_from, end));
157            }
158
159            // Stale entry check
160            if cost > *dist.get(&node).unwrap_or(&f32::INFINITY) {
161                continue;
162            }
163
164            for edge_id in self.db.get_outgoing_edges(node) {
165                // Ignore edge fetch errors
166                if let Ok(neighbor) = self.db.get_edge_target(edge_id) {
167                    // Calculate cost to enter neighbor based on semantic alignment
168                    let focus_score = self.focus(neighbor, lens, vector_prop).unwrap_or(-1.0); // High penalty for missing info
169
170                    // Map [-1, 1] to [2, 0] cost
171                    // We add a tiny epsilon to ensure non-zero cost for cycles unless perfect alignment
172                    let step_cost = 1.0 - focus_score + 0.001;
173                    let new_cost = cost + step_cost;
174
175                    if new_cost < *dist.get(&neighbor).unwrap_or(&f32::INFINITY) {
176                        dist.insert(neighbor, new_cost);
177                        came_from.insert(neighbor, node);
178                        pq.push(State {
179                            cost: new_cost,
180                            node: neighbor,
181                        });
182                    }
183                }
184            }
185        }
186
187        Ok(Vec::new()) // No path found
188    }
189
190    fn reconstruct_path(&self, came_from: HashMap<NodeId, NodeId>, current: NodeId) -> Vec<NodeId> {
191        let mut path = vec![current];
192        let mut curr = current;
193        while let Some(&prev) = came_from.get(&curr) {
194            path.push(prev);
195            curr = prev;
196        }
197        path.reverse();
198        path
199    }
200}

204    use super::*;
205    use crate::core::property::PropertyMapBuilder;

208    fn create_db() -> AletheiaDB {
209        let db = AletheiaDB::new().unwrap();
210        db.enable_vector_index("vec", HnswConfig::new(2, DistanceMetric::Cosine))
211            .unwrap();
212        db
213    }

215    #[test]
216    fn test_spectre_focus() {
217        let db = create_db();
218        // Node: [1, 0]
219        let props = PropertyMapBuilder::new()
220            .insert_vector("vec", &[1.0, 0.0])
221            .build();
222        let node = db.create_node("Node", props).unwrap();
223
224        let spectre = Spectre::new(&db);
225
226        // Lens X: [1, 0] -> Focus 1.0
227        let lens_x = Lens::new(&[1.0, 0.0]);
228        let focus_x = spectre.focus(node, &lens_x, "vec").unwrap();
229        assert!((focus_x - 1.0).abs() < 1e-5);
230
231        // Lens Y: [0, 1] -> Focus 0.0
232        let lens_y = Lens::new(&[0.0, 1.0]);
233        let focus_y = spectre.focus(node, &lens_y, "vec").unwrap();
234        assert!((focus_y - 0.0).abs() < 1e-5);
235    }

237    #[test]
238    fn test_spectre_subjective_traversal() {
239        let db = create_db();
240
241        // Topology:
242        // Start -> HighRoad -> End
243        // Start -> LowRoad -> End
244
245        // HighRoad: [0, 1] (Aligned with Y)
246        // LowRoad:  [1, 0] (Aligned with X)
247
248        let p_start = PropertyMapBuilder::new()
249            .insert_vector("vec", &[0.0, 0.0])
250            .build();
251        let start = db.create_node("Start", p_start).unwrap();
252
253        let p_end = PropertyMapBuilder::new()
254            .insert_vector("vec", &[0.0, 0.0])
255            .build();
256        let end = db.create_node("End", p_end).unwrap();
257
258        let p_high = PropertyMapBuilder::new()
259            .insert_vector("vec", &[0.0, 1.0])
260            .build();
261        let high = db.create_node("HighRoad", p_high).unwrap();
262
263        let p_low = PropertyMapBuilder::new()
264            .insert_vector("vec", &[1.0, 0.0])
265            .build();
266        let low = db.create_node("LowRoad", p_low).unwrap();
267
268        // Edges
269        db.create_edge(start, high, "path", Default::default())
270            .unwrap();
271        db.create_edge(high, end, "path", Default::default())
272            .unwrap();
273
274        db.create_edge(start, low, "path", Default::default())
275            .unwrap();
276        db.create_edge(low, end, "path", Default::default())
277            .unwrap();
278
279        let spectre = Spectre::new(&db);
280
281        // 1. Perspective Y (Favor HighRoad)
282        let lens_y = Lens::new(&[0.0, 1.0]);
283        let path_y = spectre.traverse(start, end, &lens_y, "vec").unwrap();
284        assert_eq!(
285            path_y,
286            vec![start, high, end],
287            "Should take HighRoad with Y lens"
288        );
289
290        // 2. Perspective X (Favor LowRoad)
291        let lens_x = Lens::new(&[1.0, 0.0]);
292        let path_x = spectre.traverse(start, end, &lens_x, "vec").unwrap();
293        assert_eq!(
294            path_x,
295            vec![start, low, end],
296            "Should take LowRoad with X lens"
297        );
298    }
