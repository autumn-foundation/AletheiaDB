1//! Tapestry: Semantic Path Weaving Engine.
2//!
3//! "Weave a path through specific concepts."
4//!
5//! Tapestry builds on `SemanticNavigator` to find paths that don't just go from A to B,
6//! but explicitly visit a sequence of "waypoints" along the way.
7//! This allows forcing a semantic trajectory through specific conceptual neighborhoods.
8//!
9//! # Example
10//! ```rust,no_run
11//! use aletheiadb::AletheiaDB;
12//! use aletheiadb::semantic_search::tapestry::TapestryEngine;
13//! use aletheiadb::core::id::NodeId;
14//!
15//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
16//! let db = AletheiaDB::new()?;
17//! let tapestry = TapestryEngine::new(&db);
18//!
19//! # let start = NodeId::new(1).unwrap();
20//! # let waypoint = NodeId::new(2).unwrap();
21//! # let end = NodeId::new(3).unwrap();
22//! // Weave a path from start to end, passing through 'waypoint'
23//! let path = tapestry.weave(start, &[waypoint], end, "embedding")?;
24//!
25//! println!("Wove a path of {} nodes", path.len());
26//! # Ok(())
27//! # }
28//! ```
29
30use crate::AletheiaDB;
31use crate::core::error::{Error, Result};
32use crate::core::id::NodeId;
33use crate::semantic_search::semantic_navigator::SemanticNavigator;
34
35/// The Tapestry Engine for waypoint-based semantic pathfinding.
36pub struct TapestryEngine<'a> {
37    db: &'a AletheiaDB,
38}
39
40impl<'a> TapestryEngine<'a> {
41    /// Create a new TapestryEngine.
42    pub fn new(db: &'a AletheiaDB) -> Self {
43        Self { db }
44    }
45
46    /// Weave a semantic path from `start_node` to `end_node`, visiting `waypoints` in order.
47    ///
48    /// # Arguments
49    /// * `start_node` - The beginning of the path.
50    /// * `waypoints` - A slice of intermediate nodes to visit in sequence.
51    /// * `end_node` - The final destination.
52    /// * `vector_property` - The name of the property containing vector embeddings.
53    ///
54    /// # Returns
55    /// A single continuous `Vec<NodeId>` containing the full path, with duplicate
56    /// waypoints (where segments join) merged.
57    pub fn weave(
58        &self,
59        start_node: NodeId,
60        waypoints: &[NodeId],
61        end_node: NodeId,
62        vector_property: &str,
63    ) -> Result<Vec<NodeId>> {
64        let navigator = SemanticNavigator::new(self.db);
65        let mut full_path = Vec::new();
66
67        // Build the sequence of points to visit: Start -> W1 -> W2 -> ... -> End
68        let mut sequence = Vec::with_capacity(waypoints.len() + 2);
69        sequence.push(start_node);
70        sequence.extend_from_slice(waypoints);
71        sequence.push(end_node);
72
73        for window in sequence.windows(2) {
74            let from = window[0];
75            let to = window[1];
76
77            let segment = navigator.find_path(from, to, vector_property)?;
78
79            if full_path.is_empty() {
80                full_path.extend(segment);
81            } else {
82                // Segment starts with 'from', which is already the last element of full_path
83                // Skip the first element to avoid duplicates
84                full_path.extend(segment.into_iter().skip(1));
85            }
86        }
87
88        if full_path.is_empty() {
89            // Edge case: start_node == end_node and no waypoints
90            if start_node == end_node && waypoints.is_empty() {
91                return Ok(vec![start_node]);
92            } else {
93                return Err(Error::other("Failed to weave path"));
94            }
95        }
96
97        Ok(full_path)
98    }
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104    use crate::config::AletheiaDBConfig;
105    use crate::config::WalConfigBuilder;
106    use crate::core::property::PropertyMapBuilder;
107    use tempfile::tempdir;
108
109    fn create_test_db() -> (AletheiaDB, tempfile::TempDir) {
110        let dir = tempdir().unwrap();
111        let wal_path = dir.path().join("wal");
112        let data_path = dir.path().join("data");
113        std::fs::create_dir_all(&wal_path).unwrap();
114        std::fs::create_dir_all(&data_path).unwrap();
115
116        let persistence_config = crate::storage::index_persistence::PersistenceConfig {
117            data_dir: data_path,
118            enabled: false,
119            ..Default::default()
120        };
121
122        let config = AletheiaDBConfig::builder()
123            .wal(WalConfigBuilder::new().wal_dir(wal_path).build())
124            .persistence(persistence_config)
125            .build();
126        let db = AletheiaDB::with_unified_config(config).unwrap();
127        (db, dir)
128    }
129
130    #[test]
131    fn test_tapestry_weave() {
132        let (db, _dir) = create_test_db();
133
134        // Topology:
135        // A -> B -> C -> D -> E
136        //
137        // Waypoints: Weave from A to E through C.
138
139        let props = PropertyMapBuilder::new()
140            .insert_vector("vec", &[1.0, 0.0]) // Vectors don't matter much here, just need them for A*
141            .build();
142
143        let a = db.create_node("Node", props.clone()).unwrap();
144        let b = db.create_node("Node", props.clone()).unwrap();
145        let c = db.create_node("Node", props.clone()).unwrap();
146        let d = db.create_node("Node", props.clone()).unwrap();
147        let e = db.create_node("Node", props.clone()).unwrap();
148
149        db.create_edge(a, b, "NEXT", PropertyMapBuilder::new().build())
150            .unwrap();
151        db.create_edge(b, c, "NEXT", PropertyMapBuilder::new().build())
152            .unwrap();
153        db.create_edge(c, d, "NEXT", PropertyMapBuilder::new().build())
154            .unwrap();
155        db.create_edge(d, e, "NEXT", PropertyMapBuilder::new().build())
156            .unwrap();
157
158        let tapestry = TapestryEngine::new(&db);
159        let path = tapestry.weave(a, &[c], e, "vec").unwrap();
160
161        assert_eq!(path, vec![a, b, c, d, e]);
162    }
