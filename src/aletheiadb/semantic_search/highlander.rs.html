1//! Highlander: Semantic Entity Resolution.
2//!
3//! "There can be only one."
4//!
5//! This module provides tools to identify and merge duplicate nodes in the graph.
6//! It uses vector similarity to find candidates and graph transactions to merge them.
7//!
8//! # Use Cases
9//! - **Entity Resolution**: "J. Smith" vs "John Smith".
10//! - **Deduplication**: Cleaning up data ingestion errors.
11//! - **Knowledge Fusion**: Merging two knowledge graphs.
12

20/// Detector for finding potential duplicate nodes.
21///
22/// # Example
23///
24/// ```rust,no_run
25/// use aletheiadb::AletheiaDB;
26/// use aletheiadb::semantic_search::highlander::HighlanderDetector;
27///
28/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
29/// let db = AletheiaDB::new()?;
30/// let detector = HighlanderDetector::new(&db);
31///
32/// // Find candidates similar to node_id
33/// // let candidates = detector.find_duplicates(node_id, 0.9, 5)?;
34/// # Ok(())
35/// # }
36/// ```
37pub struct HighlanderDetector<'a> {
38    db: &'a AletheiaDB,
39}
40
41impl<'a> HighlanderDetector<'a> {
42    /// Create a new HighlanderDetector.
43    pub fn new(db: &'a AletheiaDB) -> Self {
44        Self { db }
45    }
46
47    /// Find potential duplicates for a given node.
48    ///
49    /// # Arguments
50    /// * `target` - The node to check.
51    /// * `threshold` - Minimum similarity score (0.0 to 1.0).
52    /// * `limit` - Maximum number of candidates.
53    pub fn find_duplicates(
54        &self,
55        target: NodeId,
56        threshold: f32,
57        limit: usize,
58    ) -> Result<Vec<(NodeId, f32)>> {
59        self.db
60            .similarity_search(crate::SimilarityQuery::from_node(target).k(limit))
61            .map(|candidates| {
62                candidates
63                    .into_iter()
64                    .filter(|&(_, score)| score >= threshold)
65                    .collect()
66            })
67    }
68}

70/// Merger for combining two nodes into one.
71pub struct EntityMerger<'a> {
72    db: &'a AletheiaDB,
73}
74
75impl<'a> EntityMerger<'a> {
76    /// Create a new EntityMerger.
77    pub fn new(db: &'a AletheiaDB) -> Self {
78        Self { db }
79    }
80
81    /// Merge a victim node into a survivor node.
82    ///
83    /// This moves all edges from the victim to the survivor, merges properties,
84    /// and deletes the victim.
85    ///
86    /// # Arguments
87    /// * `survivor` - The node that will remain.
88    /// * `victim` - The node that will be merged and deleted.
89    ///
90    /// # Errors
91    /// Returns `Error::Other` if `survivor` and `victim` are the same node.
92    pub fn merge(&self, survivor: NodeId, victim: NodeId) -> Result<()> {
93        if survivor == victim {
94            return Err(Error::other(
95                "Cannot merge a node into itself (survivor == victim)",
96            ));
97        }
98
99        self.db.write(|tx| {
100            // 1. Move Edges
101            // We collect all edges connected to the victim.
102            // Note: Self-loops appear in both outgoing and incoming lists.
103            // We handle them by deduplicating or checking ID.
104            let mut edges_processed = std::collections::HashSet::new();
105
106            // Outgoing: Victim -> Target
107            let outgoing = tx.get_outgoing_edges(victim)?;
108            for edge_id in outgoing {
109                if edges_processed.insert(edge_id) {
110                    let edge = tx.get_edge(edge_id)?;
111                    let new_target = if edge.target == victim {
112                        survivor // Handle self-loop: Victim->Victim becomes Survivor->Survivor
113                    } else {
114                        edge.target
115                    };
116                    // Resolve label
117                    let label = GLOBAL_INTERNER
118                        .resolve_with(edge.label, |s| s.to_string())
119                        .ok_or_else(|| {
120                            Error::Storage(StorageError::InconsistentState {
121                                reason: format!(
122                                    "Edge label with ID {} not found in interner",
123                                    edge.label.as_u32()
124                                ),
125                            })
126                        })?;
127
128                    // Create edge from Survivor -> New Target
129                    tx.create_edge(survivor, new_target, &label, edge.properties)?;
130                    // Delete old edge
131                    tx.delete_edge(edge_id)?;
132                }
133            }
134
135            // Incoming: Source -> Victim
136            let incoming = tx.get_incoming_edges(victim)?;
137            for edge_id in incoming {
138                if edges_processed.insert(edge_id) {
139                    let edge = tx.get_edge(edge_id)?;
140                    let new_source = if edge.source == victim {
141                        survivor // Handle self-loop (should be caught above, but for safety)
142                    } else {
143                        edge.source
144                    };
145                    // Resolve label
146                    let label = GLOBAL_INTERNER
147                        .resolve_with(edge.label, |s| s.to_string())
148                        .ok_or_else(|| {
149                            Error::Storage(StorageError::InconsistentState {
150                                reason: format!(
151                                    "Edge label with ID {} not found in interner",
152                                    edge.label.as_u32()
153                                ),
154                            })
155                        })?;
156
157                    // Create edge from New Source -> Survivor
158                    tx.create_edge(new_source, survivor, &label, edge.properties)?;
159                    // Delete old edge
160                    tx.delete_edge(edge_id)?;
161                }
162            }
163
164            // 2. Merge Properties
165            // Strategy: Survivor Wins. We only add properties from Victim that Survivor lacks.
166            let victim_node = tx.get_node(victim)?;
167            let survivor_node = tx.get_node(survivor)?;
168
169            let mut props_to_add = PropertyMapBuilder::new();
170            let mut has_changes = false;
171
172            for (key, val) in victim_node.properties.iter() {
173                // Optimized check using interned keys
174                if !survivor_node.properties.contains_interned_key(key) {
175                    props_to_add = props_to_add.try_insert_by_key(*key, val.clone())?;
176                    has_changes = true;
177                }
178            }
179
180            if has_changes {
181                tx.update_node(survivor, props_to_add.build())?;
182            }
183
184            // 3. Delete Victim
185            // We have manually deleted all edges, so use delete_node to avoid cascade issues with self-loops
186            tx.delete_node(victim)?;
187
188            Ok(())
189        })
190    }
191}

224    #[test]
225    fn test_detect_duplicates() {
226        let (db, _dir) = create_test_db();
227
228        // Enable vector index
229        let config = HnswConfig::new(2, DistanceMetric::Cosine);
230        db.enable_vector_index("embedding", config).unwrap();
231
232        // 1. Create Target Node: "John Smith" [1.0, 0.0]
233        let props1 = PropertyMapBuilder::new()
234            .insert("name", "John Smith")
235            .insert_vector("embedding", &[1.0, 0.0])
236            .build();
237        let target = db.create_node("Person", props1).unwrap();
238
239        // 2. Create Duplicate: "J. Smith" [0.99, 0.01] (Very similar)
240        let props2 = PropertyMapBuilder::new()
241            .insert("name", "J. Smith")
242            .insert_vector("embedding", &[0.99, 0.01])
243            .build();
244        let duplicate = db.create_node("Person", props2).unwrap();
245
246        // 3. Create Distinct: "Alice" [0.0, 1.0] (Orthogonal)
247        let props3 = PropertyMapBuilder::new()
248            .insert("name", "Alice")
249            .insert_vector("embedding", &[0.0, 1.0])
250            .build();
251        let distinct = db.create_node("Person", props3).unwrap();
252
253        let detector = HighlanderDetector::new(&db);
254        let duplicates = detector.find_duplicates(target, 0.9, 5).unwrap();
255
256        // Should find "J. Smith"
257        assert!(!duplicates.is_empty(), "Should find at least one duplicate");
258        assert_eq!(duplicates[0].0, duplicate);
259        assert!(duplicates[0].1 > 0.9);
260
261        // Should NOT find "Alice"
262        assert!(
263            !duplicates.iter().any(|(id, _)| *id == distinct),
264            "Should not find orthogonal vector"
265        );
266    }

268    #[test]
269    fn test_merge_entities() {
270        let (db, _dir) = create_test_db();
271
272        // 1. Create Survivor: "Survivor"
273        let props1 = PropertyMapBuilder::new()
274            .insert("name", "Survivor")
275            .insert("age", 30)
276            .build();
277        let survivor = db.create_node("Person", props1).unwrap();
278
279        // 2. Create Victim: "Victim"
280        let props2 = PropertyMapBuilder::new()
281            .insert("name", "Victim") // Should be overwritten? Or ignored? Strategy dependent.
282            .insert("city", "London") // Should be added to Survivor
283            .build();
284        let victim = db.create_node("Person", props2).unwrap();
285
286        // 3. Create Edge to Victim: Friend -> Victim
287        let friend = db
288            .create_node("Person", PropertyMapBuilder::new().build())
289            .unwrap();
290        db.create_edge(
291            friend,
292            victim,
293            "KNOWS",
294            PropertyMapBuilder::new().insert("since", 2020).build(),
295        )
296        .unwrap();
297
298        // 4. Create Edge from Victim: Victim -> Place
299        let place = db
300            .create_node("Place", PropertyMapBuilder::new().build())
301            .unwrap();
302        db.create_edge(victim, place, "LIVES_IN", PropertyMapBuilder::new().build())
303            .unwrap();
304
305        let merger = EntityMerger::new(&db);
306        merger.merge(survivor, victim).unwrap();
307
308        // Verify Victim is deleted
309        assert!(
310            db.get_node(victim).is_err(),
311            "Victim node should be deleted"
312        );
313
314        // Verify Edges Moved
315        // Friend -> Survivor
316        let outgoing_friend = db.get_outgoing_edges_with_label(friend, "KNOWS");
317        assert_eq!(outgoing_friend.len(), 1);
318        let edge_id = outgoing_friend[0];
319        let edge = db
320            .get_edge(edge_id)
321            .unwrap_or_else(|_| panic!("Failed to get friend edge {:?}", edge_id));
322        assert_eq!(edge.target, survivor, "Edge should point to Survivor");
323        assert_eq!(
324            edge.properties.get("since").unwrap().as_int(),
325            Some(2020),
326            "Edge properties should be preserved"
327        );
328
329        // Survivor -> Place
330        let outgoing_survivor = db.get_outgoing_edges_with_label(survivor, "LIVES_IN");
331        assert_eq!(outgoing_survivor.len(), 1);
332        let edge_id = outgoing_survivor[0];
333        let edge = db
334            .get_edge(edge_id)
335            .unwrap_or_else(|_| panic!("Failed to get edge {:?}", edge_id));
336        assert_eq!(edge.target, place, "Edge should start from Survivor");
337
338        // Verify Properties Merged
339        let survivor_node = db.get_node(survivor).unwrap();
340        assert_eq!(
341            survivor_node.properties.get("name").unwrap().as_str(),
342            Some("Survivor"),
343            "Survivor name should be preserved"
344        );
345        assert_eq!(
346            survivor_node.properties.get("city").unwrap().as_str(),
347            Some("London"),
348            "City should be copied from Victim"
349        );
350    }

364    #[test]
365    fn test_merge_self_loops() {
366        let (db, _dir) = create_test_db();
367        let survivor = db
368            .create_node("Survivor", PropertyMapBuilder::new().build())
369            .unwrap();
370        let victim = db
371            .create_node("Victim", PropertyMapBuilder::new().build())
372            .unwrap();
373
374        // Create self-loop on victim: Victim -> Victim
375        let _old_edge = db
376            .create_edge(victim, victim, "SELF", PropertyMapBuilder::new().build())
377            .unwrap();
378
379        let merger = EntityMerger::new(&db);
380        merger.merge(survivor, victim).unwrap();
381
382        // Should become Survivor -> Survivor
383        let outgoing = db.get_outgoing_edges_with_label(survivor, "SELF");
384        assert_eq!(outgoing.len(), 1);
385        let edge = db.get_edge(outgoing[0]).unwrap();
386        assert_eq!(edge.source, survivor);
387        assert_eq!(edge.target, survivor);
388    }

413        assert_eq!(outgoing_link.len(), 1);
414        let edge_link = db.get_edge(outgoing_link[0]).unwrap();
415        assert_eq!(edge_link.source, survivor);
416        assert_eq!(edge_link.target, survivor);
417
418        // Victim -> Survivor should become Survivor -> Survivor (Self loop)
419        let outgoing_back = db.get_outgoing_edges_with_label(survivor, "BACK");
420        assert_eq!(outgoing_back.len(), 1);
421        let edge_back = db.get_edge(outgoing_back[0]).unwrap();
