3//! Horizon: Semantic Event Horizon.
4//!
5//! "Where does a concept end?"
6//!
7//! The Horizon Engine maps the "Semantic Event Horizon" of a seed node.
8//! It explores the graph outward using Breadth-First Search (BFS) and stops traversing
9//! along a path as soon as the semantic similarity between the seed node and the current
10//! node drops below a specified threshold.
11//!
12//! The nodes just outside this similarity bubble form the "Event Horizon",
13//! clearly defining the boundary where a concept fundamentally shifts into something else.
14//!
15//! # Use Cases
16//! - **Concept Boundaries**: Finding where discussions about "Machine Learning" turn into "General AI".
17//! - **Topic Segmentation**: Understanding the hard edges of a cluster or community.
18//! - **Semantic Reach**: Measuring how far an idea can travel before losing its meaning.
19//!
20//! # Example
21//! ```rust,no_run
22//! // Requires features = ["nova"]
23//! use aletheiadb::AletheiaDB;
24//! use aletheiadb::semantic_search::horizon::HorizonEngine;
25//! use aletheiadb::core::id::NodeId;
26//!
27//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
28//! let db = AletheiaDB::new()?;
29//! # let node_id = db.create_node("Concept", Default::default())?;
30//! let horizon = HorizonEngine::new(&db);
31//!
32//! // Find where the similarity drops below 0.6. Max 5 hops.
33//! let result = horizon.map_horizon(node_id, "embedding", 0.6, 5)?;
34//!
35//! println!("Interior nodes: {}", result.interior.len());
36//! println!("Horizon nodes (boundary): {}", result.horizon.len());
37//! # Ok(())
38//! # }
39//! ```

41use crate::AletheiaDB;
42use crate::api::transaction::ReadOps;
43use crate::core::error::{Error, Result, StorageError};
44use crate::core::id::NodeId;
45use crate::core::vector::ops::cosine_similarity;
46use std::collections::{HashSet, VecDeque};
47
48/// DoS-protection cap on the number of distinct nodes a single
49/// [`HorizonEngine::map_horizon`] call may visit.
50///
51/// `map_horizon` runs a bidirectional BFS bounded only by `max_depth` (clamped
52/// to 20 at the MCP surface) and a visited set. On a dense component even a
53/// modest depth can touch an enormous fraction of the graph, so an untrusted
54/// caller could force an expensive traversal. Once the visited set exceeds this
55/// generous cap the search aborts with a [`StorageError::CapacityExceeded`]
56/// resource-limit error rather than continuing. The cap is deliberately large
57/// so it never trips on realistic concept-boundary queries; it exists purely to
58/// bound adversarial worst cases.
59pub const MAX_HORIZON_VISITED_NODES: usize = 100_000;
60
61/// The result of mapping the Semantic Event Horizon.
62#[derive(Debug, Clone, PartialEq)]
63pub struct HorizonResult {
64    /// Nodes that are inside the semantic bubble (similarity >= threshold).
65    pub interior: HashSet<NodeId>,
66    /// Nodes that represent the boundary (similarity < threshold), where exploration stopped.
67    pub horizon: HashSet<NodeId>,
68}
69
70/// The Horizon Engine.
71pub struct HorizonEngine<'a> {
72    db: &'a AletheiaDB,
73}
74
75impl<'a> HorizonEngine<'a> {
76    /// Create a new Horizon Engine.
77    pub fn new(db: &'a AletheiaDB) -> Self {
78        Self { db }
79    }
80
81    /// Map the semantic event horizon of a seed node.
82    ///
83    /// Performance Note: Iterating over outgoing and incoming edges uses `.chain()` instead
84    /// of collecting into an intermediate vector, eliminating unnecessary heap allocations and improving performance on large graphs.
85    ///
86    /// # Arguments
87    /// * `seed` - The starting node.
88    /// * `property_name` - The vector property to use for semantic comparison.
89    /// * `threshold` - The similarity threshold (e.g., 0.6). Nodes below this form the horizon.
90    /// * `max_depth` - The maximum number of hops to explore.
91    pub fn map_horizon(
92        &self,
93        seed: NodeId,
94        property_name: &str,
95        threshold: f32,
96        max_depth: usize,
97    ) -> Result<HorizonResult> {
98        let mut interior = HashSet::new();
99        let mut horizon = HashSet::new();
100        let mut visited = HashSet::new();
101        let mut queue = VecDeque::new();
102
103        self.db.read(|tx| {
104            // Get seed vector
105            let seed_node = tx.get_node(seed)?;
106            let seed_vec = match seed_node
107                .get_property(property_name)
108                .and_then(|p| p.as_vector())
109            {
110                Some(v) => v,
111                None => {
112                    return Err(Error::other(format!(
113                        "Seed node {} does not have vector property '{}'",
114                        seed, property_name
115                    )));
116                }
117            };
118
119            // Start BFS
120            queue.push_back((seed, 0));
121            visited.insert(seed);
122            interior.insert(seed); // Seed is always in the interior
123
124            while let Some((current_node_id, depth)) = queue.pop_front() {
125                if depth >= max_depth {
126                    continue;
127                }
128
129                // Get all neighbors (both outgoing and incoming for true semantic reach)
130                let outgoing = tx
131                    .get_outgoing_edges(current_node_id)?
132                    .into_iter()
133                    .filter_map(|e| tx.get_edge(e).ok().map(|edge| edge.target));
134
135                let incoming = tx
136                    .get_incoming_edges(current_node_id)?
137                    .into_iter()
138                    .filter_map(|e| tx.get_edge(e).ok().map(|edge| edge.source));
139
140                for neighbor_id in outgoing.chain(incoming) {
141                    if visited.contains(&neighbor_id) {
142                        continue;
143                    }
144
145                    visited.insert(neighbor_id);
146
147                    // DoS protection: abort if the traversal has touched more
148                    // than the visited-node cap (a dense component can otherwise
149                    // reach a huge fraction of the graph within max_depth).
150                    if visited.len() > MAX_HORIZON_VISITED_NODES {
151                        return Err(StorageError::CapacityExceeded {
152                            resource: "semantic_horizon visited nodes".to_string(),
153                            current: visited.len(),
154                            limit: MAX_HORIZON_VISITED_NODES,
155                        }
156                        .into());
157                    }
158
159                    if let Ok(neighbor_node) = tx.get_node(neighbor_id) {
160                        if let Some(neighbor_vec) = neighbor_node
161                            .get_property(property_name)
162                            .and_then(|p| p.as_vector())
163                        {
164                            // Calculate similarity with the *seed* node
165                            let sim = cosine_similarity(seed_vec, neighbor_vec)?;
166
167                            if sim >= threshold {
168                                // Still inside the bubble
169                                interior.insert(neighbor_id);
170                                queue.push_back((neighbor_id, depth + 1));
171                            } else {
172                                // Crossed the event horizon
173                                horizon.insert(neighbor_id);
174                            }
175                        }
176                    }
177                }
178            }
179
180            Ok::<(), Error>(())
181        })?;
182
183        Ok(HorizonResult { interior, horizon })
184    }
185}

187#[cfg(test)]
188mod tests {
189    use super::*;
190    use crate::api::transaction::WriteOps;
191    use crate::core::property::PropertyMapBuilder;
192
193    #[test]
194    fn test_horizon_mapping() {
195        let db = AletheiaDB::new().unwrap();
196
197        let mut a = NodeId::new(0).unwrap();
198        let mut b = NodeId::new(0).unwrap();
199        let mut c = NodeId::new(0).unwrap();
200        let mut d = NodeId::new(0).unwrap();
201        let mut e = NodeId::new(0).unwrap();
202
203        db.write(|tx| {
204            // A (seed): Core Concept
205            a = tx
206                .create_node(
207                    "Concept",
208                    PropertyMapBuilder::new()
209                        .insert_vector("vec", &[1.0, 0.0, 0.0])
210                        .build(),
211                )
212                .unwrap();
213
214            // B: Very similar to A (cosine sim = 0.9)
215            // dot product = 0.9, norm = 1.0 * 1.0
216            // x * 1.0 = 0.9 => x = 0.9
217            // y^2 + z^2 = 1 - 0.9^2 = 1 - 0.81 = 0.19
218            // let y = sqrt(0.19) ~ 0.4358
219            b = tx
220                .create_node(
221                    "Concept",
222                    PropertyMapBuilder::new()
223                        .insert_vector("vec", &[0.9, 0.435889, 0.0])
224                        .build(),
225                )
226                .unwrap();
227
228            // C: Somewhat similar to A (cosine sim = 0.6)
229            // dot product = 0.6 => x = 0.6
230            // y = sqrt(1 - 0.36) = 0.8
231            c = tx
232                .create_node(
233                    "Concept",
234                    PropertyMapBuilder::new()
235                        .insert_vector("vec", &[0.6, 0.8, 0.0])
236                        .build(),
237                )
238                .unwrap();
239
240            // D: Different from A (cosine sim = 0.2)
241            // x = 0.2
242            // y = sqrt(1 - 0.04) = 0.9797
243            d = tx
244                .create_node(
245                    "Concept",
246                    PropertyMapBuilder::new()
247                        .insert_vector("vec", &[0.2, 0.979795, 0.0])
248                        .build(),
249                )
250                .unwrap();
251
252            // E: Completely different (cosine sim = 0.0)
253            e = tx
254                .create_node(
255                    "Concept",
256                    PropertyMapBuilder::new()
257                        .insert_vector("vec", &[0.0, 1.0, 0.0])
258                        .build(),
259                )
260                .unwrap();
261
262            // Graph structure: A -> B -> C -> D -> E
263            tx.create_edge(a, b, "LINK", Default::default()).unwrap();
264            tx.create_edge(b, c, "LINK", Default::default()).unwrap();
265            tx.create_edge(c, d, "LINK", Default::default()).unwrap();
266            tx.create_edge(d, e, "LINK", Default::default()).unwrap();
267
268            Ok::<(), Error>(())
269        })
270        .unwrap();
271
272        let engine = HorizonEngine::new(&db);
273
274        // Threshold = 0.5.
275        // A=1.0, B=0.9, C=0.6 are inside.
276        // D=0.2 is the boundary (horizon). E is never reached because traversal stops at D.
277        let result = engine.map_horizon(a, "vec", 0.5, 10).unwrap();
278
279        assert!(result.interior.contains(&a));
280        assert!(result.interior.contains(&b));
281        assert!(result.interior.contains(&c));
282        assert!(!result.interior.contains(&d));
283        assert!(!result.interior.contains(&e));
284
285        assert!(result.horizon.contains(&d));
286        assert!(!result.horizon.contains(&e)); // Not reached
287
288        // Threshold = 0.8
289        // A=1.0, B=0.9 are inside.
290        // C=0.6 is the boundary. D and E are not reached.
291        let result2 = engine.map_horizon(a, "vec", 0.8, 10).unwrap();
292        assert!(result2.interior.contains(&a));
293        assert!(result2.interior.contains(&b));
294        assert!(result2.horizon.contains(&c));
295        assert!(!result2.horizon.contains(&d));
296    }
