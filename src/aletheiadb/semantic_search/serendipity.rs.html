1//! Serendipity: The Scenic Route Finder 🧭
2//!
3//! "Sometimes the best path is not the shortest, but the most surprising."
4//!
5//! Serendipity finds paths between two nodes that intentionally maximize
6//! semantic divergence along the way, while still eventually reaching the destination.
7//! It answers the question: "How can I get from A to B while encountering the most
8//! unexpected (but structurally valid) concepts?"
9//!
10//! # Concepts
11//! - **Scenic Route**: A path where adjacent nodes have *low* cosine similarity,
12//!   representing a "leap" in meaning, rather than a gradual shift.
13//! - **Serendipity Score**: The sum of semantic distances (1.0 - similarity) along the path.
14//!   Higher score = more surprising journey.
15//!
16//! # Use Cases
17//! - **Creative Ideation**: "Connect 'Black Hole' to 'Pancake' using the most surprising conceptual jumps."
18//! - **Recommendation Diversity**: Breaking out of "filter bubbles" by suggesting a path
19//!   that traverses diverse topics before arriving at a related interest.
20//! - **Educational Exploration**: Teaching by finding lateral connections between subjects.
21//!
22//! # Example
23//!
24//! ```rust
25//! // Requires features = ["nova"]
26//! use aletheiadb::AletheiaDB;
27//! use aletheiadb::semantic_search::serendipity::SerendipityEngine;
28//!
29//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
30//! let db = AletheiaDB::new()?;
31//! // ... setup graph ...
32//! # let start = aletheiadb::core::id::NodeId::new(0)?;
33//! # let end = aletheiadb::core::id::NodeId::new(0)?;
34//!
35//! let serendipity = SerendipityEngine::new(&db);
36//!
37//! // Find a path from start to end, max 5 hops, maximizing semantic distance
38//! // Note: This is an expensive operation on large graphs, use carefully!
39//! # /*
40//! let path = serendipity.find_scenic_route(start, end, "embedding", 5)?;
41//!
42//! println!("Found a scenic route with {} hops!", path.len());
43//! # */
44//! # Ok(())
45//! # }
46//! ```

48use crate::AletheiaDB;
49use crate::core::error::{Error, Result};
50use crate::core::id::NodeId;
51use crate::core::vector::ops;
52use std::collections::{HashSet, VecDeque};

54/// A discovered scenic route.
55#[derive(Debug, Clone)]
56pub struct ScenicRoute {
57    /// The sequence of nodes forming the path.
58    pub path: Vec<NodeId>,
59    /// The total serendipity score (sum of semantic distances).
60    pub serendipity_score: f32,
61}
62
63/// The Serendipity Engine.
64pub struct SerendipityEngine<'a> {
65    db: &'a AletheiaDB,
66}
67
68impl<'a> SerendipityEngine<'a> {
69    /// Create a new SerendipityEngine.
70    pub fn new(db: &'a AletheiaDB) -> Self {
71        Self { db }
72    }
73
74    /// Find a path from `start` to `end` that maximizes semantic divergence (serendipity).
75    ///
76    /// This uses a modified breadth-first search / beam search that tracks all simple paths
77    /// up to `max_depth`. It evaluates the serendipity score of each valid path and returns
78    /// the one with the highest score.
79    ///
80    /// # Arguments
81    /// * `start` - The starting node.
82    /// * `end` - The target node.
83    /// * `vector_property` - The name of the vector property to use for semantic comparisons.
84    /// * `max_depth` - The maximum number of hops allowed. High values will be very slow.
85    pub fn find_scenic_route(
86        &self,
87        start: NodeId,
88        end: NodeId,
89        vector_property: &str,
90        max_depth: usize,
91    ) -> Result<Option<ScenicRoute>> {
92        if start == end {
93            return Ok(Some(ScenicRoute {
94                path: vec![start],
95                serendipity_score: 0.0,
96            }));
97        }
98
99        if max_depth == 0 {
100            return Ok(None);
101        }
102
103        // To calculate serendipity, we need to know the semantic distance between adjacent nodes.
104        // We'll perform a BFS that maintains the path and the accumulated score.
105        // Since we want to find the *most* serendipitous path, we explore all paths up to max_depth
106        // that reach the destination. To prevent combinatorial explosion, this should technically
107        // be a Beam Search or bounded DFS, but for Nova we'll do a simple exhaustive BFS with a depth limit
108        // and a visited set per path (simple paths only).
109
110        // Queue stores: (current_node, current_path, current_score, current_depth)
111        let mut queue = VecDeque::new();
112        queue.push_back((start, vec![start], 0.0_f32, 0));
113
114        let mut best_route: Option<ScenicRoute> = None;
115
116        while let Some((current, path, score, depth)) = queue.pop_front() {
117            if current == end {
118                // We found a route! Check if it's the best so far.
119                // We prefer routes with the highest serendipity score.
120                // Note: Longer paths naturally get higher scores if we just sum them.
121                // We could average it, but taking the "longest weirdest path" is perfectly serendipitous.
122                if let Some(best) = &best_route {
123                    if score > best.serendipity_score {
124                        best_route = Some(ScenicRoute {
125                            path: path.clone(),
126                            serendipity_score: score,
127                        });
128                    }
129                } else {
130                    best_route = Some(ScenicRoute {
131                        path: path.clone(),
132                        serendipity_score: score,
133                    });
134                }
135                continue;
136            }
137
138            if depth >= max_depth {
139                continue; // Reached depth limit without hitting end
140            }
141
142            // Get current node's vector for distance calculation
143            let current_vec = match self.get_vector(current, vector_property) {
144                Ok(v) => v,
145                Err(_) => continue, // Skip nodes without the vector property
146            };
147
148            // Explore neighbors (outgoing edges only for simplicity, though undirected might make sense)
149            let outgoing = self.db.get_outgoing_edges(current);
150            let mut neighbors = HashSet::new();
151
152            for edge_id in outgoing {
153                if let Ok(target) = self.db.get_edge_target(edge_id) {
154                    neighbors.insert(target);
155                }
156            }
157
158            for neighbor in neighbors {
159                // Prevent cycles (ensure simple path)
160                if path.contains(&neighbor) {
161                    continue;
162                }
163
164                // Calculate semantic distance
165                if let Ok(neighbor_vec) = self.get_vector(neighbor, vector_property) {
166                    // Distance = 1.0 - Cosine Similarity
167                    let similarity =
168                        ops::cosine_similarity(&current_vec, &neighbor_vec).unwrap_or(0.0);
169                    let distance = (1.0 - similarity).max(0.0);
170
171                    let mut new_path = path.clone();
172                    new_path.push(neighbor);
173
174                    queue.push_back((neighbor, new_path, score + distance, depth + 1));
175                }
176            }
177        }
178
179        Ok(best_route)
180    }
181
182    /// Helper to fetch a node's vector safely.
183    fn get_vector(&self, node_id: NodeId, property: &str) -> Result<Vec<f32>> {
184        let node = self.db.get_node(node_id)?;
185        let val = node.properties.get(property).ok_or_else(|| {
186            Error::other(format!(
187                "Property '{}' not found on node {}",
188                property, node_id
189            ))
190        })?;
191        let vec = val
192            .as_vector()
193            .ok_or_else(|| Error::other(format!("Property '{}' is not a vector", property)))?;
194        Ok(vec.to_vec())
195    }
196}

224    #[test]
225    fn test_serendipity_route() {
226        let db = AletheiaDB::new().unwrap();
227
228        // Let's build a graph where A -> B -> C is the shortest path (and semantically smooth)
229        // But A -> X -> Y -> C is a longer, highly divergent path.
230
231        // Vectors:
232        // A: [1.0, 0.0]
233        // B: [0.9, 0.1] (Similar to A)
234        // C: [0.8, 0.2] (Similar to B)
235        // X: [0.0, 1.0] (Orthogonal to A -> High serendipity!)
236        // Y: [-1.0, 0.0] (Opposite to X -> High serendipity!)
237
238        let a = db
239            .create_node(
240                "Node",
241                PropertyMapBuilder::new()
242                    .insert_vector("vec", &[1.0, 0.0])
243                    .build(),
244            )
245            .unwrap();
246        let b = db
247            .create_node(
248                "Node",
249                PropertyMapBuilder::new()
250                    .insert_vector("vec", &[0.9, 0.1])
251                    .build(),
252            )
253            .unwrap();
254        let c = db
255            .create_node(
256                "Node",
257                PropertyMapBuilder::new()
258                    .insert_vector("vec", &[0.8, 0.2])
259                    .build(),
260            )
261            .unwrap();
262
263        let x = db
264            .create_node(
265                "Node",
266                PropertyMapBuilder::new()
267                    .insert_vector("vec", &[0.0, 1.0])
268                    .build(),
269            )
270            .unwrap();
271        let y = db
272            .create_node(
273                "Node",
274                PropertyMapBuilder::new()
275                    .insert_vector("vec", &[-1.0, 0.0])
276                    .build(),
277            )
278            .unwrap();
279
280        // Smooth Path: A -> B -> C
281        db.create_edge(a, b, "LINK", Default::default()).unwrap();
282        db.create_edge(b, c, "LINK", Default::default()).unwrap();
283
284        // Scenic Path: A -> X -> Y -> C
285        db.create_edge(a, x, "LINK", Default::default()).unwrap();
286        db.create_edge(x, y, "LINK", Default::default()).unwrap();
287        db.create_edge(y, c, "LINK", Default::default()).unwrap();
288
289        let engine = SerendipityEngine::new(&db);
290
291        // Let's find the route!
292        let result = engine.find_scenic_route(a, c, "vec", 4).unwrap().unwrap();
293
294        // The scenic path A->X->Y->C should have a much higher score than A->B->C
295        assert_eq!(result.path, vec![a, x, y, c]);
296
297        // A to X: Cosine([1,0], [0,1]) = 0.0 -> Distance = 1.0
298        // X to Y: Cosine([0,1], [-1,0]) = 0.0 -> Distance = 1.0
299        // Y to C: Cosine([-1,0], [0.8, 0.2]) = -0.8 -> Distance = 1.8
300        // Total expected score: ~3.8
301        assert!(result.serendipity_score > 3.0);
302    }
