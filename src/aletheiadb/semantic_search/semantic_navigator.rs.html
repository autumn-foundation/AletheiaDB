1//! Semantic Navigator
2//!
3//! This experimental module implements A* pathfinding on the graph using vector similarity
4//! as the heuristic and cost function. It enables finding "semantically smooth" paths
5//! between concepts, prioritizing edges that maintain semantic coherence.
6//!
7//! # How it works
8//!
9//! The navigator treats the graph as a state space where:
10//! - **Nodes** are states.
11//! - **Edges** are transitions.
12//! - **Cost** of a transition (A -> B) is `1.0 - cosine_similarity(A, B)`.
13//! - **Heuristic** (h-score) for a node N to goal G is `1.0 - cosine_similarity(N, G)`.
14//!
15//! This approach favors paths where adjacent nodes are semantically similar, and where
16//! the path moves progressively closer to the goal's semantic meaning.
17//!
18//! # Example
19//!
20//! ```rust
21//! # use aletheiadb::AletheiaDB;
22//! # use aletheiadb::core::property::PropertyMapBuilder;
23//! # use aletheiadb::semantic_search::semantic_navigator::SemanticNavigator;
24//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
25//! // 1. Setup database
26//! let db = AletheiaDB::new()?;
27//!
28//! // 2. Create nodes with embeddings
29//! // A: "King" -> B: "Queen" -> C: "Royalty"
30//! // ( Simplified vectors for illustration )
31//! let vec_a = vec![1.0, 0.0];
32//! let vec_b = vec![0.9, 0.1]; // Similar to A
33//! let vec_c = vec![0.8, 0.2]; // Similar to B
34//!
35//! let a = db.create_node("Word", PropertyMapBuilder::new().insert_vector("vec", &vec_a).build())?;
36//! let b = db.create_node("Word", PropertyMapBuilder::new().insert_vector("vec", &vec_b).build())?;
37//! let c = db.create_node("Word", PropertyMapBuilder::new().insert_vector("vec", &vec_c).build())?;
38//!
39//! db.create_edge(a, b, "RELATED", Default::default())?;
40//! db.create_edge(b, c, "RELATED", Default::default())?;
41//!
42//! // 3. Find semantic path
43//! let navigator = SemanticNavigator::new(&db);
44//! let path = navigator.find_path(a, c, "vec")?;
45//!
46//! assert_eq!(path, vec![a, b, c]);
47//! # Ok(())
48//! # }
49//! ```

58/// A navigator that finds semantically meaningful paths through the graph.
59///
60/// Wraps a reference to `AletheiaDB` to perform graph traversals and property lookups.
61pub struct SemanticNavigator<'a> {
62    db: &'a AletheiaDB,
63}
64
65/// Search state for A* algorithm.
66#[derive(Clone, Copy, PartialEq)]
67struct State {
68    /// Total estimated cost (f-score = g-score + h-score).
69    cost: f32,
70    /// Current node ID.
71    node: NodeId,
72}
73
74impl Eq for State {}
75
76// Priority queue depends on `Ord`.
77// We flip the ordering on costs because BinaryHeap is a max-heap,
78// but A* requires a min-heap (lowest cost first).
79impl Ord for State {
80    fn cmp(&self, other: &Self) -> Ordering {
81        // Compare costs inversely
82        other
83            .cost
84            .partial_cmp(&self.cost)
85            .unwrap_or(Ordering::Equal)
86    }
87}
88
89impl PartialOrd for State {
90    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
91        Some(self.cmp(other))
92    }
93}

95/// Outcome of a bounded semantic path search.
96///
97/// The bounded search has three terminal states that callers frequently need
98/// to distinguish. Two of them are perfectly **normal** outcomes (not server
99/// faults): a genuinely disconnected pair of nodes, and a search that hit its
100/// DoS-protection expansion budget before finding the goal. Conflating these
101/// with hard errors (invalid node id, missing vector property) — as an
102/// `Err(Error::other(..))` for both would — misleads a caller into treating a
103/// normal "no route exists" answer as an internal bug. This enum makes the
104/// three cases explicit so the MCP surface can map them to the right
105/// structured error code.
106///
107/// Hard errors (start/end invalid, missing vector) are still surfaced as
108/// `Err` from the search; only the three *reachability* outcomes are modeled
109/// here.
110#[derive(Debug, Clone, PartialEq, Eq)]
111pub enum PathSearchOutcome {
112    /// A path from `start` to `end` (inclusive) was found.
113    Found(Vec<NodeId>),
114    /// The two nodes are not connected: the search exhausted the reachable
115    /// component (within budget) without reaching `end`.
116    NoPath,
117    /// The expansion budget was exhausted before `end` was reached. The graph
118    /// may still contain a path; the bounded search simply refused to keep
119    /// exploring. Carries the budget that was hit so the caller can report it.
120    BudgetExhausted {
121        /// The `max_expansions` budget that was reached.
122        max_expansions: usize,
123    },
124}

132    /// Find a path from `start` to `end` using vector similarity on `vector_prop`.
133    ///
134    /// # Algorithm
135    ///
136    /// Implements A* search where:
137    /// - **g_score(n)**: Cumulative cost from start to n. Cost of edge (u, v) is `1.0 - similarity(u, v)`.
138    /// - **h_score(n)**: Estimated cost from n to end. Heuristic is `1.0 - similarity(n, end)`.
139    /// - **f_score(n)**: `g_score(n) + h_score(n)`.
140    ///
141    /// # Constraints
142    ///
143    /// - Both `start` and `end` nodes must have the specified `vector_prop`.
144    /// - Nodes missing the `vector_prop` encountered during traversal are penalized with high cost (1.0).
145    ///
146    /// # Arguments
147    ///
148    /// * `start` - Starting node ID.
149    /// * `end` - Target node ID.
150    /// * `vector_prop` - Name of the property containing the vector embedding.
151    ///
152    /// # Returns
153    ///
154    /// - `Ok(Vec<NodeId>)`: The path from start to end (inclusive).
155    /// - `Err`: If start/end nodes are invalid, missing vectors, or no path exists.
156    pub fn find_path(&self, start: NodeId, end: NodeId, vector_prop: &str) -> Result<Vec<NodeId>> {
157        // The unbounded variant is a thin wrapper over the bounded search with an
158        // effectively-infinite expansion budget, so the two share one code path.
159        // With `usize::MAX` the budget can never be exhausted, so the only
160        // non-success reachability outcome is a genuinely disconnected pair,
161        // which this variant reports (unchanged external behavior) as the
162        // historical `Err(Error::other("No path found"))`.
163        match self.search(start, end, vector_prop, usize::MAX)? {
164            PathSearchOutcome::Found(path) => Ok(path),
165            PathSearchOutcome::NoPath | PathSearchOutcome::BudgetExhausted { .. } => {
166                Err(Error::other("No path found"))
167            }
168        }
169    }
170
171    /// Bounded variant of [`find_path`](Self::find_path) for untrusted callers
172    /// (e.g. the MCP surface).
173    ///
174    /// The plain A* search is **unbounded**: on a large or adversarial graph it
175    /// can expand an arbitrary number of nodes before concluding, a denial-of-
176    /// service risk when the endpoints are attacker-controlled. This variant caps
177    /// the number of node expansions (pops from the open set) at
178    /// `max_expansions`; once the budget is exhausted without reaching `end`, it
179    /// returns an error instead of continuing to explore. All other semantics are
180    /// identical to [`find_path`](Self::find_path).
181    ///
182    /// # Arguments
183    ///
184    /// * `start` - Starting node ID.
185    /// * `end` - Target node ID.
186    /// * `vector_prop` - Name of the property containing the vector embedding.
187    /// * `max_expansions` - Maximum number of node expansions before giving up.
188    ///
189    /// # Returns
190    ///
191    /// - `Ok(PathSearchOutcome::Found(path))`: The path from start to end
192    ///   (inclusive).
193    /// - `Ok(PathSearchOutcome::NoPath)`: The two nodes are not connected —
194    ///   a normal outcome, not an error.
195    /// - `Ok(PathSearchOutcome::BudgetExhausted { .. })`: The expansion budget
196    ///   was exhausted before `end` was reached — a DoS-protection refusal, not
197    ///   an error.
198    /// - `Err`: Only for hard faults — start/end invalid or missing the vector
199    ///   property. Reachability outcomes are reported via [`PathSearchOutcome`],
200    ///   so callers (e.g. the MCP surface) can map "no path" and "budget
201    ///   exhausted" to distinct, non-`INTERNAL` error codes.
202    pub fn find_path_bounded(
203        &self,
204        start: NodeId,
205        end: NodeId,
206        vector_prop: &str,
207        max_expansions: usize,
208    ) -> Result<PathSearchOutcome> {
209        self.search(start, end, vector_prop, max_expansions)
210    }
211
212    /// Core bounded A* search shared by [`find_path`](Self::find_path) and
213    /// [`find_path_bounded`](Self::find_path_bounded).
214    ///
215    /// Returns a [`PathSearchOutcome`] for the three reachability outcomes;
216    /// only hard faults (invalid nodes, missing vectors) are surfaced as `Err`.
217    fn search(
218        &self,
219        start: NodeId,
220        end: NodeId,
221        vector_prop: &str,
222        max_expansions: usize,
223    ) -> Result<PathSearchOutcome> {
224        // 1. Validate Start/End and get Goal Vector
225        let start_node = self.db.get_node(start)?;
226        let end_node = self.db.get_node(end)?;
227
228        // Ensure start and end have vectors (we need goal vector for heuristic)
229        let _start_vec = start_node
230            .properties
231            .get(vector_prop)
232            .and_then(|v| v.as_arc_vector())
233            .ok_or_else(|| {
234                Error::other(format!(
235                    "Start node {} missing vector property '{}'",
236                    start, vector_prop
237                ))
238            })?;
239
240        let end_vec = end_node
241            .properties
242            .get(vector_prop)
243            .and_then(|v| v.as_arc_vector())
244            .ok_or_else(|| {
245                Error::other(format!(
246                    "End node {} missing vector property '{}'",
247                    end, vector_prop
248                ))
249            })?;
250
251        // 2. Initialize A*
252        let mut open_set = BinaryHeap::new();
253        open_set.push(State {
254            cost: 0.0,
255            node: start,
256        });
257
258        let mut came_from: HashMap<NodeId, NodeId> = HashMap::new();
259        let mut g_score: HashMap<NodeId, f32> = HashMap::new();
260        g_score.insert(start, 0.0);
261
262        let mut f_score: HashMap<NodeId, f32> = HashMap::new();
263        // h(start) = 1.0 - sim(start, end)
264        // We know start has a vector.
265        let h_start = 1.0 - cosine_similarity(&_start_vec, &end_vec)?;
266        f_score.insert(start, h_start);
267
268        // Track visited nodes to prevent cycles and redundant processing
269        // While A* with consistent heuristic doesn't strictly need a closed set for correctness,
270        // it helps performance on general graphs.
271        // However, g_score check implicitly handles this.
272
273        let mut expansions: usize = 0;
274        while let Some(State {
275            cost: _current_f,
276            node: current,
277        }) = open_set.pop()
278        {
279            if current == end {
280                return Ok(PathSearchOutcome::Found(
281                    self.reconstruct_path(came_from, current),
282                ));
283            }
284
285            // Enforce the expansion budget (DoS protection for untrusted
286            // callers). We count a node the moment it is popped for expansion.
287            expansions = expansions.saturating_add(1);
288            if expansions > max_expansions {
289                return Ok(PathSearchOutcome::BudgetExhausted { max_expansions });
290            }
291
292            // Get current vector for cost calculation
293            // Optimization Note: We fetch it repeatedly. In a production version,
294            // we might want to cache these or batch-fetch.
295            let current_node = if current == start {
296                start_node.clone() // We have it already (cheap clone of Arc properties)
297            } else {
298                self.db.get_node(current)?
299            };
300
301            let current_vec = current_node
302                .properties
303                .get(vector_prop)
304                .and_then(|v| v.as_arc_vector());
305
306            for edge_id in self.db.get_outgoing_edges(current) {
307                let neighbor = self.db.get_edge_target(edge_id)?;
308
309                // Calculate tentative_g_score
310                let neighbor_node = self.db.get_node(neighbor)?;
311                let neighbor_vec = neighbor_node
312                    .properties
313                    .get(vector_prop)
314                    .and_then(|v| v.as_arc_vector());
315
316                // Cost(current, neighbor)
317                let distance_cost = match (&current_vec, &neighbor_vec) {
318                    (Some(a), Some(b)) => 1.0 - cosine_similarity(a, b)?,
319                    _ => 1.0, // Penalize missing vectors (max distance)
320                };
321
322                let tentative_g = g_score.get(&current).unwrap_or(&f32::INFINITY) + distance_cost;
323
324                if tentative_g < *g_score.get(&neighbor).unwrap_or(&f32::INFINITY) {
325                    came_from.insert(neighbor, current);
326                    g_score.insert(neighbor, tentative_g);
327
328                    // h(neighbor) = 1.0 - sim(neighbor, goal)
329                    let h_score = match &neighbor_vec {
330                        Some(vec) => 1.0 - cosine_similarity(vec, &end_vec)?,
331                        None => 1.0, // High heuristic if missing vector
332                    };
333
334                    let f = tentative_g + h_score;
335                    f_score.insert(neighbor, f);
336                    open_set.push(State {
337                        cost: f,
338                        node: neighbor,
339                    });
340                }
341            }
342        }
343
344        Ok(PathSearchOutcome::NoPath)
345    }
346
347    fn reconstruct_path(
348        &self,
349        came_from: HashMap<NodeId, NodeId>,
350        mut current: NodeId,
351    ) -> Vec<NodeId> {
352        let mut total_path = vec![current];
353        while let Some(&prev) = came_from.get(&current) {
354            current = prev;
355            total_path.push(current);
356        }
357        total_path.reverse();
358        total_path
359    }

391    #[test]
392    fn test_semantic_path_linear() {
393        let (db, _dir) = create_test_db();
394
395        // A -> B -> C
396        // Vectors:
397        // A: [1.0, 0.0]
398        // B: [0.707, 0.707] (45 degrees, "Halfway" semantically)
399        // C: [0.0, 1.0] (90 degrees)
400
401        // Graph Topology:
402        // A -> B -> C
403        // A -> D -> C (Alternative path)
404        // D: [0.0, -1.0] (Opposite direction, semantically distant from C)
405
406        // Semantic Navigator should prefer A -> B -> C because B is semantically closer to C than D is.
407
408        let props_a = PropertyMapBuilder::new()
409            .insert_vector("vec", &[1.0, 0.0])
410            .build();
411        let a = db.create_node("Node", props_a).unwrap();
412
413        let props_b = PropertyMapBuilder::new()
414            .insert_vector("vec", &[0.707, 0.707])
415            .build();
416        let b = db.create_node("Node", props_b).unwrap();
417
418        let props_c = PropertyMapBuilder::new()
419            .insert_vector("vec", &[0.0, 1.0])
420            .build();
421        let c = db.create_node("Node", props_c).unwrap();
422
423        let props_d = PropertyMapBuilder::new()
424            .insert_vector("vec", &[0.0, -1.0])
425            .build();
426        let d = db.create_node("Node", props_d).unwrap();
427
428        db.create_edge(a, b, "NEXT", PropertyMapBuilder::new().build())
429            .unwrap();
430        db.create_edge(b, c, "NEXT", PropertyMapBuilder::new().build())
431            .unwrap();
432
433        db.create_edge(a, d, "NEXT", PropertyMapBuilder::new().build())
434            .unwrap();
435        db.create_edge(d, c, "NEXT", PropertyMapBuilder::new().build())
436            .unwrap();
437
438        let nav = SemanticNavigator::new(&db);
439        let path = nav.find_path(a, c, "vec").unwrap();
440
441        assert_eq!(
442            path,
443            vec![a, b, c],
444            "Should choose the semantically closer path via B"
445        );
446    }

500    #[test]
501    fn bounded_budget_exhausted_reports_budget_outcome() {
502        let (db, _dir) = create_test_db();
503        // Build a long chain A -> B -> C -> ... so a tiny budget is exhausted
504        // before the far end is reached.
505        let mut ids = Vec::new();
506        for i in 0..50u64 {
507            let x = (i as f32) / 50.0;
508            let id = db
509                .create_node(
510                    "Node",
511                    PropertyMapBuilder::new()
512                        .insert_vector("vec", &[1.0 - x, x])
513                        .build(),
514                )
515                .unwrap();
516            ids.push(id);
517        }
518        for w in ids.windows(2) {
519            db.create_edge(w[0], w[1], "NEXT", PropertyMapBuilder::new().build())
520                .unwrap();
521        }
522
523        let nav = SemanticNavigator::new(&db);
524        let outcome = nav
525            .find_path_bounded(ids[0], ids[ids.len() - 1], "vec", 2)
526            .unwrap();
527        assert_eq!(
528            outcome,
529            PathSearchOutcome::BudgetExhausted { max_expansions: 2 },
530            "a tiny budget on a long chain must report BudgetExhausted"
531        );
532
533        // A generous budget still finds the path.
534        let found = nav
535            .find_path_bounded(ids[0], ids[ids.len() - 1], "vec", usize::MAX)
536            .unwrap();
537        assert!(matches!(found, PathSearchOutcome::Found(_)));
538    }
