1//! Voyager: Maximal Novelty Traversal Engine.
2//!
3//! "To boldly go where the semantics haven't."
4//!
5//! Most pathfinding and recommendation systems in AletheiaDB (and graph DBs in general)
6//! rely on *similarity* — finding nodes that are close in vector space. This creates
7//! filter bubbles and predictable trajectories.
8//!
9//! Voyager takes the opposite approach. It explores the graph by intentionally
10//! choosing the connected node that is **most semantically different** from the current one.
11//! It's a "semantic chaos" explorer designed to break out of local minimums and
12//! find the most surprising connections.
13//!
14//! # Use Cases
15//! - **Creative Brainstorming**: Jump between radically different concepts.
16//! - **Story Generation**: Introduce unexpected plot twists by forcing characters into unfamiliar domains.
17//! - **Anomaly Discovery**: Find paths that abruptly shift context.
18
19use crate::AletheiaDB;
20use crate::core::error::{Error, Result, StorageError, VectorError};
21use crate::core::id::NodeId;
22use crate::core::vector::ops::cosine_similarity;
23use std::collections::HashSet;
24
25/// The Voyager Engine for finding the path of maximal novelty.
26pub struct Voyager<'a> {
27    db: &'a AletheiaDB,
28}
29
30impl<'a> Voyager<'a> {
31    /// Create a new Voyager instance.
32    pub fn new(db: &'a AletheiaDB) -> Self {
33        Self { db }
34    }
35
36    /// Traverse the graph starting from a seed node, always choosing the neighbor
37    /// with the *lowest* cosine similarity to the current node.
38    ///
39    /// The traversal stops if:
40    /// - `max_steps` is reached.
41    /// - The current node has no outgoing edges.
42    /// - All neighbors have already been visited (prevents infinite loops).
43    /// - A node lacks the specified vector property.
44    ///
45    /// # Arguments
46    /// * `start_node` - The node to start exploring from.
47    /// * `vector_property` - The name of the vector property to use for distance.
48    /// * `max_steps` - The maximum length of the path to return.
49    ///
50    /// # Returns
51    /// A sequential list of `NodeId`s representing the path taken, starting with `start_node`.
52    ///
53    /// # Examples
54    ///
55    /// ```rust,no_run
56    /// # fn run() -> Result<(), Box<dyn std::error::Error>> {
57    /// use aletheiadb::AletheiaDB;
58    /// use aletheiadb::semantic_search::voyager::Voyager;
59    /// use aletheiadb::core::id::NodeId;
60    ///
61    /// let db = AletheiaDB::new()?;
62    /// let voyager = Voyager::new(&db);
63    /// let start_node = NodeId::new(1)?; // Assuming this node exists
64    ///
65    /// // Traverse up to 10 steps, always picking the most semantically novel neighbor
66    /// let path = voyager.traverse(start_node, "embedding", 10)?;
67    ///
68    /// println!("Voyager explored {} nodes.", path.len());
69    /// # Ok(())
70    /// # }
71    /// ```
72    pub fn traverse(
73        &self,
74        start_node: NodeId,
75        vector_property: &str,
76        max_steps: usize,
77    ) -> Result<Vec<NodeId>> {
78        let mut path = Vec::new();
79        let mut visited = HashSet::new();
80
81        let mut current = start_node;
82        path.push(current);
83        visited.insert(current);
84
85        for _ in 0..max_steps {
86            // Get current node's vector
87            let current_vec = self.get_vector(current, vector_property)?;
88
89            let edges = self.db.get_outgoing_edges(current);
90            if edges.is_empty() {
91                break; // Dead end
92            }
93
94            let mut least_similar_neighbor = None;
95            let mut lowest_similarity = f32::MAX;
96
97            for edge_id in edges {
98                if let Ok(edge) = self.db.get_edge(edge_id) {
99                    let neighbor = edge.target;
100
101                    // Skip visited to prevent cycles
102                    if visited.contains(&neighbor) {
103                        continue;
104                    }
105
106                    // Try to get neighbor's vector. If it doesn't have one, we can either
107                    // ignore it or treat it as highly novel. Let's ignore it to keep it semantic.
108                    if let Ok(neighbor_vec) = self.get_vector(neighbor, vector_property) {
109                        let sim = cosine_similarity(&current_vec, &neighbor_vec)?;
110
111                        if sim < lowest_similarity {
112                            lowest_similarity = sim;
113                            least_similar_neighbor = Some(neighbor);
114                        }
115                    }
116                }
117            }
118
119            match least_similar_neighbor {
120                Some(next_node) => {
121                    path.push(next_node);
122                    visited.insert(next_node);
123                    current = next_node;
124                }
125                None => {
126                    // All neighbors visited or lack vectors
127                    break;
128                }
129            }
130        }
131
132        Ok(path)
133    }
134
135    /// Helper to extract a vector from a node.
136    fn get_vector(&self, node_id: NodeId, property: &str) -> Result<Vec<f32>> {
137        let node = self
138            .db
139            .get_node(node_id)
140            .map_err(|_| Error::Storage(StorageError::NodeNotFound(node_id)))?;
141
142        let val = node.properties.get(property).ok_or_else(|| {
143            Error::Vector(VectorError::IndexError(format!(
144                "Property '{}' missing on node {}",
145                property, node_id
146            )))
147        })?;
148
149        let vec = val.as_vector().ok_or_else(|| {
150            Error::Vector(VectorError::IndexError(format!(
151                "Property '{}' is not a vector on node {}",
152                property, node_id
153            )))
154        })?;
155
156        Ok(vec.to_vec())
157    }
158}

160#[cfg(test)]
161mod tests {
162    use super::*;
163    use crate::AletheiaDB;
164    use crate::core::property::PropertyMapBuilder;
165    use crate::index::vector::{DistanceMetric, HnswConfig};
166
167    #[test]
168    fn test_voyager_chooses_lowest_similarity() {
169        let db = AletheiaDB::new().unwrap();
170        db.enable_vector_index("vec", HnswConfig::new(2, DistanceMetric::Cosine))
171            .unwrap();
172
173        // Node A: [1.0, 0.0] (Start)
174        let props_a = PropertyMapBuilder::new()
175            .insert_vector("vec", &[1.0, 0.0])
176            .build();
177        let a = db.create_node("Node", props_a).unwrap();
178
179        // Node B: [0.9, 0.1] (High similarity to A, ~0.99)
180        let props_b = PropertyMapBuilder::new()
181            .insert_vector("vec", &[0.9, 0.1])
182            .build();
183        let b = db.create_node("Node", props_b).unwrap();
184
185        // Node C: [0.0, 1.0] (Low similarity to A, 0.0)
186        let props_c = PropertyMapBuilder::new()
187            .insert_vector("vec", &[0.0, 1.0])
188            .build();
189        let c = db.create_node("Node", props_c).unwrap();
190
191        // Node D: [-1.0, 0.0] (Lowest similarity to C, -1.0 relative to C? No, C is [0,1], D is [-1,0]. Sim is 0.0)
192        // Wait, C is [0.0, 1.0]. D is [-1.0, 0.0]. Cosine sim(C, D) = 0.0.
193        // Let's make D perfectly opposite to C: [0.0, -1.0]. Sim = -1.0.
194        let props_d = PropertyMapBuilder::new()
195            .insert_vector("vec", &[0.0, -1.0])
196            .build();
197        let d = db.create_node("Node", props_d).unwrap();
198
199        // Edges from A -> B and A -> C
200        db.create_edge(a, b, "LINK", PropertyMapBuilder::new().build())
201            .unwrap();
202        db.create_edge(a, c, "LINK", PropertyMapBuilder::new().build())
203            .unwrap();
204
205        // Edges from C -> B and C -> D
206        db.create_edge(c, b, "LINK", PropertyMapBuilder::new().build())
207            .unwrap();
208        db.create_edge(c, d, "LINK", PropertyMapBuilder::new().build())
209            .unwrap();
210
211        let voyager = Voyager::new(&db);
212
213        // Start at A. Max steps = 2.
214        // Step 1: A has neighbors B (sim ~0.99) and C (sim 0.0). Voyager chooses C.
215        // Step 2: C has neighbors B (sim ~0.1) and D (sim -1.0). Voyager chooses D.
216        let path = voyager.traverse(a, "vec", 2).unwrap();
217
218        assert_eq!(path.len(), 3); // Start + 2 steps
219        assert_eq!(path[0], a);
220        assert_eq!(path[1], c);
221        assert_eq!(path[2], d);
222    }

224    #[test]
225    fn test_voyager_avoids_cycles() {
226        let db = AletheiaDB::new().unwrap();
227        db.enable_vector_index("vec", HnswConfig::new(2, DistanceMetric::Cosine))
228            .unwrap();
229
230        // Node A: [1.0, 0.0]
231        let props_a = PropertyMapBuilder::new()
232            .insert_vector("vec", &[1.0, 0.0])
233            .build();
234        let a = db.create_node("Node", props_a).unwrap();
235
236        // Node B: [0.0, 1.0] (Opposite/orthogonal to A)
237        let props_b = PropertyMapBuilder::new()
238            .insert_vector("vec", &[0.0, 1.0])
239            .build();
240        let b = db.create_node("Node", props_b).unwrap();
241
242        // Connect A <-> B to form a cycle
243        db.create_edge(a, b, "LINK", PropertyMapBuilder::new().build())
244            .unwrap();
245        db.create_edge(b, a, "LINK", PropertyMapBuilder::new().build())
246            .unwrap();
247
248        let voyager = Voyager::new(&db);
249
250        // Even with max_steps = 10, it should stop after A -> B because A is already visited.
251        let path = voyager.traverse(a, "vec", 10).unwrap();
252
253        assert_eq!(path.len(), 2);
254        assert_eq!(path[0], a);
255        assert_eq!(path[1], b);
256    }
