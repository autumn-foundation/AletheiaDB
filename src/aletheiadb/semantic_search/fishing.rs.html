1//! Associative Retrieval ("Fishing") Module
2//!
3//! This module implements a "fishing" algorithm that combines vector similarity
4//! with graph traversal to find related nodes. It's designed to simulate
5//! associative memory, where recalling one concept ("the bait") pulls up
6//! related concepts ("the catch") based on both semantic similarity and
7//! structural connections.
8//!
9//! # The Metaphor
10//! - **Bait**: The starting point (a Node ID or a raw Vector).
11//! - **Casting**: Finding the initial set of nodes similar to the bait.
12//! - **Spreading the Net**: traversing edges from the initial set.
13//! - **Freshness**: Preferring recently updated information.
14
15use crate::AletheiaDB;
16use crate::core::error::{Error, Result};
17use crate::core::id::NodeId;
18use crate::core::temporal::time;
19use std::collections::HashMap;

21/// Configuration for a fishing trip.
22#[derive(Debug, Clone)]
23pub struct FishingTrip {
24    /// Maximum number of results to return.
25    pub limit: usize,
26    /// Maximum depth of graph traversal from the "school" (vector results).
27    /// Currently only depth 1 is supported (direct neighbors).
28    pub depth: usize,
29    /// Weight given to vector similarity (0.0 to 1.0).
30    pub vector_weight: f32,
31    /// Weight given to graph connection (0.0 to 1.0).
32    pub graph_weight: f32,
33    /// Weight given to freshness/recency (0.0 to 1.0).
34    pub freshness_weight: f32,
35    /// Optional edge labels to follow during graph expansion. If None, follows all edges.
36    pub edge_labels: Option<Vec<String>>,
37}
38
39impl Default for FishingTrip {
40    fn default() -> Self {
41        Self {
42            limit: 10,
43            depth: 1,
44            vector_weight: 1.0,
45            graph_weight: 0.5,
46            freshness_weight: 0.1,
47            edge_labels: None,
48        }
49    }
50}

52/// The input to the fishing algorithm.
53#[derive(Debug, Clone)]
54pub enum Bait {
55    /// Start with an existing node in the graph.
56    /// Optionally specify which vector property to use for similarity search.
57    /// If None, it will attempt to use the first available vector index or rely on default behavior.
58    Node {
59        /// The Node ID to start fishing from.
60        id: NodeId,
61        /// Optional vector property to use for similarity search.
62        property: Option<String>,
63    },
64    /// Start with a raw embedding vector.
65    /// Optionally specify the target property name.
66    /// If None, it will use the first available vector index.
67    Vector {
68        /// The raw vector embedding to start fishing from.
69        vector: Vec<f32>,
70        /// Optional vector property to target.
71        property: Option<String>,
72    },
73}
74
75/// A result from the fishing algorithm.
76#[derive(Debug, Clone)]
77pub struct Catch {
78    /// The caught node.
79    pub node_id: NodeId,
80    /// The total relevance score.
81    pub score: f32,
82    /// Explanation of why this node was caught.
83    pub provenance: String,
84}
85
86/// The main tool for associative retrieval.
87pub struct FishingRod<'a> {
88    db: &'a AletheiaDB,
89}
90
91impl<'a> FishingRod<'a> {
92    /// Create a new FishingRod.
93    pub fn new(db: &'a AletheiaDB) -> Self {
94        Self { db }
95    }
96
97    /// Cast the line and retrieve related nodes.
98    pub fn cast(&self, bait: Bait, config: FishingTrip) -> Result<Vec<Catch>> {
99        if config.depth > 1 {
100            return Err(Error::not_implemented(
101                "Multi-hop traversal (depth > 1)",
102                "Not yet implemented in FishingRod",
103            ));
104        }
105
106        // Step 1: Cast the Line (Vector Search)
107        let school = match bait {
108            Bait::Node { id, property } => {
109                if let Some(prop) = property {
110                    self.db.find_similar_in(&prop, id, config.limit)?
111                } else {
112                    // Fallback to finding similar across any index (default behavior)
113                    self.db
114                        .similarity_search(crate::SimilarityQuery::from_node(id).k(config.limit))?
115                }
116            }
117            Bait::Vector { vector, property } => {
118                let property_name = if let Some(p) = property {
119                    p
120                } else {
121                    // Auto-detect index
122                    let indexes = self.db.list_vector_indexes();
123                    if let Some(idx_info) = indexes.first() {
124                        idx_info.property_name.clone()
125                    } else {
126                        return Err(crate::core::error::Error::Vector(
127                            crate::core::error::VectorError::IndexError(
128                                "No vector indexes configured. Call db.vector_index(\"...\").hnsw(...).enable() first."
129                                    .to_string(),
130                            ),
131                        ));
132                    }
133                };
134                self.db
135                    .search_vectors_in(&property_name, &vector, config.limit)?
136            }
137        };
138
139        let mut candidate_scores: HashMap<NodeId, f32> = HashMap::new();
140        let mut provenance: HashMap<NodeId, String> = HashMap::new();
141
142        // Add the "school" (vector matches) to candidates
143        for (node_id, similarity) in school.iter() {
144            let score = similarity * config.vector_weight;
145            candidate_scores.insert(*node_id, score);
146            provenance.insert(*node_id, format!("Vector Similarity: {:.4}", similarity));
147        }
148
149        // Step 2: Spread the Net (Graph Traversal)
150        if config.depth > 0 {
151            let mut neighbors: Vec<(NodeId, NodeId)> = Vec::new(); // (neighbor, source)
152
153            for (source_node, _) in school.iter() {
154                let edges = if let Some(ref labels) = config.edge_labels {
155                    let mut filtered_edges = Vec::new();
156                    for label in labels {
157                        filtered_edges
158                            .extend(self.db.get_outgoing_edges_with_label(*source_node, label));
159                    }
160                    filtered_edges
161                } else {
162                    self.db.get_outgoing_edges(*source_node)
163                };
164
165                for edge_id in edges {
166                    if let Ok(target) = self.db.get_edge_target(edge_id) {
167                        neighbors.push((target, *source_node));
168                    }
169                }
170            }
171
172            for (target, source) in neighbors {
173                let current_score = *candidate_scores.get(&target).unwrap_or(&0.0);
174                // Simple additive boost for being a neighbor
175                let new_score = current_score + config.graph_weight;
176                candidate_scores.insert(target, new_score);
177
178                provenance
179                    .entry(target)
180                    .and_modify(|p| *p += &format!("\nLinked from Node {}", source))
181                    .or_insert_with(|| format!("Linked from Node {}", source));
182            }
183        }
184
185        // Step 3: Check Freshness
186        if config.freshness_weight > 0.0 {
187            let now_micros = time::now().wallclock();
188
189            // We need to fetch nodes to check timestamps.
190            // Collect all candidate IDs
191            let all_candidates: Vec<NodeId> = candidate_scores.keys().cloned().collect();
192
193            for node_id in all_candidates {
194                // Not collapsing if to keep it simple and compatible with potentially older rust versions
195                // or just to be explicit. But clippy complained, so let's try to satisfy it without let_chains
196                // if they are experimental (as per memory), but `if let && let` is stable?
197                // Memory says: "The project targets stable Rust and does not support experimental features like let_chains (if let ... && ...). Use nested if statements or Option::is_some_and for complex conditionals."
198                // So I CANNOT collapse it using `if let ... && let ...`.
199                // I will suppress the lint.
200                #[allow(clippy::collapsible_if)]
201                if let Ok(node) = self.db.get_node(node_id) {
202                    if let Some(ts) = node.metadata.commit_timestamp {
203                        let age_micros = now_micros.saturating_sub(ts.wallclock());
204                        let age_seconds = age_micros as f32 / 1_000_000.0;
205                        // Decay function: 1 / (1 + age in hours)
206                        let age_hours = age_seconds / 3600.0;
207                        let freshness_score = 1.0 / (1.0 + age_hours);
208
209                        candidate_scores.entry(node_id).and_modify(|s| {
210                            *s += freshness_score * config.freshness_weight;
211                        });
212                    }
213                }
214            }
215        }
216
217        // Step 4: Format Results
218        let mut catches: Vec<Catch> = candidate_scores
219            .into_iter()
220            .map(|(node_id, score)| Catch {
221                node_id,
222                score,
223                provenance: provenance.remove(&node_id).unwrap_or_default(),
224            })
225            .collect();
226
227        // Sort by score descending
228        // SAFETY: partial_cmp only returns None for NaN, which we handle with unwrap_or
229        catches.sort_by(|a, b| {
230            b.score
231                .partial_cmp(&a.score)
232                .unwrap_or(std::cmp::Ordering::Equal)
233        });
234
235        // Limit
236        catches.truncate(config.limit);
237
238        Ok(catches)
239    }
240}

249    #[test]
250    fn test_fishing_workflow() {
251        let db = AletheiaDB::new().unwrap();
252
253        // Enable vector index
254        let config = HnswConfig::new(2, DistanceMetric::Cosine);
255        db.enable_vector_index("embedding", config).unwrap();
256
257        // Create nodes
258        // Node 1: "The Bait" (we'll use its vector)
259        let props1 = PropertyMapBuilder::new()
260            .insert("name", "Fish A")
261            .insert_vector("embedding", &[1.0, 0.0])
262            .build();
263        let _n1 = db.create_node("Fish", props1).unwrap();
264
265        // Node 2: "The Catch" (Similar vector)
266        let props2 = PropertyMapBuilder::new()
267            .insert("name", "Fish B")
268            .insert_vector("embedding", &[0.9, 0.1])
269            .build();
270        let n2 = db.create_node("Fish", props2).unwrap();
271
272        // Node 3: "The Neighbor" (Linked from Fish B)
273        let props3 = PropertyMapBuilder::new().insert("name", "Coral").build(); // No vector needed for graph traversal catch
274        let n3 = db.create_node("Coral", props3).unwrap();
275
276        // Link N2 -> N3
277        db.create_edge(n2, n3, "HIDES_IN", PropertyMapBuilder::new().build())
278            .unwrap();
279
280        // Go fishing!
281        let rod = FishingRod::new(&db);
282
283        // Fish with a vector similar to N1/N2
284        let bait = Bait::Vector {
285            vector: vec![1.0, 0.0],
286            property: Some("embedding".to_string()),
287        };
288        let trip = FishingTrip {
289            limit: 5,
290            depth: 1,
291            vector_weight: 1.0,
292            graph_weight: 0.5,
293            freshness_weight: 0.0, // Ignore time for this deterministic test
294            edge_labels: None,
295        };
296
297        let catches = rod.cast(bait, trip).unwrap();
298
299        // We expect:
300        // 1. N1 (Perfect match or close)
301        // 2. N2 (Close match)
302        // 3. N3 (Linked from N2)
303
304        assert!(catches.len() >= 2);
305
306        let n2_catch = catches.iter().find(|c| c.node_id == n2);
307        assert!(n2_catch.is_some(), "Should catch Fish B via vector");
308
309        let n3_catch = catches.iter().find(|c| c.node_id == n3);
310        assert!(n3_catch.is_some(), "Should catch Coral via graph link");
311
312        if let Some(c) = n3_catch {
313            assert!(c.provenance.contains("Linked from Node"));
314        }
315    }

318    fn test_fishing_with_edge_label_filter() {
319        let db = AletheiaDB::new().unwrap();
320
321        // Enable vector index
322        let config = HnswConfig::new(2, DistanceMetric::Cosine);
323        db.enable_vector_index("embedding", config).unwrap();
324
325        // Node 1: "The Bait"
326        let props1 = PropertyMapBuilder::new()
327            .insert("name", "Bait")
328            .insert_vector("embedding", &[1.0, 0.0])
329            .build();
330        let n1 = db.create_node("Node", props1).unwrap();
331
332        // Node 2: "The School" (Similar to Bait)
333        let props2 = PropertyMapBuilder::new()
334            .insert("name", "Similar")
335            .insert_vector("embedding", &[0.99, 0.01])
336            .build();
337        let n2 = db.create_node("Node", props2).unwrap();
338
339        // Node 3: Connected to School via "KNOWS"
340        let n3 = db
341            .create_node("Node", PropertyMapBuilder::new().build())
342            .unwrap();
343        db.create_edge(n2, n3, "KNOWS", PropertyMapBuilder::new().build())
344            .unwrap();
345
346        // Node 4: Connected to School via "HATES"
347        let n4 = db
348            .create_node("Node", PropertyMapBuilder::new().build())
349            .unwrap();
350        db.create_edge(n2, n4, "HATES", PropertyMapBuilder::new().build())
351            .unwrap();
352
353        let rod = FishingRod::new(&db);
354
355        // Use Node 1 as bait.
356        // It will find Node 2 (similar).
357        // Then from Node 2, it will expand to neighbors.
358        let bait = Bait::Node {
359            id: n1,
360            property: None,
361        };
362
363        // Filter only for "KNOWS" edges
364        let trip = FishingTrip {
365            limit: 10,
366            depth: 1,
367            vector_weight: 1.0,
368            graph_weight: 1.0,
369            freshness_weight: 0.0,
370            edge_labels: Some(vec!["KNOWS".to_string()]),
371        };
372
373        let catches = rod.cast(bait, trip).unwrap();
374
375        // Expectation:
376        // - N2 is caught (Vector similarity)
377        // - N3 is caught (Graph neighbor via KNOWS)
378        // - N4 is NOT caught (Graph neighbor via HATES)
379        // - N1 is NOT caught (Self is excluded from find_similar)
380
381        assert!(
382            catches.iter().any(|c| c.node_id == n2),
383            "Should catch N2 (School)"
384        );
385        assert!(
386            catches.iter().any(|c| c.node_id == n3),
387            "Should catch N3 (KNOWS neighbor)"
388        );
389        assert!(
390            !catches.iter().any(|c| c.node_id == n4),
391            "Should NOT catch N4 (HATES neighbor)"
392        );
393    }
