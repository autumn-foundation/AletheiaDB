1//! Per-query resource limits for the MCP `query` tool (Issue #3368).
2//!
3//! This is the MCP-surface counterpart to the HTTP `/query` resource limits in
4//! [`crate::http::config`]. The two surfaces are compiled under independent
5//! feature flags (`mcp-server` vs `http-server`), so the semantic types are
6//! deliberately duplicated here rather than shared — a shared module would
7//! force one feature to pull in the other. The merge/ceiling semantics are kept
8//! byte-for-byte consistent with the HTTP surface so an operator configuring
9//! both surfaces gets identical behavior.
10//!
11//! Each of the three limit dimensions — **wall-clock timeout**, **result rows**,
12//! and **result bytes** (the working-memory proxy for a read-only query, see the
13//! coverage matrix in `docs/guides/mcp-query-tool.md`) — carries a server-level
14//! **default**, an operator **ceiling** that a per-call override cannot exceed,
15//! and a per-call **override**. A value of `0` on any dimension means
16//! "unlimited"; a ceiling of `0` means "no ceiling". The master
17//! [`enabled`](QueryLimitsConfig::enabled) switch turns all enforcement off.
18
19use rmcp::schemars::{self, JsonSchema};
20use serde::{Deserialize, Serialize};

22/// Default per-query wall-clock timeout, in milliseconds (30 s).
23pub const DEFAULT_QUERY_TIMEOUT_MS: u64 = 30_000;
24/// Operator hard ceiling for a per-call timeout override, in milliseconds (5 min).
25pub const DEFAULT_MAX_QUERY_TIMEOUT_MS: u64 = 300_000;
26/// Default cap on the number of rows returned by a single query.
27///
28/// Aligned with the pre-existing `MAX_RESULT_LIMIT` hard cap on the `query`
29/// tool so enabling limits by default changes no existing behavior.
30pub const DEFAULT_MAX_RESULT_ROWS: usize = 10_000;
31/// Operator hard ceiling for a per-call `max_result_rows` override.
32pub const DEFAULT_MAX_RESULT_ROWS_CEILING: usize = 100_000;
33/// Default cap on the serialized response size of a single query, in bytes (8 MiB).
34pub const DEFAULT_MAX_RESPONSE_BYTES: usize = 8 * 1024 * 1024;
35/// Operator hard ceiling for a per-call `max_response_bytes` override (64 MiB).
36pub const DEFAULT_MAX_RESPONSE_BYTES_CEILING: usize = 64 * 1024 * 1024;
37/// Default cap on a query's estimated working memory, in bytes (Issue #3368
38/// memory-budget dimension).
39///
40/// `0` = unlimited. This is **default-off**: unlike the timeout / row / byte
41/// dimensions, the memory budget is not engaged under the default config, so an
42/// operator must explicitly opt in. See the documented proxy definition in
43/// `docs/guides/mcp-query-tool.md` — the "working memory attributable to a
44/// query's evaluation" is a *proxy* (serialized response length scaled by an
45/// in-memory expansion factor), honestly post-hoc exactly like the byte cap, not
46/// true per-task allocation accounting.
47pub const DEFAULT_MAX_QUERY_MEMORY_BYTES: usize = 0;
48/// Operator hard ceiling for the memory budget (`0` = no ceiling). Default-off:
49/// no ceiling in v1 (there is no per-call memory override to bound).
50pub const DEFAULT_MAX_QUERY_MEMORY_BYTES_CEILING: usize = 0;
51/// Default cap on the number of `query` calls concurrently occupying a
52/// timeout-race worker thread (Issue #3368 DoS guard).
53///
54/// A configured wall-clock timeout runs each query on a dedicated, detached
55/// worker thread (it cannot be cooperatively cancelled). Without a bound, a
56/// caller sending tiny `timeout_ms` overrides — or simply hammering the
57/// documented retriable-timeout retry loop — could pile up unbounded still-
58/// running expensive queries and exhaust threads/CPU/memory. This caps the
59/// number of live workers; at the cap a new timed query is rejected
60/// `UNAVAILABLE` (retriable) instead of spawning. `0` = unbounded.
61pub const DEFAULT_MAX_IN_FLIGHT_QUERIES: usize = 64;
62
63/// Which per-query resource-limit dimension a value applies to (Issue #3368).
64///
65/// The [`as_str`](Self::as_str) token is the stable `details.dimension` value in
66/// the structured error body, so callers can branch on it without string
67/// matching the human message.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub enum LimitDimension {
70    /// The wall-clock time budget for producing the response.
71    WallClockTimeout,
72    /// The number of rows in the result.
73    ResultRows,
74    /// The serialized byte size of the result (working-memory proxy).
75    ResultBytes,
76    /// The estimated working-set memory consumed materializing the result
77    /// (Issue #3368 memory-budget dimension; a documented proxy, default-off).
78    Memory,
79}
80
81impl LimitDimension {
82    /// Stable snake_case token used in `error.details.dimension`.
83    #[must_use]
84    pub fn as_str(self) -> &'static str {
85        match self {
86            Self::WallClockTimeout => "wall_clock_timeout",
87            Self::ResultRows => "result_rows",
88            Self::ResultBytes => "result_bytes",
89            Self::Memory => "memory_bytes",
90        }
91    }
92}

94/// Per-call limit override carried on a `query` tool request under `"limits"`.
95///
96/// Every field is optional and additive, so existing request bodies that omit
97/// `"limits"` parse and behave exactly as before. An override present but
98/// *larger* than the operator ceiling is rejected (`INVALID_ARGUMENT`); an
99/// override *smaller* than the server default is honored (a caller self-limiting
100/// tighter). A value of `0` means "unlimited" for that dimension and is only
101/// accepted when the operator ceiling is itself unbounded.
102#[derive(Debug, Clone, Default, Deserialize, Serialize, PartialEq, Eq, JsonSchema)]
103pub struct QueryLimitsOverride {
104    /// Per-call wall-clock timeout in milliseconds (`0` = unlimited).
105    #[serde(default)]
106    #[schemars(
107        description = "Per-call wall-clock timeout in milliseconds (0 = unlimited). Rejected \
108                              with INVALID_ARGUMENT if it exceeds the operator ceiling."
109    )]
110    pub timeout_ms: Option<u64>,
111    /// Per-call maximum result rows (`0` = unlimited).
112    #[serde(default)]
113    #[schemars(
114        description = "Per-call maximum result rows (0 = unlimited). Over-cap results are \
115                              truncated with truncated:true, not rejected."
116    )]
117    pub max_result_rows: Option<usize>,
118    /// Per-call maximum serialized response bytes (`0` = unlimited).
119    #[serde(default)]
120    #[schemars(
121        description = "Per-call maximum serialized response bytes (0 = unlimited). Exceeding \
122                              it fails closed with a RESOURCE_EXHAUSTED error."
123    )]
124    pub max_response_bytes: Option<usize>,
125}
126
127impl QueryLimitsOverride {
128    /// True when no override fields are set (equivalent to no `"limits"` key).
129    #[must_use]
130    pub fn is_empty(&self) -> bool {
131        self.timeout_ms.is_none()
132            && self.max_result_rows.is_none()
133            && self.max_response_bytes.is_none()
134    }
135}

137/// A per-call override that exceeded the operator hard ceiling (Issue #3368).
138///
139/// Maps to a `INVALID_ARGUMENT` query-tool error with
140/// `details: {dimension, requested, ceiling}`.
141#[derive(Debug, Clone, PartialEq, Eq)]
142pub struct LimitOverrideError {
143    /// The dimension whose override was rejected.
144    pub dimension: LimitDimension,
145    /// The value the caller requested.
146    pub requested: u64,
147    /// The operator ceiling it exceeded.
148    pub ceiling: u64,
149}
150
151/// The concrete limits in force for a single query, after folding the server
152/// defaults, the per-call override, and the operator ceilings (Issue #3368). A
153/// value of `0` on any dimension means "unlimited".
154#[derive(Debug, Clone, Copy, PartialEq, Eq)]
155pub struct EffectiveQueryLimits {
156    /// Wall-clock timeout in milliseconds (`0` = unlimited).
157    pub timeout_ms: u64,
158    /// Maximum result rows (`0` = unlimited).
159    pub max_result_rows: usize,
160    /// Maximum serialized response bytes (`0` = unlimited).
161    pub max_response_bytes: usize,
162    /// Maximum estimated working-set memory in bytes (`0` = unlimited; Issue
163    /// #3368 memory-budget dimension, default-off).
164    pub max_query_memory_bytes: usize,
165}
166
167impl EffectiveQueryLimits {
168    /// No enforcement on any dimension.
169    #[must_use]
170    pub fn unlimited() -> Self {
171        Self {
172            timeout_ms: 0,
173            max_result_rows: 0,
174            max_response_bytes: 0,
175            max_query_memory_bytes: 0,
176        }
177    }
178}

180/// Fold one dimension's server default, operator ceiling, and per-call override
181/// into a single effective value.
182///
183/// Semantics (all in the dimension's native unit; `0` = unlimited, a ceiling of
184/// `0` = no ceiling), kept consistent with `crate::http::config::merge_dimension`:
185///
186/// - **Override present**: rejected when a ceiling exists and the override
187///   either requests unlimited (`0`) or exceeds the ceiling; otherwise the
188///   override wins (it may be *below* the default — a tighter self-limit).
189/// - **Override absent**: the server default, clamped down to the ceiling if the
190///   default is unlimited or above it.
191fn merge_dimension(
192    default_v: u64,
193    ceiling: u64,
194    override_v: Option<u64>,
195    dimension: LimitDimension,
196) -> Result<u64, LimitOverrideError> {
197    match override_v {
198        Some(requested) => {
199            if ceiling != 0 && (requested == 0 || requested > ceiling) {
200                return Err(LimitOverrideError {
201                    dimension,
202                    requested,
203                    ceiling,
204                });
205            }
206            Ok(requested)
207        }
208        None => {
209            let mut effective = default_v;
210            if ceiling != 0 && (effective == 0 || effective > ceiling) {
211                effective = ceiling;
212            }
213            Ok(effective)
214        }
215    }
216}

218/// Server-level configuration for per-query resource limits on the MCP `query`
219/// tool (Issue #3368).
220///
221/// Each dimension has a **default** (applied when the request supplies no
222/// override), a **ceiling** (the largest value a per-call override may request;
223/// `0` = no ceiling). A value of `0` on a default/override means "unlimited".
224///
225/// The master [`enabled`](Self::enabled) switch disables all enforcement when
226/// `false` (see [`disabled`](Self::disabled)): the effective limits are then
227/// unconditionally unlimited and per-call overrides are ignored.
228#[derive(Debug, Clone, PartialEq, Eq)]
229pub struct QueryLimitsConfig {
230    /// Master switch. When `false`, no limit is enforced.
231    pub enabled: bool,
232    /// Default wall-clock timeout in milliseconds (`0` = unlimited).
233    pub default_timeout_ms: u64,
234    /// Operator ceiling for a per-call timeout override (`0` = no ceiling).
235    pub max_timeout_ms: u64,
236    /// Default maximum result rows (`0` = unlimited).
237    pub default_max_result_rows: usize,
238    /// Operator ceiling for a per-call row override (`0` = no ceiling).
239    pub max_result_rows: usize,
240    /// Default maximum serialized response bytes (`0` = unlimited).
241    pub default_max_response_bytes: usize,
242    /// Operator ceiling for a per-call byte override (`0` = no ceiling).
243    pub max_response_bytes: usize,
244    /// Default cap on a query's estimated working memory in bytes (`0` =
245    /// unlimited; Issue #3368 memory-budget dimension, default-off).
246    ///
247    /// There is no per-call memory override in v1, so this default (clamped by
248    /// [`max_query_memory_bytes`](Self::max_query_memory_bytes)) is the only
249    /// value that ever applies.
250    pub default_max_query_memory_bytes: usize,
251    /// Operator ceiling for the memory budget (`0` = no ceiling; Issue #3368).
252    pub max_query_memory_bytes: usize,
253    /// Cap on the number of `query` calls concurrently occupying a
254    /// timeout-race worker thread (Issue #3368 DoS guard; `0` = unbounded).
255    ///
256    /// This is a server-level (not per-call) bound: it is not part of
257    /// [`EffectiveQueryLimits`] and cannot be overridden per request. It only
258    /// affects the spawned-worker path taken when a wall-clock timeout is in
259    /// force; the inline `timeout_ms == 0` path never spawns and is unguarded.
260    pub max_in_flight_queries: usize,
261}
262
263impl QueryLimitsConfig {
264    /// A config that enforces nothing: every dimension unlimited, overrides
265    /// ignored. Useful for trusted embedded deployments and tests.
266    #[must_use]
267    pub fn disabled() -> Self {
268        Self {
269            enabled: false,
270            default_timeout_ms: 0,
271            max_timeout_ms: 0,
272            default_max_result_rows: 0,
273            max_result_rows: 0,
274            default_max_response_bytes: 0,
275            max_response_bytes: 0,
276            default_max_query_memory_bytes: 0,
277            max_query_memory_bytes: 0,
278            max_in_flight_queries: 0,
279        }
280    }
281
282    /// Fold this config and an optional per-call override into the concrete
283    /// [`EffectiveQueryLimits`] for one query.
284    ///
285    /// When [`enabled`](Self::enabled) is `false`, this always returns
286    /// [`EffectiveQueryLimits::unlimited`] and never errors (overrides ignored).
287    ///
288    /// # Errors
289    ///
290    /// Returns [`LimitOverrideError`] if `override_` requests a value above the
291    /// corresponding ceiling (or requests unlimited under a finite ceiling).
292    pub fn effective(
293        &self,
294        override_: Option<&QueryLimitsOverride>,
295    ) -> Result<EffectiveQueryLimits, LimitOverrideError> {
296        if !self.enabled {
297            return Ok(EffectiveQueryLimits::unlimited());
298        }
299        let ov = override_.filter(|o| !o.is_empty());
300        let timeout_ms = merge_dimension(
301            self.default_timeout_ms,
302            self.max_timeout_ms,
303            ov.and_then(|o| o.timeout_ms),
304            LimitDimension::WallClockTimeout,
305        )?;
306        let max_result_rows = merge_dimension(
307            self.default_max_result_rows as u64,
308            self.max_result_rows as u64,
309            ov.and_then(|o| o.max_result_rows).map(|v| v as u64),
310            LimitDimension::ResultRows,
311        )? as usize;
312        let max_response_bytes = merge_dimension(
313            self.default_max_response_bytes as u64,
314            self.max_response_bytes as u64,
315            ov.and_then(|o| o.max_response_bytes).map(|v| v as u64),
316            LimitDimension::ResultBytes,
317        )? as usize;
318        // Memory budget has no per-call override in v1 (default-off, server
319        // defaults only), so it is folded with `None` — only the default,
320        // clamped down to a finite ceiling, ever applies.
321        let max_query_memory_bytes = merge_dimension(
322            self.default_max_query_memory_bytes as u64,
323            self.max_query_memory_bytes as u64,
324            None,
325            LimitDimension::Memory,
326        )? as usize;
327        Ok(EffectiveQueryLimits {
328            timeout_ms,
329            max_result_rows,
330            max_response_bytes,
331            max_query_memory_bytes,
332        })
333    }
334}

336impl Default for QueryLimitsConfig {
337    /// Enabled by default, but with generous limits chosen so no existing
338    /// request behavior changes (Issue #3368).
339    fn default() -> Self {
340        Self {
341            enabled: true,
342            default_timeout_ms: DEFAULT_QUERY_TIMEOUT_MS,
343            max_timeout_ms: DEFAULT_MAX_QUERY_TIMEOUT_MS,

348            // Memory budget is default-off (unlimited) even under the default
349            // config: an operator must explicitly opt in (Issue #3368).
350            default_max_query_memory_bytes: DEFAULT_MAX_QUERY_MEMORY_BYTES,
351            max_query_memory_bytes: DEFAULT_MAX_QUERY_MEMORY_BYTES_CEILING,
352            max_in_flight_queries: DEFAULT_MAX_IN_FLIGHT_QUERIES,
353        }
354    }

357/// Per-dimension counters of over-limit query terminations (Issue #3368
358/// observability). Cheap atomics incremented at the MCP surface so an operator
359/// (or a test) can spot chronic offenders. Surfacing these through
360/// `database_stats` requires a storage-layer counter and is a cross-lane
361/// follow-up; the MCP-surface accessor is the interim.
362#[derive(Debug, Default)]
363pub struct LimitCounters {
364    /// Queries terminated by the wall-clock timeout.
365    pub wall_clock_timeout: std::sync::atomic::AtomicU64,
366    /// Responses rejected for exceeding the result-byte cap.
367    pub result_bytes: std::sync::atomic::AtomicU64,
368    /// Responses rejected for exceeding the estimated-memory budget (Issue
369    /// #3368 memory-budget dimension).
370    pub memory_bytes: std::sync::atomic::AtomicU64,
371    /// Per-call overrides rejected for exceeding an operator ceiling.
372    pub override_rejected: std::sync::atomic::AtomicU64,
373}
374
375/// An immutable snapshot of [`LimitCounters`] (Issue #3368).
376#[derive(Debug, Clone, Copy, PartialEq, Eq)]
377pub struct LimitCountsSnapshot {
378    /// Queries terminated by the wall-clock timeout.
379    pub wall_clock_timeout: u64,
380    /// Responses rejected for exceeding the result-byte cap.
381    pub result_bytes: u64,
382    /// Responses rejected for exceeding the estimated-memory budget (Issue
383    /// #3368 memory-budget dimension).
384    pub memory_bytes: u64,
385    /// Per-call overrides rejected for exceeding an operator ceiling.
386    pub override_rejected: u64,
387}
388
389impl LimitCounters {
390    /// Increment the counter for `dimension` (a terminated query).
391    pub fn record_termination(&self, dimension: LimitDimension) {
392        use std::sync::atomic::Ordering;
393        match dimension {
394            LimitDimension::WallClockTimeout => {
395                self.wall_clock_timeout.fetch_add(1, Ordering::Relaxed);
396            }
397            LimitDimension::ResultBytes => {
398                self.result_bytes.fetch_add(1, Ordering::Relaxed);
399            }
400            LimitDimension::Memory => {
401                self.memory_bytes.fetch_add(1, Ordering::Relaxed);
402            }
403            // Row-cap breaches degrade to a disclosed truncation, not a
404            // termination, so they are not counted here.
405            LimitDimension::ResultRows => {}
406        }
407    }
408
409    /// Increment the rejected-override counter.
410    pub fn record_override_rejected(&self) {
411        self.override_rejected
412            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
413    }
414
415    /// Read a consistent-enough snapshot of the counters.
416    #[must_use]
417    pub fn snapshot(&self) -> LimitCountsSnapshot {
418        use std::sync::atomic::Ordering;
419        LimitCountsSnapshot {
420            wall_clock_timeout: self.wall_clock_timeout.load(Ordering::Relaxed),
421            result_bytes: self.result_bytes.load(Ordering::Relaxed),
422            memory_bytes: self.memory_bytes.load(Ordering::Relaxed),
423            override_rejected: self.override_rejected.load(Ordering::Relaxed),
424        }
425    }
426}
