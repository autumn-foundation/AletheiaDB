1//! Atomic multi-write batches with local refs (`apply_batch`, Issue #3231).
2//!
3//! An LLM building a subgraph through the single-op MCP tools must issue one
4//! call per write, each its own transaction — a mid-sequence failure leaves a
5//! half-built, referentially-inconsistent graph. `apply_batch` accepts an
6//! **ordered** array of write operations that commit **all-or-nothing** in a
7//! single `WriteTransaction` (one WAL batch append, one GroupCommit fsync),
8//! where later operations may reference nodes created earlier in the same
9//! batch via symbolic local refs (`"$alias"` or positional `"$0"`), in the
10//! spirit of Datomic tempids / XTDB `submit-tx`.
11//!
12//! # Two-phase execution
13//!
14//! 1. **Static pre-validation** (no database access): the batch cap, op
15//!    shapes, alias/ref resolution (duplicates, forward refs, unknown refs),
16//!    per-op timestamps/properties/provenance, v1 scope rules, and
17//!    parameter-combination rules are all checked up front. Every rejection
18//!    reports a precise `failed_op_index`, and a structurally invalid batch
19//!    never opens a transaction (and never burns entity IDs).
20//! 2. **Execution**: one `db.write(|tx| ...)` closure issues the operations
21//!    in order. Any failure aborts the whole transaction (drop-rollback —
22//!    nothing was applied, nothing is ever visible to concurrent readers) and
23//!    reports the failing op index. Commit-phase failures (conflicts,
24//!    constraint violations) also abort atomically.
25//!
26//! # v1 scope (documented limitations)
27//!
28//! - Later ops may **reference** batch-created nodes as edge endpoints, but
29//!   may not **update/delete/re-write** a batch-created entity: the core
30//!   transaction's update/delete paths read committed storage, not the write
31//!   buffer (read-your-writes inside an uncommitted batch is explicitly out
32//!   of scope for #3231). Such ops are rejected up front with a clear error.
33//! - At most one write per committed entity per batch (a second write to the
34//!   same node/edge id is rejected — the second op would act on a stale read
35//!   of the pre-batch state).
36//! - Delete results carry no `version_id` (tombstone versions are allocated
37//!   at apply time inside the commit path and are not surfaced by the
38//!   transaction API).
39//! - A batch-created edge removed by a later `delete_node` + `detach: true`
40//!   of one of its endpoints is elided from the transaction entirely (its
41//!   per-op result reports `removed_by_delete_at_index`): the committed
42//!   record matches the batch's atomic net effect, and no history is written
43//!   for an edge that never existed outside the batch.

66/// Request to apply an ordered batch of write operations atomically.
67#[derive(Debug, Clone, Deserialize, Serialize, JsonSchema)]
68pub struct ApplyBatchRequest {
69    /// Ordered list of write operations. The whole batch commits
70    /// all-or-nothing; operations may reference nodes created earlier in the
71    /// same batch via `"$alias"` (a `ref` supplied on a `create_node` op) or
72    /// positionally via `"$<index>"`.
73    ///
74    /// Kept as raw JSON values so that a malformed operation can be rejected
75    /// with its precise array index (`failed_op_index`); the advertised JSON
76    /// schema is the typed [`BatchOperation`] shape.
77    #[schemars(
78        with = "Vec<BatchOperation>",
79        description = "Ordered list of write operations, applied atomically (all-or-nothing). \
80                       Each element is a tagged object whose `op` field selects the operation: \
81                       create_node, create_edge, update_node, update_edge, delete_node, \
82                       delete_edge. Edge endpoints (and only endpoints) may reference nodes \
83                       created earlier in the same batch via '$alias' (see create_node's `ref`) \
84                       or positional '$<index>' local references."
85    )]
86    pub operations: Vec<serde_json::Value>,
87}

323/// Closure error carrying the failing op index out of `db.write`.
324///
325/// Commit-phase errors (validation, conflicts, constraints) arrive through
326/// the `From<Error>` impl `db.write` requires and carry no op index
327/// (`failed_op_index` is reported as JSON `null` for those).
328enum BatchAbort {
329    /// An operation failed with a database error at its op index.
330    Op { index: usize, source: Error },
331    /// The #3209 safe-by-default refusal: the node targeted by the
332    /// `delete_node` op at `index` still has connected edges (committed
333    /// and/or batch-created) and `detach` was not passed.
334    Refused {
335        index: usize,
336        node_id: u64,
337        connected_edges: usize,
338    },
339    /// A `delete_node` + `detach: true` cascade would discard an edge update
340    /// made earlier in this batch — rejected rather than silently dropping
341    /// the update (commit validation would abort the batch anyway, with a
342    /// less precise error).
343    UpdatedEdgeCascade {
344        index: usize,
345        edge_id: u64,
346        node_id: u64,
347        updated_at_index: usize,
348    },
349    /// The op targets a committed edge that an earlier `delete_node` +
350    /// `detach: true` cascade in this batch already removed.
351    EdgeRemovedByCascade {
352        index: usize,
353        edge_id: u64,
354        removed_at_index: usize,
355    },
356    /// The op at `index` targets a committed integer id that a `create_node`/
357    /// `create_edge` earlier in THIS batch allocated — a guessed-id write
358    /// against an entity not yet committed. This is the documented v1-scope
359    /// rejection ("no update/delete of batch-created refs") that the static
360    /// `$alias`/`$<index>` prevalidation enforces for the ref *spelling*
361    /// (`batch_committed_node`); enforced explicitly here for the integer-id
362    /// form because buffer-aware transaction reads (Issue #3417) no longer
363    /// surface it incidentally as `NOT_FOUND`.
364    BatchCreatedRefWrite {
365        index: usize,
366        entity: &'static str,
367        id: u64,
368        created_at_index: usize,
369        verb: &'static str,
370    },
371    /// A commit-phase failure (no attributable op index).
372    Commit { source: Error },
373    /// An internal invariant breach in the batch handler itself (never the
374    /// caller's fault) — reported as `INTERNAL`, not as a caller-fault code.
375    Internal { index: usize, message: String },
376}
377
378impl From<Error> for BatchAbort {
379    fn from(source: Error) -> Self {
380        BatchAbort::Commit { source }
381    }
382}

384/// Batch-local adjacency ledger entry for an edge created in this batch.
385struct BatchEdge {
386    source: NodeId,
387    target: NodeId,
388    annihilated_by: Option<usize>,
389}
390
391impl BatchEdge {
392    fn touches(&self, node: NodeId) -> bool {
393        self.source == node || self.target == node
394    }
395
396    /// Whether the edge still (logically) exists when the op at `index`
397    /// executes: it has not been removed by an earlier detach-delete.
398    fn alive_at(&self, index: usize) -> bool {
399        match self.annihilated_by {
400            None => true,
401            Some(k) => k >= index,
402        }
403    }
404}

410impl AletheiaMcpServer {
411    /// Apply an ordered batch of write operations atomically (Issue #3231).
412    ///
413    /// See the [module docs](self) for the contract; returns the JSON
414    /// response as a string, like the other typed tool wrappers.
415    pub fn apply_batch(&self, req: ApplyBatchRequest) -> String {
416        Self::extract_text(self.handle_apply_batch(
417            serde_json::to_value(req).expect("request serialization should not fail"),
418        ))
419    }
420
421    /// Single dispatch entry for the `apply_batch` tool.
422    pub(crate) fn handle_apply_batch(&self, args: serde_json::Value) -> CallToolResult {
423        // AUTHZ SEAM: a future authorization check / principal stamp for
424        // batched writes belongs here — one choke point before any operation
425        // is parsed, validated, or executed.
426        let req: ApplyBatchRequest = match serde_json::from_value(args) {
427            Ok(r) => r,
428            Err(e) => return self.invalid_argument(&format!("Invalid arguments: {}", e)),
429        };
430
431        // Phase 1: static pre-validation. No database access; a rejected
432        // batch never opens a transaction and never burns entity IDs.
433        let planned = match self.prevalidate_batch(&req.operations) {
434            Ok(p) => p,
435            Err(result) => return *result,
436        };
437
438        // An empty batch is a harmless, idempotent no-op success.
439        if planned.is_empty() {
440            return self.success_json(json!({
441                "success": true,
442                "operation_count": 0,
443                "results": [],
444                "ref_map": {}
445            }));
446        }
447
448        // Phase 2: execute all operations in one write transaction. Any
449        // error rolls the whole transaction back (drop-rollback): none of
450        // the batch's writes are ever applied or visible.
451        let outcome = self.db().write(
452            |tx| -> Result<
453                (
454                    Vec<serde_json::Value>,
455                    serde_json::Map<String, serde_json::Value>,
456                ),
457                BatchAbort,
458            > { self.execute_batch(tx, &planned) },
459        );
460
461        match outcome {
462            Ok((results, ref_map)) => self.success_json(json!({
463                "success": true,
464                "operation_count": planned.len(),
465                "results": results,
466                "ref_map": ref_map
467            })),
468            Err(abort) => self.batch_abort_result(abort),
469        }
470    }
471
472    // ------------------------------------------------------------------
473    // Phase 1: static pre-validation
474    // ------------------------------------------------------------------
475
476    /// Statically validate the raw operation array into executable
477    /// [`PlannedOp`]s, or produce the structured rejection.
478    ///
479    /// Boxed error to keep the happy-path return small (clippy
480    /// result_large_err).
481    fn prevalidate_batch(
482        &self,
483        raw_ops: &[serde_json::Value],
484    ) -> Result<Vec<PlannedOp>, Box<CallToolResult>> {
485        // Batch cap first (Issue #3226 convention: echo the limit).
486        let limit = self.max_batch_operations;
487        if raw_ops.len() > limit {
488            return Err(Box::new(self.error_result(
489                McpError::new(
490                    McpErrorCode::InvalidArgument,
491                    format!(
492                        "Batch of {} operations exceeds the maximum of {} per apply_batch call; \
493                         split it into smaller batches.",
494                        raw_ops.len(),
495                        limit
496                    ),
497                )
498                .details(json!({
499                    "limit": limit,
500                    "submitted": raw_ops.len(),
501                    // No single op is at fault; null keeps the details shape
502                    // uniform with every other apply_batch error.
503                    "failed_op_index": serde_json::Value::Null
504                })),
505            )));
506        }
507
508        // Parse each op individually so a malformed element is rejected with
509        // its precise index (a wholesale Vec deserialization would lose it).
510        let mut ops = Vec::with_capacity(raw_ops.len());
511        for (index, raw) in raw_ops.iter().enumerate() {
512            match serde_json::from_value::<BatchOperation>(raw.clone()) {
513                Ok(op) => ops.push(op),
514                Err(e) => {
515                    return Err(self.batch_invalid(
516                        index,
517                        format!("Invalid operation at index {index}: {e}"),
518                        serde_json::Map::new(),
519                    ));
520                }
521            }
522        }
523
524        // Collect alias definitions (create_node ops only) and validate them.
525        let mut aliases: HashMap<&str, usize> = HashMap::new();
526        for (index, op) in ops.iter().enumerate() {
527            if let BatchOperation::CreateNode {
528                alias: Some(alias), ..
529            } = op
530            {
531                if alias.is_empty() {
532                    return Err(self.batch_invalid(
533                        index,
534                        format!("Operation {index}: `ref` must not be empty"),
535                        serde_json::Map::new(),
536                    ));
537                }
538                if alias.starts_with('$') {
539                    return Err(self.batch_invalid(
540                        index,
541                        format!(
542                            "Operation {index}: `ref` \"{alias}\" must not start with '$' \
543                             (the '$' prefix is used when referencing, not when defining)"
544                        ),
545                        serde_json::Map::new(),
546                    ));
547                }
548                if alias.chars().all(|c| c.is_ascii_digit()) {
549                    return Err(self.batch_invalid(
550                        index,
551                        format!(
552                            "Operation {index}: `ref` \"{alias}\" must not be purely numeric; \
553                             numeric references are reserved for positional '$<index>' refs"
554                        ),
555                        serde_json::Map::new(),
556                    ));
557                }
558                if let Some(&first) = aliases.get(alias.as_str()) {
559                    let mut details = serde_json::Map::new();
560                    details.insert("ref".to_string(), json!(alias));
561                    details.insert("first_defined_at_index".to_string(), json!(first));
562                    return Err(self.batch_invalid(
563                        index,
564                        format!(
565                            "Operation {index}: duplicate `ref` \"{alias}\" \
566                             (first defined at operation {first})"
567                        ),
568                        details,
569                    ));
570                }
571                aliases.insert(alias.as_str(), index);
572            }
573        }
574
575        // Main pass: resolve refs, parse timestamps/properties/provenance,
576        // and enforce the v1 scope rules.
577        let mut planned = Vec::with_capacity(ops.len());
578        // Committed node ids deleted by an earlier op -> that op's index.
579        let mut deleted_nodes: HashMap<u64, (usize, bool)> = HashMap::new(); // id -> (index, detach)
580        // Committed entity ids already written by an earlier op (one write
581        // per committed entity per batch).
582        let mut written_nodes: HashMap<u64, usize> = HashMap::new();
583        let mut written_edges: HashMap<u64, usize> = HashMap::new();
584
585        for (index, op) in ops.iter().enumerate() {
586            let planned_op = match op {
587                BatchOperation::CreateNode {
588                    label,
589                    properties,
590                    valid_time,
591                    provenance,
592                    alias,
593                } => PlannedOp::CreateNode {
594                    label: label.clone(),
595                    properties: self.batch_properties(index, properties.as_ref())?,
596                    valid_from: self.batch_valid_time(index, valid_time)?,
597                    provenance: self.batch_provenance(index, provenance.clone())?,
598                    alias: alias.clone(),
599                },
600                BatchOperation::CreateEdge {
601                    source_id,
602                    target_id,
603                    label,
604                    properties,
605                    valid_time,
606                    provenance,
607                } => {
608                    let source =
609                        self.batch_resolve_ref(index, source_id, &aliases, &ops, &deleted_nodes)?;
610                    let target =
611                        self.batch_resolve_ref(index, target_id, &aliases, &ops, &deleted_nodes)?;
612                    PlannedOp::CreateEdge {
613                        source,
614                        target,
615                        label: label.clone(),
616                        properties: self.batch_properties(index, properties.as_ref())?,
617                        valid_from: self.batch_valid_time(index, valid_time)?,
618                        provenance: self.batch_provenance(index, provenance.clone())?,
619                        annihilated_by: None, // filled below, after all deletes are known
620                    }
621                }
622                BatchOperation::UpdateNode {
623                    node_id,
624                    properties,
625                    valid_time,
626                    provenance,
627                } => {
628                    let node_id =
629                        self.batch_committed_node(index, node_id, "update", &mut written_nodes)?;
630                    PlannedOp::UpdateNode {
631                        node_id,
632                        properties: self.batch_properties(index, Some(properties))?,
633                        valid_from: self.batch_valid_time(index, valid_time)?,
634                        provenance: self.batch_provenance(index, provenance.clone())?,
635                    }
636                }
637                BatchOperation::UpdateEdge {
638                    edge_id,
639                    properties,
640                    valid_time,
641                    provenance,
642                } => {
643                    let edge_id = self.batch_committed_edge(index, *edge_id, &mut written_edges)?;
644                    PlannedOp::UpdateEdge {
645                        edge_id,
646                        properties: self.batch_properties(index, Some(properties))?,
647                        valid_from: self.batch_valid_time(index, valid_time)?,
648                        provenance: self.batch_provenance(index, provenance.clone())?,
649                    }
650                }
651                BatchOperation::DeleteNode {
652                    node_id,
653                    detach,
654                    valid_time,
655                } => {
656                    let detach = detach.unwrap_or(false);
657                    let valid_from = self.batch_valid_time(index, valid_time)?;
658                    if detach && valid_from.is_some() {
659                        return Err(self.batch_invalid(
660                            index,
661                            format!(
662                                "Operation {index}: valid_time is not supported together with \
663                                 detach:true; cascade delete does not support backdating. Delete \
664                                 the connected edges individually with valid_time, or omit \
665                                 valid_time to cascade-delete at now."
666                            ),
667                            serde_json::Map::new(),
668                        ));
669                    }
670                    let node_id =
671                        self.batch_committed_node(index, node_id, "delete", &mut written_nodes)?;
672                    deleted_nodes.insert(node_id.as_u64(), (index, detach));
673                    PlannedOp::DeleteNode {
674                        node_id,
675                        detach,
676                        valid_from,
677                    }
678                }
679                BatchOperation::DeleteEdge {
680                    edge_id,
681                    valid_time,
682                } => {
683                    let edge_id = self.batch_committed_edge(index, *edge_id, &mut written_edges)?;
684                    PlannedOp::DeleteEdge {
685                        edge_id,
686                        valid_from: self.batch_valid_time(index, valid_time)?,
687                    }
688                }
689            };
690            planned.push(planned_op);
691        }
692
693        // Annihilation pass: a batch-created edge whose committed endpoint is
694        // later removed by `delete_node` + `detach: true` is elided from the
695        // transaction — the cascade "removes" it (module docs). The earliest
696        // such delete wins. A to-be-elided edge carrying an explicit
697        // valid_time is REJECTED instead: eliding it would silently erase a
698        // backdated bi-temporal fact that single-op sequencing (create,
699        // commit, then delete) preserves in history.
700        let mut annihilators: Vec<Option<usize>> = vec![None; planned.len()];
701        for (index, op) in planned.iter().enumerate() {
702            let PlannedOp::CreateEdge {
703                source,
704                target,
705                valid_from,
706                ..
707            } = op
708            else {
709                continue;
710            };
711            let mut annihilator: Option<usize> = None;
712            for endpoint in [source, target] {
713                if let ResolvedNode::Committed(node_id) = endpoint
714                    && let Some(&(k, detach)) = deleted_nodes.get(&node_id.as_u64())
715                    && detach
716                    && k > index
717                    && annihilator.is_none_or(|prev| k < prev)
718                {
719                    annihilator = Some(k);
720                }
721            }
722            if let Some(k) = annihilator {
723                if valid_from.is_some() {
724                    let mut details = serde_json::Map::new();
725                    details.insert("removed_by_delete_at_index".to_string(), json!(k));
726                    return Err(self.batch_invalid(
727                        index,
728                        format!(
729                            "Operation {index}: this create_edge carries an explicit valid_time, \
730                             but operation {k} detach-deletes one of its endpoints later in the \
731                             batch, which would elide the edge and silently erase the backdated \
732                             fact. Drop the create_edge from this batch, or commit it in a \
733                             separate call before the delete (single-op sequencing preserves the \
734                             fact in history)."
735                        ),
736                        details,
737                    ));
738                }
739                annihilators[index] = Some(k);
740            }
741        }
742        for (op, annihilator) in planned.iter_mut().zip(annihilators) {
743            if let (Some(k), PlannedOp::CreateEdge { annihilated_by, .. }) = (annihilator, op) {
744                *annihilated_by = Some(k);
745            }
746        }
747
748        Ok(planned)
749    }

786    /// Convert an op's optional properties object.
787    fn batch_properties(
788        &self,
789        index: usize,
790        properties: Option<&HashMap<String, serde_json::Value>>,
791    ) -> Result<PropertyMap, Box<CallToolResult>> {
792        match properties {
793            None => Ok(PropertyMap::default()),
794            Some(p) => self.json_to_property_map(p).map_err(|e| {
795                // A property-KEY interner-cap breach is a typed
796                // `StorageError::CapacityExceeded` (configurable interner cap,
797                // Issue #3716) that must render the SAME structured
798                // `FAILED_PRECONDITION` envelope the single-op create_node/HTTP
799                // surfaces emit — naming `persistence.max_interned_strings` with
800                // `{resource, current, limit}` details — not the generic
801                // `INVALID_ARGUMENT` this path previously flattened it to (Issue
802                // #3723). It is routed through `batch_db_error` (which reuses the
803                // same `McpError::from_db_error` mapping create_node uses) so the
804                // envelope is identical, plus the batch's `failed_op_index`.
805                //
806                // This runs in Phase 1 static pre-validation, before any
807                // transaction opens, so a breach leaves ZERO writes (all-or-
808                // nothing is trivially preserved). Every other property error
809                // (bad value, recursion depth) stays a caller-fault
810                // `INVALID_ARGUMENT`, preserving prior behavior.
811                if matches!(
812                    &e,
813                    Error::Storage(crate::core::error::StorageError::CapacityExceeded { .. })
814                ) {
815                    Box::new(self.batch_db_error(Some(index), &e))
816                } else {
817                    self.batch_invalid(
818                        index,
819                        format!("Operation {index}: invalid properties: {e}"),
820                        serde_json::Map::new(),
821                    )
822                }
823            }),
824        }
825    }

853    /// Resolve an edge-endpoint reference statically.
854    fn batch_resolve_ref(
855        &self,
856        index: usize,
857        node_ref: &BatchNodeRef,
858        aliases: &HashMap<&str, usize>,
859        ops: &[BatchOperation],
860        deleted_nodes: &HashMap<u64, (usize, bool)>,
861    ) -> Result<ResolvedNode, Box<CallToolResult>> {
862        match node_ref {
863            BatchNodeRef::Id(id) => {
864                let node_id = NodeId::new(*id).map_err(|e| {
865                    self.batch_invalid(
866                        index,
867                        format!("Operation {index}: {e}"),
868                        serde_json::Map::new(),
869                    )
870                })?;
871                // Referencing a node this batch already deleted can never
872                // commit (validation would abort at commit time); reject it
873                // here with the precise index instead.
874                if let Some(&(deleted_at, _)) = deleted_nodes.get(id) {
875                    let mut details = serde_json::Map::new();
876                    details.insert("node_id".to_string(), json!(id));
877                    details.insert("deleted_at_index".to_string(), json!(deleted_at));
878                    return Err(self.batch_invalid(
879                        index,
880                        format!(
881                            "Operation {index}: references node {id}, which operation \
882                             {deleted_at} deletes earlier in this batch"
883                        ),
884                        details,
885                    ));
886                }
887                Ok(ResolvedNode::Committed(node_id))
888            }
889            BatchNodeRef::Ref(s) => {
890                let Some(name) = s.strip_prefix('$') else {
891                    return Err(self.batch_invalid(
892                        index,
893                        format!(
894                            "Operation {index}: node reference \"{s}\" must be an integer node \
895                             id or a local reference starting with '$'"
896                        ),
897                        serde_json::Map::new(),
898                    ));
899                };
900                // Caller-supplied aliases first; purely numeric names are
901                // positional indices (aliases are validated non-numeric).
902                // The positional form is all-ASCII-digits only, so "$+5"
903                // never silently parses positionally (usize::from_str would
904                // accept a leading '+') — it falls through to the
905                // unknown-ref error. Leading zeros are fine: "$007" == "$7".
906                let defined_at = if let Some(&d) = aliases.get(name) {
907                    Some(d)
908                } else if !name.is_empty() && name.bytes().all(|b| b.is_ascii_digit()) {
909                    name.parse::<usize>()
910                        .ok()
911                        .and_then(|p| (p < ops.len()).then_some(p))
912                } else {
913                    None
914                };
915                let Some(defined_at) = defined_at else {
916                    let mut details = serde_json::Map::new();
917                    details.insert("ref".to_string(), json!(s));
918                    return Err(self.batch_invalid(
919                        index,
920                        format!(
921                            "Operation {index}: unknown local reference \"{s}\" (no create_node \
922                             in this batch defines it)"
923                        ),
924                        details,
925                    ));
926                };
927                if !matches!(ops[defined_at], BatchOperation::CreateNode { .. }) {
928                    let mut details = serde_json::Map::new();
929                    details.insert("ref".to_string(), json!(s));
930                    details.insert("defined_at_index".to_string(), json!(defined_at));
931                    return Err(self.batch_invalid(
932                        index,
933                        format!(
934                            "Operation {index}: local reference \"{s}\" points at operation \
935                             {defined_at}, which is not a create_node"
936                        ),
937                        details,
938                    ));
939                }
940                if defined_at >= index {
941                    let mut details = serde_json::Map::new();
942                    details.insert("ref".to_string(), json!(s));
943                    details.insert("defined_at_index".to_string(), json!(defined_at));
944                    return Err(self.batch_invalid(
945                        index,
946                        format!(
947                            "Operation {index}: forward reference \"{s}\" — the node it names \
948                             is only created later, at operation {defined_at}; reorder the \
949                             batch so creations precede references"
950                        ),
951                        details,
952                    ));
953                }
954                Ok(ResolvedNode::Local(defined_at))
955            }
956        }
957    }

959    /// Validate a committed-node target for update/delete ops: local refs
960    /// are a documented v1 rejection, and each committed entity accepts at
961    /// most one write per batch.
962    fn batch_committed_node(
963        &self,
964        index: usize,
965        node_ref: &BatchNodeRef,
966        verb: &str,
967        written_nodes: &mut HashMap<u64, usize>,
968    ) -> Result<NodeId, Box<CallToolResult>> {
969        let id = match node_ref {
970            BatchNodeRef::Id(id) => *id,
971            // Only '$'-prefixed strings are local refs and get the v1-scope
972            // message; any other string (e.g. a quoted number like "123") is
973            // a plain type error, reported as such.
974            BatchNodeRef::Ref(s) if s.starts_with('$') => {
975                let mut details = serde_json::Map::new();
976                details.insert("ref".to_string(), json!(s));
977                return Err(self.batch_invalid(
978                    index,
979                    format!(
980                        "Operation {index}: cannot {verb} \"{s}\" — updating or deleting a node \
981                         created in the same batch is not supported in v1. Commit the creation \
982                         first, then {verb} it in a follow-up call."
983                    ),
984                    details,
985                ));
986            }
987            BatchNodeRef::Ref(s) => {
988                return Err(self.batch_invalid(
989                    index,
990                    format!(
991                        "Operation {index}: node_id must be an integer node id, got string \
992                         \"{s}\""
993                    ),
994                    serde_json::Map::new(),
995                ));
996            }
997        };
998        let node_id = NodeId::new(id).map_err(|e| {
999            self.batch_invalid(
1000                index,
1001                format!("Operation {index}: {e}"),
1002                serde_json::Map::new(),
1003            )
1004        })?;
1005        if let Some(&first) = written_nodes.get(&id) {
1006            let mut details = serde_json::Map::new();
1007            details.insert("node_id".to_string(), json!(id));
1008            details.insert("first_write_op_index".to_string(), json!(first));
1009            return Err(self.batch_invalid(
1010                index,
1011                format!(
1012                    "Operation {index}: node {id} is already written by operation {first}; a \
1013                     batch accepts at most one write per committed entity (the second write \
1014                     would act on a stale read of the pre-batch state)"
1015                ),
1016                details,
1017            ));
1018        }
1019        written_nodes.insert(id, index);
1020        Ok(node_id)
1021    }

1060    /// Execute the pre-validated operations in order inside `tx`, returning
1061    /// the per-op result objects (input order) and the alias -> real-id map.
1062    #[allow(clippy::type_complexity)]
1063    fn execute_batch(
1064        &self,
1065        tx: &mut crate::api::transaction::WriteTransaction,
1066        planned: &[PlannedOp],
1067    ) -> Result<
1068        (
1069            Vec<serde_json::Value>,
1070            serde_json::Map<String, serde_json::Value>,
1071        ),
1072        BatchAbort,
1073    > {
1074        // Real NodeId for each create_node op, by op index (local-ref target).
1075        let mut created_nodes: Vec<Option<NodeId>> = vec![None; planned.len()];
1076        // Batch-local adjacency ledger: edges created in this batch, with
1077        // resolved endpoints. `count_connected_edges`/`delete_node_cascade`
1078        // only see committed storage, so the #3209 contract inside a batch
1079        // is enforced from this ledger (Issue #3231).
1080        let mut batch_edges: Vec<BatchEdge> = Vec::new();
1081        // Committed edges deleted so far (explicitly or by cascade) -> op index.
1082        let mut deleted_committed_edges: HashMap<u64, usize> = HashMap::new();
1083        // Committed edges updated so far -> (endpoints, op index), to refuse
1084        // a later cascade silently discarding the update.
1085        let mut updated_committed_edges: HashMap<u64, (NodeId, NodeId, usize)> = HashMap::new();
1086        // Integer ids of nodes/edges CREATED earlier in this batch, mapped to
1087        // the op index that created them. An update/delete targeting one of
1088        // these is a guessed-id write against an entity not yet committed —
1089        // the v1-scope rejection the static `$alias` prevalidation
1090        // (`batch_committed_node`) enforces for the ref SPELLING. Enforced
1091        // explicitly here because buffer-aware transaction reads (Issue #3417)
1092        // removed the incidental `NOT_FOUND` that used to catch the integer-id
1093        // form (pinned by
1094        // apply_batch_guessed_id_of_batch_created_node_aborts_whole_batch).
1095        let mut created_node_ids: HashMap<u64, usize> = HashMap::new();
1096        let mut created_edge_ids: HashMap<u64, usize> = HashMap::new();
1097
1098        let mut results = Vec::with_capacity(planned.len());
1099        let mut ref_map = serde_json::Map::new();
1100
1101        let op_err = |index: usize| move |source: Error| BatchAbort::Op { index, source };
1102
1103        for (index, op) in planned.iter().enumerate() {
1104            match op {
1105                PlannedOp::CreateNode {
1106                    label,
1107                    properties,
1108                    valid_from,
1109                    provenance,
1110                    alias,
1111                } => {
1112                    let options = write_options(*valid_from, provenance.clone());
1113                    let node_id = tx
1114                        .create_node_with_options(label, properties.clone(), options)
1115                        .map_err(op_err(index))?;
1116                    let version_id = tx.buffered_node_version(node_id).map(|v| v.as_u64());
1117                    created_nodes[index] = Some(node_id);
1118                    created_node_ids.insert(node_id.as_u64(), index);
1119                    if let Some(alias) = alias {
1120                        ref_map.insert(alias.clone(), json!(node_id.as_u64()));
1121                    }
1122                    results.push(json!({
1123                        "op": "create_node",
1124                        "index": index,
1125                        "ref": alias,
1126                        "node_id": node_id.as_u64(),
1127                        "version_id": version_id
1128                    }));
1129                }
1130                PlannedOp::CreateEdge {
1131                    source,
1132                    target,
1133                    label,
1134                    properties,
1135                    valid_from,
1136                    provenance,
1137                    annihilated_by,
1138                } => {
1139                    let source = resolve_local(index, *source, &created_nodes)?;
1140                    let target = resolve_local(index, *target, &created_nodes)?;
1141                    if let Some(k) = annihilated_by {
1142                        // Elided edge: a later detach-delete of an endpoint
1143                        // removes it (see module docs). Never issued to the
1144                        // transaction; recorded in the ledger for counting.
1145                        batch_edges.push(BatchEdge {
1146                            source,
1147                            target,
1148                            annihilated_by: Some(*k),
1149                        });
1150                        results.push(json!({
1151                            "op": "create_edge",
1152                            "index": index,
1153                            "edge_id": serde_json::Value::Null,
1154                            "version_id": serde_json::Value::Null,
1155                            "source_id": source.as_u64(),
1156                            "target_id": target.as_u64(),
1157                            "removed_by_delete_at_index": k
1158                        }));
1159                        continue;
1160                    }
1161                    let options = write_options(*valid_from, provenance.clone());
1162                    let edge_id = tx
1163                        .create_edge_with_options(
1164                            source,
1165                            target,
1166                            label,
1167                            properties.clone(),
1168                            options,
1169                        )
1170                        .map_err(op_err(index))?;
1171                    let version_id = tx.buffered_edge_version(edge_id).map(|v| v.as_u64());
1172                    created_edge_ids.insert(edge_id.as_u64(), index);
1173                    batch_edges.push(BatchEdge {
1174                        source,
1175                        target,
1176                        annihilated_by: None,
1177                    });
1178                    results.push(json!({
1179                        "op": "create_edge",
1180                        "index": index,
1181                        "edge_id": edge_id.as_u64(),
1182                        "version_id": version_id,
1183                        "source_id": source.as_u64(),
1184                        "target_id": target.as_u64()
1185                    }));
1186                }
1187                PlannedOp::UpdateNode {
1188                    node_id,
1189                    properties,
1190                    valid_from,
1191                    provenance,
1192                } => {
1193                    if let Some(&created_at) = created_node_ids.get(&node_id.as_u64()) {
1194                        return Err(BatchAbort::BatchCreatedRefWrite {
1195                            index,
1196                            entity: "node",
1197                            id: node_id.as_u64(),
1198                            created_at_index: created_at,
1199                            verb: "update",
1200                        });
1201                    }
1202                    let options = write_options(*valid_from, provenance.clone());
1203                    tx.update_node_with_options(*node_id, properties.clone(), options)
1204                        .map_err(op_err(index))?;
1205                    let version_id = tx.buffered_node_version(*node_id).map(|v| v.as_u64());
1206                    results.push(json!({
1207                        "op": "update_node",
1208                        "index": index,
1209                        "node_id": node_id.as_u64(),
1210                        "version_id": version_id
1211                    }));
1212                }
1213                PlannedOp::UpdateEdge {
1214                    edge_id,
1215                    properties,
1216                    valid_from,
1217                    provenance,
1218                } => {
1219                    if let Some(&created_at) = created_edge_ids.get(&edge_id.as_u64()) {
1220                        return Err(BatchAbort::BatchCreatedRefWrite {
1221                            index,
1222                            entity: "edge",
1223                            id: edge_id.as_u64(),
1224                            created_at_index: created_at,
1225                            verb: "update",
1226                        });
1227                    }
1228                    if let Some(&removed_at) = deleted_committed_edges.get(&edge_id.as_u64()) {
1229                        return Err(BatchAbort::EdgeRemovedByCascade {
1230                            index,
1231                            edge_id: edge_id.as_u64(),
1232                            removed_at_index: removed_at,
1233                        });
1234                    }
1235                    // Endpoints recorded so a later detach-delete of either
1236                    // endpoint refuses rather than discarding this update.
1237                    let edge = tx.get_edge(*edge_id).map_err(op_err(index))?;
1238                    let options = write_options(*valid_from, provenance.clone());
1239                    tx.update_edge_with_options(*edge_id, properties.clone(), options)
1240                        .map_err(op_err(index))?;
1241                    let version_id = tx.buffered_edge_version(*edge_id).map(|v| v.as_u64());
1242                    updated_committed_edges
1243                        .insert(edge_id.as_u64(), (edge.source, edge.target, index));
1244                    results.push(json!({
1245                        "op": "update_edge",
1246                        "index": index,
1247                        "edge_id": edge_id.as_u64(),
1248                        "version_id": version_id
1249                    }));
1250                }
1251                PlannedOp::DeleteNode {
1252                    node_id,
1253                    detach,
1254                    valid_from,
1255                } => {
1256                    if let Some(&created_at) = created_node_ids.get(&node_id.as_u64()) {
1257                        return Err(BatchAbort::BatchCreatedRefWrite {
1258                            index,
1259                            entity: "node",
1260                            id: node_id.as_u64(),
1261                            created_at_index: created_at,
1262                            verb: "delete",
1263                        });
1264                    }
1265                    let edges_removed = self.batch_delete_node(
1266                        tx,
1267                        index,
1268                        *node_id,
1269                        *detach,
1270                        *valid_from,
1271                        &batch_edges,
1272                        &mut deleted_committed_edges,
1273                        &updated_committed_edges,
1274                    )?;
1275                    results.push(json!({
1276                        "op": "delete_node",
1277                        "index": index,
1278                        "node_id": node_id.as_u64(),
1279                        "detached": detach,
1280                        "edges_removed": edges_removed
1281                    }));
1282                }
1283                PlannedOp::DeleteEdge {
1284                    edge_id,
1285                    valid_from,
1286                } => {
1287                    if let Some(&created_at) = created_edge_ids.get(&edge_id.as_u64()) {
1288                        return Err(BatchAbort::BatchCreatedRefWrite {
1289                            index,
1290                            entity: "edge",
1291                            id: edge_id.as_u64(),
1292                            created_at_index: created_at,
1293                            verb: "delete",
1294                        });
1295                    }
1296                    if let Some(&removed_at) = deleted_committed_edges.get(&edge_id.as_u64()) {
1297                        return Err(BatchAbort::EdgeRemovedByCascade {
1298                            index,
1299                            edge_id: edge_id.as_u64(),
1300                            removed_at_index: removed_at,
1301                        });
1302                    }
1303                    tx.delete_edge_with_valid_time(*edge_id, *valid_from)
1304                        .map_err(op_err(index))?;
1305                    deleted_committed_edges.insert(edge_id.as_u64(), index);
1306                    results.push(json!({
1307                        "op": "delete_edge",
1308                        "index": index,
1309                        "edge_id": edge_id.as_u64()
1310                    }));
1311                }
1312            }
1313        }
1314
1315        Ok((results, ref_map))
1316    }

1318    /// Execute a `delete_node` op under the #3209 safe-by-default contract,
1319    /// counted against committed **and** batch-created adjacency.
1320    ///
1321    /// Returns the number of edges removed (committed edges cascaded now,
1322    /// plus batch-created edges elided by this op).
1323    #[allow(clippy::too_many_arguments)]
1324    fn batch_delete_node(
1325        &self,
1326        tx: &mut crate::api::transaction::WriteTransaction,
1327        index: usize,
1328        node_id: NodeId,
1329        detach: bool,
1330        valid_from: Option<Timestamp>,
1331        batch_edges: &[BatchEdge],
1332        deleted_committed_edges: &mut HashMap<u64, usize>,
1333        updated_committed_edges: &HashMap<u64, (NodeId, NodeId, usize)>,
1334    ) -> Result<usize, BatchAbort> {
1335        // Existence check through the transaction (same snapshot the delete
1336        // will use), so a nonexistent node reports NOT_FOUND at this index.
1337        // A guessed numeric id that a create earlier in this batch allocated
1338        // is rejected BEFORE reaching here by the explicit batch-created-ref
1339        // guard in `execute_batch` (Issue #3417) — this read only sees
1340        // genuinely committed nodes.
1341        tx.get_node(node_id)
1342            .map_err(|source| BatchAbort::Op { index, source })?;
1343
1344        // DISTINCT committed edges (a self-loop appears in both adjacency
1345        // directions but is one edge — mirrors the retract_node dedup),
1346        // minus committed edges this batch already deleted.
1347        let mut committed_edges = tx
1348            .get_outgoing_edges(node_id)
1349            .map_err(|source| BatchAbort::Op { index, source })?;
1350        committed_edges.extend(
1351            tx.get_incoming_edges(node_id)
1352                .map_err(|source| BatchAbort::Op { index, source })?,
1353        );
1354        committed_edges.sort_unstable();
1355        committed_edges.dedup();
1356        let live_committed: Vec<EdgeId> = committed_edges
1357            .into_iter()
1358            .filter(|e| !deleted_committed_edges.contains_key(&e.as_u64()))
1359            .collect();
1360
1361        // Batch-created edges still alive at this op (ledger; one entry per
1362        // edge, so a batch self-loop naturally counts once).
1363        let live_batch = batch_edges
1364            .iter()
1365            .filter(|e| e.touches(node_id) && e.alive_at(index))
1366            .count();
1367
1368        let connected_edges = live_committed.len() + live_batch;
1369
1370        // Refuse-by-default: never orphan committed or batch-created edges.
1371        if connected_edges > 0 && !detach {
1372            return Err(BatchAbort::Refused {
1373                index,
1374                node_id: node_id.as_u64(),
1375                connected_edges,
1376            });
1377        }
1378
1379        if detach {
1380            // A cascade would silently discard an edge update made earlier
1381            // in this batch; refuse with the precise indices instead (commit
1382            // validation would abort the batch anyway, less precisely).
1383            // Deterministic attribution: when several updated edges touch
1384            // this node, report the EARLIEST update (minimum
1385            // updated_at_index), never arbitrary HashMap iteration order.
1386            let mut conflict: Option<(u64, usize)> = None;
1387            for (edge_id, (source, target, updated_at)) in updated_committed_edges {
1388                if (*source == node_id || *target == node_id)
1389                    && !deleted_committed_edges.contains_key(edge_id)
1390                    && conflict.is_none_or(|(_, prev)| *updated_at < prev)
1391                {
1392                    conflict = Some((*edge_id, *updated_at));
1393                }
1394            }
1395            if let Some((edge_id, updated_at_index)) = conflict {
1396                return Err(BatchAbort::UpdatedEdgeCascade {
1397                    index,
1398                    edge_id,
1399                    node_id: node_id.as_u64(),
1400                    updated_at_index,
1401                });
1402            }
1403
1404            // Manual cascade over committed edges (delete_node_cascade would
1405            // re-enumerate committed adjacency and double-delete edges this
1406            // batch already removed). Batch-created edges touching this node
1407            // were elided at their create op (annihilated_by == index).
1408            for edge_id in &live_committed {
1409                tx.delete_edge(*edge_id)
1410                    .map_err(|source| BatchAbort::Op { index, source })?;
1411                deleted_committed_edges.insert(edge_id.as_u64(), index);
1412            }
1413            tx.delete_node_with_valid_time(node_id, None)
1414                .map_err(|source| BatchAbort::Op { index, source })?;
1415
1416            let elided_batch = batch_edges
1417                .iter()
1418                .filter(|e| e.touches(node_id) && e.annihilated_by == Some(index))
1419                .count();
1420            Ok(live_committed.len() + elided_batch)
1421        } else {
1422            // No connected edges: a plain delete cannot orphan anything.
1423            tx.delete_node_with_valid_time(node_id, valid_from)
1424                .map_err(|source| BatchAbort::Op { index, source })?;
1425            Ok(0)
1426        }
1427    }

1433    /// Map a [`BatchAbort`] to the structured error response. Every shape
1434    /// carries `details.failed_op_index` (JSON `null` for commit-phase
1435    /// failures, which have no attributable op).
1436    fn batch_abort_result(&self, abort: BatchAbort) -> CallToolResult {
1437        match abort {
1438            BatchAbort::Op { index, source } => self.batch_db_error(Some(index), &source),
1439            BatchAbort::Commit { source } => self.batch_db_error(None, &source),
1440            BatchAbort::Internal { index, message } => self.error_result(
1441                McpError::new(McpErrorCode::Internal, message)
1442                    .details(json!({ "failed_op_index": index })),
1443            ),
1444            BatchAbort::Refused {
1445                index,
1446                node_id,
1447                connected_edges,
1448            } => {
1449                // The #3209 refusal in the #3234 structured shape, with the
1450                // batch's failed_op_index added and the legacy top-level
1451                // fields preserved (mirrors handle_delete_node).
1452                let message = format!(
1453                    "Operation {index}: node {node_id} has {connected_edges} connected edge(s) \
1454                     at this point in the batch; refusing to delete. Pass `detach: true` on \
1455                     this operation to delete the node and its connected edges, or remove the \
1456                     edges first. The whole batch was rolled back."
1457                );
1458                let mut top_level = serde_json::Map::new();
1459                top_level.insert("success".to_string(), json!(false));
1460                top_level.insert("node_id".to_string(), json!(node_id));
1461                top_level.insert("connected_edges".to_string(), json!(connected_edges));
1462                top_level.insert("detach_required".to_string(), json!(true));
1463                Self::error_result_with_top_level(
1464                    McpError::new(McpErrorCode::FailedPrecondition, message).details(json!({
1465                        "failed_op_index": index,
1466                        "node_id": node_id,
1467                        "connected_edges": connected_edges,
1468                        "detach_required": true
1469                    })),
1470                    top_level,
1471                )
1472            }
1473            BatchAbort::UpdatedEdgeCascade {
1474                index,
1475                edge_id,
1476                node_id,
1477                updated_at_index,
1478            } => self.error_result(
1479                McpError::new(
1480                    McpErrorCode::FailedPrecondition,
1481                    format!(
1482                        "Operation {index}: detach-deleting node {node_id} would discard the \
1483                         update to edge {edge_id} made by operation {updated_at_index}; remove \
1484                         either the update or the delete. The whole batch was rolled back."
1485                    ),
1486                )
1487                .details(json!({
1488                    "failed_op_index": index,
1489                    "node_id": node_id,
1490                    "edge_id": edge_id,
1491                    "updated_at_index": updated_at_index
1492                })),
1493            ),
1494            BatchAbort::EdgeRemovedByCascade {
1495                index,
1496                edge_id,
1497                removed_at_index,
1498            } => self.error_result(
1499                McpError::new(
1500                    McpErrorCode::FailedPrecondition,
1501                    format!(
1502                        "Operation {index}: edge {edge_id} was already removed earlier in this \
1503                         batch by the detach delete at operation {removed_at_index}. The whole \
1504                         batch was rolled back."
1505                    ),
1506                )
1507                .details(json!({
1508                    "failed_op_index": index,
1509                    "edge_id": edge_id,
1510                    "removed_at_index": removed_at_index
1511                })),
1512            ),
1513            // Mirrors the static `$alias` prevalidation in
1514            // `batch_committed_node`: same INVALID_ARGUMENT code and the same
1515            // "created in the same batch is not supported in v1" phrasing, for
1516            // the guessed integer-id form.
1517            BatchAbort::BatchCreatedRefWrite {
1518                index,
1519                entity,
1520                id,
1521                created_at_index,
1522                verb,
1523            } => {
1524                let mut details = serde_json::Map::new();
1525                details.insert("failed_op_index".to_string(), json!(index));
1526                details.insert(format!("{entity}_id"), json!(id));
1527                details.insert("created_at_index".to_string(), json!(created_at_index));
1528                self.error_result(
1529                    McpError::new(
1530                        McpErrorCode::InvalidArgument,
1531                        format!(
1532                            "Operation {index}: cannot {verb} {entity} {id} — updating or \
1533                             deleting a {entity} created in the same batch (operation \
1534                             {created_at_index}) is not supported in v1. Commit the creation \
1535                             first, then {verb} it in a follow-up call. The whole batch was \
1536                             rolled back."
1537                        ),
1538                    )
1539                    .details(serde_json::Value::Object(details)),
1540                )
1541            }
1542        }
1543    }

1545    /// Classify a database error per #3234, augmented with the batch's
1546    /// `failed_op_index`. Mirrors `db_error`'s unique-violation enrichment
1547    /// (structured details plus legacy top-level fields).
1548    fn batch_db_error(&self, index: Option<usize>, e: &Error) -> CallToolResult {
1549        // `with_detail` MERGES (rather than replaces), so any structured details
1550        // the classifier already attached survive — e.g. a configurable
1551        // interner-cap `CapacityExceeded` keeps its `{resource, current, limit}`
1552        // (Issue #3723), byte-identical to the create_node/HTTP envelope, with
1553        // the batch's `failed_op_index` added alongside.
1554        let mut err = McpError::from_db_error(e).with_detail("failed_op_index", json!(index));
1555        let mut top_level = serde_json::Map::new();
1556
1557        if let Some(crate::core::error::ConstraintError::UniqueViolation {
1558            label,
1559            property,
1560            value,
1561            existing_node_id,
1562        }) = e.as_constraint()
1563        {
1564            err = err
1565                .with_detail("label", json!(label))
1566                .with_detail("property", json!(property))
1567                .with_detail("value", json!(value))
1568                .with_detail("existing_node_id", json!(existing_node_id.as_u64()));
1569            top_level.insert("success".to_string(), json!(false));
1570            top_level.insert("constraint_violation".to_string(), json!(true));
1571            top_level.insert("label".to_string(), json!(label));
1572            top_level.insert("property".to_string(), json!(property));
1573            top_level.insert("value".to_string(), json!(value));
1574            top_level.insert(
1575                "existing_node_id".to_string(),
1576                json!(existing_node_id.as_u64()),
1577            );
1578        }
1579
1580        Self::error_result_with_top_level(err, top_level)
1581    }

1603/// Turn a statically resolved endpoint into a real `NodeId` at execution
1604/// time. A `Local` target always names an earlier, already-executed
1605/// `create_node` op (pre-validation guarantees it); a missing id here is an
1606/// internal invariant breach, reported as such rather than panicking.
1607fn resolve_local(
1608    index: usize,
1609    resolved: ResolvedNode,
1610    created_nodes: &[Option<NodeId>],
1611) -> Result<NodeId, BatchAbort> {
1612    match resolved {
1613        ResolvedNode::Committed(id) => Ok(id),
1614        ResolvedNode::Local(defined_at) => created_nodes
1615            .get(defined_at)
1616            .copied()
1617            .flatten()
1618            .ok_or_else(|| BatchAbort::Internal {
1619                index,
1620                message: format!(
1621                    "internal invariant breach: local ref target (op {defined_at}) has no \
1622                     allocated node id"
1623                ),
1624            }),
1625    }
