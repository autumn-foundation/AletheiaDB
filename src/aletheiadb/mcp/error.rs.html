1//! Structured MCP error codes with a retriable flag (Issue #3234).
2//!
3//! Every MCP tool error response is a JSON object of the shape:
4//!
5//! ```json
6//! {
7//!   "error": {
8//!     "code": "NOT_FOUND",
9//!     "message": "Storage error: Node not found: Node(123)",
10//!     "retriable": false,
11//!     "details": { "...": "optional, structured, per-code metadata" }
12//!   }
13//! }
14//! ```
15//!
16//! - `code` is drawn from the small, stable [`McpErrorCode`] enum, modeled on
17//!   gRPC's canonical codes. Clients branch on `code`, never on `message`.
18//! - `message` preserves the human-readable free text previously returned as
19//!   the whole `error` value (additive change — no information lost).
20//! - `retriable` is an explicit, advisory server-side classification: `true`
21//!   only for transient failure classes (timeouts, clock skew, serialization
22//!   conflicts), `false` for caller-fault classes (not-found,
23//!   invalid-argument, constraint violations). The client owns the retry
24//!   loop; the server never retries on its behalf.
25//! - `details` carries optional structured metadata (e.g. the DETACH-delete
26//!   refusal's `connected_edges`, a unique violation's `existing_node_id`).
27//!
28//! The classification from AletheiaDB's internal [`Error`] taxonomy to MCP
29//! codes happens only at this boundary — the library's `thiserror` enums are
30//! unchanged.
31
32use crate::core::error::{
33    ConstraintError, Error, QueryError, StorageError, TemporalError, TransactionError, VectorError,
34};

36/// Stable, machine-readable error codes for the MCP surface.
37///
38/// The wire representation ([`McpErrorCode::as_str`]) is SCREAMING_SNAKE_CASE
39/// and is a **stability contract**: codes may be added, but existing codes
40/// never change meaning or spelling.
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub enum McpErrorCode {
43    /// The referenced entity (node, edge, version, tool, ...) does not exist,
44    /// or did not exist at the requested bi-temporal coordinate.
45    NotFound,
46    /// The request itself is malformed: bad JSON arguments, an out-of-range
47    /// ID, an unparseable timestamp, inconsistent parameter combinations, or
48    /// an invalid query statement. Fix the arguments before retrying.
49    InvalidArgument,
50    /// A declared uniqueness constraint rejected the write.
51    ConstraintViolation,
52    /// The operation is valid but the system is not in the required state:
53    /// e.g. a vector index is not enabled, a node still has connected edges
54    /// and `detach` was not passed, or a required build feature is not
55    /// compiled in. Change the state (or the call), then retry.
56    FailedPrecondition,
57    /// A concurrency conflict: serialization failure, write-write conflict,
58    /// aborted transaction, or duplicate ID race. Usually retriable.
59    Conflict,
60    /// A transient condition: query timeout, clock skew, or other
61    /// should-heal-itself failure. Retriable.
62    Unavailable,
63    /// An unexpected internal failure (I/O, corruption, poisoned lock,
64    /// serialization of a response, ...). Not retriable; report it.
65    Internal,
66    /// The caller is not authenticated (Issue #3350): the surface requires a
67    /// credential and none was supplied, or the supplied credential is
68    /// unknown or revoked. Deliberately uniform across all of those cases —
69    /// the error never reveals whether a presented key exists and never
70    /// echoes credential material. Not retriable as-is: obtain a valid
71    /// credential, then retry.
72    Unauthenticated,
73    /// The caller is authenticated but their role does not permit the
74    /// operation's access class (Issue #3350). `details` carries
75    /// `{required_class, principal_role}`. Not retriable: the same call
76    /// under the same principal can never succeed.
77    PermissionDenied,
78    /// A per-query resource limit was exceeded (Issue #3368): the wall-clock
79    /// timeout elapsed, or the result exceeded the response-byte cap.
80    /// `details` carries `{dimension, limit, consumed?}`. Retriability is
81    /// case-specific and set explicitly (a read-only wall-clock timeout is
82    /// retriable; a byte-cap breach is not), so the default is the
83    /// conservative non-retriable — treat an unqualified `RESOURCE_EXHAUSTED`
84    /// as non-retriable.
85    ResourceExhausted,
86}
87
88impl McpErrorCode {
89    /// The stable wire representation of this code.
90    pub fn as_str(self) -> &'static str {
91        match self {
92            McpErrorCode::NotFound => "NOT_FOUND",
93            McpErrorCode::InvalidArgument => "INVALID_ARGUMENT",
94            McpErrorCode::ConstraintViolation => "CONSTRAINT_VIOLATION",
95            McpErrorCode::FailedPrecondition => "FAILED_PRECONDITION",
96            McpErrorCode::Conflict => "CONFLICT",
97            McpErrorCode::Unavailable => "UNAVAILABLE",
98            McpErrorCode::Internal => "INTERNAL",
99            McpErrorCode::Unauthenticated => "UNAUTHENTICATED",
100            McpErrorCode::PermissionDenied => "PERMISSION_DENIED",
101            McpErrorCode::ResourceExhausted => "RESOURCE_EXHAUSTED",
102        }
103    }
104
105    /// Default retriable classification for this code.
106    ///
107    /// `Conflict` and `Unavailable` are transient by default; every
108    /// caller-fault and internal class defaults to non-retriable. Individual
109    /// errors may override (e.g. a `DuplicateId` maps to `Conflict` but is
110    /// *not* retriable, because retrying the same ID cannot succeed).
111    pub fn default_retriable(self) -> bool {
112        matches!(self, McpErrorCode::Conflict | McpErrorCode::Unavailable)
113    }
114}

116impl std::fmt::Display for McpErrorCode {
117    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
118        f.write_str(self.as_str())
119    }
120}

122/// A structured MCP error payload: code + message + retriable (+ details).
123#[derive(Debug, Clone)]
124pub struct McpError {
125    code: McpErrorCode,
126    message: String,
127    retriable: bool,
128    details: Option<serde_json::Value>,
129}
130
131impl McpError {
132    /// Build an error with the code's default retriable classification.
133    pub fn new(code: McpErrorCode, message: impl Into<String>) -> Self {
134        Self {
135            code,
136            message: message.into(),
137            retriable: code.default_retriable(),
138            details: None,
139        }
140    }
141
142    /// Override the retriable flag (e.g. non-retriable `Conflict`).
143    pub fn retriable(mut self, retriable: bool) -> Self {
144        self.retriable = retriable;
145        self
146    }
147
148    /// Attach structured, per-code metadata under `error.details`.
149    pub fn details(mut self, details: serde_json::Value) -> Self {
150        self.details = Some(details);
151        self
152    }
153
154    /// Merge a single `key`/`value` into the existing `error.details` object,
155    /// preserving any structured details already present (e.g. the
156    /// `{resource, current, limit}` a `from_db_error` capacity error carries).
157    ///
158    /// If no details object exists yet — or the existing details is not a JSON
159    /// object — a fresh single-key object is created. This is the additive
160    /// counterpart to [`Self::details`] (which replaces): call sites that need
161    /// to enrich a classified error with call-site metadata (e.g. the batch
162    /// surface's `failed_op_index`) use this so the classifier's own details are
163    /// not clobbered.
164    pub fn with_detail(mut self, key: impl Into<String>, value: serde_json::Value) -> Self {
165        match &mut self.details {
166            Some(serde_json::Value::Object(map)) => {
167                map.insert(key.into(), value);
168            }
169            _ => {
170                let mut map = serde_json::Map::new();
171                map.insert(key.into(), value);
172                self.details = Some(serde_json::Value::Object(map));
173            }
174        }
175        self
176    }
177
178    /// Replace the message (e.g. to add call-site context around the
179    /// classified error's own text) without changing the classification.
180    pub fn with_message(mut self, message: impl Into<String>) -> Self {
181        self.message = message.into();
182        self
183    }
184
185    /// The code assigned to this error.
186    pub fn code(&self) -> McpErrorCode {
187        self.code
188    }
189
190    /// Whether this error is advisorily retriable.
191    pub fn is_retriable(&self) -> bool {
192        self.retriable
193    }
194
195    /// The human-readable message.
196    pub fn message(&self) -> &str {
197        &self.message
198    }
199
200    /// Serialize the inner error object (the value of the `"error"` key).
201    pub fn to_json(&self) -> serde_json::Value {
202        let mut obj = serde_json::Map::new();
203        obj.insert("code".to_string(), self.code.as_str().into());
204        obj.insert("message".to_string(), self.message.clone().into());
205        obj.insert("retriable".to_string(), self.retriable.into());
206        if let Some(details) = &self.details {
207            obj.insert("details".to_string(), details.clone());
208        }
209        serde_json::Value::Object(obj)
210    }
211
212    /// Classify an internal database [`Error`] into an MCP error.
213    ///
214    /// This is the single boundary mapping from the library's `thiserror`
215    /// taxonomy to the stable MCP codes; the message is always
216    /// `e.to_string()`, preserving the pre-#3234 free text verbatim.
217    ///
218    /// Schema-constraint violations (Issue #3378) additionally carry their
219    /// structured `details` here — `TypeViolation`'s `expected_type`/
220    /// `actual_type`, `MissingRequiredKey`'s `missing_keys`, and
221    /// `NonConformingOnEnable`'s bounded violation report — so a caller can
222    /// self-repair without parsing the free-text `message`. This mirrors the
223    /// `UniqueViolation` precedent (whose `existing_node_id` details are built
224    /// at the `server::db_error` call site, along with its legacy top-level
225    /// fields); that path is unchanged because
226    /// [`ConstraintError::structured_details`] returns `None` for it.
227    pub fn from_db_error(e: &Error) -> Self {
228        let (code, retriable) = classify_db_error(e);
229        let details = e
230            .as_constraint()
231            .and_then(|ce| ce.structured_details())
232            .or_else(|| principal_quota_details(e))
233            .or_else(|| namespace_details(e))
234            .or_else(|| capacity_exceeded_details(e));
235        // A string-interner capacity exhaustion (configurable interner cap) gets
236        // an actionable message naming the knob and the restart requirement, so a
237        // caller/LLM can resolve it without parsing the raw Display text.
238        let message = interner_capacity_message(e).unwrap_or_else(|| e.to_string());
239        Self {
240            code,
241            message,
242            retriable,
243            details,
244        }
245    }
246}

248/// Map an internal error to `(code, retriable)`.
249///
250/// `retriable` is `true` **only** for transient classes: concurrency
251/// conflicts that a fresh attempt can win (serialization failures, write
252/// conflicts, aborted transactions) and self-healing conditions (timeouts,
253/// clock skew). Caller-fault classes (not-found, invalid-argument,
254/// constraint, failed-precondition) and persistent internal failures
255/// (corruption, poisoned locks, I/O) are never retriable.
256fn classify_db_error(e: &Error) -> (McpErrorCode, bool) {
257    match e {
258        Error::Storage(se) => classify_storage_error(se),
259        Error::Temporal(te) => classify_temporal_error(te),
260        Error::Query(qe) => classify_query_error(qe),
261        Error::Transaction(te) => classify_transaction_error(te),
262        Error::Vector(ve) => classify_vector_error(ve),
263        Error::Constraint(ce) => classify_constraint_error(ce),
264        // A provenance bundle failing validation is a caller fault.
265        Error::Provenance(_) => (McpErrorCode::InvalidArgument, false),
266        Error::Lineage(le) => classify_lineage_error(le),
267        Error::Namespace(ne) => classify_namespace_error(ne),
268        // A PITR (#3374) target outside the achievable window, or a window that
269        // crosses a post-backup vocabulary change, is a caller-fault precondition
270        // failure; each error's Display explains the remediation. All other
271        // backup errors remain Internal.
272        Error::Backup(
273            crate::storage::backup::BackupError::TargetOutsideWindow { .. }
274            | crate::storage::backup::BackupError::WindowCrossesVocabularyChange { .. },
275        ) => (McpErrorCode::FailedPrecondition, false),
276        Error::Io(_) | Error::Backup(_) => (McpErrorCode::Internal, false),
277        // The feature exists but this build/deployment doesn't provide it.
278        Error::NotImplemented { .. } => (McpErrorCode::FailedPrecondition, false),
279        // An opt-in feature is disabled (e.g. the provenance hash chain).
280        Error::FailedPrecondition(_) => (McpErrorCode::FailedPrecondition, false),
281        Error::Other(_) => (McpErrorCode::Internal, false),
282    }
283}

285/// Map a derivation-lineage error (Issue #3371). All lineage errors are
286/// caller faults and never retriable: a dangling source reference is
287/// `NOT_FOUND`, a self/cyclic declaration is `INVALID_ARGUMENT`, and
288/// re-declaring lineage for an already-recorded version is a
289/// `FAILED_PRECONDITION`.
290fn classify_lineage_error(e: &crate::core::lineage::LineageError) -> (McpErrorCode, bool) {
291    use crate::core::lineage::LineageError;
292    match e {
293        LineageError::SourceNotFound { .. } => (McpErrorCode::NotFound, false),
294        LineageError::SelfDerivation(_) | LineageError::CycleDetected { .. } => {
295            (McpErrorCode::InvalidArgument, false)
296        }
297        LineageError::AlreadyRecorded(_) => (McpErrorCode::FailedPrecondition, false),
298    }
299}

301/// Map a namespace error (Issue #3349). All are caller faults and never
302/// retriable: an invalid name or a forged engine-reserved property key is
303/// `INVALID_ARGUMENT`, an unknown namespace is `NOT_FOUND`, and a duplicate
304/// `create_namespace` is `CONFLICT`.
305fn classify_namespace_error(e: &crate::core::namespace::NamespaceError) -> (McpErrorCode, bool) {
306    use crate::core::namespace::NamespaceError;
307    match e {
308        NamespaceError::InvalidName { .. }
309        | NamespaceError::ReservedPropertyKey { .. }
310        | NamespaceError::Immutable => (McpErrorCode::InvalidArgument, false),
311        NamespaceError::NotFound { .. } => (McpErrorCode::NotFound, false),
312        NamespaceError::AlreadyExists { .. } => (McpErrorCode::Conflict, false),
313    }
314}

316/// Structured `details` for a per-principal changefeed quota breach (Issue
317/// #3678): `{principal, current, limit}`. Shared by the MCP and HTTP surfaces so
318/// both render byte-identical metadata under `error.details`. Returns `None` for
319/// every other error.
320fn principal_quota_details(e: &Error) -> Option<serde_json::Value> {
321    if let Error::Storage(StorageError::PrincipalQuotaExceeded {
322        principal,
323        current,
324        limit,
325    }) = e
326    {
327        Some(serde_json::json!({
328            "principal": principal,
329            "current": current,
330            "limit": limit,
331        }))
332    } else {
333        None
334    }
335}

339/// surfaces so both render byte-identical metadata under `error.details`.

349            "resource": resource,
350            "current": current,
351            "limit": limit,

358/// Actionable replacement message for a **string-interner** capacity exhaustion
359/// (configurable interner cap). Names the `persistence.max_interned_strings`
360/// knob and the restart requirement. Returns `None` for any other error
361/// (including non-interner `CapacityExceeded` resources), preserving their
362/// Display text.
363fn interner_capacity_message(e: &Error) -> Option<String> {
364    if let Error::Storage(StorageError::CapacityExceeded {
365        resource,
366        current,
367        limit,
368    }) = e
369        && resource.contains("interner")
370    {
371        return Some(format!(
372            "String interner at capacity ({current}/{limit}). Raise \
373             `persistence.max_interned_strings` above {limit} and restart to intern more \
374             unique strings. No data is lost — the WAL is the source of truth."
375        ));
376    }
377    None
378}

380/// Structured `details` for a namespace error (Issue #3349). An unknown
381/// namespace (`NOT_FOUND`) and a malformed / empty scope (`INVALID_ARGUMENT`)
382/// both carry the offending `namespace` value so a caller can self-correct
383/// without substring-parsing the message; the `InvalidName` case also carries
384/// the validation `reason`. Shared by the MCP and HTTP surfaces so both render
385/// byte-identical metadata under `error.details`. Returns `None` for every
386/// other error.
387fn namespace_details(e: &Error) -> Option<serde_json::Value> {
388    use crate::core::namespace::NamespaceError;
389    match e {
390        Error::Namespace(NamespaceError::NotFound { namespace }) => {
391            Some(serde_json::json!({ "namespace": namespace }))
392        }
393        Error::Namespace(NamespaceError::InvalidName { name, reason }) => {
394            Some(serde_json::json!({ "namespace": name, "reason": reason }))
395        }
396        _ => None,
397    }
398}

400fn classify_storage_error(e: &StorageError) -> (McpErrorCode, bool) {
401    match e {
402        StorageError::NodeNotFound(_)
403        | StorageError::EdgeNotFound(_)
404        | StorageError::VersionNotFound(_)
405        | StorageError::PropertyNotFound(_) => (McpErrorCode::NotFound, false),
406        StorageError::InvalidId { .. } | StorageError::InvalidProperty { .. } => {
407            (McpErrorCode::InvalidArgument, false)
408        }
409        // A duplicate ID is a conflict, but retrying the identical request
410        // cannot succeed — override the code's default retriability.
411        StorageError::DuplicateId { .. } => (McpErrorCode::Conflict, false),
412        // Resource limits (DoS protection): the request is well-formed but
413        // the system refuses in its current state.
414        StorageError::CapacityExceeded { .. } => (McpErrorCode::FailedPrecondition, false),
415        // Enabling encryption on an already-encrypted WAL is a caller precondition
416        // failure, not an internal fault (Issue #3616 PR3) — and retrying cannot
417        // succeed, so it is non-retriable.
418        StorageError::WalKeyringAlreadyInstalled { .. } => {
419            (McpErrorCode::FailedPrecondition, false)
420        }
421        // Disabling encryption on an already-plaintext WAL is a caller precondition
422        // failure, not an internal fault (Issue #3616 PR4) — the mirror of the
423        // already-installed rejection above, and likewise non-retriable.
424        StorageError::WalKeyringNotInstalled { .. } => (McpErrorCode::FailedPrecondition, false),
425        // A per-principal changefeed quota breach (Issue #3678) is a transient
426        // fairness limit: another of this principal's subscriptions may drop, so
427        // retrying with backoff can succeed → RESOURCE_EXHAUSTED, retriable.
428        StorageError::PrincipalQuotaExceeded { .. } => (McpErrorCode::ResourceExhausted, true),
429        StorageError::InconsistentState { .. }
430        | StorageError::WalError { .. }
431        | StorageError::CheckpointError { .. }
432        | StorageError::IoError(_)
433        | StorageError::CorruptedData(_)
434        | StorageError::PersistenceErrorWithKind { .. }
435        | StorageError::PersistenceError(_)
436        | StorageError::LockPoisoned { .. }
437        | StorageError::Encryption(_)
438        | StorageError::KeyProvider(_) => (McpErrorCode::Internal, false),
439    }
440}

442fn classify_temporal_error(e: &TemporalError) -> (McpErrorCode, bool) {
443    match e {
444        TemporalError::NodeNotFoundAtTime { .. } | TemporalError::VersionNotFound(_) => {
445            (McpErrorCode::NotFound, false)
446        }
447        TemporalError::InvalidTimeRange { .. }
448        | TemporalError::InvalidTimestamp { .. }
449        | TemporalError::ValidTimeBeforeCreation { .. }
450        | TemporalError::ValidTimeTooFarInFuture { .. }
451        | TemporalError::ValidTimeBeforeEntityCreation { .. } => {
452            (McpErrorCode::InvalidArgument, false)
453        }
454        TemporalError::VersionAlreadyClosed { .. } => (McpErrorCode::FailedPrecondition, false),
455        // Clock-adjacent hiccups heal on their own; a fresh attempt at a
456        // later wallclock/logical tick can succeed.
457        TemporalError::NonMonotonicTransactionTime { .. }
458        | TemporalError::LogicalCounterOverflow { .. } => (McpErrorCode::Unavailable, true),
459        // Despite its caller-fault-sounding name, `TemporalParadox` is only
460        // raised when the system clock reads before the Unix epoch — a
461        // server-environment fault, not something the caller can repair.
462        TemporalError::TemporalParadox { .. }
463        | TemporalError::CorruptedVersionChain { .. }
464        | TemporalError::MissingAnchor { .. }
465        | TemporalError::MaxDepthExceeded { .. } => (McpErrorCode::Internal, false),
466    }
467}

469/// Classify a [`ConstraintError`] per variant.
470///
471/// Only a `UniqueViolation` is a true `CONSTRAINT_VIOLATION` (a declared
472/// constraint rejecting *this* write). `UnsupportedKeyType` is a caller
473/// fault in the enable request itself, and `DuplicateOnEnable` is a state
474/// problem — duplicates already exist in the graph, so the caller must fix
475/// the data (or drop the request) before the enable can succeed.
476fn classify_constraint_error(e: &ConstraintError) -> (McpErrorCode, bool) {
477    match e {
478        ConstraintError::UniqueViolation { .. } => (McpErrorCode::ConstraintViolation, false),
479        ConstraintError::UnsupportedKeyType { .. } => (McpErrorCode::InvalidArgument, false),
480        ConstraintError::DuplicateOnEnable { .. } => (McpErrorCode::FailedPrecondition, false),
481        // Schema constraints (Issue #3378). A type/required-key violation on a
482        // write is a declared constraint rejecting *this* write
483        // (CONSTRAINT_VIOLATION); non-conformance on enable is a state problem
484        // the caller must fix first (FAILED_PRECONDITION). All non-retriable.
485        ConstraintError::TypeViolation { .. } => (McpErrorCode::ConstraintViolation, false),
486        ConstraintError::MissingRequiredKey { .. } => (McpErrorCode::ConstraintViolation, false),
487        ConstraintError::NonConformingOnEnable { .. } => (McpErrorCode::FailedPrecondition, false),
488    }
489}

491fn classify_query_error(e: &QueryError) -> (McpErrorCode, bool) {
492    match e {
493        QueryError::SyntaxError { .. }
494        | QueryError::UnsupportedFeature { .. }
495        | QueryError::InvalidParameter { .. }
496        | QueryError::LimitExceeded { .. }
497        | QueryError::InvalidTraversal { .. }
498        | QueryError::TypeMismatch { .. } => (McpErrorCode::InvalidArgument, false),
499        QueryError::Timeout { .. } => (McpErrorCode::Unavailable, true),
500        QueryError::IndexNotFound { .. } => (McpErrorCode::FailedPrecondition, false),
501        QueryError::ExecutionError { .. } => (McpErrorCode::Internal, false),
502    }
503}

505fn classify_transaction_error(e: &TransactionError) -> (McpErrorCode, bool) {
506    match e {
507        TransactionError::SerializationFailure { .. }
508        | TransactionError::WriteConflict { .. }
509        | TransactionError::Aborted { .. } => (McpErrorCode::Conflict, true),
510        TransactionError::InvalidState { .. }
511        | TransactionError::AlreadyCommitted { .. }
512        | TransactionError::ValidationFailed { .. }
513        // A lost compare-and-set (Issue #3577) is a caller-fault precondition
514        // failure: retrying the same call with the same stale `expected` version
515        // can never succeed, so it is NON-retriable (unlike SerializationFailure).
516        | TransactionError::CasMismatch { .. } => (McpErrorCode::FailedPrecondition, false),
517        TransactionError::ClockSkew { .. } => (McpErrorCode::Unavailable, true),
518        TransactionError::CommitFailed { .. }
519        | TransactionError::RollbackFailed { .. }
520        | TransactionError::LockPoisoned { .. } => (McpErrorCode::Internal, false),
521    }
522}

525    match e {
526        VectorError::NotFound { .. } => (McpErrorCode::NotFound, false),
527        VectorError::DimensionMismatch { .. }
528        | VectorError::ContainsNaN { .. }
529        | VectorError::ContainsInfinity { .. }
530        | VectorError::DimensionTooLarge { .. }

534        VectorError::IndexError(_) => (McpErrorCode::Internal, false),

538/// Map a `query` tool error `kind` (Issue #3213) to `(code, retriable)`.
539///
540/// The query tool keeps its own `kind` field verbatim (its published
541/// contract); `code`/`retriable` are added additively so the tool also
542/// satisfies the uniform #3234 contract. Every kind is a caller-fixable
543/// request problem except `language_unavailable` (a build/deployment
544/// precondition) and `runtime_error` (engine-side, classified from the
545/// underlying error where available).
546pub(crate) fn query_kind_classification(kind: &str) -> (McpErrorCode, bool) {
547    match kind {
548        "invalid_request"
549        | "read_only_violation"
550        | "parse_error"
551        | "unsupported_construct"
552        | "invalid_params" => (McpErrorCode::InvalidArgument, false),
553        "language_unavailable" => (McpErrorCode::FailedPrecondition, false),
554        "runtime_error" => (McpErrorCode::Internal, false),
