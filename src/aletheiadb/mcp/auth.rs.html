1//! MCP-surface authentication and role-based authorization (Issue #3350,
2//! Phase 2).
3//!
4//! # Session-scoped credential model
5//!
6//! The MCP server runs over **stdio**: the credential is supplied once at
7//! process start (a *session-scoped* credential), not per request as on the
8//! HTTP surface. Enforcement happens at the single dispatch point
9//! ([`AletheiaMcpServer::dispatch_tool`]) **before** any tool executes:
10//!
11//! 1. **Authentication** — in [`AuthMode::Required`] the session credential
12//!    is re-verified against the shared [`AuthStore`] on *every* tool call
13//!    (a SHA-256 digest plus a constant-time record scan — comfortably
14//!    within the sub-millisecond budget), so revoking the key in the store
15//!    takes effect on the very next call. A missing, unknown, or revoked
16//!    credential yields a uniform `UNAUTHENTICATED` error for every tool —
17//!    including unknown tool names, so unauthenticated callers cannot probe
18//!    tool *existence* via error-shape differences (the inventory itself is
19//!    available through the ungated `tools/list`; the uniformity here is
20//!    about not leaking anything through per-name error variance) — and the
21//!    error never reveals whether a presented key exists nor echoes
22//!    credential material.
23//! 2. **Authorization** — the tool name is classified into an
24//!    [`AccessClass`] via [`TOOL_ACCESS_CLASSES`] and checked against the
25//!    principal's [`Role::allows`] matrix; a denial yields
26//!    `PERMISSION_DENIED` with `details: {required_class, principal_role}`.
27//!
28//! [`AuthMode::Anonymous`] preserves the pre-#3350 behavior (full access)
29//! and is the implicit mode of [`AletheiaMcpServer::new`] for embedded/
30//! programmatic use — a Rust caller holding the `Arc<AletheiaDB>` can
31//! already do anything the tools can. The `aletheia-mcp` **binary** is
32//! where auth-on-by-default applies (see its env-var docs).
33//!
34//! # Key lifecycle
35//!
36//! Key management (create/list/revoke) is served by the HTTP admin surface
37//! (Phase 1: `POST/GET /admin/keys`, `POST /admin/keys/revoke`). Point both
38//! surfaces at the same persisted store path (`{data_dir}/auth/keys.json`)
39//! and keys minted over HTTP are usable here. No MCP-side *key-lifecycle*
40//! tools exist yet. The `Admin`-class MCP tools that do exist are the GDPR
41//! crypto-shred surface (`designate_subject` / `erase_subject`, Issue #3359).
42//!
43//! The documented classification lives in
44//! `docs/guides/access-control-matrix.md`; a conformance test mechanically
45//! compares it against [`TOOL_ACCESS_CLASSES`] — keep them in sync.
46//!
47//! [`AletheiaMcpServer::dispatch_tool`]: super::AletheiaMcpServer
48//! [`AletheiaMcpServer::new`]: super::AletheiaMcpServer::new

153/// Look up the [`AccessClass`] a tool requires. `None` for unknown tool
154/// names (the caller lets those fall through to the normal unknown-tool
155/// error — *after* authentication has passed).
156pub(crate) fn tool_access_class(name: &str) -> Option<AccessClass> {
157    TOOL_ACCESS_CLASSES
158        .iter()
159        .find(|(tool, _)| *tool == name)
160        .map(|(_, class)| *class)
161}

163/// Authentication configuration for
164/// [`AletheiaMcpServer::with_auth`](super::AletheiaMcpServer::with_auth).
165///
166/// Bundles the [`AuthMode`], the shared [`AuthStore`] (share the same
167/// `Arc`/persisted path with the HTTP surface to use keys minted there), and
168/// the optional session credential presented by this stdio session.
169///
170/// # Example
171///
172/// ```rust,no_run
173/// use std::sync::Arc;
174/// use aletheiadb::AletheiaDB;
175/// use aletheiadb::auth::{AuthMode, AuthStore, Role, SecretString};
176/// use aletheiadb::mcp::{AletheiaMcpServer, McpAuthConfig};
177///
178/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
179/// let store = Arc::new(AuthStore::new());
180/// let (_principal, key) = store.create_key("agent", Role::Reader)?;
181///
182/// let server = AletheiaMcpServer::with_auth(
183///     Arc::new(AletheiaDB::new()?),
184///     McpAuthConfig::new(AuthMode::Required, store)
185///         .with_credential(SecretString::new(key.as_str())),
186/// );
187/// # Ok(())
188/// # }
189/// ```
190#[derive(Debug, Clone)]
191pub struct McpAuthConfig {
192    mode: AuthMode,
193    store: Arc<AuthStore>,
194    credential: Option<SecretString>,
195}
196
197impl McpAuthConfig {
198    /// Create an auth configuration with no session credential yet.
199    ///
200    /// In [`AuthMode::Required`] a config without a credential produces a
201    /// server whose every tool call returns `UNAUTHENTICATED` — the server
202    /// still runs, mirroring the HTTP surface's uniform-401 contract.
203    pub fn new(mode: AuthMode, store: Arc<AuthStore>) -> Self {
204        Self {
205            mode,
206            store,
207            credential: None,
208        }
209    }
210
211    /// Attach the session credential (e.g. from `ALETHEIADB_MCP_API_KEY`).
212    ///
213    /// The credential is re-verified against the store on every tool call,
214    /// so a revocation in the (possibly shared) store takes effect on the
215    /// next call.
216    #[must_use]
217    pub fn with_credential(mut self, credential: SecretString) -> Self {
218        self.credential = Some(credential);
219        self
220    }
221
222    /// The configured [`AuthMode`].
223    pub fn mode(&self) -> AuthMode {
224        self.mode
225    }
226}

228/// The resolved per-session authentication state held by the server.
229#[derive(Clone)]
230pub(crate) enum SessionAuth {
231    /// No authentication: every tool call is fully privileged.
232    Anonymous,
233    /// Every tool call must authenticate via the session credential.
234    Required {
235        store: Arc<AuthStore>,
236        credential: Option<SecretString>,
237    },
238}
239
240impl From<McpAuthConfig> for SessionAuth {
241    fn from(config: McpAuthConfig) -> Self {
242        match config.mode {
243            AuthMode::Anonymous => SessionAuth::Anonymous,
244            AuthMode::Required => SessionAuth::Required {
245                store: config.store,
246                credential: config.credential,
247            },
248        }
249    }
250}
251
252impl SessionAuth {
253    /// Authorize one tool call: authenticate the session credential, then
254    /// check the tool's [`AccessClass`] against the principal's role.
255    ///
256    /// Ordering is deliberate: authentication happens before (and
257    /// independently of) tool-name resolution, so unauthenticated callers
258    /// cannot probe tool existence via error-shape differences (`tools/list`
259    /// itself is ungated — the property enforced here is uniform errors,
260    /// not inventory secrecy).
261    /// For authenticated callers an unknown name returns `Ok(())` here and
262    /// falls through to the dispatcher's normal unknown-tool error.
263    pub(crate) fn authorize_tool(&self, tool_name: &str) -> Result<(), McpError> {
264        let Some(principal) = self.authenticate()? else {
265            // Anonymous mode: fully privileged.
266            return Ok(());
267        };
268        match tool_access_class(tool_name) {
269            None => Ok(()),
270            Some(class) if principal.role.allows(class) => Ok(()),
271            Some(class) => Err(permission_denied_error(principal.role, class)),
272        }
273    }
274
275    /// Authenticate the session credential.
276    ///
277    /// `Ok(None)` = anonymous mode (no principal, full access);
278    /// `Ok(Some(p))` = verified principal; `Err` = `UNAUTHENTICATED`.
279    fn authenticate(&self) -> Result<Option<Principal>, McpError> {
280        match self {
281            SessionAuth::Anonymous => Ok(None),
282            SessionAuth::Required { store, credential } => credential
283                .as_ref()
284                .and_then(|c| store.verify(c.expose()))
285                .map(Some)
286                .ok_or_else(unauthenticated_error),
287        }
288    }
289
290    /// The currently verified session principal, if any (re-verifies the
291    /// credential, so a revoked key yields `None`).
292    pub(crate) fn principal(&self) -> Option<Principal> {
293        self.authenticate().ok().flatten()
294    }
295}

297/// The uniform `UNAUTHENTICATED` error.
298///
299/// Uniform by construction (a constant message, no details), so it can
300/// never reveal whether a presented key exists nor echo the credential.
301pub(crate) fn unauthenticated_error() -> McpError {
302    McpError::new(McpErrorCode::Unauthenticated, "authentication required")
303}
304
305/// A `PERMISSION_DENIED` error with `{required_class, principal_role}`
306/// details, mirroring the HTTP 403 body's message text.
307pub(crate) fn permission_denied_error(role: Role, class: AccessClass) -> McpError {
308    McpError::new(
309        McpErrorCode::PermissionDenied,
310        format!("role '{role}' does not permit {class} access"),
311    )
312    .details(json!({
313        "required_class": class.to_string(),
314        "principal_role": role.to_string(),
315    }))
316}

318/// Startup validation for the MCP server's auth configuration.
319///
320/// [`AuthMode::Required`] (the conservative default) with zero credentials
321/// available means every tool call would be rejected — refuse to start
322/// instead so the operator notices immediately. Mirrors the HTTP surface's
323/// `validate_auth_startup`.
324///
325/// # Errors
326///
327/// Returns a human-readable operator-guidance message when `mode` is
328/// `Required` and `store` holds no keys.
329pub fn validate_mcp_auth_startup(mode: AuthMode, store: &AuthStore) -> Result<(), String> {
330    match mode {
331        AuthMode::Anonymous => Ok(()),
332        AuthMode::Required => {
333            if store.is_empty() {
334                Err(
335                    "authentication is required (the default) but no credentials are \
336                     available: set a bootstrap admin key (ALETHEIADB_BOOTSTRAP_ADMIN_KEY), \
337                     point the server at an existing persisted key store \
338                     ({data_dir}/auth/keys.json under ALETHEIADB_DATA_DIR — keys minted \
339                     via the HTTP admin endpoints work here), or explicitly opt into \
340                     anonymous mode (ALETHEIADB_AUTH_MODE=anonymous)"
341                        .to_string(),
342                )
343            } else {
344                Ok(())
345            }
346        }
347    }
