1//! Stateless, snapshot-anchored keyset continuation cursors (Issue #3360).
2//!
3//! Offset pagination (#3226) over a live, concurrently-written database is
4//! quietly broken for the caller that matters most: between page 1 and page 2
5//! other agents write, offsets shift, and the reader sees duplicates or misses
6//! rows -- each page computed against a *different* database state. This module
7//! replaces that with an opaque **cursor** contract that is
8//!
9//! - **snapshot-anchored** -- every page of one scan is evaluated at the
10//!   bi-temporal coordinate captured on the first page, leveraging the existing
11//!   MVCC / temporal read semantics, so the union of all pages equals exactly
12//!   the result of an unbounded query *at that one moment*; concurrent writes
13//!   committed after the anchor are invisible to the scan (no duplicates, no
14//!   gaps, no post-cursor leakage);
15//! - **keyset continuation** -- continuation carries the last-returned id, so a
16//!   page never re-emits rows from the preceding pages (no duplicates, no gaps),
17//!   unlike offset pagination which re-materializes and discards them. Note this
18//!   is *result-page* deduplication, not a depth-independent seek: in v1 the
19//!   candidate enumeration is still O(total) per page (the node path re-runs the
20//!   full `find_nodes_at_time` candidate scan, and adjacency re-resolves the
21//!   whole edge set, on every page); a true depth-independent keyset seek that
22//!   skips prior candidates is a follow-up;
23//! - **stateless-safe for the client** -- the token is a printable,
24//!   bounded-length, self-describing string an LLM can echo back verbatim with
25//!   no escaping hazards, and continuing needs no other parameters;
26//! - **tamper-evident** -- the token is signed with a per-process secret, so a
27//!   mangled or forged token is rejected with a structured error, never served
28//!   wrong data.
29//!
30//! # Design choice: stateless token + minimal in-process registry
31//!
32//! The cursor is **stateless**: everything needed to resume -- the originating
33//! tool, the pinned bi-temporal snapshot, the keyset position, the page size,
34//! and the discriminating query parameters -- is encoded *in the token itself*,
35//! not in server-side session state. This means a resume works even if the
36//! server forgot it, and the design has no unbounded server memory growth.
37//!
38//! A small in-process **registry** is kept purely to enforce the
39//! per-connection *cap on concurrently live cursors* (AC5) and to make resource
40//! reclamation observable: it maps a cursor id to its expiry and is pruned of
41//! expired entries on every issue. It is authoritative for the *cap*; the
42//! token's embedded `iat` (issued-at) is authoritative for the *TTL*, so a
43//! token whose registry entry was already reclaimed still expires correctly by
44//! its own timestamp (stateless-safe). The registry pins no storage and only
45//! ever holds tiny `(id, expiry)` pairs bounded by the cap.
46//!
47//! ## Snapshot semantics under concurrent writes (documented, not hand-waved)
48//!
49//! The first page records `(valid_time, transaction_time)` -- for the
50//! current-state tools this is simply "now" at first-page time; for
51//! `find_nodes_at_time` it is the caller's requested coordinate. Every
52//! continuation re-evaluates the query *as of that pinned transaction time*.
53//! Consequently, for the whole lifetime of a cursor scan:
54//!
55//! - a node/edge **created** after the anchor (transaction time later than the
56//!   pin) is **not** seen by any page -- it did not exist in the snapshot;
57//! - a node/edge **deleted** after the anchor is **still** seen -- the snapshot
58//!   predates the deletion (bi-temporal append-only history);
59//! - a node/edge **updated** after the anchor is returned **as it was** at the
60//!   anchor, not its latest state.
61//!
62//! The scan therefore reflects exactly one consistent moment. This is a
63//! deliberate contrast with Qdrant/Weaviate/Milvus scroll APIs, which drift
64//! under concurrent writes because they have no temporal coordinate to anchor
65//! to.
66//!
67//! ## Lifecycle
68//!
69//! Cursors have a bounded **TTL** (default 5 minutes) and cross-restart
70//! durability is intentionally *not* provided: the signing secret is
71//! per-process, so a token minted before a restart fails signature
72//! verification afterward and the caller simply re-issues the query. Resuming
73//! after expiry, exceeding the live-cursor cap, or presenting a tampered token
74//! all return a structured [`McpError`] naming the cause with remediation
75//! guidance.

90/// Opaque token prefix and version tag. Bumping the version invalidates old
91/// tokens (they parse-fail cleanly with a structured error).
92const TOKEN_PREFIX: &str = "aletheiadb.cursor.v1.";
93
94/// Hard ceiling on an accepted token's length. A cursor payload is tiny; a
95/// token far larger than this is malformed (or hostile) and is rejected before
96/// any allocation-heavy decode.
97const MAX_TOKEN_LEN: usize = 8 * 1024;
98
99/// Default cursor time-to-live: 5 minutes. Long enough for an agent to page
100/// through a large result set, short enough that abandoned cursors reclaim
101/// their registry slot promptly.
102pub(crate) const DEFAULT_CURSOR_TTL: Duration = Duration::from_secs(300);
103
104/// Default cap on concurrently live cursors per connection. Exceeding it is a
105/// `FAILED_PRECONDITION` telling the caller to finish or abandon an existing
106/// scan (or re-issue the query) rather than a silent success.
107pub(crate) const DEFAULT_MAX_LIVE_CURSORS: usize = 128;

109/// The self-describing, signed contents of a cursor token.
110///
111/// Field names are kept terse because the struct is serialized into every
112/// token the client round-trips.
113#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
114pub(crate) struct CursorPayload {
115    /// Payload schema version (matches [`TOKEN_PREFIX`]'s tag).
116    pub v: u8,
117    /// The originating tool name (e.g. `"list_nodes"`). A token minted by one
118    /// tool is rejected if replayed against another -- the snapshot and filter
119    /// shape only make sense for the tool that produced them.
120    pub tool: String,
121    /// Pinned snapshot valid-time (microseconds since epoch).
122    pub svt: i64,
123    /// Pinned snapshot transaction-time (microseconds since epoch).
124    pub stt: i64,
125    /// Exclusive keyset lower bound: the last id returned on the prior page.
126    /// The next page returns ids strictly greater than this. `None` means
127    /// "from the beginning" -- kept distinct from `Some(0)` because ids are
128    /// 0-based, so id 0 is a real row that a `0` sentinel would wrongly skip.
129    pub after: Option<u64>,
130    /// Offset fallback for tools whose result order is not a simple id keyset
131    /// (e.g. `traverse`'s DFS order). Snapshot-anchored, so still consistent;
132    /// see the module docs' per-tool notes.
133    pub off: u64,
134    /// Page size carried across the whole scan so every page is the same size.
135    pub limit: usize,
136    /// Issued-at (microseconds since epoch); authoritative for TTL.
137    pub iat: i64,
138    /// Random per-cursor id; the registry key for cap + reclamation.
139    pub cid: String,
140    /// The discriminating request parameters (label, property filter, edge
141    /// label, direction, ...) so continuation needs no other arguments.
142    pub filters: serde_json::Value,
143}

145impl CursorPayload {
146    /// Build the seed of a first-page cursor. `iat`/`cid` are stamped by
147    /// [`CursorManager::issue`]; the keyset position starts at the beginning
148    /// (`after = None`, `off = 0`).
149    pub(crate) fn seed(
150        tool: &str,
151        snapshot: (i64, i64),
152        limit: usize,
153        filters: serde_json::Value,
154    ) -> Self {
155        Self {
156            v: 1,
157            tool: tool.to_string(),
158            svt: snapshot.0,
159            stt: snapshot.1,
160            after: None,
161            off: 0,
162            limit,
163            iat: 0,
164            cid: String::new(),
165            filters,
166        }
167    }
168}

170/// Issues, signs, verifies and lifecycle-manages [`CursorPayload`] tokens.
171///
172/// One manager is shared (via `Arc`) across a server instance == one MCP
173/// connection, so its live-cursor cap is a per-connection cap.
174#[derive(Debug)]
175pub(crate) struct CursorManager {
176    /// Per-process HMAC key. Randomized at construction, never exposed; makes
177    /// tokens unforgeable and (deliberately) non-durable across restarts.
178    secret: [u8; 32],
179    ttl: Duration,
180    max_live: usize,
181    /// Live cursor ids -> expiry (microseconds since epoch). Pruned on issue.
182    live: Mutex<HashMap<String, i64>>,
183}
184
185impl CursorManager {
186    /// Create a manager with the default TTL and cap and a fresh random secret.
187    pub(crate) fn new() -> Self {
188        Self::with_config(DEFAULT_CURSOR_TTL, DEFAULT_MAX_LIVE_CURSORS)
189    }
190
191    /// Create a manager with an explicit TTL and live-cursor cap (used by the
192    /// server's `with_cursor_config` builder and by tests exercising expiry
193    /// and the cap).
194    pub(crate) fn with_config(ttl: Duration, max_live: usize) -> Self {
195        let mut secret = [0u8; 32];
196        rand::thread_rng().fill_bytes(&mut secret);
197        Self {
198            secret,
199            ttl,
200            max_live: max_live.max(1),
201            live: Mutex::new(HashMap::new()),
202        }
203    }
204
205    /// The configured TTL (documented in tool responses / guide).
206    pub(crate) fn ttl(&self) -> Duration {
207        self.ttl
208    }
209
210    /// Number of currently-live (unexpired, registered) cursors. Used by
211    /// resource-reclamation tests.
212    #[cfg(test)]
213    pub(crate) fn live_count(&self) -> usize {
214        let mut live = self.live.lock().unwrap_or_else(|e| e.into_inner());
215        let now = time::now().wallclock();
216        live.retain(|_, &mut expiry| expiry > now);
217        live.len()
218    }
219
220    /// Issue a signed token for `payload`, stamping a fresh `iat` and
221    /// registering it for cap enforcement + reclamation.
222    ///
223    /// A **first-page** cursor (empty `payload.cid`) is minted a fresh id and
224    /// counted against the per-connection cap; issuing one when the cap is
225    /// already reached (after pruning expired entries) is a
226    /// `FAILED_PRECONDITION`. A **continuation** cursor (non-empty `cid`,
227    /// carried over from the page the caller is resuming) reuses that id and
228    /// only refreshes its expiry, so the cap counts distinct *scans*, not
229    /// pages -- a thousand-page scan holds exactly one slot. The refreshed
230    /// expiry makes the TTL an idle timeout between successive pages.
231    pub(crate) fn issue(&self, mut payload: CursorPayload) -> Result<String, McpError> {
232        let now = time::now().wallclock();
233        let expiry = now.saturating_add(self.ttl.as_micros() as i64);
234
235        {
236            let mut live = self.live.lock().unwrap_or_else(|e| e.into_inner());
237            // Reclaim expired cursors first: expired cursors never pin a slot.
238            live.retain(|_, &mut e| e > now);
239            if payload.cid.is_empty() {
240                // First page: enforce the cap and mint a fresh scan id.
241                if live.len() >= self.max_live {
242                    return Err(McpError::new(
243                        McpErrorCode::FailedPrecondition,
244                        format!(
245                            "Too many live cursors (cap {}). Finish or abandon an open scan, or \
246                             re-issue the query without use_cursor.",
247                            self.max_live
248                        ),
249                    )
250                    .details(serde_json::json!({
251                        "max_live_cursors": self.max_live,
252                        "live_cursors": live.len(),
253                    })));
254                }
255                payload.cid = Self::random_id();
256            }
257            // Register (first page) or refresh (continuation) this scan's slot.
258            live.insert(payload.cid.clone(), expiry);
259        }
260        payload.iat = now;
261
262        Ok(self.encode(&payload))
263    }
264
265    /// Decode, verify and lifecycle-check a token presented for `expected_tool`.
266    ///
267    /// Failure modes, each a structured [`McpError`] naming the cause:
268    /// - malformed / wrong-prefix / bad base64 / bad signature / wrong version
269    ///   / wrong tool -> `INVALID_ARGUMENT` (a caller/tampering fault: re-issue
270    ///   the original query);
271    /// - expired past TTL -> `FAILED_PRECONDITION` (re-issue the query).
272    ///
273    /// A verification failure never returns a (wrong) payload.
274    pub(crate) fn decode(
275        &self,
276        token: &str,
277        expected_tool: &str,
278    ) -> Result<CursorPayload, McpError> {
279        if token.len() > MAX_TOKEN_LEN {
280            return Err(Self::invalid("cursor token is too long to be valid"));
281        }
282        let body = token
283            .strip_prefix(TOKEN_PREFIX)
284            .ok_or_else(|| Self::invalid("cursor token has an unrecognized format"))?;
285        let (payload_b64, sig_b64) = body
286            .split_once('.')
287            .ok_or_else(|| Self::invalid("cursor token is malformed (missing signature)"))?;
288
289        let payload_bytes = B64
290            .decode(payload_b64)
291            .map_err(|_| Self::invalid("cursor token payload is not valid base64url"))?;
292        let sig = B64
293            .decode(sig_b64)
294            .map_err(|_| Self::invalid("cursor token signature is not valid base64url"))?;
295
296        // Verify the signature over the exact encoded payload bytes before
297        // trusting anything inside it (reject tampering up front).
298        let expected_sig = self.sign(payload_b64.as_bytes());
299        if !ct_eq(&sig, &expected_sig) {
300            return Err(Self::invalid(
301                "cursor token signature is invalid (tampered or issued by another server)",
302            ));
303        }
304
305        let payload: CursorPayload = serde_json::from_slice(&payload_bytes)
306            .map_err(|_| Self::invalid("cursor token payload is not decodable"))?;
307
308        if payload.v != 1 {
309            return Err(Self::invalid("cursor token version is unsupported"));
310        }
311        if payload.tool != expected_tool {
312            return Err(Self::invalid(&format!(
313                "cursor was issued for tool '{}', not '{}'; re-issue the query with this tool",
314                payload.tool, expected_tool
315            )));
316        }
317
318        // TTL is authoritative from the embedded issued-at, so an expired
319        // token is rejected even if its registry slot was already reclaimed.
320        let now = time::now().wallclock();
321        let expiry = payload.iat.saturating_add(self.ttl.as_micros() as i64);
322        if now >= expiry {
323            return Err(McpError::new(
324                McpErrorCode::FailedPrecondition,
325                "cursor has expired; re-issue the original query to start a new scan",
326            )
327            .details(serde_json::json!({ "ttl_seconds": self.ttl.as_secs() })));
328        }
329
330        Ok(payload)
331    }
332
333    /// Encode + sign a fully-stamped payload into an opaque token string.
334    fn encode(&self, payload: &CursorPayload) -> String {
335        let json = serde_json::to_vec(payload).expect("cursor payload serialization is infallible");
336        let payload_b64 = B64.encode(json);
337        let sig = self.sign(payload_b64.as_bytes());
338        let sig_b64 = B64.encode(sig);
339        format!("{TOKEN_PREFIX}{payload_b64}.{sig_b64}")
340    }
341
342    /// HMAC-SHA256 of `msg` under the per-process secret.
343    fn sign(&self, msg: &[u8]) -> [u8; 32] {
344        hmac_sha256(&self.secret, msg)
345    }
346
347    fn random_id() -> String {
348        let mut bytes = [0u8; 12];
349        rand::thread_rng().fill_bytes(&mut bytes);
350        B64.encode(bytes)
351    }
352
353    fn invalid(msg: &str) -> McpError {
354        McpError::new(McpErrorCode::InvalidArgument, msg)
355    }
356}

358/// Constant-time byte-slice equality (avoids leaking signature-match position
359/// via timing). Dependency-free so it never gates on an optional crate.
360fn ct_eq(a: &[u8], b: &[u8]) -> bool {
361    if a.len() != b.len() {
362        return false;
363    }
364    let mut diff = 0u8;
365    for (x, y) in a.iter().zip(b.iter()) {
366        diff |= x ^ y;
367    }
368    diff == 0
369}
370
371/// HMAC-SHA256 (RFC 2104) using the crate's already-present `sha2`. The key is
372/// 32 bytes, shorter than SHA-256's 64-byte block, so it is zero-padded rather
373/// than hashed.
374fn hmac_sha256(key: &[u8; 32], msg: &[u8]) -> [u8; 32] {
375    const BLOCK: usize = 64;
376    let mut ipad = [0x36u8; BLOCK];
377    let mut opad = [0x5cu8; BLOCK];
378    for i in 0..key.len() {
379        ipad[i] ^= key[i];
380        opad[i] ^= key[i];
381    }
382    let mut inner = Sha256::new();
383    inner.update(ipad);
384    inner.update(msg);
385    let inner_digest = inner.finalize();
386
387    let mut outer = Sha256::new();
388    outer.update(opad);
389    outer.update(inner_digest);
390    outer.finalize().into()
391}

406    #[test]
407    fn round_trips_a_valid_token() {
408        let mgr = CursorManager::new();

410        let decoded = mgr.decode(&token, "list_nodes").expect("decode");

438        let mgr = CursorManager::new();

440        // Flip a character in the payload segment.

442        let (payload_b64, sig_b64) = body.split_once('.').unwrap();
443        let mut chars: Vec<char> = payload_b64.chars().collect();
444        // Mutate a middle char to a different valid base64url char.
445        let idx = chars.len() / 2;

447        let mutated: String = chars.into_iter().collect();
448        let tampered = format!("{TOKEN_PREFIX}{mutated}.{sig_b64}");

515    #[test]
516    fn continuation_pages_reuse_the_scan_slot_and_do_not_consume_the_cap() {
517        // Cap of 1: a single scan must be able to page many times without
518        // tripping the cap, because continuation reuses the scan's cursor id.
519        let mgr = CursorManager::with_config(DEFAULT_CURSOR_TTL, 1);
520        let first = mgr.issue(seed("list_nodes")).expect("first page");
521        let mut decoded = mgr.decode(&first, "list_nodes").expect("decode first");
522        for page in 0..50u64 {
523            decoded.after = Some(page + 1);
524            let token = mgr
525                .issue(decoded.clone())
526                .expect("continuation must not trip the cap");
527            decoded = mgr
528                .decode(&token, "list_nodes")
529                .expect("decode continuation");
530        }
531        assert_eq!(mgr.live_count(), 1, "one scan holds exactly one slot");
532    }
