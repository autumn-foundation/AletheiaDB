1//! API-key store: creation, verification, revocation, optional persistence
2//! (Issue #3350).
3//!
4//! Security invariants:
5//!
6//! - Plaintext keys are returned exactly once at creation and **never
7//!   stored** — the store holds only SHA-256 digests.
8//! - Digest comparison uses [`subtle::ConstantTimeEq`]; verification scans
9//!   every record without early exit on mismatch, so no string comparison of
10//!   key material is timing-observable.
11//! - The persisted JSON file contains digests (hex), never plaintext, is
12//!   written atomically (temp file + rename) and chmodded `0600` on unix.
13//! - `Debug` output contains counts only — no digests, no prefixes.
14

27/// Scheme prefix all generated API keys start with.
28pub const KEY_SCHEME_PREFIX: &str = "aletheia_sk_";
29
30/// How many characters of the random part are kept in the masked
31/// [`Principal::key_prefix`] (out of 43 base64url characters).
32const KEY_PREFIX_RANDOM_CHARS: usize = 8;
33
34/// Random bytes per generated key (base64url-encoded into the key string).
35const KEY_RANDOM_BYTES: usize = 32;
36
37/// Maximum accepted length (in characters) of a key/principal display name.
38const MAX_KEY_NAME_CHARS: usize = 256;
39
40/// A named identity bound to a role. Never holds key material, so listing
41/// principals is masked by construction.
42///
43/// Marked `#[non_exhaustive]`: future work (e.g. multi-tenancy, Issue #3365)
44/// will add fields. Out-of-crate code obtains principals from
45/// [`AuthStore`] operations rather than constructing them.
46#[derive(Debug, Clone, Serialize, Deserialize)]
47#[non_exhaustive]
48pub struct Principal {
49    /// Short random identifier (stable across restarts for persisted keys).
50    pub id: String,
51    /// Operator-chosen display name.
52    pub name: String,
53    /// The role deciding which access classes this principal may exercise.
54    pub role: Role,
55    /// Masked key prefix (scheme + first characters) for identification.
56    pub key_prefix: String,
57    /// RFC 3339 creation timestamp.
58    pub created_at: String,
59}
60
61impl Principal {
62    /// Construct a synthetic, fully-privileged **anonymous** principal for
63    /// surfaces running in explicit anonymous mode, where no credential is
64    /// presented but a [`Principal`] value is still required (e.g. the
65    /// autumn-web 0.5 server's class-parameterized auth extractor). Carries the
66    /// `admin` role so it may exercise every access class, matching anonymous
67    /// mode's documented "full, unauthenticated access". This is the only
68    /// out-of-crate way to obtain a principal not backed by a stored key, since
69    /// [`Principal`] is `#[non_exhaustive]`.
70    ///
71    // exposed for autumn-web migration (Issue #3524)
72    #[must_use]
73    pub fn anonymous() -> Self {
74        Self {
75            id: "anonymous".to_string(),
76            name: "anonymous".to_string(),
77            role: Role::Admin,
78            key_prefix: String::new(),
79            created_at: String::new(),
80        }
81    }
82}

84/// Errors from [`AuthStore`] operations.
85#[derive(Debug, thiserror::Error)]
86pub enum AuthError {
87    /// I/O failure reading or writing the persisted store.
88    #[error("auth store I/O error: {0}")]
89    Io(#[from] std::io::Error),
90    /// The in-memory store could not be serialized for persistence.
91    #[error("auth store serialization error: {0}")]
92    Serde(#[from] serde_json::Error),
93    /// The persisted store file could not be parsed.
94    #[error("invalid auth store file: {0}")]
95    InvalidFile(String),
96    /// Caller-supplied input was invalid.
97    #[error("{0}")]
98    InvalidInput(String),
99}

101/// One stored key: the digest plus its principal.
102struct KeyRecord {
103    /// SHA-256 digest of the full plaintext key string.
104    digest: [u8; 32],
105    principal: Principal,
106    /// Whether this record is written to the persisted store. Bootstrap
107    /// (env/config-supplied) keys are memory-only.
108    persisted: bool,
109}
110
111/// On-disk representation of one key (digest only — NEVER plaintext).
112#[derive(Serialize, Deserialize)]
113struct PersistedKey {
114    id: String,
115    name: String,
116    role: Role,
117    /// Hex-encoded SHA-256 digest of the full key string.
118    key_hash: String,
119    key_prefix: String,
120    created_at: String,
121}
122
123/// On-disk representation of the whole store.
124#[derive(Serialize, Deserialize)]
125struct PersistedStore {
126    version: u32,
127    keys: Vec<PersistedKey>,
128}
129
130const PERSIST_FORMAT_VERSION: u32 = 1;
131
132/// Thread-safe store of API keys, keyed by the SHA-256 digest of the full
133/// key string. See the module docs for the security invariants.
134pub struct AuthStore {
135    /// Records keyed by principal id (revocation is by id).
136    records: DashMap<String, KeyRecord>,
137    /// When `Some`, admin mutations are persisted to this JSON file.
138    persist_path: Option<PathBuf>,
139    /// Serializes snapshot+write cycles so concurrent saves can't interleave.
140    save_lock: parking_lot::Mutex<()>,
141}
142
143impl std::fmt::Debug for AuthStore {
144    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
145        // Counts only: no key material, no hashes, no prefixes.
146        f.debug_struct("AuthStore")
147            .field("keys", &self.records.len())
148            .field("persistent", &self.persist_path.is_some())
149            .finish()
150    }
151}

159impl AuthStore {
160    /// Create an empty, memory-only store.
161    #[must_use]
162    pub fn new() -> Self {
163        Self {
164            records: DashMap::new(),
165            persist_path: None,
166            save_lock: parking_lot::Mutex::new(()),
167        }
168    }
169
170    /// Create a store persisted at `path`, loading existing keys if the file
171    /// exists.
172    ///
173    /// # Errors
174    ///
175    /// Returns [`AuthError::Io`] if the file exists but cannot be read, or
176    /// [`AuthError::InvalidFile`] if it cannot be parsed.
177    pub fn with_persistence(path: impl AsRef<Path>) -> Result<Self, AuthError> {
178        let path = path.as_ref().to_path_buf();
179        let store = Self {
180            records: DashMap::new(),
181            persist_path: Some(path.clone()),
182            save_lock: parking_lot::Mutex::new(()),
183        };
184        if path.exists() {
185            let contents = std::fs::read_to_string(&path)?;
186            let parsed: PersistedStore = serde_json::from_str(&contents)
187                .map_err(|e| AuthError::InvalidFile(e.to_string()))?;
188            if parsed.version != PERSIST_FORMAT_VERSION {
189                return Err(AuthError::InvalidFile(format!(
190                    "unsupported auth store version {} (expected {})",
191                    parsed.version, PERSIST_FORMAT_VERSION
192                )));
193            }
194            for key in parsed.keys {
195                let digest = hex_decode_32(&key.key_hash).ok_or_else(|| {
196                    AuthError::InvalidFile(format!("invalid key hash for id '{}'", key.id))
197                })?;
198                let principal = Principal {
199                    id: key.id.clone(),
200                    name: key.name,
201                    role: key.role,
202                    key_prefix: key.key_prefix,
203                    created_at: key.created_at,
204                };
205                store.records.insert(
206                    key.id,
207                    KeyRecord {
208                        digest,
209                        principal,
210                        persisted: true,
211                    },
212                );
213            }
214        }
215        Ok(store)
216    }

218    /// Create a new key with the given display name and role.
219    ///
220    /// Returns the principal plus the plaintext key — **exactly once**. The
221    /// plaintext is never stored; only its SHA-256 digest is retained.
222    ///
223    /// # Errors
224    ///
225    /// Returns [`AuthError::InvalidInput`] for an empty name or a name
226    /// longer than 256 characters, or a persistence error if the store is
227    /// file-backed and saving fails (in which case the key is rolled back
228    /// and does not take effect).
229    pub fn create_key(
230        &self,
231        name: &str,
232        role: Role,
233    ) -> Result<(Principal, Zeroizing<String>), AuthError> {
234        let name = name.trim();
235        if name.is_empty() {
236            return Err(AuthError::InvalidInput(
237                "key name must not be empty".to_string(),
238            ));
239        }
240        if name.chars().count() > MAX_KEY_NAME_CHARS {
241            return Err(AuthError::InvalidInput(format!(
242                "key name must be at most {MAX_KEY_NAME_CHARS} characters"
243            )));
244        }
245
246        let mut random_bytes = [0u8; KEY_RANDOM_BYTES];
247        OsRng.fill_bytes(&mut random_bytes);
248        let mut random_part = URL_SAFE_NO_PAD.encode(random_bytes);
249        random_bytes.zeroize();
250
251        let key = Zeroizing::new(format!("{KEY_SCHEME_PREFIX}{random_part}"));
252        // The random part is pure ASCII base64url, so byte slicing is safe.
253        let key_prefix = format!(
254            "{KEY_SCHEME_PREFIX}{}",
255            &random_part[..KEY_PREFIX_RANDOM_CHARS.min(random_part.len())]
256        );
257        random_part.zeroize();
258
259        let principal = Principal {
260            id: self.generate_unique_id(),
261            name: name.to_string(),
262            role,
263            key_prefix,
264            created_at: now_rfc3339(),
265        };
266        let record = KeyRecord {
267            digest: digest_key(&key),
268            principal: principal.clone(),
269            persisted: true,
270        };
271        self.records.insert(principal.id.clone(), record);
272
273        if let Err(e) = self.save() {
274            // Keep memory and disk consistent: a key we couldn't persist
275            // must not silently exist only until the next restart.
276            self.records.remove(&principal.id);
277            return Err(e);
278        }
279        Ok((principal, key))
280    }

282    /// List all principals, sorted by id. Masked by construction —
283    /// [`Principal`] never holds key material.
284    #[must_use]
285    pub fn list(&self) -> Vec<Principal> {
286        let mut principals: Vec<Principal> = self
287            .records
288            .iter()
289            .map(|r| r.value().principal.clone())
290            .collect();
291        principals.sort_by(|a, b| a.id.cmp(&b.id));
292        principals
293    }

295    /// Revoke a key by principal id. Takes effect for subsequent
296    /// verifications immediately. Returns whether the id existed.
297    ///
298    /// # Errors
299    ///
300    /// Returns a persistence error if the store is file-backed and saving
301    /// fails. The revocation still holds in memory (fail-secure): the key is
302    /// gone for this process even if the file could not be updated.
303    pub fn revoke(&self, id: &str) -> Result<bool, AuthError> {
304        match self.records.remove(id) {
305            None => Ok(false),
306            Some((_, record)) => {
307                if record.persisted {
308                    self.save()?;
309                }
310                Ok(true)
311            }
312        }
313    }

315    /// Verify a presented key, returning its principal if (and only if) the
316    /// SHA-256 digest of the presented string matches a stored digest.
317    ///
318    /// Constant-time discipline: every record is compared via
319    /// [`subtle::ConstantTimeEq`] and the scan never exits early on a
320    /// mismatch.
321    #[must_use]
322    pub fn verify(&self, presented: &str) -> Option<Principal> {
323        let digest = digest_key(presented);
324        let mut found: Option<Principal> = None;
325        for entry in self.records.iter() {
326            if bool::from(entry.value().digest.ct_eq(&digest)) {
327                found = Some(entry.value().principal.clone());
328            }
329        }
330        found
331    }

333    /// Install a bootstrap key supplied via env/config at startup.
334    ///
335    /// Bootstrap keys are **memory-only**: they are never written to the
336    /// persisted store (the operator re-supplies them on each start).
337    /// Idempotent for the same plaintext — re-inserting returns the existing
338    /// principal instead of duplicating.
339    ///
340    /// # Errors
341    ///
342    /// Returns [`AuthError::InvalidInput`] for an empty name or plaintext.
343    pub fn insert_bootstrap_key(
344        &self,
345        name: &str,
346        role: Role,
347        plaintext: &str,
348    ) -> Result<Principal, AuthError> {
349        let name = name.trim();
350        if name.is_empty() {
351            return Err(AuthError::InvalidInput(
352                "bootstrap key name must not be empty".to_string(),
353            ));
354        }
355        if plaintext.is_empty() {
356            return Err(AuthError::InvalidInput(
357                "bootstrap key must not be empty".to_string(),
358            ));
359        }
360
361        let digest = digest_key(plaintext);
362        // Idempotency: same plaintext already installed → return it.
363        for entry in self.records.iter() {
364            if bool::from(entry.value().digest.ct_eq(&digest)) {
365                let existing = entry.value().principal.clone();
366                if existing.role != role {
367                    // Fail-closed: the requested role is IGNORED, the
368                    // existing principal (and its role) is kept — a
369                    // re-supplied bootstrap key can never escalate a key
370                    // that already exists. Warn loudly so the operator
371                    // notices the collision. NEVER log key material here.
372                    eprintln!(
373                        "WARNING: bootstrap key '{name}' matches an existing key belonging to \
374                         principal '{}' with role '{}'; the requested role '{role}' is ignored \
375                         and the existing principal is returned (no privilege change)",
376                        existing.name, existing.role
377                    );
378                    #[cfg(feature = "http-server")]
379                    tracing::warn!(
380                        existing_principal = %existing.name,
381                        existing_role = %existing.role,
382                        requested_role = %role,
383                        "bootstrap key matches an existing key with a different role; \
384                         keeping the existing principal"
385                    );
386                }
387                return Ok(existing);
388            }
389        }
390
391        // Reveal at most 4 characters of an operator-chosen secret in the
392        // masked prefix (generated keys expose 8 of 43 random chars, but a
393        // bootstrap key may be low-entropy).
394        let prefix: String = plaintext.chars().take(4).collect();
395        let principal = Principal {
396            id: self.generate_unique_id(),
397            name: name.to_string(),
398            role,
399            key_prefix: format!("{prefix}\u{2026}"),
400            created_at: now_rfc3339(),
401        };
402        self.records.insert(
403            principal.id.clone(),
404            KeyRecord {
405                digest,
406                principal: principal.clone(),
407                persisted: false,
408            },
409        );
410        Ok(principal)
411    }

425    /// Generate a fresh random principal id not currently in use.
426    fn generate_unique_id(&self) -> String {
427        loop {
428            let mut bytes = [0u8; 8];
429            OsRng.fill_bytes(&mut bytes);
430            let id = hex_encode(&bytes);
431            if !self.records.contains_key(&id) {
432                return id;
433            }
434        }
435    }

437    /// Persist the current set of `persisted` records, atomically
438    /// (temp file + rename), with `0600` permissions on unix.
439    ///
440    /// A no-op for memory-only stores.
441    fn save(&self) -> Result<(), AuthError> {
442        let Some(path) = &self.persist_path else {
443            return Ok(());
444        };
445        let _guard = self.save_lock.lock();
446
447        let mut keys: Vec<PersistedKey> = self
448            .records
449            .iter()
450            .filter(|r| r.value().persisted)
451            .map(|r| {
452                let p = &r.value().principal;
453                PersistedKey {
454                    id: p.id.clone(),
455                    name: p.name.clone(),
456                    role: p.role,
457                    key_hash: hex_encode(&r.value().digest),
458                    key_prefix: p.key_prefix.clone(),
459                    created_at: p.created_at.clone(),
460                }
461            })
462            .collect();
463        keys.sort_by(|a, b| a.id.cmp(&b.id));
464
465        let serialized = serde_json::to_vec_pretty(&PersistedStore {
466            version: PERSIST_FORMAT_VERSION,
467            keys,
468        })?;
469
470        if let Some(parent) = path.parent()
471            && !parent.as_os_str().is_empty()
472        {
473            std::fs::create_dir_all(parent)?;
474        }
475
476        // Atomic write: write to a sibling temp file, fsync, then rename.
477        let tmp_path = path.with_extension("tmp");
478        // Remove any stale temp file first: `mode(0o600)` only applies when
479        // the file is *created*, so a pre-existing temp file (e.g. from a
480        // crashed save) could otherwise keep looser permissions.
481        let _ = std::fs::remove_file(&tmp_path);
482        {
483            use std::io::Write as _;
484            let mut options = std::fs::OpenOptions::new();
485            options.write(true).create(true).truncate(true);
486            #[cfg(unix)]
487            {
488                use std::os::unix::fs::OpenOptionsExt as _;
489                options.mode(0o600);
490            }
491            let mut file = options.open(&tmp_path)?;
492            file.write_all(&serialized)?;
493            file.sync_all()?;
494        }
495        std::fs::rename(&tmp_path, path)?;
496        // Make the rename itself durable: fsync the parent directory so the
497        // new directory entry survives a crash (unix only; directories are
498        // not opened for sync on other platforms).
499        #[cfg(unix)]
500        if let Some(parent) = path.parent()
501            && !parent.as_os_str().is_empty()
502        {
503            std::fs::File::open(parent)?.sync_all()?;
504        }
505        Ok(())
506    }
507}

509/// SHA-256 digest of the full key string.
510fn digest_key(key: &str) -> [u8; 32] {
511    Sha256::digest(key.as_bytes()).into()
512}
513
514/// Current wallclock time as RFC 3339 (UTC).
515fn now_rfc3339() -> String {
516    chrono::Utc::now().to_rfc3339_opts(chrono::SecondsFormat::Micros, true)
517}
518
519/// Lowercase hex encoding.
520fn hex_encode(bytes: &[u8]) -> String {
521    use std::fmt::Write as _;
522    bytes
523        .iter()
524        .fold(String::with_capacity(bytes.len() * 2), |mut out, byte| {
525            // Writing to a String cannot fail.
526            let _ = write!(out, "{byte:02x}");
527            out
528        })
529}

531/// Decode a 64-char hex string into 32 bytes. `None` on any malformation.
532fn hex_decode_32(s: &str) -> Option<[u8; 32]> {
533    if s.len() != 64 || !s.is_ascii() {
534        return None;
535    }
536    let mut out = [0u8; 32];
537    for (slot, chunk) in out.iter_mut().zip(s.as_bytes().chunks(2)) {
538        let pair = std::str::from_utf8(chunk).ok()?;
539        *slot = u8::from_str_radix(pair, 16).ok()?;
540    }
541    Some(out)
542}
