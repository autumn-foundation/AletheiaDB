1//! Authentication and role-based access control core (Issue #3350, Phase 1).
2//!
3//! This module is transport-agnostic: it knows nothing about HTTP or MCP.
4//! It provides:
5//!
6//! - [`Role`] — the four operator-facing roles (`admin`, `writer`, `reader`,
7//!   `metrics`) and the [`Role::allows`] permission matrix over
8//!   [`AccessClass`]es.
9//! - [`Principal`] — a named identity bound to a role. Never holds key
10//!   material; listing principals is masked by construction.
11//! - [`AuthStore`] — a thread-safe store of API keys, keyed by the SHA-256
12//!   digest of the full key string. Plaintext keys are returned exactly once
13//!   at creation and never stored. Optional JSON file persistence writes
14//!   only digests (atomic write, `0600` on unix).
15//! - [`AuthMode`] — whether a serving surface requires credentials
16//!   ([`AuthMode::Required`], the default) or has been explicitly opted into
17//!   anonymous operation ([`AuthMode::Anonymous`]).
18//!
19//! Key format: `aletheia_sk_` followed by the base64url (no padding)
20//! encoding of 32 bytes from the OS CSPRNG.
21//!
22//! Enforcement (extractors, per-operation authorization) lives in the
23//! transport layers: `crate::http` today, `crate::mcp` in Phase 2.
24

33/// Whether a serving surface requires authentication.
34///
35/// The conservative default is [`AuthMode::Required`]: a server refuses to
36/// start without at least one credential unless the operator explicitly opts
37/// into [`AuthMode::Anonymous`].
38#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
39pub enum AuthMode {
40    /// Every request must present a valid API key (default).
41    #[default]
42    Required,
43    /// No authentication: every request is treated as fully privileged.
44    /// Explicit opt-in only; the server logs a prominent warning.
45    Anonymous,
46}
47
48impl FromStr for AuthMode {
49    type Err = String;
50
51    fn from_str(s: &str) -> Result<Self, Self::Err> {
52        match s.trim().to_ascii_lowercase().as_str() {
53            "required" => Ok(Self::Required),
54            "anonymous" => Ok(Self::Anonymous),
55            other => Err(format!(
56                "invalid auth mode '{other}': expected 'required' or 'anonymous'"
57            )),
58        }
59    }
60}
61
62impl std::fmt::Display for AuthMode {
63    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
64        match self {
65            Self::Required => f.write_str("required"),
66            Self::Anonymous => f.write_str("anonymous"),
67        }
68    }
69}

71/// Resolve an optionally-supplied auth-mode string (typically the value of
72/// `ALETHEIADB_AUTH_MODE`) into an [`AuthMode`], fail-closed.
73///
74/// - `None` (variable unset) → [`AuthMode::Required`] (the default).
75/// - A valid value (`"required"` / `"anonymous"`, case-insensitive,
76///   whitespace-tolerant) → that mode.
77/// - An invalid value → a warning on stderr and [`AuthMode::Required`] —
78///   **never** anonymous, so a typo cannot silently disable authentication.
79#[must_use]
80pub fn parse_auth_mode_value(raw: Option<&str>) -> AuthMode {
81    match raw {
82        None => AuthMode::Required,
83        Some(raw) => match raw.parse::<AuthMode>() {
84            Ok(mode) => mode,
85            Err(e) => {
86                eprintln!("WARNING: {e}. Falling back to auth mode 'required'.");
87                AuthMode::Required
88            }
89        },
90    }
91}
92
93/// Read `ALETHEIADB_AUTH_MODE` from the environment via
94/// [`parse_auth_mode_value`] (fail-closed: unset or invalid → `Required`).
95///
96/// Shared by the HTTP and MCP server binaries so both surfaces resolve the
97/// variable with identical semantics.
98#[must_use]
99pub fn auth_mode_from_env() -> AuthMode {
100    parse_auth_mode_value(std::env::var("ALETHEIADB_AUTH_MODE").ok().as_deref())
101}

103/// A secret string wrapper that never exposes its contents through `Debug`
104/// and zeroizes its buffer on drop.
105///
106/// Used for plaintext credentials that must transit configuration (e.g. a
107/// bootstrap admin key supplied via environment variable).
108#[derive(Clone)]
109pub struct SecretString(zeroize::Zeroizing<String>);
110
111impl SecretString {
112    /// Wrap a plaintext secret.
113    pub fn new(value: impl Into<String>) -> Self {
114        Self(zeroize::Zeroizing::new(value.into()))
115    }
116
117    /// Expose the plaintext. Callers must not log or persist the result.
118    #[must_use]
119    pub fn expose(&self) -> &str {
120        &self.0
121    }
122}
123
124impl std::fmt::Debug for SecretString {
125    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
126        f.write_str("SecretString(<redacted>)")
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    #[test]
135    fn auth_mode_default_is_required() {
136        assert_eq!(AuthMode::default(), AuthMode::Required);
137    }
138
139    #[test]
140    fn auth_mode_from_str_roundtrip() {
141        assert_eq!("required".parse::<AuthMode>(), Ok(AuthMode::Required));
142        assert_eq!("anonymous".parse::<AuthMode>(), Ok(AuthMode::Anonymous));
143        assert_eq!("REQUIRED".parse::<AuthMode>(), Ok(AuthMode::Required));
144        assert_eq!(" Anonymous ".parse::<AuthMode>(), Ok(AuthMode::Anonymous));
145        assert!("open".parse::<AuthMode>().is_err());
146        assert!("".parse::<AuthMode>().is_err());
147    }
148
149    #[test]
150    fn auth_mode_display() {
151        assert_eq!(AuthMode::Required.to_string(), "required");
152        assert_eq!(AuthMode::Anonymous.to_string(), "anonymous");
153    }
154
155    #[test]
156    fn parse_auth_mode_value_is_fail_closed() {
157        // Unset → Required.
158        assert_eq!(parse_auth_mode_value(None), AuthMode::Required);
159        // Valid values pass through (case-insensitive, trimmed).
160        assert_eq!(parse_auth_mode_value(Some("required")), AuthMode::Required);
161        assert_eq!(
162            parse_auth_mode_value(Some("anonymous")),
163            AuthMode::Anonymous
164        );
165        assert_eq!(
166            parse_auth_mode_value(Some(" ANONYMOUS ")),
167            AuthMode::Anonymous
168        );
169        // Invalid values MUST fall back to Required, never Anonymous.
170        assert_eq!(parse_auth_mode_value(Some("open")), AuthMode::Required);
171        assert_eq!(
172            parse_auth_mode_value(Some("anonymos")),
173            AuthMode::Required,
174            "a typo near 'anonymous' must still fail closed"
175        );
176        assert_eq!(parse_auth_mode_value(Some("")), AuthMode::Required);
177    }
178
179    #[test]
180    fn secret_string_debug_is_redacted() {
181        let secret = SecretString::new("aletheia_sk_super_secret_value");
182        let debug = format!("{secret:?}");
183        assert!(!debug.contains("super_secret_value"));
184        assert!(debug.contains("redacted"));
185        assert_eq!(secret.expose(), "aletheia_sk_super_secret_value");
186    }
