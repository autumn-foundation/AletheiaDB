38/// Converter from SQL AST to AletheiaDB Query.
39///
40/// # Example
41///
42/// ```rust
43/// use aletheiadb::sql::SqlConverter;
44///
45/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
46/// let converter = SqlConverter::new();
47/// let query = converter.convert_sql("SELECT * FROM nodes WHERE label = 'Person'")?;
48/// # Ok(())
49/// # }
50/// ```
51pub struct SqlConverter {
52    /// Parameter bindings
53    parameters: HashMap<String, SqlParameterValue>,
54}

75    /// Convert a SQL string to a Query.
76    pub fn convert_sql(&self, sql: &str) -> Result<Query, SqlError> {
77        // Pipeline: SQL → temporal → match → vector → sqlparser → convert → wire up ops
78        let extracted_temporal = temporal_parser::extract_temporal_clauses(sql)?;
79        let extracted_match = match_parser::extract_match_clauses(&extracted_temporal.cleaned_sql)?;
80        let extracted_vector = vector_parser::extract_vector_clauses(&extracted_match.cleaned_sql)?;
81
82        // Parse the cleaned SQL (without temporal, MATCH, or vector clauses)
83        let stmt = SqlParser::parse(&extracted_vector.cleaned_sql)?;
84
85        // Convert to Query and add temporal context
86        let mut query = self.convert(&stmt)?;
87        query.temporal_context = extracted_temporal.to_temporal_context()?;
88
89        // Insert graph traversal ops after source ops (ScanNodes/ScanEdges)
90        // but before filter/projection/sort/limit ops.
91        for pattern in extracted_match.patterns.iter().rev() {
92            let insert_pos = query
93                .ops
94                .iter()
95                .position(|op| !matches!(op, QueryOp::ScanNodes { .. } | QueryOp::ScanEdges { .. }))
96                .unwrap_or(query.ops.len());
97            query.ops.insert(insert_pos, pattern.to_query_op());
98        }
99
100        // Resolve and insert vector ops
101        for vector_op in &extracted_vector.vector_ops {
102            self.apply_vector_op(&mut query, vector_op)?;
103        }
104
105        Ok(query)
106    }

108    /// Convert a SQL statement to a Query.
109    pub fn convert(&self, stmt: &Statement) -> Result<Query, SqlError> {
110        match stmt {
111            Statement::Query(query) => self.convert_query(query),
112            _ => Err(SqlError::UnsupportedFeature(format!(
113                "Only SELECT queries are supported, got: {:?}",
114                stmt
115            ))),
116        }
117    }

119    /// Convert a SQL SELECT query to a Query.
120    fn convert_query(&self, query: &SqlQuery) -> Result<Query, SqlError> {
121        // Handle the body of the query
122        let select = match query.body.as_ref() {
123            SetExpr::Select(select) => select,
124            _ => {
125                return Err(SqlError::UnsupportedFeature(
126                    "Only simple SELECT queries are supported".to_string(),
127                ));
128            }
129        };
130
131        let mut ops = Vec::new();
132
133        // Convert FROM clause
134        self.convert_from(&select.from, &mut ops)?;
135
136        // Convert WHERE clause. Edge-property `WHERE` over a `FROM edges` scan is
137        // evaluated for real (Issue #3622): the executor detects an
138        // `EdgeScan`-rooted stream and runs the shared `FilterIterator` in
139        // edge-property mode, matching each property leaf against the edge's own
140        // properties. (SQL only ever sources edges via a bare `FROM edges`, a
141        // pure edge stream, so every bare property leaf unambiguously refers to
142        // the edge.)
143        if let Some(ref selection) = select.selection {
144            let predicate = self.convert_expr_to_predicate(selection)?;
145            ops.push(QueryOp::Filter(predicate));
146        }
147
148        // Convert SELECT projection
149        self.convert_projection(&select.projection, &mut ops)?;
150
151        // Convert ORDER BY. A property-key `ORDER BY` over `FROM edges` is
152        // likewise evaluated for real (Issue #3622): the `SortIterator` reads the
153        // edge's own properties when its input is `EdgeScan`-rooted.
154        for order_by in &query.order_by {
155            self.convert_order_by(order_by, &mut ops)?;
156        }
157
158        // Convert OFFSET (must come before LIMIT for correct SQL semantics:
159        // skip N rows first, then take M rows from the result)
160        if let Some(ref offset) = query.offset {
161            let n = self.expr_to_usize(&offset.value)?;
162            ops.push(QueryOp::Skip(n));
163        }
164
165        // Convert LIMIT
166        if let Some(ref limit) = query.limit {
167            let n = self.expr_to_usize(limit)?;
168            ops.push(QueryOp::Limit(n));
169        }
170
171        Ok(Query {
172            ops,
173            // Temporal context is set by convert_sql() after extraction
174            temporal_context: None,
175            hints: QueryHints::default(),
176            // SQL namespace scoping is a follow-up (PR2b); no scope today.
177            scope: None,
178        })
179    }

181    /// Convert FROM clause to source operations.
182    fn convert_from(
183        &self,
184        from: &[TableWithJoins],
185        ops: &mut Vec<QueryOp>,
186    ) -> Result<(), SqlError> {
187        if from.is_empty() {
188            return Err(SqlError::MissingClause(
189                "FROM clause is required".to_string(),
190            ));
191        }
192
193        if from.len() > 1 {
194            return Err(SqlError::UnsupportedFeature(
195                "Multiple tables (joins) not yet supported".to_string(),
196            ));
197        }
198
199        let table = &from[0];
200        if !table.joins.is_empty() {
201            return Err(SqlError::UnsupportedFeature(
202                "JOIN clauses are not yet supported. Use MATCH for graph traversal: \
203                 MATCH (source)-[:EDGE_TYPE]->(target)"
204                    .to_string(),
205            ));
206        }
207
208        match &table.relation {
209            TableFactor::Table { name, .. } => {
210                let table_name = name.to_string().to_lowercase();
211                match table_name.as_str() {
212                    "nodes" => {
213                        ops.push(QueryOp::ScanNodes { label: None });
214                    }
215                    "edges" => {
216                        ops.push(QueryOp::ScanEdges { edge_type: None });
217                    }
218                    _ => {
219                        // Treat other table names as label filters
220                        // e.g., "SELECT * FROM Person" scans nodes with label 'Person'
221                        ops.push(QueryOp::ScanNodes {
222                            label: Some(table_name),
223                        });
224                    }
225                }
226            }
227            _ => {
228                return Err(SqlError::UnsupportedFeature(
229                    "Complex table expressions not supported".to_string(),
230                ));
231            }
232        }
233
234        Ok(())
235    }

237    /// Convert SELECT projection.
238    fn convert_projection(
239        &self,
240        projection: &[SelectItem],
241        ops: &mut Vec<QueryOp>,
242    ) -> Result<(), SqlError> {
243        let mut columns = Vec::new();
244        let mut is_star = false;
245
246        for item in projection {
247            match item {
248                SelectItem::Wildcard(_) => {
249                    is_star = true;
250                }
251                SelectItem::UnnamedExpr(expr) => {
252                    if let Some(col) = self.expr_to_column_name(expr) {
253                        columns.push(col);
254                    }
255                }
256                SelectItem::ExprWithAlias { expr, alias } => {
257                    if self.expr_to_column_name(expr).is_some() {
258                        columns.push(alias.value.clone());
259                    }
260                }
261                SelectItem::QualifiedWildcard(_, _) => {
262                    is_star = true;
263                }
264            }
265        }
266
267        // If not SELECT *, add projection
268        if !is_star && !columns.is_empty() {
269            ops.push(QueryOp::Project(columns));
270        }
271
272        Ok(())
273    }

275    /// Convert ORDER BY clause.
276    ///
277    /// A property-key `ORDER BY` over a `FROM edges` scan is honored: the
278    /// executor runs the `SortIterator` in edge-property mode for an
279    /// `EdgeScan`-rooted stream, so edge rows are sorted by their own properties
280    /// (Issue #3622). `ORDER BY score`/`timestamp` are unaffected.
281    fn convert_order_by(
282        &self,
283        order_by: &OrderByExpr,
284        ops: &mut Vec<QueryOp>,
285    ) -> Result<(), SqlError> {
286        let key = match &order_by.expr {
287            Expr::Identifier(ident) => {
288                let name = ident.value.to_lowercase();
289                match name.as_str() {
290                    "score" => SortKey::Score,
291                    "timestamp" => SortKey::Timestamp,
292                    _ => SortKey::Property(ident.value.clone()),
293                }
294            }
295            Expr::CompoundIdentifier(parts) => {
296                // Handle table.column syntax
297                let col = parts.last().map(|p| p.value.clone()).ok_or_else(|| {
298                    SqlError::InvalidColumn("Empty compound identifier in ORDER BY".to_string())
299                })?;
300                SortKey::Property(col)
301            }
302            _ => {
303                return Err(SqlError::UnsupportedFeature(
304                    "Complex ORDER BY expressions not yet supported. Use simple column names (e.g., ORDER BY name DESC)".to_string(),
305                ));
306            }
307        };
308
309        let descending = order_by.asc.map(|asc| !asc).unwrap_or(false);
310
311        ops.push(QueryOp::Sort { key, descending });
312
313        Ok(())
314    }

316    /// Convert a SQL expression to a predicate.
317    fn convert_expr_to_predicate(&self, expr: &Expr) -> Result<Predicate, SqlError> {
318        match expr {
319            Expr::BinaryOp { left, op, right } => self.convert_binary_op(left, op, right),
320            Expr::Nested(inner) => self.convert_expr_to_predicate(inner),
321            Expr::IsNull(inner) => {
322                let key = self.expr_to_property_key(inner)?;
323                Ok(Predicate::Eq {
324                    key,
325                    value: PredicateValue::Null,
326                })
327            }
328            Expr::IsNotNull(inner) => {
329                let key = self.expr_to_property_key(inner)?;
330                Ok(Predicate::Ne {
331                    key,
332                    value: PredicateValue::Null,
333                })
334            }
335            Expr::InList {
336                expr,
337                list,
338                negated,
339            } => {
340                let key = self.expr_to_property_key(expr)?;
341                let values: Result<Vec<PredicateValue>, SqlError> =
342                    list.iter().map(|e| self.expr_to_value(e)).collect();
343                let pred = Predicate::In {
344                    key,
345                    values: values?,
346                };
347                if *negated { Ok(!pred) } else { Ok(pred) }
348            }
349            Expr::Like {
350                expr,
351                pattern,
352                negated,
353                ..
354            } => {
355                let key = self.expr_to_property_key(expr)?;
356                let pattern_str = self.expr_to_string(pattern)?;
357
358                // Convert LIKE pattern to appropriate predicate using slicing
359                // to correctly handle patterns like 'a%b'
360                let pred = if pattern_str.starts_with('%')
361                    && pattern_str.ends_with('%')
362                    && pattern_str.len() > 1
363                {
364                    // %substring% -> Contains
365                    let substring = pattern_str[1..pattern_str.len() - 1].to_string();
366                    Predicate::Contains { key, substring }
367                } else if pattern_str.ends_with('%') && !pattern_str.starts_with('%') {
368                    // prefix% -> StartsWith
369                    let prefix = pattern_str[..pattern_str.len() - 1].to_string();
370                    Predicate::StartsWith { key, prefix }
371                } else if pattern_str.starts_with('%') && !pattern_str.ends_with('%') {
372                    // %suffix -> EndsWith
373                    let suffix = pattern_str[1..].to_string();
374                    Predicate::EndsWith { key, suffix }
375                } else {
376                    // Exact match (no wildcards or complex pattern)
377                    Predicate::Eq {
378                        key,
379                        value: PredicateValue::String(pattern_str),
380                    }
381                };
382
383                if *negated { Ok(!pred) } else { Ok(pred) }
384            }
385            _ => Err(SqlError::UnsupportedFeature(format!(
386                "Expression type not supported in WHERE: {:?}",
387                expr
388            ))),
389        }
390    }

392    /// Convert a binary operation to a predicate.
393    fn convert_binary_op(
394        &self,
395        left: &Expr,
396        op: &BinaryOperator,
397        right: &Expr,
398    ) -> Result<Predicate, SqlError> {
399        // Handle logical operators
400        match op {
401            BinaryOperator::And => {
402                let l = self.convert_expr_to_predicate(left)?;
403                let r = self.convert_expr_to_predicate(right)?;
404                return Ok(l.and(r));
405            }
406            BinaryOperator::Or => {
407                let l = self.convert_expr_to_predicate(left)?;
408                let r = self.convert_expr_to_predicate(right)?;
409                return Ok(l.or(r));
410            }
411            _ => {}
412        }
413
414        // Handle comparison operators
415        let key = self.expr_to_property_key(left)?;
416        let value = self.expr_to_value(right)?;
417
418        match op {
419            BinaryOperator::Eq => Ok(Predicate::Eq { key, value }),
420            BinaryOperator::NotEq => Ok(Predicate::Ne { key, value }),
421            BinaryOperator::Lt => Ok(Predicate::Lt { key, value }),
422            BinaryOperator::LtEq => Ok(Predicate::Lte { key, value }),
423            BinaryOperator::Gt => Ok(Predicate::Gt { key, value }),
424            BinaryOperator::GtEq => Ok(Predicate::Gte { key, value }),
425            _ => Err(SqlError::UnsupportedFeature(format!(
426                "Operator not supported: {:?}",
427                op
428            ))),
429        }
430    }

432    /// Extract property key from expression.
433    fn expr_to_property_key(&self, expr: &Expr) -> Result<String, SqlError> {
434        match expr {
435            Expr::Identifier(ident) => Ok(ident.value.clone()),
436            Expr::CompoundIdentifier(parts) => {
437                // Handle table.column syntax - return just the column name
438                parts
439                    .last()
440                    .map(|p| p.value.clone())
441                    .ok_or_else(|| SqlError::InvalidColumn("Empty compound identifier".to_string()))
442            }
443            _ => Err(SqlError::InvalidColumn(format!(
444                "Cannot extract property key from: {:?}",
445                expr
446            ))),
447        }
448    }

450    /// Convert expression to predicate value.
451    fn expr_to_value(&self, expr: &Expr) -> Result<PredicateValue, SqlError> {
452        match expr {
453            Expr::Value(value) => self.value_to_predicate_value(value),
454            Expr::Identifier(ident) => {
455                // Check if it's a parameter reference
456                if let Some(param) = self.parameters.get(&ident.value) {
457                    match param {
458                        SqlParameterValue::Scalar(v) => Ok(v.clone()),
459                        _ => Err(SqlError::TypeError(
460                            "Expected scalar parameter value".to_string(),
461                        )),
462                    }
463                } else {
464                    Err(SqlError::ParameterError(format!(
465                        "Unknown parameter: {}",
466                        ident.value
467                    )))
468                }
469            }
470            Expr::UnaryOp { op, expr } => {
471                // Handle negative numbers
472                match op {
473                    sqlparser::ast::UnaryOperator::Minus => {
474                        let inner = self.expr_to_value(expr)?;
475                        match inner {
476                            PredicateValue::Int(n) => Ok(PredicateValue::Int(-n)),
477                            PredicateValue::Float(n) => Ok(PredicateValue::Float(-n)),
478                            _ => Err(SqlError::TypeError(
479                                "Cannot negate non-numeric value".to_string(),
480                            )),
481                        }
482                    }
483                    _ => Err(SqlError::UnsupportedFeature(format!(
484                        "Unary operator not supported: {:?}",
485                        op
486                    ))),
487                }
488            }
489            _ => Err(SqlError::UnsupportedFeature(format!(
490                "Expression type not supported as value: {:?}",
491                expr
492            ))),
493        }
494    }

496    /// Convert SQL value to predicate value.
497    fn value_to_predicate_value(&self, value: &Value) -> Result<PredicateValue, SqlError> {
498        match value {
499            Value::Null => Ok(PredicateValue::Null),
500            Value::Boolean(b) => Ok(PredicateValue::Bool(*b)),
501            Value::Number(n, _) => {
502                // Try parsing as i64 first, then f64
503                if let Ok(i) = n.parse::<i64>() {
504                    Ok(PredicateValue::Int(i))
505                } else if let Ok(f) = n.parse::<f64>() {
506                    Ok(PredicateValue::Float(f))
507                } else {
508                    Err(SqlError::TypeError(format!("Invalid number: {}", n)))
509                }
510            }
511            Value::SingleQuotedString(s) | Value::DoubleQuotedString(s) => {
512                Ok(PredicateValue::String(s.clone()))
513            }
514            _ => Err(SqlError::UnsupportedFeature(format!(
515                "Value type not supported: {:?}",
516                value
517            ))),
518        }
519    }

560    /// Resolve an `EmbeddingRef` to a concrete `Arc<[f32]>` using bound parameters.
561    fn resolve_embedding(&self, r: &EmbeddingRef) -> Result<Arc<[f32]>, SqlError> {
562        match r {
563            EmbeddingRef::Literal(v) => Ok(v.clone().into()),
564            EmbeddingRef::Parameter(name) => {
565                // Strip leading $ if present
566                let clean_name = name.strip_prefix('$').unwrap_or(name);
567                match self.parameters.get(clean_name) {
568                    Some(SqlParameterValue::Embedding(e)) => Ok(e.clone()),
569                    Some(_) => Err(SqlError::TypeError(format!(
570                        "Parameter '{}' is not an embedding",
571                        clean_name
572                    ))),
573                    None => Err(SqlError::ParameterError(format!(
574                        "Unbound embedding parameter: {}",
575                        clean_name
576                    ))),
577                }
578            }
579        }
580    }

582    /// Apply a parsed vector operation to the query ops pipeline.
583    fn apply_vector_op(&self, query: &mut Query, op: &VectorOp) -> Result<(), SqlError> {
584        match op {
585            VectorOp::KnnOrderBy {
586                property_key,
587                embedding_ref,
588                metric,
589            } => {
590                let embedding = self.resolve_embedding(embedding_ref)?;
591
592                // Detect whether we have graph traversal ops in the pipeline.
593                // If so, this is a hybrid query: rank traversal results by similarity.
594                let has_traversal = query.ops.iter().any(|op| {
595                    matches!(
596                        op,
597                        QueryOp::TraverseOut { .. }
598                            | QueryOp::TraverseIn { .. }
599                            | QueryOp::TraverseBoth { .. }
600                    )
601                });
602
603                let k = query.ops.iter().find_map(|op| {
604                    if let QueryOp::Limit(n) = op {
605                        Some(*n)
606                    } else {
607                        None
608                    }
609                });
610
611                if has_traversal {
612                    // Hybrid mode: rank traversal results by similarity.
613                    // Insert RankBySimilarity before the Limit op.
614                    let pos = query
615                        .ops
616                        .iter()
617                        .position(|op| matches!(op, QueryOp::Limit(_)))
618                        .unwrap_or(query.ops.len());
619                    query.ops.insert(
620                        pos,
621                        QueryOp::RankBySimilarity {
622                            embedding,
623                            top_k: k,
624                            property_key: Some(property_key.clone()),
625                            metric: crate::core::vector::DistanceMetric::Cosine,
626                            threshold: None,
627                            score_alias: None,
628                        },
629                    );
630                } else {
631                    // Standalone k-NN search.
632                    let limit = k.unwrap_or(10);
633
634                    // If OFFSET is present, VectorSearch must fetch enough rows so
635                    // that after Skip(offset) we still have `limit` results.
636                    let offset = query.ops.iter().find_map(|op| {
637                        if let QueryOp::Skip(n) = op {
638                            Some(*n)
639                        } else {
640                            None
641                        }
642                    });
643                    let effective_k = limit + offset.unwrap_or(0);
644
645                    // Remove the Limit op since VectorSearch has its own k.
646                    // Keep Skip so offset semantics are preserved.
647                    query.ops.retain(|op| !matches!(op, QueryOp::Limit(_)));
648
649                    // Insert VectorSearch after source ops.
650                    let pos = query
651                        .ops
652                        .iter()
653                        .position(|op| {
654                            !matches!(op, QueryOp::ScanNodes { .. } | QueryOp::ScanEdges { .. })
655                        })
656                        .unwrap_or(query.ops.len());
657                    query.ops.insert(
658                        pos,
659                        QueryOp::VectorSearch {
660                            embedding,
661                            k: effective_k,
662                            metric: *metric,
663                            property_key: Some(property_key.clone()),
664                        },
665                    );
666
667                    // Re-add Limit after Skip so the final row count equals `limit`.
668                    if offset.is_some() {
669                        query.ops.push(QueryOp::Limit(limit));
670                    }
671                }
672            }
673            VectorOp::KnnFunction {
674                property_key,
675                embedding_ref,
676                k,
677                ..
678            } => {
679                let embedding = self.resolve_embedding(embedding_ref)?;
680
681                // Preserve label filter from the ScanNodes we are about to remove.
682                // e.g. `FROM KNN('Documents', ...)` produces ScanNodes { label: Some("documents") }.
683                let label = query.ops.iter().find_map(|op| {
684                    if let QueryOp::ScanNodes { label: Some(l) } = op {
685                        Some(l.clone())
686                    } else {
687                        None
688                    }
689                });
690
691                // Replace the scan with VectorSearch.
692                query
693                    .ops
694                    .retain(|op| !matches!(op, QueryOp::ScanNodes { .. }));
695                query.ops.insert(
696                    0,
697                    QueryOp::VectorSearch {
698                        embedding,
699                        k: *k,
700                        metric: DistanceMetric::Cosine,
701                        property_key: Some(property_key.clone()),
702                    },
703                );
704
705                // Re-apply label filter so results are scoped to the original label.
706                if let Some(l) = label {
707                    query.ops.insert(1, QueryOp::FilterLabel(l));
708                }
709            }
710            VectorOp::SimilarToFilter {
711                property_key,
712                embedding_ref,
713                threshold,
714            } => {
715                // A full implementation would add a dedicated VectorFilter QueryOp.
716                // For now we over-fetch with VectorSearch and post-filter by threshold.
717                let embedding = self.resolve_embedding(embedding_ref)?;
718
719                let k = SIMILAR_TO_DEFAULT_K;
720                let pos = query
721                    .ops
722                    .iter()
723                    .position(|op| {
724                        !matches!(op, QueryOp::ScanNodes { .. } | QueryOp::ScanEdges { .. })
725                    })
726                    .unwrap_or(query.ops.len());
727                query.ops.insert(
728                    pos,
729                    QueryOp::VectorSearch {
730                        embedding,
731                        k,
732                        metric: DistanceMetric::Cosine,
733                        property_key: Some(property_key.clone()),
734                    },
735                );
736
737                // Enforce the similarity threshold as a post-search filter.
738                // VectorSearch returns a "score" property for each result;
739                // we keep only results whose score >= the user-specified threshold.
740                query.ops.insert(
741                    pos + 1,
742                    QueryOp::Filter(Predicate::Gte {
743                        key: "score".to_string(),
744                        value: PredicateValue::Float(*threshold),
745                    }),
746                );
747            }
748        }
749
750        Ok(())
751    }
752}
753
754impl Default for SqlConverter {
755    fn default() -> Self {
756        Self::new()
757    }
758}

760/// Parse a SQL query and convert it to a AletheiaDB Query.
761///
762/// This is the simplest way to execute a SQL query.
763///
764/// # Example
765///
766/// ```rust
767/// use aletheiadb::sql::parse_sql;
768///
769/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
770/// let query = parse_sql("SELECT * FROM nodes WHERE label = 'Person' LIMIT 10")?;
771/// # Ok(())
772/// # }
773/// ```
774pub fn parse_sql(sql: &str) -> Result<Query, SqlError> {
775    let converter = SqlConverter::new();
776    converter.convert_sql(sql)
777}

779/// Parse a SQL query with parameters.
780///
781/// # Example
782///
783/// ```rust
784/// use aletheiadb::sql::{parse_sql_with_params, SqlParameterValue};
785/// use aletheiadb::query::ir::PredicateValue;
786/// use std::collections::HashMap;
787///
788/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
789/// let mut params = HashMap::new();
790/// params.insert("min_age".to_string(), SqlParameterValue::Scalar(PredicateValue::Int(21)));
791///
792/// let query = parse_sql_with_params(
793///     "SELECT * FROM nodes WHERE age > min_age",
794///     params
795/// )?;
796/// # Ok(())
797/// # }
798/// ```
799pub fn parse_sql_with_params(
800    sql: &str,
801    params: HashMap<String, SqlParameterValue>,
802) -> Result<Query, SqlError> {
803    let converter = SqlConverter::with_parameters(params);
804    converter.convert_sql(sql)
