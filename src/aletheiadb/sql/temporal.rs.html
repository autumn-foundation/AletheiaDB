1//! SQL:2011 Temporal Clause Handling.
2//!
3//! This module handles parsing and conversion of SQL:2011 temporal clauses:
4//! - `FOR SYSTEM_TIME AS OF timestamp` - Point-in-time transaction time query
5//! - `FOR SYSTEM_TIME BETWEEN t1 AND t2` - Transaction time range query
6//! - `FOR VALID_TIME AS OF timestamp` - Point-in-time valid time query
7//! - Combined temporal specifications
8
9use crate::core::temporal::{TimeRange, Timestamp};
10use chrono::{DateTime, FixedOffset, NaiveDate, NaiveDateTime, Utc};
11
12use super::error::SqlError;
13
14/// Represents a parsed SQL:2011 temporal clause.
15#[derive(Debug, Clone, PartialEq)]
16pub enum TemporalClause {
17    /// Point-in-time query for system (transaction) time.
18    /// `FOR SYSTEM_TIME AS OF timestamp`
19    SystemTimeAsOf(Timestamp),
20
21    /// Time range query for system (transaction) time.
22    /// `FOR SYSTEM_TIME BETWEEN t1 AND t2`
23    SystemTimeBetween(TimeRange),
24
25    /// Point-in-time query for valid (application) time.
26    /// `FOR VALID_TIME AS OF timestamp`
27    ValidTimeAsOf(Timestamp),
28
29    /// Time range query for valid (application) time.
30    /// `FOR VALID_TIME BETWEEN t1 AND t2`
31    ValidTimeBetween(TimeRange),
32
33    /// Combined bi-temporal query.
34    BiTemporal {
35        /// System time specification
36        system_time: Box<TemporalClause>,
37        /// Valid time specification
38        valid_time: Box<TemporalClause>,
39    },
40}
41
42impl TemporalClause {
43    /// Validates that a string matches the SQL timestamp format strictly.
44    ///
45    /// Accepts:
46    /// - `YYYY-MM-DD HH:MM:SS` (exactly 19 characters)
47    /// - `YYYY-MM-DD HH:MM:SS.f` (20-26 characters, where f is 1-6 digits)
48    ///
49    /// This prevents parse_from_str from accepting invalid inputs with trailing characters.
50    /// Uses byte-level validation to avoid allocations.
51    fn is_valid_sql_timestamp(s: &str) -> bool {
52        let bytes = s.as_bytes();
53        let len = bytes.len();
54
55        // Must be at least 19 characters for "YYYY-MM-DD HH:MM:SS"
56        if len < 19 {
57            return false;
58        }
59
60        // Must be at most 26 characters for "YYYY-MM-DD HH:MM:SS.ffffff"
61        if len > 26 {
62            return false;
63        }
64
65        // Check structure: YYYY-MM-DD HH:MM:SS
66        if bytes[4] != b'-'
67            || bytes[7] != b'-'
68            || bytes[10] != b' '
69            || bytes[13] != b':'
70            || bytes[16] != b':'
71        {
72            return false;
73        }
74
75        // Check digit positions
76        let digit_positions = [0, 1, 2, 3, 5, 6, 8, 9, 11, 12, 14, 15, 17, 18];
77        for &pos in &digit_positions {
78            if !bytes[pos].is_ascii_digit() {
79                return false;
80            }
81        }
82
83        // Check fractional seconds if present
84        if len > 19 {
85            if bytes[19] != b'.' {
86                return false;
87            }
88            for &byte in &bytes[20..] {
89                if !byte.is_ascii_digit() {
90                    return false;
91                }
92            }
93        }
94
95        true
96    }

98    /// Parse a timestamp string.
99    ///
100    /// Supports the following formats:
101    /// - **Unix microseconds**: `1705315200000000` (fast path)
102    /// - **ISO 8601 UTC**: `2024-01-15T10:00:00Z`
103    /// - **ISO 8601 with offset**: `2024-01-15T15:30:00+05:30`
104    /// - **SQL timestamp**: `2024-01-15 10:00:00` (assumes UTC)
105    /// - **Date only**: `2024-01-15` (assumes midnight UTC)
106    ///
107    /// All formats support optional single or double quotes and whitespace.
108    ///
109    /// # Examples
110    ///
111    /// ```
112    /// use aletheiadb::sql::TemporalClause;
113    ///
114    /// // Unix microseconds
115    /// let ts = TemporalClause::parse_timestamp("1705315200000000").unwrap();
116    ///
117    /// // ISO 8601 UTC
118    /// let ts = TemporalClause::parse_timestamp("2024-01-15T10:00:00Z").unwrap();
119    ///
120    /// // ISO 8601 with timezone
121    /// let ts = TemporalClause::parse_timestamp("2024-01-15T15:30:00+05:30").unwrap();
122    ///
123    /// // SQL timestamp format
124    /// let ts = TemporalClause::parse_timestamp("2024-01-15 10:00:00").unwrap();
125    ///
126    /// // Date only
127    /// let ts = TemporalClause::parse_timestamp("2024-01-15").unwrap();
128    /// ```
129    pub fn parse_timestamp(s: &str) -> Result<Timestamp, SqlError> {
130        let trimmed = s.trim().trim_matches('\'').trim_matches('"');
131
132        // Fast path: Try parsing as Unix microseconds first
133        if let Ok(micros) = trimmed.parse::<i64>() {
134            return Ok(Timestamp::from(micros));
135        }
136
137        // Try parsing as ISO 8601 with UTC timezone (e.g., "2024-01-15T10:00:00Z")
138        if let Ok(dt) = trimmed.parse::<DateTime<Utc>>() {
139            return Ok(Timestamp::from(dt.timestamp_micros()));
140        }
141
142        // Try parsing as ISO 8601 with timezone offset (e.g., "2024-01-15T15:30:00+05:30")
143        if let Ok(dt) = trimmed.parse::<DateTime<FixedOffset>>() {
144            return Ok(Timestamp::from(dt.with_timezone(&Utc).timestamp_micros()));
145        }
146
147        // Try parsing as a naive datetime with ISO-like format (T separator)
148        // e.g., "2024-01-15T10:00:00" or "2024-01-15T10:00:00.123456"
149        // This parse is strict and does not allow trailing characters.
150        // We assume UTC for naive timestamps.
151        if let Ok(dt) = trimmed.parse::<NaiveDateTime>() {
152            return Ok(Timestamp::from(dt.and_utc().timestamp_micros()));
153        }
154
155        // Try parsing as SQL timestamp format with space separator (e.g., "2024-01-15 10:00:00")
156        // NaiveDateTime::FromStr doesn't support space separator, so we use parse_from_str.
157        // To ensure strictness, we validate the format by checking length and characters.
158        if Self::is_valid_sql_timestamp(trimmed) {
159            // Try with fractional seconds first (more specific)
160            if let Ok(dt) = NaiveDateTime::parse_from_str(trimmed, "%Y-%m-%d %H:%M:%S%.f") {
161                return Ok(Timestamp::from(dt.and_utc().timestamp_micros()));
162            }
163            // Try without fractional seconds
164            if let Ok(dt) = NaiveDateTime::parse_from_str(trimmed, "%Y-%m-%d %H:%M:%S") {
165                return Ok(Timestamp::from(dt.and_utc().timestamp_micros()));
166            }
167        }
168
169        // Try parsing as date-only format (e.g., "2024-01-15").
170        // This parse is strict.
171        // Assume midnight UTC.
172        if let Ok(date) = trimmed.parse::<NaiveDate>() {
173            // `and_hms_opt` is safe and will not panic.
174            if let Some(dt) = date.and_hms_opt(0, 0, 0) {
175                return Ok(Timestamp::from(dt.and_utc().timestamp_micros()));
176            }
177        }
178
179        // None of the formats worked, return detailed error
180        Err(SqlError::InvalidTimestamp(format!(
181            "Cannot parse timestamp '{}'. Supported formats:\n\
182             - Unix microseconds: 1705315200000000\n\
183             - ISO 8601 UTC: 2024-01-15T10:00:00Z\n\
184             - ISO 8601 with offset: 2024-01-15T15:30:00+05:30\n\
185             - SQL timestamp: 2024-01-15 10:00:00 (assumes UTC)\n\
186             - Date only: 2024-01-15 (assumes midnight UTC)",
187            s
188        )))
189    }

191    /// Create a SYSTEM_TIME AS OF clause.
192    pub fn system_time_as_of(timestamp: Timestamp) -> Self {
193        TemporalClause::SystemTimeAsOf(timestamp)
194    }
195
196    /// Create a VALID_TIME AS OF clause.
197    pub fn valid_time_as_of(timestamp: Timestamp) -> Self {
198        TemporalClause::ValidTimeAsOf(timestamp)
199    }
200
201    /// Create a SYSTEM_TIME BETWEEN clause.
202    pub fn system_time_between(start: Timestamp, end: Timestamp) -> Result<Self, SqlError> {
203        let range = TimeRange::new(start, end)
204            .map_err(|e| SqlError::InvalidTemporalClause(format!("Invalid time range: {}", e)))?;
205        Ok(TemporalClause::SystemTimeBetween(range))
206    }
207
208    /// Create a VALID_TIME BETWEEN clause.
209    pub fn valid_time_between(start: Timestamp, end: Timestamp) -> Result<Self, SqlError> {
210        let range = TimeRange::new(start, end)
211            .map_err(|e| SqlError::InvalidTemporalClause(format!("Invalid time range: {}", e)))?;
212        Ok(TemporalClause::ValidTimeBetween(range))
213    }

220    #[test]
221    fn should_validate_sql_timestamps_correctly() {
222        struct TestCase<'a> {
223            input: &'a str,
224            expected: bool,
225            desc: &'a str,
226        }
227
228        let cases = vec![
229            TestCase {
230                input: "2024-01-15 10:00:00",
231                expected: true,
232                desc: "Valid exact length timestamp",
233            },
234            TestCase {
235                input: "2024-01-15 10:00:00.1",
236                expected: true,
237                desc: "Valid 1 fractional digit",
238            },
239            TestCase {
240                input: "2024-01-15 10:00:00.123456",
241                expected: true,
242                desc: "Valid 6 fractional digits",
243            },
244            TestCase {
245                input: "2024-01-15 10:00:00.1234567",
246                expected: false,
247                desc: "Invalid 7 fractional digits (too long)",
248            },
249            TestCase {
250                input: "2024-01-15 10:00:0",
251                expected: false,
252                desc: "Invalid short length",
253            },
254            TestCase {
255                input: "2024/01/15 10:00:00",
256                expected: false,
257                desc: "Invalid separator (slash instead of dash)",
258            },
259            TestCase {
260                input: "2024-01-15T10:00:00",
261                expected: false,
262                desc: "Invalid separator (T instead of space)",
263            },
264            TestCase {
265                input: "2024-01-15 10-00-00",
266                expected: false,
267                desc: "Invalid time separator",
268            },
269            TestCase {
270                input: "abcd-ef-gh ij:kl:mn",
271                expected: false,
272                desc: "Invalid non-digit characters",
273            },
274            TestCase {
275                input: "2024-01-15 10:00:00x123",
276                expected: false,
277                desc: "Invalid fractional separator",
278            },
279            TestCase {
280                input: "2024-01-15 10:00:00.abc",
281                expected: false,
282                desc: "Invalid fractional non-digits",
283            },
284            TestCase {
285                input: "2024-01-15 10:00:00 invalid",
286                expected: false,
287                desc: "Invalid trailing characters",
288            },
289        ];
290
291        for case in cases {
292            assert_eq!(
293                TemporalClause::is_valid_sql_timestamp(case.input),
294                case.expected,
295                "Failed on: {}",
296                case.desc
297            );
298        }
299    }

328        let start = Timestamp::from(1000);

330

337
338        let valid_clause = TemporalClause::valid_time_between(start, end).unwrap();
339        assert!(matches!(valid_clause, TemporalClause::ValidTimeBetween(_)));
340        if let TemporalClause::ValidTimeBetween(range) = valid_clause {
341            assert_eq!(range.start(), start);
342            assert_eq!(range.end(), end);
343        }
