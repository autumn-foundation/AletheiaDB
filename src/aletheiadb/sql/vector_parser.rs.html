1//! Vector operation preprocessing for SQL queries.
2//!
3//! Extracts vector-related syntax from SQL before passing to sqlparser-rs,
4//! similar to how `temporal_parser.rs` handles temporal clauses and
5//! `match_parser.rs` handles graph patterns.
6//!
7//! # Supported Syntax
8//!
9//! ## Distance operators in ORDER BY
10//!
11//! ```sql
12//! ORDER BY embedding <=> '[0.1, 0.2, 0.3]'::vector LIMIT 10
13//! ORDER BY embedding <=> $query_embedding LIMIT 10
14//! ORDER BY embedding <-> '[0.1, 0.2, 0.3]'::vector LIMIT 10
15//! ```
16//!
17//! - `<=>` = cosine distance
18//! - `<->` = L2 (Euclidean) distance
19//!
20//! ## KNN function in FROM clause
21//!
22//! ```sql
23//! SELECT * FROM KNN('Documents', 'embedding', $query, 10)
24//! ```
25//!
26//! ## SIMILAR_TO in WHERE clause
27//!
28//! ```sql
29//! WHERE SIMILAR_TO(embedding, $query_embedding, 0.8)
30//! ```
31
32use super::error::SqlError;
33use crate::index::vector::DistanceMetric;

35/// Result of extracting vector clauses from SQL.
36#[derive(Debug)]
37pub struct ExtractedVector {
38    /// The SQL string with vector clauses removed/replaced.
39    pub cleaned_sql: String,
40    /// Parsed vector operations.
41    pub vector_ops: Vec<VectorOp>,
42}
43
44/// A parsed vector operation extracted from SQL.
45#[derive(Debug)]
46pub enum VectorOp {
47    /// `ORDER BY property <=> vector LIMIT k`
48    KnnOrderBy {
49        property_key: String,
50        embedding_ref: EmbeddingRef,
51        metric: DistanceMetric,
52    },
53    /// `FROM KNN('label', 'property', embedding, k)`
54    KnnFunction {
55        /// The node label extracted from KNN's first argument.
56        /// Used during SQL rewriting; kept here for debugging/display.
57        #[allow(dead_code)]
58        label: String,
59        property_key: String,
60        embedding_ref: EmbeddingRef,
61        k: usize,
62    },
63    /// `WHERE SIMILAR_TO(property, embedding, threshold)`
64    SimilarToFilter {
65        property_key: String,
66        embedding_ref: EmbeddingRef,
67        threshold: f64,
68    },
69}
70
71/// Reference to an embedding value — either inline literal or parameter.
72#[derive(Debug, Clone)]
73pub enum EmbeddingRef {
74    /// A parameter reference like `$query_embedding`.
75    Parameter(String),
76    /// An inline vector literal like `[0.1, 0.2]`.
77    Literal(Vec<f32>),
78}

80/// Extract vector clauses from a SQL string.
81///
82/// Detects and removes:
83/// 1. `ORDER BY prop <=> vector_ref` distance operators
84/// 2. `FROM KNN(...)` function calls
85/// 3. `WHERE SIMILAR_TO(...)` function calls
86///
87/// Returns cleaned SQL suitable for sqlparser-rs plus parsed vector ops.
88pub fn extract_vector_clauses(sql: &str) -> Result<ExtractedVector, SqlError> {
89    let mut cleaned = sql.to_string();
90    let mut vector_ops = Vec::new();
91
92    // 1. Extract KNN function from FROM clause
93    extract_knn_function(&mut cleaned, &mut vector_ops)?;
94
95    // 2. Extract SIMILAR_TO from WHERE clause
96    extract_similar_to(&mut cleaned, &mut vector_ops)?;
97
98    // 3. Extract ORDER BY distance operators
99    extract_order_by_distance(&mut cleaned, &mut vector_ops)?;
100
101    Ok(ExtractedVector {
102        cleaned_sql: cleaned,
103        vector_ops,
104    })
105}

107/// Extract `FROM KNN('label', 'property', $embedding, k)` from SQL.
108///
109/// Replaces the KNN function call with the label as a normal table reference.
110fn extract_knn_function(sql: &mut String, ops: &mut Vec<VectorOp>) -> Result<(), SqlError> {
111    // Case-insensitive search for KNN( outside string literals.
112    // Use the original string directly; find_outside_strings handles case-insensitive
113    // matching and returns byte offsets valid for the original string.
114    let Some(knn_pos) = find_outside_strings(sql, "KNN(") else {
115        return Ok(());
116    };
117
118    // Find the matching closing paren
119    let open_paren = knn_pos + 3; // position of '('
120    let close_paren = find_matching_paren(sql, open_paren).ok_or_else(|| {
121        SqlError::ParseError("Unmatched parenthesis in KNN() function".to_string())
122    })?;
123
124    // Extract the arguments between the parens
125    let args_str = &sql[open_paren + 1..close_paren];
126    let args = split_args(args_str)?;
127
128    if args.len() != 4 {
129        return Err(SqlError::ParseError(format!(
130            "KNN() requires 4 arguments (label, property, embedding, k), got {}",
131            args.len()
132        )));
133    }
134
135    let label = unquote_string(&args[0])?;
136    let property_key = unquote_string(&args[1])?;
137    let embedding_ref = parse_embedding_ref(&args[2])?;
138    let k = args[3].trim().parse::<usize>().map_err(|_| {
139        SqlError::ParseError(format!("Invalid k value in KNN(): '{}'", args[3].trim()))
140    })?;
141
142    ops.push(VectorOp::KnnFunction {
143        label: label.clone(),
144        property_key,
145        embedding_ref,
146        k,
147    });
148
149    // Replace `KNN(...)` with the label name as a plain table reference
150    let replacement = label.to_lowercase();
151    sql.replace_range(knn_pos..close_paren + 1, &replacement);
152
153    Ok(())
154}

156/// Extract `SIMILAR_TO(property, $embedding, threshold)` from WHERE clause.
157///
158/// Replaces the SIMILAR_TO call with `1=1` so sqlparser can still parse the WHERE.
159fn extract_similar_to(sql: &mut String, ops: &mut Vec<VectorOp>) -> Result<(), SqlError> {
160    // Use the original string directly; find_outside_strings handles case-insensitive
161    // matching and returns byte offsets valid for the original string.
162    let Some(st_pos) = find_outside_strings(sql, "SIMILAR_TO(") else {
163        return Ok(());
164    };
165
166    let open_paren = st_pos + 10; // position of '('
167    let close_paren = find_matching_paren(sql, open_paren).ok_or_else(|| {
168        SqlError::ParseError("Unmatched parenthesis in SIMILAR_TO() function".to_string())
169    })?;
170
171    let args_str = &sql[open_paren + 1..close_paren];
172    let args = split_args(args_str)?;
173
174    if args.len() != 3 {
175        return Err(SqlError::ParseError(format!(
176            "SIMILAR_TO() requires 3 arguments (property, embedding, threshold), got {}",
177            args.len()
178        )));
179    }
180
181    let property_key = args[0].trim().to_string();
182    let embedding_ref = parse_embedding_ref(&args[1])?;
183    let threshold = args[2].trim().parse::<f64>().map_err(|_| {
184        SqlError::ParseError(format!(
185            "Invalid threshold in SIMILAR_TO(): '{}'",
186            args[2].trim()
187        ))
188    })?;
189
190    ops.push(VectorOp::SimilarToFilter {
191        property_key,
192        embedding_ref,
193        threshold,
194    });
195
196    // Replace SIMILAR_TO(...) with TRUE so sqlparser can handle the WHERE clause
197    sql.replace_range(st_pos..close_paren + 1, "TRUE");
198
199    // Clean up "WHERE TRUE" if it's the only predicate remaining
200    cleanup_where_true(sql);
201
202    Ok(())
203}

205/// Extract `ORDER BY property <=> vector_ref` from SQL.
206///
207/// Removes the distance-based ORDER BY clause from the SQL, leaving only
208/// LIMIT/OFFSET for sqlparser to handle.
209fn extract_order_by_distance(sql: &mut String, ops: &mut Vec<VectorOp>) -> Result<(), SqlError> {
210    // Look for `<=>` or `<->` outside string literals
211    let (op_pos, op_len, metric) = if let Some(pos) = find_outside_strings(sql, "<=>") {
212        (pos, 3, DistanceMetric::Cosine)
213    } else if let Some(pos) = find_outside_strings(sql, "<->") {
214        (pos, 3, DistanceMetric::Euclidean)
215    } else {
216        return Ok(());
217    };
218
219    // Find the ORDER BY that governs this distance operator.
220    // Search backwards from the operator position for "ORDER BY".
221    // Use rfind_ascii_ci on the original string to get a byte offset that is
222    // valid for slicing `sql` even when multi-byte characters are present.
223    let order_by_pos = rfind_ascii_ci(&sql[..op_pos], "ORDER BY").ok_or_else(|| {
224        SqlError::ParseError(
225            "Distance operator <=> / <-> must appear in an ORDER BY clause".to_string(),
226        )
227    })?;
228
229    // Extract property name: between "ORDER BY" and the operator.
230    let between = sql[order_by_pos + 8..op_pos].trim();
231    let property_key = extract_property_from_order_by(between)?;
232
233    // Extract vector reference: from after the operator to the next keyword or end.
234    let after_op = op_pos + op_len;
235    let (embedding_ref, ref_end) = extract_vector_ref_from_sql(sql, after_op)?;
236
237    ops.push(VectorOp::KnnOrderBy {
238        property_key,
239        embedding_ref,
240        metric,
241    });
242
243    // Remove the ORDER BY distance clause.
244    // Keep everything before ORDER BY and everything after the vector ref.
245    let remainder = sql[ref_end..].trim_start().to_string();
246    let prefix = sql[..order_by_pos].to_string();
247
248    *sql = format!("{}{}", prefix, remainder);
249
250    Ok(())
251}

257/// If the WHERE clause is now just `WHERE TRUE` (or `WHERE TRUE AND ...` was reduced),
258/// remove the standalone `WHERE TRUE` to avoid confusing the SQL converter.
259fn cleanup_where_true(sql: &mut String) {
260    // Use the original string directly; find_outside_strings handles case-insensitive
261    // matching and returns byte offsets valid for the original string.
262    let Some(where_pos) = find_outside_strings(sql, "WHERE") else {
263        return;
264    };
265
266    // Find the start of the predicate content after WHERE
267    let predicate_start = where_pos + 5; // len("WHERE")
268    let whitespace_len = sql[predicate_start..]
269        .len()
270        .saturating_sub(sql[predicate_start..].trim_start().len());
271    let true_pos = predicate_start + whitespace_len;
272
273    let remainder_upper = sql[true_pos..].to_uppercase();
274    let Some(after_true) = remainder_upper.strip_prefix("TRUE") else {
275        return;
276    };
277
278    let true_end = true_pos + 4; // len("TRUE")
279    let after_true_trimmed = after_true.trim_start();
280
281    if after_true_trimmed.is_empty()
282        || after_true_trimmed.starts_with("ORDER")
283        || after_true_trimmed.starts_with("LIMIT")
284        || after_true_trimmed.starts_with("OFFSET")
285        || after_true_trimmed.starts_with(';')
286    {
287        // Standalone WHERE TRUE — remove "WHERE TRUE" entirely
288        sql.replace_range(where_pos..true_end, "");
289        let trimmed = sql.replace("  ", " ");
290        *sql = trimmed.trim().to_string();
291    } else if after_true_trimmed.starts_with("AND") {
292        // WHERE TRUE AND other → WHERE other
293        // Remove from "TRUE" through "AND " (including trailing space)
294        let and_pos = true_end + (after_true.len() - after_true_trimmed.len());
295        let and_end = and_pos + 3; // len("AND")
296        let after_and_ws = sql[and_end..].len() - sql[and_end..].trim_start().len();
297        sql.replace_range(true_pos..and_end + after_and_ws, "");
298    }
299}

301/// Find the last byte position of an ASCII `needle` in `haystack` using
302/// ASCII-case-insensitive comparison.
303///
304/// The returned offset is a valid byte index into `haystack` regardless of
305/// any multi-byte Unicode characters in the haystack, because the search only
306/// considers char-boundary-aligned positions and `needle` is pure ASCII.
307fn rfind_ascii_ci(haystack: &str, needle: &str) -> Option<usize> {
308    debug_assert!(needle.is_ascii(), "rfind_ascii_ci: needle must be ASCII");
309    let n = needle.len();
310    if n == 0 {
311        return Some(haystack.len());
312    }
313    let haystack_bytes = haystack.as_bytes();
314    let needle_bytes = needle.as_bytes();
315    let mut last = None;
316    for (byte_pos, _) in haystack.char_indices() {
317        if byte_pos + n <= haystack_bytes.len()
318            && haystack_bytes[byte_pos..byte_pos + n].eq_ignore_ascii_case(needle_bytes)
319        {
320            last = Some(byte_pos);
321        }
322    }
323    last
324}

326/// Find a substring outside of single-quoted string literals.
327/// Returns the byte position of the first match, or None.
328fn find_outside_strings(sql: &str, needle: &str) -> Option<usize> {
329    let chars: Vec<char> = sql.chars().collect();
330    let needle_chars: Vec<char> = needle.chars().collect();
331    let needle_len = needle_chars.len();
332
333    let mut i = 0;
334    while i < chars.len() {
335        // Skip single-quoted strings
336        if chars[i] == '\'' {
337            i += 1;
338            while i < chars.len() {
339                if chars[i] == '\'' {
340                    if i + 1 < chars.len() && chars[i + 1] == '\'' {
341                        i += 2; // escaped quote
342                    } else {
343                        i += 1;
344                        break;
345                    }
346                } else {
347                    i += 1;
348                }
349            }
350            continue;
351        }
352
353        // Check for needle match
354        if i + needle_len <= chars.len() {
355            let candidate: String = chars[i..i + needle_len].iter().collect();
356            if candidate == needle {
357                // Convert char index to byte offset
358                let byte_offset: usize = chars[..i].iter().map(|c| c.len_utf8()).sum();
359                return Some(byte_offset);
360            }
361            // Also check case-insensitive for keyword-like needles (not operators)
362            if !needle.starts_with('<')
363                && !needle.starts_with('>')
364                && candidate.to_uppercase() == needle.to_uppercase()
365            {
366                let byte_offset: usize = chars[..i].iter().map(|c| c.len_utf8()).sum();
367                return Some(byte_offset);
368            }
369        }
370
371        i += 1;
372    }
373
374    None
375}

377/// Find the matching closing parenthesis for an opening paren at `open_pos`.
378fn find_matching_paren(sql: &str, open_pos: usize) -> Option<usize> {
379    let bytes = sql.as_bytes();
380    if bytes.get(open_pos) != Some(&b'(') {
381        return None;
382    }
383
384    let mut depth = 1;
385    let mut i = open_pos + 1;
386    let mut in_string = false;
387
388    while i < bytes.len() {
389        if in_string {
390            if bytes[i] == b'\'' {
391                if i + 1 < bytes.len() && bytes[i + 1] == b'\'' {
392                    i += 2;
393                    continue;
394                }
395                in_string = false;
396            }
397        } else {
398            match bytes[i] {
399                b'\'' => in_string = true,
400                b'(' => depth += 1,
401                b')' => {
402                    depth -= 1;
403                    if depth == 0 {
404                        return Some(i);
405                    }
406                }
407                _ => {}
408            }
409        }
410        i += 1;
411    }
412
413    None
414}

416/// Split function arguments by comma, respecting parentheses and string literals.
417fn split_args(args: &str) -> Result<Vec<String>, SqlError> {
418    let mut result = Vec::new();
419    let mut current = String::new();
420    let mut depth = 0;
421    let mut in_string = false;
422
423    for ch in args.chars() {
424        if in_string {
425            current.push(ch);
426            if ch == '\'' {
427                in_string = false;
428            }
429            continue;
430        }
431
432        match ch {
433            '\'' => {
434                in_string = true;
435                current.push(ch);
436            }
437            '(' => {
438                depth += 1;
439                current.push(ch);
440            }
441            ')' => {
442                depth -= 1;
443                current.push(ch);
444            }
445            ',' if depth == 0 => {
446                result.push(current.trim().to_string());
447                current = String::new();
448            }
449            _ => current.push(ch),
450        }
451    }
452
453    let trimmed = current.trim().to_string();
454    if !trimmed.is_empty() {
455        result.push(trimmed);
456    }
457
458    Ok(result)
459}

461/// Remove surrounding single or double quotes from a string literal.
462fn unquote_string(s: &str) -> Result<String, SqlError> {
463    let s = s.trim();
464    if (s.starts_with('\'') && s.ends_with('\'')) || (s.starts_with('"') && s.ends_with('"')) {
465        Ok(s[1..s.len() - 1].to_string())
466    } else {
467        Err(SqlError::ParseError(format!(
468            "Expected quoted string, got: '{}'",
469            s
470        )))
471    }
472}

474/// Parse an embedding reference — either `$param_name` or `'[0.1, 0.2]'::vector`.
475fn parse_embedding_ref(s: &str) -> Result<EmbeddingRef, SqlError> {
476    let s = s.trim();
477
478    // Parameter reference: $name
479    if s.starts_with('$') {
480        return Ok(EmbeddingRef::Parameter(s.to_string()));
481    }
482
483    // Vector literal: '[0.1, 0.2, 0.3]'::vector or '[0.1, 0.2, 0.3]'
484    if s.starts_with('\'') {
485        let values = parse_vector_literal(s)?;
486        return Ok(EmbeddingRef::Literal(values));
487    }
488
489    // Bare identifier (could be a parameter without $)
490    if s.chars().all(|c| c.is_alphanumeric() || c == '_') {
491        return Ok(EmbeddingRef::Parameter(format!("${}", s)));
492    }
493
494    Err(SqlError::ParseError(format!(
495        "Cannot parse embedding reference: '{}'",
496        s
497    )))
498}

500/// Parse a vector literal like `'[0.1, 0.2, 0.3]'::vector` or `'[0.1, 0.2, 0.3]'`.
501fn parse_vector_literal(s: &str) -> Result<Vec<f32>, SqlError> {
502    let s = s.trim();
503
504    // Strip ::vector suffix if present
505    let s = if let Some(cast_pos) = s.find("::") {
506        &s[..cast_pos]
507    } else {
508        s
509    };
510
511    // Strip surrounding single quotes
512    let s = if s.starts_with('\'') && s.ends_with('\'') {
513        &s[1..s.len() - 1]
514    } else {
515        s
516    };
517
518    // Strip brackets
519    let s = s.trim();
520    let s = s
521        .strip_prefix('[')
522        .and_then(|s| s.strip_suffix(']'))
523        .ok_or_else(|| {
524            SqlError::ParseError(format!("Vector literal must be enclosed in []: '{}'", s))
525        })?;
526
527    // Parse comma-separated floats
528    let values: Result<Vec<f32>, _> = s.split(',').map(|v| v.trim().parse::<f32>()).collect();
529    values.map_err(|e| SqlError::ParseError(format!("Invalid vector literal: {}", e)))
530}

532/// Extract property name from ORDER BY text before the distance operator.
533///
534/// Handles plain identifiers and `table.column` forms.
535fn extract_property_from_order_by(text: &str) -> Result<String, SqlError> {
536    let text = text.trim();
537    if text.is_empty() {
538        return Err(SqlError::ParseError(
539            "Missing property name in ORDER BY distance clause".to_string(),
540        ));
541    }
542
543    // Handle `table.column` form — take the column part
544    if let Some(dot_pos) = text.rfind('.') {
545        let col = text[dot_pos + 1..].trim();
546        if !col.is_empty() {
547            return Ok(col.to_string());
548        }
549    }
550
551    // Plain identifier
552    Ok(text.to_string())
553}

555/// Extract a vector reference from SQL starting at `start`, returning the
556/// parsed reference and the byte position just past it.
557fn extract_vector_ref_from_sql(sql: &str, start: usize) -> Result<(EmbeddingRef, usize), SqlError> {
558    let rest = sql[start..].trim_start();
559    let trimmed_offset = start + (sql.len() - start - rest.len());
560
561    // Parameter reference: $name
562    if rest.starts_with('$') {
563        let end = rest
564            .find(|c: char| !c.is_alphanumeric() && c != '_' && c != '$')
565            .unwrap_or(rest.len());
566        let param = &rest[..end];
567        return Ok((
568            EmbeddingRef::Parameter(param.to_string()),
569            trimmed_offset + end,
570        ));
571    }
572
573    // Vector literal: '[...]'::vector
574    if let Some(after_quote) = rest.strip_prefix('\'') {
575        // Find the closing quote (position relative to rest, not after_quote)
576        let close_quote = after_quote.find('\'').map(|p| p + 1).ok_or_else(|| {
577            SqlError::ParseError("Unterminated vector literal in ORDER BY".to_string())
578        })?;
579
580        let mut end = close_quote + 1;
581
582        // Check for ::vector cast suffix
583        if rest[end..].starts_with("::") {
584            // Find end of cast type
585            let cast_end = rest[end + 2..]
586                .find(|c: char| !c.is_alphanumeric() && c != '_')
587                .map(|p| end + 2 + p)
588                .unwrap_or(rest.len());
589            end = cast_end;
590        }
591
592        let literal_text = &rest[..end];
593        let values = parse_vector_literal(literal_text)?;
594        return Ok((EmbeddingRef::Literal(values), trimmed_offset + end));
595    }
596
597    Err(SqlError::ParseError(
598        "Expected vector literal ('[...]'::vector) or parameter ($name) after distance operator"
599            .to_string(),
600    ))
601}

767    fn test_extract_similar_to() {
768        let result = extract_vector_clauses(
769            "SELECT * FROM Documents WHERE SIMILAR_TO(embedding, $query_embedding, 0.8)",
770        )
771        .unwrap();
772
773        assert_eq!(result.vector_ops.len(), 1);
774        assert!(matches!(
775            &result.vector_ops[0],
776            VectorOp::SimilarToFilter {
777                property_key,
778                threshold,
779                ..
780            } if property_key == "embedding" && (*threshold - 0.8).abs() < f64::EPSILON
781        ));
782
783        // SIMILAR_TO should be removed from the SQL
784        assert!(!result.cleaned_sql.to_uppercase().contains("SIMILAR_TO"));
785        // WHERE TRUE is cleaned up, so no WHERE should remain
786        assert!(
787            !result.cleaned_sql.to_uppercase().contains("WHERE"),
788            "WHERE clause should be removed when SIMILAR_TO was the only predicate, got: '{}'",
789            result.cleaned_sql
790        );
791    }
