1//! Fault injection for deterministic simulation testing.
2//!
3//! `FaultConfig` describes what faults to inject; `FaultInjector` applies them
4//! deterministically using a seeded PRNG so every run with the same seed
5//! reproduces the identical fault sequence.
6
7use rand::rngs::SmallRng;
8use rand::{Rng, SeedableRng};
9use std::cell::RefCell;
10
11/// Taxonomy of injectable faults.
12#[derive(Debug, Clone, PartialEq, Eq)]
13pub enum FaultType {
14    /// Flip random bits in storage data.
15    StorageCorruption,
16    /// Simulate slow I/O by recording an artificial latency.
17    StorageLatency {
18        /// Simulated latency in microseconds.
19        micros: u64,
20    },
21    /// Simulate a process crash once `byte_offset` bytes have been written.
22    CrashAtOffset {
23        /// Cumulative byte offset at which the crash is triggered.
24        byte_offset: u64,
25    },
26    /// Inject a clock discontinuity.
27    ClockJump {
28        /// Microseconds to jump (positive = forward, negative = backward).
29        delta_micros: i64,
30    },
31}
32
33/// Configuration for the fault injector.
34///
35/// Build with the builder methods:
36/// ```ignore
37/// let cfg = FaultConfig::default()
38///     .with_corruption_rate(0.1)
39///     .with_crash_at_offset(4096);
40/// ```
41#[derive(Debug, Clone)]
42pub struct FaultConfig {
43    /// Probability [0.0, 1.0] that a storage write is corrupted.
44    pub corruption_rate: f64,
45    /// If set, writing past this cumulative byte offset triggers a simulated crash.
46    pub crash_at_byte_offset: Option<u64>,
47}
48
49impl Default for FaultConfig {
50    fn default() -> Self {
51        Self {
52            corruption_rate: 0.0,
53            crash_at_byte_offset: None,
54        }
55    }
56}
57
58impl FaultConfig {
59    /// Set the probability that any individual storage write is corrupted.
60    pub fn with_corruption_rate(mut self, rate: f64) -> Self {
61        assert!(
62            (0.0..=1.0).contains(&rate),
63            "corruption_rate must be in [0.0, 1.0]"
64        );
65        self.corruption_rate = rate;
66        self
67    }
68
69    /// Trigger a simulated crash once the cumulative bytes written reach `offset`.
70    pub fn with_crash_at_offset(mut self, offset: u64) -> Self {
71        self.crash_at_byte_offset = Some(offset);
72        self
73    }
74}
75
76/// Applies faults described by a [`FaultConfig`] using a seeded PRNG.
77///
78/// All decisions are drawn from the same PRNG in order, so the same seed
79/// always produces the same fault sequence.
80#[derive(Debug)]
81pub struct FaultInjector {
82    config: FaultConfig,
83    /// Interior-mutable so `try_corrupt` can be called via `&self`.
84    rng: RefCell<SmallRng>,
85}
86
87impl FaultInjector {
88    /// Create a new injector from `config` and a deterministic `seed`.
89    pub fn new(config: FaultConfig, seed: u64) -> Self {
90        Self {
91            config,
92            rng: RefCell::new(SmallRng::seed_from_u64(seed)),
93        }
94    }
95
96    /// Return the underlying fault configuration.
97    pub fn config(&self) -> &FaultConfig {
98        &self.config
99    }
100
101    /// Possibly corrupt `data` in-place according to the configured corruption rate.
102    ///
103    /// At `rate = 0.0` the data is never modified.
104    /// At `rate = 1.0` every byte is XOR'd with a random non-zero mask.
105    pub fn try_corrupt(&self, data: &mut [u8]) {
106        if data.is_empty() || self.config.corruption_rate == 0.0 {
107            return;
108        }
109
110        let mut rng = self.rng.borrow_mut();
111        if self.config.corruption_rate >= 1.0
112            || rng.gen_range(0.0_f64..1.0_f64) < self.config.corruption_rate
113        {
114            // Flip a random bit in each byte to guarantee at least one change.
115            for byte in data.iter_mut() {
116                let mask: u8 = rng.gen_range(1_u8..=255_u8);
117                *byte ^= mask;
118            }
119        }
120    }
121
122    /// Return `true` if `bytes_written_so_far` has reached or exceeded the
123    /// configured crash-at-byte-offset threshold.
124    pub fn should_crash_at(&self, bytes_written_so_far: u64) -> bool {
125        match self.config.crash_at_byte_offset {
126            Some(threshold) => bytes_written_so_far >= threshold,
127            None => false,
128        }
129    }
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    #[test]
137    fn zero_rate_never_corrupts() {
138        let fi = FaultInjector::new(FaultConfig::default(), 0);
139        let mut data = vec![0xAB_u8; 32];
140        let original = data.clone();
141        for _ in 0..100 {
142            fi.try_corrupt(&mut data);
143        }
144        assert_eq!(data, original);
145    }
146
147    #[test]
148    fn full_rate_always_corrupts() {
149        let cfg = FaultConfig::default().with_corruption_rate(1.0);
150        let fi = FaultInjector::new(cfg, 1);
151        let mut data = vec![0x00_u8; 8];
152        fi.try_corrupt(&mut data);
153        assert!(data.iter().any(|&b| b != 0));
154    }
155
156    #[test]
157    fn crash_threshold_triggers_at_offset() {
158        let cfg = FaultConfig::default().with_crash_at_offset(100);
159        let fi = FaultInjector::new(cfg, 0);
160        assert!(!fi.should_crash_at(99));
161        assert!(fi.should_crash_at(100));
162        assert!(fi.should_crash_at(101));
163    }
164
165    #[test]
166    fn same_seed_same_output() {
167        let cfg = FaultConfig::default().with_corruption_rate(0.5);
168        let fi_a = FaultInjector::new(cfg.clone(), 42);
169        let fi_b = FaultInjector::new(cfg, 42);
170        let mut da = vec![0xFF_u8; 16];
171        let mut db = da.clone();
172        fi_a.try_corrupt(&mut da);
173        fi_b.try_corrupt(&mut db);
174        assert_eq!(da, db);
175    }
176}
