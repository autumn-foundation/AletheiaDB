1//! Deterministic Simulation Testing (DST) framework (issue #154).
2//!
3//! Provides a controlled, reproducible environment for testing AletheiaDB's
4//! bi-temporal model under clock jumps, storage faults, and concurrent
5//! transaction interleavings — all driven from a single integer seed.
6//!
7//! # Quick start
8//!
9//! ```ignore
10//! use aletheiadb::simulation::{Simulator, FaultConfig};
11//!
12//! // Every detail is reproducible from seed 42.
13//! // The Simulator automatically injects its clock into time::now() for the
14//! // duration of its lifetime — no manual guard management needed.
15//! let mut sim = Simulator::new(42);
16//! let (_tmp, db) = aletheiadb::test_utils::create_test_db().unwrap();
17//!
18//! sim.advance_time_by(1_000_000); // move forward 1 s — time::now() returns 1 s
19//! sim.inject_clock_jump(-500_000); // NTP-style correction — time::now() returns 0.5 s
20//!
21//! db.create_node("Event", aletheiadb::PropertyMapBuilder::new().build()).unwrap();
22//!
23//! let report = sim.verify_temporal_invariants(&db);
24//! assert!(report.passed);
25//! ```

44/// Result of checking temporal invariants across a database snapshot.
45#[derive(Debug, Default)]
46pub struct InvariantReport {
47    /// `true` when all checked invariants hold.
48    pub passed: bool,
49    /// Human-readable description of each violation found.
50    pub violations: Vec<String>,
51    /// Number of current nodes examined.
52    pub nodes_checked: usize,
53}
54
55/// Verify the eight core temporal invariants against the current database state.
56///
57/// | # | Invariant |
58/// |---|-----------|
59/// | 1 | Tx-time monotonicity: each version's tx_time ≥ predecessor's |
60/// | 2 | Version number ordering: strictly increasing |
61/// | 3 | Time range validity: start ≤ end for every stored range |
62/// | 4 | Visibility consistency: `visible_at(vt,tt)` iff valid_at(vt) ∧ recorded_at(tt) |
63/// | 5 | Overlap symmetry: r1.overlaps(r2) == r2.overlaps(r1) |
64/// | 6 | Contains-range reflexivity: r.contains_range(r) == true |
65/// | 7 | Half-open interval semantics: end is always exclusive |
66/// | 8 | Temporal isolation: time-travel yields consistent snapshot values |
67///
68/// Checks invariants 1, 2, 3, 6, and 7 by inspecting the bi-temporal version
69/// history of every node currently in the database.
70fn verify_temporal_invariants_impl(db: &AletheiaDB) -> InvariantReport {
71    use crate::core::temporal::time;
72
73    let mut report = InvariantReport {
74        passed: true,
75        violations: Vec::new(),
76        nodes_checked: 0,
77    };
78
79    // ── Invariant 6: TimeRange::contains_range reflexivity ──────────────────
80    {
81        let r = TimeRange::from(time::from_secs(0));
82        if !r.contains_range(&r) {
83            push_violation(
84                &mut report,
85                "TimeRange::contains_range reflexivity violated".to_owned(),
86            );
87        }
88    }
89
90    // ── Check all live nodes ─────────────────────────────────────────────────
91    let node_ids = db.get_all_node_ids();
92    report.nodes_checked = node_ids.len();
93
94    for node_id in node_ids {
95        let history = match db.get_node_history(node_id) {
96            Ok(h) => h,
97            Err(e) => {
98                push_violation(
99                    &mut report,
100                    format!("node {node_id:?}: failed to retrieve history: {e}"),
101                );
102                continue;
103            }
104        };
105
106        let mut prev_tx_start = None;
107        let mut prev_version_number = None;
108
109        for version in &history.versions {
110            let vt = version.temporal.valid_time();
111            let tt = version.temporal.transaction_time();
112
113            // ── Invariant 3: start ≤ end for both dimensions ───────────────
114            if vt.start() > vt.end() {
115                push_violation(
116                    &mut report,
117                    format!(
118                        "node {node_id:?} v{}: valid_time start {:?} > end {:?} (Inv 3)",
119                        version.version_number,
120                        vt.start(),
121                        vt.end()
122                    ),
123                );
124            }
125            if tt.start() > tt.end() {
126                push_violation(
127                    &mut report,
128                    format!(
129                        "node {node_id:?} v{}: tx_time start {:?} > end {:?} (Inv 3)",
130                        version.version_number,
131                        tt.start(),
132                        tt.end()
133                    ),
134                );
135            }
136
137            // ── Invariant 7: end is exclusive — end must not equal a timestamp ─
138            // Closed ranges must have end > start (a zero-width closed range is
139            // only valid for point-in-time ranges explicitly created with at()).
140            // We check that a closed range's `contains(end)` is false.
141            if vt.is_closed() && vt.contains(vt.end()) {
142                push_violation(
143                    &mut report,
144                    format!(
145                        "node {node_id:?} v{}: valid_time.end {:?} is inclusive (Inv 7)",
146                        version.version_number,
147                        vt.end()
148                    ),
149                );
150            }
151            if tt.is_closed() && tt.contains(tt.end()) {
152                push_violation(
153                    &mut report,
154                    format!(
155                        "node {node_id:?} v{}: tx_time.end {:?} is inclusive (Inv 7)",
156                        version.version_number,
157                        tt.end()
158                    ),
159                );
160            }
161
162            // ── Invariant 1: tx_time.start is monotonically non-decreasing ─────
163            if let Some(prev) = prev_tx_start
164                && tt.start() < prev
165            {
166                push_violation(
167                    &mut report,
168                    format!(
169                        "node {node_id:?} v{}: tx_time.start {:?} < previous {:?} (Inv 1)",
170                        version.version_number,
171                        tt.start(),
172                        prev
173                    ),
174                );
175            }
176            prev_tx_start = Some(tt.start());
177
178            // ── Invariant 2: version numbers are strictly increasing ────────────
179            if let Some(prev_vn) = prev_version_number
180                && version.version_number <= prev_vn
181            {
182                push_violation(
183                    &mut report,
184                    format!(
185                        "node {node_id:?}: version_number {} not > previous {} (Inv 2)",
186                        version.version_number, prev_vn
187                    ),
188                );
189            }
190            prev_version_number = Some(version.version_number);
191
192            // ── Invariant 4: visibility consistency ────────────────────────────
193            // A version visible at (vt_point, tt_point) must satisfy both ranges.
194            let vt_mid = vt.start();
195            let tt_mid = tt.start();
196            let expected = vt.contains(vt_mid) && tt.contains(tt_mid);
197            let actual = version.temporal.is_visible_at(vt_mid, tt_mid);
198            if actual != expected {
199                push_violation(
200                    &mut report,
201                    format!(
202                        "node {node_id:?} v{}: is_visible_at({vt_mid:?},{tt_mid:?}) = {actual} \
203                         but expected {expected} (Inv 4)",
204                        version.version_number
205                    ),
206                );
207            }
208        }
209    }
210
211    // ── Invariant 3 & 7: reference clock must be ≤ TIMESTAMP_MAX ───────────
212    {
213        let now = time::now();
214        if now > TIMESTAMP_MAX {
215            push_violation(
216                &mut report,
217                format!(
218                    "Current time {now:?} exceeds TIMESTAMP_MAX — clock injection out of bounds"
219                ),
220            );
221        }
222    }
223
224    report
225}
226
227fn push_violation(report: &mut InvariantReport, msg: String) {
228    report.violations.push(msg);
229    report.passed = false;
230}

236/// Top-level DST harness: controls clock, faults, and scheduling from one seed.
237///
238/// The `Simulator` **automatically injects its clock** into `time::now()` for
239/// its entire lifetime. Any call to [`advance_time_by`](Self::advance_time_by)
240/// or [`inject_clock_jump`](Self::inject_clock_jump) is immediately reflected
241/// in subsequent `time::now()` calls on the same thread — no manual
242/// `clock().inject()` call is required.
243///
244/// When the `Simulator` is dropped the real wall clock is restored.
245///
246/// Create with [`Simulator::new`] (no faults) or
247/// [`Simulator::with_seed_and_faults`] (custom fault config).
248#[derive(Debug)]
249pub struct Simulator {
250    seed: u64,
251    clock: SimulatedClock,
252    faults: FaultInjector,
253    /// Holds the thread-local override active for the lifetime of this Simulator.
254    /// Declared after `clock` so the clock is still alive when the guard drops
255    /// (though the guard does not reference the clock, this keeps drop order clear).
256    _clock_guard: ClockInjectionGuard,
257}
258
259impl Simulator {
260    /// Create a fault-free simulator seeded with `seed`.
261    ///
262    /// Immediately injects the simulated clock into `time::now()` on the
263    /// current thread.
264    pub fn new(seed: u64) -> Self {
265        let clock = SimulatedClock::new(0);
266        let guard = clock.inject();
267        Self {
268            seed,
269            clock,
270            faults: FaultInjector::new(FaultConfig::default(), seed),
271            _clock_guard: guard,
272        }
273    }
274
275    /// Create a simulator with the given seed and custom fault configuration.
276    ///
277    /// Immediately injects the simulated clock into `time::now()` on the
278    /// current thread.
279    pub fn with_seed_and_faults(seed: u64, config: FaultConfig) -> Self {
280        let clock = SimulatedClock::new(0);
281        let guard = clock.inject();
282        Self {
283            seed,
284            clock,
285            faults: FaultInjector::new(config, seed),
286            _clock_guard: guard,
287        }
288    }
289
290    /// The seed used to initialise this simulator.
291    pub fn seed(&self) -> u64 {
292        self.seed
293    }
294
295    /// Immutable access to the simulated clock.
296    pub fn clock(&self) -> &SimulatedClock {
297        &self.clock
298    }
299
300    /// Mutable access to the simulated clock.
301    pub fn clock_mut(&mut self) -> &mut SimulatedClock {
302        &mut self.clock
303    }
304
305    /// Reference to the fault injector.
306    pub fn faults(&self) -> &FaultInjector {
307        &self.faults
308    }
309
310    /// Advance simulated time forward by `delta_micros` microseconds.
311    ///
312    /// `time::now()` on the current thread will return the updated time
313    /// immediately.
314    pub fn advance_time_by(&mut self, delta_micros: i64) {
315        self.clock.advance_by(delta_micros);
316    }
317
318    /// Apply a clock discontinuity of `delta_micros` (positive = forward, negative = backward).
319    ///
320    /// `time::now()` on the current thread will return the updated time
321    /// immediately.
322    pub fn inject_clock_jump(&mut self, delta_micros: i64) {
323        self.clock.jump_by(delta_micros);
324    }
325
326    /// Check that `db` satisfies all verifiable temporal invariants.
327    ///
328    /// Returns an [`InvariantReport`] describing any violations found.
329    pub fn verify_temporal_invariants(&self, db: &AletheiaDB) -> InvariantReport {
330        verify_temporal_invariants_impl(db)
331    }
332}

334#[cfg(test)]
335mod tests {
336    use super::*;
337    use crate::core::temporal::time;

340    #[test]
341    fn new_simulator_seed_accessible() {
342        let s = Simulator::new(99);
343        assert_eq!(s.seed(), 99);
344    }
345
346    #[test]
347    fn simulator_auto_injects_clock() {
348        let sim = Simulator::new(0);
349        // Clock starts at 0 — time::now() must return the simulated value.
350        assert_eq!(time::now().wallclock(), 0);
351        drop(sim);
352    }
353
354    #[test]
355    fn advance_time_by_updates_time_now() {
356        let mut sim = Simulator::new(0);
357        sim.advance_time_by(500_000);
358        assert_eq!(time::now().wallclock(), 500_000);
359    }
360
361    #[test]
362    fn fresh_db_passes_invariants() {
363        let s = Simulator::new(0);
364        let (_tmp, db) = create_test_db().unwrap();
365        let r = s.verify_temporal_invariants(&db);
366        assert!(r.passed, "{:?}", r.violations);
367    }
368
369    #[test]
370    fn invariants_check_real_node_versions() {
371        use crate::PropertyMapBuilder;
372        use crate::api::WriteOps;
373
374        let s = Simulator::new(1_000_000); // start at 1s
375        let (_tmp, db) = create_test_db().unwrap();
376
377        let props = PropertyMapBuilder::new().build();
378        let node = db.create_node("Item", props).unwrap();
379        db.write(|tx: &mut crate::WriteTransaction| {
380            tx.update_node(node, PropertyMapBuilder::new().insert("v", 2_i64).build())
381        })
382        .unwrap();
383
384        let r = s.verify_temporal_invariants(&db);
385        assert_eq!(r.nodes_checked, 1);
386        assert!(r.passed, "{:?}", r.violations);
387    }
