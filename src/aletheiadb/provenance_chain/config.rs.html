1//! Configuration for the opt-in provenance hash chain (Issue #3351).
2//!
3//! Mirrors the shape of the other unified-config structs: `#[derive(Default)]`
4//! with the feature **disabled** by default so enabling the chain is an explicit
5//! opt-in and a disabled database keeps byte-identical behavior.
6
7use std::path::PathBuf;
8
9/// When the sidecar chain log is flushed to durable storage.
10///
11/// The default (`Batched`) keeps hashing/fsync off the commit critical path;
12/// `PerTransaction` fsyncs every appended transaction record for the strongest
13/// durability, and `Never` leaves flushing to the OS (fastest, weakest).
14#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
15#[cfg_attr(
16    feature = "serde",
17    derive(serde::Serialize, serde::Deserialize),
18    serde(rename_all = "snake_case")
19)]
20pub enum ChainFsyncMode {
21    /// fsync after every appended transaction record.
22    PerTransaction,
23    /// Do not fsync per append; rely on an explicit `flush()` / OS writeback.
24    #[default]
25    Batched,
26    /// Never fsync from the store; durability is entirely the OS's decision.
27    Never,
28}
29
30/// Opt-in configuration for the tamper-evident provenance hash chain.
31///
32/// Disabled by default. When `enabled` is `false`, no chain directory is touched
33/// and no hashing work is performed, so behavior and on-disk formats are
34/// byte-identical to a build without the feature.
35#[derive(Debug, Clone, PartialEq, Eq, Default)]
36#[cfg_attr(
37    feature = "serde",
38    derive(serde::Serialize, serde::Deserialize),
39    serde(default)
40)]
41pub struct ChainConfig {
42    /// Whether the provenance hash chain is active.
43    pub enabled: bool,
44    /// Flush policy for the sidecar log.
45    pub fsync: ChainFsyncMode,
46    /// Override for the chain directory. When `None`, the default
47    /// `<data_dir>/chain` is used.
48    pub dir: Option<PathBuf>,
49}
50
51impl ChainConfig {
52    /// Construct an enabled config with the default flush policy and directory.
53    #[must_use]
54    pub fn enabled() -> Self {
55        ChainConfig {
56            enabled: true,
57            ..Default::default()
58        }
59    }
60
61    /// Resolve the chain directory against a data directory: the explicit `dir`
62    /// override if set, otherwise `<data_dir>/chain`.
63    #[must_use]
64    pub fn resolve_dir(&self, data_dir: &std::path::Path) -> PathBuf {
65        self.dir.clone().unwrap_or_else(|| data_dir.join("chain"))
66    }
67}

69#[cfg(test)]
70mod tests {
71    use super::*;
72
73    #[test]
74    fn default_is_disabled_and_batched() {
75        let c = ChainConfig::default();
76        assert!(!c.enabled);
77        assert_eq!(c.fsync, ChainFsyncMode::Batched);
78        assert!(c.dir.is_none());
79    }
80
81    #[test]
82    fn resolve_dir_defaults_under_data_dir() {
83        let c = ChainConfig::default();
84        let resolved = c.resolve_dir(std::path::Path::new("/data"));
85        assert_eq!(resolved, PathBuf::from("/data/chain"));
86    }
87
88    #[test]
89    fn resolve_dir_honors_override() {
90        let c = ChainConfig {
91            enabled: true,
92            fsync: ChainFsyncMode::PerTransaction,
93            dir: Some(PathBuf::from("/custom/place")),
94        };
95        assert_eq!(
96            c.resolve_dir(std::path::Path::new("/data")),
97            PathBuf::from("/custom/place")
98        );
99    }
