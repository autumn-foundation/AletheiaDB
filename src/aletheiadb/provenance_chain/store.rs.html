1//! Append-only persistence for the provenance hash chain (Issue #3351).
2//!
3//! Layout under the chain directory:
4//! - `chain.log` — a header (`ALCHAIN1` magic + format-version `u32`) followed by
5//!   length-prefixed [`ChainTxRecord`]s (`u32` big-endian frame length, then the
6//!   record's binary payload).
7//! - `head` — an atomically-rewritten [`ChainHead`] checkpoint (`ALCHEAD1` magic +
8//!   version + payload), enabling fast startup without replaying the whole log.
9//!
10//! Torn-tail tolerance: [`ChainStore::load`] returns the longest valid prefix of
11//! records. A partially-written trailing frame (crash mid-append) is dropped
12//! rather than failing the load, mirroring WAL recovery.
13
14use std::fs::{File, OpenOptions};
15use std::io::{Read, Write};
16use std::path::{Path, PathBuf};
17use std::sync::Mutex;
18
19use super::config::ChainFsyncMode;
20use super::error::{ChainError, ChainResult};
21use super::record::{ChainHead, ChainTxRecord};
22
23const LOG_MAGIC: &[u8; 8] = b"ALCHAIN1";
24const HEAD_MAGIC: &[u8; 8] = b"ALCHEAD1";
25const FORMAT_VERSION: u32 = 1;
26const HEADER_LEN: usize = 12; // magic(8) + version(4)
27
28/// Append-only store for the sealed portion of a provenance hash chain.
29pub struct ChainStore {
30    log_path: PathBuf,
31    head_path: PathBuf,
32    fsync: ChainFsyncMode,
33    /// Append handle guarded for cross-thread appends.
34    append: Mutex<File>,
35}
36
37impl ChainStore {
38    /// Open (creating if necessary) the chain store under `dir`.
39    ///
40    /// Creates the directory and writes a fresh log header if the log does not
41    /// yet exist. If the log exists, its header is validated.
42    pub fn open(dir: impl AsRef<Path>, fsync: ChainFsyncMode) -> ChainResult<Self> {
43        let dir = dir.as_ref();
44        std::fs::create_dir_all(dir)?;
45        let log_path = dir.join("chain.log");
46        let head_path = dir.join("head");
47
48        if !log_path.exists() {
49            // Create with header.
50            let mut f = OpenOptions::new()
51                .create_new(true)
52                .write(true)
53                .open(&log_path)?;
54            let mut header = Vec::with_capacity(HEADER_LEN);
55            header.extend_from_slice(LOG_MAGIC);
56            header.extend_from_slice(&FORMAT_VERSION.to_be_bytes());
57            f.write_all(&header)?;
58            f.sync_all()?;
59        } else {
60            // Validate the existing header eagerly.
61            let mut f = File::open(&log_path)?;
62            let mut header = [0u8; HEADER_LEN];
63            f.read_exact(&mut header)
64                .map_err(|_| ChainError::HeaderMismatch("chain log too short for header".into()))?;
65            validate_header(&header)?;
66        }
67
68        let append = OpenOptions::new().append(true).open(&log_path)?;
69        Ok(ChainStore {
70            log_path,
71            head_path,
72            fsync,
73            append: Mutex::new(append),
74        })
75    }
76
77    /// Append one sealed transaction record, flushing per the configured mode.
78    pub fn append(&self, record: &ChainTxRecord) -> ChainResult<()> {
79        let payload = record.encode();
80        let len = u32::try_from(payload.len())
81            .map_err(|_| ChainError::BadFraming("record too large to frame".into()))?;
82
83        let mut framed = Vec::with_capacity(4 + payload.len());
84        framed.extend_from_slice(&len.to_be_bytes());
85        framed.extend_from_slice(&payload);
86
87        let mut guard = self
88            .append
89            .lock()
90            .map_err(|_| ChainError::Io("chain append lock poisoned".into()))?;
91        guard.write_all(&framed)?;
92        match self.fsync {
93            ChainFsyncMode::PerTransaction => guard.sync_all()?,
94            ChainFsyncMode::Batched | ChainFsyncMode::Never => {}
95        }
96        Ok(())
97    }
98
99    /// Atomically rewrite the entire log to `records` (header + framed records),
100    /// replacing the append handle with one positioned at the new tail.
101    ///
102    /// Used by the sealer's rare out-of-order path (Issue #3351 finding 4): when
103    /// a transaction arrives with a commit timestamp that sorts *before* an
104    /// already-sealed record, the in-memory suffix is re-folded and the log is
105    /// rewritten so the persisted order stays the canonical commit-timestamp
106    /// order. The common in-order case still uses [`append`](Self::append).
107    pub fn rewrite(&self, records: &[ChainTxRecord]) -> ChainResult<()> {
108        let mut payload = Vec::new();
109        payload.extend_from_slice(LOG_MAGIC);
110        payload.extend_from_slice(&FORMAT_VERSION.to_be_bytes());
111        for record in records {
112            let body = record.encode();
113            let len = u32::try_from(body.len())
114                .map_err(|_| ChainError::BadFraming("record too large to frame".into()))?;
115            payload.extend_from_slice(&len.to_be_bytes());
116            payload.extend_from_slice(&body);
117        }
118
119        let tmp_path = self.log_path.with_extension("log.tmp");
120        {
121            let mut f = OpenOptions::new()
122                .create(true)
123                .truncate(true)
124                .write(true)
125                .open(&tmp_path)?;
126            f.write_all(&payload)?;
127            f.sync_all()?;
128        }
129        std::fs::rename(&tmp_path, &self.log_path)?;
130
131        // Re-point the append handle at the rewritten log.
132        let new_append = OpenOptions::new().append(true).open(&self.log_path)?;
133        let mut guard = self
134            .append
135            .lock()
136            .map_err(|_| ChainError::Io("chain append lock poisoned".into()))?;
137        *guard = new_append;
138        if self.fsync == ChainFsyncMode::PerTransaction {
139            guard.sync_all()?;
140        }
141        Ok(())
142    }
143
144    /// Force any buffered/os-cached writes to durable storage. Useful for
145    /// [`ChainFsyncMode::Batched`] callers that batch appends and flush
146    /// explicitly.
147    pub fn flush(&self) -> ChainResult<()> {
148        let guard = self
149            .append
150            .lock()
151            .map_err(|_| ChainError::Io("chain append lock poisoned".into()))?;
152        guard.sync_all()?;
153        Ok(())
154    }

156    /// Load all valid records from the log, tolerating a torn trailing frame.
157    pub fn load(&self) -> ChainResult<Vec<ChainTxRecord>> {
158        let bytes = std::fs::read(&self.log_path)?;
159        if bytes.len() < HEADER_LEN {
160            return Err(ChainError::HeaderMismatch(
161                "chain log too short for header".into(),
162            ));
163        }
164        validate_header(&bytes[..HEADER_LEN])?;
165
166        let mut records = Vec::new();
167        let mut pos = HEADER_LEN;
168        while pos < bytes.len() {
169            // Need at least 4 bytes for the frame length.
170            if pos + 4 > bytes.len() {
171                // Torn length prefix at the tail — drop it.
172                break;
173            }
174            let len =
175                u32::from_be_bytes([bytes[pos], bytes[pos + 1], bytes[pos + 2], bytes[pos + 3]])
176                    as usize;
177            let body_start = pos + 4;
178            let body_end = match body_start.checked_add(len) {
179                Some(e) => e,
180                None => break, // absurd length; stop
181            };
182            if body_end > bytes.len() {
183                // Torn record body at the tail — drop it and keep the prefix.
184                break;
185            }
186            // A corrupt (non-tail) record is a hard error, not a torn tail.
187            let rec = ChainTxRecord::decode(&bytes[body_start..body_end])?;
188            records.push(rec);
189            pos = body_end;
190        }
191        Ok(records)
192    }

194    /// Atomically write the head checkpoint (temp file + rename), fsyncing the
195    /// temp file first.
196    pub fn write_head_checkpoint(&self, head: &ChainHead) -> ChainResult<()> {
197        let mut payload = Vec::new();
198        payload.extend_from_slice(HEAD_MAGIC);
199        payload.extend_from_slice(&FORMAT_VERSION.to_be_bytes());
200        payload.extend_from_slice(&head.encode());
201
202        let tmp_path = self.head_path.with_extension("tmp");
203        {
204            let mut f = OpenOptions::new()
205                .create(true)
206                .truncate(true)
207                .write(true)
208                .open(&tmp_path)?;
209            f.write_all(&payload)?;
210            f.sync_all()?;
211        }
212        std::fs::rename(&tmp_path, &self.head_path)?;
213        Ok(())
214    }
215
216    /// Read the head checkpoint, returning `None` if none has been written.
217    pub fn read_head_checkpoint(&self) -> ChainResult<Option<ChainHead>> {
218        let bytes = match std::fs::read(&self.head_path) {
219            Ok(b) => b,
220            Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(None),
221            Err(e) => return Err(e.into()),
222        };
223        if bytes.len() < HEADER_LEN {
224            return Err(ChainError::Corrupt("head checkpoint too short".into()));
225        }
226        if &bytes[..8] != HEAD_MAGIC {
227            return Err(ChainError::HeaderMismatch("bad head magic".into()));
228        }
229        let version = u32::from_be_bytes([bytes[8], bytes[9], bytes[10], bytes[11]]);
230        if version != FORMAT_VERSION {
231            return Err(ChainError::HeaderMismatch(format!(
232                "unsupported head version {version}"
233            )));
234        }
235        let head = ChainHead::decode(&bytes[HEADER_LEN..])?;
236        Ok(Some(head))
237    }
238}
239
240fn validate_header(header: &[u8]) -> ChainResult<()> {
241    if header.len() < HEADER_LEN {
242        return Err(ChainError::HeaderMismatch("header too short".into()));
243    }
244    if &header[..8] != LOG_MAGIC {
245        return Err(ChainError::HeaderMismatch("bad chain log magic".into()));
246    }
247    let version = u32::from_be_bytes([header[8], header[9], header[10], header[11]]);
248    if version != FORMAT_VERSION {
249        return Err(ChainError::HeaderMismatch(format!(
250            "unsupported chain log version {version}"
251        )));
252    }
253    Ok(())
254}

274    #[test]
275    fn append_and_load_round_trip() {
276        let dir = tempfile::tempdir().unwrap();
277        let store = ChainStore::open(dir.path(), ChainFsyncMode::PerTransaction).unwrap();
278        let records: Vec<_> = (1..=5).map(rec).collect();
279        for r in &records {
280            store.append(r).unwrap();
281        }
282        let loaded = store.load().unwrap();
283        assert_eq!(loaded, records);
284    }

301    fn torn_trailing_record_yields_valid_prefix() {
302        let dir = tempfile::tempdir().unwrap();
303        let store = ChainStore::open(dir.path(), ChainFsyncMode::Never).unwrap();
304        store.append(&rec(1)).unwrap();
305        store.append(&rec(2)).unwrap();
306        store.flush().unwrap();
307
308        // Corrupt the tail: append a partial frame (a length prefix promising more
309        // bytes than are present).
310        let log_path = dir.path().join("chain.log");
311        let mut f = OpenOptions::new().append(true).open(&log_path).unwrap();
312        f.write_all(&999u32.to_be_bytes()).unwrap(); // frame len = 999
313        f.write_all(&[0xAB, 0xCD]).unwrap(); // only 2 bytes of body
314        f.sync_all().unwrap();
315
316        let loaded = store.load().unwrap();
317        assert_eq!(loaded, vec![rec(1), rec(2)], "torn tail must be dropped");
318    }
