1//! Tamper-evident provenance hash chain (Issue #3351).
2//!
3//! An opt-in, self-contained sidecar that binds every committed transaction into
4//! a domain-separated SHA-256 hash chain over the database's recorded history, so
5//! that any post-hoc alteration of stored versions — editing a byte of a version,
6//! deleting/reordering/inserting a transaction, truncating the tail — is
7//! detectable, and (against an exported anchor) rollback and fork are provable.
8//!
9//! # Layers (this module is the pure core; no database wiring)
10//!
11//! - [`config`] — [`ChainConfig`] / [`ChainFsyncMode`], opt-in and disabled by
12//!   default.
13//! - [`canonical`] — the injective, deterministic canonical encoder over a
14//!   normalized [`VersionHashInput`], plus the folding primitives
15//!   ([`version_leaf`], [`tx_digest`], [`chain_step`], [`genesis_digest`]).
16//! - [`record`] — [`ChainTxRecord`] and [`ChainHead`] with binary + serde codecs.
17//! - [`store`] — the append-only [`ChainStore`] (header + length-framed records +
18//!   head checkpoint), tolerant of a torn trailing record.
19//! - [`verify`] — DB-independent verification: [`verify_full`],
20//!   [`verify_entity`], and [`verify_against_anchor`] over a [`VersionSource`].
21//!
22//! The chain unit is a committed transaction. Versions of one transaction share
23//! one commit timestamp, which is the grouping and ordering key. Each version is
24//! reduced to a leaf hash; the leaves fold into a per-transaction digest; each
25//! transaction digest chains onto the previous, seeded by an explicit genesis
26//! record (enable LSN + timestamp).
27

51#[cfg(test)]
52mod integration_tests {
53    //! End-to-end core flow: build a chain, persist it, reload, verify.
54    use super::*;
55    use crate::core::property::PropertyValue;
56    use std::collections::HashMap;
57
58    struct MapSource(HashMap<(EntityKind, u64, u64), VersionHashInput>);
59    impl VersionSource for MapSource {
60        fn fetch(&self, kind: EntityKind, id: u64, version_id: u64) -> Option<VersionHashInput> {
61            self.0.get(&(kind, id, version_id)).cloned()
62        }
63    }
64
65    fn version(id: u64, vid: u64, name: &str) -> VersionHashInput {
66        VersionHashInput {
67            entity_kind: EntityKind::Node,
68            entity_id: id,
69            version_id: vid,
70            prev_version_id: None,
71            label: "Person".to_string(),
72            source: None,
73            target: None,
74            valid_from: Some(10),
75            valid_to: None,
76            transaction_from: Some(10),
77            transaction_to: None,
78            is_current: true,
79            is_tombstone: false,
80            provenance: None,
81            properties: vec![("name".to_string(), PropertyValue::string(name))],
82        }
83    }
84
85    #[test]
86    fn persist_reload_and_verify_full_chain() {
87        let dir = tempfile::tempdir().unwrap();
88        let genesis = ChainHead::genesis(1, 0);
89        let mut src = MapSource(HashMap::new());
90
91        // Seal three transactions.
92        let store = ChainStore::open(dir.path(), ChainFsyncMode::PerTransaction).unwrap();
93        let mut prev = genesis.digest;
94        for seq in 1..=3u64 {
95            let v = version(seq, 1, &format!("n{seq}"));
96            src.0
97                .insert((v.entity_kind, v.entity_id, v.version_id), v.clone());
98            let leaf = version_leaf(&v);
99            let txd = tx_digest(1000 + seq as i64, 0, &[leaf]);
100            let digest = chain_step(&prev, &txd);
101            let rec = ChainTxRecord {
102                seq,
103                commit_ts: 1000 + seq as i64,
104                commit_ts_logical: 0,
105                tx_id: seq,
106                anchor_lsn: seq,
107                leaves: vec![leaf],
108                entity_refs: vec![(EntityKind::Node, seq, 1)],
109                digest,
110            };
111            store.append(&rec).unwrap();
112            prev = digest;
113        }
114        // Checkpoint the head.
115        let head = ChainHead {
116            seq: 3,
117            digest: prev,
118            commit_ts: 1003,
119            anchor_lsn: 3,
120            genesis_digest: genesis.genesis_digest,
121        };
122        store.write_head_checkpoint(&head).unwrap();
123
124        // Reopen, reload, verify.
125        let store2 = ChainStore::open(dir.path(), ChainFsyncMode::PerTransaction).unwrap();
126        let records = store2.load().unwrap();
127        assert_eq!(records.len(), 3);
128        let checkpoint = store2.read_head_checkpoint().unwrap().unwrap();
129        assert_eq!(checkpoint, head);
130
131        let result = verify_full(&records, &src, &genesis);
132        assert!(result.passed, "reason: {:?}", result.reason);
133        assert_eq!(result.head_seq, 3);
134        assert_eq!(result.head_digest_hex, to_hex(&head.digest));
135
136        // The reloaded head extends the exported anchor.
137        assert!(verify_against_anchor(&records, &genesis, &checkpoint).passed);
138    }
