1//! Injective, deterministic, domain-separated SHA-256 canonical encoder for a
2//! version, plus the folding primitives that build a transaction digest and
3//! chain step (Issue #3351).
4//!
5//! # Why a normalized [`VersionHashInput`]
6//!
7//! Both the seal path and the verify path hash the **same** normalized form so a
8//! digest is reproducible from reconstructed history alone. [`From`] conversions
9//! turn a live [`NodeVersion`]/[`EdgeVersion`] into a [`VersionHashInput`], and a
10//! [`VersionSource`](super::verify::VersionSource) reconstructs one at verify
11//! time; the encoder never sees the storage structs directly.
12//!
13//! # Injectivity
14//!
15//! The encoder is designed so that two *distinct logical values* can never
16//! produce identical bytes:
17//! - every value carries a 1-byte **type tag**,
18//! - every variable-length field is **length-prefixed** with a big-endian `u64`,
19//! - `Option`s emit a `0`/`1` presence byte,
20//! - properties are **sorted by key** before encoding so map iteration order does
21//!   not change the digest,
22//! - `Bytes` (tag 5) and `String` (tag 4) differ by tag, so the same underlying
23//!   bytes as a byte string vs. a UTF-8 string hash **differently** (the boundary
24//!   the audit encoder's `Bytes` fallback previously violated — see the sibling
25//!   fix in `src/audit/canonical.rs`).
26

41/// Whether a version belongs to a node or an edge. The tag byte is part of the
42/// canonical encoding so a node and an edge that happen to share an id and
43/// content still hash differently.
44#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
45#[cfg_attr(
46    feature = "serde",
47    derive(serde::Serialize, serde::Deserialize),
48    serde(rename_all = "snake_case")
49)]
50pub enum EntityKind {
51    /// A graph node.
52    Node,
53    /// A graph edge.
54    Edge,
55}
56
57impl EntityKind {
58    /// The 1-byte discriminator used in the canonical encoding and on disk.
59    #[must_use]
60    pub const fn tag(self) -> u8 {
61        match self {
62            EntityKind::Node => 0,
63            EntityKind::Edge => 1,
64        }
65    }
66
67    /// Reconstruct an [`EntityKind`] from its on-disk tag byte.
68    #[must_use]
69    pub const fn from_tag(tag: u8) -> Option<Self> {
70        match tag {
71            0 => Some(EntityKind::Node),
72            1 => Some(EntityKind::Edge),
73            _ => None,
74        }
75    }
76}

105/// Normalized, hashable form of a single version.
106///
107/// This is the ONLY structure the canonical encoder reads, so seal-time and
108/// verify-time necessarily hash identical bytes. Timestamps are microseconds
109/// since the Unix epoch; a `None` end means an open (still-current) interval.
110#[derive(Debug, Clone, PartialEq)]
111pub struct VersionHashInput {
112    /// Whether this is a node or edge version.
113    pub entity_kind: EntityKind,
114    /// The owning entity's numeric id.
115    pub entity_id: u64,
116    /// This version's id.
117    pub version_id: u64,
118    /// The previous version's id in the chain, if any.
119    pub prev_version_id: Option<u64>,
120    /// The entity's label at this version (node label / edge type), resolved to
121    /// a string. Bound into the leaf so a relabel is tamper-evident (Issue
122    /// #3351: leaf previously omitted the label).
123    pub label: String,
124    /// For an edge, the source node id; `None` for a node. Bound so an edge
125    /// endpoint rewire is tamper-evident (Issue #3351).
126    pub source: Option<u64>,
127    /// For an edge, the target node id; `None` for a node. Bound so an edge
128    /// endpoint rewire is tamper-evident (Issue #3351).
129    pub target: Option<u64>,
130    /// Valid-time start (micros); `None` only for a degenerate open start.
131    pub valid_from: Option<i64>,
132    /// Valid-time end (micros); `None` == open.
133    ///
134    /// A *live* version's valid end is closed by a later supersession, so it is
135    /// normalized out of the leaf (see `db::chain_source::normalize_immutable`).
136    /// A **born-closed terminal** version (delete tombstone or retraction) has an
137    /// immutable valid end (never re-closed by supersession because
138    /// `close_previous_version_intervals` only closes *open* valid intervals), so
139    /// for those it is retained and bound — closing the un-retract / re-validate
140    /// tamper hole (Issue #3351 finding 2).
141    pub valid_to: Option<i64>,
142    /// Transaction-time start (micros).
143    pub transaction_from: Option<i64>,
144    /// Transaction-time end (micros); `None` == open. Always normalized out of
145    /// the leaf (a supersession closes it), retained here only for structural
146    /// consistency checks.
147    pub transaction_to: Option<i64>,
148    /// Whether this version is current in both temporal dimensions. Normalized
149    /// out of the leaf (flips on supersession).
150    pub is_current: bool,
151    /// Whether this version is a delete tombstone (its valid-time interval is
152    /// empty — `start == end`). Immutable after creation and bound into the leaf
153    /// so an un-delete (re-opening the tombstone interval) is tamper-evident
154    /// (Issue #3351 finding 2).
155    pub is_tombstone: bool,
156    /// Write-time provenance, if any.
157    pub provenance: Option<ProvenanceHashInput>,
158    /// Materialized properties `(key, value)`. Encoding sorts by key, so the
159    /// caller may supply them in any order.
160    pub properties: Vec<(String, PropertyValue)>,
161}

163/// Extract `(key, value)` property pairs from a version's [`VersionData`].
164///
165/// For an anchor this is the full property snapshot. For a delta only the
166/// changed (non-removed) properties are available from the version in isolation;
167/// callers that need full-state leaves (the seal path) should reconstruct the
168/// state first and build the [`VersionHashInput`] from that. This helper keeps
169/// the [`From`] conversions total and lossless for anchors.
170fn properties_from_data(data: &VersionData) -> Vec<(String, PropertyValue)> {
171    use crate::core::interning::GLOBAL_INTERNER;
172    let mut out: Vec<(String, PropertyValue)> = Vec::new();
173    match data {
174        VersionData::Anchor { properties, .. } => {
175            for (key, value) in properties.iter() {
176                if let Some(name) = GLOBAL_INTERNER.resolve_with(*key, |s| s.to_string()) {
177                    out.push((name, value.clone()));
178                }
179            }
180        }
181        VersionData::Delta { delta } => {
182            for (key, value) in &delta.changed {
183                if let Some(name) = GLOBAL_INTERNER.resolve_with(*key, |s| s.to_string()) {
184                    out.push((name, value.clone()));
185                }
186            }
187        }
188    }
189    out
190}

192/// Resolve an [`InternedString`](crate::core::interning::InternedString) label to
193/// an owned string, defaulting to empty when the intern table cannot resolve it
194/// (a resolvable label always round-trips; the empty default keeps the leaf total).
195fn resolve_label(label: crate::core::interning::InternedString) -> String {
196    use crate::core::interning::GLOBAL_INTERNER;
197    GLOBAL_INTERNER
198        .resolve_with(label, |s| s.to_string())
199        .unwrap_or_default()
200}

202/// Whether a bi-temporal interval's valid-time dimension is empty
203/// (`start == end`), which is exactly how a delete tombstone is encoded (see
204/// `apply::create_temporal_interval` / `HistoricalStorage::add_node_version`).
205fn is_tombstone_interval(interval: &crate::core::temporal::BiTemporalInterval) -> bool {
206    let vt = interval.valid_time();
207    vt.start() == vt.end()
208}

256/// Decompose a bi-temporal interval into `(valid_from, valid_to, tx_from,
257/// tx_to)` micros, mapping the open-interval sentinel (`TIMESTAMP_MAX`) to
258/// `None`.
259fn temporal_micros(
260    interval: &crate::core::temporal::BiTemporalInterval,
261) -> (Option<i64>, Option<i64>, Option<i64>, Option<i64>) {
262    use crate::core::temporal::TIMESTAMP_MAX;
263    let end_micros = |ts: crate::core::temporal::Timestamp| -> Option<i64> {
264        if ts == TIMESTAMP_MAX {
265            None
266        } else {
267            Some(ts.wallclock())
268        }
269    };
270    let vt = interval.valid_time();
271    let tt = interval.transaction_time();
272    (
273        Some(vt.start().wallclock()),
274        end_micros(vt.end()),
275        Some(tt.start().wallclock()),
276        end_micros(tt.end()),
277    )
278}

280/// A tiny append-only canonical byte writer (mirrors `audit::canonical::Canon`).
281#[derive(Default)]
282struct Canon {
283    buf: Vec<u8>,
284}
285
286impl Canon {
287    fn tag(&mut self, t: u8) {
288        self.buf.push(t);
289    }
290    fn u32(&mut self, v: u32) {
291        self.buf.extend_from_slice(&v.to_be_bytes());
292    }
293    fn u64(&mut self, v: u64) {
294        self.buf.extend_from_slice(&v.to_be_bytes());
295    }
296    fn i64(&mut self, v: i64) {
297        self.buf.extend_from_slice(&v.to_be_bytes());
298    }
299    fn f32bits(&mut self, v: f32) {
300        self.buf.extend_from_slice(&v.to_bits().to_be_bytes());
301    }
302    fn f64bits(&mut self, v: f64) {
303        self.buf.extend_from_slice(&v.to_bits().to_be_bytes());
304    }
305    fn bool(&mut self, v: bool) {
306        self.buf.push(u8::from(v));
307    }
308    fn bytes(&mut self, b: &[u8]) {
309        self.u64(b.len() as u64);
310        self.buf.extend_from_slice(b);
311    }
312    fn str(&mut self, s: &str) {
313        self.bytes(s.as_bytes());
314    }
315    fn opt_u64(&mut self, v: Option<u64>) {
316        match v {
317            None => self.buf.push(0),
318            Some(x) => {
319                self.buf.push(1);
320                self.u64(x);
321            }
322        }
323    }
324    fn opt_i64(&mut self, v: Option<i64>) {
325        match v {
326            None => self.buf.push(0),
327            Some(x) => {
328                self.buf.push(1);
329                self.i64(x);
330            }
331        }
332    }
333    fn opt_str(&mut self, v: Option<&str>) {
334        match v {
335            None => self.buf.push(0),
336            Some(s) => {
337                self.buf.push(1);
338                self.str(s);
339            }
340        }
341    }
342    fn opt_f64(&mut self, v: Option<f64>) {
343        match v {
344            None => self.buf.push(0),
345            Some(x) => {
346                self.buf.push(1);
347                self.f64bits(x);
348            }
349        }
350    }
351
352    /// Encode a property value with a per-variant type tag. Every variable-length
353    /// payload is length-prefixed, so the encoding is injective across variants.
354    fn value(&mut self, v: &PropertyValue) {
355        match v {
356            PropertyValue::Null => self.tag(0),
357            PropertyValue::Bool(b) => {
358                self.tag(1);
359                self.bool(*b);
360            }
361            PropertyValue::Int(i) => {
362                self.tag(2);
363                self.i64(*i);
364            }
365            PropertyValue::Float(f) => {
366                self.tag(3);
367                self.f64bits(*f);
368            }
369            PropertyValue::String(s) => {
370                self.tag(4);
371                self.str(s);
372            }
373            PropertyValue::Bytes(b) => {
374                // Distinct tag from String: identical underlying bytes as a byte
375                // string vs. a UTF-8 string hash differently.
376                self.tag(5);
377                self.bytes(b);
378            }
379            PropertyValue::Array(items) => {
380                self.tag(6);
381                self.u64(items.len() as u64);
382                for item in items.iter() {
383                    self.value(item);
384                }
385            }
386            PropertyValue::Vector(vec) => {
387                self.tag(7);
388                self.u64(vec.len() as u64);
389                for f in vec.iter() {
390                    self.f32bits(*f);
391                }
392            }
393            PropertyValue::SparseVector(sparse) => {
394                self.tag(8);
395                self.u64(sparse.dimension() as u64);
396                let indices = sparse.indices();
397                self.u64(indices.len() as u64);
398                for i in indices {
399                    self.u32(*i);
400                }
401                let values = sparse.values();
402                self.u64(values.len() as u64);
403                for f in values {
404                    self.f32bits(*f);
405                }
406            }
407        }
408    }
409
410    fn provenance(&mut self, p: &Option<ProvenanceHashInput>) {
411        match p {
412            None => self.buf.push(0),
413            Some(pr) => {
414                self.buf.push(1);
415                self.opt_str(pr.source.as_deref());
416                self.opt_f64(pr.confidence);
417                self.opt_str(pr.note.as_deref());
418                self.opt_str(pr.correlation_id.as_deref());
419                self.opt_str(pr.principal.as_deref());
420            }
421        }
422    }
423}

425/// Deterministic canonical bytes for a version (independent of property order).
426fn version_canonical(input: &VersionHashInput) -> Vec<u8> {
427    let mut c = Canon::default();
428    c.tag(input.entity_kind.tag());
429    c.u64(input.entity_id);
430    c.u64(input.version_id);
431    c.opt_u64(input.prev_version_id);
432    // Label (length-prefixed str) — binds node label / edge type so a relabel
433    // changes the leaf (Issue #3351 finding 1).
434    c.str(&input.label);
435    // Edge endpoints (presence-gated u64) — bind source/target so an endpoint
436    // rewire changes the leaf; a node emits two `0` presence bytes (Issue #3351
437    // finding 1).
438    c.opt_u64(input.source);
439    c.opt_u64(input.target);
440    c.opt_i64(input.valid_from);
441    c.opt_i64(input.valid_to);
442    c.opt_i64(input.transaction_from);
443    c.opt_i64(input.transaction_to);
444    c.bool(input.is_current);
445    // Tombstone discriminator — binds the delete flag so re-opening a tombstone
446    // interval (un-delete) changes the leaf (Issue #3351 finding 2).
447    c.bool(input.is_tombstone);
448    c.provenance(&input.provenance);
449
450    // Sort properties by key so map iteration order is not signed content.
451    let mut props: Vec<&(String, PropertyValue)> = input.properties.iter().collect();
452    props.sort_by(|a, b| a.0.cmp(&b.0));
453    c.u64(props.len() as u64);
454    for (key, value) in props {
455        c.str(key);
456        c.value(value);
457    }
458    c.buf
459}

461/// SHA-256 of `domain || payload`.
462fn hash_domain(domain: &[u8], payload: &[u8]) -> [u8; 32] {
463    let mut h = Sha256::new();
464    h.update(domain);
465    h.update(payload);
466    h.finalize().into()
467}

469/// The per-version leaf hash: `SHA256(LEAF_DOMAIN || canon(version))`.
470#[must_use]
471pub fn version_leaf(input: &VersionHashInput) -> [u8; 32] {
472    hash_domain(CHAIN_LEAF_DOMAIN, &version_canonical(input))
473}

475/// The per-transaction digest, order- and count-dependent over its leaves:
476/// `SHA256(TX_DOMAIN || commit_ts_micros || commit_ts_logical || len || leaf_0
477/// || .. || leaf_n)`.
478///
479/// # Determinism across live-seal and post-crash rebuild (Issue #3351 finding 4)
480///
481/// The digest binds the **full HLC commit timestamp** (`commit_ts_micros` *and*
482/// the logical counter), NOT the transaction id. The transaction id survives as
483/// record metadata only. This is deliberate:
484/// - the live seal path knows the real `tx_id`, but a post-crash rebuild folds
485///   from replayed history where the original `tx_id` is not recoverable — so
486///   binding `tx_id` made a rebuilt chain diverge from a pre-crash anchor;
487/// - the commit timestamp plus the (sorted) leaves uniquely identify a
488///   committed transaction, and the logical counter keeps two commits that
489///   share a wallclock microsecond distinct in *both* the live and rebuild
490///   paths.
491#[must_use]
492pub fn tx_digest(commit_ts_micros: i64, commit_ts_logical: u32, leaves: &[[u8; 32]]) -> [u8; 32] {
493    let mut h = Sha256::new();
494    h.update(CHAIN_TX_DOMAIN);
495    h.update(commit_ts_micros.to_be_bytes());
496    h.update(commit_ts_logical.to_be_bytes());
497    h.update((leaves.len() as u64).to_be_bytes());
498    for leaf in leaves {
499        h.update(leaf);
500    }
501    h.finalize().into()
502}
503
504/// One chain step: `SHA256(NODE_DOMAIN || prev || tx_digest)`.
505#[must_use]
506pub fn chain_step(prev: &[u8; 32], tx_digest: &[u8; 32]) -> [u8; 32] {
507    let mut h = Sha256::new();
508    h.update(CHAIN_NODE_DOMAIN);
509    h.update(prev);
510    h.update(tx_digest);
511    h.finalize().into()
512}
513
514/// The genesis digest seeding a chain: `SHA256(GENESIS_DOMAIN || enable_lsn ||
515/// enable_ts)`.
516#[must_use]
517pub fn genesis_digest(enable_lsn: u64, enable_ts: i64) -> [u8; 32] {
518    let mut h = Sha256::new();
519    h.update(CHAIN_GENESIS_DOMAIN);
520    h.update(enable_lsn.to_be_bytes());
521    h.update(enable_ts.to_be_bytes());
522    h.finalize().into()
523}

531/// Decode a 64-char lowercase-hex string into a 32-byte digest.
532#[must_use]
533pub fn from_hex_32(s: &str) -> Option<[u8; 32]> {
534    let v = hex::decode(s)?;
535    if v.len() != 32 {
536        return None;
537    }
538    let mut out = [0u8; 32];
539    out.copy_from_slice(&v);
540    Some(out)
541}

676    #[test]
677    fn tx_digest_is_order_and_count_dependent() {
678        let l1 = version_leaf(&base());
679        let mut other = base();
680        other.version_id = 2;
681        let l2 = version_leaf(&other);
682
683        let d_12 = tx_digest(500, 7, &[l1, l2]);
684        let d_21 = tx_digest(500, 7, &[l2, l1]);
685        let d_1 = tx_digest(500, 7, &[l1]);
686        assert_ne!(d_12, d_21, "leaf order must matter");
687        assert_ne!(d_12, d_1, "leaf count must matter");
688
689        // The full HLC commit timestamp (wallclock micros AND logical counter)
690        // is bound into the digest; the tx id is not (Issue #3351 finding 4).
691        assert_ne!(d_12, tx_digest(501, 7, &[l1, l2]));
692        assert_ne!(d_12, tx_digest(500, 8, &[l1, l2]));
693    }
