1//! HTTP error type for AletheiaDB JSON API.

3//! Provides a single error enum with `IntoResponse` so handlers return

14/// A per-query resource limit was exceeded at the HTTP layer (Issue #3368).
15///
16/// Carries enough structure to render the `{code, retriable, details}` body:
17/// the [`LimitDimension`], the effective `limit`, (for the byte/row caps) how
18/// much was `consumed`, and whether a retry could usefully succeed. The HTTP
19/// status is derived from the dimension:
20///
21/// - [`LimitDimension::WallClockTimeout`] → `429`
22/// - [`LimitDimension::ResultBytes`] / [`LimitDimension::ResultRows`] → `413`
23///
24/// The [`retriable`](Self::retriable) flag is carried explicitly rather than
25/// derived from the dimension: a wall-clock timeout is transient and normally
26/// `retriable: true`, **but** a timeout on a *write*-class operation is
27/// `retriable: false` — the write may already have committed on the blocking
28/// pool, so a naive retry would duplicate it (Issue #3368 review). Result-row
29/// and result-byte caps are always `retriable: false`, and (post-review) only
30/// ever apply to read-class operations.
31#[derive(Debug, Clone, PartialEq, Eq)]
32pub struct ResourceLimitExceeded {
33    /// Which dimension's cap was hit.
34    pub dimension: LimitDimension,
35    /// The effective limit that was exceeded (ms for timeout, else count/bytes).
36    pub limit: u64,
37    /// How much was consumed, when known (row count / byte size).
38    pub consumed: Option<u64>,
39    /// Whether a retry could usefully succeed. `true` only for a read-class
40    /// wall-clock timeout; `false` for a write-class timeout and for every
41    /// row/byte cap.
42    pub retriable: bool,
43}
44
45impl std::fmt::Display for ResourceLimitExceeded {
46    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
47        match self.dimension {
48            LimitDimension::WallClockTimeout => {
49                write!(
50                    f,
51                    "query exceeded the wall-clock timeout of {} ms",
52                    self.limit
53                )
54            }
55            LimitDimension::ResultRows => match self.consumed {
56                Some(c) => write!(
57                    f,
58                    "query result exceeded the row limit of {} (produced {})",
59                    self.limit, c
60                ),
61                None => write!(f, "query result exceeded the row limit of {}", self.limit),
62            },
63            LimitDimension::ResultBytes => match self.consumed {
64                Some(c) => write!(
65                    f,
66                    "query response exceeded the byte limit of {} (serialized {})",
67                    self.limit, c
68                ),
69                None => write!(
70                    f,
71                    "query response exceeded the byte limit of {} bytes",
72                    self.limit
73                ),
74            },
75        }
76    }
77}

79/// Errors raised by AletheiaDB's HTTP handlers.
80#[derive(Debug, thiserror::Error)]
81pub enum AletheiaHttpError {
82    /// Request validation failed (bad IDs, pagination overflow, malformed input).
83    #[error("{0}")]
84    BadRequest(String),
85
86    /// A resource lookup that the client reasonably expected to exist failed.
87    #[error("{0}")]
88    NotFound(String),
89
90    /// Query syntax / parse errors surfaced from the AQL parser.
91    #[error("{0}")]
92    QueryParse(String),
93
94    /// Any unexpected backend or serialization failure.
95    #[error("{0}")]
96    Internal(String),
97
98    /// Shared application state was not installed at startup.
99    ///
100    /// Treated as HTTP 500 — this is a boot-time invariant; reaching it in a
101    /// running server indicates the app was built incorrectly.
102    #[error("application state not installed")]
103    StateMissing,
104
105    /// Authentication failed (missing/malformed/unknown/revoked credential).
106    ///
107    /// Deliberately a unit variant: every authentication failure produces a
108    /// byte-identical 401 body so callers can't distinguish "no such key"
109    /// from "revoked key", and the presented credential is never echoed.
110    #[error("authentication required")]
111    Unauthorized,
112
113    /// The authenticated principal's role does not allow this operation.
114    ///
115    /// Carries the operation's required [`AccessClass`] and the principal's
116    /// [`Role`] so the error body can surface a structured
117    /// `details: {required_class, principal_role}` block (matching the MCP
118    /// surface, Issue #3234). The `Display` free text is byte-identical to the
119    /// legacy `"role '{role}' does not permit {class} access"` message.
120    #[error("role '{principal_role}' does not permit {required_class} access")]
121    PermissionDenied {
122        /// The access class the operation required.
123        required_class: AccessClass,
124        /// The authenticated principal's role.
125        principal_role: Role,
126    },
127
128    /// A per-query resource limit (timeout / rows / bytes) was exceeded
129    /// (Issue #3368). Renders `429` (timeout) or `413` (rows/bytes).
130    #[error("{0}")]
131    ResourceLimitExceeded(ResourceLimitExceeded),
132
133    /// A per-call limit override exceeded the operator ceiling (Issue #3368).
134    /// Renders `422 INVALID_ARGUMENT`.
135    #[error(
136        "limit override for '{}' ({}) exceeds the maximum allowed ({})",
137        .0.dimension.as_str(), .0.requested, .0.ceiling
138    )]
139    InvalidLimitOverride(LimitOverrideError),
140
141    /// The bounded in-flight-query worker pool is full (Issue #3368 DoS guard):
142    /// a new *timed* `/query` was rejected instead of spawning yet another
143    /// detached, non-cancellable worker. Renders `503 UNAVAILABLE`, `retriable:
144    /// true` — the caller should back off and retry once load subsides. Carries
145    /// the configured `cap` so the caller can see the operative limit.
146    #[error(
147        "server is at its cap of {cap} concurrently executing timed queries; retry with backoff"
148    )]
149    InFlightCapacityExceeded {
150        /// The configured `max_in_flight_queries` cap that was reached.
151        cap: usize,
152    },
153
154    /// A per-principal changefeed subscription quota was exceeded (Issue #3678).
155    /// Renders `429 RESOURCE_EXHAUSTED`, `retriable: true`, with
156    /// `details {principal, current, limit}` — byte-shape-identical to the MCP
157    /// surface's classification of [`StorageError::PrincipalQuotaExceeded`].
158    #[error(
159        "principal '{principal}' has reached its changefeed subscription quota (current={current}, limit={limit})"
160    )]
161    PrincipalQuotaExceeded {
162        /// The principal id whose quota was exceeded.
163        principal: String,
164        /// The principal's current live-subscription count.
165        current: usize,
166        /// The principal's configured maximum.
167        limit: usize,
168    },
169
170    /// A db-layer error pre-classified into the unified #3234 envelope, so the
171    /// HTTP body carries the same `code`/`retriable`/`details` as the MCP
172    /// surface. Built via [`Self::from_db_error`], which classifies the
173    /// schema-constraint (Issue #3378), uniqueness, and transaction
174    /// conflict/precondition classes; every other db error keeps its prior HTTP
175    /// mapping (a 400 `BadRequest`).
176    #[error("{message}")]
177    Structured {
178        /// The #3234 code string (e.g. `CONSTRAINT_VIOLATION`,
179        /// `FAILED_PRECONDITION`, `CONFLICT`, `UNAVAILABLE`).
180        code: &'static str,
181        /// HTTP status derived from the code (`409` / `412` / `503` / …).
182        status: StatusCode,
183        /// Whether a fresh attempt could usefully succeed. `true` only for the
184        /// transient conflict / clock-skew classes; `false` for the caller-fault
185        /// constraint / precondition classes.
186        retriable: bool,
187        /// Free-text message: the db error's `Display` (preserved verbatim).
188        message: String,
189        /// Structured, machine-readable metadata (shared with the MCP surface).
190        details: Option<Value>,
191    },
192}

194impl AletheiaHttpError {
195    /// Classify a db-layer [`Error`](crate::core::error::Error) surfaced from a
196    /// write handler into the HTTP error envelope, using the same #3234 code
197    /// vocabulary as the MCP surface (`src/mcp/error.rs`).
198    ///
199    /// Three classes are pre-classified into [`Self::Structured`] so the
200    /// response carries the same `code`/`retriable`/`details` as the MCP
201    /// surface:
202    ///
203    /// - **Constraint violations** (Issue #3378 + uniqueness): a
204    ///   type/required-key/unique violation is a declared constraint rejecting
205    ///   *this* write → `CONSTRAINT_VIOLATION` (`409`); a non-conforming enable
206    ///   or a pre-existing duplicate is a state problem the caller must fix
207    ///   first → `FAILED_PRECONDITION` (`412`). All non-retriable.
208    /// - **Transaction conflicts / preconditions**: a serialization failure /
209    ///   write conflict / abort is transient → `CONFLICT` (`409`, retriable); a
210    ///   validation failure (e.g. the #3416 concurrent-orphan guard), invalid
211    ///   state, already-committed, or lost CAS is a caller-fault precondition →
212    ///   `FAILED_PRECONDITION` (`412`, non-retriable); clock skew heals on its
213    ///   own → `UNAVAILABLE` (`503`, retriable).
214    ///
215    /// Everything else preserves the prior write-path mapping (`400 BadRequest`
216    /// with the free-text message) — critically, this keeps genuinely bad input
217    /// (a malformed property map wrapped as [`Error::Other`], an invalid id) at
218    /// `400`, so extending the classifier never over-broadens a real
219    /// bad-request into a constraint or internal error.
220    ///
221    /// [`Error::Other`]: crate::core::error::Error::Other
222    #[must_use]
223    pub(crate) fn from_db_error(e: &crate::core::error::Error) -> Self {
224        use crate::core::error::ConstraintError as CE;
225        use crate::core::error::Error as E;
226        use crate::core::error::TransactionError as TE;
227
228        // --- Constraint violations (Issue #3378 schema + uniqueness). ---
229        if let Some(ce) = e.as_constraint() {
230            return match ce {
231                // A declared type/required-key violation rejects *this* write.
232                CE::TypeViolation { .. } | CE::MissingRequiredKey { .. } => Self::Structured {
233                    code: "CONSTRAINT_VIOLATION",
234                    status: StatusCode::CONFLICT,
235                    retriable: false,
236                    message: e.to_string(),
237                    details: ce.structured_details(),
238                },
239                // A uniqueness violation is likewise CONSTRAINT_VIOLATION. Its
240                // `structured_details()` is `None`, so build the machine-readable
241                // metadata here, mirroring the MCP surface's `db_error` call site
242                // (`label`/`property`/`value`/`existing_node_id`).
243                CE::UniqueViolation {
244                    label,
245                    property,
246                    value,
247                    existing_node_id,
248                } => Self::Structured {
249                    code: "CONSTRAINT_VIOLATION",
250                    status: StatusCode::CONFLICT,
251                    retriable: false,
252                    message: e.to_string(),
253                    details: Some(json!({
254                        "label": label,
255                        "property": property,
256                        "value": value,
257                        "existing_node_id": existing_node_id.as_u64(),
258                    })),
259                },
260                // A non-conforming enable or a pre-existing duplicate is a state
261                // problem the caller must fix first.
262                CE::NonConformingOnEnable { .. } | CE::DuplicateOnEnable { .. } => {
263                    Self::Structured {
264                        code: "FAILED_PRECONDITION",
265                        status: StatusCode::PRECONDITION_FAILED,
266                        retriable: false,
267                        message: e.to_string(),
268                        // `None` for DuplicateOnEnable (no shared structured details).
269                        details: ce.structured_details(),
270                    }
271                }
272                // An unsupported key type in the *enable request itself* is a
273                // genuine bad request, not a constraint rejecting a write.
274                CE::UnsupportedKeyType { .. } => Self::BadRequest(e.to_string()),
275            };
276        }
277
278        // --- Transaction conflicts / preconditions (mirrors the MCP
279        // classifier in src/mcp/error.rs::classify_transaction_error). ---
280        if let E::Transaction(te) = e {
281            return match te {
282                // Concurrency conflicts a fresh attempt can win.
283                TE::SerializationFailure { .. } | TE::WriteConflict { .. } | TE::Aborted { .. } => {
284                    Self::Structured {
285                        code: "CONFLICT",
286                        status: StatusCode::CONFLICT,
287                        retriable: true,
288                        message: e.to_string(),
289                        details: None,
290                    }
291                }
292                // Well-formed but forbidden by current state (e.g. the #3416
293                // concurrent-orphan validation guard, a lost compare-and-set):
294                // retrying the identical call cannot succeed.
295                TE::ValidationFailed { .. }
296                | TE::InvalidState { .. }
297                | TE::AlreadyCommitted { .. }
298                | TE::CasMismatch { .. } => Self::Structured {
299                    code: "FAILED_PRECONDITION",
300                    status: StatusCode::PRECONDITION_FAILED,
301                    retriable: false,
302                    message: e.to_string(),
303                    details: None,
304                },
305                // A clock-adjacent hiccup heals on its own at a later tick.
306                TE::ClockSkew { .. } => Self::Structured {
307                    code: "UNAVAILABLE",
308                    status: StatusCode::SERVICE_UNAVAILABLE,
309                    retriable: true,
310                    message: e.to_string(),
311                    details: None,
312                },
313                // Genuine internal failures (commit/rollback/poisoned lock).
314                TE::CommitFailed { .. } | TE::RollbackFailed { .. } | TE::LockPoisoned { .. } => {
315                    Self::Internal(e.to_string())
316                }
317            };
318        }
319
320        // --- Storage capacity exhaustion (configurable interner cap). ---
321        // A string-interner capacity breach is well-formed input the system
322        // refuses in its current state; retrying is futile until the operator
323        // raises the cap and restarts → FAILED_PRECONDITION (412, non-retriable),
324        // with the same actionable message + `{resource,current,limit}` details
325        // the MCP surface emits (byte-shape-identical envelope).
326        if let E::Storage(crate::core::error::StorageError::CapacityExceeded {
327            resource,
328            current,
329            limit,
330        }) = e
331        {
332            let message = if resource.contains("interner") {
333                format!(
334                    "String interner at capacity ({current}/{limit}). Raise \
335                     `persistence.max_interned_strings` above {limit} and restart to intern more \
336                     unique strings. No data is lost — the WAL is the source of truth."
337                )
338            } else {
339                e.to_string()
340            };
341            return Self::Structured {
342                code: "FAILED_PRECONDITION",
343                status: StatusCode::PRECONDITION_FAILED,
344                retriable: false,
345                message,
346                details: Some(json!({
347                    "resource": resource,
348                    "current": current,
349                    "limit": limit,
350                })),
351            };
352        }
353
354        // Every other db error keeps the prior write-path mapping (400
355        // BadRequest), preserving the contract that genuine bad input stays 400.
356        Self::BadRequest(e.to_string())
357    }

359    fn status(&self) -> StatusCode {
360        match self {
361            Self::BadRequest(_) | Self::QueryParse(_) => StatusCode::BAD_REQUEST,
362            Self::NotFound(_) => StatusCode::NOT_FOUND,
363            Self::Internal(_) | Self::StateMissing => StatusCode::INTERNAL_SERVER_ERROR,
364            Self::Unauthorized => StatusCode::UNAUTHORIZED,
365            Self::PermissionDenied { .. } => StatusCode::FORBIDDEN,
366            Self::InvalidLimitOverride(_) => StatusCode::UNPROCESSABLE_ENTITY,
367            // A pre-classified db error carries its own status (409 / 412 / 503).
368            Self::Structured { status, .. } => *status,
369            // Transient overload → 503 Service Unavailable (retriable).
370            Self::InFlightCapacityExceeded { .. } => StatusCode::SERVICE_UNAVAILABLE,
371            // A per-principal quota breach → 429 Too Many Requests (retriable).
372            Self::PrincipalQuotaExceeded { .. } => StatusCode::TOO_MANY_REQUESTS,
373            Self::ResourceLimitExceeded(e) => match e.dimension {
374                // Timeout is transient → 429 Too Many Requests.
375                LimitDimension::WallClockTimeout => StatusCode::TOO_MANY_REQUESTS,
376                // Result too large → 413 Payload Too Large.
377                LimitDimension::ResultRows | LimitDimension::ResultBytes => {
378                    StatusCode::PAYLOAD_TOO_LARGE
379                }
380            },
381        }
382    }
383
384    /// Whether a caller may usefully retry. `true` only for transient failure
385    /// classes: a read-class wall-clock timeout (carried on the
386    /// [`ResourceLimitExceeded`] itself) and a full in-flight worker pool
387    /// (transient overload). A write-class timeout is not retriable (a
388    /// committed write must not be duplicated), and every caller-fault class
389    /// (bad-request, not-found, auth, override, row/byte caps) is `false`.
390    ///
391    /// Mirrors the MCP surface's explicit per-error classification (Issue
392    /// #3234): the nested error envelope always carries a concrete `retriable`
393    /// boolean.
394    fn retriable(&self) -> bool {
395        match self {
396            Self::ResourceLimitExceeded(e) => e.retriable,
397            // Transient overload: backing off and retrying can succeed.
398            Self::InFlightCapacityExceeded { .. } => true,
399            // A fairness quota: another of this principal's subscriptions may
400            // drop, so a backed-off retry can succeed (Issue #3678).
401            Self::PrincipalQuotaExceeded { .. } => true,
402            // A pre-classified db error carries its own retriability (true only
403            // for the transient conflict / clock-skew classes).
404            Self::Structured { retriable, .. } => *retriable,
405            _ => false,
406        }
407    }
408
409    /// Structured, per-code metadata (additive; Issue #3368 and, for
410    /// `PermissionDenied`, Issue #3234). Rendered under `error.details`.
411    fn details(&self) -> Option<Value> {
412        match self {
413            // A role denial names the required class and the principal's role,
414            // mirroring the MCP-surface 403 details (Issue #3234).
415            Self::PermissionDenied {
416                required_class,
417                principal_role,
418            } => Some(json!({
419                "required_class": required_class.to_string(),
420                "principal_role": principal_role.to_string(),
421            })),
422            Self::ResourceLimitExceeded(e) => {
423                let mut d = json!({
424                    "dimension": e.dimension.as_str(),
425                });
426                match e.dimension {
427                    LimitDimension::WallClockTimeout => {
428                        d["limit_ms"] = json!(e.limit);
429                    }
430                    LimitDimension::ResultRows | LimitDimension::ResultBytes => {
431                        d["limit"] = json!(e.limit);
432                        if let Some(c) = e.consumed {
433                            d["consumed"] = json!(c);
434                        }
435                    }
436                }
437                Some(d)
438            }
439            Self::InvalidLimitOverride(e) => Some(json!({
440                "dimension": e.dimension.as_str(),
441                "requested": e.requested,
442                "ceiling": e.ceiling,
443            })),
444            Self::InFlightCapacityExceeded { cap } => Some(json!({
445                "reason": "in_flight_query_cap",
446                "max_in_flight_queries": cap,
447            })),
448            Self::PrincipalQuotaExceeded {
449                principal,
450                current,
451                limit,
452            } => Some(json!({
453                "principal": principal,
454                "current": current,
455                "limit": limit,
456            })),
457            // A pre-classified db error forwards its structured details (shared
458            // with the MCP surface), so the HTTP body is byte-shape-identical.
459            Self::Structured { details, .. } => details.clone(),
460            _ => None,
461        }
462    }

464    /// Stable machine-readable code for *every* variant, using the Issue #3234
465    /// vocabulary. Rendered as `error.code` in the response body (unified with
466    /// the MCP surface) and stamped as `aletheiadb.error.code` onto the trace
467    /// span (Issue #3376).
468    #[must_use]
469    pub(crate) fn code_str(&self) -> &'static str {
470        match self {
471            Self::BadRequest(_) | Self::QueryParse(_) | Self::InvalidLimitOverride(_) => {
472                "INVALID_ARGUMENT"
473            }
474            Self::NotFound(_) => "NOT_FOUND",
475            Self::Internal(_) | Self::StateMissing => "INTERNAL",
476            Self::Unauthorized => "UNAUTHENTICATED",
477            Self::PermissionDenied { .. } => "PERMISSION_DENIED",
478            Self::ResourceLimitExceeded(_) => "RESOURCE_EXHAUSTED",
479            Self::PrincipalQuotaExceeded { .. } => "RESOURCE_EXHAUSTED",
480            Self::InFlightCapacityExceeded { .. } => "UNAVAILABLE",
481            // The #3234 code was fixed when the variant was classified.
482            Self::Structured { code, .. } => code,
483        }
484    }

486    /// Convert to a response with the unified nested error envelope (Issue
487    /// #3234): `{"error": {"code", "message", "retriable", "details"?}}`,
488    /// byte-shape-identical to the MCP surface. The active trace id (Issue
489    /// #3376) is carried additively as a **top-level** `trace_id` sibling of
490    /// `error` (the SDK reads it top-level) and as an `x-trace-id` header when
491    /// one is available, so a failing call can be looked up in the trace
492    /// backend directly.
493    #[must_use]
494    pub(crate) fn into_response_with_trace(self, trace_id: Option<String>) -> Response {
495        let status = self.status();
496        // Build the nested `error` object, mirroring `McpError::to_json`'s
497        // field order: code, message, retriable, then optional details.
498        let mut error_obj = serde_json::Map::new();
499        error_obj.insert("code".to_string(), json!(self.code_str()));
500        error_obj.insert("message".to_string(), json!(self.to_string()));
501        error_obj.insert("retriable".to_string(), json!(self.retriable()));
502        if let Some(details) = self.details() {
503            error_obj.insert("details".to_string(), details);
504        }
505        let mut body = json!({ "error": Value::Object(error_obj) });
506        // Additively carry the active trace id (Issue #3376) as a *top-level*
507        // `trace_id` field (sibling of `error`, not nested) so a failing call
508        // can be correlated in the trace backend and the SDK can read it.
509        if let (Some(map), Some(tid)) = (body.as_object_mut(), trace_id.as_deref()) {
510            map.insert("trace_id".to_string(), json!(tid));
511        }
512        let mut response = (status, Json(body)).into_response();
513        if matches!(self, Self::Unauthorized) {
514            response.headers_mut().insert(
515                axum::http::header::WWW_AUTHENTICATE,
516                axum::http::HeaderValue::from_static("Bearer"),
517            );
518        }
519        // Additively surface the trace id as an `x-trace-id` response header
520        // (Issue #3376) so limit errors (429/413/422) are correlatable too.
521        if let Some(tid) = trace_id.and_then(|t| axum::http::HeaderValue::from_str(&t).ok()) {
522            response
523                .headers_mut()
524                .insert(axum::http::HeaderName::from_static("x-trace-id"), tid);
525        }
526        // A transient overload — a wall-clock timeout (429) or a full in-flight
527        // worker pool (503) — hints clients to back off briefly before retrying
528        // (Issue #3368). Fixed conservative 1 s.
529        if matches!(
530            &self,
531            Self::ResourceLimitExceeded(e) if e.dimension == LimitDimension::WallClockTimeout
532        ) || matches!(
533            &self,
534            Self::InFlightCapacityExceeded { .. } | Self::PrincipalQuotaExceeded { .. }
535        ) {
536            response.headers_mut().insert(
537                axum::http::header::RETRY_AFTER,
538                axum::http::HeaderValue::from_static("1"),
539            );
540        }
541        response
542    }
543}

545impl IntoResponse for AletheiaHttpError {
546    fn into_response(self) -> Response {
547        // Auto-conversion path (extractor rejections, `?` in handlers not
548        // wrapped in a root span): include a trace id if one happens to be
549        // active on this thread.
550        let trace_id = crate::http::trace::active_trace_id();
551        self.into_response_with_trace(trace_id)
552    }
553}

560    async fn body_json(err: AletheiaHttpError) -> (StatusCode, Value) {
561        let resp = err.into_response();
562        let status = resp.status();
563        let bytes = to_bytes(resp.into_body(), usize::MAX).await.unwrap();
564        (status, serde_json::from_slice(&bytes).unwrap())
565    }

567    #[tokio::test]
568    async fn timeout_maps_to_429_retriable_with_details() {
569        let (status, body) = body_json(AletheiaHttpError::ResourceLimitExceeded(
570            ResourceLimitExceeded {
571                dimension: LimitDimension::WallClockTimeout,
572                limit: 100,
573                consumed: None,
574                retriable: true,
575            },
576        ))
577        .await;
578        assert_eq!(status, StatusCode::TOO_MANY_REQUESTS);
579        assert!(
580            body.get("success").is_none(),
581            "flat `success` field dropped"
582        );
