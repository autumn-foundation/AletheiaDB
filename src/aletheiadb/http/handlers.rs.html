61/// Health check endpoint. Returns `{"status": "healthy"}` with HTTP 200.
62///
63/// Classified [`AccessClass::Metrics`]: every role may probe liveness, but
64/// (in required-auth mode) only with a valid credential.
65#[get("/status")]
66pub async fn health_check(
67    trace: HttpTrace,
68    auth: AuthContext,
69) -> Result<Json<HealthResponse>, AletheiaHttpError> {
70    trace.record_operation("health");
71    // Enter the root span so the probe is a first-class (non-orphaned) span.
72    // There is no `.await` inside this block, so holding the guard is sound.
73    #[cfg(feature = "observability")]
74    let _entered = trace.span().entered();
75    auth.authorize(AccessClass::Metrics)?;
76    Ok(Json(HealthResponse {
77        status: "healthy".to_string(),
78    }))
79}

85/// Optional provenance filter parameters (Issue #3348) accepted by the
86/// read operations (`find_node`, `get_node`, `find_neighbors`).
87///
88/// Flattened into each read variant so callers pass the four keys inline;
89/// all default to absent, so omitting them reproduces prior behavior exactly.
90/// Validated via [`ProvenanceFilterParams::into_filter`], sharing the
91/// surface-agnostic [`ProvenanceFilter`](crate::core::ProvenanceFilter)
92/// semantics with the MCP surface.
93#[derive(Debug, Default, Deserialize)]
94pub struct ProvenanceFilterParams {
95    /// Exact-match provenance source.
96    #[serde(default)]
97    pub provenance_source: Option<String>,
98    /// Any-of provenance sources (unioned with `provenance_source`).
99    #[serde(default)]
100    pub provenance_sources: Option<Vec<String>>,
101    /// Inclusive minimum recorded confidence, in `[0.0, 1.0]`.
102    #[serde(default)]
103    pub min_confidence: Option<f64>,
104    /// Re-include versions with no recorded provenance.
105    #[serde(default)]
106    pub include_unattributed: Option<bool>,
107}
108
109impl ProvenanceFilterParams {
110    /// Validate into a [`ProvenanceFilter`](crate::core::ProvenanceFilter),
111    /// returning `Ok(None)` when no constraint is active (byte-identical
112    /// legacy behavior). Invalid values map to a `400 BadRequest`
113    /// (INVALID_ARGUMENT) whose message names the offending field (Issue
114    /// #3348 AC5), never a silently-empty success.
115    fn into_filter(self) -> Result<Option<crate::core::ProvenanceFilter>, AletheiaHttpError> {
116        crate::core::ProvenanceFilter::validated(
117            self.provenance_source,
118            self.provenance_sources,
119            self.min_confidence,
120            self.include_unattributed.unwrap_or(false),
121        )
122        .map_err(|e| AletheiaHttpError::BadRequest(format!("{} (field: {})", e, e.field())))
123    }
124}

207/// Wrapper for all JSON responses: `{ success: bool, data?: ..., error?: ... }`.
208#[derive(Debug, Serialize)]
209pub struct ApiResponse {
210    success: bool,
211    #[serde(skip_serializing_if = "Option::is_none")]
212    data: Option<Value>,
213    #[serde(skip_serializing_if = "Option::is_none")]
214    error: Option<String>,
215    /// `truncated` is additive (Issue #3368): present and `true` only when a
216    /// result row cap under [`RowOverflowPolicy::Truncate`] dropped rows;
217    /// omitted otherwise, so existing responses are unchanged.
218    #[serde(skip_serializing_if = "Option::is_none")]
219    truncated: Option<bool>,
220}
221
222impl ApiResponse {
223    /// Build a `{success: true, data}` success envelope.
224    ///
225    /// Exposed for the autumn-web migration spike (Issue #3524): the isolated
226    /// `aletheia-autumn-spike` crate wraps its `GET /nodes/{id}` node in this
227    /// exact envelope so its response is byte-identical to `POST /query`.
228    /// Widening from `pub(crate)` to `pub` is additive and changes no behavior.
229    #[must_use]
230    pub fn success(data: Value) -> Self {
231        Self {
232            success: true,
233            data: Some(data),
234            error: None,
235            truncated: None,
236        }
237    }
238
239    /// Success response that flags `truncated: true` when the row cap dropped
240    /// rows (Issue #3368). When `truncated` is `false`, the field is omitted.
241    pub(crate) fn success_with_truncation(data: Value, truncated: bool) -> Self {
242        Self {
243            success: true,
244            data: Some(data),
245            error: None,
246            truncated: truncated.then_some(true),
247        }
248    }
249}

251/// Convert a `serde_json::Value` into a `PredicateValue`, or `None` for
252/// unsupported kinds (arrays/objects aren't valid equality predicates).
253fn json_to_predicate_value(v: &Value) -> Option<PredicateValue> {
254    match v {
255        Value::Null => Some(PredicateValue::Null),
256        Value::Bool(b) => Some(PredicateValue::Bool(*b)),
257        Value::Number(n) => {
258            if let Some(i) = n.as_i64() {
259                Some(PredicateValue::Int(i))
260            } else {
261                n.as_f64().map(PredicateValue::Float)
262            }
263        }
264        Value::String(s) => Some(PredicateValue::String(s.clone())),
265        _ => None,
266    }
267}

269/// Run a CPU/IO-bound closure on the blocking pool and unwrap the join error.
270///
271/// The caller's current `tracing` span is propagated into the blocking thread
272/// and entered for the duration of `f`, so the database's own child spans
273/// (`aletheiadb.query.execute`, `aletheiadb.transaction.commit`, …) nest under
274/// the HTTP root span even though they run on a different thread (Issue #3376).
275async fn blocking<F, T>(f: F) -> Result<T, AletheiaHttpError>
276where
277    F: FnOnce() -> Result<T, AletheiaHttpError> + Send + 'static,
278    T: Send + 'static,
279{
280    #[cfg(feature = "observability")]
281    let span = tracing::Span::current();
282    tokio::task::spawn_blocking(move || {
283        #[cfg(feature = "observability")]
284        let _entered = span.enter();
285        f()
286    })
287    .await
288    .map_err(|e| AletheiaHttpError::Internal(e.to_string()))?
289}

295/// Build the [`WriteRequestOptions`](crate::api::transaction::WriteRequestOptions)
296/// for an authenticated `/query` write: when the request carried a verified
297/// principal, its name is stamped into a write-time provenance bundle
298/// (Issue #3350) so "who wrote this fact" is answerable from history.
299/// Anonymous-mode requests get default options (no provenance).
300fn write_options_for(principal: Option<&str>) -> crate::api::transaction::WriteRequestOptions {
301    let options = crate::api::transaction::WriteRequestOptions::new();
302    match principal.and_then(|name| {
303        // A principal-only bundle cannot fail validation (only `confidence`
304        // is validated, and it is unset here); `.ok()` keeps this
305        // non-panicking regardless.
306        crate::core::provenance::Provenance::builder()
307            .principal(name)
308            .build()
309            .ok()
310    }) {
311        Some(provenance) => options.with_provenance(provenance),
312        None => options,
313    }
314}

316async fn handle_create_node(
317    db: Arc<crate::AletheiaDB>,
318    label: String,
319    properties: Option<HashMap<String, Value>>,
320    principal: Option<String>,
321) -> Result<Value, AletheiaHttpError> {
322    let props = match properties {
323        // A property-KEY interner capacity breach must render as `412
324        // FAILED_PRECONDITION` (configurable interner cap); a bad value stays
325        // `400 INVALID_ARGUMENT`. `from_db_error` routes each correctly.
326        Some(p) => json_to_property_map(&p).map_err(|e| AletheiaHttpError::from_db_error(&e))?,
327        None => crate::core::PropertyMap::new(),
328    };
329
330    blocking(move || {
331        let node_id = db
332            .create_node_with_options(&label, props, write_options_for(principal.as_deref()))
333            // Issue #3378: a schema-constraint violation renders with the
334            // structured #3234 envelope (code/details); every other write
335            // error keeps the prior 400 BadRequest mapping.
336            .map_err(|e| AletheiaHttpError::from_db_error(&e))?;
337        let node = db
338            .get_node(node_id)
339            .map_err(|e| AletheiaHttpError::Internal(e.to_string()))?;
340        let props_json =
341            property_map_to_json(&node.properties).map_err(AletheiaHttpError::Internal)?;
342        Ok(json!({
343            "id": node.id.as_u64(),
344            "label": interned_to_string(node.label),
345            "properties": props_json,
346        }))
347    })
348    .await
349}

351/// Load a node's write-time provenance bundle for the exact version the
352/// snapshot is holding (Issue #3348). Best-effort: an unreadable metadata
353/// record degrades to `None` (treated as unattributed by the filter),
354/// mirroring the MCP surface, rather than failing the whole read.
355fn node_provenance(
356    db: &crate::AletheiaDB,
357    node: &crate::core::Node,
358) -> Option<crate::core::Provenance> {
359    match db.get_node_version_read_metadata(node.current_version) {
360        Ok(Some((prov, _interval))) => prov,
361        _ => None,
362    }

365async fn handle_get_node(
366    db: Arc<crate::AletheiaDB>,
367    node_id: u64,
368    provenance_filter: ProvenanceFilterParams,
369) -> Result<Value, AletheiaHttpError> {
370    let nid = NodeId::new(node_id).map_err(|e| AletheiaHttpError::BadRequest(e.to_string()))?;
371    // Issue #3348: validate the filter up front (400 on bad input, AC5).
372    let prov_filter = provenance_filter.into_filter()?;
373
374    blocking(move || {
375        let node = db
376            .get_node(nid)
377            .map_err(|_| AletheiaHttpError::NotFound(format!("Node {node_id} not found")))?;
378        // Issue #3348: a node that exists but fails the provenance filter is
379        // not in the filtered view -> an honest 404 (never a fabricated node).
380        if let Some(filter) = &prov_filter
381            && !filter.matches(node_provenance(&db, &node).as_ref())
382        {
383            return Err(AletheiaHttpError::NotFound(format!(
384                "Node {node_id} not found"
385            )));
386        }
387        // Issue #3524: reuse the shared serializer so the migration-spike's
388        // `GET /nodes/{id}` produces byte-identical output. Behavior unchanged.
389        crate::http::converters::node_to_query_json(&node).map_err(AletheiaHttpError::Internal)
390    })
391    .await
392}

394async fn handle_find_node(
395    db: Arc<crate::AletheiaDB>,
396    label: Option<String>,
397    properties: Option<HashMap<String, Value>>,
398    limit: Option<usize>,
399    offset: Option<usize>,
400    provenance_filter: ProvenanceFilterParams,
401) -> Result<Value, AletheiaHttpError> {
402    let limit_val = limit.unwrap_or(100);
403    let offset_val = offset.unwrap_or(0);
404    // Issue #3348: validate the filter up front (400 on bad input, AC5).
405    let prov_filter = provenance_filter.into_filter()?;
406
407    if offset_val.saturating_add(limit_val) > MAX_DEEP_PAGINATION {
408        return Err(AletheiaHttpError::BadRequest(format!(
409            "Pagination limit exceeded: offset + limit must be <= {MAX_DEEP_PAGINATION}"
410        )));
411    }
412
413    // Issue #3348 (FIX A): the HTTP find_node response is a bare array with no
414    // `has_more`/`next_offset` signal, so a page that filters down below
415    // `limit_val` mid-scan would make a client using the standard "short page
416    // => end of results" heuristic stop early and under-read later matches.
417    // When a provenance filter is active we therefore OVER-FETCH and REFILL:
418    // scan forward past `limit_val` until the page holds `limit_val`
419    // filter-passing rows (or the source is exhausted), so a short/empty page
420    // genuinely means end-of-data. The refill scan is bounded by
421    // `MAX_DEEP_PAGINATION` (the same horizon enforced on `offset + limit`
422    // above) -- it is never unbounded; if that cap is reached before the page
423    // fills, we stop and return the shorter page (documented boundary). With no
424    // filter the behavior is byte-identical to before (scan exactly `limit_val`).
425    let scan_limit = if prov_filter.is_some() {
426        MAX_DEEP_PAGINATION.saturating_sub(offset_val)
427    } else {
428        limit_val
429    };
430
431    blocking(move || {
432        let mut builder = if let Some(lbl) = label {
433            QueryBuilder::new().scan_label(&lbl)
434        } else {
435            QueryBuilder::new().scan(None)
436        };
437
438        if let Some(props) = properties {
439            for (key, value) in props {
440                if let Some(pred_value) = json_to_predicate_value(&value) {
441                    builder = builder.filter(Predicate::eq(key, pred_value));
442                }
443            }
444        }
445
446        if let Some(skip) = offset {
447            builder = builder.skip(skip);
448        }
449
450        let results = builder
451            .limit(scan_limit)
452            .execute(&db)
453            .map_err(|e| AletheiaHttpError::Internal(e.to_string()))?;
454
455        // NOTE: explicit `row_result?` rather than `.flatten()` so storage
456        // errors mid-scan propagate as 500 instead of being silently dropped
457        // and producing a partial `success: true` response.
458        let mut nodes = Vec::new();
459        for row_result in results {
460            let row = row_result.map_err(|e| AletheiaHttpError::Internal(e.to_string()))?;
461            if let crate::query::executor::EntityResult::Node(node) = row.entity {
462                // Issue #3348: per-page provenance filter. The scan over-fetches
463                // (see `scan_limit` above) so the page can be refilled to
464                // `limit_val` passing rows; iteration is lazy, so we stop
465                // consuming the scan as soon as the page is full.
466                if let Some(filter) = &prov_filter
467                    && !filter.matches(node_provenance(&db, &node).as_ref())
468                {
469                    continue;
470                }
471                let props_json =
472                    property_map_to_json(&node.properties).map_err(AletheiaHttpError::Internal)?;
473                nodes.push(json!({
474                    "id": node.id.as_u64(),
475                    "label": interned_to_string(node.label),
476                    "properties": props_json,
477                }));
478                if nodes.len() >= limit_val {
479                    break;
480                }
481            }
482        }
483        Ok(Value::Array(nodes))
484    })
485    .await
486}

488async fn handle_find_neighbors(
489    db: Arc<crate::AletheiaDB>,
490    node_id: u64,
491    limit: Option<usize>,
492    offset: Option<usize>,
493    provenance_filter: ProvenanceFilterParams,
494) -> Result<Value, AletheiaHttpError> {
495    let nid = NodeId::new(node_id).map_err(|e| AletheiaHttpError::BadRequest(e.to_string()))?;
496
497    let limit_val = limit.unwrap_or(100).min(MAX_NEIGHBOR_LIMIT);
498    let offset_val = offset.unwrap_or(0);
499    // Issue #3348: validate the filter up front (400 on bad input, AC5).
500    let prov_filter = provenance_filter.into_filter()?;
501
502    if offset_val.saturating_add(limit_val) > MAX_DEEP_PAGINATION {
503        return Err(AletheiaHttpError::BadRequest(format!(
504            "Pagination limit exceeded: offset + limit must be <= {MAX_DEEP_PAGINATION}"
505        )));
506    }
507
508    // Issue #3348 (FIX A): like find_node, find_neighbors returns a bare array
509    // with no completeness signal, so a provenance-filtered page that would
510    // fall short mid-scan is refilled by over-fetching neighbor candidates
511    // until the page holds `limit_val` passing rows (or candidates are
512    // exhausted). The refill scan is bounded by `MAX_DEEP_PAGINATION` (the same
513    // horizon enforced on `offset + limit` above) so it is never unbounded; if
514    // that cap is reached before the page fills we stop (documented boundary).
515    // With no filter the behavior is unchanged (take exactly `limit_val`).
516    let scan_limit = if prov_filter.is_some() {
517        MAX_DEEP_PAGINATION.saturating_sub(offset_val)
518    } else {
519        limit_val
520    };
521
522    blocking(move || {
523        let mut seen_ids = HashSet::new();
524        let mut neighbors = Vec::with_capacity(limit_val);
525
526        let outgoing_iter = db
527            .get_outgoing_edges_iter(nid)
528            .map(|edge_id| db.get_edge_target(edge_id).ok());
529        let incoming_iter = db
530            .get_incoming_edges_iter(nid)
531            .map(|edge_id| db.get_edge_source(edge_id).ok());
532
533        // `take(scan_limit)` bounds the candidate scan; when a filter is active
534        // this over-fetches past `limit_val` so the page can be refilled, and
535        // the explicit `break` below stops once `limit_val` passing rows are
536        // collected (lazy iteration => no wasted candidate reads).
537        let combined_iter = outgoing_iter
538            .chain(incoming_iter)
539            .flatten()
540            .filter(|&neighbor_id| seen_ids.insert(neighbor_id))
541            .skip(offset_val)
542            .take(scan_limit);
543
544        for neighbor_id in combined_iter {
545            let node = db
546                .get_node(neighbor_id)
547                .map_err(|e| AletheiaHttpError::Internal(e.to_string()))?;
548            // Issue #3348: keep only neighbor nodes whose provenance passes.
549            if let Some(filter) = &prov_filter
550                && !filter.matches(node_provenance(&db, &node).as_ref())
551            {
552                continue;
553            }
554            let props_json =
555                property_map_to_json(&node.properties).map_err(AletheiaHttpError::Internal)?;
556            neighbors.push(json!({
557                "id": node.id.as_u64(),
558                "label": interned_to_string(node.label),
559                "properties": props_json,
560            }));
561            if neighbors.len() >= limit_val {
562                break;
563            }
564        }
565        Ok(Value::Array(neighbors))
566    })
567    .await
568}

570async fn handle_execute_query(
571    db: Arc<crate::AletheiaDB>,
572    query: String,
573    parameters: Option<HashMap<String, Value>>,
574) -> Result<Value, AletheiaHttpError> {
575    blocking(move || {
576        let parsed_query = if let Some(params_json) = parameters {
577            let params =
578                json_to_parameter_map(&params_json).map_err(AletheiaHttpError::BadRequest)?;
579            parse_query_with_params(&query, params)
580                .map_err(|e| AletheiaHttpError::QueryParse(e.to_string()))?
581        } else {
582            parse_query(&query).map_err(|e| classify_query_error(e.to_string()))?
583        };
584
585        let results = db
586            .execute_query(parsed_query)
587            .map_err(|e| classify_query_error(e.to_string()))?;
588
589        let rows = results
590            .take(MAX_EXEC_RESULTS)
591            .map(|row_result| {
592                let row = row_result.map_err(|e| AletheiaHttpError::Internal(e.to_string()))?;
593                query_row_to_json(row).map_err(AletheiaHttpError::Internal)
594            })
595            .collect::<Result<Vec<_>, AletheiaHttpError>>()?;
596
597        Ok(Value::Array(rows))
598    })
599    .await

776async fn handle_bulk_execute_query(
777    db: Arc<crate::AletheiaDB>,
778    queries: Vec<ExecuteQueryInput>,
779) -> Result<Value, AletheiaHttpError> {
780    validate_bulk_size(&queries, "queries")?;
781    blocking(move || {
782        let mut all_results = Vec::with_capacity(queries.len());
783        let mut total_rows = 0usize;
784        for (idx, query_item) in queries.into_iter().enumerate() {
785            let parsed_query = if let Some(params_json) = query_item.parameters {
786                let params =
787                    json_to_parameter_map(&params_json).map_err(AletheiaHttpError::BadRequest)?;
788                parse_query_with_params(&query_item.query, params)
789                    .map_err(|e| AletheiaHttpError::QueryParse(e.to_string()))?
790            } else {
791                parse_query(&query_item.query).map_err(|e| classify_query_error(e.to_string()))?
792            };
793
794            let rows = db
795                .execute_query(parsed_query)
796                .map_err(|e| classify_query_error(e.to_string()))?
797                .take(MAX_EXEC_RESULTS)
798                .map(|row_result| {
799                    let row = row_result.map_err(|e| AletheiaHttpError::Internal(e.to_string()))?;
800                    query_row_to_json(row).map_err(AletheiaHttpError::Internal)
801                })
802                .collect::<Result<Vec<_>, AletheiaHttpError>>()?;
803
804            total_rows = total_rows.saturating_add(rows.len());
805            if total_rows > MAX_BULK_EXEC_TOTAL_ROWS {
806                return Err(AletheiaHttpError::BadRequest(format!(
807                    "Bulk query result budget exceeded at query {}: total rows {} > {}",
808                    idx + 1,
809                    total_rows,
810                    MAX_BULK_EXEC_TOTAL_ROWS
811                )));
812            }
813
814            all_results.push(json!({
815                "query": query_item.query,
816                "rows": rows,
817            }));
818        }
819        Ok(Value::Array(all_results))
820    })
821    .await
822}

834/// Classify a `/query` operation into the [`AccessClass`] it requires.
835///
836/// Exhaustive match — no wildcard arm — so adding a new `QueryRequest`
837/// variant forces an explicit classification decision at compile time.
838/// This mapping is the seed of the authorization conformance matrix.
839fn query_access_class(req: &QueryRequest) -> AccessClass {
840    match req {
841        // Read-only lookups and traversals.
842        QueryRequest::FindNode { .. }
843        | QueryRequest::GetNode { .. }
844        | QueryRequest::BulkGetNodes { .. }
845        | QueryRequest::FindNeighbors { .. } => AccessClass::Read,
846        // Graph mutations.
847        QueryRequest::CreateNode { .. }
848        | QueryRequest::BulkCreateNodes { .. }
849        | QueryRequest::BulkUpdateNodes { .. }
850        | QueryRequest::BulkDeleteNodes { .. } => AccessClass::Write,
851        // Query-string execution: classify per statement using the shared
852        // read-only guard (the same one the MCP `query` tool uses). A
853        // statement with no mutating clause needs only Read.
854        QueryRequest::ExecuteQuery { query, .. } => {
855            if detect_mutating_clause(query).is_none() {
856                AccessClass::Read
857            } else {
858                AccessClass::Write
859            }
860        }
861        // A bulk batch is Read only if *every* statement is read-only.
862        QueryRequest::BulkExecuteQuery { queries } => {
863            if queries
864                .iter()
865                .all(|q| detect_mutating_clause(&q.query).is_none())
866            {
867                AccessClass::Read
868            } else {
869                AccessClass::Write
870            }
871        }
872    }
873}

878/// and its fields stay at the top level) and an optional per-call `limits`
879/// override. Both `limits` and its inner fields are `#[serde(default)]`, so a

889/// Apply the effective row cap to a result `Value` (Issue #3368).
890///
891/// Caps the **top-level array length only**: row limits apply to
892/// array-shaped results (list-like reads); a single-entity object result is
893/// returned unchanged, and nested arrays inside an object are never counted or
894/// truncated. Under [`RowOverflowPolicy::Truncate`] the array is capped and
895/// `truncated` is `true`; under [`RowOverflowPolicy::Reject`] an over-cap
896/// result is a structured `413` error (never retriable — the result is simply
897/// too large).
898///
899/// Only ever invoked for read-class operations (Issue #3368 review MUST-FIX 1):
900/// a committed write's acknowledgement is never truncated or rejected here.
901fn apply_row_cap(
902    data: Value,
903    limits: &EffectiveQueryLimits,
904) -> Result<(Value, bool), AletheiaHttpError> {
905    if limits.max_result_rows == 0 {
906        return Ok((data, false));
907    }
908    match data {
909        Value::Array(mut rows) if rows.len() > limits.max_result_rows => {
910            let consumed = rows.len();
911            match limits.row_overflow {
912                RowOverflowPolicy::Truncate => {
913                    rows.truncate(limits.max_result_rows);
914                    Ok((Value::Array(rows), true))
915                }
916                RowOverflowPolicy::Reject => Err(AletheiaHttpError::ResourceLimitExceeded(
917                    ResourceLimitExceeded {
918                        dimension: LimitDimension::ResultRows,
919                        limit: limits.max_result_rows as u64,
920                        consumed: Some(consumed as u64),
921                        retriable: false,
922                    },
923                )),
924            }
925        }
926        other => Ok((other, false)),
927    }
928}

930/// Enforce the time and result-shaping limit dimensions around an operation
931/// future (Issue #3368):
932///
933/// 1. **Wall-clock timeout** — the operation is raced against `timeout_ms`; on
934///    elapse the client gets a prompt `429`. Applies to **all** operations
935///    (it bounds the client's wait). (The underlying blocking computation is
936///    not cancelled — it runs to completion on the blocking pool — but the HTTP
937///    response deadline is bounded. True engine-level cancellation is the
938///    query-executor lane's concern.) The timeout's `retriable` flag is `true`
939///    only for read-class ops: a **write** may have committed on the blocking
940///    pool, so its timeout is `retriable: false` to avoid a duplicate retry
941///    (review MUST-FIX 1).
942/// 2. **Result-row cap** — truncate-or-reject per policy, **read-class only**.
943///    A committed write's small acknowledgement (ids / version ids) is never
944///    truncated or `413`'d after the fact.
945///
946/// The **result-byte cap** is enforced by the caller ([`handle_query`]) on the
947/// fully-serialized response envelope, so it too is read-class only and is not
948/// applied here.
949///
950/// # In-flight worker DoS guard (Issue #3368)
951///
952/// Because a timed op's worker is **detached and non-cancellable** (see above),
953/// a caller sending tiny `timeout_ms` overrides — or hammering the documented
954/// retriable-timeout retry loop — could pile up unbounded still-running
955/// expensive queries and exhaust threads/CPU/memory. So the **timed** branch is
956/// bounded: a slot is reserved from `in_flight` (cap = `cap`, `0` = unbounded)
957/// **before** the worker is spawned; at the cap the query is rejected `503`
958/// `UNAVAILABLE` (retriable) instead of spawning yet another worker. The
959/// [`InFlightGuard`](crate::http::state) is moved **into** the spawned worker so
960/// its slot is released only when the WORKER finishes — a caller that times out
961/// and discards the result keeps holding its slot until the work actually
962/// completes. The inline/unlimited branch (`timeout_ms == 0`) never spawns and
963/// is therefore never guarded.
964///
965/// A dimension whose effective value is `0` is unlimited and skipped. Returns
966/// the (possibly truncated) result plus a `truncated` flag.
967async fn enforce_query_limits<F>(
968    in_flight: InFlightCounter,
969    cap: usize,
970    limits: &EffectiveQueryLimits,
971    is_write: bool,
972    op: F,
973) -> Result<(Value, bool), AletheiaHttpError>
974where
975    F: Future<Output = Result<Value, AletheiaHttpError>> + Send + 'static,
976{
977    // 1. Wall-clock timeout (all ops — it bounds the client's wait).
978    let data = if limits.timeout_ms > 0 {
979        // Admission control: reserve an in-flight slot BEFORE spawning the
980        // detached worker; at the cap, reject rather than pile up another
981        // non-cancellable query (Issue #3368 DoS guard). `cap == 0` is
982        // unbounded (a slot is always granted).
983        let guard = in_flight
984            .try_acquire(cap)
985            .ok_or(AletheiaHttpError::InFlightCapacityExceeded { cap })?;
986
987        // Spawn the op on its own task with the guard moved in, so the slot is
988        // released only when the WORKER finishes — even if the caller below
989        // times out and discards this handle, the worker runs to completion
990        // still holding its slot. Instrument with the current span so the DB's
991        // child spans keep nesting under the HTTP root span (Issue #3376).
992        #[cfg(feature = "observability")]
993        let worker = {
994            use tracing::Instrument;
995            async move {
996                let _guard = guard;
997                op.await
998            }
999            .instrument(tracing::Span::current())
1000        };
1001        #[cfg(not(feature = "observability"))]
1002        let worker = async move {
1003            let _guard = guard;
1004            op.await
1005        };
1006        let handle = tokio::spawn(worker);
1007
1008        match tokio::time::timeout(Duration::from_millis(limits.timeout_ms), handle).await {
1009            // Worker finished within the deadline (guard already dropped inside
1010            // it, so its slot is released before we observe completion).
1011            Ok(Ok(result)) => result?,
1012            // The worker task panicked or was cancelled without a result. Do
1013            // NOT treat this as a timeout — a deterministic panic must not be
1014            // reported as an infinitely-retriable timeout. The guard still drops
1015            // as the panic unwinds, so the slot is released. Carry no
1016            // engine-internal detail (no panic payload) in the message, matching
1017            // the MCP surface's worker-died error (Issue #3368).
1018            Ok(Err(_join_err)) => {
1019                return Err(AletheiaHttpError::Internal(
1020                    "query worker terminated unexpectedly before producing a result".to_string(),
1021                ));
1022            }
1023            // Deadline elapsed first. Return promptly; we deliberately do NOT
1024            // abort `handle` — the detached worker keeps running to completion,
1025            // holding its in-flight slot until it finishes.
1026            Err(_elapsed) => {
1027                return Err(AletheiaHttpError::ResourceLimitExceeded(
1028                    ResourceLimitExceeded {
1029                        dimension: LimitDimension::WallClockTimeout,
1030                        limit: limits.timeout_ms,
1031                        consumed: None,
1032                        // A write may already have committed — do not invite a
1033                        // duplicate retry.
1034                        retriable: !is_write,
1035                    },
1036                ));
1037            }
1038        }
1039    } else {
1040        // Inline / unlimited path: never spawns, so it is never guarded.
1041        op.await?
1042    };
1043
1044    // 2. Result-row cap — READ-class ops only. A write acknowledgement is never
1045    // truncated or rejected after the write has committed.
1046    if is_write {
1047        return Ok((data, false));
1048    }
1049    let (data, truncated) = apply_row_cap(data, limits)?;
1050    Ok((data, truncated))
1051}

1053/// Polymorphic `/query` endpoint. Dispatches on the `operation` tag.
1054///
1055/// Authorization happens *before* any database work: the request's operation
1056/// is classified via [`query_access_class`] and checked against the caller's
1057/// role. Per-query resource limits (Issue #3368) are then resolved from the
1058/// server defaults + the optional per-call `limits` override (a request that
1059/// asks for more than the operator ceiling is rejected `422` here, before any
1060/// DB work), and enforced around execution by [`enforce_query_limits`].
1061///
1062/// The whole handler body runs inside the [`HttpTrace`] root span (Issue
1063/// #3376), so the database's query/write child spans nest underneath it, the
1064/// operation name / result count / error code are stamped onto the span, and a
1065/// failure carries the active trace id back to the caller for correlation.
1066#[post("/query")]
1067pub async fn handle_query(
1068    trace: HttpTrace,
1069    auth: AuthContext,
1070    state: AppState,
1071    Json(envelope): Json<QueryEnvelope>,
1072) -> Response {
1073    let QueryEnvelope {
1074        request: req,
1075        limits,
1076    } = envelope;
1077
1078    // Compute and stamp per-request attributes only when the span will actually
1079    // be recorded (Issue #3376, MED2 / Perf). A feature-absent build compiles
1080    // this block out entirely; a runtime-disabled or head-sampled-out span
1081    // short-circuits `is_recording()` so the (otherwise-wasted) attribute
1082    // computation — including `request_is_temporal`'s statement scan — never
1083    // runs.
1084    #[cfg(feature = "observability")]
1085    let recording = trace.is_recording();
1086    #[cfg(feature = "observability")]
1087    if recording {
1088        trace.record_operation(operation_name(&req));
1089        trace.record_temporal_scope(request_is_temporal(&req));
1090    }
1091    #[cfg(feature = "otel")]
1092    maybe_capture_statement(&trace, &req);
1093    // In an observability-off build the extractor still runs (zero-sized no-op),
1094    // but nothing consumes `trace`; keep it "used" without renaming the field
1095    // (which is genuinely used under `observability`).
1096    #[cfg(not(feature = "observability"))]
1097    let _ = &trace;
1098
1099    #[cfg(feature = "observability")]
1100    let span = trace.span();
1101
1102    // Classify the operation once. The access class drives BOTH authorization
1103    // and limit semantics: a write is exempt from the result row/byte caps and
1104    // its wall-clock timeout is non-retriable (the write may already have
1105    // committed — Issue #3368 review MUST-FIX 1).
1106    let access_class = query_access_class(&req);
1107    let is_write = matches!(access_class, AccessClass::Write);
1108
1109    // Authorization + limit resolution + dispatch, wrapped in the trace root
1110    // span so DB child spans nest underneath it and the trace id is available
1111    // for correlation on any failure (Issue #3376). Resource limits (Issue
1112    // #3368) are resolved and enforced inside, so a limit rejection (422/413/
1113    // 429) is stamped and trace-correlated exactly like any other error.
1114    let work = async move {
1115        // Authorization first: auth rejects (401/403) before limit logic runs.
1116        auth.authorize(access_class)?;
1117
1118        // Resolve effective limits; an over-ceiling per-call override is a 422
1119        // rejected up front, before any database work.
1120        let effective = state
1121            .query_limits()
1122            .effective(limits.as_ref())
1123            .map_err(AletheiaHttpError::InvalidLimitOverride)?;
1124
1125        // In-flight worker DoS guard (Issue #3368): the shared counter + the
1126        // configured cap, passed to `enforce_query_limits` which reserves a slot
1127        // before spawning the timed worker (or rejects `503` at the cap).
1128        let in_flight = state.in_flight_counter();
1129        let in_flight_cap = state.query_limits().max_in_flight_queries;
1130
1131        let db = state.db_arc();
1132        // Verified principal name (if any) for provenance stamping on write
1133        // operations (Issue #3350). Anonymous mode yields None.
1134        let principal = auth.principal().map(|p| p.name.clone());
1135
1136        // Build (but do not yet await) the operation future so the wall-clock
1137        // timeout in `enforce_query_limits` wraps the actual execution.
1138        let op = async move {
1139            match req {
1140                QueryRequest::CreateNode { label, properties } => {
1141                    handle_create_node(db, label, properties, principal).await
1142                }
1143                QueryRequest::GetNode {
1144                    node_id,
1145                    provenance_filter,
1146                } => handle_get_node(db, node_id, provenance_filter).await,
1147                QueryRequest::BulkCreateNodes { nodes } => {
1148                    handle_bulk_create_nodes(db, nodes, principal).await
1149                }
1150                QueryRequest::BulkGetNodes { node_ids } => {
1151                    handle_bulk_get_nodes(db, node_ids).await
1152                }
1153                QueryRequest::BulkUpdateNodes { updates } => {
1154                    handle_bulk_update_nodes(db, updates, principal).await
1155                }
1156                QueryRequest::BulkDeleteNodes { node_ids, cascade } => {
1157                    handle_bulk_delete_nodes(db, node_ids, cascade, principal).await
1158                }
1159                QueryRequest::FindNode {
1160                    label,
1161                    properties,
1162                    limit,
1163                    offset,
1164                    provenance_filter,
1165                } => {
1166                    handle_find_node(db, label, properties, limit, offset, provenance_filter).await
1167                }
1168                QueryRequest::FindNeighbors {
1169                    node_id,
1170                    limit,
1171                    offset,
1172                    provenance_filter,
1173                } => handle_find_neighbors(db, node_id, limit, offset, provenance_filter).await,
1174                QueryRequest::ExecuteQuery { query, parameters } => {
1175                    handle_execute_query(db, query, parameters).await
1176                }
1177                QueryRequest::BulkExecuteQuery { queries } => {
1178                    handle_bulk_execute_query(db, queries).await
1179                }
1180            }
1181        };
1182
1183        let (data, truncated) =
1184            enforce_query_limits(in_flight, in_flight_cap, &effective, is_write, op).await?;
1185        Ok::<(Value, bool, EffectiveQueryLimits), AletheiaHttpError>((data, truncated, effective))
1186    };
1187
1188    #[cfg(feature = "observability")]
1189    let result = {
1190        use tracing::Instrument;
1191        work.instrument(span.clone()).await
1192    };
1193    #[cfg(not(feature = "observability"))]
1194    let result = work.await;
1195
1196    // Build the final response. On success the result count is stamped on the
1197    // span (Issue #3376) and the wire envelope is serialized exactly ONCE
1198    // (Issue #3368 perf-High) so the result-byte cap is measured against the
1199    // real wire bytes. Any error — including a limit rejection surfaced here —
1200    // is funneled through the single trace-aware error path below so it carries
1201    // the `error.code` span attribute and the `trace_id`/`x-trace-id`.
1202    let outcome: Result<Response, AletheiaHttpError> = match result {
1203        Ok((data, truncated, effective)) => {
1204            #[cfg(feature = "observability")]
1205            if recording {
1206                trace.record_result_count(result_count(&data));
1207            }
1208            let body = ApiResponse::success_with_truncation(data, truncated);
1209            match serde_json::to_vec(&body) {
1210                Ok(bytes) => {
1211                    // Result-byte cap — READ-class only, and only when a finite
1212                    // cap is set. A committed write's acknowledgement has already
1213                    // taken effect; a post-commit 413 would lie about durability,
1214                    // so writes are exempt. Measured against the real wire bytes
1215                    // (never back through `Json(..)`, which would serialize a
1216                    // second time — these bytes are byte-identical to it).
1217                    if !is_write
1218                        && effective.max_response_bytes > 0
1219                        && bytes.len() > effective.max_response_bytes
1220                    {
1221                        Err(AletheiaHttpError::ResourceLimitExceeded(
1222                            ResourceLimitExceeded {
1223                                dimension: LimitDimension::ResultBytes,
1224                                limit: effective.max_response_bytes as u64,
1225                                consumed: Some(bytes.len() as u64),
1226                                retriable: false,
1227                            },
1228                        ))
1229                    } else {
1230                        Ok((
1231                            StatusCode::OK,
1232                            [(
1233                                header::CONTENT_TYPE,
1234                                HeaderValue::from_static("application/json"),
1235                            )],
1236                            bytes,
1237                        )
1238                            .into_response())
1239                    }
1240                }
1241                Err(e) => Err(AletheiaHttpError::Internal(e.to_string())),
1242            }
1243        }
1244        Err(e) => Err(e),
1245    };
1246
1247    match outcome {
1248        Ok(response) => response,
1249        Err(e) => {
1250            #[cfg(feature = "observability")]
1251            if recording {
1252                trace.record_error(e.code_str());
1253            }
1254            #[cfg(feature = "observability")]
1255            let trace_id = span.in_scope(crate::http::trace::active_trace_id);
1256            #[cfg(not(feature = "observability"))]
1257            let trace_id = crate::http::trace::active_trace_id();
1258            e.into_response_with_trace(trace_id)
1259        }
1260    }
1261}

1295/// Whether a request carries a temporal (`AS OF`) scope. Only statement-based
1296/// operations can; detection is a cheap case-insensitive substring check that
1297/// never inspects property values.
1298#[cfg(feature = "observability")]
1299fn request_is_temporal(req: &QueryRequest) -> bool {
1300    /// Allocation-free, case-insensitive scan for the `as of` temporal marker.
1301    /// The previous `to_ascii_lowercase().contains(..)` allocated a full
1302    /// lowercased copy of the statement on every request; scanning byte windows
1303    /// with `eq_ignore_ascii_case` matches identically for the ASCII needle
1304    /// while allocating nothing (Issue #3376, MED2 / Perf).
1305    fn is_as_of(query: &str) -> bool {
1306        const NEEDLE: &[u8] = b"as of";
1307        query
1308            .as_bytes()
1309            .windows(NEEDLE.len())
1310            .any(|w| w.eq_ignore_ascii_case(NEEDLE))
1311    }
1312    match req {
1313        QueryRequest::ExecuteQuery { query, .. } => is_as_of(query),
1314        QueryRequest::BulkExecuteQuery { queries } => queries.iter().any(|q| is_as_of(&q.query)),
1315        _ => false,
1316    }
1317}
