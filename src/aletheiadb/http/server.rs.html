1//! HTTP server creation and bootstrap.
2//!
3//! AletheiaDB's HTTP surface is built on [`autumn-web`](autumn_web), which in
4//! turn wraps [`axum`]. This module assembles the routes, state, and
5//! middleware stack once, and exposes:
6//!
7//! - [`run_server`] — bind and serve in the `aletheia-server` binary,
8//! - [`build_test_router`] — a fully-wired `axum::Router` for integration
9//!   tests using [`autumn_web::test::TestApp::from_router`].
10//!
11//! `run_server` applies middleware via autumn's `AppBuilder::layer` (so
12//! autumn's native middleware — request IDs, security headers, etc. — still
13//! runs). `build_test_router` rebuilds just our routes + middleware as a
14//! plain axum Router so tests can exercise it through
15//! [`tower::ServiceExt::oneshot`] without autumn's lifecycle.
16//!
17//! **Rate limiting status**: the pre-migration actix stack shipped
18//! `actix-governor` for per-IP rate limiting. autumn 0.2 does not yet ship
19//! rate limiting out of the box, and `tower-governor` 0.8 does not satisfy
20//! autumn's sealed `IntoAppLayer` bound (its `S::Response: From<GovernorError>`
21//! constraint doesn't propagate through autumn's strict service signature).
22//! Rather than write a short-lived shim, rate limiting is not applied at the
23//! HTTP layer in this PR. `RateLimitConfig` is preserved in the public API;
24//! it will be wired to autumn's native per-IP limiter when 0.3 ships. In the
25//! interim, operators should enforce rate limits at the reverse-proxy layer
26//! (nginx, Caddy, Envoy). See ADR 0055 for the full trade-off.
27

54/// Run the HTTP server to completion.
55///
56/// Creates a fresh `AletheiaDB`, wires it into shared state, assembles the
57/// autumn app, and hands control to `autumn_web::app`
58/// — which handles the TCP listener, graceful shutdown on SIGINT/SIGTERM, and
59/// startup/shutdown tracing. Returns only after the server exits cleanly.
60///
61/// # Environment variables consumed
62///
63/// | Name | Purpose | Default |
64/// |------|---------|---------|
65/// | `ALETHEIADB_PORT` | Listening port | `1963` (in honor of the first Doctor Who broadcast, 23 Nov 1963) |
66/// | `ALETHEIADB_HOST` | Bind address | `0.0.0.0` |
67/// | `ALETHEIADB_CORS_PERMISSIVE` | Allow any origin | `false` |
68/// | `ALETHEIADB_CORS_ORIGINS` | Comma-separated CORS allow-list | — |
69///
70/// All of these are parsed by the binary entry point into [`ServerConfig`]
71/// before reaching this function — `run_server` itself reads no environment
72/// variables of its own.
73///
74/// # Errors
75///
76/// Returns an IO error if the database fails to initialize or if the
77/// rate-limit configuration is invalid. Autumn's own startup failures
78/// (e.g. failing to bind the port) terminate the process via
79/// [`std::process::exit`]; they are not surfaced as `Err` here.
80pub async fn run_server(config: ServerConfig) -> std::io::Result<()> {
81    // Install OpenTelemetry tracing from the standard OTEL_* environment
82    // (Issue #3376). Disabled unless ALETHEIADB_OTEL is truthy or an OTLP
83    // endpoint is configured; the guard flushes + shuts the exporter down when
84    // `run_server` returns. A subscriber-already-installed error is non-fatal
85    // (the host process may own the subscriber).
86    #[cfg(feature = "otel")]
87    let _otel_guard =
88        match crate::observability::otel::init(&crate::observability::otel::OtelConfig::from_env())
89        {
90            Ok(guard) => guard,
91            Err(e) => {
92                eprintln!("WARNING: OpenTelemetry tracing not installed: {e}");
93                None
94            }
95        };
96
97    // Validate config before wiring anything else.
98    config
99        .rate_limit()
100        .validate()
101        .map_err(std::io::Error::other)?;
102
103    // Assemble the auth store: persisted keys (if a path is configured or
104    // derivable from data_dir), plus an optional env/config bootstrap admin
105    // key (memory-only, never persisted).
106    let auth_state = build_auth_state(&config).map_err(std::io::Error::other)?;
107    match auth_state.mode() {
108        AuthMode::Anonymous => {
109            // PROMINENT warning: the operator explicitly opted out of auth.
110            eprintln!(
111                "WARNING: AUTHENTICATION IS DISABLED (auth mode: anonymous). \
112                 Every request has full, unauthenticated access to the database. \
113                 Do not expose this server to untrusted networks."
114            );
115            tracing::warn!(
116                "authentication disabled (auth mode: anonymous); every request has full access"
117            );
118        }
119        AuthMode::Required => {
120            // Conservative default: refuse to start with zero credentials.
121            validate_auth_startup(AuthMode::Required, auth_state.store())
122                .map_err(std::io::Error::other)?;
123        }
124    }
125
126    let db = Arc::new(build_database(&config)?);
127    // Deliver the per-query resource limits (Issue #3368) to handlers via
128    // shared state, exactly as the request-body-size limit is delivered via
129    // config into the layer stack below.
130    let our_state = AppState::new(db).with_query_limits(config.query_limits().clone());
131    let startup_state = our_state.clone();
132    let startup_auth = auth_state.clone();
133    let shutdown_state = our_state.clone();
134    let persist_on_shutdown = config.data_dir().is_some();
135
136    eprintln!(
137        "Starting AletheiaDB HTTP server on {}",
138        config.bind_address()
139    );
140    match config.data_dir() {
141        Some(path) => eprintln!("Data directory: {}", path.display()),
142        None => eprintln!(
143            "WARNING: no data directory configured — running in-memory; state is lost on shutdown."
144        ),
145    }
146    if config.cors().is_permissive() {
147        eprintln!(
148            "WARNING: CORS is configured in permissive mode (any origin allowed). \
149             This is not recommended for production."
150        );
151    }
152
153    // Bridge our ServerConfig into autumn's config via its AUTUMN_*__* env
154    // vars (host/port/CORS/CSRF; the actuator hardening below goes through
155    // a ConfigLoader because `[actuator]` has no AUTUMN_* env mapping).
156    //
157    // TODO: fold this env-var bridge into `HardenedConfigLoader` — it's
158    // cleaner and retires the `unsafe` block.
159    //
160    // SAFETY: `set_var` is unsafe in edition 2024 because concurrent reads
161    // from other threads are UB. These calls happen before any autumn code
162    // runs (autumn's own startup reads env only inside `.run()`), and
163    // AletheiaDB performs no env reads on other threads during this window.
164    unsafe {
165        apply_autumn_env(&config);
166    }
167
168    // Framework endpoints autumn mounts outside our AuthContext extractor
169    // (they never see AletheiaDB credentials). With the sensitive actuator
170    // group forced off by `HardenedConfigLoader`, what remains is
171    // health/metadata only — but operators should still block them at the
172    // reverse proxy if they must not be publicly reachable.
173    eprintln!(
174        "Note: framework endpoints served WITHOUT AletheiaDB authentication: \
175         /health, /live, /ready, /startup, /actuator/health, /actuator/info, \
176         /actuator/metrics, /actuator/a11y, /actuator/ui, /actuator/ui/metrics. \
177         Sensitive actuator endpoints (/actuator/env, /actuator/configprops, \
178         /actuator/loggers, /actuator/tasks, /actuator/jobs, /actuator/prometheus) \
179         are disabled in every profile. Block /actuator and the probe paths at \
180         your reverse proxy if they must not be publicly reachable."
181    );
182
183    // TODO(autumn-0.3): wire per-IP rate limiting here when autumn ships it
184    // natively. See the module-level doc.
185    //
186    // Request tracing, metrics, and security headers are provided by autumn's
187    // baseline middleware. CORS is driven by the AUTUMN_CORS__* env vars set
188    // in `apply_autumn_env` above, so operator-facing `ALETHEIADB_CORS_*`
189    // settings flow end-to-end into autumn's CorsLayer.
190
191    autumn_web::app()
192        .with_config_loader(HardenedConfigLoader)
193        .on_startup(move |autumn_state| {
194            let installed = startup_state.clone();
195            let installed_auth = startup_auth.clone();
196            async move {
197                autumn_state.insert_extension(installed);
198                autumn_state.insert_extension(installed_auth);
199                Ok(())
200            }
201        })
202        // Graceful-shutdown checkpoint. Without this, SIGTERM leaves the
203        // string interner and index state behind at the last
204        // mutation-threshold snapshot (defaults: 500 new strings / 5-10
205        // minute cadence). A Docker-style stop/start workflow would lose
206        // label strings until a checkpoint happened organically, which
207        // manifests as `label: "<unknown:N>"` after restart. Flushing on
208        // shutdown makes restart semantics feel like postgres: stop, start,
209        // your data is exactly as you left it.
210        .on_shutdown(move || {
211            let db = shutdown_state.db_arc();
212            let should_persist = persist_on_shutdown;
213            async move {
214                if !should_persist {
215                    return;
216                }
217                match tokio::task::spawn_blocking(move || db.persist_indexes()).await {
218                    Ok(Ok(())) => eprintln!("Shutdown: indexes persisted."),
219                    Ok(Err(e)) => eprintln!("Shutdown: persist_indexes failed: {e}"),
220                    Err(e) => eprintln!("Shutdown: persist_indexes task panicked: {e}"),
221                }
222            }
223        })
224        .routes(all_routes())
225        // Reject oversized request bodies with 413 before they are buffered /
226        // deserialized (payload-amplification DoS prevention — Issue #3108).
227        // `DefaultBodyLimit` is axum-native and keeps the request type as
228        // `axum::extract::Request`, so it satisfies autumn's `IntoAppLayer`
229        // bound (unlike `RequestBodyLimitLayer`, which rewrites the body type).
230        .layer(DefaultBodyLimit::max(config.max_request_body_bytes()))
231        .run()
232        .await;
233
234    Ok(())
235}

237/// Autumn [`ConfigLoader`](autumn_web::config::ConfigLoader) that runs the
238/// default five-layer load (framework defaults → profile smart defaults →
239/// `autumn.toml` → `autumn-{profile}.toml` → `AUTUMN_*` env vars) and then
240/// forces security-relevant framework settings to safe values.
241///
242/// Autumn mounts its own routes (probes, actuator) *outside* AletheiaDB's
243/// `AuthContext` extractor, so they are never authenticated by our API-key
244/// layer. Its `dev` profile smart default sets `actuator.sensitive = true`,
245/// which additionally exposes `/actuator/env`, `/actuator/configprops`, an
246/// unauthenticated `PUT /actuator/loggers/{name}`, `/actuator/tasks`,
247/// `/actuator/jobs`, and `/actuator/prometheus` — and `[actuator]` has **no**
248/// `AUTUMN_*` env-var mapping, so `AUTUMN_ACTUATOR__SENSITIVE=false` cannot
249/// turn it off. This loader pins `actuator.sensitive = false` in every
250/// profile: the config-dump and log-mutation endpoints are never served,
251/// debug build or not. The remaining always-mounted framework endpoints
252/// (health probes, `/actuator/health|info|metrics|a11y|ui`) are documented in
253/// `docs/guides/security-quickstart.md` with reverse-proxy guidance.
254struct HardenedConfigLoader;
255
256impl autumn_web::config::ConfigLoader for HardenedConfigLoader {
257    async fn load(
258        &self,
259    ) -> Result<autumn_web::config::AutumnConfig, autumn_web::config::ConfigError> {
260        let mut config = autumn_web::config::TomlEnvConfigLoader::new()
261            .load()
262            .await?;
263        // Never serve the sensitive actuator group, regardless of profile
264        // or operator TOML: these endpoints bypass AletheiaDB auth entirely.
265        config.actuator.sensitive = false;
266        Ok(config)
267    }
268}

270/// Assemble the [`AuthState`] a [`ServerConfig`] implies.
271///
272/// - Loads the persisted key store from
273///   [`ServerConfig::resolved_auth_persist_path`] when one is configured or
274///   derivable, otherwise starts a memory-only store.
275/// - Installs the bootstrap admin key (principal name `bootstrap-admin`,
276///   role `admin`) when configured. Bootstrap keys are memory-only: they are
277///   re-supplied via env/config on every start and never persisted.
278///
279/// Startup *refusal* (required mode with zero credentials) is enforced by
280/// [`run_server`], not here — `build_test_router` deliberately skips it so
281/// tests can exercise the uniform-401 behavior of a keyless required-mode
282/// server.
283fn build_auth_state(config: &ServerConfig) -> Result<AuthState, String> {
284    let store = match config.resolved_auth_persist_path() {
285        Some(path) => AuthStore::with_persistence(&path)
286            .map_err(|e| format!("failed to open auth key store: {e}"))?,
287        None => AuthStore::new(),
288    };
289    if let Some(key) = config.bootstrap_admin_key() {
290        store
291            .insert_bootstrap_key("bootstrap-admin", Role::Admin, key.expose())
292            .map_err(|e| format!("failed to install bootstrap admin key: {e}"))?;
293    }
294    Ok(AuthState::new(Arc::new(store), config.auth_mode()))
295}

297/// Construct the [`AletheiaDB`] instance the HTTP server will share.
298///
299/// Resolution order:
300/// 1. If `ALETHEIADB_CONFIG` or `ALETHEIADB_DATA_DIR` is set in the environment,
301///    delegate to [`AletheiaDB::open_from_env`] so the server picks up the same
302///    config the MCP server, CLI, and Python SDK do.
303/// 2. Otherwise use [`ServerConfig::to_unified_config`] — this is the path
304///    integration tests take when they pass `data_dir` programmatically without
305///    touching the environment.
306fn build_database(config: &ServerConfig) -> std::io::Result<AletheiaDB> {
307    if crate::config::config_path_from_env().is_some()
308        || crate::config::data_dir_from_env().is_some()
309    {
310        return AletheiaDB::open_from_env().map_err(|e| std::io::Error::other(e.to_string()));
311    }
312    match config.to_unified_config() {
313        None => AletheiaDB::new().map_err(|e| std::io::Error::other(e.to_string())),
314        Some(unified) => AletheiaDB::with_unified_config(unified)
315            .map_err(|e| std::io::Error::other(e.to_string())),
316    }
317}

319/// Set the `AUTUMN_SERVER__*` and `AUTUMN_CORS__*` environment variables
320/// autumn's config loader reads, based on our [`ServerConfig`].
321///
322/// # Safety
323///
324/// Calls [`std::env::set_var`], which is `unsafe` under Rust edition 2024
325/// because concurrent reads from other threads result in UB. Callers must
326/// ensure no other thread is reading environment variables concurrently.
327/// [`run_server`] calls this exactly once, before handing control to autumn.
328unsafe fn apply_autumn_env(config: &ServerConfig) {
329    // SAFETY: see function-level docs; ordering is the caller's responsibility.
330    unsafe {
331        std::env::set_var("AUTUMN_SERVER__HOST", config.host());
332        std::env::set_var("AUTUMN_SERVER__PORT", config.port().to_string());
333
334        let cors = config.cors();
335        // `permissive` maps to wildcard origin. `restrictive` with zero explicit
336        // origins would disable CORS entirely, which is fine.
337        let origins: Vec<&str> = if cors.is_permissive() {
338            vec!["*"]
339        } else {
340            cors.allowed_origins().iter().map(String::as_str).collect()
341        };
342        std::env::set_var("AUTUMN_CORS__ALLOWED_ORIGINS", origins.join(","));
343        std::env::set_var(
344            "AUTUMN_CORS__ALLOWED_METHODS",
345            cors.get_allowed_methods().join(","),
346        );
347        std::env::set_var(
348            "AUTUMN_CORS__ALLOWED_HEADERS",
349            cors.get_allowed_headers().join(","),
350        );
351        std::env::set_var("AUTUMN_CORS__MAX_AGE_SECS", cors.get_max_age().to_string());
352
353        // Disable CSRF protection. autumn's `prod` profile enables it by
354        // default (appropriate for session-authenticated browser forms),
355        // but AletheiaDB's `/query` endpoint is a token-less JSON API —
356        // no sessions, no forms, nothing a cross-site POST could
357        // impersonate. CSRF protection adds no security here and only
358        // breaks legitimate POSTs. When the dashboard PR lands with
359        // session-backed routes, it should re-enable CSRF scoped to those
360        // routes rather than flipping this flag globally.
361        std::env::set_var("AUTUMN_SECURITY__CSRF__ENABLED", "false");
362    }
363}

365/// Build a fully-wired `axum::Router` suitable for integration tests.
366///
367/// Registers the crate's HTTP routes on a fresh [`autumn_web::prelude::AppState`]
368/// with the provided [`AppState`] pre-installed as an extension, then resolves
369/// the state via [`Router::with_state`]. Pass the result to
370/// [`autumn_web::test::TestApp::from_router`] to get a `TestClient`.
371///
372/// `RateLimitConfig` validation still runs so tests catch misconfiguration,
373/// but no rate-limit layer is attached (see the module doc).
374///
375/// Auth state is derived from `config` (mode, bootstrap key, persist path),
376/// exactly as `run_server` derives it — but *without* `run_server`'s
377/// "required mode needs at least one credential" startup refusal, so tests
378/// can exercise the uniform 401 of a keyless required-mode server. Use
379/// [`build_test_router_with_auth`] to inject a pre-populated
380/// [`AuthState`]/[`AuthStore`] instead.
381///
382/// # Errors
383///
384/// Returns an error string if the rate-limit or auth configuration is invalid.
385pub fn build_test_router(state: AppState, config: &ServerConfig) -> Result<Router, String> {
386    let auth_state = build_auth_state(config)?;
387    build_test_router_with_auth(state, auth_state, config)
388}
389
390/// [`build_test_router`] with an explicit [`AuthState`] — for tests that
391/// create/revoke keys against a shared [`AuthStore`] handle.
392///
393/// # Errors
394///
395/// Returns an error string if the rate-limit configuration is invalid.
396pub fn build_test_router_with_auth(
397    state: AppState,
398    auth_state: AuthState,
399    config: &ServerConfig,
400) -> Result<Router, String> {
401    config.rate_limit().validate()?;
402
403    // Fold the per-query resource limits (Issue #3368) from `config` into the
404    // shared state, mirroring how `run_server` wires them — so tests that pass
405    // a custom `QueryLimitsConfig` through `ServerConfig` see it enforced.
406    let state = state.with_query_limits(config.query_limits().clone());
407
408    let autumn_state = AutumnAppState::detached();
409    autumn_state.insert_extension(state);
410    autumn_state.insert_extension(auth_state);
411
412    let mut router: Router<AutumnAppState> = Router::new();
413    for route in all_routes() {
414        router = router.route(route.path, route.handler);
415    }
416
417    // Layer order matches run_server (minus the rate limiter).
418    let router = router
419        .layer(SetResponseHeaderLayer::if_not_present(
420            HeaderName::from_static("x-content-type-options"),
421            HeaderValue::from_static("nosniff"),
422        ))
423        .layer(SetResponseHeaderLayer::if_not_present(
424            HeaderName::from_static("x-frame-options"),
425            HeaderValue::from_static("DENY"),
426        ))
427        .layer(SetResponseHeaderLayer::if_not_present(
428            HeaderName::from_static("content-security-policy"),
429            HeaderValue::from_static("default-src 'none'; frame-ancestors 'none'"),
430        ))
431        .layer(build_cors_layer(config.cors()))
432        .layer(TraceLayer::new_for_http())
433        // Reject oversized request bodies with 413 before they are buffered /
434        // deserialized, bounding the memory a single request can force us to
435        // allocate (payload-amplification DoS prevention — Issue #3108).
436        .layer(DefaultBodyLimit::max(config.max_request_body_bytes()));
437
438    Ok(router.with_state(autumn_state))
439}

445/// Build a tower-http CORS layer from our [`CorsConfig`].
446fn build_cors_layer(cors: &CorsConfig) -> CorsLayer {
447    let origin = if cors.is_permissive() {
448        AllowOrigin::any()
449    } else {
450        let values: Vec<HeaderValue> = cors
451            .allowed_origins()
452            .iter()
453            .filter_map(|o| HeaderValue::from_str(o).ok())
454            .collect();
455        AllowOrigin::list(values)
456    };
457
458    let methods: Vec<Method> = cors
459        .get_allowed_methods()
460        .iter()
461        .filter_map(|m| m.parse().ok())
462        .collect();
463
464    let headers: Vec<HeaderName> = cors
465        .get_allowed_headers()
466        .iter()
467        .filter_map(|h| h.parse().ok())
468        .collect();
469
470    CorsLayer::new()
471        .allow_origin(origin)
472        .allow_methods(AllowMethods::list(methods))
473        .allow_headers(AllowHeaders::list(headers))
474        .max_age(Duration::from_secs(u64::from(cors.get_max_age())))
475}
