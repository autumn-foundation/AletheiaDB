1//! Cypher write-statement execution (Issue #560).
2//!
3//! Executes the mutating Cypher subset -- `CREATE`, `SET`, `DELETE`, and
4//! `DETACH DELETE` -- by driving AletheiaDB's **native write APIs** (via a single
5//! [`WriteTransaction`](crate::api::transaction::WriteTransaction)) rather than
6//! lowering into the read-only [`Query`](crate::query::Query) IR. Going through
7//! the native APIs means every mutation records the correct bi-temporal version
8//! (a new transaction-time record; a `SET`/`DELETE` supersedes without erasing
9//! history) exactly as the direct Rust API would.
10//!
11//! # Reachability / read-only guarantee
12//!
13//! This path is reached **only** through
14//! [`AletheiaDB::execute_cypher`](crate::db::AletheiaDB::execute_cypher) /
15//! `execute_cypher_with_params`. The MCP `query` tool and HTTP `/query` endpoint
16//! reject every mutating clause *before the parser runs*
17//! (`crate::query::read_only::detect_mutating_clause`), so a write statement can
18//! never execute through those read-only surfaces.
19//!
20//! # Atomicity
21//!
22//! All writes produced by one statement commit in a single
23//! [`AletheiaDB::write`](crate::db::AletheiaDB::write) transaction: they are
24//! applied all-or-nothing and share one commit timestamp. The reading `MATCH`
25//! is evaluated against current state *before* the write transaction opens
26//! (a v1 check-then-act window, acceptable because the writes themselves are
27//! atomic).
28//!
29//! # v1 scope (honest structured rejections, never a silently-wrong write)
30//!
31//! - `CREATE` of single-labelled nodes and typed, directed relationships, with
32//!   inline properties; endpoints may reference matched/earlier-created
33//!   variables.
34//! - `SET n.prop = value` (PATCH-merge onto the existing property map).
35//! - `DELETE` / `DETACH DELETE` of node and relationship variables, with the
36//!   openCypher safety rule: a plain `DELETE` of a node that still has
37//!   relationships is refused (use `DETACH DELETE`).
38//! - `RETURN` of bound variables (bare or `AS`-aliased).
39//!
40//! - `MERGE <pattern> [ON CREATE SET ...] [ON MATCH SET ...]` (Issue #3548):
41//!   match the pattern if present, otherwise create the *whole* pattern
42//!   (openCypher whole-pattern semantics). A bare match records no new version;
43//!   a create records new versions; `ON MATCH SET` on a matched entity records
44//!   a version (it is an update). Matching reuses the current-state matcher
45//!   (`match_bindings`) as a pre-transaction (check-then-act) read.
46//!
47//!   For a **single-node** MERGE pattern (one node, no relationship), matching
48//!   additionally consults a statement-local ledger of nodes created earlier in
49//!   the same statement, so repeated single-node MERGEs of the same key --
50//!   across multiple reading-clause rows OR multiple MERGE clauses -- dedup to a
51//!   single node (openCypher requires exactly one). A **relationship/path**
52//!   MERGE keeps whole-pattern semantics: the end node is (re)created per row
53//!   when the whole path is absent (the well-known openCypher MERGE-on-a-path
54//!   behavior), so it is *not* deduped against the ledger.
55//!
56//!   `ON` and `MERGE` are now **reserved keywords** (a backward-incompatible
57//!   consequence of adding the MERGE grammar): they can no longer be used as
58//!   bare identifiers, labels, relationship types, or property keys.
59//!
60//! Deferred to follow-ups and rejected cleanly: `REMOVE`, label mutation
61//! (`SET n:Label`), whole-entity replacement (`SET n = {...}`), variable-length
62//! relationships in a write, multi-label MERGE nodes, and aggregate/property
63//! `RETURN` projections.

85/// Execute a parsed Cypher write statement (Issue #560).
86///
87/// # Errors
88///
89/// Returns a query error for any unsupported construct (mapped from
90/// [`CypherError`]) or a storage/transaction error from the underlying write.
91pub fn execute(
92    db: &AletheiaDB,
93    statement: &CypherStatement,
94    params: &Params,
95) -> Result<QueryResults> {
96    let CypherStatement::Write(write) = statement else {
97        return Err(CypherError::UnsupportedFeature(
98            "mutation executor requires a write statement".to_string(),
99        )
100        .into());
101    };
102
103    // Static validation independent of the data (fails before any transaction).
104    validate(write)?;
105
106    // Cap the fan-out working set (Issue #3623 review): a MERGE that matches
107    // many committed entities -- or several chained MERGEs -- multiplies the
108    // working set combinatorially (up to N^k for k MERGEs over N matches each),
109    // and every binding buffers writes in this single transaction. Bound it with
110    // the SAME configurable `max_schema_as_of_entities` limit the read matcher
111    // uses (see `match_bindings` in `multi_pattern.rs`), turning a potential OOM
112    // into an honest structured rejection. Read once, up front.
113    let binding_cap = db.historical.read().max_schema_as_of_entities();
114
115    // Read phase: matched rows drive the write clauses. A statement with no
116    // reading part (a bare `CREATE`) runs once against a single empty binding.
117    let base_rows: Vec<Binding> = match &write.reading {
118        Some(reading) => {
119            match_bindings(db, &reading.pattern, reading.where_clause.as_ref(), params)?
120        }
121        None => vec![Vec::new()],
122    };
123
124    // Write phase: apply every clause per row inside ONE transaction so the whole
125    // statement commits all-or-nothing. RETURN snapshots are captured here and
126    // materialized (with fresh post-commit reads) afterwards.
127    let mut return_snapshots: Vec<Vec<(String, EntityResult)>> = Vec::new();
128    db.write::<_, (), Error>(|tx| {
129        // Track entities deleted by this statement so a later clause/row does not
130        // double-delete, and so the plain-DELETE safety check ignores edges this
131        // same statement already removed.
132        let mut deleted_nodes: HashSet<NodeId> = HashSet::new();
133        let mut deleted_edges: HashSet<EdgeId> = HashSet::new();
134        // Relationships CREATEd by THIS statement, as (edge, source, target).
135        // The committed adjacency reads used by the plain-DELETE safety check do
136        // not yet see a buffered same-statement CREATE, so this statement-local
137        // ledger lets a plain DELETE of such an endpoint refuse cleanly instead
138        // of aborting at commit with a cryptic dangling-edge error (Minor 1).
139        let mut created_edges: Vec<(EdgeId, NodeId, NodeId)> = Vec::new();
140        // Nodes CREATEd by THIS statement, as (node, label, resolved properties).
141        // A single-node MERGE consults this ledger (in addition to committed
142        // current state) so repeated single-node MERGEs of the same key across
143        // rows/clauses dedup to one node -- the committed `match_bindings` read
144        // never sees this transaction's own buffered creates (Core bug fix).
145        let mut created_nodes: Vec<(NodeId, String, PropertyMap)> = Vec::new();
146
147        for base in &base_rows {
148            // Working set of bindings for this base row. A clause maps the set:
149            // CREATE/SET/DELETE are 1->1, but a MERGE whose pattern matches
150            // multiple committed entities fans one binding into N (openCypher
151            // whole-pattern MATCH semantics, Issue #3623), so later clauses and
152            // RETURN run once per fanned row.
153            let mut rows: Vec<Binding> = vec![base.clone()];
154            for clause in &write.clauses {
155                rows = apply_clause(
156                    tx,
157                    db,
158                    clause,
159                    rows,
160                    params,
161                    &mut deleted_nodes,
162                    &mut deleted_edges,
163                    &mut created_edges,
164                    &mut created_nodes,
165                    binding_cap,
166                )?;
167                // Aggregate cap after each clause (Issue #3623 review): every
168                // clause is 1->1 except MERGE (which fans 1->N), so this bounds
169                // the whole working set as it grows clause-over-clause. Defense
170                // in depth alongside the mid-extend check inside the MERGE arm.
171                if rows.len() > binding_cap {
172                    return Err(fanout_cap_error(binding_cap).into());
173                }
174            }
175            if let Some(ret) = &write.return_clause {
176                // The RETURN order of fanned rows is intentionally left
177                // unasserted: openCypher gives no row order without ORDER BY, so
178                // the emission order of a multi-match MERGE's fan-out is not a
179                // contract (tests assert the SET of rows, not their sequence).
180                for binding in &rows {
181                    return_snapshots.push(collect_return_snapshots(ret, binding)?);
182                }
183            }
184        }
185        Ok(())
186    })?;
187
188    // Materialize RETURN rows after commit: re-read each entity for its
189    // post-write state, falling back to the captured snapshot for entities the
190    // statement deleted.
191    let mut rows = Vec::with_capacity(return_snapshots.len());
192    for snapshot in return_snapshots {
193        rows.push(materialize_row(db, snapshot));
194    }
195    Ok(QueryResults::from_rows(rows))
196}

202/// Reject unsupported write shapes up front, before any transaction opens.
203fn validate(write: &CypherWriteStatement) -> std::result::Result<(), CypherError> {
204    if let Some(reading) = &write.reading {
205        for pattern in &reading.pattern {
206            reject_varlength(pattern, "the reading MATCH of a write statement")?;
207        }
208    }
209    for clause in &write.clauses {
210        match clause {
211            CypherWriteClause::Create(patterns) => {
212                for pattern in patterns {
213                    reject_varlength(pattern, "a CREATE pattern")?;
214                    validate_create_pattern(pattern)?;
215                }
216            }
217            CypherWriteClause::Merge { pattern, .. } => {
218                reject_varlength(pattern, "a MERGE pattern")?;
219                validate_create_pattern(pattern)?;
220                validate_merge_pattern(pattern)?;
221            }
222            CypherWriteClause::Set(_) | CypherWriteClause::Delete { .. } => {}
223        }
224    }
225    if let Some(ret) = &write.return_clause {
226        validate_return(ret)?;
227    }
228    Ok(())
229}

231/// Reject a variable-length relationship anywhere in a pattern.
232fn reject_varlength(pattern: &CypherPattern, ctx: &str) -> std::result::Result<(), CypherError> {
233    for element in &pattern.elements {
234        if let CypherPatternElement::Relationship(rel) = element
235            && rel.depth.is_some()
236        {
237            return Err(CypherError::UnsupportedFeature(format!(
238                "variable-length relationships are not supported in {ctx}"
239            )));
240        }
241    }
242    Ok(())
243}
244
245/// Statically validate a `CREATE` pattern's relationships (direction + type).
246fn validate_create_pattern(pattern: &CypherPattern) -> std::result::Result<(), CypherError> {
247    for element in &pattern.elements {
248        if let CypherPatternElement::Relationship(rel) = element {
249            if matches!(rel.direction, CypherDirection::Both) {
250                return Err(CypherError::UnsupportedFeature(
251                    "CREATE requires a directed relationship (use -> or <-), not an undirected -"
252                        .to_string(),
253                ));
254            }
255            if rel.rel_types.len() != 1 {
256                return Err(CypherError::UnsupportedFeature(
257                    "CREATE requires exactly one relationship type, e.g. -[:KNOWS]->".to_string(),
258                ));
259            }
260        }
261    }
262    Ok(())
263}

265/// Statically validate a `MERGE` pattern's nodes: AletheiaDB nodes are
266/// single-labelled, so a multi-label node (`(n:A:B)`) can never be created and
267/// is rejected before any transaction opens. A node with zero labels is allowed
268/// (it must resolve to a variable already bound by a leading `MATCH`; that is
269/// checked at runtime).
270fn validate_merge_pattern(pattern: &CypherPattern) -> std::result::Result<(), CypherError> {
271    for element in &pattern.elements {
272        if let CypherPatternElement::Node(node) = element
273            && node.labels.len() > 1
274        {
275            return Err(CypherError::UnsupportedFeature(
276                "MERGE does not support multi-label nodes (AletheiaDB nodes are single-labelled)"
277                    .to_string(),
278            ));
279        }
280    }
281    Ok(())
282}

284/// Validate the `RETURN` of a write statement: only bound variables (bare or
285/// `AS`-aliased) or `*`, with no ordering/pagination/deduplication in v1.
286fn validate_return(ret: &CypherReturn) -> std::result::Result<(), CypherError> {
287    if ret.distinct || !ret.order_by.is_empty() || ret.skip.is_some() || ret.limit.is_some() {
288        return Err(CypherError::UnsupportedFeature(
289            "DISTINCT / ORDER BY / SKIP / LIMIT are not supported on the RETURN of a write \
290             statement in v1"
291                .to_string(),
292        ));
293    }
294    for item in &ret.items {
295        match item {
296            CypherReturnItem::Star | CypherReturnItem::Variable(_) => {}
297            CypherReturnItem::Expression {
298                expr: super::ast::CypherExpr::Variable(_),
299                ..
300            } => {}
301            _ => {
302                return Err(CypherError::UnsupportedFeature(
303                    "RETURN after a write supports only bound variables (optionally AS-aliased) \
304                     or *; property, function, and aggregate projections are not supported in v1"
305                        .to_string(),
306                ));
307            }
308        }
309    }
310    Ok(())
311}

317/// The structured rejection returned when a statement's fan-out working set
318/// exceeds the configured cap (Issue #3623 review). Mirrors the read matcher's
319/// [`match_bindings`] cap error shape (a [`CypherError::UnsupportedFeature`]
320/// keyed on the same `max_schema_as_of_entities` knob) so the write phase
321/// reports a combinatorial MERGE blow-up as an honest structured error rather
322/// than exhausting memory.
323fn fanout_cap_error(binding_cap: usize) -> CypherError {
324    CypherError::UnsupportedFeature(format!(
325        "MERGE fan-out working set exceeds {binding_cap} bindings; a MERGE matching \
326         many committed entities (or multiple chained MERGEs) multiplies the working \
327         set combinatorially, each binding buffering writes in one transaction -- add \
328         a more selective pattern or labels to narrow the match (configurable via \
329         max_schema_as_of_entities)"
330    ))
331}

333/// Apply one write clause to the whole working set of bindings, returning the
334/// resulting set. Most clauses are 1->1 (each input binding maps to itself after
335/// its side effects), but a MERGE whose pattern matches multiple committed
336/// entities fans one binding into N (Issue #3623), so this maps `Vec<Binding>`
337/// to `Vec<Binding>` rather than mutating a single binding in place. `binding_cap`
338/// bounds the MERGE fan-out mid-extend (Issue #3623 review).
339#[allow(clippy::too_many_arguments)]
340fn apply_clause(
341    tx: &mut crate::api::transaction::WriteTransaction,
342    db: &AletheiaDB,
343    clause: &CypherWriteClause,
344    rows: Vec<Binding>,
345    params: &Params,
346    deleted_nodes: &mut HashSet<NodeId>,
347    deleted_edges: &mut HashSet<EdgeId>,
348    created_edges: &mut Vec<(EdgeId, NodeId, NodeId)>,
349    created_nodes: &mut Vec<(NodeId, String, PropertyMap)>,
350    binding_cap: usize,
351) -> Result<Vec<Binding>> {
352    match clause {
353        CypherWriteClause::Create(patterns) => {
354            let mut out = Vec::with_capacity(rows.len());
355            for mut binding in rows {
356                for pattern in patterns {
357                    create_pattern(
358                        tx,
359                        pattern,
360                        &mut binding,
361                        params,
362                        created_edges,
363                        created_nodes,
364                    )?;
365                }
366                out.push(binding);
367            }
368            Ok(out)
369        }
370        CypherWriteClause::Set(items) => {
371            let mut out = Vec::with_capacity(rows.len());
372            for binding in rows {
373                apply_set(tx, items, &binding, params, deleted_nodes, deleted_edges)?;
374                out.push(binding);
375            }
376            Ok(out)
377        }
378        CypherWriteClause::Delete { detach, targets } => {
379            let mut out = Vec::with_capacity(rows.len());
380            for binding in rows {
381                apply_delete(
382                    tx,
383                    *detach,
384                    targets,
385                    &binding,
386                    deleted_nodes,
387                    deleted_edges,
388                    created_edges,
389                )?;
390                out.push(binding);
391            }
392            Ok(out)
393        }
394        CypherWriteClause::Merge {
395            pattern,
396            on_create,
397            on_match,
398        } => {
399            let mut out = Vec::with_capacity(rows.len());
400            for binding in rows {
401                out.extend(apply_merge(
402                    tx,
403                    db,
404                    pattern,
405                    on_create,
406                    on_match,
407                    binding,
408                    params,
409                    deleted_nodes,
410                    deleted_edges,
411                    created_edges,
412                    created_nodes,
413                )?);
414                // Bound the fan-out mid-extend (Issue #3623 review): a single
415                // MERGE matching many committed entities across many input
416                // bindings could otherwise balloon `out` to ~cap^2 before the
417                // caller's post-clause check runs. Reject as soon as it exceeds
418                // the cap, matching the read matcher's per-step bound.
419                if out.len() > binding_cap {
420                    return Err(fanout_cap_error(binding_cap).into());
421                }
422            }
423            Ok(out)
424        }
425    }
426}

428/// Apply a `MERGE` clause (Issue #3548): match the pattern if it already exists,
429/// otherwise create the *entire* pattern (openCypher whole-pattern semantics).
430///
431/// The match is a pre-transaction (check-then-act) read against committed
432/// current state via [`match_bindings`], filtered to rows consistent with the
433/// variables already bound in this row (so a leading variable bound by a prior
434/// `MATCH` constrains the match rather than re-scanning freely).
435///
436/// # Bi-temporal correctness
437///
438/// - A bare **match** records **no** new version: `on_match` is empty, so
439///   [`apply_set`] performs no update.
440/// - A **create** records new versions (one per created node/edge).
441/// - `ON MATCH SET` on a matched entity records a new version (it is an update).
442/// - `ON CREATE SET` folds into the same first version window as the create's
443///   own `SET` coalescing (a separate `update_node`, mirroring `CREATE ... SET`).
444///
445/// # Same-statement dedup (single-node MERGE)
446///
447/// The committed [`match_bindings`] read cannot see this transaction's own
448/// buffered creates, so a naive per-row MERGE would create openCypher-illegal
449/// duplicates for a **single-node** pattern (e.g.
450/// `MATCH (p:Person) MERGE (c:City {name:'NYC'})` over N persons must yield ONE
451/// NYC, not N). To fix this, when the pattern is a single node and no committed
452/// match exists, we also consult a statement-local ledger of nodes created
453/// earlier in this same statement (`created_nodes`), matching by label and the
454/// pattern's specified properties. A ledger hit is treated as a MATCH (bind it,
455/// apply `on_match`, no new node/version); a miss creates and records the node.
456///
457/// A **relationship/path** MERGE deliberately does NOT dedup against the ledger:
458/// openCypher matches the *whole path*, so
459/// `MATCH (p:Person) MERGE (p)-[:R]->(c:City {name:'NYC'})` correctly creates
460/// one NYC per person (a NYC connected only to p1 does not satisfy p2's path).
461///
462/// # Multi-match binding (Issue #3623)
463///
464/// A MERGE whose pattern matches MORE THAN ONE committed entity binds **all**
465/// matches (openCypher whole-pattern MATCH semantics): it fans the input
466/// binding into one output binding per match and applies `ON MATCH SET` to
467/// every one. The write executor carries a working set of bindings
468/// (`Vec<Binding>`) precisely so this fan-out composes with subsequent clauses
469/// and `RETURN` (each fanned row runs the rest of the statement independently).
470///
471/// ## Combinatorial cost and the aggregate cap
472///
473/// Fan-out is **multiplicative**: `M` base rows times `N` matches per MERGE, and
474/// `k` chained MERGEs can grow the working set to `M * N^k` bindings, each
475/// buffering its own writes inside the one transaction. To keep a pathological
476/// statement from exhausting memory, the working set is bounded by the same
477/// configurable `max_schema_as_of_entities` limit the read matcher applies to
478/// its intermediate binding count. The bound is checked mid-extend here and
479/// again after each clause in [`execute`]; exceeding it returns the structured
480/// [`fanout_cap_error`] ([`CypherError::UnsupportedFeature`]) rather than an OOM.
481///
482/// ## Bi-temporal multi-versioning of a single entity
483///
484/// Because `ON MATCH SET` (and any following write clause) runs **once per
485/// fanned row**, a single statement can record MULTIPLE new bi-temporal
486/// versions of ONE entity when that entity is bound by more than one match/row
487/// (e.g. a node reached by two matched relationship paths, or the far node of a
488/// base-row cross product). Each per-row update is a distinct `update_node`, so
489/// the entity is versioned once per binding -- a known deviation from
490/// openCypher's set-once-per-write semantics that is documented, not deduped, in
491/// v1.
492///
493/// # v1 concurrency
494///
495/// Without a unique constraint on the merged property, the pre-transaction
496/// match is a documented check-then-act window (mirrors Issue #560): two racing
497/// MERGE-creates can both create. Declaring
498/// `db.unique_constraint(label, prop).enable()` closes the window --- the second
499/// committer aborts (first-committer-wins).
500#[allow(clippy::too_many_arguments)]
501fn apply_merge(
502    tx: &mut crate::api::transaction::WriteTransaction,
503    db: &AletheiaDB,
504    pattern: &CypherPattern,
505    on_create: &[CypherSetItem],
506    on_match: &[CypherSetItem],
507    binding: Binding,
508    params: &Params,
509    deleted_nodes: &HashSet<NodeId>,
510    deleted_edges: &HashSet<EdgeId>,
511    created_edges: &mut Vec<(EdgeId, NodeId, NodeId)>,
512    created_nodes: &mut Vec<(NodeId, String, PropertyMap)>,
513) -> Result<Vec<Binding>> {
514    // Pre-transaction match against committed current state, filtered to rows
515    // consistent with variables already bound in this row. Fully drained into a
516    // Vec before any write so no read borrow is held across the tx mutations.
517    let committed: Vec<Binding> = match_bindings(db, std::slice::from_ref(pattern), None, params)?
518        .into_iter()
519        .filter(|row| consistent_with(row, &binding))
520        .collect();
521
522    // MATCH branch (Issue #3623): a pattern matching one OR MORE committed
523    // entities binds ALL of them (openCypher whole-pattern MATCH semantics),
524    // fanning the input binding into one output binding per match and applying
525    // ON MATCH SET to every one. A bare match (empty on_match) records nothing.
526    if !committed.is_empty() {
527        let mut out = Vec::with_capacity(committed.len());
528        for row in committed {
529            let mut b = binding.clone();
530            for (name, entity) in row {
531                bind(&mut b, &name, entity);
532            }
533            apply_set(tx, on_match, &b, params, deleted_nodes, deleted_edges)?;
534            out.push(b);
535        }
536        return Ok(out);
537    }
538
539    // No committed match: a single output binding (create or ledger-match).
540    let mut binding = binding;
541
542    // For a single-node pattern that would be created (an unbound/labelled
543    // node), consult this statement's created-node ledger so repeated
544    // single-node MERGEs of the same key dedup to one node.
545    if let Some(node) = single_creatable_node(pattern, &binding)
546        && let Some(id) = find_created_single_node(node, created_nodes, params)?
547    {
548        // Ledger MATCH branch: bind the buffered-created node and apply
549        // ON MATCH SET only (no create, no whole-pattern duplicate).
550        if let Some(var) = &node.variable {
551            bind(&mut binding, var, EntityResult::NodeId(id));
552        }
553        apply_set(tx, on_match, &binding, params, deleted_nodes, deleted_edges)?;
554        return Ok(vec![binding]);
555    }
556
557    // CREATE branch: create the whole pattern (a bound leading variable is
558    // reused; the unbound remainder is created), then ON CREATE SET.
559    create_pattern(
560        tx,
561        pattern,
562        &mut binding,
563        params,
564        created_edges,
565        created_nodes,
566    )?;
567    apply_set(
568        tx,
569        on_create,
570        &binding,
571        params,
572        deleted_nodes,
573        deleted_edges,
574    )?;
575    Ok(vec![binding])
576}

578/// If a MERGE pattern is a single node that would be *created* (exactly one
579/// label and not an already-bound variable), return that node element; else
580/// `None`. A bound leading variable (`MATCH (a) MERGE (a) ...`) already resolves
581/// to an existing node, and a relationship/path pattern keeps whole-pattern
582/// semantics, so neither participates in single-node ledger dedup.
583fn single_creatable_node<'p>(
584    pattern: &'p CypherPattern,
585    binding: &Binding,
586) -> Option<&'p CypherNodePattern> {
587    let [CypherPatternElement::Node(node)] = pattern.elements.as_slice() else {
588        return None;
589    };
590    if node.labels.len() != 1 {
591        return None;
592    }
593    if let Some(var) = &node.variable
594        && lookup(binding, var).is_some()
595    {
596        return None; // already bound to an existing node
597    }
598    Some(node)
599}

601/// Find a node created earlier in this same statement whose label equals the
602/// pattern node's label and whose stored properties satisfy every property the
603/// pattern node specifies. Mirrors the current-state matcher's label+property
604/// check against buffered creates.
605fn find_created_single_node(
606    node: &CypherNodePattern,
607    created_nodes: &[(NodeId, String, PropertyMap)],
608    params: &Params,
609) -> Result<Option<NodeId>> {
610    let want_label = &node.labels[0];
611    for (id, label, props) in created_nodes {
612        if label != want_label {
613            continue;
614        }
615        if node_props_satisfied(&node.properties, props, params)? {
616            return Ok(Some(*id));
617        }
618    }
619    Ok(None)
620}

622/// Whether every `(key, value)` the pattern node specifies is present in a
623/// created node's stored property map and equal (with int/float coercion,
624/// matching the current-state matcher's `loosely_equal`).
625fn node_props_satisfied(
626    props: &[(String, CypherValue)],
627    stored: &PropertyMap,
628    params: &Params,
629) -> Result<bool> {
630    for (key, want) in props {
631        let want = cypher_value_to_property(want, params)?;
632        match stored.get(key) {
633            Some(actual) if loosely_equal(actual, &want) => {}
634            _ => return Ok(false),
635        }
636    }
637    Ok(true)
638}

640/// A matched candidate row is usable only if it agrees with every variable
641/// already bound in the current row: for each shared variable the entity ids
642/// must be identical. Variables bound only in the candidate (the freshly
643/// matched remainder) impose no constraint.
644fn consistent_with(row: &Binding, binding: &Binding) -> bool {
645    binding
646        .iter()
647        .all(|(name, entity)| match lookup(row, name) {
648            Some(other) => entity_same(entity, other),
649            None => true,
650        })
651}

653/// Two [`EntityResult`]s refer to the same graph entity (same node id or same
654/// edge id).
655fn entity_same(a: &EntityResult, b: &EntityResult) -> bool {
656    if let (Some(x), Some(y)) = (node_id_of(a), node_id_of(b)) {
657        return x == y;
658    }
659    if let (Some(x), Some(y)) = (edge_id_of(a), edge_id_of(b)) {
660        return x == y;
661    }
662    false
663}

665/// Create the nodes and relationships described by one `CREATE` pattern,
666/// binding each named element into `binding`.
667fn create_pattern(
668    tx: &mut crate::api::transaction::WriteTransaction,
669    pattern: &CypherPattern,
670    binding: &mut Binding,
671    params: &Params,
672    created_edges: &mut Vec<(EdgeId, NodeId, NodeId)>,
673    created_nodes: &mut Vec<(NodeId, String, PropertyMap)>,
674) -> Result<()> {
675    let mut prev_node: Option<NodeId> = None;
676    let mut idx = 0;
677    while idx < pattern.elements.len() {
678        match &pattern.elements[idx] {
679            CypherPatternElement::Node(node) => {
680                let id = resolve_or_create_node(tx, node, binding, params, created_nodes)?;
681                prev_node = Some(id);
682                idx += 1;
683            }
684            CypherPatternElement::Relationship(rel) => {
685                let left = prev_node.ok_or_else(|| {
686                    Error::from(CypherError::SemanticError(
687                        "a relationship must be preceded by a node in a CREATE pattern".to_string(),
688                    ))
689                })?;
690                let Some(CypherPatternElement::Node(right_pat)) = pattern.elements.get(idx + 1)
691                else {
692                    return Err(CypherError::SemanticError(
693                        "a relationship must be followed by a node in a CREATE pattern".to_string(),
694                    )
695                    .into());
696                };
697                let right = resolve_or_create_node(tx, right_pat, binding, params, created_nodes)?;
698
699                // Direction/type validated statically; exactly one type present.
700                let rel_type = &rel.rel_types[0];
701                let (source, target) = match rel.direction {
702                    CypherDirection::Outgoing => (left, right),
703                    CypherDirection::Incoming => (right, left),
704                    CypherDirection::Both => {
705                        return Err(CypherError::UnsupportedFeature(
706                            "CREATE requires a directed relationship".to_string(),
707                        )
708                        .into());
709                    }
710                };
711                let props = build_props(&rel.properties, params)?;
712                let edge_id = tx.create_edge(source, target, rel_type, props)?;
713                created_edges.push((edge_id, source, target));
714                if let Some(var) = &rel.variable {
715                    bind(binding, var, EntityResult::EdgeId(edge_id));
716                }
717                prev_node = Some(right);
718                idx += 2;
719            }
720        }
721    }
722    Ok(())
723}

725/// Resolve a node element to an existing bound node, or create a new node and
726/// bind it. A bound variable is reused; an unbound/anonymous node is created and
727/// must carry exactly one label.
728fn resolve_or_create_node(
729    tx: &mut crate::api::transaction::WriteTransaction,
730    node: &CypherNodePattern,
731    binding: &mut Binding,
732    params: &Params,
733    created_nodes: &mut Vec<(NodeId, String, PropertyMap)>,
734) -> Result<NodeId> {
735    if let Some(var) = &node.variable
736        && let Some(existing) = lookup(binding, var)
737    {
738        // Reusing a bound variable: it must already be a node, and CREATE must
739        // not also (re)specify a label/properties on it.
740        let Some(id) = node_id_of(existing) else {
741            return Err(CypherError::SemanticError(format!(
742                "variable '{var}' is bound to a relationship and cannot be used as a node in CREATE"
743            ))
744            .into());
745        };
746        if !node.labels.is_empty() || !node.properties.is_empty() {
747            return Err(CypherError::UnsupportedFeature(format!(
748                "CREATE cannot re-declare labels or properties on already-bound variable '{var}'"
749            ))
750            .into());
751        }
752        return Ok(id);
753    }
754
755    // Creating a new node: AletheiaDB nodes are single-labelled.
756    if node.labels.len() != 1 {
757        return Err(CypherError::UnsupportedFeature(
758            "CREATE requires exactly one label per new node (AletheiaDB nodes are single-labelled)"
759                .to_string(),
760        )
761        .into());
762    }
763    let props = build_props(&node.properties, params)?;
764    let id = tx.create_node(&node.labels[0], props.clone())?;
765    // Record in the statement-local ledger so a later single-node MERGE of the
766    // same (label, properties) key can dedup to this buffered-created node.
767    created_nodes.push((id, node.labels[0].clone(), props));
768    if let Some(var) = &node.variable {
769        bind(binding, var, EntityResult::NodeId(id));
770    }
771    Ok(id)
772}

774/// Apply a `SET` clause: PATCH-merge each `n.prop = value` onto its bound
775/// entity. Assignments to the same variable are coalesced into one update so a
776/// multi-property `SET` records a single new version.
777fn apply_set(
778    tx: &mut crate::api::transaction::WriteTransaction,
779    items: &[CypherSetItem],
780    binding: &Binding,
781    params: &Params,
782    deleted_nodes: &HashSet<NodeId>,
783    deleted_edges: &HashSet<EdgeId>,
784) -> Result<()> {
785    // Accumulate per-target property maps in first-seen order.
786    let mut node_updates: Vec<(NodeId, PropertyMapBuilder)> = Vec::new();
787    let mut edge_updates: Vec<(EdgeId, PropertyMapBuilder)> = Vec::new();
788
789    for item in items {
790        let entity = lookup(binding, &item.variable).ok_or_else(|| {
791            Error::from(CypherError::SemanticError(format!(
792                "SET references unbound variable '{}'",
793                item.variable
794            )))
795        })?;
796        let value = cypher_value_to_property(&item.value, params)?;
797
798        if let Some(node_id) = node_id_of(entity) {
799            if deleted_nodes.contains(&node_id) {
800                continue; // no-op on an entity this statement already deleted
801            }
802            let slot = match node_updates.iter_mut().find(|(id, _)| *id == node_id) {
803                Some((_, builder)) => builder,
804                None => {
805                    node_updates.push((node_id, PropertyMapBuilder::new()));
806                    &mut node_updates.last_mut().expect("just pushed").1
807                }
808            };
809            take_insert(slot, &item.property, value);
810        } else if let Some(edge_id) = edge_id_of(entity) {
811            if deleted_edges.contains(&edge_id) {
812                continue; // no-op on an edge this statement already deleted
813            }
814            let slot = match edge_updates.iter_mut().find(|(id, _)| *id == edge_id) {
815                Some((_, builder)) => builder,
816                None => {
817                    edge_updates.push((edge_id, PropertyMapBuilder::new()));
818                    &mut edge_updates.last_mut().expect("just pushed").1
819                }
820            };
821            take_insert(slot, &item.property, value);
822        } else {
823            return Err(CypherError::SemanticError(format!(
824                "SET target variable '{}' is not bound to a node or relationship",
825                item.variable
826            ))
827            .into());
828        }
829    }
830
831    for (node_id, builder) in node_updates {
832        tx.update_node(node_id, builder.build())?;
833    }
834    for (edge_id, builder) in edge_updates {
835        tx.update_edge(edge_id, builder.build())?;
836    }
837    Ok(())
838}

840/// Apply a `DELETE` / `DETACH DELETE` clause.
841///
842/// Relationships are deleted before nodes so that `DELETE r, a` and `DELETE a, r`
843/// behave identically (openCypher deletes are order-independent). A plain
844/// `DELETE` of a node that still has relationships (excluding any this statement
845/// already deleted) is refused; `DETACH DELETE` cascade-removes them.
846fn apply_delete(
847    tx: &mut crate::api::transaction::WriteTransaction,
848    detach: bool,
849    targets: &[String],
850    binding: &Binding,
851    deleted_nodes: &mut HashSet<NodeId>,
852    deleted_edges: &mut HashSet<EdgeId>,
853    created_edges: &[(EdgeId, NodeId, NodeId)],
854) -> Result<()> {
855    // Partition targets into edges and nodes.
856    let mut edge_targets: Vec<EdgeId> = Vec::new();
857    let mut node_targets: Vec<NodeId> = Vec::new();
858    for var in targets {
859        let entity = lookup(binding, var).ok_or_else(|| {
860            Error::from(CypherError::SemanticError(format!(
861                "DELETE references unbound variable '{var}'"
862            )))
863        })?;
864        if let Some(edge_id) = edge_id_of(entity) {
865            edge_targets.push(edge_id);
866        } else if let Some(node_id) = node_id_of(entity) {
867            node_targets.push(node_id);
868        } else {
869            return Err(CypherError::SemanticError(format!(
870                "DELETE target variable '{var}' is not bound to a node or relationship"
871            ))
872            .into());
873        }
874    }
875
876    // Delete relationships first.
877    for edge_id in edge_targets {
878        if deleted_edges.contains(&edge_id) || tx.get_edge(edge_id).is_err() {
879            continue; // already gone
880        }
881        tx.delete_edge(edge_id)?;
882        deleted_edges.insert(edge_id);
883    }
884
885    // Then nodes.
886    for node_id in node_targets {
887        if deleted_nodes.contains(&node_id) || tx.get_node(node_id).is_err() {
888            continue;
889        }
890        if detach {
891            // Record the connected edges as deleted so a later plain DELETE does
892            // not mis-count them, then cascade. `delete_node_cascade` itself
893            // unions committed adjacency with same-transaction buffered CREATEs,
894            // so a relationship created earlier in this statement is removed too;
895            // we mirror that here into `deleted_edges` for the plain-DELETE check.
896            for edge_id in connected_edges(tx, node_id)? {
897                deleted_edges.insert(edge_id);
898            }
899            for (edge_id, source, target) in created_edges {
900                if *source == node_id || *target == node_id {
901                    deleted_edges.insert(*edge_id);
902                }
903            }
904            tx.delete_node_cascade(node_id)?;
905        } else {
906            // openCypher safety rule: refuse to orphan edges. Ignore edges this
907            // same statement already deleted. Committed adjacency does not yet
908            // include a relationship CREATEd earlier in this same statement
909            // (it is still buffered), so fold the statement-local created-edge
910            // ledger in too — otherwise this plain DELETE would slip past the
911            // check and abort at commit with a cryptic dangling-edge error
912            // (Minor 1).
913            let mut remaining: Vec<EdgeId> = connected_edges(tx, node_id)?
914                .into_iter()
915                .filter(|e| !deleted_edges.contains(e))
916                .collect();
917            for (edge_id, source, target) in created_edges {
918                if (*source == node_id || *target == node_id)
919                    && !deleted_edges.contains(edge_id)
920                    && !remaining.contains(edge_id)
921                {
922                    remaining.push(*edge_id);
923                }
924            }
925            if !remaining.is_empty() {
926                return Err(TransactionError::ValidationFailed {
927                    reason: format!(
928                        "Cannot delete node {} because it still has {} relationship(s); use \
929                         DETACH DELETE to remove the node and its relationships",
930                        node_id.as_u64(),
931                        remaining.len()
932                    ),
933                }
934                .into());
935            }
936            tx.delete_node(node_id)?;
937        }
938        deleted_nodes.insert(node_id);
939    }
940    Ok(())
941}

943/// Distinct edge ids connected to a node (a self-loop counts once).
944fn connected_edges(
945    tx: &mut crate::api::transaction::WriteTransaction,
946    node_id: NodeId,
947) -> Result<Vec<EdgeId>> {
948    let mut ids = tx.get_outgoing_edges(node_id)?;
949    ids.extend(tx.get_incoming_edges(node_id)?);
950    ids.sort_unstable();
951    ids.dedup();
952    Ok(ids)
953}

959/// Capture the bound entities a `RETURN` projects, as `(output_name, snapshot)`
960/// pairs, from the row's final binding. Snapshots are materialized (re-read)
961/// after commit.
962fn collect_return_snapshots(
963    ret: &CypherReturn,
964    binding: &Binding,
965) -> Result<Vec<(String, EntityResult)>> {
966    let mut out = Vec::new();
967    for item in &ret.items {
968        match item {
969            CypherReturnItem::Star => {
970                for (name, entity) in binding {
971                    out.push((name.clone(), entity.clone()));
972                }
973            }
974            CypherReturnItem::Variable(name) => {
975                out.push((name.clone(), lookup_owned(binding, name)?));
976            }
977            CypherReturnItem::Expression {
978                expr: super::ast::CypherExpr::Variable(name),
979                alias,
980            } => {
981                let output = alias.clone().unwrap_or_else(|| name.clone());
982                out.push((output, lookup_owned(binding, name)?));
983            }
984            _ => {
985                return Err(CypherError::UnsupportedFeature(
986                    "RETURN after a write supports only bound variables or *".to_string(),
987                )
988                .into());
989            }
990        }
991    }
992    Ok(out)
993}

995/// Materialize one RETURN row: re-read each captured entity for its post-write
996/// state, falling back to the captured snapshot when the entity was deleted.
997fn materialize_row(db: &AletheiaDB, snapshot: Vec<(String, EntityResult)>) -> QueryRow {
998    let fresh: Vec<(String, EntityResult)> = snapshot
999        .into_iter()
1000        .map(|(name, entity)| (name, refresh_entity(db, entity)))
1001        .collect();
1002
1003    if fresh.len() == 1 {
1004        let (_, entity) = fresh.into_iter().next().expect("len == 1");
1005        QueryRow::from_entity(entity)
1006    } else {
1007        QueryRow::from_bindings(fresh, None)
1008    }
1009}

1011/// Re-read an entity's current state by id, keeping the snapshot if it is gone.
1012fn refresh_entity(db: &AletheiaDB, entity: EntityResult) -> EntityResult {
1013    if let Some(node_id) = node_id_of(&entity) {
1014        return db
1015            .get_node(node_id)
1016            .map(EntityResult::Node)
1017            .unwrap_or(entity);
1018    }
1019    if let Some(edge_id) = edge_id_of(&entity) {
1020        return db
1021            .get_edge(edge_id)
1022            .map(EntityResult::Edge)
1023            .unwrap_or(entity);
1024    }
1025    entity
1026}

1032/// Build a [`PropertyMap`] from pattern/`SET` `(key, value)` pairs.
1033fn build_props(
1034    props: &[(String, CypherValue)],
1035    params: &Params,
1036) -> std::result::Result<PropertyMap, CypherError> {
1037    let mut builder = PropertyMapBuilder::new();
1038    for (key, value) in props {
1039        let pv = cypher_value_to_property(value, params)?;
1040        builder = builder.insert(key, pv);
1041    }
1042    Ok(builder.build())
1043}

1045/// Insert into a builder held behind a mutable reference (builder methods are
1046/// by-value, so swap through a temporary).
1047fn take_insert(
1048    slot: &mut PropertyMapBuilder,
1049    key: &str,
1050    value: crate::core::property::PropertyValue,
1051) {
1052    let taken = std::mem::take(slot);
1053    *slot = taken.insert(key, value);
1054}

1056/// Look up a variable's binding by name.
1057fn lookup<'b>(binding: &'b Binding, var: &str) -> Option<&'b EntityResult> {
1058    binding.iter().find(|(name, _)| name == var).map(|(_, e)| e)
1059}

1061/// Look up a variable, cloning its binding, erroring if unbound.
1062fn lookup_owned(binding: &Binding, var: &str) -> Result<EntityResult> {
1063    lookup(binding, var).cloned().ok_or_else(|| {
1064        CypherError::SemanticError(format!("RETURN references unbound variable '{var}'")).into()
1065    })
1066}

1068/// Bind (or rebind) a variable to an entity.
1069fn bind(binding: &mut Binding, var: &str, entity: EntityResult) {
1070    if let Some(slot) = binding.iter_mut().find(|(name, _)| name == var) {
1071        slot.1 = entity;
1072    } else {
1073        binding.push((var.to_string(), entity));
1074    }
1075}

1079    match entity {
1080        EntityResult::Node(n) => Some(n.id),
1081        EntityResult::NodeId(id) => Some(*id),
1082        _ => None,
