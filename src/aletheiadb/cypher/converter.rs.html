1//! Cypher AST → Query IR Converter
2//!
3//! Converts a parsed [`CypherStatement`] AST into the internal [`Query`] IR
4//! that the query planner and executor understand. This bridges the gap between
5//! the user-facing Cypher syntax and AletheiaDB's native query pipeline.
6//!
7//! # Architecture
8//!
9//! ```text
10//! Cypher String ─► [Lexer] ─► Tokens ─► [Parser] ─► CypherStatement
11//!                                                          │
12//!                                                   [CypherConverter]
13//!                                                          │
14//!                                                          ▼
15//!                                                    Query { ops, hints }
16//! ```
17//!
18//! # Usage
19//!
20//! ```rust
21//! use aletheiadb::cypher::{parse_cypher, parse_cypher_with_params, CypherParameterValue};
22//! use std::collections::HashMap;
23//!
24//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
25//! // Simple: no parameters
26//! let query = parse_cypher("MATCH (n:Person) RETURN n")?;
27//!
28//! // With parameters
29//! let mut params = HashMap::new();
30//! params.insert("name".to_string(), CypherParameterValue::String("Alice".into()));
31//! let query = parse_cypher_with_params("MATCH (n:Person {name: $name}) RETURN n", params)?;
32//! # Ok(())
33//! # }
34//! ```

55/// Defensive recursion bound for expression → predicate conversion.
56///
57/// The parser already caps expression nesting at 128, so for any AST it
58/// produces this never fires. It exists solely because `convert` is `pub`: a
59/// caller constructing a `CypherExpr` by hand (bypassing the parser) could nest
60/// `Not`/`Grouped` arbitrarily deep and overflow the stack during conversion.
61/// Kept clearly above the parser's cap so it only ever rejects such hand-built
62/// pathological input, never a legitimately parsed query.
63const MAX_CONVERT_DEPTH: usize = 256;

65/// Map a Cypher [`CypherCompOp`] to the IR [`ProvenanceCmp`] (Issue #3354b).
66fn cypher_comp_to_provenance_cmp(op: CypherCompOp) -> ProvenanceCmp {
67    match op {
68        CypherCompOp::Eq => ProvenanceCmp::Eq,
69        CypherCompOp::Ne => ProvenanceCmp::Ne,
70        CypherCompOp::Lt => ProvenanceCmp::Lt,
71        CypherCompOp::Le => ProvenanceCmp::Le,
72        CypherCompOp::Gt => ProvenanceCmp::Gt,
73        CypherCompOp::Ge => ProvenanceCmp::Ge,
74    }
75}

77/// Flip a Cypher comparison operator when its operands are swapped, so
78/// `value <op> x` evaluates identically to `x <flipped> value` (an accessor on
79/// the right-hand side lowers like the accessor-on-left form). Equality/
80/// inequality are symmetric (Issue #3354b). Shared by every Cypher operand-swap
81/// site (provenance lowering, vector-score comparison, and the edge-property
82/// evaluator, Issue #3622).
83pub(crate) fn flip_cypher_comp_op(op: CypherCompOp) -> CypherCompOp {
84    match op {
85        CypherCompOp::Eq => CypherCompOp::Eq,
86        CypherCompOp::Ne => CypherCompOp::Ne,
87        CypherCompOp::Lt => CypherCompOp::Gt,
88        CypherCompOp::Le => CypherCompOp::Ge,
89        CypherCompOp::Gt => CypherCompOp::Lt,
90        CypherCompOp::Ge => CypherCompOp::Le,
91    }
92}

94/// Map the surface-agnostic [`ProvenanceLoweringError`] into the Cypher error
95/// vocabulary (Issue #3354b).
96///
97/// Chosen so the MCP `query` tool surfaces the same structured kinds as the AQL
98/// side (via [`CypherError`] → `crate::core::error::Error` → `map_query_error`):
99/// type mismatches and out-of-range confidence become
100/// [`CypherError::ParameterError`] (→ `invalid_params`); the ordering-op-on-
101/// string rejection becomes [`CypherError::SemanticError`] (→ `parse_error`).
102fn map_provenance_lowering_error(e: ProvenanceLoweringError) -> CypherError {
103    match e {
104        ProvenanceLoweringError::ConfidenceOutOfRange { value, .. } => CypherError::ParameterError(
105            format!("confidence(x) must be between 0.0 and 1.0, got {value}"),
106        ),
107        ProvenanceLoweringError::ConfidenceNotNumeric => CypherError::ParameterError(
108            "type mismatch: confidence(x) compares only to a numeric value".to_string(),
109        ),
110        ProvenanceLoweringError::OrderingOpOnStringField { field } => {
111            CypherError::SemanticError(format!(
112                "ordering operators (<, <=, >, >=) are not supported for {}(x); \
113                 only = and <> are supported for source/reason",
114                field.accessor_name()
115            ))
116        }
117        ProvenanceLoweringError::StringFieldNotString { field } => {
118            CypherError::ParameterError(format!(
119                "type mismatch: {}(x) compares only to a string value",
120                field.accessor_name()
121            ))
122        }
123        ProvenanceLoweringError::FilterValidation { message, .. } => {
124            CypherError::ParameterError(message)
125        }
126    }
127}

133/// A runtime parameter value that can be bound to `$param` references in Cypher
134/// queries.
135///
136/// Parameters allow safe, injection-free query execution with dynamic values.
137#[derive(Debug, Clone, PartialEq)]
138pub enum CypherParameterValue {
139    /// SQL/Cypher NULL.
140    Null,
141    /// A boolean value.
142    Bool(bool),
143    /// A 64-bit signed integer.
144    Int(i64),
145    /// A 64-bit floating-point number.
146    Float(f64),
147    /// A UTF-8 string.
148    String(String),
149    /// A dense vector embedding for similarity search.
150    Embedding(Arc<[f32]>),
151    /// A homogeneous or heterogeneous list of parameter values.
152    ///
153    /// Bound to the `$list` position of a standalone `UNWIND $list AS x`
154    /// (Issue #559). Lists are not valid as scalar predicate values.
155    List(Vec<CypherParameterValue>),
156}
157
158impl CypherParameterValue {
159    /// Convert this parameter value into a [`PredicateValue`] for use in
160    /// query filters.
161    ///
162    /// # Errors
163    ///
164    /// Returns [`CypherError::ParameterError`] if the value type cannot be
165    /// converted (e.g., embeddings are not valid predicate values).
166    fn to_predicate_value(&self) -> Result<PredicateValue, CypherError> {
167        match self {
168            CypherParameterValue::Null => Ok(PredicateValue::Null),
169            CypherParameterValue::Bool(b) => Ok(PredicateValue::Bool(*b)),
170            CypherParameterValue::Int(n) => Ok(PredicateValue::Int(*n)),
171            CypherParameterValue::Float(f) => Ok(PredicateValue::Float(*f)),
172            CypherParameterValue::String(s) => Ok(PredicateValue::String(s.clone())),
173            CypherParameterValue::Embedding(_) => Err(CypherError::ParameterError(
174                "embedding parameters cannot be used as predicate values".to_string(),
175            )),
176            CypherParameterValue::List(_) => Err(CypherError::ParameterError(
177                "list parameters cannot be used as predicate values".to_string(),
178            )),
179        }
180    }
181}

183/// The projection shape of a `WITH` clause, as resolved for the positional
184/// pipeline (Issue #556). The pipeline carries a single entity per row, so a
185/// `WITH` either carries everything (`*`) or carries the current binding under
186/// its original or an aliased name.
187enum WithProjection {
188    /// `WITH *` -- carry every visible variable forward unchanged.
189    All,
190    /// `WITH <source>` or `WITH <source> AS <output>` -- carry the current
191    /// positional binding forward, renaming it to `output`.
192    Single {
193        /// The source variable being projected (must be the current binding).
194        source: String,
195        /// The output name it is bound to downstream (equal to `source` when
196        /// no `AS` alias is given).
197        output: String,
198    },
199}

205/// Converts a Cypher AST ([`CypherStatement`]) into a [`Query`] suitable for
206/// the AletheiaDB query planner.
207///
208/// The converter walks the AST and emits a flat sequence of [`QueryOp`]s that
209/// describe the query pipeline:
210///
211/// 1. A `ScanNodes` op for the first node in each pattern.
212/// 2. A `TraverseOut`/`TraverseIn`/`TraverseBoth` op for each relationship.
213/// 3. `Filter` ops for inline property constraints and `WHERE` clauses.
214/// 4. `Distinct`, `Sort`, `Skip`, `Limit` ops from the `RETURN` clause.
215pub struct CypherConverter {
216    /// Bound parameters for `$param` references.
217    params: HashMap<String, CypherParameterValue>,
218}
219
220impl Default for CypherConverter {
221    fn default() -> Self {
222        Self::new()
223    }
224}

241    /// Convert a [`CypherStatement`] into a [`Query`].
242    ///
243    /// # Errors
244    ///
245    /// Returns [`CypherError`] if:
246    /// - A `$param` reference has no binding in the parameter map.
247    /// - An unsupported AST construct is encountered.
248    pub fn convert(&self, stmt: CypherStatement) -> Result<Query, CypherError> {
249        match stmt {
250            CypherStatement::Match {
251                optional,
252                pattern,
253                where_clause,
254                return_clause,
255                temporal,
256                with_clauses,
257                optional_matches,
258            } => {
259                let mut ops = Vec::new();
260
261                // 1. Convert graph patterns → ScanNodes + Traverse + Filter ops,
262                //    followed by the base WHERE clause → Filter op.
263                //
264                //    For a leading OPTIONAL MATCH the whole base segment
265                //    (scan + filters + WHERE) is wrapped in a single
266                //    QueryOp::Optional so an empty result yields one null row.
267                let mut base_ops = Vec::new();
268                for pat in &pattern {
269                    self.convert_pattern(pat, &mut base_ops)?;
270                }
271                if let Some(expr) = where_clause {
272                    // Lifts any vector similarity threshold (Issue #553) into a
273                    // RankBySimilarity; otherwise emits a single Filter exactly
274                    // as before.
275                    self.convert_where_clause(&expr, &mut base_ops)?;
276                }
277                if optional {
278                    ops.push(QueryOp::Optional { ops: base_ops });
279                } else {
280                    ops.append(&mut base_ops);
281                }
282
283                // Track the variable the pipeline is positionally "on" after
284                // each clause: the last node of the base pattern, then the
285                // last node of each OPTIONAL MATCH pattern (or a WITH
286                // projection). Subsequent OPTIONAL MATCH clauses must
287                // continue from this binding (see convert_optional_pattern).
288                let mut current_var: Option<String> = pattern
289                    .last()
290                    .and_then(Self::last_node_variable)
291                    .map(str::to_string);
292
293                // Variable scoping (Issue #556). `visible` holds the names in
294                // scope at the current point; `dropped` accumulates names a
295                // `WITH` has projected away. A downstream reference to a dropped
296                // name is a scope violation (rejected, never answered against a
297                // stale binding). The base pattern's variables seed the scope.
298                let mut visible: HashSet<String> = HashSet::new();
299                for pat in &pattern {
300                    Self::collect_pattern_variables(pat, &mut visible);
301                }
302                let mut dropped: HashSet<String> = HashSet::new();
303
304                // 2. Convert temporal clause → TemporalContext + ops
305                let temporal_context = if let Some(ref temporal_clause) = temporal {
306                    Some(self.convert_temporal(temporal_clause, &mut ops)?)
307                } else {
308                    None
309                };
310
311                // 3. Interleave WITH clauses (→ projection + Distinct/Sort/Skip/
312                //    Limit/Filter ops) and OPTIONAL MATCH clauses (→ Optional
313                //    ops) in source order.
314                let mut with_idx = 0;
315                for opt_match in &optional_matches {
316                    while with_idx < opt_match.preceding_withs && with_idx < with_clauses.len() {
317                        let order_by_overridden = Self::order_by_overridden_without_window(
318                            &with_clauses,
319                            with_idx,
320                            &return_clause,
321                        );
322                        self.convert_with_clause(
323                            &with_clauses[with_idx],
324                            &mut ops,
325                            &mut visible,
326                            &mut dropped,
327                            &mut current_var,
328                            order_by_overridden,
329                        )?;
330                        with_idx += 1;
331                    }
332                    // Without variable-binding analysis, comma-separated
333                    // patterns in one OPTIONAL MATCH clause would silently
334                    // chain positionally (the second pattern would traverse
335                    // the first pattern's result instead of re-seeding from
336                    // the bound variable), returning wrong entities. Reject
337                    // until binding analysis exists.
338                    if opt_match.pattern.len() > 1 {
339                        return Err(CypherError::UnsupportedFeature(
340                            "comma-separated patterns in an OPTIONAL MATCH clause; \
341                             use one pattern per OPTIONAL MATCH clause instead"
342                                .to_string(),
343                        ));
344                    }
345                    let mut sub_ops = Vec::new();
346                    for pat in &opt_match.pattern {
347                        self.convert_optional_pattern(pat, &mut sub_ops, current_var.as_deref())?;
348                        // The optional pattern's variables (re-)enter scope: add
349                        // them to `visible` and clear any earlier `dropped` mark.
350                        let mut pat_vars = HashSet::new();
351                        Self::collect_pattern_variables(pat, &mut pat_vars);
352                        for v in &pat_vars {
353                            dropped.remove(v);
354                        }
355                        visible.extend(pat_vars);
356                    }
357                    // The next clause continues from this pattern's last node.
358                    current_var = opt_match
359                        .pattern
360                        .last()
361                        .and_then(Self::last_node_variable)
362                        .map(str::to_string);
363                    if let Some(ref where_expr) = opt_match.where_clause {
364                        // Per openCypher, the clause's WHERE is part of the
365                        // optional pattern: it must run before the
366                        // matched/unmatched decision, hence inside the op.
367                        self.reject_dropped_refs(where_expr, &dropped, "OPTIONAL MATCH ... WHERE")?;
368                        let predicate = self.convert_expr_to_predicate(where_expr)?;
369                        sub_ops.push(QueryOp::Filter(predicate));
370                    }
371                    ops.push(QueryOp::Optional { ops: sub_ops });
372                }
373                while with_idx < with_clauses.len() {
374                    let order_by_overridden = Self::order_by_overridden_without_window(
375                        &with_clauses,
376                        with_idx,
377                        &return_clause,
378                    );
379                    self.convert_with_clause(
380                        &with_clauses[with_idx],
381                        &mut ops,
382                        &mut visible,
383                        &mut dropped,
384                        &mut current_var,
385                        order_by_overridden,
386                    )?;
387                    with_idx += 1;
388                }
389
390                // Enforce scoping on the final RETURN: a RETURN item (or its
391                // ORDER BY) that references a variable a `WITH` dropped is a
392                // scope violation. (The positional pipeline returns the current
393                // entity regardless of which in-scope variable is named -- a
394                // pre-existing limitation -- but a reference to an
395                // out-of-scope, dropped variable is unambiguously wrong and is
396                // rejected here rather than answered against the wrong entity.)
397                for item in &return_clause.items {
398                    match item {
399                        CypherReturnItem::Expression { expr, .. } => {
400                            self.reject_dropped_refs(expr, &dropped, "RETURN")?;
401                        }
402                        // Defense-in-depth for a manually-constructed AST: the
403                        // parser currently emits `Expression` for a bare
404                        // variable, but a hand-built `Variable` item must be
405                        // scope-checked too. `Star` binds nothing to reject.
406                        CypherReturnItem::Variable(name) => {
407                            if dropped.contains(name) {
408                                return Err(CypherError::SemanticError(format!(
409                                    "RETURN references variable '{name}', which is not in \
410                                     scope after a WITH projection dropped it; project it \
411                                     through the WITH (e.g. `WITH ..., {name}`) to keep it \
412                                     visible"
413                                )));
414                            }
415                        }
416                        CypherReturnItem::Star => {}
417                    }
418                }
419                for order_item in &return_clause.order_by {
420                    self.reject_dropped_refs(&order_item.expr, &dropped, "RETURN ... ORDER BY")?;
421                }
422
423                // 4. Aggregation (openCypher implicit grouping). If any RETURN
424                //    item is an aggregate function call, the RETURN lowers to a
425                //    single Aggregate op grouping on the non-aggregate items.
426                //    Aggregation subsumes DISTINCT and changes what the pipeline
427                //    yields (computed column rows), so RETURN DISTINCT and the
428                //    ORDER BY / vector-rank projection are only applied on the
429                //    non-aggregate path.
430                let aggregate_op = self.build_aggregate(&return_clause)?;
431                let aggregated = aggregate_op.is_some();
432                if let Some(aggregate_op) = aggregate_op {
433                    ops.push(aggregate_op);
434                    // ORDER BY over aggregate output: sort by the output column
435                    // name / aggregate alias (RETURN DISTINCT is subsumed by
436                    // grouping; vector rank does not apply to computed rows).
437                    for order_item in &return_clause.order_by {
438                        let key = self.aggregate_order_key(&return_clause, order_item)?;
439                        ops.push(QueryOp::Sort {
440                            key: SortKey::Property(key),
441                            descending: order_item.descending,
442                        });
443                    }
444                } else {
445                    // 4a. RETURN DISTINCT
446                    if return_clause.distinct {
447                        ops.push(QueryOp::Distinct);
448                    }
449
450                    // 4b. Check if ORDER BY contains a vector function call that
451                    // should be converted to RankBySimilarity instead of Sort.
452                    let vector_rank_emitted =
453                        self.try_emit_vector_rank(&return_clause, &mut ops)?;
454
455                    if !vector_rank_emitted {
456                        for order_item in &return_clause.order_by {
457                            let sort_key = self.convert_order_item_to_sort_key(order_item)?;
458                            ops.push(QueryOp::Sort {
459                                key: sort_key,
460                                descending: order_item.descending,
461                            });
462                        }
463                    }
464                }
465
466                // SKIP / LIMIT page the final rows in both the aggregate and the
467                // ordinary case.
468                if let Some(skip) = return_clause.skip {
469                    ops.push(QueryOp::Skip(skip));
470                }
471
472                if let Some(limit) = return_clause.limit {
473                    // If we already emitted a RankBySimilarity (which includes top_k),
474                    // still emit the Limit for the general pipeline.
475                    ops.push(QueryOp::Limit(limit));
476                }
477
478                // Provenance accessor projection (Issue #3354) runs LAST so it
479                // never perturbs ordering/pagination. `build_provenance_projection`
480                // is always consulted (even when aggregated) so an accessor mixed
481                // with aggregation is REJECTED rather than silently dropped; a
482                // normal aggregate query with no accessor still yields `None`.
483                if let Some(op) =
484                    self.build_provenance_projection(&return_clause, &pattern, aggregated)?
485                {
486                    ops.push(op);
487                }
488
489                Ok(Query {
490                    ops,
491                    temporal_context,
492                    hints: QueryHints::default(),
493                    // Cypher namespace scoping is a follow-up (PR2b).
494                    scope: None,
495                })
496            }
497            // A standalone `UNWIND` produces scalar rows, not stored entities,
498            // so it has no representation in the entity-oriented `Query` IR. It
499            // is executed by the dedicated Cypher UNWIND runtime instead (see
500            // [`crate::cypher::plan_cypher`] / `AletheiaDB::execute_cypher`).
501            // Reject conversion explicitly rather than answer silently wrong.
502            CypherStatement::Unwind { .. } => Err(CypherError::UnsupportedFeature(
503                "UNWIND does not lower into the graph query pipeline; execute it \
504                 via AletheiaDB::execute_cypher (or cypher::plan_cypher), which \
505                 evaluates the list-expansion runtime directly"
506                    .to_string(),
507            )),
508            // `EXPLAIN`/`PROFILE` are planning/execution *directives*, not
509            // queries: they are unwrapped and dispatched by
510            // [`crate::cypher::plan_cypher`] (which lowers the inner statement
511            // and returns a dedicated `CypherExecution::Explain`/`Profile`), so
512            // the converter never sees them directly. Reject rather than answer
513            // silently wrong if one reaches here.
514            CypherStatement::Explain(_) | CypherStatement::Profile(_) => {
515                Err(CypherError::UnsupportedFeature(
516                    "EXPLAIN/PROFILE are query directives handled by \
517                     cypher::plan_cypher, not lowered by the converter"
518                        .to_string(),
519                ))
520            }
521            // Write statements (Issues #560, #3548) are executed directly
522            // against the native write APIs by `crate::cypher::mutation`, not
523            // lowered into the read-only `Query` IR. Reject conversion
524            // explicitly rather than fabricate a read plan for a mutation.
525            CypherStatement::Write(_) => Err(CypherError::UnsupportedFeature(
526                "write statements (CREATE / MERGE / SET / DELETE) do not lower into the \
527                 read-only query pipeline; execute them via \
528                 AletheiaDB::execute_cypher, which applies the mutation through \
529                 the native write APIs"
530                    .to_string(),
531            )),
532        }
533    }

539    /// Convert a single graph pattern into query operations.
540    ///
541    /// A pattern like `(a:Person)-[:KNOWS]->(b:Person {name: 'Bob'})` produces:
542    /// 1. `ScanNodes { label: Some("Person") }` for node `a`
543    /// 2. Filter ops for node `a`'s inline properties (if any)
544    /// 3. `TraverseOut { label: Some("KNOWS"), depth: Exact(1) }` for the rel
545    /// 4. Filter ops for node `b`'s inline properties
546    fn convert_pattern(
547        &self,
548        pattern: &CypherPattern,
549        ops: &mut Vec<QueryOp>,
550    ) -> Result<(), CypherError> {
551        let mut first_node = true;
552
553        for element in &pattern.elements {
554            match element {
555                CypherPatternElement::Node(node) => {
556                    if first_node {
557                        // The first node in the pattern produces a ScanNodes op.
558                        let label = node.labels.first().cloned();
559                        ops.push(QueryOp::ScanNodes { label });
560                        first_node = false;
561                    }
562                    // Emit filters for inline property constraints on any node.
563                    self.convert_node_properties(node, ops)?;
564                }
565                CypherPatternElement::Relationship(rel) => {
566                    self.convert_relationship(rel, ops)?;
567                }
568            }
569        }
570
571        Ok(())
572    }

574    /// Convert a graph pattern from a *subsequent* `OPTIONAL MATCH` clause
575    /// into query operations for the optional sub-pipeline.
576    ///
577    /// Unlike [`Self::convert_pattern`], the first node does **not** emit a
578    /// `ScanNodes` op: it is assumed to refer to the current row produced by
579    /// the preceding pipeline (e.g. `a` in
580    /// `MATCH (a) OPTIONAL MATCH (a)-[:KNOWS]->(x)`). Its inline property
581    /// constraints still become `Filter` ops, which run against the seed row
582    /// inside the optional segment and therefore participate in the
583    /// matched/unmatched decision.
584    ///
585    /// Limitation and design choice: the converter performs no variable
586    /// binding analysis, so the first node is bound purely positionally to
587    /// the current row. Two constructs that would silently produce wrong
588    /// rows are therefore **rejected** with
589    /// [`CypherError::UnsupportedFeature`] rather than answered wrongly:
590    ///
591    /// - A **label** on the first node cannot be honored (there is no label
592    ///   predicate to lower it to), and silently dropping it would turn
593    ///   `MATCH (a:Person) OPTIONAL MATCH (b:City) RETURN b` into a query
594    ///   that returns `a`.
595    /// - A first-node **variable that does not name the positionally-current
596    ///   binding** (`current_var`, the last node of the previous
597    ///   MATCH/OPTIONAL MATCH or a `WITH` projection) would silently bind to
598    ///   whatever the pipeline is positioned on: in `MATCH (a:Person)
599    ///   OPTIONAL MATCH (a)-[:KNOWS]->(x) OPTIONAL MATCH (a)-[:OWNS]->(y)`
600    ///   the second `(a)` would bind to `x` (or null) and traverse `OWNS`
601    ///   from the wrong node. Re-anchoring on an earlier variable requires
602    ///   real binding analysis and is not yet supported.
603    ///
604    /// An unnamed first node `()` (explicit positional reference) and a
605    /// first node naming `current_var` are accepted; inline properties on
606    /// the first node remain supported because they lower cheaply and
607    /// correctly to seed-row filters.
608    fn convert_optional_pattern(
609        &self,
610        pattern: &CypherPattern,
611        ops: &mut Vec<QueryOp>,
612        current_var: Option<&str>,
613    ) -> Result<(), CypherError> {
614        if let Some(CypherPatternElement::Node(first)) = pattern.elements.first() {
615            // Reject a labeled first node: it is bound positionally to the
616            // seed row and its label would be silently ignored (see rustdoc
617            // above).
618            if !first.labels.is_empty() {
619                return Err(CypherError::UnsupportedFeature(format!(
620                    "label ':{}' on the first node of a subsequent OPTIONAL MATCH; \
621                     the first node refers to the current row and must be an \
622                     unlabeled bound variable (e.g. `(a)`) -- move the constraint \
623                     into the preceding MATCH or a WHERE clause",
624                    first.labels.join(":"),
625                )));
626            }
627            // Reject a first-node variable that re-anchors on something other
628            // than the positionally-current binding (see rustdoc above).
629            if let Some(ref first_var) = first.variable
630                && current_var != Some(first_var.as_str())
631            {
632                return Err(CypherError::UnsupportedFeature(match current_var {
633                    Some(cur) => format!(
634                        "OPTIONAL MATCH must continue from the previous clause's \
635                         binding '{cur}'; re-anchoring on '{first_var}' is not \
636                         yet supported -- start the pattern with `({cur})` or \
637                         restructure the query"
638                    ),
639                    None => format!(
640                        "OPTIONAL MATCH must continue from the previous clause's \
641                         binding, but its last node is unnamed; re-anchoring on \
642                         '{first_var}' is not yet supported -- name the previous \
643                         pattern's last node or restructure the query"
644                    ),
645                }));
646            }
647        }
648        for element in &pattern.elements {
649            match element {
650                CypherPatternElement::Node(node) => {
651                    // No ScanNodes: the seed row is the pattern's entry point.
652                    self.convert_node_properties(node, ops)?;
653                }
654                CypherPatternElement::Relationship(rel) => {
655                    self.convert_relationship(rel, ops)?;
656                }
657            }
658        }
659        Ok(())
660    }

662    /// The variable bound to a pattern's last node element, if named.
663    ///
664    /// This is the binding the pipeline is positionally "on" after the
665    /// pattern converts: each relationship traversal moves the current row
666    /// to its target node, so the last node of the pattern is where a
667    /// subsequent `OPTIONAL MATCH` continues from.
668    fn last_node_variable(pattern: &CypherPattern) -> Option<&str> {
669        pattern
670            .elements
671            .iter()
672            .rev()
673            .find_map(|element| match element {
674                CypherPatternElement::Node(node) => Some(node.variable.as_deref()),
675                CypherPatternElement::Relationship(_) => None,
676            })
677            .flatten()
678    }

680    /// Convert a `WITH` clause into query operations, threading variable scope
681    /// through the pipeline (Issue #556).
682    ///
683    /// openCypher evaluates a `WITH` body as: project the items, apply
684    /// `DISTINCT`, then `ORDER BY`, `SKIP`, `LIMIT`, and finally the trailing
685    /// `WHERE` (which filters the *projected* rows). This method emits that
686    /// pipeline as ops and updates `visible` / `dropped` / `current_var`.
687    ///
688    /// # Projection boundary (positional pipeline)
689    ///
690    /// AletheiaDB's query pipeline carries exactly one entity per row, so a
691    /// `WITH` can only *carry forward* the current positional binding: a bare
692    /// or aliased reference to `current_var`, or `*` (carry everything). Any
693    /// other projection -- multiple items, a property or computed expression,
694    /// an aggregate, or a re-projection of a non-current variable -- has no
695    /// faithful representation and is **rejected** with a structured
696    /// [`CypherError::UnsupportedFeature`] rather than answered against the
697    /// wrong entity.
698    fn convert_with_clause(
699        &self,
700        with: &CypherWith,
701        ops: &mut Vec<QueryOp>,
702        visible: &mut HashSet<String>,
703        dropped: &mut HashSet<String>,
704        current_var: &mut Option<String>,
705        order_by_overridden: bool,
706    ) -> Result<(), CypherError> {
707        // 1. Resolve the projection shape and update the scope.
708        match Self::with_projection(with)? {
709            WithProjection::All => {
710                // `WITH *` carries every visible variable forward, but the
711                // positional pipeline carries only ONE entity per row. When
712                // more than one variable is in scope, `*` would silently
713                // forward the wrong entity (e.g. `MATCH (a)-[:KNOWS]->(b)
714                // WITH * RETURN a` returns `b`'s data as `a`). Reject rather
715                // than answer wrongly; `*` with a single visible variable is a
716                // faithful no-op (scope and positional binding unchanged).
717                if visible.len() > 1 {
718                    return Err(CypherError::UnsupportedFeature(format!(
719                        "WITH * with more than one variable in scope ({} visible) is \
720                         not supported: the positional execution pipeline carries a \
721                         single entity per row, so '*' would forward the wrong \
722                         entity. Project the single variable you need (optionally \
723                         aliased) instead.",
724                        visible.len()
725                    )));
726                }
727            }
728            WithProjection::Single { source, output } => {
729                // The pipeline is positioned on `current_var`; it cannot
730                // reposition to a different variable, so only the current
731                // binding (optionally aliased) can be carried forward.
732                if current_var.as_deref() != Some(source.as_str()) {
733                    return Err(CypherError::UnsupportedFeature(match current_var {
734                        Some(cur) => format!(
735                            "WITH can only carry forward the current pattern variable \
736                             '{cur}' (optionally aliased) or '*'; re-projecting \
737                             '{source}' would require repositioning the positional \
738                             pipeline, which is not supported"
739                        ),
740                        None => format!(
741                            "WITH can only carry forward the current pattern variable \
742                             (optionally aliased) or '*', but the preceding clause's \
743                             last node is unnamed; projecting '{source}' is not \
744                             supported"
745                        ),
746                    }));
747                }
748                // Everything except the projected output leaves scope. Drain
749                // `visible` to move the owned names straight into `dropped`
750                // (no clone), then re-seed the scope with the output.
751                for name in visible.drain() {
752                    if name != output {
753                        dropped.insert(name);
754                    }
755                }
756                visible.insert(output.clone());
757                dropped.remove(&output);
758                *current_var = Some(output);
759            }
760        }
761
762        // 2. DISTINCT deduplicates the projected rows.
763        if with.distinct {
764            ops.push(QueryOp::Distinct);
765        }
766
767        // 3. ORDER BY / SKIP / LIMIT on the projected rows.
768        //
769        //    A `WITH` ORDER BY is meaningful when it selects a SKIP/LIMIT
770        //    window (in this clause OR any clause before the next re-sort), or
771        //    when it is the last ordering applied before output. It may be
772        //    dropped ONLY when the next ordering-affecting event is a later
773        //    re-sort with NO intervening SKIP/LIMIT window
774        //    (`order_by_overridden`): that later ordering fully replaces it and
775        //    openCypher leaves the overridden tie order unspecified. Dropping
776        //    the pure-consecutive-sort case also avoids an incorrect fold in
777        //    the physical planner, which merges *consecutive* Sort ops into one
778        //    multi-key Sort; emitting a Sort when a window intervenes is
779        //    fold-safe because the intervening Skip/Limit op breaks that fold.
780        if !with.order_by.is_empty() {
781            for order_item in &with.order_by {
782                self.reject_dropped_refs(&order_item.expr, dropped, "WITH ... ORDER BY")?;
783            }
784            let has_window = with.skip.is_some() || with.limit.is_some();
785            if has_window || !order_by_overridden {
786                for order_item in &with.order_by {
787                    let sort_key = self.convert_order_item_to_sort_key(order_item)?;
788                    ops.push(QueryOp::Sort {
789                        key: sort_key,
790                        descending: order_item.descending,
791                    });
792                }
793            }
794        }
795        if let Some(skip) = with.skip {
796            ops.push(QueryOp::Skip(skip));
797        }
798        if let Some(limit) = with.limit {
799            ops.push(QueryOp::Limit(limit));
800        }
801
802        // 4. The trailing WHERE filters the projected rows.
803        if let Some(ref where_expr) = with.where_clause {
804            self.reject_dropped_refs(where_expr, dropped, "WITH ... WHERE")?;
805            let predicate = self.convert_expr_to_predicate(where_expr)?;
806            ops.push(QueryOp::Filter(predicate));
807        }
808        Ok(())
809    }

811    /// Resolve the projection shape of a `WITH` clause, rejecting any shape the
812    /// positional pipeline cannot carry (see [`Self::convert_with_clause`]).
813    fn with_projection(with: &CypherWith) -> Result<WithProjection, CypherError> {
814        match with.items.as_slice() {
815            [CypherReturnItem::Star] => Ok(WithProjection::All),
816            [single] => Self::single_projection(single),
817            _ => Err(CypherError::UnsupportedFeature(
818                "WITH projecting multiple items is not supported: the positional \
819                 execution pipeline carries a single entity per row. Project one \
820                 variable (optionally aliased) or '*'."
821                    .to_string(),
822            )),
823        }
824    }

826    /// Resolve a single `WITH` item into its `(source, output)` variable names.
827    ///
828    /// Only a bare variable (`WITH a`) or an aliased variable (`WITH a AS b`)
829    /// is representable; a computed expression, property access, aggregate, or
830    /// `*` in this position is rejected.
831    fn single_projection(item: &CypherReturnItem) -> Result<WithProjection, CypherError> {
832        match item {
833            CypherReturnItem::Variable(v) => Ok(WithProjection::Single {
834                source: v.clone(),
835                output: v.clone(),
836            }),
837            CypherReturnItem::Expression {
838                expr: CypherExpr::Variable(v),
839                alias,
840            } => Ok(WithProjection::Single {
841                source: v.clone(),
842                output: alias.clone().unwrap_or_else(|| v.clone()),
843            }),
844            CypherReturnItem::Star => Err(CypherError::UnsupportedFeature(
845                "WITH '*' cannot be mixed with other projected items".to_string(),
846            )),
847            CypherReturnItem::Expression { expr, .. } => {
848                Err(CypherError::UnsupportedFeature(format!(
849                    "WITH can only project a bound variable (optionally aliased) or \
850                     '*'; projecting the expression {expr:?} is not supported -- \
851                     computed columns and aggregates in WITH are a follow-up"
852                )))
853            }
854        }
855    }

857    /// Whether a bare intermediate `WITH ... ORDER BY` (one with no SKIP/LIMIT
858    /// window of its own) at `with_idx` is fully overridden by a later re-sort
859    /// with **no intervening SKIP/LIMIT window**, and may therefore be dropped.
860    ///
861    /// A `WITH`/`RETURN` clause applies its own `ORDER BY` *before* its own
862    /// `SKIP`/`LIMIT`, so scanning forward from the next clause:
863    ///
864    /// - the first later clause carrying its own `ORDER BY` is a **re-sort**
865    ///   that replaces this ordering (droppable) -- its order runs before any
866    ///   paging that clause also has, so it counts as a re-sort, not a window;
867    /// - a `SKIP`/`LIMIT` in a clause *before* that re-sort **consumes** this
868    ///   ordering (it selects a window against these ordered rows), making the
869    ///   sort observable -- do not drop;
870    /// - reaching the `RETURN` with no earlier re-sort: the `RETURN`'s own
871    ///   `ORDER BY` (if any) is the next re-sort (droppable); otherwise this
872    ///   ordering is **terminal** (the final result order) and is observable.
873    ///
874    /// The caller additionally guards on this `WITH`'s *own* SKIP/LIMIT (a
875    /// same-clause window makes its ORDER BY observable); this look-ahead only
876    /// considers the clauses that follow it.
877    fn order_by_overridden_without_window(
878        with_clauses: &[CypherWith],
879        with_idx: usize,
880        return_clause: &CypherReturn,
881    ) -> bool {
882        for later in with_clauses.iter().skip(with_idx + 1) {
883            if !later.order_by.is_empty() {
884                // Next re-sort reached with no intervening window: overridden.
885                return true;
886            }
887            if later.skip.is_some() || later.limit.is_some() {
888                // A window consumes this ordering before any later re-sort.
889                return false;
890            }
891        }
892        // No later `WITH` re-sorts before the `RETURN`: the `RETURN`'s own
893        // `ORDER BY` (applied before its paging) is the next re-sort; absent
894        // that, this ordering is terminal and therefore observable.
895        !return_clause.order_by.is_empty()
896    }

917    /// Reject an expression that references any variable a `WITH` has dropped
918    /// from scope (Issue #556 scoping enforcement). `context` names the clause
919    /// for the error message.
920    fn reject_dropped_refs(
921        &self,
922        expr: &CypherExpr,
923        dropped: &HashSet<String>,
924        context: &str,

934                )));
935            }
936        }
937        Ok(())
938    }

940    /// Collect the variable names referenced by an expression (both bare
941    /// variables and the base of a property access) into `out`.
942    ///
943    /// Depth-guarded like the predicate converter: `convert` is `pub`, so a
944    /// hand-built expression could nest past the parser's own cap. The bound
945    /// mirrors [`MAX_CONVERT_DEPTH`] so it never fires for parsed input.
946    fn collect_expr_variables(
947        expr: &CypherExpr,
948        out: &mut HashSet<String>,
949        depth: usize,
950    ) -> Result<(), CypherError> {
951        if depth > MAX_CONVERT_DEPTH {
952            return Err(CypherError::ParseError {
953                position: 0,
954                message: "expression nesting too deep for scope analysis (max 256)".to_string(),
955            });
956        }
957        match expr {
958            CypherExpr::Variable(name) => {
959                out.insert(name.clone());
960            }
961            CypherExpr::Property { variable, .. } => {
962                out.insert(variable.clone());
963            }
964            CypherExpr::Value(_) | CypherExpr::Star => {}
965            CypherExpr::Comparison { left, right, .. } => {
966                Self::collect_expr_variables(left, out, depth + 1)?;
967                Self::collect_expr_variables(right, out, depth + 1)?;
968            }
969            CypherExpr::And(left, right) | CypherExpr::Or(left, right) => {
970                Self::collect_expr_variables(left, out, depth + 1)?;
971                Self::collect_expr_variables(right, out, depth + 1)?;
972            }
973            CypherExpr::Not(inner)
974            | CypherExpr::IsNull(inner)
975            | CypherExpr::IsNotNull(inner)
976            | CypherExpr::Grouped(inner) => {
977                Self::collect_expr_variables(inner, out, depth + 1)?;
978            }
979            CypherExpr::In { expr, values } => {
980                Self::collect_expr_variables(expr, out, depth + 1)?;
981                for value in values {
982                    Self::collect_expr_variables(value, out, depth + 1)?;
983                }
984            }
985            CypherExpr::Contains { expr, .. }
986            | CypherExpr::StartsWith { expr, .. }
987            | CypherExpr::EndsWith { expr, .. } => {
988                Self::collect_expr_variables(expr, out, depth + 1)?;
989            }
990            CypherExpr::FunctionCall { args, .. } | CypherExpr::List(args) => {
991                for arg in args {
992                    Self::collect_expr_variables(arg, out, depth + 1)?;
993                }
994            }
995        }
996        Ok(())
997    }

1015    /// Convert a relationship pattern into a traversal op.
1016    ///
1017    /// Also emits `Filter` ops for any inline property constraints on the
1018    /// relationship (e.g., `-[:TYPE {since: 2020}]->`).
1019    fn convert_relationship(
1020        &self,
1021        rel: &CypherRelPattern,
1022        ops: &mut Vec<QueryOp>,
1023    ) -> Result<(), CypherError> {
1024        let label = rel.rel_types.first().cloned();
1025        let depth = self.convert_depth(&rel.depth);
1026
1027        let op = match rel.direction {
1028            CypherDirection::Outgoing => QueryOp::TraverseOut { label, depth },
1029            CypherDirection::Incoming => QueryOp::TraverseIn { label, depth },
1030            CypherDirection::Both => QueryOp::TraverseBoth { label, depth },
1031        };
1032
1033        ops.push(op);
1034
1035        // Emit filters for inline relationship properties: -[:TYPE {key: val}]->
1036        for (key, value) in &rel.properties {
1037            let pred_value = self.convert_value_to_predicate_value(value)?;
1038            ops.push(QueryOp::Filter(Predicate::Eq {
1039                key: key.clone(),
1040                value: pred_value,
1041            }));
1042        }
1043
1044        Ok(())
1045    }

1047    /// Map a Cypher depth specifier to a [`TraversalDepth`].
1048    fn convert_depth(&self, depth: &Option<CypherDepth>) -> TraversalDepth {
1049        match depth {
1050            None => TraversalDepth::Exact(1),
1051            Some(CypherDepth::Unbounded) => TraversalDepth::Variable,
1052            Some(CypherDepth::Exact(n)) => TraversalDepth::Exact(*n),
1053            Some(CypherDepth::Max(n)) => TraversalDepth::Max(*n),
1054            Some(CypherDepth::Min(n)) => TraversalDepth::Range {
1055                min: *n,
1056                max: usize::MAX,
1057            },
1058            Some(CypherDepth::Range { min, max }) => TraversalDepth::Range {
1059                min: *min,
1060                max: *max,
1061            },
1062        }
1063    }

1069    /// Convert a Cypher expression into a [`Predicate`].
1070    ///
1071    /// This handles comparisons, logical operators, string predicates, etc.
1072    fn convert_expr_to_predicate(&self, expr: &CypherExpr) -> Result<Predicate, CypherError> {
1073        self.convert_expr_to_predicate_at(expr, 0)
1074    }
1075
1076    /// Depth-guarded worker for [`Self::convert_expr_to_predicate`].
1077    ///
1078    /// The `Not` / `Grouped` arms recurse structurally; `convert` is `pub`, so
1079    /// this guards against a caller handing us an AST nested past the parser's
1080    /// own [`MAX_EXPRESSION_DEPTH`]. The bound is set above the parser's cap so
1081    /// it never fires for parser-produced ASTs, only for hand-built ones.
1082    fn convert_expr_to_predicate_at(
1083        &self,
1084        expr: &CypherExpr,
1085        depth: usize,
1086    ) -> Result<Predicate, CypherError> {
1087        if depth > MAX_CONVERT_DEPTH {
1088            return Err(CypherError::ParseError {
1089                position: 0,
1090                message: "expression nesting too deep for conversion (max 256)".to_string(),
1091            });
1092        }
1093        match expr {
1094            CypherExpr::Comparison { left, op, right } => self.convert_comparison(left, *op, right),
1095            CypherExpr::And(_, _) => {
1096                // Iteratively flatten the (possibly deeply left-nested) AND
1097                // spine so that a long `a AND b AND c ...` chain lowers to a
1098                // single n-ary predicate without recursing per operator.
1099                let operands = flatten_logical_operands(expr, |e| match e {
1100                    CypherExpr::And(left, right) => Some((left, right)),
1101                    _ => None,
1102                });
1103                let preds = operands
1104                    .into_iter()
1105                    .map(|operand| self.convert_expr_to_predicate_at(operand, depth + 1))
1106                    .collect::<Result<Vec<_>, _>>()?;
1107                Ok(Predicate::And(preds))
1108            }
1109            CypherExpr::Or(_, _) => {
1110                // Same iterative flattening for the OR spine.
1111                let operands = flatten_logical_operands(expr, |e| match e {
1112                    CypherExpr::Or(left, right) => Some((left, right)),
1113                    _ => None,
1114                });
1115                let preds = operands
1116                    .into_iter()
1117                    .map(|operand| self.convert_expr_to_predicate_at(operand, depth + 1))
1118                    .collect::<Result<Vec<_>, _>>()?;
1119                Ok(Predicate::Or(preds))
1120            }
1121            CypherExpr::Not(inner) => {
1122                let inner_pred = self.convert_expr_to_predicate_at(inner, depth + 1)?;
1123                Ok(Predicate::Not(Box::new(inner_pred)))
1124            }
1125            CypherExpr::IsNull(inner) => {
1126                // `provenance(x) IS NULL` (Issue #3354b): selects the versions
1127                // with no provenance bundle at all (unattributed rows).
1128                if self.is_provenance_bundle_accessor(inner)? {
1129                    return Ok(Predicate::Provenance(ProvenancePredicate::IsNull {
1130                        negated: false,
1131                    }));
1132                }
1133                match inner.as_ref() {
1134                    // Property access (`n.email IS NULL`): openCypher treats a
1135                    // property as null when it is ABSENT *or* present with an
1136                    // explicit null value. The data model represents both (a
1137                    // missing key and a stored `PropertyValue::Null`), so lower to
1138                    // `NotExists(key) OR Eq(key, Null)`. The old `Eq(key, Null)`-only
1139                    // lowering matched present-null but silently missed absence.
1140                    CypherExpr::Property { property, .. } => Ok(Predicate::Or(vec![
1141                        Predicate::NotExists(property.clone()),
1142                        Predicate::Eq {
1143                            key: property.clone(),
1144                            value: PredicateValue::Null,
1145                        },
1146                    ])),
1147                    // Bare variable (`x IS NULL`): a node/edge-level null check
1148                    // (only an OPTIONAL MATCH unmatched binding is ever null). Keep
1149                    // the `Eq(var, Null)` lowering: a matched entity has no property
1150                    // literally named after the variable (so it is not-null), and a
1151                    // null binding is handled by the executor's null-row path.
1152                    _ => {
1153                        let key = self.extract_property_key(inner)?;
1154                        Ok(Predicate::Eq {
1155                            key,
1156                            value: PredicateValue::Null,
1157                        })
1158                    }
1159                }
1160            }
1161            CypherExpr::IsNotNull(inner) => {
1162                // `provenance(x) IS NOT NULL` (Issue #3354b): selects the
1163                // versions that carry a provenance bundle (attributed rows).
1164                if self.is_provenance_bundle_accessor(inner)? {
1165                    return Ok(Predicate::Provenance(ProvenancePredicate::IsNull {
1166                        negated: true,
1167                    }));
1168                }
1169                match inner.as_ref() {
1170                    // Property access (`n.email IS NOT NULL`): openCypher requires
1171                    // the property to be PRESENT with a non-null value. Lower to
1172                    // `Exists(key) AND Ne(key, Null)`; the old `Ne(key, Null)`-only
1173                    // lowering wrongly matched absent properties (the executor treats
1174                    // a missing property as `!= null`).
1175                    CypherExpr::Property { property, .. } => Ok(Predicate::And(vec![
1176                        Predicate::Exists(property.clone()),
1177                        Predicate::Ne {
1178                            key: property.clone(),
1179                            value: PredicateValue::Null,
1180                        },
1181                    ])),
1182                    // Bare variable (`x IS NOT NULL`): node/edge-level null check
1183                    // (see IsNull above). Keep the `Ne(var, Null)` lowering.
1184                    _ => {
1185                        let key = self.extract_property_key(inner)?;
1186                        Ok(Predicate::Ne {
1187                            key,
1188                            value: PredicateValue::Null,
1189                        })
1190                    }
1191                }
1192            }
1193            CypherExpr::In { expr, values } => {
1194                let key = self.extract_property_key(expr)?;
1195                let pred_values = values
1196                    .iter()
1197                    .map(|v| self.convert_expr_to_predicate_value(v))
1198                    .collect::<Result<Vec<_>, _>>()?;
1199                Ok(Predicate::In {
1200                    key,
1201                    values: pred_values,
1202                })
1203            }
1204            CypherExpr::Contains { expr, substring } => {
1205                let key = self.extract_property_key(expr)?;
1206                Ok(Predicate::Contains {
1207                    key,
1208                    substring: substring.clone(),
1209                })
1210            }
1211            CypherExpr::StartsWith { expr, prefix } => {
1212                let key = self.extract_property_key(expr)?;
1213                Ok(Predicate::StartsWith {
1214                    key,
1215                    prefix: prefix.clone(),
1216                })
1217            }
1218            CypherExpr::EndsWith { expr, suffix } => {
1219                let key = self.extract_property_key(expr)?;
1220                Ok(Predicate::EndsWith {
1221                    key,
1222                    suffix: suffix.clone(),
1223                })
1224            }
1225            CypherExpr::Grouped(inner) => self.convert_expr_to_predicate_at(inner, depth + 1),
1226            CypherExpr::Value(CypherValue::Bool(true)) => Ok(Predicate::True),
1227            CypherExpr::Value(CypherValue::Bool(false)) => Ok(Predicate::False),
1228            _ => Err(CypherError::UnsupportedFeature(format!(
1229                "expression cannot be converted to predicate: {expr:?}"
1230            ))),
1231        }
1232    }

1234    /// Convert a comparison expression into a [`Predicate`].
1235    fn convert_comparison(
1236        &self,
1237        left: &CypherExpr,
1238        op: CypherCompOp,
1239        right: &CypherExpr,
1240    ) -> Result<Predicate, CypherError> {
1241        // Provenance accessors (Issue #3354b): `source(x)`/`confidence(x)`/
1242        // `reason(x)` appear as function calls, never property accesses, so a
1243        // property literally named `source` (`n.source`) is unaffected.
1244        // Recognize an accessor on either side and lower to a provenance
1245        // predicate before the ordinary property-access handling below.
1246        let left_accessor = self.as_provenance_accessor(left)?;
1247        let right_accessor = self.as_provenance_accessor(right)?;
1248        match (left_accessor, right_accessor) {
1249            (Some(_), Some(_)) => {
1250                return Err(CypherError::SemanticError(
1251                    "comparing two provenance accessors is not supported".to_string(),
1252                ));
1253            }
1254            (Some(field), None) => {
1255                return self.build_provenance_comparison(field, op, right);
1256            }
1257            (None, Some(field)) => {
1258                // Accessor on the right (`0.9 <= confidence(n)`): flip the
1259                // comparison so the accessor is the logical left-hand side.
1260                return self.build_provenance_comparison(field, flip_cypher_comp_op(op), left);
1261            }
1262            (None, None) => {}
1263        }
1264
1265        let key = self.extract_property_key(left)?;
1266        let value = self.convert_expr_to_predicate_value(right)?;
1267
1268        Ok(match op {
1269            CypherCompOp::Eq => Predicate::Eq { key, value },
1270            CypherCompOp::Ne => Predicate::Ne { key, value },
1271            CypherCompOp::Gt => Predicate::Gt { key, value },
1272            CypherCompOp::Ge => Predicate::Gte { key, value },
1273            CypherCompOp::Lt => Predicate::Lt { key, value },
1274            CypherCompOp::Le => Predicate::Lte { key, value },
1275        })
1276    }

1278    /// Recognize a provenance accessor expression `source(x)`/`confidence(x)`/
1279    /// `reason(x)` (Issue #3354b).
1280    ///
1281    /// Returns:
1282    /// - `Ok(Some(field))` for a well-formed accessor over a single bound
1283    ///   variable,
1284    /// - `Ok(None)` for any other expression (including unknown function calls,
1285    ///   which fall through to the ordinary comparison path),
1286    /// - `Err(..)` for a provenance-named accessor with a malformed argument
1287    ///   (e.g. `source(n.foo)`, `confidence('x')`, `confidence(r, s)`,
1288    ///   `source()`), so the mistake surfaces as a structured error rather than
1289    ///   silently degrading.
1290    fn as_provenance_accessor(
1291        &self,
1292        expr: &CypherExpr,
1293    ) -> Result<Option<ProvenanceField>, CypherError> {
1294        let CypherExpr::FunctionCall {
1295            name,
1296            args,
1297            distinct,
1298        } = expr
1299        else {
1300            return Ok(None);
1301        };
1302        let Some(field) = provenance_field_name(name) else {
1303            return Ok(None);
1304        };
1305        // `DISTINCT` is only meaningful for aggregates; reject it here rather
1306        // than silently ignoring it.
1307        if *distinct {
1308            return Err(CypherError::SemanticError(format!(
1309                "{}(x) provenance accessor does not accept DISTINCT",
1310                field.accessor_name()
1311            )));
1312        }
1313        match args.as_slice() {
1314            [CypherExpr::Variable(_)] => Ok(Some(field)),
1315            _ => Err(CypherError::SemanticError(format!(
1316                "{}(x) provenance accessor requires exactly one bound variable argument",
1317                field.accessor_name()
1318            ))),
1319        }
1320    }

1322    /// Recognize a `provenance(x)` bundle accessor for the `IS [NOT] NULL` form
1323    /// (Issue #3354b).
1324    ///
1325    /// Returns `Ok(true)` for a well-formed `provenance(<bound variable>)` call,
1326    /// `Ok(false)` for any other expression, and `Err(..)` for a
1327    /// `provenance(...)` call with a malformed argument. Unlike the field
1328    /// accessors, `provenance` addresses the whole bundle and is only valid in
1329    /// the `IS [NOT] NULL` position.
1330    fn is_provenance_bundle_accessor(&self, expr: &CypherExpr) -> Result<bool, CypherError> {
1331        let CypherExpr::FunctionCall {
1332            name,
1333            args,
1334            distinct,
1335        } = expr
1336        else {
1337            return Ok(false);
1338        };
1339        if !name.eq_ignore_ascii_case("provenance") {
1340            return Ok(false);
1341        }
1342        if *distinct {
1343            return Err(CypherError::SemanticError(
1344                "provenance(x) does not accept DISTINCT".to_string(),

1346        }

1348            [CypherExpr::Variable(_)] => Ok(true),
1349            _ => Err(CypherError::SemanticError(

1352        }

1456    /// Build a [`QueryOp::Aggregate`] from the `RETURN` clause if it contains
1457    /// any aggregate function call, applying openCypher implicit grouping:
1458    /// every non-aggregate item becomes a grouping key and every aggregate
1459    /// item becomes an [`AggregateSpec`]. Returns `Ok(None)` when the clause
1460    /// has no aggregates (the ordinary projection path).
1461    fn build_aggregate(&self, ret: &CypherReturn) -> Result<Option<QueryOp>, CypherError> {
1462        let has_aggregate = ret.items.iter().any(|item| {
1463            matches!(
1464                item,
1465                CypherReturnItem::Expression { expr, .. }
1466                    if Self::aggregate_call(expr).is_some()
1467            )
1468        });
1469        if !has_aggregate {
1470            return Ok(None);
1471        }
1472
1473        let mut group_keys = Vec::new();
1474        let mut aggregates = Vec::new();
1475
1476        for item in &ret.items {
1477            match item {
1478                CypherReturnItem::Star => {
1479                    return Err(CypherError::UnsupportedFeature(
1480                        "RETURN * combined with aggregation; project explicit \
1481                         grouping keys instead"
1482                            .to_string(),
1483                    ));
1484                }
1485                CypherReturnItem::Variable(name) => {
1486                    // Grouping by a whole node/edge cannot be expressed by the
1487                    // single-entity row model (it would collapse into one
1488                    // Null-keyed group). Reject rather than silently mis-group.
1489                    return Err(CypherError::UnsupportedFeature(format!(
1490                        "grouping by a whole node/edge is not supported \
1491                         (RETURN {name}, <aggregate>); group by a property such \
1492                         as {name}.<key> instead"
1493                    )));
1494                }
1495                CypherReturnItem::Expression { expr, alias } => {
1496                    if let Some((func, args, distinct)) = Self::aggregate_call(expr) {
1497                        let arg = Self::aggregate_arg(func, args)?;
1498                        let default_alias = Self::aggregate_default_alias(func, args, distinct);
1499                        aggregates.push(AggregateSpec {
1500                            func,
1501                            arg,
1502                            distinct,
1503                            alias: alias.clone().unwrap_or(default_alias),
1504                        });
1505                    } else {
1506                        let (property_key, default_alias) = Self::group_key_of(expr)?;
1507                        group_keys.push(AggregateGroupKey {
1508                            property_key,
1509                            alias: alias.clone().unwrap_or(default_alias),
1510                        });
1511                    }
1512                }
1513            }
1514        }
1515
1516        Ok(Some(QueryOp::Aggregate {
1517            group_keys,
1518            aggregates,
1519        }))
1520    }

1552    ///
1553    /// `count(*)` -> [`AggregateArg::Star`] (count every row); a bare-variable
1554    /// `count(n)` -> [`AggregateArg::Entity`] (count rows with a non-null bound
1555    /// entity, openCypher semantics); `func(v.prop)` ->
1556    /// [`AggregateArg::Property`]. Non-property arguments to value aggregates
1557    /// are rejected in v1.
1558    fn aggregate_arg(
1559        func: AggregateFunc,
1560        args: &[CypherExpr],
1561    ) -> Result<AggregateArg, CypherError> {
1562        match args {
1563            // `*` (or the degenerate empty arg list) is only valid for count.
1564            [CypherExpr::Star] | [] if func == AggregateFunc::Count => Ok(AggregateArg::Star),
1565            [CypherExpr::Property { property, .. }] => Ok(AggregateArg::Property(property.clone())),
1566            // A bare variable argument (`count(n)`) references the bound entity,
1567            // not a property: count non-null bindings (distinct from count(*)).
1568            [CypherExpr::Variable(_)] if func == AggregateFunc::Count => Ok(AggregateArg::Entity),
1569            _ => Err(CypherError::UnsupportedFeature(format!(
1570                "aggregate argument must be `*` (count only) or a property access \
1571                 (e.g. n.age); got {args:?}"
1572            ))),

1577    /// `AS` alias is supplied), e.g. `count(*)`, `sum(n.age)`,
1578    /// `count(DISTINCT n.dept)`.
1579    fn aggregate_default_alias(func: AggregateFunc, args: &[CypherExpr], distinct: bool) -> String {
1580        let func_name = match func {
1581            AggregateFunc::Count => "count",
1582            AggregateFunc::Sum => "sum",

1584            AggregateFunc::Min => "min",

1596        format!("{func_name}({distinct_prefix}{arg_text})")

1622    /// column name it should sort by (the aggregate/group alias or generated

1625        &self,

1628    ) -> Result<String, CypherError> {
1629        let expr = &order_item.expr;
1630

1649        for item in &ret.items {

1651                expr: item_expr,

1654                && item_expr == expr

1692    /// Detect provenance accessors in a `RETURN` clause and build a
1693    /// [`QueryOp::ProjectProvenance`] op (Issue #3354), or `None` when the
1694    /// clause projects none. A provenance-named accessor with a malformed
1695    /// argument is a structured error (never silently dropped), reusing the same
1696    /// [`as_provenance_accessor`](Self::as_provenance_accessor) recognition the
1697    /// `WHERE` half uses.
1698    ///
1699    /// v1 supports only the clean single-entity forms
1700    /// (`RETURN <entity>, <accessor(s)>` and `RETURN <accessor(s)>` over the
1701    /// single bound node). Building general property-projection-into-columns is
1702    /// out of scope, so an accessor mixed with a property projection
1703    /// (`RETURN n.name, source(n)`), an aggregate
1704    /// (`RETURN n.dept, count(*), source(n)`), or a variable other than the
1705    /// single bound entity (`RETURN a, source(b)`) is **rejected** with a
1706    /// structured [`CypherError::UnsupportedFeature`] rather than silently
1707    /// dropping a column or resolving the wrong entity. (Edge/traversal-variable
1708    /// accessors never reach this path: a multi-variable MATCH routes to the
1709    /// multi-pattern evaluator, so provenance projection is node-entity-scoped in
1710    /// v1.)
1711    fn build_provenance_projection(
1712        &self,
1713        return_clause: &CypherReturn,
1714        patterns: &[CypherPattern],
1715        aggregated: bool,
1716    ) -> Result<Option<QueryOp>, CypherError> {
1717        let mut items = Vec::new();
1718        let mut entity_binding: Option<String> = None;
1719        // Distinct entity variables referenced across the bare entity and every
1720        // accessor argument; the supported form references exactly one.
1721        let mut referenced_vars: BTreeSet<String> = BTreeSet::new();
1722        let mut has_property_projection = false;
1723        for item in &return_clause.items {
1724            let (expr, alias) = match item {
1725                CypherReturnItem::Expression { expr, alias } => (expr, alias.clone()),
1726                // A bare `RETURN n` variable item -- preserve it as the entity
1727                // binding so it stays observable alongside projected columns.
1728                CypherReturnItem::Variable(name) => {
1729                    if entity_binding.is_none() {
1730                        entity_binding = Some(name.clone());
1731                    }
1732                    referenced_vars.insert(name.clone());
1733                    continue;
1734                }
1735                CypherReturnItem::Star => continue,
1736            };
1737            match expr {
1738                // The parser emits a bare variable as an `Expression` item.
1739                CypherExpr::Variable(name) => {
1740                    if entity_binding.is_none() {
1741                        entity_binding = Some(name.clone());
1742                    }
1743                    referenced_vars.insert(name.clone());
1744                }
1745                // A property projection (`n.name`) alongside an accessor is the
1746                // unsupported mix (rejected below once an accessor is present).
1747                CypherExpr::Property { .. } => {
1748                    has_property_projection = true;
1749                }
1750                CypherExpr::FunctionCall { args, .. } => {
1751                    if let Some(field) = self.as_provenance_accessor(expr)? {
1752                        // `as_provenance_accessor` guarantees exactly one bound
1753                        // variable argument.
1754                        let var = match args.as_slice() {
1755                            [CypherExpr::Variable(v)] => v.clone(),
1756                            _ => String::new(),
1757                        };
1758                        referenced_vars.insert(var.clone());
1759                        let output_name =
1760                            alias.unwrap_or_else(|| format!("{}({})", field.accessor_name(), var));
1761                        items.push(ProvenanceProjectionItem { output_name, field });
1762                    }
1763                }
1764                _ => {}
1765            }
1766        }
1767        if items.is_empty() {
1768            return Ok(None);
1769        }
1770
1771        // An accessor is projected: enforce the v1 single-entity contract.
1772        if aggregated {
1773            return Err(CypherError::UnsupportedFeature(
1774                "combining a provenance accessor with an aggregate in RETURN is not supported \
1775                 (v1); an aggregating RETURN produces its own computed columns"
1776                    .to_string(),
1777            ));
1778        }
1779        if has_property_projection {
1780            return Err(CypherError::UnsupportedFeature(
1781                "combining a property projection with a provenance accessor in RETURN is not \
1782                 supported (v1); return the entity itself (RETURN n, source(n)) or the accessors \
1783                 alone (RETURN source(n))"
1784                    .to_string(),
1785            ));
1786        }
1787        // The supported form binds exactly one entity: a single MATCH node with a
1788        // variable. Every referenced variable (bare entity + accessor args) must
1789        // be that node. This rejects the multi-variable mismatch
1790        // (`RETURN a, source(b)`, including an accessor over an unbound variable),
1791        // which the single-entity positional pipeline would otherwise resolve
1792        // against the wrong (or a nonexistent) entity.
1793        let sole_entity = Self::sole_pattern_node_var(patterns);
1794        let matches_sole = match &sole_entity {
1795            Some(v) => referenced_vars.len() == 1 && referenced_vars.contains(v),
1796            None => false,
1797        };
1798        if !matches_sole {
1799            return Err(CypherError::UnsupportedFeature(
1800                "provenance projection in RETURN is supported only for a single bound entity \
1801                 whose accessor variable matches the returned node (RETURN n, source(n)); \
1802                 projecting an accessor over a different variable or a multi-entity match is not \
1803                 supported (v1)"
1804                    .to_string(),
1805            ));
1806        }
1807
1808        Ok(Some(QueryOp::ProjectProvenance(ProvenanceProjection {
1809            entity_binding,
1810            items,
1811        })))
1812    }
1813
1814    /// The variable of the single bound node when the MATCH is exactly one
1815    /// pattern consisting of one node element with a variable and no
1816    /// relationships; `None` otherwise. This is the one entity the single-entity
1817    /// positional pipeline can correctly attribute a provenance accessor to.
1818    fn sole_pattern_node_var(patterns: &[CypherPattern]) -> Option<String> {
1819        let [pattern] = patterns else {
1820            return None;
1821        };
1822        let [CypherPatternElement::Node(node)] = pattern.elements.as_slice() else {
1823            return None;
1824        };

1826    }

1881    /// Check if the ORDER BY clause contains a vector similarity function call
1882    /// and, if so, emit a [`QueryOp::RankBySimilarity`] instead of a regular Sort.
1883    ///
1884    /// Returns `true` if a vector rank was emitted (meaning the caller should
1885    /// skip normal sort emission for this ORDER BY).
1886    fn try_emit_vector_rank(
1887        &self,
1888        return_clause: &CypherReturn,
1889        ops: &mut Vec<QueryOp>,
1890    ) -> Result<bool, CypherError> {
1891        if return_clause.order_by.len() != 1 {
1892            return Ok(false);
1893        }
1894
1895        // Every `vector.fn(...) AS <alias>` item in the RETURN list (Issue #555).
1896        // v1 supports at most ONE such projection, and it must be the SAME
1897        // function (name + args) as the one driving the rank -- otherwise the
1898        // materialized score would come from a different metric/embedding than
1899        // the ordering, a silent wrong answer (adversarial finding #2).
1900        let aliased = Self::aliased_vector_return_items(return_clause);
1901        if aliased.len() > 1 {
1902            return Err(CypherError::UnsupportedFeature(
1903                "multiple vector score projections in a single RETURN are not \
1904                 supported in v1 (project a single vector.fn(...) AS <alias>)"
1905                    .to_string(),
1906            ));
1907        }
1908
1909        let order_item = &return_clause.order_by[0];
1910
1911        // Arm 1: `ORDER BY vector.fn(entity.prop, <emb>) <DIR>` directly.
1912        if let CypherExpr::FunctionCall { name, args, .. } = &order_item.expr
1913            && is_vector_function(name)
1914        {
1915            let metric = vector_metric_for(name);
1916            // Reject an ordering direction that would request farthest-first
1917            // (silent wrong answer #1): the rank always yields nearest-first, so
1918            // the direction must match the metric's natural nearest-first sense.
1919            Self::check_vector_order_direction(name, metric, order_item.descending)?;
1920            // Attach the projected alias only when it is the SAME function+args.
1921            let score_alias = match aliased.first() {
1922                None => None,
1923                Some(&(a_name, a_args, alias)) => {
1924                    if a_name == name && a_args == args.as_slice() {
1925                        Some(alias.clone())
1926                    } else {
1927                        return Err(CypherError::UnsupportedFeature(
1928                            "the RETURN projects a vector function that differs \
1929                             from the ORDER BY vector function; independent \
1930                             multi-score projection is not supported in v1 \
1931                             (project and order by the same vector.fn(...))"
1932                                .to_string(),
1933                        ));
1934                    }
1935                }
1936            };
1937            let (property_key, embedding) = self.extract_vector_args(args)?;
1938            ops.push(QueryOp::RankBySimilarity {
1939                embedding,
1940                top_k: return_clause.limit,
1941                property_key: Some(property_key),
1942                metric,
1943                threshold: None,
1944                score_alias,
1945            });
1946            return Ok(true);
1947        }
1948
1949        // Arm 2: `ORDER BY <alias>` where `<alias>` was defined as a vector
1950        // function in the RETURN items (e.g. `vector.cosine(...) AS score` then
1951        // `ORDER BY score DESC`). The score is materialized under `<alias>`.
1952        if let CypherExpr::Variable(ref alias_name) = order_item.expr
1953            && let Some(&(name, args, alias)) = aliased
1954                .iter()
1955                .find(|entry| entry.2.as_str() == alias_name.as_str())
1956        {
1957            let metric = vector_metric_for(name);
1958            Self::check_vector_order_direction(name, metric, order_item.descending)?;
1959            let (property_key, embedding) = self.extract_vector_args(args)?;
1960            ops.push(QueryOp::RankBySimilarity {
1961                embedding,
1962                top_k: return_clause.limit,
1963                property_key: Some(property_key),
1964                metric,
1965                threshold: None,
1966                score_alias: Some(alias.clone()),
1967            });
1968            return Ok(true);
1969        }
1970
1971        Ok(false)
1972    }

1974    /// Every `RETURN` item of the form `vector.fn(...) AS <alias>`, as
1975    /// `(name, args, alias)`. Used to materialize the similarity score under its
1976    /// alias (Issue #555) and to reject unsupported multi-/mismatched-projection
1977    /// forms (adversarial finding #2).
1978    fn aliased_vector_return_items(
1979        return_clause: &CypherReturn,
1980    ) -> Vec<(&String, &[CypherExpr], &String)> {
1981        return_clause
1982            .items
1983            .iter()
1984            .filter_map(|item| match item {
1985                CypherReturnItem::Expression {
1986                    expr: CypherExpr::FunctionCall { name, args, .. },
1987                    alias: Some(alias),
1988                } if is_vector_function(name) => Some((name, args.as_slice(), alias)),
1989                _ => None,

1994    /// Reject a vector `ORDER BY` direction that would request farthest-first
1995    /// results (adversarial finding #1). The rank always yields **nearest-first**,
1996    /// so the ordering direction must match the metric's natural nearest-first
1997    /// sense: `DESC` for similarity metrics (cosine / similarity / dot_product,
1998    /// higher = nearer) and `ASC` for the distance-flavored `vector.euclidean`
1999    /// (lower = nearer, matching the Issue #553 example). Any other direction
2000    /// asks for farthest-first, which v1 does not support (rejected clearly
2001    /// rather than silently returning nearest-first).
2002    fn check_vector_order_direction(
2003        func_name: &str,
2004        metric: VectorMetric,
2005        descending: bool,
2006    ) -> Result<(), CypherError> {
2007        let natural_descending = !matches!(metric, VectorMetric::Euclidean);
2008        if descending != natural_descending {
2009            let (wanted, other) = if natural_descending {
2010                ("DESC", "ASC")
2011            } else {
2012                ("ASC", "DESC")
2013            };
2014            return Err(CypherError::UnsupportedFeature(format!(
2015                "vector ranking over {func_name} supports only {wanted} ordering \
2016                 (nearest-first); {other} (farthest-first) vector ordering is not \
2017                 supported in v1"
2018            )));
2019        }
2020        Ok(())
2021    }

2111    /// Convert a base-pattern `WHERE` clause into ops, lifting any vector
2112    /// similarity **threshold** comparison (`vector.similarity(d.emb, $q) > 0.8`,
2113    /// Issue #553) out of the predicate into a [`QueryOp::RankBySimilarity`]
2114    /// carrying a [`ScoreThreshold`], and emitting a `Filter` for the residual
2115    /// (non-vector) part of the predicate.
2116    ///
2117    /// A threshold conjunct may appear alone or as one term of a top-level
2118    /// `AND` chain (`vector.similarity(...) > 0.8 AND d.lang = 'en'`); the
2119    /// residual conjuncts become an ordinary `Filter` evaluated **before** the
2120    /// rank so the score is only computed over surviving rows.
2121    fn convert_where_clause(
2122        &self,
2123        expr: &CypherExpr,
2124        ops: &mut Vec<QueryOp>,
2125    ) -> Result<(), CypherError> {
2126        // Split the top-level AND spine so a vector threshold can be lifted out
2127        // of `A AND B AND ...` while the rest becomes an ordinary Filter.
2128        let conjuncts = flatten_logical_operands(expr, |e| match e {
2129            CypherExpr::And(left, right) => Some((left.as_ref(), right.as_ref())),
2130            _ => None,
2131        });
2132
2133        let mut ranks: Vec<QueryOp> = Vec::new();
2134        let mut residual: Vec<&CypherExpr> = Vec::new();
2135        for conj in conjuncts {
2136            if let Some(rank) = self.try_vector_threshold(conj)? {
2137                ranks.push(rank);
2138            } else {
2139                residual.push(conj);
2140            }
2141        }
2142
2143        // No vector threshold: preserve the original single-Filter behavior.
2144        if ranks.is_empty() {
2145            let predicate = self.convert_expr_to_predicate(expr)?;
2146            ops.push(QueryOp::Filter(predicate));
2147            return Ok(());
2148        }
2149
2150        // Residual property predicate first (filter, then score survivors)...
2151        if !residual.is_empty() {
2152            let predicate = if residual.len() == 1 {
2153                self.convert_expr_to_predicate(residual[0])?
2154            } else {
2155                let preds = residual
2156                    .iter()
2157                    .map(|e| self.convert_expr_to_predicate(e))
2158                    .collect::<Result<Vec<_>, _>>()?;
2159                Predicate::And(preds)
2160            };
2161            ops.push(QueryOp::Filter(predicate));
2162        }
2163        // ...then the vector similarity threshold rank(s).
2164        //
2165        // v1 BEHAVIOR (adversarial finding #6): the threshold is implemented as a
2166        // RankBySimilarity with `top_k: None`, which computes the score for every
2167        // surviving row and returns them **sorted by similarity descending** (the
2168        // rank's inherent output order). So a bare
2169        // `WHERE vector.similarity(...) > t RETURN d` (no ORDER BY) returns the
2170        // passing rows nearest-first rather than in scan order, and a trailing
2171        // `LIMIT n` therefore yields the top-n *nearest* passing rows rather than
2172        // the first-n in scan order. openCypher leaves row order unspecified
2173        // without ORDER BY, so this is a superset-correct (not wrong) result; it
2174        // is pinned by a test and documented in the compatibility guide. A
2175        // score-unaware Filter-only lowering is a possible follow-up but would
2176        // still need the score computed to evaluate the predicate.
2177        ops.append(&mut ranks);
2178        Ok(())
2179    }

2181    /// If `expr` is a vector-similarity threshold comparison
2182    /// (`vector.fn(entity.prop, <emb>) <cmp> <number>`, or the mirrored
2183    /// `<number> <cmp> vector.fn(...)`), build the [`QueryOp::RankBySimilarity`]
2184    /// that computes the score and keeps only rows satisfying the threshold.
2185    /// Returns `Ok(None)` for any non-vector-threshold expression.
2186    fn try_vector_threshold(&self, expr: &CypherExpr) -> Result<Option<QueryOp>, CypherError> {
2187        let CypherExpr::Comparison { left, op, right } = expr else {
2188            return Ok(None);
2189        };
2190
2191        // Identify which side is the vector function and orient the comparison
2192        // so it reads `score <cmp> value`.
2193        let (func_name, args, cmp) = match (left.as_ref(), right.as_ref()) {
2194            (CypherExpr::FunctionCall { name, args, .. }, _) if is_vector_function(name) => {
2195                (name, args, *op)
2196            }
2197            (_, CypherExpr::FunctionCall { name, args, .. }) if is_vector_function(name) => {
2198                // `value <cmp> score` == `score <flipped-cmp> value`.
2199                (name, args, flip_cypher_comp_op(*op))
2200            }
2201            _ => return Ok(None),
2202        };
2203
2204        let Some(threshold_op) = score_comparison_for(cmp) else {
2205            // `=`/`<>` against a similarity score is not a threshold filter.
2206            return Err(CypherError::UnsupportedFeature(
2207                "vector similarity comparisons support only >, >=, <, <= \
2208                 thresholds (not = or <>)"
2209                    .to_string(),
2210            ));
2211        };
2212
2213        // The other operand must be a numeric literal threshold value.
2214        let value_expr = if matches!(left.as_ref(), CypherExpr::FunctionCall { name, .. } if is_vector_function(name))
2215        {
2216            right.as_ref()
2217        } else {
2218            left.as_ref()
2219        };
2220        let value = match value_expr {
2221            CypherExpr::Value(CypherValue::Float(f)) => *f as f32,
2222            CypherExpr::Value(CypherValue::Int(i)) => *i as f32,
2223            other => {
2224                return Err(CypherError::SemanticError(format!(
2225                    "vector similarity threshold must compare against a numeric \
2226                     literal, got: {other:?}"
2227                )));
2228            }
2229        };
2230
2231        let (property_key, embedding) = self.extract_vector_args(args)?;
2232        Ok(Some(QueryOp::RankBySimilarity {
2233            embedding,
2234            top_k: None,
2235            property_key: Some(property_key),
2236            metric: vector_metric_for(func_name),
2237            threshold: Some(ScoreThreshold {
2238                op: threshold_op,
2239                value,
2240            }),
2241            score_alias: None,
2242        }))
2243    }

2250/// Flatten a left-nested chain of a single logical operator (`AND` or `OR`, as
2251/// built by the parser's `parse_and_expr` / `parse_or_expr`) into its ordered
2252/// operand list, iteratively — without recursing down the operator spine.
2253///
2254/// `split` returns `Some((left, right))` for a node of the operator being
2255/// flattened and `None` for any other node. Nodes for which `split` returns
2256/// `None` (including a sub-tree of the *other* logical operator) are returned
2257/// as opaque operands, to be converted individually by the caller. Operand
2258/// order is preserved left-to-right.
2259///
2260/// This keeps converting an arbitrarily long `a AND b AND c ...` (or `OR`)
2261/// chain from overflowing the stack (issue #3404): the spine is walked with an
2262/// explicit heap stack instead of one call frame per operator.
2263fn flatten_logical_operands<'a>(
2264    root: &'a CypherExpr,
2265    split: impl Fn(&'a CypherExpr) -> Option<(&'a CypherExpr, &'a CypherExpr)>,
2266) -> Vec<&'a CypherExpr> {
2267    let mut operands = Vec::new();
2268    let mut stack = vec![root];
2269    // Pushing right-then-left means the left operand is popped first, so leaves
2270    // are collected in source (left-to-right) order.
2271    while let Some(node) = stack.pop() {
2272        if let Some((left, right)) = split(node) {
2273            stack.push(right);
2274            stack.push(left);
2275        } else {
2276            operands.push(node);
2277        }
2278    }
2279    operands
2280}

2286/// Parse a timestamp string into a [`Timestamp`] (HybridTimestamp).
2287///
2288/// Supports:
2289/// - Unix microseconds: `"1705312800000000"`
2290/// - ISO 8601 with timezone: `"2024-01-15T10:00:00Z"`
2291/// - Date only: `"2024-01-15"` (midnight UTC)
2292fn parse_timestamp_string(s: &str) -> Result<Timestamp, CypherError> {
2293    let trimmed = s.trim().trim_matches('\'').trim_matches('"');
2294
2295    // Try unix microseconds first
2296    if let Ok(micros) = trimmed.parse::<i64>() {
2297        return Ok(Timestamp::from(micros));
2298    }
2299
2300    // ISO 8601 with timezone: 2024-01-15T10:00:00Z
2301    if let Ok(dt) = trimmed.parse::<chrono::DateTime<chrono::Utc>>() {
2302        return Ok(Timestamp::from(dt.timestamp_micros()));
2303    }
2304
2305    // Date only: 2024-01-15
2306    if let Ok(date) = trimmed.parse::<chrono::NaiveDate>()
2307        && let Some(dt) = date.and_hms_opt(0, 0, 0)
2308    {
2309        return Ok(Timestamp::from(dt.and_utc().timestamp_micros()));
2310    }
2311
2312    Err(CypherError::InvalidTimestamp(s.to_string()))
2313}

2324///

2336    }
