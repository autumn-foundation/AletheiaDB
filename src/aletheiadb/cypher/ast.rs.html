1//! Cypher Query Language Abstract Syntax Tree (AST)
2//!
3//! This module defines the AST types produced by the Cypher parser. Every type
4//! in this module derives [`Debug`], [`Clone`], and [`PartialEq`] so that ASTs
5//! can be inspected, duplicated, and compared in tests.
6//!
7//! # Structure
8//!
9//! ```text
10//! CypherStatement
11//!   ├── CypherPattern          (graph pattern: nodes + relationships)
12//!   │     └── CypherPatternElement
13//!   │           ├── CypherNodePattern
14//!   │           └── CypherRelPattern
15//!   ├── CypherExpr             (WHERE / filter expressions)
16//!   ├── CypherReturn           (projection: items, ordering, pagination)
17//!   ├── CypherTemporal         (bi-temporal qualifiers)
18//!   └── CypherWith             (intermediate projection)
19//! ```
20
21use std::sync::Arc;
22
23// ---------------------------------------------------------------------------
24// Top-level statement
25// ---------------------------------------------------------------------------
26
27/// A complete Cypher statement ready for planning and execution.
28///
29/// Reading (`MATCH`) and write (`CREATE` / `MERGE` / `SET` / `DELETE`)
30/// statements are supported; further clause types are added as new variants in
31/// future phases.
32#[derive(Debug, Clone, PartialEq)]
33pub enum CypherStatement {
34    /// A `MATCH` (or `OPTIONAL MATCH`) statement that reads from the graph.
35    Match {
36        /// Whether this is an `OPTIONAL MATCH` (returns nulls for unmatched patterns).
37        optional: bool,
38        /// One or more comma-separated graph patterns to match.
39        pattern: Vec<CypherPattern>,
40        /// An optional `WHERE` clause that filters matched results.
41        where_clause: Option<CypherExpr>,
42        /// The `RETURN` clause that defines output projection.
43        return_clause: CypherReturn,
44        /// An optional temporal qualifier (e.g., `AS OF TIMESTAMP ...`).
45        temporal: Option<CypherTemporal>,
46        /// Zero or more intermediate `WITH` projections.
47        with_clauses: Vec<CypherWith>,
48        /// Zero or more subsequent `OPTIONAL MATCH` clauses.
49        ///
50        /// Source ordering relative to `with_clauses` is preserved via
51        /// [`CypherOptionalMatch::preceding_withs`].
52        optional_matches: Vec<CypherOptionalMatch>,
53    },
54
55    /// A standalone `UNWIND <list> AS <var> RETURN ...` statement (Issue #559).
56    ///
57    /// `UNWIND` expands a list value into one row per element, binding each
58    /// element to `variable`. This variant models the *standalone* form only
59    /// (an `UNWIND` that is not preceded by `MATCH`/`WITH` and does not feed a
60    /// subsequent graph pattern); it is executed directly by the dedicated
61    /// Cypher UNWIND runtime rather than lowered into the graph query pipeline.
62    ///
63    /// openCypher list semantics apply: an empty list and the `null` value both
64    /// expand to zero rows.
65    Unwind {
66        /// The list-valued source expression. Supported forms are a list
67        /// literal (`[...]`), a parameter reference (`$list`), or the `null`
68        /// literal; other (row-context-dependent) expressions are rejected at
69        /// execution time.
70        source: CypherExpr,
71        /// The variable each list element is bound to (the `AS <var>` target).
72        variable: String,
73        /// The trailing `RETURN` projection (with optional `DISTINCT`,
74        /// `ORDER BY`, `SKIP`, `LIMIT`).
75        return_clause: CypherReturn,
76    },
77
78    /// `EXPLAIN <statement>` -- return the query plan for the wrapped statement
79    /// **without executing it** (Issue #562).
80    ///
81    /// The inner statement is any ordinary readable statement (a `MATCH`,
82    /// optionally with a leading temporal clause). A nested/duplicate prefix
83    /// (`EXPLAIN EXPLAIN`, `EXPLAIN PROFILE`) is rejected at parse time, so the
84    /// boxed inner is never itself an `Explain`/`Profile`.
85    Explain(Box<CypherStatement>),
86
87    /// `PROFILE <statement>` -- **execute** the wrapped statement and return its
88    /// plan annotated with per-operator executed statistics (row counts and
89    /// timing) (Issue #562).
90    ///
91    /// Same nesting rules as [`CypherStatement::Explain`].
92    Profile(Box<CypherStatement>),
93
94    /// A write statement (Issue #560): `CREATE` / `SET` / `DELETE` /
95    /// `DETACH DELETE`, optionally preceded by a reading `MATCH` and followed by
96    /// a `RETURN`.
97    ///
98    /// Write statements are **not** lowered into the read-only [`Query`] IR;
99    /// they are executed directly against the database's native write APIs (so
100    /// each mutation records the correct bi-temporal version). They are reachable
101    /// only through `AletheiaDB::execute_cypher` / `execute_cypher_with_params`;
102    /// the MCP `query` tool rejects every mutating clause *before* the parser
103    /// runs (`crate::query::read_only::detect_mutating_clause`), so this variant
104    /// never executes through that read-only surface.
105    ///
106    /// [`Query`]: crate::query::Query
107    Write(CypherWriteStatement),
108}
109
110/// A complete Cypher write statement (Issue #560).
111///
112/// Grammar (v1):
113///
114/// ```text
115/// write_stmt := [reading] write_clause+ [RETURN ...]
116/// reading    := MATCH pattern_list [WHERE expr]
117/// write_clause := CREATE pattern_list
118///               | SET set_item (',' set_item)*
119///               | [DETACH] DELETE variable (',' variable)*
120///               | MERGE pattern (ON CREATE SET set_item (',' set_item)*
121///                               | ON MATCH SET set_item (',' set_item)*)*
122/// set_item   := variable '.' property '=' value
123/// ```
124#[derive(Debug, Clone, PartialEq)]
125pub struct CypherWriteStatement {
126    /// An optional leading reading clause (`MATCH ... [WHERE ...]`) whose matched
127    /// rows drive the write clauses. `None` for a statement that opens directly
128    /// with `CREATE`.
129    pub reading: Option<CypherReadingClause>,
130    /// One or more write clauses, applied in source order per matched row.
131    pub clauses: Vec<CypherWriteClause>,
132    /// An optional trailing `RETURN` projecting the affected entities.
133    pub return_clause: Option<CypherReturn>,
134}
135
136/// The reading (`MATCH ... [WHERE ...]`) part of a write statement.
137#[derive(Debug, Clone, PartialEq)]
138pub struct CypherReadingClause {
139    /// One or more comma-separated graph patterns to match.
140    pub pattern: Vec<CypherPattern>,
141    /// An optional `WHERE` clause filtering matched rows.
142    pub where_clause: Option<CypherExpr>,
143}
144
145/// A single write clause within a [`CypherWriteStatement`].
146#[derive(Debug, Clone, PartialEq)]
147pub enum CypherWriteClause {
148    /// `CREATE <pattern_list>` -- create the described nodes and relationships.
149    Create(Vec<CypherPattern>),
150    /// `SET <set_item> (',' <set_item>)*` -- assign properties on bound entities.
151    Set(Vec<CypherSetItem>),
152    /// `[DETACH] DELETE <variable> (',' <variable>)*` -- delete bound entities.
153    Delete {
154        /// Whether this is a `DETACH DELETE` (cascade-remove connected edges).
155        detach: bool,
156        /// The variables to delete (nodes or relationships).
157        targets: Vec<String>,
158    },
159    /// `MERGE <pattern> [ON CREATE SET ...] [ON MATCH SET ...]` (Issue #3548) --
160    /// match the pattern if it already exists (per openCypher whole-pattern
161    /// semantics), otherwise create the entire pattern. `ON CREATE SET` applies
162    /// only on the create branch; `ON MATCH SET` only on the match branch.
163    Merge {
164        /// The single graph pattern to match-or-create.
165        pattern: CypherPattern,
166        /// Assignments applied only when the pattern is created.
167        on_create: Vec<CypherSetItem>,
168        /// Assignments applied only when the pattern is matched.
169        on_match: Vec<CypherSetItem>,
170    },
171}
172
173/// A single `SET n.prop = value` assignment.
174#[derive(Debug, Clone, PartialEq)]
175pub struct CypherSetItem {
176    /// The variable whose property is being set.
177    pub variable: String,
178    /// The property key to assign.
179    pub property: String,
180    /// The literal (or parameter) value assigned.
181    pub value: CypherValue,
182}
183
184/// A subsequent `OPTIONAL MATCH` clause within a `MATCH` statement.
185///
186/// Per openCypher semantics the clause's `WHERE` is part of the optional
187/// pattern itself: it participates in the matched/unmatched decision rather
188/// than filtering rows afterwards.
189#[derive(Debug, Clone, PartialEq)]
190pub struct CypherOptionalMatch {
191    /// One or more comma-separated graph patterns to match optionally.
192    pub pattern: Vec<CypherPattern>,
193    /// An optional `WHERE` clause scoped to this optional pattern.
194    pub where_clause: Option<CypherExpr>,
195    /// Number of `WITH` clauses (in the statement's `with_clauses`) that
196    /// appear before this clause in the query text. This preserves clause
197    /// ordering so conversion can interleave `WITH ... WHERE` filters and
198    /// optional patterns exactly as written.
199    pub preceding_withs: usize,
200}

202// ---------------------------------------------------------------------------
203// Graph patterns
204// ---------------------------------------------------------------------------
205
206/// A single graph pattern consisting of alternating node and relationship elements.
207///
208/// A valid pattern always starts and ends with a node. For example,
209/// `(a:Person)-[:KNOWS]->(b:Person)` has three elements:
210/// `[Node(a), Relationship(KNOWS), Node(b)]`.
211#[derive(Debug, Clone, PartialEq)]
212pub struct CypherPattern {
213    /// The ordered sequence of nodes and relationships that form this pattern.
214    pub elements: Vec<CypherPatternElement>,
215}
216
217/// A single element inside a graph pattern -- either a node or a relationship.
218#[derive(Debug, Clone, PartialEq)]
219pub enum CypherPatternElement {
220    /// A node pattern, e.g. `(n:Person {name: 'Alice'})`.
221    Node(CypherNodePattern),
222    /// A relationship pattern, e.g. `-[:KNOWS]->`.
223    Relationship(CypherRelPattern),
224}
225
226/// A node pattern within a `MATCH` clause.
227///
228/// Examples:
229/// - `(n)` -- bare variable, no label or properties
230/// - `(n:Person)` -- variable with label
231/// - `(:Person {name: 'Alice'})` -- anonymous node with label and properties
232#[derive(Debug, Clone, PartialEq)]
233pub struct CypherNodePattern {
234    /// An optional variable name bound to this node (e.g. `n` in `(n:Person)`).
235    pub variable: Option<String>,
236    /// Zero or more labels required on the node (e.g. `["Person"]`).
237    pub labels: Vec<String>,
238    /// Zero or more property key-value constraints (e.g. `{name: 'Alice'}`).
239    pub properties: Vec<(String, CypherValue)>,
240}
241
242/// A relationship pattern within a `MATCH` clause.
243///
244/// Examples:
245/// - `-[:KNOWS]->` -- outgoing, typed
246/// - `<-[:FOLLOWS]-` -- incoming, typed
247/// - `-[r:KNOWS|LIKES*1..3]->` -- outgoing, variable, multi-type, variable-length
248#[derive(Debug, Clone, PartialEq)]
249pub struct CypherRelPattern {
250    /// An optional variable name bound to this relationship.
251    pub variable: Option<String>,
252    /// Zero or more relationship types (e.g. `["KNOWS", "LIKES"]`).
253    pub rel_types: Vec<String>,
254    /// The traversal direction of this relationship.
255    pub direction: CypherDirection,
256    /// An optional variable-length path specifier (e.g. `*1..3`).
257    pub depth: Option<CypherDepth>,
258    /// Zero or more property key-value constraints on the relationship.
259    pub properties: Vec<(String, CypherValue)>,
260}
261
262/// The traversal direction of a relationship in a pattern.
263#[derive(Debug, Clone, Copy, PartialEq, Eq)]
264pub enum CypherDirection {
265    /// Outgoing: `(a)-[]->(b)`.
266    Outgoing,
267    /// Incoming: `(a)<-[]-(b)`.
268    Incoming,
269    /// Bidirectional (either direction): `(a)-[]-(b)`.
270    Both,
271}
272
273/// A variable-length path depth specifier.
274///
275/// Appears after `*` in a relationship pattern to indicate how many hops are
276/// allowed.
277#[derive(Debug, Clone, Copy, PartialEq, Eq)]
278pub enum CypherDepth {
279    /// `*` -- any number of hops (including zero).
280    Unbounded,
281    /// `*N` -- exactly N hops.
282    Exact(usize),
283    /// `*..M` -- at most M hops.
284    Max(usize),
285    /// `*N..` -- at least N hops.
286    Min(usize),
287    /// `*N..M` -- between N and M hops (inclusive).
288    Range {
289        /// Minimum number of hops.
290        min: usize,
291        /// Maximum number of hops.
292        max: usize,
293    },
294}

300/// A literal value or parameter reference in a Cypher expression.
301#[derive(Debug, Clone, PartialEq)]
302pub enum CypherValue {
303    /// The `NULL` literal.
304    Null,
305    /// A boolean literal (`TRUE` / `FALSE`).
306    Bool(bool),
307    /// A 64-bit signed integer literal.
308    Int(i64),
309    /// A 64-bit floating-point literal.
310    Float(f64),
311    /// A string literal (single- or double-quoted).
312    String(String),
313    /// A parameter reference (e.g. `$name`). The string is the parameter name
314    /// without the leading `$`.
315    Parameter(String),
316    /// A dense vector literal used for similarity search.
317    Vector(Arc<[f32]>),
318}

324/// An expression node in the Cypher AST.
325///
326/// Expressions appear in `WHERE` clauses, `RETURN` items, and `ORDER BY` clauses.
327/// They form a tree with operators as interior nodes and values / variables as leaves.
328#[derive(Debug, Clone, PartialEq)]
329pub enum CypherExpr {
330    /// A literal value (number, string, boolean, null, parameter, vector).
331    Value(CypherValue),
332    /// A bare variable reference (e.g. `n`).
333    Variable(String),
334    /// A property access expression (e.g. `n.name`).
335    Property {
336        /// The variable whose property is being accessed.
337        variable: String,
338        /// The property key name.
339        property: String,
340    },
341    /// A comparison expression (e.g. `n.age > 18`).
342    Comparison {
343        /// The left-hand operand.
344        left: Box<CypherExpr>,
345        /// The comparison operator.
346        op: CypherCompOp,
347        /// The right-hand operand.
348        right: Box<CypherExpr>,
349    },
350    /// Logical AND of two expressions.
351    And(Box<CypherExpr>, Box<CypherExpr>),
352    /// Logical OR of two expressions.
353    Or(Box<CypherExpr>, Box<CypherExpr>),
354    /// Logical NOT of an expression.
355    Not(Box<CypherExpr>),
356    /// `IS NULL` predicate.
357    IsNull(Box<CypherExpr>),
358    /// `IS NOT NULL` predicate.
359    IsNotNull(Box<CypherExpr>),
360    /// `IN [...]` list membership test.
361    In {
362        /// The expression to test for membership.
363        expr: Box<CypherExpr>,
364        /// The list of values to test against.
365        values: Vec<CypherExpr>,
366    },
367    /// `CONTAINS 'substring'` string predicate.
368    Contains {
369        /// The expression whose string representation is tested.
370        expr: Box<CypherExpr>,
371        /// The substring to search for.
372        substring: String,
373    },
374    /// `STARTS WITH 'prefix'` string predicate.
375    StartsWith {
376        /// The expression whose string representation is tested.
377        expr: Box<CypherExpr>,
378        /// The required prefix.
379        prefix: String,
380    },
381    /// `ENDS WITH 'suffix'` string predicate.
382    EndsWith {
383        /// The expression whose string representation is tested.
384        expr: Box<CypherExpr>,
385        /// The required suffix.
386        suffix: String,
387    },
388    /// A function call (e.g. `count(n)`, `avg(n.age)`).
389    FunctionCall {
390        /// The function name (case-insensitive at parse time).
391        name: String,
392        /// The arguments passed to the function.
393        ///
394        /// For the `count(*)` aggregate the single argument is
395        /// [`CypherExpr::Star`].
396        args: Vec<CypherExpr>,
397        /// Whether the call used the `DISTINCT` quantifier, e.g.
398        /// `count(DISTINCT n.dept)`. Only meaningful for aggregate functions;
399        /// always `false` for ordinary/vector functions.
400        distinct: bool,
401    },
402    /// The `*` wildcard argument, used only inside `count(*)`.
403    Star,
404    /// A parenthesized sub-expression used for grouping.
405    Grouped(Box<CypherExpr>),
406    /// A list literal, e.g. `[1, 2, 3]` or `[[1, 2], [3, 4]]`.
407    ///
408    /// Currently produced (and consumed) by the `UNWIND` source position; a
409    /// list literal appearing where a scalar predicate operand is expected is
410    /// rejected during conversion rather than silently mishandled.
411    List(Vec<CypherExpr>),
412}

414/// A comparison operator in a Cypher expression.
415#[derive(Debug, Clone, Copy, PartialEq, Eq)]
416pub enum CypherCompOp {
417    /// `=` equality.
418    Eq,
419    /// `<>` or `!=` inequality.
420    Ne,
421    /// `<` less than.
422    Lt,
423    /// `<=` less than or equal.
424    Le,
425    /// `>` greater than.
426    Gt,
427    /// `>=` greater than or equal.
428    Ge,
429}

435/// The `RETURN` clause of a Cypher statement.
436///
437/// Controls which data is projected out of the query, along with ordering,
438/// deduplication, and pagination.
439#[derive(Debug, Clone, PartialEq)]
440pub struct CypherReturn {
441    /// Whether `DISTINCT` was specified to deduplicate results.
442    pub distinct: bool,
443    /// The items to return (variables, expressions, or `*`).
444    pub items: Vec<CypherReturnItem>,
445    /// Zero or more `ORDER BY` items that determine result ordering.
446    pub order_by: Vec<CypherOrderItem>,
447    /// An optional `SKIP n` offset for pagination.
448    pub skip: Option<usize>,
449    /// An optional `LIMIT n` cap on result count.
450    pub limit: Option<usize>,
451}
452
453/// A single item in the `RETURN` clause.
454#[derive(Debug, Clone, PartialEq)]
455pub enum CypherReturnItem {
456    /// `RETURN *` -- return all bound variables.
457    Star,
458    /// A bare variable name (e.g. `RETURN n`).
459    Variable(String),
460    /// An arbitrary expression with an optional alias (e.g. `RETURN n.name AS name`).
461    Expression {
462        /// The expression to evaluate and return.
463        expr: CypherExpr,
464        /// An optional alias for the column (set via `AS`).
465        alias: Option<String>,
466    },
467}
468
469/// A single `ORDER BY` item specifying sort expression and direction.
470#[derive(Debug, Clone, PartialEq)]
471pub struct CypherOrderItem {
472    /// The expression to sort by.
473    pub expr: CypherExpr,
474    /// Whether to sort in descending order (`true` = DESC, `false` = ASC).
475    pub descending: bool,
476}

482/// A temporal qualifier for bi-temporal queries.
483///
484/// These extend standard Cypher with AletheiaDB's time-travel capabilities.
485#[derive(Debug, Clone, PartialEq)]
486pub enum CypherTemporal {
487    /// `AS OF TIMESTAMP '...'` -- point-in-time query (defaults to valid time).
488    AsOfTimestamp(String),
489    /// `FOR VALID_TIME AS OF '...'` -- explicit valid-time point query.
490    AsOfValidTime(String),
491    /// `FOR SYSTEM_TIME AS OF '...'` -- explicit system/transaction-time point query.
492    AsOfSystemTime(String),
493    /// Bi-temporal query combining valid time and system time.
494    BiTemporal {
495        /// The valid-time timestamp.
496        valid_time: String,
497        /// The system/transaction-time timestamp.
498        system_time: String,
499    },
500    /// `BETWEEN '...' AND '...'` -- time-range query.
501    Between {
502        /// The start of the time range (inclusive).
503        start: String,
504        /// The end of the time range (inclusive).
505        end: String,
506    },
507}

509// ---------------------------------------------------------------------------
510// WITH clause
511// ---------------------------------------------------------------------------
512
513/// An intermediate `WITH` projection clause.
514///
515/// `WITH` acts like a sub-`RETURN` that pipes results into subsequent clauses,
516/// optionally filtering with a `WHERE`. Per openCypher the clause body mirrors
517/// a `RETURN` body -- `WITH [DISTINCT] items [ORDER BY ...] [SKIP n] [LIMIT n]`
518/// -- followed by an optional trailing `WHERE` that filters the *projected*
519/// rows (Issue #556).
520#[derive(Debug, Clone, PartialEq)]
521pub struct CypherWith {
522    /// Whether `DISTINCT` was specified to deduplicate the projected rows.
523    pub distinct: bool,
524    /// The items to project through the `WITH`.
525    pub items: Vec<CypherReturnItem>,
526    /// Zero or more `ORDER BY` items ordering the projected rows.
527    pub order_by: Vec<CypherOrderItem>,
528    /// An optional `SKIP n` offset applied to the projected rows.
529    pub skip: Option<usize>,
530    /// An optional `LIMIT n` cap applied to the projected rows.
531    pub limit: Option<usize>,
532    /// An optional `WHERE` clause that filters after projection.
533    pub where_clause: Option<CypherExpr>,
534}
