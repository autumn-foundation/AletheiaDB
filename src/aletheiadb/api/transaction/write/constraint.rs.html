1//! Uniqueness constraint enforcement on the write path.
2//!
3//! Called from `commit_with_timestamp_inner` after conflict detection and
4//! before acquiring the timestamp lock.  Returns an RAII `ReservationGuard`
5//! that auto-rolls back on drop if the commit fails later.
6
7use crate::core::changefeed::EntityKind;
8use crate::core::constraint::{ConstraintRegistry, ReservationGuard};
9use crate::core::error::ConstraintError;
10use crate::core::graph::Node;
11use crate::core::id::NodeId;
12use crate::core::interning::InternedString;
13use crate::core::property::PropertyMap;
14use std::sync::Arc;
15
16use super::WriteTransaction;
17use crate::api::transaction::write_buffer::BufferedWrite;
18
19/// Enforce uniqueness constraints for the transaction's buffered node operations.
20///
21/// Returns a `ReservationGuard` that:
22/// - On drop (rollback path): releases all added reservations from the index.
23/// - After calling `guard.commit()` (success path): removes old-value reservations
24///   freed by updates/deletes.
25pub(crate) fn check_constraints(
26    tx: &WriteTransaction,
27    registry: &Arc<ConstraintRegistry>,
28) -> std::result::Result<ReservationGuard, ConstraintError> {
29    // Property type / required-key schema constraints (Issue #3378).
30    //
31    // Runs before the uniqueness reservation step and, like it, before the
32    // timestamp/WAL/apply phase, so a violation aborts the whole transaction
33    // with zero partial application. UpdateNode/UpdateEdge buffer the fully
34    // *merged* (effective) post-write property map (PATCH merge happens at
35    // buffer time), so the buffered map is exactly the state to validate — no
36    // re-merge with committed state is needed. Skipped entirely (zero cost)
37    // when no schema constraints are declared.
38    if !registry.schema_constraints_empty() {
39        for op in tx.buffer.operations() {
40            match op {
41                BufferedWrite::CreateNode {
42                    label, properties, ..
43                }
44                | BufferedWrite::UpdateNode {
45                    label, properties, ..
46                } => {
47                    registry.check_entity(EntityKind::Node, *label, properties)?;
48                }
49                BufferedWrite::CreateEdge {
50                    label, properties, ..
51                }
52                | BufferedWrite::UpdateEdge {
53                    label, properties, ..
54                } => {
55                    registry.check_entity(EntityKind::Edge, *label, properties)?;
56                }
57                _ => {}
58            }
59        }
60    }
61
62    // Borrow property maps directly from the buffer to avoid cloning on the hot path.
63    let mut added_refs: Vec<(InternedString, &PropertyMap, NodeId)> = Vec::new();
64
65    // Pre-tx nodes whose constraint keys are freed on a successful commit.
66    let mut removed_nodes: Vec<Node> = Vec::new();
67
68    for op in tx.buffer.operations() {
69        match op {
70            BufferedWrite::CreateNode {
71                node_id,
72                label,
73                properties,
74                ..
75            } => {
76                added_refs.push((*label, properties, *node_id));
77            }
78            BufferedWrite::UpdateNode {
79                node_id,
80                label,
81                properties,
82                ..
83            } => {
84                // Post-tx state is the new properties.
85                added_refs.push((*label, properties, *node_id));
86                // Pre-tx state: fetch current node to find keys being released.
87                if let Ok(current_node) = tx.current.get_node(*node_id) {
88                    removed_nodes.push(current_node);
89                }
90            }
91            BufferedWrite::DeleteNode { node_id, .. } => {
92                // All constraint keys for this node are freed on commit.
93                if let Ok(current_node) = tx.current.get_node(*node_id) {
94                    removed_nodes.push(current_node);
95                }
96            }
97            _ => {}
98        }
99    }
100
101    let removed_refs: Vec<(InternedString, &PropertyMap, NodeId)> = removed_nodes
102        .iter()
103        .map(|n| (n.label, &n.properties, n.id))
104        .collect();
105
106    registry.reserve_for_transaction(&added_refs, &removed_refs)
107}
