1//! Write transactions with ACID guarantees.
2//!
3//! Write transactions provide full ACID properties:
4//! - **Atomicity**: All-or-nothing commit via write buffering
5//! - **Consistency**: Referential integrity validation before commit
6//! - **Isolation**: Snapshot Isolation with write-write conflict detection
7//! - **Durability**: WAL with fsync guarantees

328    /// Seal designated property values in the write buffer (Issue #3359, PR-1b).
329    ///
330    /// Called from `commit_with_timestamp_inner` AFTER validation + constraint
331    /// checks (which must see plaintext values) but BEFORE WAL logging and apply,
332    /// so the WAL segment, the current tier, and the historical tier all receive
333    /// **byte-identical** sealed ciphertext (each envelope is sealed exactly once;
334    /// its random per-encryption nonce is shared across every tier). No-op when no
335    /// sealing context is attached (no active designation) or when no buffered
336    /// entity carries a designated key.
337    ///
338    /// # Errors
339    /// Propagates the seal error (fail-closed: a seal failure aborts the commit
340    /// rather than persisting plaintext of erasable data).
341    #[cfg(feature = "audit-export")]
342    pub(crate) fn seal_buffer(&mut self) -> Result<()> {
343        use super::BufferedWrite as BW;
344        // Cheap clone (two Arc + a Copy enum) so we can borrow the buffer mutably.
345        let Some(ctx) = self.sealing.clone() else {
346            return Ok(());
347        };
348        for op in self.buffer.operations_mut() {
349            let (is_node, entity_id) = match op {
350                BW::CreateNode { node_id, .. } | BW::UpdateNode { node_id, .. } => {
351                    (true, node_id.as_u64())
352                }
353                BW::CreateEdge { edge_id, .. } | BW::UpdateEdge { edge_id, .. } => {
354                    (false, edge_id.as_u64())
355                }
356                // Deletes/retracts re-persist already-sealed bytes read back from
357                // storage — no re-seal needed.
358                _ => continue,
359            };
360            if let Some(props_ref) = op.properties_mut() {
361                let props = std::mem::take(props_ref);
362                *props_ref = ctx.seal_map(is_node, entity_id, props)?;
363            }
364        }
365        Ok(())
366    }

384    fn lock_adaptive_forward_jump_limit(
385        &self,
386        observed_at: Instant,
387    ) -> Result<(std::sync::MutexGuard<'_, Instant>, i64)> {
388        let previous_observed_at =
389            self.commit_clock_observed_at
390                .lock()
391                .map_err(|_| TransactionError::LockPoisoned {
392                    resource: "commit_clock_observed_at".to_string(),
393                })?;
394        let elapsed = observed_at.duration_since(*previous_observed_at);
395        let elapsed_us = i64::try_from(elapsed.as_micros()).unwrap_or(i64::MAX);
396        Ok((
397            previous_observed_at,
398            MAX_FORWARD_JUMP_US.saturating_add(elapsed_us),
399        ))
400    }

402    /// Commit the transaction.
403    ///
404    /// This validates all buffered writes and applies them atomically
405    /// to the storage. If validation fails or any operation fails,
406    /// the transaction is rolled back.
407    ///
408    /// The durability behavior depends on the transaction's durability mode:
409    /// - **Synchronous**: Waits for fsync (maximum durability, default)
410    /// - **Async**: Returns after flush to OS cache (background thread syncs)
411    /// - **GroupCommit**: Waits for batch fsync (ACID + high throughput)
412    /// - **AsyncBatched**: Returns after flush to OS cache, batched fsync in background (<100µs latency)
413    ///
414    /// ## Examples
415    ///
416    /// ```rust,no_run
417    /// # use aletheiadb::{AletheiaDB, PropertyMapBuilder, properties};
418    /// # use aletheiadb::api::transaction::WriteOps;
419    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
420    /// let db = AletheiaDB::new()?;
421    /// let mut tx = db.write_transaction()?;
422    ///
423    /// tx.create_node("Person", properties! { "name" => "Alice" })?;
424    ///
425    /// // Commit the changes to the database
426    /// tx.commit()?;
427    /// # Ok(())
428    /// # }
429    /// ```
430    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
431    pub fn commit(self) -> Result<()> {
432        self.commit_with_timestamp().map(|_| ())

460    /// - **Async**: Returns after flush to OS cache (background thread syncs)
461    /// - **GroupCommit**: Waits for batch fsync (ACID + high throughput)

468    fn commit_with_timestamp_inner(&mut self) -> Result<Timestamp> {
469        #[cfg(feature = "observability")]
470        let tx_id = self.tx_id.to_string();
471
472        #[cfg(feature = "observability")]
473        let durability_mode = format!("{:?}", self.durability_mode);
474
475        #[cfg(feature = "observability")]
476        let _span =
477            crate::observability::transaction_commit_span(&tx_id, &durability_mode).entered();
478
479        #[cfg(feature = "observability")]
480        let commit_start = std::time::Instant::now();
481
482        #[cfg(feature = "observability")]
483        let operations_count = self.buffer.operations().len();
484
485        // Check transaction state
486        if self.state != TxState::Active {
487            return Err(TransactionError::InvalidState {
488                current: format!("{:?}", self.state),
489                expected: "Active".to_string(),
490            }
491            .into());
492        }
493
494        // Transition to preparing state
495        self.state = TxState::Preparing;
496
497        // Validate all buffered writes
498        self.validate()?;
499
500        // Detect write-write conflicts (Snapshot Isolation)
501        self.detect_conflicts()?;
502
503        // Enforce uniqueness constraints (reservation step).
504        // Must sit before the timestamp lock so we never hold a DashMap shard
505        // guard across `current_timestamp` (preserves lock-ordering contract).
506        // The guard is intentionally kept alive (not prefixed with `_`) so its
507        // Drop impl releases reserved keys on any subsequent error path.
508        let constraint_guard = if let Some(ref registry) = self.constraint_registry {
509            Some(
510                constraint::check_constraints(self, registry)
511                    .map_err(crate::core::error::Error::Constraint)?,
512            )
513        } else {
514            None
515        };
516
517        // GDPR crypto-shred (Issue #3359, PR-1b): seal designated property values
518        // in the buffer AFTER validation + constraint checks (which must see
519        // plaintext) but BEFORE WAL logging + apply, so every tier (WAL segment,
520        // current, historical) records byte-identical sealed ciphertext. Runs
521        // outside the `current_timestamp` critical section, keeping the keyring
522        // lock off the timestamp-held path. Fail-closed. No-op without an active
523        // designation.
524        #[cfg(feature = "audit-export")]
525        self.seal_buffer()?;
526
527        // Issue #3406: pre-generate the closing (delete tombstone / retraction)
528        // version ids BEFORE WAL logging, so the SAME ids are both (a) recorded
529        // in the WAL delete/retract payloads and (b) applied to historical
530        // storage. This makes crash recovery reproduce the live version chain
531        // bit-for-bit instead of synthesizing ids that can diverge from — or
532        // collide with — later-logged version ids. Generated here (id
533        // generators are lock-free atomics, order class 5) before the
534        // `current_timestamp` lock is taken, preserving the lock-order contract.
535        let closing_version_ids = self.pregenerate_closing_version_ids()?;
536
537        // In-flight watermark guard (lost-write persist race fix). Declared in
538        // the OUTER function scope so it lives past the `current_timestamp`
539        // critical section and across `apply_changes` + finalization: the LSN is
540        // registered AFTER the WAL append allocates it but BEFORE the durability
541        // fsync (so a durable write is always registered), and this guard is only
542        // dropped at end-of-function (after apply + finalize on the success path,
543        // or on any early-return / panic path), so a *deregistered* LSN is
544        // guaranteed present in any current+historical snapshot.
545        let mut in_flight_guard: Option<in_flight::InFlightGuard> = None;
546
547        // Issue #3375 (ordered emit, HIGH review fix): the push-changefeed emit ticket.
548        // Reserved UNDER the `current_timestamp` lock below (right after the commit timestamp
549        // is assigned) so its sequence totally orders commits by commit timestamp; consumed on
550        // the success path by `EmitTicket::submit`. Declared in the OUTER scope so that if the
551        // commit aborts/`?`-returns after reservation but before submit (a WAL failure, or the
552        // Issue #3416 commit-time write-skew re-check in `apply_changes`), dropping this
553        // `Option` releases an empty slot for the reserved seq — the sequencer never stalls on
554        // a reserved-but-never-submitted sequence.
555        let mut emit_ticket: Option<crate::core::changefeed_subscription::EmitTicket> = None;
556
557        // Acquire commit timestamp and perform mode-aware WAL flush.
558        //
559        // CRITICAL: We must hold the timestamp lock until WAL logging is complete
560        // to prevent a race condition where transactions commit out-of-order.
561        //
562        // Timestamp management for Bi-Temporal Database:
563        //
564        // For temporal queries to work correctly, we MUST use wallclock timestamps
565        // for transaction_time, not a logical clock. This allows querying historical
566        // state at specific points in time (e.g., "what was the state at 2PM yesterday?").
567        //
568        // Monotonicity: Wallclock time is monotonically increasing (assuming NTP is working),
569        // which satisfies the ordering requirements for Snapshot Isolation:
570        // - commit_ts > snapshot_ts for transactions that started before this commit
571        // - future snapshots will have timestamp >= this commit
572        //
573        // DURABILITY MODES (handled by ConcurrentWalSystem):
574        // - Synchronous: Appends drain and flush immediately with fsync
575        // - Async: Appends go to ring buffers, background thread syncs
576        // - GroupCommit: Appends go to ring buffers, wait for epoch completion
577        let commit_timestamp = {
578            #[cfg(feature = "observability")]
579            let ts_lock_start = std::time::Instant::now();
580
581            let mut ts =
582                self.current_timestamp
583                    .lock()
584                    .map_err(|_| TransactionError::LockPoisoned {
585                        resource: "current_timestamp".to_string(),
586                    })?;
587
588            #[cfg(feature = "observability")]
589            let ts_lock_acquired = std::time::Instant::now();
590
591            // Phase 2: Use HLC for distributed temporal consistency
592            // Get current physical wallclock
593            let current_wallclock = crate::core::temporal::time::now();
594            let observed_at = Instant::now();
595            let (mut previous_observed_at, adaptive_forward_limit_us) =
596                self.lock_adaptive_forward_jump_limit(observed_at)?;
597
598            let self_heal_clock_skew = is_clock_skew_self_heal_enabled();
599            let skew_decision = evaluate_clock_skew(
600                current_wallclock.wallclock(),
601                ts.wallclock(),
602                Some(adaptive_forward_limit_us),
603                self_heal_clock_skew,
604            )
605            .map_err(|violation| TransactionError::ClockSkew {
606                wallclock: current_wallclock.wallclock(),
607                previous: ts.wallclock(),
608                drift_us: violation.drift_us,
609                max_allowed: violation.max_allowed,
610            })?;
611
612            if self_heal_clock_skew && let Some(_direction) = skew_decision.healed_direction {
613                #[cfg(feature = "observability")]
614                tracing::warn!(
615                    wallclock_ts = %current_wallclock,
616                    prev_ts = %ts,
617                    drift_us = skew_decision.drift_us,
618                    reason = _direction.as_str(),
619                    "Self-healing clock skew by clamping to local HLC frontier"
620                );
621            }
622
623            // Phase 2: Use HLC .send() method for monotonic timestamp generation
624            // This ensures: if wallclock advances, reset logical; otherwise increment logical
625            let commit = send_with_overflow_self_heal(
626                &ts,
627                skew_decision.effective_wallclock,
628                self_heal_clock_skew,
629                |error| match error {
630                    SendWithSelfHealError::InitialSend(error) => TransactionError::CommitFailed {
631                        reason: format!("HLC timestamp generation failed: {}", error),
632                    },
633                    SendWithSelfHealError::FallbackWallclockOverflow {
634                        wallclock,
635                        current_logical,
636                    } => TransactionError::CommitFailed {
637                        reason: format!(
638                            "HLC logical counter overflow at wallclock={}: {}",
639                            wallclock, current_logical
640                        ),
641                    },
642                    SendWithSelfHealError::FallbackSend(fallback_error) => {
643                        TransactionError::CommitFailed {
644                            reason: format!(
645                                "HLC timestamp generation failed while self-healing: {}",
646                                fallback_error
647                            ),
648                        }
649                    }
650                },
651            )?;
652
653            // Observability: Warn about clock skew issues
654            #[cfg(feature = "observability")]
655            {
656                // Clock went backwards: wallclock < previous wallclock
657                if current_wallclock.wallclock() < ts.wallclock() {
658                    tracing::warn!(
659                        wallclock_ts = %current_wallclock,
660                        prev_ts = %ts,
661                        skew_us = ts.wallclock() - current_wallclock.wallclock(),
662                        logical_counter = commit.logical(),
663                        "Clock skew detected: wallclock went backwards (NTP adjustment?)"
664                    );
665                } else if commit.wallclock() > ts.wallclock() + 60_000_000 {
666                    // Large forward jump (>60 seconds)
667                    tracing::warn!(
668                        wallclock_ts = %current_wallclock,
669                        prev_ts = %ts,
670                        jump_us = commit.wallclock() - ts.wallclock(),
671                        "Large clock jump detected: timestamps will be lumpy"
672                    );
673                }
674            }
675
676            // Update current_timestamp for next transaction's snapshot
677            *ts = commit;
678            // Persist observation only after we successfully advanced the frontier.
679            *previous_observed_at = observed_at;
680            drop(previous_observed_at);
681
682            // Issue #3375 (ordered emit): reserve the push-changefeed emit ticket HERE, while
683            // the `current_timestamp` lock is still held and monotonic ordering is guaranteed,
684            // so the ticket's sequence == this commit's timestamp order == ChangeCursor order.
685            // Gated on `has_subscribers()` so the zero-subscriber fast path stays a single
686            // atomic load with no sequencer work (no reservation, no ticket). The reservation
687            // itself is a lock-free `fetch_add`, so taking it under `current_timestamp` does
688            // not violate the write-path lock order.
689            if let Some(broadcaster) = &self.changefeed
690                && broadcaster.has_subscribers()
691            {
692                emit_ticket = Some(broadcaster.reserve_emit_ticket());
693            }
694
695            #[cfg(feature = "observability")]
696            let wal_start = std::time::Instant::now();
697
698            // Log operations to WAL (lock-free striped append!)
699            // This must happen BEFORE applying changes for durability.
700            // Returns the base (lowest) LSN allocated for this commit, if any.
701            let base_lsn = wal::log_operations_to_wal(self, commit, &closing_version_ids)?;
702
703            // Register the commit's base LSN as in-flight BEFORE the durability
704            // fsync below, so a write that becomes durable is always registered.
705            // The guard survives to end-of-function (see its declaration above).
706            if let (Some(tracker), Some(base_lsn)) = (self.in_flight.as_ref(), base_lsn) {
707                in_flight_guard = Some(tracker.register(base_lsn.0));
708            }
709
710            #[cfg(feature = "observability")]
711            let wal_logged = std::time::Instant::now();
712
713            // Commit with configured durability mode
714            // For Sync: drains and flushes immediately
715            // For Async: returns immediately
716            // For GroupCommit: registers and returns epoch
717            let wait_epoch = self.wal.commit()?;
718
719            #[cfg(feature = "observability")]
720            let wal_commit_completed = std::time::Instant::now();
721
722            // For GroupCommit mode, wait for the epoch to be flushed.
723            // AsyncBatched mode returns an epoch but does NOT wait.
724            if let Some(epoch) = wait_epoch
725                && let Some(gc) = self.wal.group_commit_coordinator()
726                && self.durability_mode.waits_for_durability()
727            {
728                gc.wait_for_flush(epoch)?;
729            }
730
731            #[cfg(feature = "observability")]
732            {
733                // Record detailed breakdown for tracing and metrics.
734                let ts_lock_wait_us =
735                    ts_lock_acquired.duration_since(ts_lock_start).as_micros() as u64;
736                let wal_log_us = wal_logged.duration_since(wal_start).as_micros() as u64;
737                let wal_commit_us =
738                    wal_commit_completed.duration_since(wal_logged).as_micros() as u64;
739                let total_us = wal_commit_completed
740                    .duration_since(ts_lock_start)
741                    .as_micros() as u64;
742
743                let total_commit_us = commit_start.elapsed().as_micros() as u64;
744                tracing::info!(
745                    ts_lock_wait_us,
746                    wal_log_us,
747                    wal_commit_us,
748                    total_us,
749                    total_commit_us,
750                    operations_count,
751                    commit_ts = %commit,
752                    durability_mode = ?self.durability_mode,
753                    "Transaction commit breakdown (concurrent WAL)"
754                );
755            }
756
757            commit
758        };
759
760        // Apply all changes atomically.
761        // Nodes/edges are written with commit_timestamp: None during this phase.
762        //
763        // Issue #3425: `apply_changes` returns the `historical.write()` guard
764        // instead of dropping it internally. We hold it across finalization below
765        // so the current-side writes and their `commit_timestamp` finalization are
766        // atomic w.r.t. any snapshot (checkpoint / backup) holding `historical.read()`
767        // — closing the *in-memory* window in which a committed node/edge could be
768        // captured with `commit_timestamp: None`. (The persisted index format does
769        // not serialize `commit_timestamp` and reconstructs it as committed on
770        // restore, so this fix hardens the in-memory snapshot invariant rather than
771        // fixing any on-disk corruption.) This respects the documented lock order
772        // (`historical` precedes `snapshot_lock` and adjacency), so no lock-order
773        // inversion is introduced.
774        //
775        // The pre-generated closing version ids (Issue #3406) are consumed here
776        // in the same buffer order they were logged to the WAL.
777        //
778        // Test-only interleaving hook (Issue #3416 Pt1): fires AFTER
779        // validate/detect_conflicts/WAL but BEFORE `apply_changes` acquires the
780        // `historical.write()` guard, so a test can commit a concurrent
781        // delete/create in this window and exercise the commit-time write-skew
782        // re-checks. Production builds compile this away.
783        #[cfg(test)]
784        commit_test_hooks::run_pre_apply_hook();
785
786        // Always-compiled one-shot interleaving seam for the lost-write persist
787        // race regression tests (which live in `tests/` and therefore cannot see
788        // the `#[cfg(test)]` hooks above). This fires at the SAME point — durable
789        // + acknowledged (WAL fsynced), in-flight LSN registered, but NOT yet
790        // applied — so an integration test can park exactly one commit here while
791        // it drives a racing `persist_indexes()`. In production the seam is a
792        // single relaxed atomic load (unarmed), so it is effectively zero-cost.
793        race_seam::run_pre_apply_once();
794
795        let historical_guard = apply::apply_changes(self, commit_timestamp, &closing_version_ids)?;
796
797        // Test-only interleaving hook (Issue #3425): fires at the exact point
798        // between the historical writes and the commit-timestamp finalization,
799        // while `historical_guard` is held. Production builds compile this away.
800        #[cfg(test)]
801        commit_test_hooks::run_pre_finalize_hook();
802
803        // Finalize commit timestamps in current storage.
804        // Only reached on the success path — sets commit_timestamp: Some(T) on all
805        // written nodes/edges, making them visible to future snapshot readers.
806        apply::finalize_current_commit_timestamps(self, commit_timestamp);
807
808        // Apply + finalization are complete: this write is now present in both
809        // current and historical storage. Deregister the in-flight LSN so index
810        // persistence no longer needs to hold the manifest watermark below it.
811        // Deregistering strictly AFTER finalize is the invariant that lets the
812        // watermark guarantee "every deregistered LSN is in the snapshot".
813        drop(in_flight_guard.take());
814
815        // Finalization complete: release the historical write guard. Snapshots
816        // blocked on `historical.read()` now proceed and observe resolved timestamps.
817        //
818        // Issue #3425: only `finalize_current_commit_timestamps` must run under the
819        // guard to close the snapshot finalize window. The temporal-vector notify
820        // below is deliberately kept OUTSIDE the guard: it touches only the vector
821        // index's own locks and is independent of current-side commit_timestamp
822        // finalization, so serializing its (potentially O(N log N)) HNSW snapshot
823        // build under the global historical write lock would only add latency
824        // without any correctness benefit.
825        drop(historical_guard);
826
827        // Notify temporal vector index of transaction completion (for snapshot creation).
828        // Only call this if the transaction modified vector properties to avoid unnecessary overhead.
829        // No ordering dependency on finalize: the vectors were already added to the
830        // index during `apply_changes`; this only triggers snapshot creation.
831        if self.buffer.has_vector_operations() {
832            self.current.on_temporal_vector_transaction()?;
833        }
834
835        // Commit the constraint guard before registering with the visibility manager.
836        // Committing first ensures that freed old-value slots are released before
837        // other transactions observe this commit as visible — otherwise a concurrent
838        // delete+create of the same key could see the old reservation still present
839        // and raise a spurious UniqueViolation.
840        if let Some(guard) = constraint_guard {
841            guard.commit();
842        }
843
844        // Register commit with visibility manager
845        self.visibility_manager.register_commit(self.tx_id);
846
847        // Mark as committed
848        self.state = TxState::Committed;
849
850        // Issue #3375: broadcast this transaction's committed change set to push-changefeed
851        // subscribers. This runs on the committed path only (never for an aborted/rolled-back
852        // transaction, which returns early before reaching here), and strictly AFTER the write
853        // is durable (WAL fsynced), applied (current + historical), and visible
854        // (`register_commit`). No write-path lock is held: the records are built via a fresh,
855        // short `historical.read()` (a leaf acquisition — nothing later in the lock order is
856        // held) whose guard is dropped before submitting.
857        //
858        // HIGH review fix (ordered emit): we consume the ticket reserved under the
859        // `current_timestamp` lock. `submit` releases records to subscriber buffers in strict
860        // reserved-sequence (== commit-timestamp == ChangeCursor) order, so per-subscriber
861        // delivery is cursor-ascending even when concurrent commits finish their record-build
862        // out of order — the precondition that makes `last_delivered` a zero-loss resume
863        // high-water-mark. A slow subscriber cannot back-pressure this writer. We submit even
864        // an empty record set so the reserved sequence is released and never stalls the
865        // sequencer.
866        if let Some(ticket) = emit_ticket.take() {
867            let (node_vids, edge_vids) =
868                self.committed_changefeed_version_ids(&closing_version_ids);
869            let mut records: Vec<crate::core::changefeed::ChangeRecord> =
870                if node_vids.is_empty() && edge_vids.is_empty() {
871                    Vec::new()
872                } else {
873                    let hist = self.historical.read();
874                    hist.collect_committed_changes(&node_vids, &edge_vids)
875                        .into_iter()
876                        .map(|raw| raw.into_record())
877                        .collect()
878                };
879            // Deterministic #3216 total order (tx-time, kind, entity, version).
880            records.sort_by_key(|r| r.cursor());
881            ticket.submit(records);
882        }
883
884        #[cfg(feature = "observability")]
885        {
886            let total_commit_us = commit_start.elapsed().as_micros() as u64;
887            crate::observability::record_transaction_commit(
888                commit_start.elapsed().as_secs_f64(),
889                operations_count as u64,
890                &durability_mode,
891                "committed",
892            );
893            tracing::debug!(
894                total_commit_us,
895                tx_id = %self.tx_id,
896                "Transaction committed"
897            );
898        }
899
900        Ok(commit_timestamp)
901    }

903    /// Pre-generate the closing version ids for every buffered delete/retract
904    /// operation, in buffer order (Issue #3406).
905    ///
906    /// Delete tombstones and valid-time retractions each append exactly one
907    /// closing version; both draw from the shared `version_id_gen`. Generating
908    /// them here — before the WAL log phase — lets the exact same id be written
909    /// into the WAL payload AND applied to historical storage, so crash
910    /// recovery reproduces the live version chain instead of synthesizing a
911    /// (potentially colliding) id.
912    ///
913    /// The returned vector has one id per delete/retract op in the buffer,
914    /// ordered to match the iteration order used by both
915    /// [`wal::log_operations_to_wal`] and [`apply::apply_changes`].
916    fn pregenerate_closing_version_ids(&self) -> Result<Vec<VersionId>> {
917        let num_closing = self
918            .buffer
919            .operations()
920            .iter()
921            .filter(|op| {
922                matches!(
923                    op,
924                    super::BufferedWrite::DeleteNode { .. }
925                        | super::BufferedWrite::DeleteEdge { .. }
926                        | super::BufferedWrite::RetractNode { .. }
927                        | super::BufferedWrite::RetractEdge { .. }
928                )
929            })
930            .count();
931
932        let mut ids = Vec::with_capacity(num_closing);
933        for _ in 0..num_closing {
934            ids.push(VersionId::new_unchecked(self.version_id_gen.next()?));
935        }
936        Ok(ids)
937    }

939    /// Collect the node and edge version ids this transaction just committed, split by
940    /// kind, for the push-changefeed broadcast (Issue #3375).
941    ///
942    /// Create/update ops carry their own `version_id`; delete/retract ops draw their
943    /// closing (tombstone) `version_id` from `closing_version_ids` in buffer order — the
944    /// exact order [`pregenerate_closing_version_ids`](Self::pregenerate_closing_version_ids)
945    /// generated them and [`apply::apply_changes`] consumed them. Iterating the buffer in
946    /// order and advancing a cursor into `closing_version_ids` for each delete/retract keeps
947    /// the pairing correct.
948    fn committed_changefeed_version_ids(
949        &self,
950        closing_version_ids: &[VersionId],
951    ) -> (Vec<VersionId>, Vec<VersionId>) {
952        use super::BufferedWrite as BW;
953        let mut node_vids = Vec::new();
954        let mut edge_vids = Vec::new();
955        let mut closing_cursor = 0usize;
956        let mut next_closing = || {
957            let id = closing_version_ids.get(closing_cursor).copied();
958            closing_cursor += 1;
959            id
960        };
961        for op in self.buffer.operations() {
962            match op {
963                BW::CreateNode { version_id, .. } | BW::UpdateNode { version_id, .. } => {
964                    node_vids.push(*version_id);
965                }
966                BW::CreateEdge { version_id, .. } | BW::UpdateEdge { version_id, .. } => {
967                    edge_vids.push(*version_id);
968                }
969                BW::DeleteNode { .. } | BW::RetractNode { .. } => {
970                    if let Some(id) = next_closing() {
971                        node_vids.push(id);
972                    }
973                }
974                BW::DeleteEdge { .. } | BW::RetractEdge { .. } => {
975                    if let Some(id) = next_closing() {
976                        edge_vids.push(id);
977                    }
978                }
979            }
980        }
981        (node_vids, edge_vids)
982    }

984    /// Rollback the transaction.
985    ///
986    /// Discards all buffered writes. This is automatically called
987    /// if the transaction is dropped without committing.
988    ///
989    /// ## Examples
990    ///
991    /// ```rust,no_run
992    /// # use aletheiadb::{AletheiaDB, PropertyMapBuilder, properties};
993    /// # use aletheiadb::api::transaction::WriteOps;
994    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
995    /// let db = AletheiaDB::new()?;
996    /// let mut tx = db.write_transaction()?;
997    ///
998    /// tx.create_node("Person", properties! { "name" => "Alice" })?;
999    ///
1000    /// // Discard the changes
1001    /// tx.rollback()?;
1002    /// # Ok(())
1003    /// # }
1004    /// ```
1005    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
1006    pub fn rollback(mut self) -> Result<()> {
1007        if self.state == TxState::Committed {
1008            return Err(TransactionError::AlreadyCommitted {
1009                tx_id: self.tx_id.as_u64(),
1010            }
1011            .into());
1012        }
1013
1014        // Clear the write buffer
1015        self.buffer.clear();
1016
1017        // Register abort with visibility manager
1018        self.visibility_manager.register_abort(self.tx_id);
1019
1020        self.state = TxState::Aborted;
1021
1022        Ok(())
1023    }

1091                    label,
1092                    properties,
1093                    version_id,
1094                    ..
1095                }
1096                | super::BufferedWrite::UpdateNode {
1097                    node_id,
1098                    label,

1102                } => Some(Ok(Node::with_metadata(

1104                    *label,

1106                    *version_id,

1109                        commit_timestamp: None, // Not yet committed
1110                    },
1111                ))),
1112                super::BufferedWrite::DeleteNode { .. }
1113                | super::BufferedWrite::RetractNode { .. } => {

1115                }
1116                _ => None, // Not a node operation
1117            })

1193    /// [`remove_node_property`](WriteOps::remove_node_property) can reuse the
1194    /// node's existing label without a (deprecated) interner `resolve`.
1195    ///
1196    /// Does not record the error metric — the public `WriteOps` entry points
1197    /// wrap the call and do so.
1198    fn buffer_node_replace(
1199        &mut self,
1200        node_id: NodeId,
1201        label_interned: crate::core::interning::InternedString,
1202        properties: PropertyMap,
1203        options: WriteRequestOptions,
1204    ) -> Result<()> {
1205        // Check transaction state
1206        if self.state != TxState::Active {
1207            return Err(TransactionError::InvalidState {
1208                current: format!("{:?}", self.state),
1209                expected: "Active".to_string(),
1210            }
1211            .into());
1212        }
1213
1214        // Verify the node exists (read-your-own-writes, Issue #3417) — a
1215        // replace targets an existing entity, never conjures one. Unlike PATCH
1216        // `update_node`, the existing map is NOT seeded into the new version:
1217        // `properties` becomes the node's map *exactly*, so any prior key it
1218        // omits is removed from current state (removal is encoded natively by
1219        // the anchor/delta diff and survives WAL replay). The label is
1220        // overwritten too.
1221        let existing = self.read_own_node(node_id)?;
1222        let version_id = VersionId::new_unchecked(self.version_id_gen.next()?);
1223
1224        // Re-stamp the immutable namespace from the existing node (Issue #3349)
1225        // so a full overwrite (replace / remove-property) can never drop it.
1226        let properties = namespace::restamp_namespace(properties, &existing.properties);
1227
1228        // A replace may add, change, OR remove vector properties; mark the
1229        // buffer so the temporal vector index is notified on commit if either
1230        // the old or the new state carries a vector.
1231        if !self.buffer.has_vector_operations()
1232            && (existing.properties.contains_vector() || properties.contains_vector())
1233        {
1234            self.buffer.mark_has_vector_operations();
1235        }
1236
1237        // Get timestamp: use provided valid_from or default to transaction start time
1238        let timestamp = self.start_timestamp;
1239        let valid_from = options.valid_from.unwrap_or(timestamp);
1240
1241        // Validate valid_from is not too far in future
1242        validation::validate_valid_from_future(valid_from)?;
1243
1244        // Validate valid_from is not before entity creation.
1245        let creation_time = {
1246            let historical = self.historical.read();
1247            historical.node_creation_time(node_id)
1248        };
1249        if let Some(creation_time) = creation_time {
1250            validation::validate_valid_from_not_before_creation(
1251                &format!("node:{}", node_id.as_u64()),
1252                creation_time,
1253                valid_from,
1254            )?;
1255        }
1256
1257        // Normalize an all-absent provenance bundle to `None` (Issue #3224).
1258        let provenance = options
1259            .provenance
1260            .filter(|p| !p.is_empty())
1261            .map(std::sync::Arc::new);
1262
1263        // Buffer the FULL overwrite. Reuses the UpdateNode op (no new WAL
1264        // variant / format bump, Issue #3549): the exact target map plus a
1265        // possibly-new label is the payload.
1266        self.buffer.add(super::BufferedWrite::UpdateNode {
1267            node_id,
1268            version_id,
1269            label: label_interned,
1270            properties,
1271            valid_from,
1272            provenance,
1273        })?;
1274
1275        Ok(())
1276    }
1277
1278    /// Shared body for edge replace (Issue #3549): buffer a full-overwrite
1279    /// `UpdateEdge` carrying the *exact* `properties` map (no PATCH merge),
1280    /// preserving the already-read `edge`'s immutable source/target/type.
1281    /// Callers pass an already-read `edge` so
1282    /// [`remove_edge_property`](WriteOps::remove_edge_property) need not read it
1283    /// a second time (mirrors [`buffer_node_replace`](Self::buffer_node_replace)).
1284    ///
1285    /// Does not record the error metric — the public `WriteOps` entry points
1286    /// wrap the call and do so.
1287    fn buffer_edge_replace(
1288        &mut self,
1289        edge: Edge,
1290        properties: PropertyMap,
1291        options: WriteRequestOptions,
1292    ) -> Result<()> {
1293        // Check transaction state
1294        if self.state != TxState::Active {
1295            return Err(TransactionError::InvalidState {
1296                current: format!("{:?}", self.state),
1297                expected: "Active".to_string(),
1298            }
1299            .into());
1300        }
1301
1302        let edge_id = edge.id;
1303        let version_id = VersionId::new_unchecked(self.version_id_gen.next()?);
1304
1305        // Re-stamp the immutable namespace from the existing edge (Issue #3349)
1306        // so a full overwrite (replace / remove-property) can never drop it.
1307        let properties = namespace::restamp_namespace(properties, &edge.properties);
1308
1309        // A replace may add/change/remove vector properties.
1310        if !self.buffer.has_vector_operations()
1311            && (edge.properties.contains_vector() || properties.contains_vector())
1312        {
1313            self.buffer.mark_has_vector_operations();
1314        }
1315
1316        // Get timestamp: use provided valid_from or default to transaction start time
1317        let timestamp = self.start_timestamp;
1318        let valid_from = options.valid_from.unwrap_or(timestamp);
1319
1320        // Validate valid_from is not too far in future
1321        validation::validate_valid_from_future(valid_from)?;
1322
1323        // Validate valid_from is not before the edge's own creation time
1324        let creation_time = {
1325            let historical = self.historical.read();
1326            historical.edge_creation_time(edge_id)
1327        };
1328        if let Some(creation_time) = creation_time {
1329            validation::validate_valid_from_not_before_creation(
1330                &format!("edge:{}", edge_id.as_u64()),
1331                creation_time,
1332                valid_from,
1333            )?;
1334        }
1335
1336        // Normalize an all-absent provenance bundle to `None` (Issue #3224).
1337        let provenance = options
1338            .provenance
1339            .filter(|p| !p.is_empty())
1340            .map(std::sync::Arc::new);
1341
1342        // Buffer the FULL overwrite via the existing UpdateEdge op (no WAL
1343        // format change, Issue #3549), preserving endpoints and type.
1344        self.buffer.add(super::BufferedWrite::UpdateEdge {
1345            edge_id,
1346            version_id,
1347            source: edge.source,
1348            target: edge.target,
1349            label: edge.label,
1350            properties,
1351            valid_from,
1352            provenance,
1353        })?;
1354
1355        Ok(())
1356    }

1358    /// Enumerate edges CREATED earlier in THIS transaction that reference
1359    /// `node_id` as source or target and are still live in the buffer (Issue
1360    /// #3416 Pt4).
1361    ///
1362    /// `delete_node_cascade` unions this with committed adjacency
1363    /// (`self.current.get_outgoing_edges` / `get_incoming_edges`) so a same-tx
1364    /// `create_node → create_edge → cascade` reaches the same-tx-created edge,
1365    /// which committed adjacency cannot yet see. Only `CreateEdge` ops matter:
1366    /// an already-committed edge (including one this tx `UpdateEdge`s) is
1367    /// already returned by the current-adjacency read. An edge created then
1368    /// deleted/retracted within this tx is filtered out (its latest buffered
1369    /// state resolves to Not-Found via [`buffered_edge`](Self::buffered_edge)),
1370    /// so the cascade never double-closes it.
1371    fn buffered_connected_edges(&self, node_id: NodeId) -> Vec<EdgeId> {
1372        let mut edges = Vec::new();
1373        for op in self.buffer.operations() {
1374            if let super::BufferedWrite::CreateEdge {
1375                edge_id,
1376                source,
1377                target,
1378                ..
1379            } = op
1380                && (*source == node_id || *target == node_id)
1381                && matches!(self.buffered_edge(*edge_id), Some(Ok(_)))
1382            {
1383                edges.push(*edge_id);
1384            }
1385        }
1386        edges
1387    }
1388}

1474    fn find_nodes_by_property(
1475        &self,
1476        label: &str,
1477        property_key: &str,
1478        property_value: &crate::core::property::PropertyValue,
1479    ) -> Vec<NodeId> {
1480        use crate::core::interning::GLOBAL_INTERNER;
1481
1482        // 1. Get committed matches, excluding nodes modified in the buffer
1483        // ⚡ Bolt Optimization: Uses `.retain()` to filter in-place and avoid allocating a new `Vec`.
1484        let mut results = self
1485            .current
1486            .find_nodes_by_property(label, property_key, property_value);
1487
1488        results.retain(|node_id| {
1489            !self.buffer.has_modified_node(*node_id) && {
1490                self.current
1491                    .get_node(*node_id)
1492                    .map(|node| {
1493                        // Use embedded commit_timestamp for visibility check (Issue #238).
1494                        self.visibility_manager.is_visible_with_embedded_ts(
1495                            &self.snapshot,
1496                            node.metadata.created_by_tx,
1497                            node.metadata.commit_timestamp,
1498                        )
1499                    })
1500                    .unwrap_or(false)
1501            }
1502        });
1503
1504        // 2. Scan buffered writes for matching CreateNode/UpdateNode
1505        let label_id = GLOBAL_INTERNER.get_id(label);
1506        let key_id = GLOBAL_INTERNER.get_id(property_key);
1507
1508        if let (Some(label_id), Some(key_id)) = (label_id, key_id) {
1509            for op in self.buffer.operations() {
1510                match op {
1511                    super::BufferedWrite::CreateNode {
1512                        node_id,
1513                        label: node_label,
1514                        properties,
1515                        ..
1516                    }
1517                    | super::BufferedWrite::UpdateNode {
1518                        node_id,
1519                        label: node_label,
1520                        properties,
1521                        ..
1522                    } => {
1523                        if *node_label == label_id
1524                            && let Some(val) = properties.get_by_interned_key(&key_id)
1525                            && val == property_value
1526                        {
1527                            results.push(*node_id);
1528                        }
1529                    }
1530                    // DeleteNode is already excluded by has_modified_node filter
1531                    _ => {}
1532                }
1533            }
1534        }
1535
1536        results
1537    }

1541///
1542/// The `apply_batch` MCP surface reports a per-operation `version_id` for
1543/// creates and updates. Those version ids are allocated eagerly at op call
1544/// time and stored in the corresponding [`BufferedWrite`](super::BufferedWrite)
1545/// variant; these accessors surface them without touching storage or any
1546/// synchronization primitive — they are plain reads of the transaction's own
1547/// write buffer (no lock-order implications).

1550    ///
1551    /// Returns `None` when the node has no buffered write in this transaction,
1552    /// or when its latest buffered write is a delete/retract (tombstone
1553    /// versions are allocated at apply time and are not surfaced here).
1554    pub fn buffered_node_version(&self, node_id: NodeId) -> Option<VersionId> {
1555        match self.buffer.get_node_write(node_id) {

1558                | super::BufferedWrite::UpdateNode { version_id, .. },
1559            ) => Some(*version_id),
1560            _ => None,
1561        }
1562    }

1567    /// or when its latest buffered write is a delete/retract.
1568    pub fn buffered_edge_version(&self, edge_id: EdgeId) -> Option<VersionId> {
1569        match self.buffer.get_edge_write(edge_id) {
1570            Some(
1571                super::BufferedWrite::CreateEdge { version_id, .. }
1572                | super::BufferedWrite::UpdateEdge { version_id, .. },

1574            _ => None,
1575        }
1576    }

1750    fn update_node_with_options(
1751        &mut self,
1752        node_id: NodeId,
1753        properties: PropertyMap,
1754        options: WriteRequestOptions,
1755    ) -> Result<()> {
1756        let result = (|| {
1757            // Check transaction state
1758            if self.state != TxState::Active {
1759                return Err(TransactionError::InvalidState {
1760                    current: format!("{:?}", self.state),
1761                    expected: "Active".to_string(),
1762                }
1763                .into());
1764            }
1765
1766            // A namespace is immutable after creation: reject an explicit
1767            // namespace on this update rather than silently ignoring it (#3349).
1768            namespace::reject_namespace_on_update(options.namespace.as_ref())?;
1769
1770            // Reject engine-reserved keys on the incoming PATCH (Issue #3349):
1771            // a user may not set/overwrite a namespace/shred ride-along key.
1772            namespace::reject_reserved_keys(&properties)?;
1773
1774            // Get current node to preserve label and existing properties.
1775            // Buffer-aware read (Issue #3417): read-your-own-writes so an
1776            // update of a node created/updated earlier in THIS transaction
1777            // merges onto the buffered state, not committed-only storage
1778            // (which would 404 a same-tx-created node).
1779            let node = self.read_own_node(node_id)?;
1780            let version_id = VersionId::new_unchecked(self.version_id_gen.next()?);
1781
1782            // PATCH semantics: Merge new properties with existing ones
1783            // Start with existing properties
1784            let mut builder = PropertyMapBuilder::from_map(node.properties.clone());
1785
1786            // Update/add properties from the incoming map
1787            for (key, value) in properties.iter() {
1788                builder = builder.insert_by_key(*key, value.clone());
1789            }
1790
1791            // Build the final merged property map, then re-stamp the immutable
1792            // namespace from the existing node (Issue #3349). A PATCH inherently
1793            // preserves the ride-along key; re-stamping makes that a hard
1794            // guarantee that no update path can ever drop or change it.
1795            let merged_properties = namespace::restamp_namespace(builder.build(), &node.properties);
1796
1797            // A PATCH may drop or downgrade a vector (e.g. overwrite the
1798            // embedding key with a scalar, or remove it): mark the buffer so
1799            // the temporal vector index snapshots on commit if EITHER the
1800            // existing node or the merged map carries a vector. `WriteBuffer::add`
1801            // only inspects the merged map, so without this a vector-removing
1802            // PATCH leaves `has_vector_operations` false and no snapshot fires,
1803            // leaking the phantom as-of-now (Issue #3621). Mirrors
1804            // `buffer_node_replace`.
1805            if !self.buffer.has_vector_operations()
1806                && (node.properties.contains_vector() || merged_properties.contains_vector())
1807            {
1808                self.buffer.mark_has_vector_operations();

2180                expected: "Active".to_string(),

2184
2185        // Verify node exists before attempting deletion.
2186        // Buffer-aware read (Issue #3417): a node created earlier in THIS
2187        // transaction can be cascade-deleted (read-your-own-writes).
2188        let _node = self.read_own_node(node_id)?;

2197        // already deleted/retracted in this tx. Without the buffer union a
2198        // same-tx create_node → create_edge → cascade would leave the

2200        // edge does not).
2201        let mut edge_ids = self.get_outgoing_edges(node_id)?;
2202        edge_ids.extend(self.get_incoming_edges(node_id)?);
2203        edge_ids.extend(self.buffered_connected_edges(node_id));
2204
2205        // Dedup by EdgeId (Issue #3416 Pt3): a self-loop (node -> node) appears
2206        // in BOTH the outgoing and incoming lists; without dedup delete_edge
2207        // would be called twice for the same edge, buffering two DeleteEdge ops
2208        // (double tombstone / double-close at apply time). Sort + dedup means
2209        // each distinct connected edge is deleted exactly once. This mirrors
2210        // retract_node_detach and apply_batch's manual cascade.
2211        edge_ids.sort_unstable();
2212        edge_ids.dedup();
2213
2214        // Delete all connected edges first to maintain referential integrity.
2215        // This prevents orphaned edges that reference a deleted node.
2216        // Performance: O(degree) where degree is the number of connected edges.
2217        //
2218        // Issue #3427: stamp the SAME options (backdated valid_from and/or the

2224        }
2225

2227        // This is safe now because all edges referencing this node have been removed
2228        self.delete_node_with_options(node_id, options)?;

2231    }

2300/// Valid-time retraction (Issue #3230).
2301///
2302/// Retraction closes an entity's valid-time interval at `valid_to` without
2303/// deleting its history. It is a first-class bi-temporal operation, distinct
2304/// from deletion:
2305///
2306/// - The pre-retraction head version's **transaction time** is closed at the
2307///   retraction's commit time while its valid interval stays open-ended, so
2308///   `AS OF SYSTEM_TIME` queries positioned before the retraction still show
2309///   the fact as currently valid (append-only — the past record is never
2310///   rewritten).
2311/// - A **new version** is appended with the same properties and the closed
2312///   valid interval `[valid_from, valid_to)`, so `AS OF VALID_TIME` queries
2313///   strictly before `valid_to` keep returning the entity, while queries at
2314///   or after `valid_to` do not.
2315/// - The entity is removed from **current storage**: like a delete, a
2316///   retracted entity is absent from current-state queries — but its full
2317///   history remains queryable.
2318impl WriteTransaction {
2319    /// Buffer a valid-time retraction of a node.
2320    ///
2321    /// Idempotency: retracting an already-retracted (or deleted) node is a
2322    /// no-op that returns the *existing* `valid_to` (with
2323    /// `already_retracted: true`) — no version is appended and no WAL entry
2324    /// is written, regardless of the `valid_to` passed here. This holds for
2325    /// writes already buffered in THIS transaction too: a second
2326    /// `retract_node` of the same node returns the first call's buffered
2327    /// `valid_to`, and retracting a node this transaction has already
2328    /// `delete_node`-ed mirrors the committed-delete case (a degenerate
2329    /// `[d, d)` interval where `d` is the buffered deletion's valid time)
2330    /// rather than double-buffering a write.
2331    ///
2332    /// # Errors
2333    ///
2334    /// - [`TemporalError::ValidTimeBeforeEntityCreation`](crate::core::error::TemporalError::ValidTimeBeforeEntityCreation)
2335    ///   if `valid_to` precedes the node's current `valid_from`
2336    ///   (`valid_to == valid_from` is allowed and yields an empty interval).
2337    /// - [`TemporalError::ValidTimeTooFarInFuture`](crate::core::error::TemporalError::ValidTimeTooFarInFuture)
2338    ///   if `valid_to` is more than one year in the future.
2339    /// - [`StorageError::NodeNotFound`](crate::core::error::StorageError::NodeNotFound)
2340    ///   if the node never existed.
2341    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
2342    pub fn retract_node(
2343        &mut self,
2344        node_id: NodeId,
2345        valid_to: Timestamp,
2346    ) -> Result<super::RetractionResult> {
2347        self.retract_node_with_provenance(node_id, valid_to, None)
2348    }
2349
2350    /// Buffer a valid-time retraction of a node, stamping a write-time
2351    /// [`Provenance`](crate::core::provenance::Provenance) bundle recording the
2352    /// acting principal onto the retraction version (Issue #3427).
2353    ///
2354    /// Behaves identically to [`retract_node`](Self::retract_node) other than
2355    /// persisting `provenance` on the appended retraction version. Passing
2356    /// `None` is exactly [`retract_node`](Self::retract_node). An all-absent
2357    /// bundle is normalized to no provenance (never a fabricated empty object).
2358    /// On the idempotent no-op path (already retracted/deleted) no version is
2359    /// appended, so the provenance is not recorded — matching the "no new
2360    /// version, no WAL entry" contract.
2361    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
2362    pub fn retract_node_with_provenance(
2363        &mut self,
2364        node_id: NodeId,
2365        valid_to: Timestamp,
2366        provenance: Option<Provenance>,
2367    ) -> Result<super::RetractionResult> {
2368        let result = (|| {
2369            // Check transaction state
2370            if self.state != TxState::Active {
2371                return Err(TransactionError::InvalidState {
2372                    current: format!("{:?}", self.state),
2373                    expected: "Active".to_string(),
2374                }
2375                .into());
2376            }
2377
2378            // A retraction or deletion already buffered in THIS transaction
2379            // is a no-op (mirrors the committed already-retracted/deleted
2380            // cases below): without this guard a double retract — or a
2381            // delete followed by a retract — would buffer two writes for
2382            // the same node, WAL-appending twice and risking a half-applied
2383            // state if the commit fails between them.
2384            match self.buffer.get_node_write(node_id) {
2385                Some(super::BufferedWrite::RetractNode {
2386                    valid_to: buffered_valid_to,
2387                    ..
2388                }) => {
2389                    let buffered_valid_to = *buffered_valid_to;
2390                    let valid_from = self.head_node_valid_from(node_id);
2391                    return Ok(super::RetractionResult {
2392                        valid_from,
2393                        valid_to: buffered_valid_to,
2394                        already_retracted: true,
2395                        edges_retracted: 0,
2396                    });
2397                }
2398                Some(super::BufferedWrite::DeleteNode { valid_from, .. }) => {
2399                    // Delete-parity: a committed delete leaves a tombstone
2400                    // with the degenerate interval [d, d); report the same
2401                    // shape for a deletion still sitting in this buffer.
2402                    let deleted_at = *valid_from;
2403                    return Ok(super::RetractionResult {
2404                        valid_from: deleted_at,
2405                        valid_to: deleted_at,
2406                        already_retracted: true,
2407                        edges_retracted: 0,
2408                    });
2409                }
2410                _ => {}
2411            }
2412
2413            // Buffer-aware read (Issue #3417): a node created earlier in THIS
2414            // transaction is retractable (read-your-own-writes). The buffered
2415            // Retract/Delete cases are already handled by the guard above; a
2416            // buffered Create/Update falls through to here.
2417            match self.read_own_node(node_id) {
2418                Ok(node) => {
2419                    // If the node being retracted contains vector properties, mark
2420                    // the buffer so the temporal vector index is notified on commit.
2421                    if !self.buffer.has_vector_operations() && node.properties.contains_vector() {
2422                        self.buffer.mark_has_vector_operations();
2423                    }
2424
2425                    // The closed interval starts at the head version's
2426                    // valid_from; `valid_to` must not precede it.
2427                    let valid_from = self.head_node_valid_from(node_id);
2428
2429                    validation::validate_valid_from_future(valid_to)?;
2430                    validation::validate_valid_from_not_before_creation(
2431                        &format!("node:{}", node_id.as_u64()),
2432                        valid_from,
2433                        valid_to,
2434                    )?;
2435
2436                    // Normalize an all-absent provenance bundle to `None` --
2437                    // never persist a fabricated empty object (Issue #3224/#3427).
2438                    let provenance = provenance
2439                        .filter(|p| !p.is_empty())
2440                        .map(std::sync::Arc::new);
2441
2442                    self.buffer.add(super::BufferedWrite::RetractNode {
2443                        node_id,
2444                        valid_to,
2445                        provenance,
2446                    })?;
2447
2448                    Ok(super::RetractionResult {
2449                        valid_from,
2450                        valid_to,
2451                        already_retracted: false,
2452                        edges_retracted: 0,
2453                    })
2454                }
2455                Err(err) => {
2456                    // Absent from current state: either already retracted or
2457                    // deleted (idempotent no-op) or it never existed.
2458                    if let Some(interval) = self.closed_node_valid_interval(node_id) {
2459                        return Ok(super::RetractionResult {
2460                            valid_from: interval.start(),
2461                            valid_to: interval.end(),
2462                            already_retracted: true,
2463                            edges_retracted: 0,
2464                        });
2465                    }
2466                    Err(err)
2467                }
2468            }
2469        })();
2470
2471        result.record_error_metric()
2472    }

2490    /// [`Provenance`](crate::core::provenance::Provenance) bundle recording the
2491    /// acting principal onto the retraction version (Issue #3427).
2492    ///
2493    /// See [`retract_node_with_provenance`](Self::retract_node_with_provenance)
2494    /// for the provenance-normalization and idempotency contract, and
2495    /// [`retract_edge`](Self::retract_edge) for the bi-temporal semantics.

2497    pub fn retract_edge_with_provenance(
2498        &mut self,
2499        edge_id: EdgeId,
2500        valid_to: Timestamp,
2501        provenance: Option<Provenance>,
2502    ) -> Result<super::RetractionResult> {

2504            // Check transaction state
2505            if self.state != TxState::Active {
2506                return Err(TransactionError::InvalidState {
2507                    current: format!("{:?}", self.state),
2508                    expected: "Active".to_string(),
2509                }
2510                .into());
2511            }

2513            // A retraction already buffered in THIS transaction (e.g. the
2514            // detach form visiting a self-loop from both adjacency
2515            // directions) is a no-op returning the buffered end; a deletion
2516            // already buffered in this transaction mirrors the
2517            // committed-delete case (degenerate [d, d) tombstone interval)
2518            // — see the identical guard in retract_node.
2519            match self.buffer.get_edge_write(edge_id) {
2520                Some(super::BufferedWrite::RetractEdge {
2521                    valid_to: buffered_valid_to,
2522                    ..
2523                }) => {

2532                }
2533                Some(super::BufferedWrite::DeleteEdge { valid_from, .. }) => {
2534                    let deleted_at = *valid_from;
2535                    return Ok(super::RetractionResult {
2536                        valid_from: deleted_at,
2537                        valid_to: deleted_at,
2538                        already_retracted: true,
2539                        edges_retracted: 0,
2540                    });
2541                }
2542                _ => {}
2543            }
2544
2545            // Buffer-aware read (Issue #3417): an edge created earlier in THIS
2546            // transaction is retractable (read-your-own-writes).
2547            match self.read_own_edge(edge_id) {
2548                Ok(edge) => {
2549                    if !self.buffer.has_vector_operations() && edge.properties.contains_vector() {
2550                        self.buffer.mark_has_vector_operations();
2551                    }
2552
2553                    let valid_from = self.head_edge_valid_from(edge_id);
2554
2555                    validation::validate_valid_from_future(valid_to)?;
2556                    validation::validate_valid_from_not_before_creation(
2557                        &format!("edge:{}", edge_id.as_u64()),
2558                        valid_from,
2559                        valid_to,
2560                    )?;
2561
2562                    // Normalize an all-absent provenance bundle to `None` --
2563                    // never persist a fabricated empty object (Issue #3224/#3427).
2564                    let provenance = provenance
2565                        .filter(|p| !p.is_empty())
2566                        .map(std::sync::Arc::new);

2568                    self.buffer.add(super::BufferedWrite::RetractEdge {
2569                        edge_id,
2570                        valid_to,
2571                        provenance,
2572                    })?;
2573
2574                    Ok(super::RetractionResult {
2575                        valid_from,
2576                        valid_to,
2577                        already_retracted: false,
2578                        edges_retracted: 0,

2580                }
2581                Err(err) => {

2585                            valid_to: interval.end(),

2587                            edges_retracted: 0,

2589                    }

2591                }
2592            }
2593        })();

2650impl Drop for WriteTransaction {
2651    fn drop(&mut self) {
2652        // Auto-rollback if the transaction never reached a terminal state.
2653        //
2654        // `Preparing` must be released here too (Issue #3415): a commit that
2655        // fails at any pre-`register_commit` step (validation, conflict,
2656        // constraint, WAL, or apply) leaves the consumed transaction in
2657        // `Preparing`. Without covering that state, its `TxId` would never be
2658        // removed from `TxVisibilityManager::active`, leaking one entry per
2659        // failed commit and pinning the snapshot horizon forever. The success
2660        // path transitions `Preparing -> Committed` before drop, so a committed
2661        // transaction never matches and is never double-aborted; an explicit
2662        // `rollback()`/`abort()` sets `Aborted` and likewise does not match.
2663        if matches!(self.state, TxState::Active | TxState::Preparing) {
2664            self.buffer.clear();
2665            // Register abort with visibility manager
2666            self.visibility_manager.register_abort(self.tx_id);
2667            self.state = TxState::Aborted;
2668        }
2669    }
2670}

2672/// Always-compiled one-shot commit-path interleaving seam for the lost-write
2673/// persist-race regression tests.
2674///
2675/// Unlike [`commit_test_hooks`] (which is `#[cfg(test)]` and thus invisible to
2676/// the crate's `tests/` integration tests), this seam is always compiled and
2677/// `#[doc(hidden)] pub`, so an integration test can arm it. It lets a test park
2678/// exactly one commit at the **durable-but-not-yet-applied** point of the commit
2679/// path (WAL fsynced + acknowledged, in-flight LSN registered, `apply_changes`
2680/// not yet called) while it drives a racing `persist_indexes()`.
2681///
2682/// # Cost
2683///
2684/// The fire path ([`run_pre_apply_once`]) is a single acquire atomic load when
2685/// unarmed (the production state), so it adds no measurable cost to the commit
2686/// hot path — the "an `Option` that's `None`" contract, implemented as an
2687/// `AtomicBool` that is `false` in production.
2688///
2689/// # Arming surface is feature-gated
2690///
2691/// The arming API (`arm_pre_apply_once`/`clear`, the `HOOK` static, and the
2692/// armed branch that locks `HOOK`) is compiled ONLY under
2693/// `#[cfg(any(test, feature = "testing"))]`. In a default build there is no way
2694/// to set `ARMED = true`, so [`run_pre_apply_once`] reduces to its inert fast
2695/// path (`if !ARMED { return; }`) and can never park the commit path — it is
2696/// inert by construction, not merely by convention. A downstream crate that
2697/// links the library without the `testing` feature cannot reach the parking
2698/// surface.
2699///
2700/// # One-shot
2701///
2702/// `arm_pre_apply_once` installs a hook fired by the **next** commit only; the
2703/// fire path atomically disarms before running the hook, so under concurrency
2704/// exactly one commit fires it. `clear` disarms without firing (test teardown).
2705#[doc(hidden)]
2706pub mod race_seam {
2707    #[cfg(any(test, feature = "testing"))]
2708    use std::sync::Mutex;
2709    use std::sync::atomic::{AtomicBool, Ordering};
2710
2711    #[cfg(any(test, feature = "testing"))]
2712    type Hook = Box<dyn FnOnce() + Send>;
2713
2714    /// Fast-path guard: `true` only while a hook is armed. Kept separate from
2715    /// the `Mutex` so the unarmed fire path never touches the lock. In a default
2716    /// (non-`testing`) build nothing can ever set this to `true`.
2717    static ARMED: AtomicBool = AtomicBool::new(false);
2718    #[cfg(any(test, feature = "testing"))]
2719    static HOOK: Mutex<Option<Hook>> = Mutex::new(None);
2720
2721    /// Arm a one-shot hook fired by the next commit at the durable-but-unapplied
2722    /// point. Intended for regression tests only; only compiled under
2723    /// `#[cfg(any(test, feature = "testing"))]`.
2724    #[cfg(any(test, feature = "testing"))]
2725    pub fn arm_pre_apply_once(hook: Box<dyn FnOnce() + Send>) {
2726        if let Ok(mut slot) = HOOK.lock() {
2727            *slot = Some(hook);
2728            ARMED.store(true, Ordering::Release);
2729        }
2730    }
2731
2732    /// Disarm any pending hook without firing it (test teardown). Only compiled
2733    /// under `#[cfg(any(test, feature = "testing"))]`.
2734    #[cfg(any(test, feature = "testing"))]
2735    pub fn clear() {
2736        ARMED.store(false, Ordering::Release);
2737        if let Ok(mut slot) = HOOK.lock() {
2738            *slot = None;
2739        }
2740    }
2741
2742    /// Fire and consume the armed hook exactly once. Called on the commit hot
2743    /// path just before `apply_changes`. Always compiled, but in a default build
2744    /// `ARMED` can never be `true` (the arming surface is gated out), so this is
2745    /// just the inert fast-path load.
2746    #[inline]
2747    pub(crate) fn run_pre_apply_once() {
2748        // Fast path: one acquire atomic load; `false` in production (the arming
2749        // surface is gated out of default builds, so `ARMED` is always `false`
2750        // and this whole body is inert by construction).
2751        if ARMED.load(Ordering::Acquire) {
2752            // Disarm atomically so concurrent commits never double-fire. This
2753            // branch is unreachable without the arming surface (test/testing only).
2754            #[cfg(any(test, feature = "testing"))]
2755            if ARMED.swap(false, Ordering::AcqRel) {
2756                let hook = HOOK.lock().ok().and_then(|mut slot| slot.take());
2757                if let Some(hook) = hook {
2758                    hook();
2759                }
2760            }
2761        }
2762    }
2763}
