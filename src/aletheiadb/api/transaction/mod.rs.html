1//! Transaction support for AletheiaDB
2//!
3//! This module provides MVCC (Multi-Version Concurrency Control) transactions
4//! with Snapshot Isolation level.
5//!
6//! # Transaction Types
7//!
8//! - [`ReadTransaction`]: Lightweight read-only transactions with zero overhead
9//! - [`WriteTransaction`]: Full ACID write transactions with write buffering and WAL
10//!
11//! # API Styles
12//!
13//! **Closure-based (recommended)**:
14//! ```rust,no_run
15//! # use aletheiadb::{AletheiaDB, PropertyMapBuilder, properties};
16//! # use aletheiadb::core::NodeId;
17//! # use aletheiadb::api::transaction::{ReadOps, WriteOps};
18//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
19//! # let db = AletheiaDB::new()?;
20//! # let id = NodeId::new(1)?;
21//! # let other = NodeId::new(2)?;
22//! # let props = PropertyMapBuilder::new().build();
23//! # let edge_props = PropertyMapBuilder::new().build();
24//! // Read-only
25//! let result = db.read(|tx| {
26//!     // get_node might fail if node doesn't exist
27//!     if let Ok(node) = tx.get_node(id) {
28//!         Ok::<_, Box<dyn std::error::Error>>(node.get_property("name").cloned())
29//!     } else {
30//!         Ok::<_, Box<dyn std::error::Error>>(None)
31//!     }
32//! })?;
33//!
34//! // Read-write (auto-commit on Ok, auto-rollback on Err)
35//! let node_id = db.write(|tx| {
36//!     let node_id = tx.create_node("Person", props)?;
37//!     tx.create_edge(node_id, other, "KNOWS", edge_props)?;
38//!     Ok::<_, Box<dyn std::error::Error>>(node_id)
39//! })?;
40//! # Ok(())
41//! # }
42//! ```
43//!
44//! **Explicit handles**:
45//! ```rust,no_run
46//! # use aletheiadb::{AletheiaDB, PropertyMapBuilder};
47//! # use aletheiadb::core::NodeId;
48//! # use aletheiadb::api::transaction::WriteOps;
49//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
50//! # let db = AletheiaDB::new()?;
51//! # let n1 = NodeId::new(1)?;
52//! # let n2 = NodeId::new(2)?;
53//! # let props = PropertyMapBuilder::new().build();
54//! let mut tx = db.write_transaction()?;
55//! tx.create_node("Person", props.clone())?;
56//! tx.create_edge(n1, n2, "KNOWS", props)?;
57//! tx.commit()?;  // or tx.rollback()
58//! # Ok(())
59//! # }
60//! ```

83/// Optional per-write settings: backdated `valid_from` and/or a write-time
84/// [`Provenance`] bundle (Issue #3224).
85///
86/// Constructed via [`WriteRequestOptions::new`] (or its [`Default`] impl) and the
87/// `with_*` builder methods. Passing `WriteRequestOptions::default()` reproduces the
88/// behavior of the plain `create_node`/`update_node`/etc. convenience methods
89/// exactly (valid time defaults to transaction start time, no provenance).
90#[derive(Debug, Clone, Default)]
91pub struct WriteRequestOptions {
92    pub(crate) valid_from: Option<Timestamp>,
93    pub(crate) provenance: Option<Provenance>,
94    /// Target namespace for a **create** (Issue #3349). `None` ⇒ the default
95    /// namespace (byte-identical to pre-namespace behavior). Ignored by update /
96    /// replace / delete / CAS paths, where the namespace is immutable and
97    /// re-stamped from the existing entity instead.
98    pub(crate) namespace: Option<crate::core::namespace::Namespace>,
99}
100
101impl WriteRequestOptions {
102    /// Create an empty set of options (equivalent to [`Default::default`]).
103    pub fn new() -> Self {
104        Self::default()
105    }
106
107    /// Set when the fact became valid in reality (`None` = transaction start time).
108    #[must_use]
109    pub fn with_valid_from(mut self, valid_from: Timestamp) -> Self {
110        self.valid_from = Some(valid_from);
111        self
112    }
113
114    /// Attach a write-time provenance bundle to this write.
115    #[must_use]
116    pub fn with_provenance(mut self, provenance: Provenance) -> Self {
117        self.provenance = Some(provenance);
118        self
119    }
120
121    /// Set the target namespace for a **create** write (Issue #3349).
122    ///
123    /// Supplying a namespace to a non-create op (update / replace / CAS /
124    /// lease-claim) is rejected with `INVALID_ARGUMENT`
125    /// ([`NamespaceError::Immutable`](crate::core::namespace::NamespaceError::Immutable)),
126    /// because a namespace is fixed at creation and those paths re-stamp it from
127    /// the existing entity.
128    ///
129    /// Note: this low-level write-path does **not** auto-register the namespace
130    /// in the registry — that is the caller's responsibility (the
131    /// `create_*_in_namespace` convenience methods and the PR3 MCP/HTTP handlers
132    /// register after a successful commit). A raw `with_namespace` write can
133    /// therefore stamp an entity in a namespace the registry does not list; the
134    /// membership index rebuilt at load reconciles that registry-vs-data
135    /// divergence.
136    #[must_use]
137    pub fn with_namespace(mut self, namespace: crate::core::namespace::Namespace) -> Self {
138        self.namespace = Some(namespace);
139        self
140    }
141}

143/// Outcome of a valid-time retraction (Issue #3230).
144///
145/// Returned by [`WriteTransaction::retract_node`]/[`WriteTransaction::retract_edge`]
146/// and the [`AletheiaDB`](crate::db::AletheiaDB) convenience wrappers. The
147/// closed valid interval is half-open: `[valid_from, valid_to)`.
148#[derive(Debug, Clone, Copy, PartialEq, Eq)]
149pub struct RetractionResult {
150    /// Start of the entity's (now closed) valid-time interval.
151    pub valid_from: Timestamp,
152    /// End of the entity's valid-time interval (exclusive). For an
153    /// idempotent re-retraction this is the *existing* end, regardless of
154    /// the `valid_to` passed to the call.
155    pub valid_to: Timestamp,
156    /// `true` when the entity was already retracted (or deleted) and this
157    /// call was a no-op: no new version was appended and no WAL entry was
158    /// written.
159    pub already_retracted: bool,
160    /// Number of connected edges co-retracted alongside a node (only set by
161    /// the detach form; `0` otherwise and for edge retractions).
162    pub edges_retracted: usize,
163}

165/// Common read operations available in all transaction types
166pub trait ReadOps {
167    /// Get a node by ID.
168    ///
169    /// This returns the node state visible in the current transaction snapshot.
170    ///
171    /// # Snapshot Isolation
172    ///
173    /// If the node has been modified or deleted by another transaction after this
174    /// transaction started, `get_node` will return the version visible at the start
175    /// of this transaction (Snapshot Isolation).
176    ///
177    /// # Performance
178    ///
179    /// - **Fast Path**: O(1) lookup in current storage (hash map)
180    /// - **Slow Path**: O(log N) lookup in historical storage if not found in current (or version not visible)
181    ///
182    /// # Example
183    ///
184    /// ```rust,no_run
185    /// # use aletheiadb::{AletheiaDB, core::NodeId, api::transaction::ReadOps};
186    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
187    /// # let db = AletheiaDB::new()?;
188    /// # let tx = db.read_transaction()?;
189    /// # let node_id = NodeId::new(1)?;
190    /// let node = tx.get_node(node_id)?;
191    /// println!("Node label: {:?}", node.label);
192    /// # Ok(())
193    /// # }
194    /// ```
195    fn get_node(&self, id: NodeId) -> Result<Node>;
196
197    /// Get an edge by ID.
198    ///
199    /// This returns the edge state visible in the current transaction snapshot.
200    ///
201    /// # Snapshot Isolation
202    ///
203    /// If the edge has been modified or deleted by another transaction after this
204    /// transaction started, `get_edge` will return the version visible at the start
205    /// of this transaction (Snapshot Isolation).
206    ///
207    /// # Example
208    ///
209    /// ```rust,no_run
210    /// # use aletheiadb::{AletheiaDB, core::EdgeId, api::transaction::ReadOps};
211    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
212    /// # let db = AletheiaDB::new()?;
213    /// # let tx = db.read_transaction()?;
214    /// # let edge_id = EdgeId::new(1)?;
215    /// let edge = tx.get_edge(edge_id)?;
216    /// println!("Edge label: {:?}", edge.label);
217    /// # Ok(())
218    /// # }
219    /// ```
220    fn get_edge(&self, id: EdgeId) -> Result<Edge>;
221
222    /// Get outgoing edges from a node.
223    ///
224    /// Returns all edges where `source == node_id` that are visible in the current snapshot.
225    ///
226    /// # Arguments
227    ///
228    /// * `node_id` - The source node to get edges from
229    ///
230    /// # Returns
231    ///
232    /// - `Ok(edge_ids)` - The node exists (is visible in this transaction's snapshot).
233    ///   The vector is **empty if the node has no outgoing edges** - that is a valid
234    ///   state, not an error.
235    /// - `Err(NodeNotFound)` - The node does not exist (or is not visible in this
236    ///   transaction's snapshot).
237    ///
238    /// This mirrors [`get_node`](Self::get_node): the same condition (a missing node)
239    /// produces an error, so callers can distinguish "node has no edges" from
240    /// "node doesn't exist" (Issue #359).
241    ///
242    /// # Ordering
243    ///
244    /// The order of edges is **not guaranteed**. Do not rely on edges being returned
245    /// in insertion order or sorted by ID. The internal storage may reorder edges
246    /// during compaction or persistence.
247    ///
248    /// # Snapshot Isolation
249    ///
250    /// In a read transaction, this method filters edges to ensure only those visible
251    /// in the current transaction snapshot are returned. Edges created by concurrent
252    /// transactions will not be seen.
253    ///
254    /// # Write transactions
255    ///
256    /// In a write transaction, only the **node existence check** is buffer-aware;
257    /// the returned edge list is read directly from committed current storage
258    /// **without snapshot filtering**. Consequently:
259    ///
260    /// - edges committed by concurrent transactions after this transaction
261    ///   started **are** returned,
262    /// - edges created in this transaction (still buffered, not yet committed)
263    ///   are **not** returned, and
264    /// - edges deleted in this transaction are **still** listed until commit.
265    ///
266    /// # Performance
267    ///
268    /// - **Time**: O(degree) to collect visible edges, plus a node existence check
269    ///   (O(1) fast path; may consult historical storage, O(log N), on a miss)
270    /// - **Space**: Allocates a new `Vec` containing all edge IDs
271    ///
272    /// # Example
273    ///
274    /// ```rust
275    /// # use aletheiadb::{AletheiaDB, properties, core::NodeId};
276    /// # use aletheiadb::api::transaction::{ReadOps, WriteOps};
277    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
278    /// # let db = AletheiaDB::new()?;
279    /// let (alice, bob) = db.write(|tx| {
280    ///     let alice = tx.create_node("Person", properties! { "name" => "Alice" })?;
281    ///     let bob = tx.create_node("Person", properties! { "name" => "Bob" })?;
282    ///     tx.create_edge(alice, bob, "KNOWS", properties! {})?;
283    ///     Ok::<_, aletheiadb::Error>((alice, bob))
284    /// })?;
285    ///
286    /// let tx = db.read_transaction()?;
287    ///
288    /// // Alice has one outgoing edge:
289    /// let edges = tx.get_outgoing_edges(alice)?;
290    /// assert_eq!(edges.len(), 1);
291    /// for edge_id in edges {
292    ///     let edge = tx.get_edge(edge_id)?;
293    ///     assert_eq!(edge.target, bob);
294    /// }
295    ///
296    /// // Bob exists but has no outgoing edges: Ok(empty), not an error.
297    /// assert!(tx.get_outgoing_edges(bob)?.is_empty());
298    ///
299    /// // A node that doesn't exist is an error, distinguishable from "no edges".
300    /// let missing = NodeId::new(999_999)?;
301    /// assert!(tx.get_outgoing_edges(missing).is_err());
302    /// # Ok(())
303    /// # }
304    /// ```
305    fn get_outgoing_edges(&self, node_id: NodeId) -> Result<Vec<EdgeId>>;
306

322    ///
323    /// # Ordering

326    ///
327    /// # Snapshot Isolation
328    ///

331    ///

335    /// the returned edge list is read directly from committed current storage
336    /// **without snapshot filtering**. Consequently:

343    ///
344    /// # Performance

443    /// Get the approximate number of nodes in the database.
444    ///
445    /// # Returns
446    ///
447    /// The count of nodes currently committed in the storage engine. Returns `0`
448    /// for an empty database. Deleted nodes are not counted. Nodes created in
449    /// this transaction but not yet committed are **not** included. Conversely,
450    /// nodes deleted in this transaction but not yet committed are still included.

453    ///

455    /// Unlike `get_node()`, this count is **NOT snapshot-isolated**. It may include
456    /// nodes created by transactions that committed after this read transaction started.
457    ///
458    /// This design choice enables O(1) performance without scanning the entire
459    /// database to filter visibility for every node.
460    ///
461    /// # Example
462    ///
463    /// ```rust,no_run

465    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
466    /// # let db = AletheiaDB::new()?;
467    /// # let tx = db.read_transaction()?;

524/// Write operations (only available in write transactions)
525pub trait WriteOps: ReadOps {
526    /// Create a new node with optional backdated valid_from time.
527    ///
528    /// # Arguments
529    ///
530    /// * `label` - Node label
531    /// * `properties` - Node properties
532    /// * `valid_from` - When the fact became valid (None = transaction time)
533    ///
534    /// # Bi-Temporal Semantics
535    ///
536    /// - If `valid_from` is None, `valid_time` starts at the **transaction start time**.
537    /// - `transaction_time` always starts at the **commit time**.
538    /// - This means by default, facts are considered valid from the moment the transaction began.
539    /// - If `valid_from` is Some(ts), valid_time starts at `ts`.
540    ///
541    /// # Example
542    ///
543    /// ```rust,no_run
544    /// # use aletheiadb::{AletheiaDB, properties, api::transaction::WriteOps};
545    /// # use aletheiadb::core::temporal::time;
546    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
547    /// # let db = AletheiaDB::new()?;
548    /// # let mut tx = db.write_transaction()?;
549    /// // Backdate a fact to 1 hour ago
550    /// let one_hour_ago = time::now().wallclock() - 3_600_000_000;
551    /// let valid_from = time::from_secs(one_hour_ago / 1_000_000);
552    ///
553    /// let node_id = tx.create_node_with_valid_time(
554    ///     "Person",
555    ///     properties! { "name" => "Alice" },
556    ///     Some(valid_from)
557    /// )?;
558    /// # Ok(())
559    /// # }
560    /// ```
561    fn create_node_with_valid_time(
562        &mut self,
563        label: &str,
564        properties: PropertyMap,
565        valid_from: Option<Timestamp>,
566    ) -> Result<NodeId> {
567        let options = WriteRequestOptions {
568            valid_from,
569            provenance: None,
570            namespace: None,
571        };
572        self.create_node_with_options(label, properties, options)
573    }
574
575    /// Create a new node with an optional [`WriteRequestOptions`] bundle (backdated
576    /// `valid_from` and/or a write-time [`Provenance`] bundle, Issue #3224).
577    ///
578    /// This is the most general node-creation method; all other
579    /// `create_node*` methods delegate to it. Passing `WriteRequestOptions::default()`
580    /// is identical to [`create_node`](Self::create_node).
581    ///
582    /// # Example
583    ///
584    /// ```rust,no_run
585    /// # use aletheiadb::{AletheiaDB, properties, Provenance, api::transaction::{WriteOps, WriteRequestOptions}};
586    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
587    /// # let db = AletheiaDB::new()?;
588    /// # let mut tx = db.write_transaction()?;
589    /// let provenance = Provenance::builder().source("hr-system").confidence(0.95).build()?;
590    /// let node_id = tx.create_node_with_options(

770    /// Compare-and-set a node's properties, conditional on its committed head
771    /// still being `expected_version` (Issue #3577).
772    ///
773    /// This is a conditional **full replace** of the node's property map (not a
774    /// PATCH merge): the whole `properties` map becomes the new state, matching
775    /// the "write the whole claim state" semantics the lease layer builds on.
776    /// The label and node id are preserved.
777    ///
778    /// # Semantics
779    ///
780    /// The precondition is enforced at **commit time, under the
781    /// commit-serialization guard** — so two claimants opened on the same
782    /// snapshot cannot both succeed (the second observes the first's new version
783    /// and aborts). On success the new version id is returned. On a lost claim
784    /// the whole transaction aborts with
785    /// [`TransactionError::CasMismatch`](crate::core::error::TransactionError::CasMismatch)
786    /// (a non-retriable precondition failure, distinct from the retriable
787    /// `SerializationFailure`) and **nothing is written**. A CAS against a
788    /// nonexistent or deleted node fails with `CasMismatch { actual: None }`
789    /// rather than a `NodeNotFound`.
790    fn compare_and_set_node(
791        &mut self,
792        node_id: NodeId,
793        expected_version: crate::core::id::VersionId,
794        properties: PropertyMap,
795    ) -> Result<crate::core::id::VersionId> {
796        self.compare_and_set_node_with_options(
797            node_id,
798            expected_version,
799            properties,
800            WriteRequestOptions::default(),
801        )
802    }

1221    /// Replace a node's **entire** property map and label (non-PATCH overwrite).
1222    ///
1223    /// Unlike [`update_node`](Self::update_node) (PATCH: merges the incoming map
1224    /// onto the existing one), this performs a full overwrite: the node's
1225    /// resulting property map is *exactly* `properties`, and its label becomes
1226    /// `label`. Any key present on the prior version but absent from
1227    /// `properties` is **removed** from current state — its history is
1228    /// preserved and still recallable `AS OF` an earlier bi-temporal coordinate
1229    /// (anchor *both* dimensions before the replace). Passing an empty map
1230    /// removes all properties while keeping the node in existence.
1231    ///
1232    /// Node label mutation is only possible through this method;
1233    /// [`update_node`](Self::update_node) preserves the label.
1234    ///
1235    /// This is the most general node-replace method; the other `replace_node*`
1236    /// methods delegate to it. The provenance recorded here describes *this*
1237    /// version only — it is not inherited from the version being replaced.
1238    fn replace_node_with_options(
1239        &mut self,
1240        node_id: NodeId,
1241        label: &str,
1242        properties: PropertyMap,
1243        options: WriteRequestOptions,
1244    ) -> Result<()>;
1245
1246    /// Replace a node's entire property map and label with an optional backdated
1247    /// `valid_from` time. See [`replace_node_with_options`](Self::replace_node_with_options).
1248    fn replace_node_with_valid_time(
1249        &mut self,
1250        node_id: NodeId,
1251        label: &str,
1252        properties: PropertyMap,
1253        valid_from: Option<Timestamp>,
1254    ) -> Result<()> {
1255        self.replace_node_with_options(
1256            node_id,
1257            label,
1258            properties,
1259            WriteRequestOptions {
1260                valid_from,
1261                provenance: None,
1262                namespace: None,
1263            },
1264        )
1265    }

1271    fn replace_node(
1272        &mut self,
1273        node_id: NodeId,
1274        label: &str,
