21/// Read-only transaction
22///
23/// Read-only transactions are lightweight:
24/// - No write buffer
25/// - No WAL logging
26/// - Snapshot-based reads for consistency
27/// - No commit overhead
28///
29/// # Example
30///
31/// ```rust,no_run
32/// # use aletheiadb::{AletheiaDB, core::NodeId, api::transaction::ReadOps};
33/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
34/// # let db = AletheiaDB::new()?;
35/// # let node_id = NodeId::new(1)?;
36/// let tx = db.read_transaction()?;
37/// let node = tx.get_node(node_id)?;
38/// // No commit needed - transaction is read-only
39/// # Ok(())
40/// # }
41/// ```
42pub struct ReadTransaction {
43    tx_id: TxId,
44    start_timestamp: crate::core::temporal::Timestamp,
45    snapshot: TransactionSnapshot,
46    current: Arc<CurrentStorage>,
47    visibility_manager: Arc<TxVisibilityManager>,
48    historical: Arc<RwLock<HistoricalStorage>>,
49}
50
51impl ReadTransaction {
52    /// Create a new read-only transaction
53    ///
54    /// # Arguments
55    ///
56    /// * `tx_id` - The unique identifier assigned to this transaction
57    /// * `snapshot` - The isolated view of the database state captured at the start of this transaction
58    /// * `current` - Reference to the current (latest) graph state storage
59    /// * `visibility_manager` - Manages which versions of entities are visible to this transaction
60    /// * `historical` - Reference to the historical graph state storage for resolving older versions
61    pub(crate) fn new(
62        tx_id: TxId,
63        snapshot: TransactionSnapshot,
64        current: Arc<CurrentStorage>,
65        visibility_manager: Arc<TxVisibilityManager>,
66        historical: Arc<RwLock<HistoricalStorage>>,
67    ) -> Self {
68        ReadTransaction {
69            tx_id,
70            start_timestamp: snapshot.snapshot_timestamp,
71            snapshot,
72            current,
73            visibility_manager,
74            historical,
75        }
76    }
77
78    /// Get transaction metadata.
79    ///
80    /// This metadata provides information about the transaction's lifecycle,
81    /// such as its unique ID, start timestamp, and current state. Since this
82    /// is a read-only transaction, the state will always be `Active` until dropped,
83    /// and the commit timestamp will be `None`.
84    ///
85    /// # Examples
86    ///
87    /// ```rust
88    /// # use aletheiadb::AletheiaDB;
89    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
90    /// # let db = AletheiaDB::new()?;
91    /// let tx = db.read_transaction()?;
92    /// let meta = tx.metadata();
93    ///
94    /// assert!(meta.is_read_only);
95    /// assert_eq!(meta.tx_id, tx.tx_id());
96    /// # Ok(())
97    /// # }
98    /// ```
99    pub fn metadata(&self) -> TxMetadata {
100        TxMetadata {
101            tx_id: self.tx_id,
102            start_timestamp: self.start_timestamp,
103            commit_timestamp: None,
104            state: TxState::Active,
105            is_read_only: true,
106        }
107    }
108
109    /// Get transaction ID.
110    ///
111    /// Returns the unique identifier assigned to this transaction when it was created.
112    ///
113    /// # Examples
114    ///
115    /// ```rust
116    /// # use aletheiadb::AletheiaDB;
117    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
118    /// # let db = AletheiaDB::new()?;
119    /// let tx = db.read_transaction()?;
120    /// let id = tx.tx_id();
121    ///
122    /// println!("Running in transaction context: {:?}", id);
123    /// # Ok(())
124    /// # }
125    /// ```
126    pub fn tx_id(&self) -> TxId {
127        self.tx_id
128    }

130    /// Query historical storage for a node version visible at snapshot time.
131    ///
132    /// This is the slow path used when the current version is not visible
133    /// or when the node has been deleted from current storage.
134    fn get_node_from_historical(&self, id: NodeId) -> Result<Node> {
135        let historical = self.historical.read();
136
137        // Find version visible at our snapshot timestamp
138        let version_id = historical.find_node_version_at_time(
139            id,
140            self.snapshot.snapshot_timestamp, // valid_time
141            self.snapshot.snapshot_timestamp, // transaction_time
142        );
143
144        match version_id {
145            Some(vid) => {
146                // Found a visible version - reconstruct it
147                let version = historical
148                    .get_node_version(vid)
149                    .ok_or(StorageError::VersionNotFound(vid))?;
150
151                // Reconstruct properties from anchor+delta
152                let properties = historical.reconstruct_node_properties(vid)?;
153
154                // Extract metadata from the temporal interval
155                // NOTE: Historical versions don't track created_by_tx, so we use TxId(0)
156                // The commit_timestamp is extracted from transaction_time.start
157                let metadata = VersionMetadata::new(
158                    super::TxId::new(0), // Historical versions don't track creating tx
159                    version.temporal.transaction_time().start(), // Extract commit timestamp
160                );
161
162                // Build Node from historical version
163                Ok(Node::with_metadata(
164                    id,
165                    version.label,
166                    properties,
167                    vid,
168                    metadata,
169                ))
170            }
171            None => {
172                // No version visible at snapshot time
173                Err(StorageError::NodeNotFound(id).into())
174            }
175        }
176    }
177
178    /// Filter a list of edge IDs to only include those visible in our snapshot.
179    ///
180    /// This prevents phantom reads by checking each edge's visibility.
181    /// Edges that don't exist or aren't visible are filtered out.
182    ///
183    /// ⚡ Bolt Optimization: Uses `.retain()` instead of `.into_iter().filter(...).collect()`
184    /// to filter in-place and avoid allocating a new `Vec`.
185    fn filter_visible_edges(&self, mut edge_ids: Vec<EdgeId>) -> Vec<EdgeId> {
186        edge_ids.retain(|&edge_id| {
187            // Use embedded commit_timestamp for visibility check (Issue #238).
188            if let Ok(edge) = self.current.get_edge(edge_id) {
189                self.visibility_manager.is_visible_with_embedded_ts(
190                    &self.snapshot,
191                    edge.metadata.created_by_tx,
192                    edge.metadata.commit_timestamp,
193                )
194            } else {
195                // Edge doesn't exist or was deleted - not visible
196                false
197            }
198        });
199        edge_ids
200    }
201
202    /// Query historical storage for an edge version visible at snapshot time.
203    ///
204    /// This is the slow path used when the current version is not visible
205    /// or when the edge has been deleted from current storage.
206    fn get_edge_from_historical(&self, id: EdgeId) -> Result<Edge> {
207        let historical = self.historical.read();
208
209        // Find version visible at our snapshot timestamp
210        let version_id = historical.find_edge_version_at_time(
211            id,
212            self.snapshot.snapshot_timestamp, // valid_time
213            self.snapshot.snapshot_timestamp, // transaction_time
214        );
215
216        match version_id {
217            Some(vid) => {
218                // Found a visible version - reconstruct it
219                let version = historical
220                    .get_edge_version(vid)
221                    .ok_or(StorageError::VersionNotFound(vid))?;
222
223                // Reconstruct properties from anchor+delta
224                let properties = historical.reconstruct_edge_properties(vid)?;
225
226                // Extract metadata from the temporal interval
227                // NOTE: Historical versions don't track created_by_tx, so we use TxId(0)
228                // The commit_timestamp is extracted from transaction_time.start
229                let metadata = VersionMetadata::new(
230                    super::TxId::new(0), // Historical versions don't track creating tx
231                    version.temporal.transaction_time().start(), // Extract commit timestamp
232                );
233
234                // Build Edge from historical version
235                Ok(Edge::with_metadata(
236                    id,
237                    version.label,
238                    version.source,
239                    version.target,
240                    properties,
241                    vid,
242                    metadata,
243                ))
244            }
245            None => {
246                // No version visible at snapshot time
247                Err(StorageError::EdgeNotFound(id).into())
248            }
249        }
250    }

253impl ReadOps for ReadTransaction {
254    fn get_node(&self, id: NodeId) -> Result<Node> {
255        let result = if let Ok(current_node) = self.current.get_node(id) {
256            // Use embedded commit_timestamp for visibility check (HyPer/TiDB pattern, Issue #238).
257            // Bypasses the TxVisibilityManager::committed map lock for the common fast path.
258            if self.visibility_manager.is_visible_with_embedded_ts(
259                &self.snapshot,
260                current_node.metadata.created_by_tx,
261                current_node.metadata.commit_timestamp,
262            ) {
263                Ok(current_node)
264            } else {
265                // If not visible, fall through to the slow path.
266                self.get_node_from_historical(id)
267            }
268        } else {
269            // If the node is not in current storage (e.g., it was deleted),
270            // we must still check historical storage.
271            self.get_node_from_historical(id)
272        };
273
274        result.record_error_metric()
275    }
276
277    fn get_edge(&self, id: EdgeId) -> Result<Edge> {
278        let result = if let Ok(current_edge) = self.current.get_edge(id) {
279            // Use embedded commit_timestamp for visibility check (HyPer/TiDB pattern, Issue #238).
280            if self.visibility_manager.is_visible_with_embedded_ts(
281                &self.snapshot,
282                current_edge.metadata.created_by_tx,
283                current_edge.metadata.commit_timestamp,
284            ) {
285                Ok(current_edge)
286            } else {
287                // If not visible, fall through to the slow path.
288                self.get_edge_from_historical(id)
289            }
290        } else {
291            // If the edge is not in current storage (e.g., it was deleted),
292            // we must still check historical storage.
293            self.get_edge_from_historical(id)
294        };
295
296        result.record_error_metric()
297    }
298
299    fn get_outgoing_edges(&self, node_id: NodeId) -> Result<Vec<EdgeId>> {
300        // Existence check (Issue #359): a missing node is an error, so callers can
301        // distinguish "node has no edges" (Ok(empty)) from "node doesn't exist".
302        self.get_node(node_id)?;
303        // Filter edges to only return those visible in our snapshot
304        // This prevents phantom reads where we see edges created after our snapshot
305        // Note: CurrentStorage::get_outgoing_edges() uses frozen view when available
306        let edge_ids = self.current.get_outgoing_edges(node_id);
307        Ok(self.filter_visible_edges(edge_ids))
308    }
309
310    fn get_incoming_edges(&self, node_id: NodeId) -> Result<Vec<EdgeId>> {
311        // Existence check (Issue #359): a missing node is an error.
312        self.get_node(node_id)?;
313        // Filter edges to only return those visible in our snapshot
314        // Note: CurrentStorage::get_incoming_edges() uses frozen view when available
315        let edge_ids = self.current.get_incoming_edges(node_id);
316        Ok(self.filter_visible_edges(edge_ids))
317    }
318
319    fn get_outgoing_edges_with_label(&self, node_id: NodeId, label: &str) -> Result<Vec<EdgeId>> {
320        // Existence check (Issue #359): a missing node is an error. An existing
321        // node with no edges matching `label` is Ok(empty) - the label is a
322        // filter, not an existence check.
323        self.get_node(node_id)?;
324        // Filter edges to only return those visible in our snapshot
325        let edge_ids = self.current.get_outgoing_edges_with_label(node_id, label);
326        Ok(self.filter_visible_edges(edge_ids))
327    }
328
329    fn node_count(&self) -> usize {
330        self.current.node_count()
331    }
332
333    fn edge_count(&self) -> usize {
334        self.current.edge_count()
335    }
336
337    fn find_nodes_by_property(
338        &self,
339        label: &str,
340        property_key: &str,
341        property_value: &PropertyValue,
342    ) -> Vec<NodeId> {
343        // ⚡ Bolt Optimization: Uses `.retain()` to filter in-place and avoid allocating a new `Vec`.
344        let mut node_ids = self
345            .current
346            .find_nodes_by_property(label, property_key, property_value);
347
348        node_ids.retain(|node_id| {
349            self.current
350                .get_node(*node_id)
351                .map(|node| {
352                    // Use embedded commit_timestamp for visibility check (Issue #238).
353                    self.visibility_manager.is_visible_with_embedded_ts(
354                        &self.snapshot,
355                        node.metadata.created_by_tx,
356                        node.metadata.commit_timestamp,
357                    )
358                })
359                .unwrap_or(false)
360        });
361
362        node_ids
363    }
364}

366impl Drop for ReadTransaction {
367    fn drop(&mut self) {
368        // Register abort to remove from active set
369        // This prevents memory leak in active transactions set
370        self.visibility_manager.register_abort(self.tx_id);
371    }
372}
