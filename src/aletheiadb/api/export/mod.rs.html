1//! Columnar Parquet export of the graph for analytics interoperability (Issue #3364).
2//!
3//! This is the export half of the Parquet import/export layer; the import half lives
4//! in [`crate::api::import`] and shares the same columnar schema so that an
5//! export → import round-trip reproduces the graph. Two modes are offered:
6//!
7//! - **Current-state export** ([`Exporter::nodes_to_parquet`] /
8//!   [`Exporter::edges_to_parquet`]) writes the live graph with a stable, documented
9//!   schema: an `id`/`label` (or `edge_type`/`source_key`/`target_key`) header, a
10//!   nullable `valid_time` column, and one typed column per property key discovered by
11//!   scanning the database. The column layout is identical to what the importer's
12//!   [`NodeMapping`](crate::api::import::NodeMapping) /
13//!   [`EdgeMapping`](crate::api::import::EdgeMapping) expect, so the file re-imports
14//!   losslessly for scalars, timestamps, and `list<float32>` embeddings.
15//! - **Full bi-temporal history export** ([`Exporter::node_history_to_parquet`] /
16//!   [`Exporter::edge_history_to_parquet`]) writes **one row per version** with
17//!   `valid_from`, `valid_to` (**null = open interval**, never a sentinel),
18//!   `transaction_time`, a `tombstone` marker, and the five provenance columns.
19//!
20//! # Streaming and memory
21//!
22//! Both modes stream: the database is scanned once to discover the property schema,
23//! then a second time to emit rows in [`ExportConfig::batch_size`]-row Arrow
24//! `RecordBatch`es, each flushed as a Parquet row group before the next is built. Peak
25//! memory is therefore bounded to roughly one batch of decoded cells plus the set of
26//! distinct property keys, not the whole graph (target: 1M nodes in well under 1 GB).
27//!
28//! # Documented lossy items
29//!
30//! - The HLC **logical counter** (the sub-microsecond tiebreaker of a
31//!   [`HybridTimestamp`](crate::core::hlc::HybridTimestamp)) is dropped; only the
32//!   microsecond wallclock is written.
33//! - Property values that have no natural columnar type — [`PropertyValue::Bytes`],
34//!   heterogeneous [`PropertyValue::Array`], and [`PropertyValue::SparseVector`] — and
35//!   any property **key whose observed type conflicts across rows** are routed to a
36//!   single `properties_json` overflow column (a reversible tagged-JSON encoding), not
37//!   a native typed column. The default importer decodes native columns; re-importing
38//!   the overflow column requires custom mapping (see the guide).
39//! - History export enumerates entities through the whole-window **changefeed**, so an
40//!   entity that was deleted or retracted (and is no longer in current state) is still
41//!   exported with its full version history and tombstone. A deleted **edge**'s
42//!   endpoints are not carried on a historical version, so `source_key`/`target_key` are
43//!   written as null for an edge that is no longer live; the residual completeness caveat
44//!   is the changefeed's own (see the guide).
45

62/// Tunables for an export run.
63#[derive(Debug, Clone)]
64pub struct ExportConfig {
65    /// Number of rows per Arrow `RecordBatch` / Parquet row group (default `8192`).
66    ///
67    /// Bounds peak export memory: a larger batch amortizes write overhead, a smaller
68    /// batch lowers the memory high-water mark. Clamped to at least `1`.
69    pub batch_size: usize,
70}
71
72impl Default for ExportConfig {
73    fn default() -> Self {
74        ExportConfig { batch_size: 8192 }
75    }
76}
77
78/// Summary of a single `*_to_parquet` call.
79#[derive(Debug, Clone, Default, PartialEq, Eq)]
80pub struct ExportReport {
81    /// Number of current-state nodes written (current-state node export).
82    pub nodes_exported: usize,
83    /// Number of current-state edges written (current-state edge export).
84    pub edges_exported: usize,
85    /// Number of node **versions** written (history node export).
86    pub node_versions_exported: usize,
87    /// Number of edge **versions** written (history edge export).
88    pub edge_versions_exported: usize,
89}

91/// A stateless Parquet exporter obtained via [`AletheiaDB::export`].
92pub struct Exporter<'a> {
93    db: &'a AletheiaDB,
94    config: ExportConfig,
95}
96
97impl AletheiaDB {
98    /// Begin a Parquet export session.
99    ///
100    /// Returns an [`Exporter`] bound to this database. See the [module docs](self) for
101    /// the schema, streaming behavior, and documented lossy items.
102    pub fn export(&self) -> Exporter<'_> {
103        Exporter {
104            db: self,
105            config: ExportConfig::default(),
106        }
107    }
108}
109
110impl<'a> Exporter<'a> {
111    /// Override the full export configuration.
112    #[must_use]
113    pub fn with_config(mut self, config: ExportConfig) -> Self {
114        self.config = config;
115        self
116    }
117
118    /// Set the per-row-group batch size (clamped to at least `1`).
119    #[must_use]
120    pub fn batch_size(mut self, batch_size: usize) -> Self {
121        self.config.batch_size = batch_size.max(1);
122        self
123    }
124
125    /// Export the current state of every node to a Parquet file (Issue #3364).
126    ///
127    /// Columns: `id: int64`, `label: string`, `valid_time: timestamp(µs, UTC)`
128    /// (nullable), then one typed column per property key (scalars → int64 / double /
129    /// bool / string, embeddings → `list<float32>`), plus a `properties_json` overflow
130    /// column when any key needs it. See the [module docs](self).
131    #[cfg(feature = "parquet")]
132    pub fn nodes_to_parquet(&self, path: impl AsRef<Path>) -> Result<ExportReport> {
133        parquet::export_nodes(self.db, path.as_ref(), self.config.batch_size.max(1))
134    }
135
136    /// Export the current state of every edge to a Parquet file (Issue #3364).
137    ///
138    /// Columns: `edge_type: string`, `source_key: int64`, `target_key: int64`,
139    /// `valid_time: timestamp(µs, UTC)` (nullable), then one typed column per property
140    /// key plus the `properties_json` overflow column when needed. Endpoint keys are
141    /// the endpoints' node ids, matching the `id` column of a node export.
142    #[cfg(feature = "parquet")]
143    pub fn edges_to_parquet(&self, path: impl AsRef<Path>) -> Result<ExportReport> {
144        parquet::export_edges(self.db, path.as_ref(), self.config.batch_size.max(1))
145    }
146
147    /// Export the full bi-temporal history of every node — one row per version.
148    ///
149    /// Columns: `id: int64`, `label: string`, `version: int64`, the typed property
150    /// columns, `valid_from`/`valid_to` (`valid_to` null = still-open interval),
151    /// `transaction_time`, `tombstone: bool`, and the five nullable `provenance_*`
152    /// columns. See the [module docs](self) for the tombstone convention and the
153    /// deleted-edge enumeration caveat (which applies to
154    /// [`edge_history_to_parquet`](Self::edge_history_to_parquet)).
155    #[cfg(feature = "parquet")]
156    pub fn node_history_to_parquet(&self, path: impl AsRef<Path>) -> Result<ExportReport> {
157        parquet::export_node_history(self.db, path.as_ref(), self.config.batch_size.max(1))
158    }
159
160    /// Export the full bi-temporal history of every edge — one row per version.
161    ///
162    /// Same shape as [`node_history_to_parquet`](Self::node_history_to_parquet) plus
163    /// `edge_type`/`source_key`/`target_key`. Edges are enumerated through
164    /// current-state adjacency; see the [module docs](self) for the deleted-edge gap.
165    #[cfg(feature = "parquet")]
166    pub fn edge_history_to_parquet(&self, path: impl AsRef<Path>) -> Result<ExportReport> {
167        parquet::export_edge_history(self.db, path.as_ref(), self.config.batch_size.max(1))
168    }
169}
