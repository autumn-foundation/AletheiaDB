1//! Neo4j **APOC Cypher-script dump** importer (Issue #3356).
2//!
3//! Loads the `create`-format output of `apoc.export.cypher.all` — a stream of
4//! `CREATE`/`MATCH ... CREATE` Cypher statements framed by `:begin`/`:commit`
5//! transaction markers — into AletheiaDB, reusing the #3211 bulk-import spine
6//! (`import_nodes_with` / `import_edges_with`) and the Neo4j fidelity report,
7//! provenance, and label-strategy machinery from the sibling CSV importer
8//! ([`super::neo4j`]).
9//!
10//! It is a **self-contained** hand-written parser: a statement splitter, a
11//! Cypher-literal value parser, and node/relationship pattern parsers, all in
12//! this module. It does **not** depend on the openCypher query engine (the
13//! `cypher` feature); the apoc `create` grammar is small and fixed, so parsing
14//! it directly keeps the importer available in every build.
15//!
16//! # Supported constructs
17//!
18//! - `:begin` / `:commit` / `:rollback` transaction markers (no-op delimiters).
19//! - Node `CREATE (:Label {..})`, including multi-label `(:A:B:C {..})`, the
20//!   backtick-quoted `` (:`Label` {`prop`: ..}) `` form, and apoc's synthetic
21//!   `` :`UNIQUE IMPORT LABEL` `` + `` `UNIQUE IMPORT ID` `` key scheme.
22//! - Relationship `MATCH (a {key}), (b {key}) CREATE (a)-[:TYPE {..}]->(b)`,
23//!   resolving endpoints by the matched key property.
24//! - The apoc-optimized `UNWIND [{..},..] AS row CREATE (n:Label) SET n += row.properties`
25//!   node batch form and its relationship `MATCH ... CREATE ... SET r += row.properties`
26//!   variant.
27//! - Cypher literal values: single/double-quoted strings (with `\n \t \r \b \f
28//!   \\ \' \" \/ \uXXXX` escapes), integers, floats (incl. exponent), booleans,
29//!   `null`, lists, nested maps (serialized to a JSON string), backtick
30//!   identifiers, and the temporal constructors `datetime(..)`,
31//!   `localdatetime(..)`, `date(..)` (→ epoch micros) and `time(..)` /
32//!   `localtime(..)` (→ string). Numeric arrays become a dense
33//!   [`PropertyValue::Vector`] **only** for a property named in
34//!   [`Neo4jCypherOptions::vector_properties`], never silently.
35//!
36//! # Out of scope (reported, never silently dropped)
37//!
38//! User-defined `CREATE CONSTRAINT` / `CREATE INDEX`, `DROP`, `CALL`, `MERGE`,
39//! spatial `point(..)`, `duration(..)`, and any unknown constructor (the form a
40//! byte array takes when apoc wraps it in a function) are enumerated in the
41//! fidelity report's `unsupported` list with counts (and set `zero_loss =
42//! false`). `--strict-types` promotes an unsupported *value* to a hard error.
43//! Note that — unlike the CSV importer, which reads a `:byte[]` type annotation
44//! from the header — a Cypher dump carries no per-value type tags, so a byte
45//! array serialized as a bare integer list is indistinguishable from a
46//! `long[]` and imports as an integer array (its data is preserved); only the
47//! constructor form is reportable as unsupported.
48//! apoc's own `UNIQUE IMPORT` scaffolding (the constraint it creates/drops and
49//! the `REMOVE`-cleanup statement) manages import-only artifacts this importer
50//! already strips, so it is ignored **losslessly** and does not affect
51//! `zero_loss`.

65/// Options controlling a Neo4j APOC Cypher-dump import (Issue #3356).

86    /// When `true`, an unsupported value (`point(..)`, `duration(..)`, an

90    /// The property apoc uses as the stable import identifier that
91    /// relationships match on (default `"UNIQUE IMPORT ID"`). Its value becomes
92    /// the node's business key and is **not** stored as a property.

100}

182/// Per-statement fidelity scratch, folded into the [`Neo4jFidelityReport`] only
183/// once the produced node/edge is known to import (so a skipped or unresolved
184/// entity never inflates the mapping / coercion / unsupported counts).
185#[derive(Default, Clone)]
186struct FidelityNote {
187    label: Option<(String, String)>,
188    edge_type: Option<String>,
189    coercions: Vec<(String, &'static str)>,
190    unsupported: Vec<(String, String)>,
191    prop_count: usize,
192}
193
194impl FidelityNote {
195    fn note_coercion(&mut self, column: &str, kind: &'static str) {
196        if !self
197            .coercions
198            .iter()
199            .any(|(c, k)| c == column && *k == kind)
200        {
201            self.coercions.push((column.to_string(), kind));
202        }
203    }
204
205    fn note_unsupported(&mut self, column: &str, ty: &str) {
206        if !self.unsupported.iter().any(|(c, _)| c == column) {
207            self.unsupported.push((column.to_string(), ty.to_string()));
208        }
209    }
210}

212/// Fold a committed entity's [`FidelityNote`] into the shared report.
213fn fold_note(report: &mut Neo4jFidelityReport, note: &FidelityNote) {
214    if let Some((neo4j_labels, chosen)) = &note.label {
215        report.note_label(neo4j_labels, chosen);
216    }
217    if let Some(edge_type) = &note.edge_type {
218        report.note_type(edge_type);
219    }
220    for (column, kind) in &note.coercions {
221        report.note_coercion(column, kind, super::neo4j::coercion_detail(kind));
222    }
223    for (column, ty) in &note.unsupported {
224        report.note_unsupported(column, ty);
225    }
226    report.properties_imported += note.prop_count;
227}

251/// One top-level statement of the dump, with the 1-based source line it began on.
252struct Statement {
253    line: usize,
254    text: String,
255}
256
257/// Split an apoc Cypher dump into top-level statements.
258///
259/// A statement is either a `:`-prefixed meta command (`:begin` / `:commit` /
260/// `:rollback`, read to end of line) or a `;`-terminated Cypher statement.
261/// Semicolons inside strings, backtick identifiers, or bracket/brace/paren
262/// nesting do not terminate a statement. A non-whitespace remainder with no
263/// terminating `;` is a **truncated** dump and errors (never silently dropped).
264fn split_statements(src: &str) -> std::result::Result<Vec<Statement>, String> {
265    let chars: Vec<char> = src.chars().collect();
266    let mut out = Vec::new();
267    let mut i = 0usize;
268    let mut line = 1usize;
269
270    // Advance over whitespace, counting newlines; returns at first non-ws char.
271    while i < chars.len() {
272        // Skip leading whitespace.
273        while i < chars.len() && chars[i].is_whitespace() {
274            if chars[i] == '\n' {
275                line += 1;
276            }
277            i += 1;
278        }
279        if i >= chars.len() {
280            break;
281        }
282        let start_line = line;
283        if chars[i] == ':' {
284            // Meta command: read to end of line.
285            let start = i;
286            while i < chars.len() && chars[i] != '\n' {
287                i += 1;
288            }
289            let text: String = chars[start..i].iter().collect();
290            out.push(Statement {
291                line: start_line,
292                text: text.trim().to_string(),
293            });
294            continue;
295        }
296
297        // Cypher statement: accumulate to a top-level ';'.
298        let start = i;
299        let mut in_string: Option<char> = None;
300        let mut in_backtick = false;
301        let mut depth: i32 = 0;
302        let mut terminated = false;
303        while i < chars.len() {
304            let c = chars[i];
305            if c == '\n' {
306                line += 1;
307            }
308            if let Some(q) = in_string {
309                if c == '\\' {
310                    // Skip the escaped char (count its newline if any).
311                    i += 1;
312                    if i < chars.len() && chars[i] == '\n' {
313                        line += 1;
314                    }
315                    i += 1;
316                    continue;
317                }
318                if c == q {
319                    in_string = None;
320                }
321                i += 1;
322                continue;
323            }
324            if in_backtick {
325                if c == '`' {
326                    in_backtick = false;
327                }
328                i += 1;
329                continue;
330            }
331            match c {
332                '\'' | '"' => in_string = Some(c),
333                '`' => in_backtick = true,
334                '(' | '[' | '{' => depth += 1,
335                ')' | ']' | '}' => depth -= 1,
336                ';' if depth <= 0 => {
337                    let text: String = chars[start..i].iter().collect();
338                    out.push(Statement {
339                        line: start_line,
340                        text: text.trim().to_string(),
341                    });
342                    i += 1; // consume ';'
343                    terminated = true;
344                    break;
345                }
346                _ => {}
347            }
348            i += 1;
349        }
350        if !terminated {
351            // Reached EOF mid-statement. If there is any non-whitespace, the dump
352            // is truncated.
353            let rem: String = chars[start..i].iter().collect();
354            if !rem.trim().is_empty() {
355                return Err(format!(
356                    "truncated Cypher dump: statement beginning at line {start_line} \
357                     has no terminating ';'"
358                ));
359            }
360        }
361    }
362    Ok(out)
363}

367/// The kind of a top-level Cypher statement.
368enum StmtClass {
369    /// A transaction marker (`:begin`/`:commit`/`:rollback`) or apoc scaffolding
370    /// — ignored losslessly.
371    Ignore,
372    /// A node `CREATE`.
373    Node,
374    /// A relationship `MATCH ... CREATE`.
375    Edge,
376    /// An `UNWIND ... CREATE` batch (node or relationship, resolved on parse).
377    Unwind,
378    /// A supported-out-of-scope construct, reported as unsupported with a count.
379    Unsupported { label: String, kind: String },
380    /// An unrecognized statement — surfaced as a parse error per the failure mode.
381    Unknown,
382}

384/// Case-insensitive first keyword of a statement (uppercased), or `""`.
385fn leading_keyword(text: &str) -> String {
386    text.trim_start()
387        .chars()
388        .take_while(|c| c.is_ascii_alphabetic())
389        .collect::<String>()
390        .to_ascii_uppercase()
391}

393/// True if `kw` appears as a whole word (case-insensitive) outside strings and
394/// backtick identifiers.
395fn contains_keyword(text: &str, kw: &str) -> bool {
396    let chars: Vec<char> = text.chars().collect();
397    let kw_lower = kw.to_ascii_lowercase();
398    let mut in_string: Option<char> = None;
399    let mut in_backtick = false;
400    let mut i = 0usize;
401    while i < chars.len() {
402        let c = chars[i];
403        if let Some(q) = in_string {
404            if c == '\\' {
405                i += 2;
406                continue;
407            }
408            if c == q {
409                in_string = None;
410            }
411            i += 1;
412            continue;
413        }
414        if in_backtick {
415            if c == '`' {
416                in_backtick = false;
417            }
418            i += 1;
419            continue;
420        }
421        match c {
422            '\'' | '"' => {
423                in_string = Some(c);
424                i += 1;
425            }
426            '`' => {
427                in_backtick = true;
428                i += 1;
429            }
430            _ if is_ident_start(c) => {
431                let start = i;
432                while i < chars.len() && is_ident_continue(chars[i]) {
433                    i += 1;
434                }
435                let word: String = chars[start..i].iter().collect();
436                if word.to_ascii_lowercase() == kw_lower {
437                    return true;
438                }
439            }
440            _ => i += 1,
441        }
442    }
443    false
444}

446/// True if the statement contains a top-level relationship arrow (`-[`),
447/// outside strings/backticks.
448fn contains_rel_arrow(text: &str) -> bool {
449    let chars: Vec<char> = text.chars().collect();
450    let mut in_string: Option<char> = None;
451    let mut in_backtick = false;
452    let mut i = 0usize;
453    while i < chars.len() {
454        let c = chars[i];
455        if let Some(q) = in_string {
456            if c == '\\' {
457                i += 2;
458                continue;
459            }
460            if c == q {
461                in_string = None;
462            }
463            i += 1;
464            continue;
465        }
466        if in_backtick {
467            if c == '`' {
468                in_backtick = false;
469            }
470            i += 1;
471            continue;
472        }
473        match c {
474            '\'' | '"' => in_string = Some(c),
475            '`' => in_backtick = true,
476            '-' if i + 1 < chars.len() && chars[i + 1] == '[' => return true,
477            _ => {}
478        }
479        i += 1;
480    }
481    false
482}

484fn classify(text: &str) -> StmtClass {
485    if text.is_empty() {
486        return StmtClass::Ignore;
487    }
488    if text.starts_with(':') {
489        // :begin / :commit / :rollback (and any other psql-style meta command).
490        return StmtClass::Ignore;
491    }
492    let kw = leading_keyword(text);
493    let lower = text.to_ascii_lowercase();
494    let has_unique_import = lower.contains("unique import");
495
496    match kw.as_str() {
497        "CREATE" => {
498            if contains_keyword(text, "CONSTRAINT") {
499                if has_unique_import {
500                    return StmtClass::Ignore; // apoc scaffolding
501                }
502                return StmtClass::Unsupported {
503                    label: "CREATE CONSTRAINT".to_string(),
504                    kind: "constraint".to_string(),
505                };
506            }
507            if contains_keyword(text, "INDEX") {
508                if has_unique_import {
509                    return StmtClass::Ignore;
510                }
511                return StmtClass::Unsupported {
512                    label: "CREATE INDEX".to_string(),
513                    kind: "index".to_string(),
514                };
515            }
516            if contains_rel_arrow(text) {
517                // Inline `CREATE (a)-[:R]->(b)` (no MATCH) is not apoc's rel
518                // format; report rather than guess node identities.
519                return StmtClass::Unsupported {
520                    label: "CREATE (inline relationship)".to_string(),
521                    kind: "inline_relationship".to_string(),
522                };
523            }
524            StmtClass::Node
525        }
526        "MATCH" => {
527            if contains_keyword(text, "CREATE") {
528                StmtClass::Edge
529            } else if has_unique_import {
530                // apoc's `MATCH (n:`UNIQUE IMPORT LABEL`) ... REMOVE ...` cleanup.
531                StmtClass::Ignore
532            } else {
533                StmtClass::Unsupported {
534                    label: "MATCH (mutation)".to_string(),
535                    kind: "match_mutation".to_string(),
536                }
537            }
538        }
539        "UNWIND" => StmtClass::Unwind,
540        "DROP" => {
541            if has_unique_import {
542                StmtClass::Ignore
543            } else {
544                StmtClass::Unsupported {
545                    label: "DROP".to_string(),
546                    kind: "drop".to_string(),
547                }
548            }
549        }
550        "CALL" => {
551            if lower.contains("awaitindex") || lower.contains("await index") {
552                StmtClass::Ignore // apoc index-await scaffolding
553            } else {
554                StmtClass::Unsupported {
555                    label: "CALL".to_string(),
556                    kind: "procedure_call".to_string(),
557                }
558            }
559        }
560        "SCHEMA" => StmtClass::Ignore, // `SCHEMA AWAIT`
561        "MERGE" => StmtClass::Unsupported {
562            label: "MERGE".to_string(),
563            kind: "merge".to_string(),
564        },
565        _ => StmtClass::Unknown,
566    }
567}

571fn is_ident_start(c: char) -> bool {
572    c.is_ascii_alphabetic() || c == '_' || c == '$'
573}
574
575fn is_ident_continue(c: char) -> bool {
576    c.is_ascii_alphanumeric() || c == '_' || c == '$'
577}

579/// A char cursor over one statement's text.
580struct Cursor {
581    s: Vec<char>,
582    pos: usize,
583}
584
585impl Cursor {
586    fn new(text: &str) -> Self {
587        Cursor {
588            s: text.chars().collect(),
589            pos: 0,
590        }
591    }
592
593    fn peek(&self) -> Option<char> {
594        self.s.get(self.pos).copied()
595    }
596
597    fn bump(&mut self) -> Option<char> {
598        let c = self.s.get(self.pos).copied();
599        if c.is_some() {
600            self.pos += 1;
601        }
602        c
603    }
604
605    fn skip_ws(&mut self) {
606        while let Some(c) = self.peek() {
607            if c.is_whitespace() {
608                self.pos += 1;
609            } else {
610                break;
611            }
612        }
613    }
614
615    fn eof(&self) -> bool {
616        self.pos >= self.s.len()
617    }
618
619    fn expect(&mut self, c: char) -> std::result::Result<(), String> {
620        self.skip_ws();
621        match self.peek() {
622            Some(got) if got == c => {
623                self.pos += 1;
624                Ok(())
625            }
626            Some(got) => Err(format!("expected '{c}', found '{got}'")),
627            None => Err(format!("expected '{c}', found end of statement")),
628        }
629    }
630
631    /// Consume `kw` (case-insensitive) if it is the next whole word.
632    fn consume_keyword(&mut self, kw: &str) -> std::result::Result<(), String> {
633        self.skip_ws();
634        let kw_chars: Vec<char> = kw.chars().collect();
635        if self.pos + kw_chars.len() > self.s.len() {
636            return Err(format!("expected keyword '{kw}'"));
637        }
638        for (k, kc) in kw_chars.iter().enumerate() {
639            if !self.s[self.pos + k].eq_ignore_ascii_case(kc) {
640                return Err(format!("expected keyword '{kw}'"));
641            }
642        }
643        // Must be a word boundary.
644        let after = self.pos + kw_chars.len();
645        if let Some(&next) = self.s.get(after)
646            && is_ident_continue(next)
647        {
648            return Err(format!("expected keyword '{kw}'"));
649        }
650        self.pos = after;
651        Ok(())
652    }
653
654    /// Parse a bare or backtick-quoted identifier.
655    fn parse_ident(&mut self) -> std::result::Result<String, String> {
656        self.skip_ws();
657        match self.peek() {
658            Some('`') => {
659                self.pos += 1;
660                let mut out = String::new();
661                loop {
662                    match self.bump() {
663                        None => return Err("unterminated backtick identifier".to_string()),
664                        Some('`') => {
665                            // Doubled backtick escapes a literal backtick.
666                            if self.peek() == Some('`') {
667                                out.push('`');
668                                self.pos += 1;
669                            } else {
670                                break;
671                            }
672                        }
673                        Some(c) => out.push(c),
674                    }
675                }
676                Ok(out)
677            }
678            Some(c) if is_ident_start(c) => {
679                let start = self.pos;
680                while let Some(c) = self.peek() {
681                    if is_ident_continue(c) {
682                        self.pos += 1;
683                    } else {
684                        break;
685                    }
686                }
687                Ok(self.s[start..self.pos].iter().collect())
688            }
689            Some(c) => Err(format!("expected identifier, found '{c}'")),
690            None => Err("expected identifier, found end of statement".to_string()),
691        }
692    }
693}

697/// Maximum nesting depth for a property value (lists / maps). A dump with
698/// deeper balanced nesting is rejected with a typed [`RowError`] rather than
699/// recursing until the native stack overflows (SIGABRT) — a hostile or
700/// corrupt dump must never crash the process, only fail the statement (so
701/// `SkipAndReport` can carry on). Neo4j property values are primitives or flat
702/// arrays in practice, so 128 is far beyond any real dump.
703const MAX_VALUE_DEPTH: usize = 128;
704
705/// Error text for a value that nests past [`MAX_VALUE_DEPTH`].
706fn depth_exceeded() -> String {
707    format!("maximum value nesting depth ({MAX_VALUE_DEPTH}) exceeded")
708}

710/// The outcome of parsing one property value.
711enum PropOutcome {
712    /// A stored value.
713    Value(PropertyValue),
714    /// `null` or an unsupported value that was dropped (and already noted).
715    Absent,
716}

718/// Parse a Cypher string literal (the opening quote at the cursor).
719fn parse_string(cur: &mut Cursor) -> std::result::Result<String, String> {
720    let quote = cur.bump().ok_or("expected string quote")?;
721    let mut out = String::new();
722    loop {
723        match cur.bump() {
724            None => return Err("unterminated string literal".to_string()),
725            Some('\\') => {
726                let esc = cur.bump().ok_or("unterminated escape in string literal")?;
727                match esc {
728                    'n' => out.push('\n'),
729                    't' => out.push('\t'),
730                    'r' => out.push('\r'),
731                    'b' => out.push('\u{0008}'),
732                    'f' => out.push('\u{000C}'),
733                    '\\' => out.push('\\'),
734                    '\'' => out.push('\''),
735                    '"' => out.push('"'),
736                    '/' => out.push('/'),
737                    '`' => out.push('`'),
738                    'u' => {
739                        let mut hex = String::new();
740                        for _ in 0..4 {
741                            let h = cur.bump().ok_or("truncated \\u escape in string literal")?;
742                            hex.push(h);
743                        }
744                        let code = u32::from_str_radix(&hex, 16)
745                            .map_err(|_| format!("invalid \\u escape '\\u{hex}'"))?;
746                        let ch = char::from_u32(code)
747                            .ok_or_else(|| format!("invalid unicode code point U+{hex}"))?;
748                        out.push(ch);
749                    }
750                    other => return Err(format!("invalid escape '\\{other}' in string literal")),
751                }
752            }
753            Some(c) if c == quote => break,
754            Some(c) => out.push(c),
755        }
756    }
757    Ok(out)
758}

760/// Parse a numeric literal at the cursor into an `Int` or `Float` value.
761fn parse_number(cur: &mut Cursor) -> std::result::Result<PropertyValue, String> {
762    let start = cur.pos;
763    if cur.peek() == Some('-') || cur.peek() == Some('+') {
764        cur.pos += 1;
765    }
766    let mut is_float = false;
767    let mut saw_digit = false;
768    while let Some(c) = cur.peek() {
769        if c.is_ascii_digit() {
770            saw_digit = true;
771            cur.pos += 1;
772        } else {
773            break;
774        }
775    }
776    if cur.peek() == Some('.') {
777        is_float = true;
778        cur.pos += 1;
779        while let Some(c) = cur.peek() {
780            if c.is_ascii_digit() {
781                saw_digit = true;
782                cur.pos += 1;
783            } else {
784                break;
785            }
786        }
787    }
788    if matches!(cur.peek(), Some('e') | Some('E')) {
789        is_float = true;
790        cur.pos += 1;
791        if matches!(cur.peek(), Some('+') | Some('-')) {
792            cur.pos += 1;
793        }
794        while let Some(c) = cur.peek() {
795            if c.is_ascii_digit() {
796                cur.pos += 1;
797            } else {
798                break;
799            }
800        }
801    }
802    if !saw_digit {
803        return Err("malformed numeric literal".to_string());
804    }
805    let text: String = cur.s[start..cur.pos].iter().collect();
806    if is_float {
807        text.parse::<f64>()
808            .map(PropertyValue::Float)
809            .map_err(|_| format!("malformed float literal '{text}'"))
810    } else {
811        match text.parse::<i64>() {
812            Ok(v) => Ok(PropertyValue::Int(v)),
813            // An integer that overflows i64 promotes to Float rather than failing.
814            Err(_) => text
815                .parse::<f64>()
816                .map(PropertyValue::Float)
817                .map_err(|_| format!("malformed integer literal '{text}'")),
818        }
819    }
820}

822/// Consume a balanced parenthesized group (the opening `(` already consumed),
823/// respecting nested brackets and string literals. Used to skip over an
824/// unsupported constructor's arguments.
825fn skip_balanced_parens(cur: &mut Cursor) -> std::result::Result<(), String> {
826    let mut depth = 1i32;
827    while let Some(c) = cur.bump() {
828        match c {
829            '\'' | '"' => {
830                // Skip a string literal.
831                let quote = c;
832                loop {
833                    match cur.bump() {
834                        None => return Err("unterminated string in arguments".to_string()),
835                        Some('\\') => {
836                            cur.bump();
837                        }
838                        Some(x) if x == quote => break,
839                        _ => {}
840                    }
841                }
842            }
843            '(' | '[' | '{' => depth += 1,
844            ')' | ']' | '}' => {
845                depth -= 1;
846                if depth == 0 {
847                    return Ok(());
848                }
849            }
850            _ => {}
851        }
852    }
853    Err("unbalanced parentheses in arguments".to_string())
854}

856/// Parse a temporal / unsupported constructor call. The identifier `name` has
857/// been read; the cursor is at the `(`.
858fn parse_function(
859    cur: &mut Cursor,
860    name: &str,
861    prop_name: &str,
862    opts: &Neo4jCypherOptions,
863    note: &mut FidelityNote,
864) -> std::result::Result<PropOutcome, String> {
865    let lower = name.to_ascii_lowercase();
866    cur.expect('(')?;
867    match lower.as_str() {
868        "datetime" | "localdatetime" | "date" | "zoneddatetime" => {
869            cur.skip_ws();
870            let arg = match cur.peek() {
871                Some('\'') | Some('"') => parse_string(cur)?,
872                _ => {
873                    skip_balanced_parens(cur)?;
874                    return unsupported_value(prop_name, &lower, opts, note);
875                }
876            };
877            cur.expect(')')?;
878            let ts = parse::parse_valid_time(strip_zone_id(arg.trim()))
879                .map_err(|m| format!("temporal value '{arg}': {m}"))?;
880            note.note_coercion(prop_name, "temporal_to_micros");
881            Ok(PropOutcome::Value(PropertyValue::Int(ts.wallclock())))
882        }
883        "time" | "localtime" => {
884            cur.skip_ws();
885            let arg = match cur.peek() {
886                Some('\'') | Some('"') => parse_string(cur)?,
887                _ => {
888                    skip_balanced_parens(cur)?;
889                    return unsupported_value(prop_name, &lower, opts, note);
890                }
891            };
892            cur.expect(')')?;
893            note.note_coercion(prop_name, "time_to_string");
894            Ok(PropOutcome::Value(PropertyValue::string(arg)))
895        }
896        _ => {
897            // point(..), duration(..), or any unknown constructor / byte-array
898            // encoding: rejected + reported (never silently dropped).
899            skip_balanced_parens(cur)?;
900            unsupported_value(prop_name, &lower, opts, note)
901        }
902    }
903}

905/// Record (or, under strict mode, hard-error on) an unsupported value.
906fn unsupported_value(
907    prop_name: &str,
908    ty: &str,
909    opts: &Neo4jCypherOptions,
910    note: &mut FidelityNote,
911) -> std::result::Result<PropOutcome, String> {
912    if opts.strict_types {
913        return Err(format!(
914            "unsupported value type '{ty}' for property '{prop_name}' (strict-types)"
915        ));
916    }
917    note.note_unsupported(prop_name, ty);
918    Ok(PropOutcome::Absent)
919}

921/// Strip a trailing bracketed IANA zone id (`...+01:00[Europe/London]`).
922fn strip_zone_id(s: &str) -> &str {
923    if s.ends_with(']')
924        && let Some(open) = s.rfind('[')
925    {
926        return s[..open].trim_end();
927    }

931/// Parse one property value.
932fn parse_value(
933    cur: &mut Cursor,
934    prop_name: &str,
935    opts: &Neo4jCypherOptions,
936    note: &mut FidelityNote,
937    depth: usize,
938) -> std::result::Result<PropOutcome, String> {
939    if depth > MAX_VALUE_DEPTH {
940        return Err(depth_exceeded());
941    }
942    cur.skip_ws();
943    match cur.peek() {
944        None => Err("unexpected end of value".to_string()),
945        Some('\'') | Some('"') => Ok(PropOutcome::Value(PropertyValue::string(parse_string(
946            cur,
947        )?))),
948        Some('[') => parse_list_value(cur, prop_name, opts, depth + 1),
949        Some('{') => {
950            let json = parse_json_value(cur, opts, depth + 1)?;
951            note.note_coercion(prop_name, "map_to_json");
952            Ok(PropOutcome::Value(PropertyValue::string(json.to_string())))
953        }
954        Some(c) if c == '-' || c == '+' || c.is_ascii_digit() => {
955            Ok(PropOutcome::Value(parse_number(cur)?))
956        }
957        Some(c) if is_ident_start(c) || c == '`' => {
958            let ident = cur.parse_ident()?;
959            cur.skip_ws();
960            if cur.peek() == Some('(') {
961                return parse_function(cur, &ident, prop_name, opts, note);
962            }
963            match ident.to_ascii_lowercase().as_str() {
964                "true" => Ok(PropOutcome::Value(PropertyValue::Bool(true))),
965                "false" => Ok(PropOutcome::Value(PropertyValue::Bool(false))),
966                "null" => Ok(PropOutcome::Absent),
967                other => Err(format!("unexpected bare literal '{other}' in value")),
968            }
969        }
970        Some(c) => Err(format!("unexpected character '{c}' in value")),
971    }
972}

974/// Parse a `[..]` list, applying vector detection for opted-in properties.
975fn parse_list_value(
976    cur: &mut Cursor,
977    prop_name: &str,
978    opts: &Neo4jCypherOptions,
979    depth: usize,
980) -> std::result::Result<PropOutcome, String> {
981    if depth > MAX_VALUE_DEPTH {
982        return Err(depth_exceeded());
983    }
984    cur.expect('[')?;
985    let mut elems: Vec<serde_json::Value> = Vec::new();
986    cur.skip_ws();
987    if cur.peek() == Some(']') {
988        cur.pos += 1;
989    } else {
990        loop {
991            let v = parse_json_value(cur, opts, depth + 1)?;
992            elems.push(v);
993            cur.skip_ws();
994            match cur.peek() {
995                Some(',') => {
996                    cur.pos += 1;
997                }
998                Some(']') => {
999                    cur.pos += 1;
1000                    break;
1001                }
1002                Some(c) => return Err(format!("expected ',' or ']' in list, found '{c}'")),
1003                None => return Err("unterminated list".to_string()),
1004            }
1005        }
1006    }
1007
1008    let all_numeric =
1009        !elems.is_empty() && elems.iter().all(|v| v.is_number() && v.as_f64().is_some());
1010    if opts.vector_properties.contains(prop_name) && all_numeric {
1011        let floats: Vec<f32> = elems
1012            .iter()
1013            .filter_map(|v| v.as_f64())
1014            .map(|f| f as f32)
1015            .collect();
1016        let vec = PropertyValue::try_vector(&floats).map_err(|e| e.to_string())?;
1017        return Ok(PropOutcome::Value(vec));
1018    }
1019    let values: Vec<PropertyValue> = elems.iter().filter_map(json_to_property_value).collect();
1020    Ok(PropOutcome::Value(PropertyValue::array(values)))
1021}

1023/// Parse a Cypher value into a `serde_json::Value` (for nested maps, list
1024/// elements, and UNWIND rows). Temporal constructors keep their string
1025/// argument; unsupported constructors become JSON `null`.
1026fn parse_json_value(
1027    cur: &mut Cursor,
1028    opts: &Neo4jCypherOptions,
1029    depth: usize,
1030) -> std::result::Result<serde_json::Value, String> {
1031    if depth > MAX_VALUE_DEPTH {
1032        return Err(depth_exceeded());
1033    }
1034    cur.skip_ws();
1035    match cur.peek() {
1036        None => Err("unexpected end of value".to_string()),
1037        Some('\'') | Some('"') => Ok(serde_json::Value::String(parse_string(cur)?)),
1038        Some('{') => parse_json_map(cur, opts, depth + 1),
1039        Some('[') => {
1040            cur.pos += 1;
1041            let mut arr = Vec::new();
1042            cur.skip_ws();
1043            if cur.peek() == Some(']') {
1044                cur.pos += 1;
1045                return Ok(serde_json::Value::Array(arr));
1046            }
1047            loop {
1048                arr.push(parse_json_value(cur, opts, depth + 1)?);
1049                cur.skip_ws();
1050                match cur.peek() {
1051                    Some(',') => cur.pos += 1,
1052                    Some(']') => {
1053                        cur.pos += 1;
1054                        break;
1055                    }
1056                    Some(c) => return Err(format!("expected ',' or ']', found '{c}'")),
1057                    None => return Err("unterminated list".to_string()),
1058                }
1059            }
1060            Ok(serde_json::Value::Array(arr))
1061        }
1062        Some(c) if c == '-' || c == '+' || c.is_ascii_digit() => match parse_number(cur)? {
1063            PropertyValue::Int(i) => Ok(serde_json::Value::from(i)),
1064            PropertyValue::Float(f) => Ok(serde_json::Number::from_f64(f)
1065                .map(serde_json::Value::Number)
1066                .unwrap_or(serde_json::Value::Null)),
1067            _ => Ok(serde_json::Value::Null),
1068        },
1069        Some(c) if is_ident_start(c) || c == '`' => {
1070            let ident = cur.parse_ident()?;
1071            cur.skip_ws();
1072            if cur.peek() == Some('(') {
1073                let lower = ident.to_ascii_lowercase();
1074                cur.expect('(')?;
1075                match lower.as_str() {
1076                    "datetime" | "localdatetime" | "date" | "zoneddatetime" | "time"
1077                    | "localtime" => {
1078                        cur.skip_ws();
1079                        let arg = match cur.peek() {
1080                            Some('\'') | Some('"') => parse_string(cur)?,
1081                            _ => {
1082                                skip_balanced_parens(cur)?;
1083                                return Ok(serde_json::Value::Null);
1084                            }
1085                        };
1086                        cur.expect(')')?;
1087                        Ok(serde_json::Value::String(arg))
1088                    }
1089                    _ => {
1090                        skip_balanced_parens(cur)?;
1091                        Ok(serde_json::Value::Null)
1092                    }
1093                }
1094            } else {
1095                match ident.to_ascii_lowercase().as_str() {
1096                    "true" => Ok(serde_json::Value::Bool(true)),
1097                    "false" => Ok(serde_json::Value::Bool(false)),
1098                    "null" => Ok(serde_json::Value::Null),
1099                    other => Err(format!("unexpected bare literal '{other}'")),
1100                }
1101            }
1102        }
1103        Some(c) => Err(format!("unexpected character '{c}' in value")),
1104    }
1105}

1107/// Parse a `{ key: value, .. }` map into a `serde_json::Value::Object`.
1108fn parse_json_map(
1109    cur: &mut Cursor,
1110    opts: &Neo4jCypherOptions,
1111    depth: usize,
1112) -> std::result::Result<serde_json::Value, String> {
1113    if depth > MAX_VALUE_DEPTH {
1114        return Err(depth_exceeded());
1115    }
1116    cur.expect('{')?;
1117    let mut map = serde_json::Map::new();
1118    cur.skip_ws();
1119    if cur.peek() == Some('}') {
1120        cur.pos += 1;
1121        return Ok(serde_json::Value::Object(map));
1122    }
1123    loop {
1124        let key = cur.parse_ident()?;
1125        cur.expect(':')?;
1126        let val = parse_json_value(cur, opts, depth + 1)?;
1127        map.insert(key, val);
1128        cur.skip_ws();
1129        match cur.peek() {
1130            Some(',') => cur.pos += 1,
1131            Some('}') => {
1132                cur.pos += 1;
1133                break;
1134            }
1135            Some(c) => return Err(format!("expected ',' or '}}' in map, found '{c}'")),
1136            None => return Err("unterminated map".to_string()),
1137        }
1138    }
1139    Ok(serde_json::Value::Object(map))
1140}

1142/// Convert a `serde_json::Value` scalar/array to a `PropertyValue`
1143/// (`None` for JSON null). Nested objects/arrays become a JSON string / array.
1144fn json_to_property_value(v: &serde_json::Value) -> Option<PropertyValue> {
1145    match v {
1146        serde_json::Value::Null => None,
1147        serde_json::Value::Bool(b) => Some(PropertyValue::Bool(*b)),
1148        serde_json::Value::Number(n) => {
1149            if let Some(i) = n.as_i64() {
1150                Some(PropertyValue::Int(i))
1151            } else {
1152                n.as_f64().map(PropertyValue::Float)
1153            }
1154        }
1155        serde_json::Value::String(s) => Some(PropertyValue::string(s)),
1156        serde_json::Value::Array(items) => Some(PropertyValue::array(
1157            items.iter().filter_map(json_to_property_value).collect(),
1158        )),
1159        serde_json::Value::Object(_) => Some(PropertyValue::string(v.to_string())),
1160    }
1161}

1163/// A property (as a JSON value) destined for the property bag, with vector
1164/// detection for opted-in names.
1165fn json_property_to_value(
1166    name: &str,
1167    v: &serde_json::Value,
1168    opts: &Neo4jCypherOptions,
1169    note: &mut FidelityNote,
1170) -> Option<PropertyValue> {
1171    if let serde_json::Value::Array(items) = v
1172        && opts.vector_properties.contains(name)
1173        && !items.is_empty()
1174        && items.iter().all(|e| e.is_number() && e.as_f64().is_some())
1175    {
1176        let floats: Vec<f32> = items
1177            .iter()
1178            .filter_map(|e| e.as_f64())
1179            .map(|f| f as f32)
1180            .collect();
1181        return PropertyValue::try_vector(&floats).ok();
1182    }
1183    if matches!(v, serde_json::Value::Object(_)) {
1184        note.note_coercion(name, "map_to_json");
1185    }
1186    json_to_property_value(v)
1187}

1213/// A parsed node pattern `( [var] (:Label)* {props}? )`.
1214struct NodePattern {
1215    var: Option<String>,
1216    labels: Vec<String>,
1217    props: Vec<(String, PropOutcome)>,
1218}
1219
1220/// Parse a node pattern at the cursor (the `(` is the next non-ws char).
1221fn parse_node_pattern(
1222    cur: &mut Cursor,
1223    opts: &Neo4jCypherOptions,
1224    note: &mut FidelityNote,
1225) -> std::result::Result<NodePattern, String> {
1226    cur.expect('(')?;
1227    cur.skip_ws();
1228    let mut var = None;
1229    // Optional variable: a bare identifier not introducing a label/prop/close.
1230    if let Some(c) = cur.peek()
1231        && c != ':'
1232        && c != '{'
1233        && c != ')'
1234    {
1235        var = Some(cur.parse_ident()?);
1236    }
1237    let mut labels = Vec::new();
1238    loop {
1239        cur.skip_ws();
1240        if cur.peek() == Some(':') {
1241            cur.pos += 1;
1242            labels.push(cur.parse_ident()?);
1243        } else {
1244            break;
1245        }
1246    }
1247    cur.skip_ws();
1248    let mut props = Vec::new();
1249    if cur.peek() == Some('{') {
1250        props = parse_prop_map(cur, opts, note)?;
1251    }
1252    cur.expect(')')?;
1253    Ok(NodePattern { var, labels, props })
1254}
1255
1256/// Parse a `{ key: value, .. }` property bag for a node/relationship.
1257fn parse_prop_map(
1258    cur: &mut Cursor,
1259    opts: &Neo4jCypherOptions,
1260    note: &mut FidelityNote,
1261) -> std::result::Result<Vec<(String, PropOutcome)>, String> {
1262    cur.expect('{')?;
1263    let mut out = Vec::new();
1264    cur.skip_ws();
1265    if cur.peek() == Some('}') {
1266        cur.pos += 1;
1267        return Ok(out);
1268    }
1269    loop {
1270        let key = cur.parse_ident()?;
1271        cur.expect(':')?;
1272        let val = parse_value(cur, &key, opts, note, 0)?;
1273        out.push((key, val));
1274        cur.skip_ws();
1275        match cur.peek() {
1276            Some(',') => cur.pos += 1,
1277            Some('}') => {
1278                cur.pos += 1;
1279                break;
1280            }
1281            Some(c) => return Err(format!("expected ',' or '}}' in property map, found '{c}'")),
1282            None => return Err("unterminated property map".to_string()),
1283        }
1284    }
1285    Ok(out)
1286}

1288/// Build a [`PreparedNode`] (and finish its fidelity note) from a parsed node
1289/// pattern. `seq` disambiguates the synthetic key of a keyless node.
1290fn build_node(
1291    pattern: NodePattern,
1292    opts: &Neo4jCypherOptions,
1293    seq: usize,
1294) -> std::result::Result<NodeUnit, String> {
1295    let mut note = FidelityNote::default();
1296
1297    let real_labels: Vec<String> = pattern
1298        .labels
1299        .iter()
1300        .filter(|l| *l != &opts.unique_import_label)
1301        .cloned()
1302        .collect();
1303
1304    let (chosen_label, neo4j_labels) = if real_labels.is_empty() {
1305        note.note_coercion(":LABEL", "default_label");
1306        (opts.default_label.clone(), "(none)".to_string())
1307    } else {
1308        let chosen = match opts.label_strategy {
1309            LabelStrategy::Concat => real_labels.join("_"),
1310            LabelStrategy::First | LabelStrategy::Property => real_labels[0].clone(),
1311        };
1312        (chosen, real_labels.join(":"))
1313    };
1314    note.label = Some((neo4j_labels, chosen_label.clone()));
1315    if real_labels.len() > 1 {
1316        note.note_coercion(":LABEL", "multi_label_flattened");
1317    }
1318
1319    let mut builder = PropertyMapBuilder::new();
1320    let mut prop_count = 0usize;
1321
1322    // Preserve the full label set as `_labels` (First: only when multi-label;
1323    // Property: always; Concat: never).
1324    let store_labels = match opts.label_strategy {
1325        LabelStrategy::First => real_labels.len() > 1,
1326        LabelStrategy::Property => !real_labels.is_empty(),
1327        LabelStrategy::Concat => false,
1328    };
1329    if store_labels {
1330        let arr = PropertyValue::array(real_labels.iter().map(PropertyValue::string).collect());
1331        builder = builder
1332            .try_insert(&opts.retained_labels_key, arr)
1333            .map_err(|e| e.to_string())?;
1334        prop_count += 1;
1335    }
1336
1337    let mut key: Option<String> = None;
1338    let mut valid_from: Option<Timestamp> = None;
1339    for (name, outcome) in pattern.props {
1340        if name == opts.key_property {
1341            if let PropOutcome::Value(v) = &outcome {
1342                key = Some(canonical_key(v));
1343            }
1344            continue; // consumed as the business key, not stored
1345        }
1346        if opts.valid_from_property.as_deref() == Some(name.as_str()) {
1347            valid_from = extract_valid_from(&outcome, &name)?;
1348            continue;
1349        }
1350        if let PropOutcome::Value(v) = outcome {
1351            builder = builder.try_insert(&name, v).map_err(|e| e.to_string())?;
1352            prop_count += 1;
1353        }
1354    }
1355
1356    note.prop_count = prop_count;
1357    let key = key.unwrap_or_else(|| format!("\0__no_key_{seq}"));
1358    Ok(NodeUnit {
1359        node: PreparedNode {
1360            key,
1361            label: chosen_label,
1362            properties: builder.build(),
1363            valid_from,
1364        },
1365        note,
1366    })
1367}

1369/// Extract a `valid_from` timestamp from a designated property's value.
1370fn extract_valid_from(
1371    outcome: &PropOutcome,
1372    name: &str,
1373) -> std::result::Result<Option<Timestamp>, String> {
1374    match outcome {
1375        PropOutcome::Value(PropertyValue::Int(micros)) => Ok(Some(Timestamp::from(*micros))),
1376        PropOutcome::Value(PropertyValue::String(s)) => parse::parse_valid_time(s)
1377            .map(Some)
1378            .map_err(|m| format!("valid-from property '{name}': {m}")),
1379        PropOutcome::Absent => Ok(None),
1380        PropOutcome::Value(other) => Err(format!(
1381            "valid-from property '{name}' is not a temporal value ({})",
1382            other.type_name()
1383        )),
1384    }
1385}

1387/// Resolve the business key referenced by a relationship-match property bag:
1388/// the configured `key_property` if present, else the sole property.
1389fn match_key(
1390    props: &[(String, PropOutcome)],
1391    opts: &Neo4jCypherOptions,
1392) -> std::result::Result<String, String> {
1393    if let Some((_, PropOutcome::Value(v))) = props.iter().find(|(k, _)| k == &opts.key_property) {
1394        return Ok(canonical_key(v));
1395    }
1396    if props.len() == 1
1397        && let (_, PropOutcome::Value(v)) = &props[0]
1398    {
1399        return Ok(canonical_key(v));
1400    }
1401    Err(format!(
1402        "cannot determine endpoint key: no '{}' property and not exactly one match property",
1403        opts.key_property
1404    ))
1405}

1407/// Parse a `MATCH (a {..}), (b {..}) CREATE (a)-[:TYPE {..}]->(b)` statement.
1408fn parse_edge(
1409    text: &str,
1410    line: usize,
1411    opts: &Neo4jCypherOptions,
1412) -> std::result::Result<EdgeUnit, String> {
1413    let mut cur = Cursor::new(text);
1414    let mut note = FidelityNote::default();
1415
1416    // MATCH clauses: one or more node patterns, separated by ',' or repeated
1417    // MATCH keywords.
1418    let mut match_vars: HashMap<String, Vec<(String, PropOutcome)>> = HashMap::new();
1419    cur.consume_keyword("MATCH")?;
1420    loop {
1421        let pat = parse_node_pattern(&mut cur, opts, &mut note)?;
1422        if let Some(var) = pat.var {
1423            match_vars.insert(var, pat.props);
1424        }
1425        cur.skip_ws();
1426        if cur.peek() == Some(',') {
1427            cur.pos += 1;
1428            continue;
1429        }
1430        if cur.consume_keyword("MATCH").is_ok() {
1431            continue;
1432        }
1433        break;
1434    }
1435
1436    cur.consume_keyword("CREATE")?;
1437    // Relationship path: (a) <-/- [ rel ] -/-> (b)
1438    cur.expect('(')?;
1439    let a_var = cur.parse_ident()?;
1440    cur.expect(')')?;
1441    cur.skip_ws();
1442    let left = cur.peek() == Some('<');
1443    if left {
1444        cur.pos += 1;
1445    }
1446    cur.expect('-')?;
1447    cur.expect('[')?;
1448    // Optional relationship variable before the ':'.
1449    cur.skip_ws();
1450    if let Some(c) = cur.peek()
1451        && c != ':'
1452    {
1453        let _relvar = cur.parse_ident()?;
1454    }
1455    cur.expect(':')?;
1456    let rel_type = cur.parse_ident()?;
1457    cur.skip_ws();
1458    let mut rel_props = Vec::new();
1459    if cur.peek() == Some('{') {
1460        rel_props = parse_prop_map(&mut cur, opts, &mut note)?;
1461    }
1462    cur.expect(']')?;
1463    cur.expect('-')?;
1464    cur.skip_ws();
1465    let right = cur.peek() == Some('>');
1466    if right {
1467        cur.pos += 1;
1468    }
1469    cur.expect('(')?;
1470    let b_var = cur.parse_ident()?;
1471    cur.expect(')')?;
1472
1473    // Direction: default a -> b; `<-...-` reverses.
1474    let (src_var, dst_var) = if left && !right {
1475        (b_var, a_var)
1476    } else {
1477        (a_var, b_var)
1478    };
1479
1480    let src_props = match_vars
1481        .get(&src_var)
1482        .ok_or_else(|| format!("relationship references unmatched variable '{src_var}'"))?;
1483    let dst_props = match_vars
1484        .get(&dst_var)
1485        .ok_or_else(|| format!("relationship references unmatched variable '{dst_var}'"))?;
1486    let source_key = match_key(src_props, opts)?;
1487    let target_key = match_key(dst_props, opts)?;
1488
1489    note.edge_type = Some(rel_type.clone());
1490
1491    // Build the relationship property map.
1492    let mut builder = PropertyMapBuilder::new();
1493    let mut prop_count = 0usize;
1494    let mut valid_from = None;
1495    for (name, outcome) in rel_props {
1496        if opts.valid_from_property.as_deref() == Some(name.as_str()) {
1497            valid_from = extract_valid_from(&outcome, &name)?;
1498            continue;
1499        }
1500        if let PropOutcome::Value(v) = outcome {
1501            builder = builder.try_insert(&name, v).map_err(|e| e.to_string())?;
1502            prop_count += 1;
1503        }
1504    }
1505    note.prop_count = prop_count;
1506
1507    Ok(EdgeUnit {
1508        source_key,
1509        target_key,
1510        label: rel_type,
1511        properties: builder.build(),
1512        valid_from,
1513        note,
1514        line,
1515    })
1516}

1520/// Parse an `UNWIND [ .. ] AS row ...` batch into node and/or relationship units.
1521fn parse_unwind(
1522    text: &str,
1523    line: usize,
1524    opts: &Neo4jCypherOptions,
1525    seq_start: usize,
1526) -> std::result::Result<(Vec<NodeUnit>, Vec<EdgeUnit>), String> {
1527    let mut cur = Cursor::new(text);
1528    cur.consume_keyword("UNWIND")?;
1529    cur.skip_ws();
1530    let rows = parse_json_value(&mut cur, opts, 0)?;
1531    let serde_json::Value::Array(rows) = rows else {
1532        return Err("UNWIND expects a list literal".to_string());
1533    };
1534    cur.consume_keyword("AS")?;
1535    let _row_var = cur.parse_ident()?;
1536
1537    // Remaining clause decides node vs relationship batch.
1538    let remainder: String = cur.s[cur.pos..].iter().collect();
1539    let is_rel = contains_keyword(&remainder, "MATCH");
1540
1541    if is_rel {
1542        let rel_type = extract_unwind_rel_type(&remainder)?;
1543        let mut edges = Vec::new();
1544        for (i, row) in rows.iter().enumerate() {
1545            let obj = row
1546                .as_object()
1547                .ok_or_else(|| format!("UNWIND row {i} is not a map"))?;
1548            let start_id = obj
1549                .get("start")
1550                .and_then(|s| s.get("_id"))
1551                .and_then(canonical_key_json)
1552                .ok_or_else(|| format!("UNWIND relationship row {i} lacks start._id"))?;
1553            let end_id = obj
1554                .get("end")
1555                .and_then(|e| e.get("_id"))
1556                .and_then(canonical_key_json)
1557                .ok_or_else(|| format!("UNWIND relationship row {i} lacks end._id"))?;
1558            let mut note = FidelityNote {
1559                edge_type: Some(rel_type.clone()),
1560                ..Default::default()
1561            };
1562            let (properties, valid_from, prop_count) =
1563                unwind_properties(obj.get("properties"), opts, &mut note)?;
1564            note.prop_count = prop_count;
1565            edges.push(EdgeUnit {
1566                source_key: start_id,
1567                target_key: end_id,
1568                label: rel_type.clone(),
1569                properties,
1570                valid_from,
1571                note,
1572                line,
1573            });
1574        }
1575        Ok((Vec::new(), edges))
1576    } else {
1577        let labels = extract_unwind_node_labels(&remainder)?;
1578        let real_labels: Vec<String> = labels
1579            .iter()
1580            .filter(|l| *l != &opts.unique_import_label)
1581            .cloned()
1582            .collect();
1583        let mut nodes = Vec::new();
1584        for (i, row) in rows.iter().enumerate() {
1585            let obj = row
1586                .as_object()
1587                .ok_or_else(|| format!("UNWIND row {i} is not a map"))?;
1588            let key = obj
1589                .get("_id")
1590                .and_then(canonical_key_json)
1591                .unwrap_or_else(|| format!("\0__no_key_{}", seq_start + i));
1592            let unit = build_unwind_node(&real_labels, obj.get("properties"), key, opts)?;
1593            nodes.push(unit);
1594        }
1595        Ok((nodes, Vec::new()))
1596    }
1597}

1599/// Extract the relationship TYPE from an UNWIND's trailing `CREATE (a)-[:TYPE]->(b)`.
1600fn extract_unwind_rel_type(remainder: &str) -> std::result::Result<String, String> {
1601    let mut cur = Cursor::new(remainder);
1602    // Skip to CREATE.
1603    while !cur.eof() {
1604        if cur.consume_keyword("CREATE").is_ok() {
1605            break;
1606        }
1607        cur.pos += 1;
1608    }
1609    if cur.eof() {
1610        return Err("UNWIND relationship batch has no CREATE".to_string());
1611    }
1612    cur.expect('(')?;
1613    let _a = cur.parse_ident()?;
1614    cur.expect(')')?;
1615    cur.skip_ws();
1616    if cur.peek() == Some('<') {
1617        cur.pos += 1;
1618    }
1619    cur.expect('-')?;
1620    cur.expect('[')?;
1621    cur.skip_ws();
1622    if let Some(c) = cur.peek()
1623        && c != ':'
1624    {
1625        let _relvar = cur.parse_ident()?;
1626    }
1627    cur.expect(':')?;
1628    cur.parse_ident()
1629}

1631/// Extract the node labels from an UNWIND's trailing `CREATE (n:Labels ..)`.
1632fn extract_unwind_node_labels(remainder: &str) -> std::result::Result<Vec<String>, String> {
1633    let mut cur = Cursor::new(remainder);
1634    while !cur.eof() {
1635        if cur.consume_keyword("CREATE").is_ok() {
1636            break;
1637        }
1638        cur.pos += 1;
1639    }
1640    if cur.eof() {
1641        return Err("UNWIND node batch has no CREATE".to_string());
1642    }
1643    cur.expect('(')?;
1644    cur.skip_ws();
1645    // Optional variable.
1646    if let Some(c) = cur.peek()
1647        && c != ':'
1648        && c != '{'
1649        && c != ')'
1650    {
1651        let _var = cur.parse_ident()?;
1652    }
1653    let mut labels = Vec::new();
1654    loop {
1655        cur.skip_ws();
1656        if cur.peek() == Some(':') {
1657            cur.pos += 1;
1658            labels.push(cur.parse_ident()?);
1659        } else {
1660            break;
1661        }
1662    }
1663    if labels.is_empty() {
1664        return Err("UNWIND node batch CREATE has no label".to_string());
1665    }
1666    Ok(labels)
1667}

1681/// Insert an UNWIND row's `properties` object into an existing builder,
1682/// consuming the designated `valid_from` property.
1683fn insert_unwind_properties(
1684    mut builder: PropertyMapBuilder,
1685    properties: Option<&serde_json::Value>,
1686    opts: &Neo4jCypherOptions,
1687    note: &mut FidelityNote,
1688) -> std::result::Result<(PropertyMapBuilder, Option<Timestamp>, usize), String> {
1689    let mut count = 0usize;
1690    let mut valid_from = None;
1691    if let Some(serde_json::Value::Object(map)) = properties {
1692        for (name, val) in map {
1693            if opts.valid_from_property.as_deref() == Some(name.as_str()) {
1694                valid_from = json_valid_from(val, name)?;
1695                continue;
1696            }
1697            if let Some(v) = json_property_to_value(name, val, opts, note) {
1698                builder = builder.try_insert(name, v).map_err(|e| e.to_string())?;
1699                count += 1;
1700            }
1701        }
1702    }
1703    Ok((builder, valid_from, count))
1704}

1757/// Extract a `valid_from` timestamp from a JSON property value.
1758fn json_valid_from(
1759    val: &serde_json::Value,
1760    name: &str,
1761) -> std::result::Result<Option<Timestamp>, String> {
1762    match val {
1763        serde_json::Value::Null => Ok(None),
1764        serde_json::Value::Number(n) => match n.as_i64() {
1765            Some(micros) => Ok(Some(Timestamp::from(micros))),

1767                "valid-from property '{name}' is not a temporal value ({n})"

1771            .map(Some)
1772            .map_err(|m| format!("valid-from property '{name}': {m}")),
1773        other => Err(format!(
1774            "valid-from property '{name}' is not a temporal value ({other})"
1775        )),
1776    }

1781impl Importer<'_> {
1782    /// Import a Neo4j **APOC Cypher-script dump** (`apoc.export.cypher.all`
1783    /// `create`-format output) from a single file (Issue #3356).
1784    ///
1785    /// Parses the whole dump, streams node `CREATE`s and relationship
1786    /// `MATCH ... CREATE`s through the shared chunked-commit spine (nodes first,
1787    /// so relationship endpoints resolve), stamps `neo4j-import::<file>`
1788    /// provenance on every version, and returns a machine-readable fidelity
1789    /// report (counts, label/type mapping tables, coercion and unsupported-
1790    /// construct notes, and a `zero_loss` flag).
1791    ///
1792    /// # Errors
1793    ///
1794    /// Returns an error if the file cannot be read; the dump is truncated
1795    /// (a statement with no terminating `;`); or — in the default
1796    /// [`FailureMode::Abort`](super::FailureMode::Abort) — the first malformed
1797    /// statement, unsupported value under `strict_types`, or unresolved
1798    /// relationship endpoint. In
1799    /// [`SkipAndReport`](super::FailureMode::SkipAndReport) mode those
1800    /// per-statement failures are collected in the report instead, each carrying
1801    /// the source file and 1-based line number.
1802    pub fn neo4j_import_cypher(
1803        &mut self,
1804        path: impl AsRef<Path>,
1805        opts: &Neo4jCypherOptions,
1806    ) -> Result<Neo4jFidelityReport> {
1807        let path = path.as_ref();
1808        let src = std::fs::read_to_string(path)
1809            .map_err(|e| Error::other(format!("opening {}: {e}", path.display())))?;
1810        let file_label = file_label_for(path);
1811
1812        let statements = split_statements(&src).map_err(Error::other)?;
1813        let mut report = Neo4jFidelityReport::default();
1814
1815        // --- Parse phase: classify + parse every statement, honoring the
1816        // failure mode with per-statement line locations. ---
1817        let mut node_units: Vec<NodeUnit> = Vec::new();
1818        let mut edge_units: Vec<EdgeUnit> = Vec::new();
1819        let mut rows_read = 0usize;
1820
1821        for stmt in statements {
1822            match classify(&stmt.text) {
1823                StmtClass::Ignore => {}
1824                StmtClass::Unsupported { label, kind } => {
1825                    report.note_unsupported(&label, &kind);
1826                }
1827                StmtClass::Node => {
1828                    rows_read += 1;
1829                    let seq = node_units.len();
1830                    match build_node_stmt(&stmt.text, opts, seq) {
1831                        Ok(unit) => node_units.push(unit),
1832                        Err(msg) => {
1833                            self.record_parse_failure(stmt.line, msg, &file_label, &mut report)?;
1834                        }
1835                    }
1836                }
1837                StmtClass::Edge => {
1838                    rows_read += 1;
1839                    match parse_edge(&stmt.text, stmt.line, opts) {
1840                        Ok(unit) => edge_units.push(unit),
1841                        Err(msg) => {
1842                            self.record_parse_failure(stmt.line, msg, &file_label, &mut report)?;
1843                        }
1844                    }
1845                }
1846                StmtClass::Unwind => {
1847                    let seq = node_units.len();
1848                    match parse_unwind(&stmt.text, stmt.line, opts, seq) {
1849                        Ok((mut ns, mut es)) => {
1850                            rows_read += ns.len() + es.len();
1851                            node_units.append(&mut ns);
1852                            edge_units.append(&mut es);
1853                        }
1854                        Err(msg) => {
1855                            rows_read += 1;
1856                            self.record_parse_failure(stmt.line, msg, &file_label, &mut report)?;
1857                        }
1858                    }
1859                }
1860                StmtClass::Unknown => {
1861                    rows_read += 1;
1862                    let kw = leading_keyword(&stmt.text);
1863                    self.record_parse_failure(
1864                        stmt.line,
1865                        format!("unrecognized statement (leading keyword '{kw}')"),
1866                        &file_label,
1867                        &mut report,
1868                    )?;
1869                }
1870            }
1871        }
1872
1873        // --- Commit phase: nodes first (build the key map), then edges. ---
1874        self.commit_nodes(node_units, path, &mut report)?;
1875        self.commit_edges(edge_units, path, &file_label, &mut report)?;
1876
1877        report.rows_read = rows_read;
1878        report.finalize();
1879        Ok(report)
1880    }
1881
1882    /// Apply the failure mode to a per-statement parse failure.
1883    fn record_parse_failure(
1884        &self,
1885        line: usize,
1886        message: String,
1887        file_label: &str,
1888        report: &mut Neo4jFidelityReport,
1889    ) -> Result<()> {
1890        let row_err = RowError {
1891            row: line,
1892            message,
1893            file: Some(file_label.to_string()),
1894        };
1895        match self.config.failure_mode {
1896            super::FailureMode::Abort => Err(super::ImportError::Row(row_err).into()),
1897            super::FailureMode::SkipAndReport => {
1898                report.skipped.push(row_err);
1899                Ok(())
1900            }
1901        }
1902    }
1903
1904    /// Commit parsed node units through the shared node-import spine.
1905    fn commit_nodes(
1906        &mut self,
1907        units: Vec<NodeUnit>,
1908        path: &Path,
1909        report: &mut Neo4jFidelityReport,
1910    ) -> Result<()> {
1911        if units.is_empty() {
1912            return Ok(());
1913        }
1914        let count = units.len();
1915        let mut iter = units.into_iter();
1916        let prev = self.config.provenance.take();
1917        self.config.provenance = Some(provenance_for(path)?);
1918
1919        let rows: parse::RowIter = Box::new((0..count).map(|_| Ok(Row::new(1, HashMap::new()))));
1920        let result = self.import_nodes_with(rows, |_row| {
1921            let unit = iter.next().ok_or_else(|| RowError {
1922                row: 0,
1923                message: "internal invariant: node unit iterator exhausted before rows".to_string(),
1924                file: None,
1925            })?;
1926            fold_note(report, &unit.note);
1927            Ok(unit.node)
1928        });
1929        self.config.provenance = prev;
1930        let import_report = result?;
1931        report.nodes_imported += import_report.nodes_imported;
1932        Ok(())
1933    }
1934
1935    /// Commit parsed relationship units through the shared edge-import spine.
1936    fn commit_edges(
1937        &mut self,
1938        units: Vec<EdgeUnit>,
1939        path: &Path,
1940        file_label: &str,
1941        report: &mut Neo4jFidelityReport,
1942    ) -> Result<()> {
1943        if units.is_empty() {
1944            return Ok(());
1945        }
1946        let count = units.len();
1947        let mut iter = units.into_iter();
1948        let prev = self.config.provenance.take();
1949        self.config.provenance = Some(provenance_for(path)?);
1950        let file = file_label.to_string();
1951
1952        let rows: parse::RowIter = Box::new((0..count).map(|_| Ok(Row::new(1, HashMap::new()))));
1953        let result = self.import_edges_with(rows, |key_to_id, _row| {
1954            let unit = iter.next().ok_or_else(|| {
1955                EdgePrepError::Row(RowError {
1956                    row: 0,
1957                    message: "internal invariant: edge unit iterator exhausted before rows"
1958                        .to_string(),
1959                    file: None,
1960                })
1961            })?;
1962            let source = key_to_id.get(&unit.source_key).copied().ok_or_else(|| {
1963                EdgePrepError::Unresolved(UnresolvedEndpoint {
1964                    row: unit.line,
1965                    key: unit.source_key.clone(),
1966                    side: Endpoint::Source,
1967                    file: Some(file.clone()),
1968                })
1969            })?;
1970            let target = key_to_id.get(&unit.target_key).copied().ok_or_else(|| {
1971                EdgePrepError::Unresolved(UnresolvedEndpoint {
1972                    row: unit.line,
1973                    key: unit.target_key.clone(),
1974                    side: Endpoint::Target,
1975                    file: Some(file.clone()),
1976                })
1977            })?;
1978            fold_note(report, &unit.note);
1979            Ok(PreparedEdge {
1980                source,
1981                target,
1982                label: unit.label,
1983                properties: unit.properties,
1984                valid_from: unit.valid_from,
1985            })
1986        });
1987        self.config.provenance = prev;
1988        let import_report = result?;
1989        report.relationships_imported += import_report.edges_imported;
1990        report
1991            .unresolved_endpoints
1992            .extend(import_report.unresolved_endpoints);
1993        Ok(())
1994    }
1995}

1997/// Parse a single node `CREATE` statement into a [`NodeUnit`].
1998fn build_node_stmt(
1999    text: &str,
2000    opts: &Neo4jCypherOptions,
2001    seq: usize,
2002) -> std::result::Result<NodeUnit, String> {
2003    let mut cur = Cursor::new(text);
2004    let mut note = FidelityNote::default();
2005    cur.consume_keyword("CREATE")?;
2006    let pattern = parse_node_pattern(&mut cur, opts, &mut note)?;
2007    // Merge any coercions recorded during pattern parsing (values) into the node.
2008    let mut unit = build_node(pattern, opts, seq)?;
2009    // Fold value-level coercions/unsupported captured on the pattern note. A
2010    // coercion on the property consumed as `valid_from` is suppressed: that
2011    // value is folded into the version's valid-time interval, not stored, so
2012    // reporting it as a property coercion would be misleading.
2013    for (c, k) in note.coercions {
2014        if opts.valid_from_property.as_deref() == Some(c.as_str()) {
2015            continue;
2016        }
2017        unit.note.note_coercion(&c, k);
2018    }
2019    for (c, t) in note.unsupported {
2020        unit.note.note_unsupported(&c, &t);
2021    }
2022    Ok(unit)
2023}
