1//! XTDB entity-history EDN importer (Issue #3384).
2//!
3//! Replays an XTDB `xt/entity-history` export taken with `:with-docs? true`,
4//! preserving valid-time, transaction-time provenance, supersession, and
5//! deletes. The export is a single top-level EDN vector of **per-entity history
6//! maps**:
7//!
8//! ```clojure
9//! [{:xt/id :alice
10//!   :history
11//!   [{:xtdb.api/valid-time #inst "2021-03-01T00:00:00Z"
12//!     :xtdb.api/tx-time    #inst "2021-03-01T09:00:00Z"
13//!     :xtdb.api/tx-id 42
14//!     :xtdb.api/doc {:xt/id :alice :type "Person" :name "Alice" :title "Engineer"}}
15//!    ...]}]
16//! ```
17//!
18//! # Mapping
19//!
20//! | Source | AletheiaDB |
21//! |--------|-----------|
22//! | history map `:xt/id` | business key + `xt/id` property |
23//! | `:xtdb.api/doc` (map) | **Node**; label from [`XtdbOptions::label_field`] (default `type`) |
24//! | doc field detected as a ref | **Edge** (see ref detection) |
25//! | `:xtdb.api/valid-time` | `valid_from` |
26//! | `:xtdb.api/valid-to` | interval close (retraction) |
27//! | `:xtdb.api/tx-time` + `:xtdb.api/tx-id` | provenance |
28//! | `:xtdb.api/doc` = nil / absent | delete → `retract_node_detach` |
29//!
30//! Keys are matched by **local name** ignoring namespace, so both the canonical
31//! `:xtdb.api/valid-time` and a short `:xt/valid-time` are accepted.
32

47/// Options controlling an XTDB entity-history import (Issue #3384).
48#[derive(Debug, Clone)]
49pub struct XtdbOptions {
50    /// Doc field (local name) whose value becomes the node label
51    /// (default `"type"`).
52    pub label_field: String,
53    /// Label used when the `label_field` is absent (default `"Entity"`).
54    pub default_label: String,
55    /// Property key under which the `:xt/id` business key is stored
56    /// (default `"xt/id"`).
57    pub id_property: String,
58    /// When `true` (default), a scalar doc field whose value resolves to another
59    /// entity's `:xt/id` is promoted to an edge automatically.
60    pub auto_detect_refs: bool,
61    /// Explicit `field-name → edge-label` overrides. A field named here is
62    /// always treated as a ref, and the mapped value is the edge label.
63    pub ref_fields: std::collections::HashMap<String, String>,
64    /// Abort on the first bad entity, or skip-and-report.
65    pub failure_mode: super::FailureMode,
66}
67
68impl Default for XtdbOptions {
69    fn default() -> Self {
70        XtdbOptions {
71            label_field: "type".to_string(),
72            default_label: "Entity".to_string(),

74            auto_detect_refs: true,
75            ref_fields: std::collections::HashMap::new(),

123/// Canonicalize an EDN scalar into a business-key string. Keywords use their
124/// local (namespace-stripped) name so a `:xt/id :alice` entity and an
125/// `:employer :alice` ref both resolve to `"alice"`.
126fn business_key(v: &Edn) -> Option<String> {
127    match v {
128        Edn::Keyword(k) => Some(edn::keyword_local(k).to_string()),
129        Edn::Str(s) => Some(s.clone()),
130        Edn::Int(i) => Some(i.to_string()),
131        Edn::Uuid(u) => Some(u.clone()),
132        _ => None,
133    }
134}

136/// Convert an EDN scalar / scalar-array into a [`PropertyValue`]. Nested maps
137/// and `nil` yield `None` (absent property).
138fn edn_to_property(v: &Edn) -> Option<PropertyValue> {
139    match v {
140        Edn::Str(s) => Some(PropertyValue::string(s)),
141        Edn::Int(i) => Some(PropertyValue::Int(*i)),
142        Edn::Float(f) => Some(PropertyValue::Float(*f)),
143        Edn::Bool(b) => Some(PropertyValue::Bool(*b)),
144        Edn::Keyword(k) => Some(PropertyValue::string(k)),
145        Edn::Uuid(u) => Some(PropertyValue::string(u)),
146        Edn::Inst(m) => Some(PropertyValue::Int(*m)),
147        Edn::Nil => None,
148        Edn::Symbol(s) => Some(PropertyValue::string(s)),
149        Edn::Vector(items) | Edn::List(items) | Edn::Set(items) => {
150            let vals: Vec<PropertyValue> = items.iter().filter_map(edn_to_property).collect();
151            Some(PropertyValue::array(vals))
152        }
153        Edn::Map(_) => None,
154    }
155}

157/// Uppercase-snake a field name for a default edge label (`employer` → `EMPLOYER`).
158fn screaming_snake(field: &str) -> String {
159    let mut out = String::with_capacity(field.len());
160    for c in field.chars() {
161        if c.is_alphanumeric() {
162            out.extend(c.to_uppercase());
163        } else {
164            out.push('_');
165        }
166    }
167    out
168}

170/// Build the provenance bundle stamped on every version of an entity.
171fn xtdb_provenance(
172    file: &str,
173    tx_id: Option<i64>,
174    tx_time_micros: Option<i64>,
175) -> Result<Provenance> {
176    let mut builder = Provenance::builder()
177        .source(format!("xtdb-import::{file}"))
178        .confidence(1.0);
179    let tx_id_str = tx_id
180        .map(|i| i.to_string())
181        .unwrap_or_else(|| "?".to_string());
182    let tx_time_str = tx_time_micros
183        .map(|m| m.to_string())
184        .unwrap_or_else(|| "?".to_string());
185    builder = builder.note(format!("xtdb tx-id={tx_id_str} tx-time={tx_time_str}"));
186    if let Some(id) = tx_id {
187        builder = builder.correlation_id(format!("xt:tx:{id}"));
188    }
189    builder.build().map_err(|e| Error::other(e.to_string()))
190}

192/// The basename used for the provenance `source`.
193fn file_basename(path: &Path) -> String {
194    path.file_name()
195        .map(|n| n.to_string_lossy().to_string())
196        .unwrap_or_else(|| path.to_string_lossy().to_string())
197}

199impl<'a> Importer<'a> {
200    /// Replay a full XTDB entity-history EDN export, preserving valid-time and
201    /// provenance (Issue #3384).
202    ///
203    /// # Errors
204    ///
205    /// Returns an error if the file cannot be read; the EDN is malformed
206    /// (a typed [`edn::EdnError`](super::edn::EdnError) surfaced with position);
207    /// an entity is already present in this session (re-import refusal); or — in
208    /// the default [`FailureMode::Abort`](super::FailureMode::Abort) — the first
209    /// malformed entity or unresolved ref endpoint.
210    pub fn xtdb_import(
211        &mut self,
212        path: impl AsRef<Path>,
213        opts: &XtdbOptions,
214    ) -> Result<ImportFidelityReport> {
215        let path = path.as_ref();
216        let text = std::fs::read_to_string(path)
217            .map_err(|e| Error::other(format!("reading {}: {e}", path.display())))?;
218        let file = file_basename(path);
219        self.xtdb_import_from_str(&text, &file, opts)
220    }
221
222    /// Parse and replay an XTDB entity-history export already in memory. Shared
223    /// by [`xtdb_import`](Self::xtdb_import) and the test suite.
224    pub(crate) fn xtdb_import_from_str(
225        &mut self,
226        text: &str,
227        file: &str,
228        opts: &XtdbOptions,
229    ) -> Result<ImportFidelityReport> {
230        // Keep the shared replay driver's failure-mode in step with the options.
231        self.config.failure_mode = opts.failure_mode;
232        let mut report = ImportFidelityReport::new("xtdb");
233        let entity_maps =
234            edn::parse_top_vector(text).map_err(|e| Error::other(format!("EDN parse: {e}")))?;
235
236        // First pass: collect all business keys so ref detection can resolve
237        // intra-export references.
238        let mut known_keys: HashSet<String> = HashSet::new();
239        for m in &entity_maps {
240            if let Some(id) = m.map_get_local("id").and_then(business_key) {
241                known_keys.insert(id);
242            }
243        }
244
245        let mut prepared: Vec<PreparedEntity> = Vec::with_capacity(entity_maps.len());
246        for (idx, entity_map) in entity_maps.iter().enumerate() {
247            report.entities_read += 1;
248            match self.prepare_xtdb_entity(
249                entity_map,
250                opts,
251                file,
252                &known_keys,
253                &mut report,
254                idx + 1,
255            ) {
256                Ok(Some(entity)) => prepared.push(entity),
257                Ok(None) => {}
258                Err(row_err) => match opts.failure_mode {
259                    super::FailureMode::Abort => {
260                        return Err(super::ImportError::Row(row_err).into());
261                    }
262                    super::FailureMode::SkipAndReport => report.skipped.push(row_err),
263                },
264            }
265        }
266
267        replay(self, &prepared, &mut report, &opts.id_property)?;
268        report.finalize();
269        Ok(report)
270    }

272    /// Prepare one per-entity history map into a [`PreparedEntity`], or `None`
273    /// if it carried no live version (pure tombstone with nothing to create).
274    #[allow(clippy::too_many_arguments)]
275    fn prepare_xtdb_entity(
276        &self,
277        entity_map: &Edn,
278        opts: &XtdbOptions,
279        file: &str,
280        known_keys: &HashSet<String>,
281        report: &mut ImportFidelityReport,
282        row: usize,
283    ) -> std::result::Result<Option<PreparedEntity>, RowError> {
284        let err = |msg: String| RowError::new(row, msg);
285
286        let id_value = entity_map
287            .map_get_local("id")
288            .ok_or_else(|| err("entity map has no :xt/id".to_string()))?;
289        let key = business_key(id_value)
290            .ok_or_else(|| err(format!("unsupported :xt/id value ({})", id_value.kind())))?;
291
292        let history = entity_map
293            .map_get_local("history")
294            .and_then(Edn::as_seq)
295            .ok_or_else(|| err("entity map has no :history vector".to_string()))?;
296
297        // Defensive sort by (valid-time, tx-id) ascending — the producer claims
298        // ascending, but a mis-ordered replay would fail the downstream
299        // valid_from-not-before-creation check, so we normalize here (AC5).
300        let mut segments: Vec<&Edn> = history.iter().collect();
301        segments.sort_by_key(|seg| {
302            let vt = seg.map_get_local("valid-time").and_then(inst_micros);
303            let tx = seg.map_get_local("tx-id").and_then(as_int);
304            (vt.unwrap_or(i64::MIN), tx.unwrap_or(i64::MIN))
305        });
306
307        let mut versions: Vec<PreparedVersion> = Vec::new();
308        let mut ref_edges: Vec<PreparedRefEdge> = Vec::new();
309        let mut seen_refs: HashSet<(String, String)> = HashSet::new();
310        let mut retraction: Option<PreparedRetraction> = None;
311        let mut last_live_valid_to: Option<i64> = None;
312
313        let n = segments.len();
314        let mut i = 0;
315        while i < n {
316            let seg = segments[i];
317            report.versions_read += 1;
318            let valid_from = seg
319                .map_get_local("valid-time")
320                .and_then(inst_micros)
321                .ok_or_else(|| err("history segment has no :xtdb.api/valid-time".to_string()))?;
322            let tx_id = seg.map_get_local("tx-id").and_then(as_int);
323            let tx_time = seg.map_get_local("tx-time").and_then(inst_micros);
324            let provenance = xtdb_provenance(file, tx_id, tx_time)
325                .map_err(|e| err(format!("provenance: {e}")))?;
326
327            let doc = seg.map_get_local("doc");
328            let is_delete = matches!(doc, None | Some(Edn::Nil));
329
330            if is_delete {
331                // Tombstone: close the entity's valid interval at this segment's
332                // valid-time (detach co-retracts any edges).
333                retraction = Some(PreparedRetraction {
334                    valid_to: Timestamp::from(valid_from),
335                    provenance: Some(provenance),
336                    detach: true,
337                });
338                last_live_valid_to = None;
339                i += 1;
340                continue;
341            }
342
343            let doc = doc.ok_or_else(|| err("delete/live classification error".to_string()))?;
344            if doc.as_map().is_none() {
345                return Err(err(format!(
346                    "history segment :doc must be a map, got {}",
347                    doc.kind()
348                )));
349            }
350
351            let (label, properties, prop_count) =
352                build_xtdb_node(doc, &key, opts, known_keys, report, &err)?;
353
354            // Collect ref edges (deduped across versions; earliest valid_from).
355            collect_xtdb_refs(
356                doc,
357                opts,
358                known_keys,
359                valid_from,
360                &provenance,
361                &mut ref_edges,
362                &mut seen_refs,
363                report,
364            );
365
366            let is_create = versions.is_empty();
367            if is_create {
368                report.note_mapping("document", &key, "node", &label);
369            }
370            versions.push(PreparedVersion {
371                label,
372                properties,
373                valid_from: Timestamp::from(valid_from),
374                provenance: Some(provenance),
375                is_create,
376                property_count: prop_count,
377            });
378
379            let this_valid_to = seg.map_get_local("valid-to").and_then(inst_micros);
380            last_live_valid_to = this_valid_to;
381
382            // Disjoint-interval detection: this live segment explicitly ends
383            // (`::xt/valid-to`) strictly BEFORE the next segment begins — a gap.
384            // The engine cannot reincarnate a retracted node (retraction removes
385            // it from current storage), so we honor THIS close (making the in-gap
386            // coordinate correctly absent instead of returning a stale fact) and
387            // report every un-revivable later segment rather than dropping it
388            // silently.
389            if let Some(vt) = this_valid_to
390                && i + 1 < n
391            {
392                let next_vf = segments[i + 1]
393                    .map_get_local("valid-time")
394                    .and_then(inst_micros)
395                    .unwrap_or(i64::MIN);
396                if next_vf > vt {
397                    let close_prov = xtdb_provenance(file, tx_id, tx_time)
398                        .map_err(|e| err(format!("provenance: {e}")))?;
399                    retraction = Some(PreparedRetraction {
400                        valid_to: Timestamp::from(vt),
401                        provenance: Some(close_prov),
402                        detach: true,
403                    });
404                    for _skipped in &segments[i + 1..] {
405                        report.versions_read += 1;
406                        report.note_unsupported(
407                            "disjoint_reincarnation",
408                            &format!(
409                                "{key}: a valid segment after a gap (closed at {vt}) cannot be \
410                                 reincarnated — the engine removes a retracted node from current \
411                                 state; segment reported, not silently dropped"
412                            ),
413                        );
414                    }
415                    break;
416                }
417            }
418            i += 1;
419        }
420
421        // A trailing live segment with an explicit valid-to closes the interval
422        // (only when no later delete/gap already set a retraction).
423        if retraction.is_none()
424            && let Some(valid_to) = last_live_valid_to
425        {
426            let provenance =
427                xtdb_provenance(file, None, None).map_err(|e| err(format!("provenance: {e}")))?;
428            retraction = Some(PreparedRetraction {
429                valid_to: Timestamp::from(valid_to),
430                provenance: Some(provenance),
431                detach: true,
432            });
433        }
434
435        if versions.is_empty() {
436            // Pure tombstone with no live version to create — nothing to do.
437            return Ok(None);
438        }
439
440        let id_value = PropertyValue::string(&key);
441        Ok(Some(PreparedEntity {
442            key,
443            id_value,
444            versions,
445            ref_edges,
446            retraction,
447        }))
448    }
449}

451/// Resolve the ref targets of a doc field: the (target_key, resolves) pairs the
452/// field promotes to edges. Empty when the field is a plain property. This is
453/// the single source of truth shared by property-building (skip a ref field) and
454/// edge-collection (emit the edges).
455fn field_ref_targets(
456    local: &str,
457    v: &Edn,
458    opts: &XtdbOptions,
459    known_keys: &HashSet<String>,
460) -> Vec<(String, bool)> {
461    let explicit = opts.ref_fields.contains_key(local);
462    let candidates: Vec<&Edn> = match v {
463        Edn::Vector(items) | Edn::List(items) | Edn::Set(items) => items.iter().collect(),
464        other => vec![other],
465    };
466    let mut targets = Vec::new();
467    for cand in candidates {
468        let Some(target_key) = business_key(cand) else {
469            continue;
470        };
471        let resolves = known_keys.contains(&target_key);
472        if explicit || (opts.auto_detect_refs && resolves) {
473            targets.push((target_key, resolves));
474        }
475    }
476    targets
477}

479/// Whether a doc field is treated as a ref (and thus excluded from properties).
480fn is_ref_field(local: &str, v: &Edn, opts: &XtdbOptions, known_keys: &HashSet<String>) -> bool {
481    !field_ref_targets(local, v, opts, known_keys).is_empty()
482}

484/// Build a node label + property map from a doc, excluding the id / label /
485/// content-hash / ref fields.
486fn build_xtdb_node(
487    doc: &Edn,
488    key: &str,
489    opts: &XtdbOptions,
490    known_keys: &HashSet<String>,
491    report: &mut ImportFidelityReport,
492    err: &impl Fn(String) -> RowError,
493) -> std::result::Result<(String, crate::core::property::PropertyMap, usize), RowError> {
494    let label = match doc.map_get_local(&opts.label_field) {
495        Some(Edn::Str(s)) => s.to_string(),
496        Some(Edn::Keyword(k)) => edn::keyword_local(k).to_string(),
497        _ => {
498            report.note_coerced("default_label", &opts.default_label);
499            opts.default_label.clone()
500        }
501    };
502
503    let mut builder = PropertyMapBuilder::new();
504    let mut prop_count = 0usize;
505    // Always store the business key as the id property.
506    builder = builder
507        .try_insert(&opts.id_property, PropertyValue::string(key))
508        .map_err(|e| err(e.to_string()))?;
509    prop_count += 1;
510
511    let entries = doc.as_map().unwrap_or(&[]);
512    for (k, v) in entries {
513        let Edn::Keyword(kw) = k else {
514            continue;
515        };
516        let local = edn::keyword_local(kw);
517        // Skip structural / already-handled fields.
518        if local == "id" || local == opts.label_field || local == "content-hash" {
519            continue;
520        }
521        // Skip fields that are refs (they become edges).
522        if is_ref_field(local, v, opts, known_keys) {
523            continue;
524        }
525        if let Some(value) = edn_to_property(v) {
526            if matches!(value, PropertyValue::Null) {
527                continue;
528            }
529            builder = builder
530                .try_insert(local, value)
531                .map_err(|e| err(e.to_string()))?;
532            prop_count += 1;
533            report.note_mapping("property", local, "property", local);
534        }
535    }
536
537    Ok((label, builder.build(), prop_count))
538}

540/// Collect ref edges from a doc's ref fields, deduped by (field, target).
541#[allow(clippy::too_many_arguments)]
542fn collect_xtdb_refs(
543    doc: &Edn,
544    opts: &XtdbOptions,
545    known_keys: &HashSet<String>,
546    valid_from: i64,
547    provenance: &Provenance,
548    ref_edges: &mut Vec<PreparedRefEdge>,
549    seen_refs: &mut HashSet<(String, String)>,
550    report: &mut ImportFidelityReport,
551) {
552    let entries = doc.as_map().unwrap_or(&[]);
553    for (k, v) in entries {
554        let Edn::Keyword(kw) = k else {
555            continue;
556        };
557        let local = edn::keyword_local(kw);
558        if local == "id" || local == opts.label_field || local == "content-hash" {
559            continue;
560        }
561        let edge_label = opts
562            .ref_fields
563            .get(local)
564            .cloned()
565            .unwrap_or_else(|| screaming_snake(local));
566
567        for (target_key, resolves) in field_ref_targets(local, v, opts, known_keys) {
568            if !resolves {
569                report.note_coerced("unresolved_ref", &format!("{local} -> {target_key}"));
570            }
571            let dedup_key = (local.to_string(), target_key.clone());
572            if !seen_refs.insert(dedup_key) {
573                continue;
574            }
575            report.note_coerced("ref_to_edge", local);
576            ref_edges.push(PreparedRefEdge {
577                target_key,
578                label: edge_label.clone(),
579                source_field: local.to_string(),
580                valid_from: Timestamp::from(valid_from),
581                provenance: Some(provenance.clone()),
582                retract_at: None,
583            });
584        }
585    }
586}
