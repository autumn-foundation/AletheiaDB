1//! Minimal, panic-free EDN reader for the XTDB / Datomic importers (Issue #3384).
2//!
3//! EDN (Extensible Data Notation) is the native serialization of both XTDB and
4//! Datomic: a Clojure export script emits it directly and it preserves
5//! keywords, `#inst`, and `#uuid` losslessly (unlike a JSON round-trip). This
6//! module ships **only** the subset the two migration artifacts need — it is
7//! deliberately *not* a full Clojure reader.
8//!
9//! # Supported
10//!
11//! - `nil`, `true`, `false`
12//! - integers (optional sign, decimal digits) and floats (sign / digits / `.` /
13//!   `e|E` exponent / trailing `M` decimal marker, stripped)
14//! - strings `"..."` with escapes `\" \\ \n \t \r \/ \uXXXX`
15//! - keywords `:kw`, `:ns/kw`, `::alias/kw` (the leading colon(s) are stripped;
16//!   a namespaced keyword keeps its `ns/name` text so the caller can normalize)
17//! - symbols (bare tokens, retained verbatim)
18//! - collections: vector `[]`, list `()`, map `{}`, set `#{}`
19//! - `;` line comments and `#_` datum-discard
20//! - tagged literals `#inst "..."` (→ epoch micros) and `#uuid "..."` (→ the
21//!   uuid string); **any other** `#tag <form>` is tolerated — the tag is
22//!   discarded and the inner form retained, never fatal
23//!
24//! # Out of scope (typed [`EdnError::Unsupported`], never a panic)
25//!
26//! `#=` eval, ratios (`1/2`), radix / bigint `N` suffixes, char literals
27//! `\c`, regex `#"..."`, metadata `^{}`, namespaced-map `#:ns{...}`, and reader
28//! conditionals `#?()`. Trailing `M` on a float is accepted (precision loss is
29//! documented); a bigint `N` suffix is rejected.
30//!
31//! # Robustness
32//!
33//! The reader tracks a 1-based line/column and attaches it to every error. It
34//! **never panics** on malformed input: recursion depth is bounded
35//! ([`MAX_DEPTH`]) so a pathological deeply-nested collection returns
36//! [`EdnError::TooDeep`] instead of overflowing the stack, and no indexing can
37//! run off the end of the buffer.
38

41/// Maximum nesting depth for collections before the reader bails with
42/// [`EdnError::TooDeep`]. Mirrors the Cypher importer's `MAX_VALUE_DEPTH` cap:
43/// adversarial input must fail loudly, not overflow the stack.
44pub(crate) const MAX_DEPTH: usize = 256;
45
46/// A parsed EDN value (the subset the importers need).
47#[derive(Debug, Clone, PartialEq)]
48pub(crate) enum Edn {
49    /// `nil`.
50    Nil,
51    /// `true` / `false`.
52    Bool(bool),
53    /// A 64-bit integer.
54    Int(i64),
55    /// A double-precision float.
56    Float(f64),
57    /// A string literal (escapes already decoded).
58    Str(String),
59    /// A keyword, stored **without** the leading `:`; a namespaced keyword keeps
60    /// its `ns/name` text (`:xtdb.api/valid-time` → `"xtdb.api/valid-time"`).
61    Keyword(String),
62    /// A bare symbol, retained verbatim.
63    Symbol(String),
64    /// A vector `[...]`.
65    Vector(Vec<Edn>),
66    /// A list `(...)`.
67    List(Vec<Edn>),
68    /// A map `{...}`, insertion-ordered; keys are compared structurally.
69    Map(Vec<(Edn, Edn)>),
70    /// A set `#{...}`.
71    Set(Vec<Edn>),
72    /// `#inst "..."` resolved to epoch **microseconds** (UTC).
73    Inst(i64),
74    /// `#uuid "..."` retained as its string form.
75    Uuid(String),
76}

78impl Edn {
79    /// Borrow the value as a keyword's local text, if it is a keyword.
80    pub(crate) fn as_keyword(&self) -> Option<&str> {
81        match self {
82            Edn::Keyword(k) => Some(k.as_str()),
83            _ => None,
84        }
85    }
86
87    /// Borrow the value as a string, if it is one.
88    pub(crate) fn as_str(&self) -> Option<&str> {
89        match self {
90            Edn::Str(s) => Some(s.as_str()),
91            _ => None,
92        }
93    }
94
95    /// Borrow the value as a map's entries, if it is a map.
96    pub(crate) fn as_map(&self) -> Option<&[(Edn, Edn)]> {
97        match self {
98            Edn::Map(entries) => Some(entries.as_slice()),
99            _ => None,
100        }
101    }
102
103    /// Borrow the value as a vector's / list's elements, if it is one of those.
104    pub(crate) fn as_seq(&self) -> Option<&[Edn]> {
105        match self {
106            Edn::Vector(v) | Edn::List(v) | Edn::Set(v) => Some(v.as_slice()),
107            _ => None,
108        }
109    }
110
111    /// Look up a map entry by the local name of a keyword key. The lookup
112    /// **ignores namespaces**: a `namespace/name` key matches on `name`, so
113    /// `:xtdb.api/valid-time` and `:xt/valid-time` both resolve for `"valid-time"`.
114    pub(crate) fn map_get_local(&self, local: &str) -> Option<&Edn> {
115        let entries = self.as_map()?;
116        entries.iter().find_map(|(k, v)| match k {
117            Edn::Keyword(name) => (keyword_local(name) == local).then_some(v),
118            _ => None,
119        })
120    }
121
122    /// A short human token naming the value's kind, for error messages.
123    pub(crate) fn kind(&self) -> &'static str {
124        match self {
125            Edn::Nil => "nil",
126            Edn::Bool(_) => "bool",
127            Edn::Int(_) => "int",
128            Edn::Float(_) => "float",
129            Edn::Str(_) => "string",
130            Edn::Keyword(_) => "keyword",
131            Edn::Symbol(_) => "symbol",
132            Edn::Vector(_) => "vector",
133            Edn::List(_) => "list",
134            Edn::Map(_) => "map",
135            Edn::Set(_) => "set",
136            Edn::Inst(_) => "inst",
137            Edn::Uuid(_) => "uuid",
138        }
139    }
140}

142/// Return the local (namespace-stripped) portion of a keyword/symbol text.
143/// `"xtdb.api/valid-time"` → `"valid-time"`; `"type"` → `"type"`.
144pub(crate) fn keyword_local(name: &str) -> &str {
145    match name.rsplit_once('/') {
146        Some((_, local)) => local,
147        None => name,
148    }
149}

151/// A typed EDN parse error, always carrying a 1-based source position.
152#[derive(Debug, Clone, PartialEq, Eq)]
153pub(crate) enum EdnError {
154    /// The input ended while a value/collection/string was still open.
155    UnexpectedEof,
156    /// An unexpected character was encountered mid-parse.
157    UnexpectedChar {
158        found: char,
159        line: usize,
160        col: usize,
161    },
162    /// A string literal was not closed before EOF or newline handling.
163    UnterminatedString { line: usize, col: usize },
164    /// A closing delimiter did not match the open one (`[` closed by `}`), or a
165    /// stray closer with no opener.
166    UnbalancedDelimiter {
167        expected: char,
168        found: char,
169        line: usize,
170        col: usize,
171    },
172    /// An unknown / malformed string escape (`\q`, truncated `\u12`).
173    BadEscape { line: usize, col: usize },
174    /// A `#inst` / `#uuid` (or other) tagged literal whose payload was invalid.
175    BadTaggedLiteral {
176        tag: String,
177        reason: String,
178        line: usize,
179        col: usize,
180    },
181    /// A numeric literal that does not fit / is malformed (e.g. a 200-digit int
182    /// beyond `i64`, or a bigint `N` suffix).
183    BadNumber {
184        reason: String,
185        line: usize,
186        col: usize,
187    },
188    /// A construct the reader deliberately does not support (ratio, regex, `#=`,
189    /// metadata, namespaced-map, reader conditional).
190    Unsupported {
191        construct: String,
192        line: usize,
193        col: usize,
194    },
195    /// Collections nested deeper than [`MAX_DEPTH`].
196    TooDeep { line: usize, col: usize },
197    /// Extra non-whitespace data after a single top-level value (`parse_one`).
198    TrailingData { line: usize, col: usize },
199}
200
201impl fmt::Display for EdnError {
202    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
203        match self {
204            EdnError::UnexpectedEof => write!(f, "unexpected end of EDN input"),
205            EdnError::UnexpectedChar { found, line, col } => {
206                write!(f, "unexpected character '{found}' at {line}:{col}")
207            }
208            EdnError::UnterminatedString { line, col } => {
209                write!(f, "unterminated string at {line}:{col}")
210            }
211            EdnError::UnbalancedDelimiter {
212                expected,
213                found,
214                line,
215                col,
216            } => write!(
217                f,
218                "unbalanced delimiter at {line}:{col}: expected '{expected}', found '{found}'"
219            ),
220            EdnError::BadEscape { line, col } => write!(f, "bad string escape at {line}:{col}"),
221            EdnError::BadTaggedLiteral {
222                tag,
223                reason,
224                line,
225                col,
226            } => write!(f, "bad #{tag} literal at {line}:{col}: {reason}"),
227            EdnError::BadNumber { reason, line, col } => {
228                write!(f, "bad number at {line}:{col}: {reason}")
229            }
230            EdnError::Unsupported {
231                construct,
232                line,
233                col,
234            } => write!(f, "unsupported EDN construct '{construct}' at {line}:{col}"),
235            EdnError::TooDeep { line, col } => {
236                write!(f, "EDN nesting too deep (>{MAX_DEPTH}) at {line}:{col}")
237            }
238            EdnError::TrailingData { line, col } => {
239                write!(f, "trailing data after top-level value at {line}:{col}")
240            }
241        }
242    }
243}

247/// The line/column reported for a given error, when it carries one. Used by
248/// tests to assert precise positions.
249impl EdnError {
250    #[cfg(test)]
251    pub(crate) fn position(&self) -> Option<(usize, usize)> {
252        match self {
253            EdnError::UnexpectedEof => None,
254            EdnError::UnexpectedChar { line, col, .. }
255            | EdnError::UnterminatedString { line, col }
256            | EdnError::UnbalancedDelimiter { line, col, .. }
257            | EdnError::BadEscape { line, col }
258            | EdnError::BadTaggedLiteral { line, col, .. }
259            | EdnError::BadNumber { line, col, .. }
260            | EdnError::Unsupported { line, col, .. }
261            | EdnError::TooDeep { line, col }
262            | EdnError::TrailingData { line, col } => Some((*line, *col)),
263        }
264    }
265}

267/// A position-tracking EDN reader over an owned char buffer.
268struct Parser {
269    chars: Vec<char>,
270    pos: usize,
271    line: usize,
272    col: usize,
273}
274
275impl Parser {
276    fn new(input: &str) -> Self {
277        Parser {
278            chars: input.chars().collect(),
279            pos: 0,
280            line: 1,
281            col: 1,
282        }
283    }
284
285    fn peek(&self) -> Option<char> {
286        self.chars.get(self.pos).copied()
287    }
288
289    fn peek_at(&self, ahead: usize) -> Option<char> {
290        self.chars.get(self.pos + ahead).copied()
291    }
292
293    /// Consume and return the next char, advancing the 1-based line/col.
294    fn bump(&mut self) -> Option<char> {
295        let c = self.chars.get(self.pos).copied()?;
296        self.pos += 1;
297        if c == '\n' {
298            self.line += 1;
299            self.col = 1;
300        } else {
301            self.col += 1;
302        }
303        Some(c)
304    }
305
306    fn pos_tuple(&self) -> (usize, usize) {
307        (self.line, self.col)
308    }
309
310    /// Skip whitespace (including `,`) and `;` line comments.
311    fn skip_ws(&mut self) {
312        while let Some(c) = self.peek() {
313            if c.is_whitespace() || c == ',' {
314                self.bump();
315            } else if c == ';' {
316                // Line comment: consume to end of line.
317                while let Some(c) = self.peek() {
318                    if c == '\n' {
319                        break;
320                    }
321                    self.bump();
322                }
323            } else {
324                break;
325            }
326        }
327    }
328
329    /// Parse a single value at the current position, honoring `#_` datum
330    /// discards that precede it.
331    fn parse_value(&mut self, depth: usize) -> Result<Edn, EdnError> {
332        loop {
333            self.skip_ws();
334            // Handle `#_` datum discard: skip the next form and continue. Each
335            // discard recurses to parse the form it drops, so it MUST count
336            // against the depth budget — otherwise `#_#_#_…0` recurses unbounded
337            // and overflows the stack (the reader's never-panic contract).
338            if self.peek() == Some('#') && self.peek_at(1) == Some('_') {
339                let (line, col) = self.pos_tuple();
340                if depth >= MAX_DEPTH {
341                    return Err(EdnError::TooDeep { line, col });
342                }
343                self.bump();
344                self.bump();
345                // Discard exactly one following form.
346                self.parse_value(depth + 1)?;
347                continue;
348            }
349            break;
350        }
351
352        let (line, col) = self.pos_tuple();
353        let c = self.peek().ok_or(EdnError::UnexpectedEof)?;
354        match c {
355            '[' => self.parse_seq('[', ']', depth),
356            '(' => self.parse_seq('(', ')', depth),
357            '{' => self.parse_map(depth),
358            '"' => self.parse_string(),
359            ':' => self.parse_keyword(),
360            '#' => self.parse_dispatch(depth),
361            ']' | '}' | ')' => Err(EdnError::UnbalancedDelimiter {
362                expected: 'x',
363                found: c,
364                line,
365                col,
366            }),
367            '\\' => Err(EdnError::Unsupported {
368                construct: "char literal".to_string(),
369                line,
370                col,
371            }),
372            '^' => Err(EdnError::Unsupported {
373                construct: "metadata".to_string(),
374                line,
375                col,
376            }),
377            c if c == '-' || c == '+' || c.is_ascii_digit() => self.parse_number(),
378            c if is_symbol_start(c) => self.parse_symbol_or_atom(),
379            _ => Err(EdnError::UnexpectedChar {
380                found: c,
381                line,
382                col,
383            }),
384        }
385    }
386
387    /// Parse a `[...]` vector or `(...)` list.
388    fn parse_seq(&mut self, open: char, close: char, depth: usize) -> Result<Edn, EdnError> {
389        let (line, col) = self.pos_tuple();
390        if depth >= MAX_DEPTH {
391            return Err(EdnError::TooDeep { line, col });
392        }
393        self.bump(); // consume opener
394        let mut items = Vec::new();
395        loop {
396            self.skip_ws();
397            match self.peek() {
398                None => return Err(EdnError::UnexpectedEof),
399                Some(c) if c == close => {
400                    self.bump();
401                    break;
402                }
403                Some(c) if c == ']' || c == '}' || c == ')' => {
404                    let (l, cc) = self.pos_tuple();
405                    return Err(EdnError::UnbalancedDelimiter {
406                        expected: close,
407                        found: c,
408                        line: l,
409                        col: cc,
410                    });
411                }
412                Some(_) => items.push(self.parse_value(depth + 1)?),
413            }
414        }
415        Ok(if open == '[' {
416            Edn::Vector(items)
417        } else {
418            Edn::List(items)
419        })
420    }
421
422    /// Parse a `{...}` map.
423    fn parse_map(&mut self, depth: usize) -> Result<Edn, EdnError> {
424        let (line, col) = self.pos_tuple();
425        if depth >= MAX_DEPTH {
426            return Err(EdnError::TooDeep { line, col });
427        }
428        self.bump(); // consume '{'
429        let mut entries = Vec::new();
430        loop {
431            self.skip_ws();
432            match self.peek() {
433                None => return Err(EdnError::UnexpectedEof),
434                Some('}') => {
435                    self.bump();
436                    break;
437                }
438                Some(c) if c == ']' || c == ')' => {
439                    let (l, cc) = self.pos_tuple();
440                    return Err(EdnError::UnbalancedDelimiter {
441                        expected: '}',
442                        found: c,
443                        line: l,
444                        col: cc,
445                    });
446                }
447                Some(_) => {
448                    let key = self.parse_value(depth + 1)?;
449                    self.skip_ws();
450                    if self.peek().is_none() {
451                        return Err(EdnError::UnexpectedEof);
452                    }
453                    if self.peek() == Some('}') {
454                        // Odd number of forms in a map.
455                        let (l, cc) = self.pos_tuple();
456                        return Err(EdnError::UnexpectedChar {
457                            found: '}',
458                            line: l,
459                            col: cc,
460                        });
461                    }
462                    let value = self.parse_value(depth + 1)?;
463                    entries.push((key, value));
464                }
465            }
466        }
467        Ok(Edn::Map(entries))
468    }
469
470    /// Parse a `#`-dispatch form: `#{` set, `#inst`, `#uuid`, `#_` (handled
471    /// earlier), or an unknown `#tag <form>` (tolerated), or an unsupported
472    /// `#"..."` / `#?(...)` / `#:ns{...}` / `#=` construct.
473    fn parse_dispatch(&mut self, depth: usize) -> Result<Edn, EdnError> {
474        let (line, col) = self.pos_tuple();
475        self.bump(); // consume '#'
476        match self.peek() {
477            None => Err(EdnError::UnexpectedEof),
478            Some('{') => self.parse_set(depth),
479            Some('"') => Err(EdnError::Unsupported {
480                construct: "regex literal".to_string(),
481                line,
482                col,
483            }),
484            Some('?') => Err(EdnError::Unsupported {
485                construct: "reader conditional".to_string(),
486                line,
487                col,
488            }),
489            Some('=') => Err(EdnError::Unsupported {
490                construct: "eval literal".to_string(),
491                line,
492                col,
493            }),
494            Some(':') => Err(EdnError::Unsupported {
495                construct: "namespaced map".to_string(),
496                line,
497                col,
498            }),
499            Some('_') => {
500                // Should have been handled in parse_value, but be defensive —
501                // and, like that path, count the discard against the depth
502                // budget so a `#_` chain reaching here cannot recurse unbounded.
503                if depth >= MAX_DEPTH {
504                    return Err(EdnError::TooDeep { line, col });
505                }
506                self.bump();
507                self.parse_value(depth + 1)?;
508                self.parse_value(depth + 1)
509            }
510            Some(c) if is_symbol_start(c) => {
511                let tag = self.read_token();
512                self.skip_ws();
513                let (tline, tcol) = (line, col);
514                let inner = self.parse_value(depth + 1)?;
515                self.apply_tag(&tag, inner, tline, tcol)
516            }
517            Some(c) => Err(EdnError::UnexpectedChar {
518                found: c,
519                line,
520                col,
521            }),
522        }
523    }
524
525    /// Parse a `#{...}` set (the `#` has been consumed; we are at `{`).
526    fn parse_set(&mut self, depth: usize) -> Result<Edn, EdnError> {
527        let (line, col) = self.pos_tuple();
528        if depth >= MAX_DEPTH {
529            return Err(EdnError::TooDeep { line, col });
530        }
531        self.bump(); // consume '{'
532        let mut items = Vec::new();
533        loop {
534            self.skip_ws();
535            match self.peek() {
536                None => return Err(EdnError::UnexpectedEof),
537                Some('}') => {
538                    self.bump();
539                    break;
540                }
541                Some(c) if c == ']' || c == ')' => {
542                    let (l, cc) = self.pos_tuple();
543                    return Err(EdnError::UnbalancedDelimiter {
544                        expected: '}',
545                        found: c,
546                        line: l,
547                        col: cc,
548                    });
549                }
550                Some(_) => items.push(self.parse_value(depth + 1)?),
551            }
552        }
553        Ok(Edn::Set(items))
554    }

556    /// Apply a tag to its already-parsed inner form.
557    fn apply_tag(&self, tag: &str, inner: Edn, line: usize, col: usize) -> Result<Edn, EdnError> {
558        match tag {
559            "inst" => {
560                let s = inner.as_str().ok_or_else(|| EdnError::BadTaggedLiteral {
561                    tag: "inst".to_string(),
562                    reason: format!("expected a string, got {}", inner.kind()),
563                    line,
564                    col,
565                })?;
566                let ts = super::parse::parse_valid_time(s).map_err(|reason| {
567                    EdnError::BadTaggedLiteral {
568                        tag: "inst".to_string(),
569                        reason,
570                        line,
571                        col,
572                    }
573                })?;
574                Ok(Edn::Inst(ts.wallclock()))
575            }
576            "uuid" => {
577                let s = inner.as_str().ok_or_else(|| EdnError::BadTaggedLiteral {
578                    tag: "uuid".to_string(),
579                    reason: format!("expected a string, got {}", inner.kind()),
580                    line,
581                    col,
582                })?;
583                Ok(Edn::Uuid(s.to_string()))
584            }
585            // Any other tag (e.g. `#crux/id`, `#audit/id`) is tolerated: the tag
586            // is discarded and the inner form retained (never fatal).
587            _ => Ok(inner),
588        }
589    }

591    /// Parse a `"..."` string, decoding escapes.
592    fn parse_string(&mut self) -> Result<Edn, EdnError> {
593        let (line, col) = self.pos_tuple();
594        self.bump(); // consume opening quote
595        let mut out = String::new();
596        loop {
597            let c = match self.bump() {
598                Some(c) => c,
599                None => return Err(EdnError::UnterminatedString { line, col }),
600            };
601            match c {
602                '"' => return Ok(Edn::Str(out)),
603                '\\' => {
604                    let (eline, ecol) = self.pos_tuple();
605                    let esc = self
606                        .bump()
607                        .ok_or(EdnError::UnterminatedString { line, col })?;
608                    match esc {
609                        '"' => out.push('"'),
610                        '\\' => out.push('\\'),
611                        '/' => out.push('/'),
612                        'n' => out.push('\n'),
613                        't' => out.push('\t'),
614                        'r' => out.push('\r'),
615                        'b' => out.push('\u{0008}'),
616                        'f' => out.push('\u{000C}'),
617                        'u' => {
618                            let mut code: u32 = 0;
619                            for _ in 0..4 {
620                                let h = self.bump().ok_or(EdnError::BadEscape {
621                                    line: eline,
622                                    col: ecol,
623                                })?;
624                                let digit = h.to_digit(16).ok_or(EdnError::BadEscape {
625                                    line: eline,
626                                    col: ecol,
627                                })?;
628                                code = code * 16 + digit;
629                            }
630                            let ch = char::from_u32(code).ok_or(EdnError::BadEscape {
631                                line: eline,
632                                col: ecol,
633                            })?;
634                            out.push(ch);
635                        }
636                        _ => {
637                            return Err(EdnError::BadEscape {
638                                line: eline,
639                                col: ecol,
640                            });
641                        }
642                    }
643                }
644                _ => out.push(c),
645            }
646        }
647    }

649    /// Parse a keyword: one or two leading colons, then the token.
650    fn parse_keyword(&mut self) -> Result<Edn, EdnError> {
651        let (line, col) = self.pos_tuple();
652        self.bump(); // consume first ':'
653        // `::alias/kw` auto-resolved keyword — consume the second colon; we keep
654        // the local+ns text and let the caller normalize by local name.
655        if self.peek() == Some(':') {
656            self.bump();
657        }
658        let token = self.read_token();
659        if token.is_empty() {
660            return Err(EdnError::UnexpectedChar {
661                found: self.peek().unwrap_or(':'),
662                line,
663                col,
664            });
665        }
666        Ok(Edn::Keyword(token))
667    }
668
669    /// Parse a bare symbol, or the atoms `nil` / `true` / `false`.
670    fn parse_symbol_or_atom(&mut self) -> Result<Edn, EdnError> {
671        let token = self.read_token();
672        match token.as_str() {
673            "nil" => Ok(Edn::Nil),
674            "true" => Ok(Edn::Bool(true)),
675            "false" => Ok(Edn::Bool(false)),
676            _ => Ok(Edn::Symbol(token)),
677        }
678    }
679
680    /// Parse an integer or float literal.
681    fn parse_number(&mut self) -> Result<Edn, EdnError> {
682        let (line, col) = self.pos_tuple();
683        let token = self.read_token();
684        // A ratio like `1/2` is unsupported.
685        if token.contains('/') {
686            return Err(EdnError::Unsupported {
687                construct: "ratio".to_string(),
688                line,
689                col,
690            });
691        }
692        let is_float = token.contains('.')
693            || token.contains('e')
694            || token.contains('E')
695            || token.ends_with('M');
696        if is_float {
697            // Strip a trailing `M` (BigDecimal marker); precision loss is
698            // documented in the module header.
699            let cleaned = token.strip_suffix('M').unwrap_or(&token);
700            cleaned
701                .parse::<f64>()
702                .map(Edn::Float)
703                .map_err(|_| EdnError::BadNumber {
704                    reason: format!("invalid float '{token}'"),
705                    line,
706                    col,
707                })
708        } else if let Some(stripped) = token.strip_suffix('N') {
709            // Bigint literal — out of scope; report rather than silently
710            // truncating (a value that fits i64 without the marker still errors,
711            // making the loss explicit).
712            let _ = stripped;
713            Err(EdnError::BadNumber {
714                reason: "arbitrary-precision integer 'N' suffix is unsupported".to_string(),
715                line,
716                col,
717            })
718        } else {
719            token
720                .parse::<i64>()
721                .map(Edn::Int)
722                .map_err(|_| EdnError::BadNumber {
723                    reason: format!("integer '{token}' does not fit in i64"),
724                    line,
725                    col,
726                })
727        }
728    }
729
730    /// Read a run of token characters (used for symbols, keywords, numbers,
731    /// tags). Stops at whitespace, `,`, or a delimiter/dispatch character.
732    fn read_token(&mut self) -> String {
733        let mut out = String::new();
734        while let Some(c) = self.peek() {
735            if c.is_whitespace() || is_token_terminator(c) {
736                break;
737            }
738            out.push(c);
739            self.bump();
740        }
741        out
742    }
743}
744
745/// Characters that terminate a bare token.
746fn is_token_terminator(c: char) -> bool {
747    matches!(c, '[' | ']' | '(' | ')' | '{' | '}' | '"' | ';' | ',')
748}
749
750/// Whether `c` can start a symbol/atom (letters, and the punctuation Clojure
751/// allows in symbols). Excludes digits/sign (numbers), `:` (keywords), and the
752/// structural characters.
753fn is_symbol_start(c: char) -> bool {
754    c.is_alphabetic()
755        || matches!(
756            c,
757            '.' | '*' | '+' | '!' | '-' | '_' | '?' | '$' | '%' | '&' | '=' | '<' | '>'
758        )
759}

761/// Parse **exactly one** top-level EDN value, rejecting trailing data.
762pub(crate) fn parse_one(input: &str) -> Result<Edn, EdnError> {
763    let mut parser = Parser::new(input);
764    let value = parser.parse_value(0)?;
765    parser.skip_ws();
766    if parser.peek().is_some() {
767        let (line, col) = parser.pos_tuple();
768        return Err(EdnError::TrailingData { line, col });
769    }
770    Ok(value)
771}
772
773/// Parse a top-level EDN **vector** `[...]` and return its elements. The XTDB
774/// entity-history export and the Datomic datom stream are both a single
775/// top-level vector; this is the artifact entry point.
776pub(crate) fn parse_top_vector(input: &str) -> Result<Vec<Edn>, EdnError> {
777    let mut parser = Parser::new(input);
778    parser.skip_ws();
779    // Capture the position of the top-level form so a non-vector error points at
780    // the real start rather than a hardcoded 1:1.
781    let (line, col) = parser.pos_tuple();
782    let value = parser.parse_value(0)?;
783    parser.skip_ws();
784    if parser.peek().is_some() {
785        let (tline, tcol) = parser.pos_tuple();
786        return Err(EdnError::TrailingData {
787            line: tline,
788            col: tcol,
789        });
790    }
791    match value {
792        Edn::Vector(items) | Edn::List(items) => Ok(items),
793        other => Err(EdnError::Unsupported {
794            construct: format!("top-level {} (expected a vector)", other.kind()),
795            line,
796            col,
797        }),
798    }
