1//! Shared history-replay machinery for the XTDB and Datomic importers
2//! (Issue #3384).
3//!
4//! Unlike the CSV/Neo4j importers, which commit **one create per row**, an
5//! XTDB entity-history sequence and a Datomic datom stream describe **multiple
6//! versions per entity** (create → N updates → optional retract). This module
7//! provides the dedicated per-entity replay driver those importers need, plus
8//! the shared [`ImportFidelityReport`] (a sibling of `Neo4jFidelityReport`,
9//! specialized for a history-preserving migration).
10//!
11//! The driver still reuses the existing spine: [`Importer::key_to_id`], the
12//! `db.write(|tx| …)` batching path, [`WriteRequestOptions`], the
13//! `create_*_with_options` / `retract_*_with_provenance` write APIs, and the
14//! `RowError` / `UnresolvedEndpoint` error model.
15
16use super::{Endpoint, Importer, RowError, UnresolvedEndpoint};
17use crate::core::error::{Error, Result};
18use crate::core::id::NodeId;
19use crate::core::property::{PropertyMap, PropertyValue};
20use crate::core::provenance::Provenance;
21use crate::core::temporal::Timestamp;

23/// One `(source → AletheiaDB)` mapping-table row in the fidelity report.
24#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
25pub struct MappingEntry {
26    /// The kind of source construct (`document`, `attribute`, `ref`, `datom`).
27    pub source_kind: String,
28    /// The source name (doc field / attribute ident / entity label).
29    pub source_name: String,
30    /// The AletheiaDB construct it became (`node`, `edge`, `property`).
31    pub aletheia_kind: String,
32    /// The AletheiaDB label / property key chosen.
33    pub aletheia_label: String,
34    /// Number of source items mapped this way.
35    pub count: usize,
36}
37
38/// A source construct that was **not** imported and is reported (never silently
39/// dropped): a Datomic schema-alteration datom, a `:db/fn` install, an excision,
40/// or an attribute retraction recorded as an explicit null.
41#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
42pub struct UnsupportedNote2 {
43    /// A stable machine token (`schema_history`, `db_fn`, `excision`,
44    /// `attr_retract_as_null`, `unknown_attribute`).
45    pub kind: String,
46    /// Human-readable detail (e.g. the offending attribute ident).
47    pub detail: String,
48    /// Number of occurrences.
49    pub count: usize,
50}
51
52/// A value that was accepted but transformed on the way in.
53#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
54pub struct CoercionNote2 {
55    /// A stable machine token (`inst_to_micros`, `ref_to_edge`,
56    /// `default_label`, `unresolved_ref`, `cardinality_many_scalar`).
57    pub kind: String,
58    /// Human-readable detail.
59    pub detail: String,
60    /// Number of occurrences.
61    pub count: usize,
62}

64/// Machine-readable fidelity report for a history-preserving import
65/// (Issue #3384), shared by the XTDB and Datomic importers.
66///
67/// Mirrors `Neo4jFidelityReport`: counts, a mapping table, skipped/unresolved
68/// items with reasons, and a `zero_loss` boolean that is `true` iff nothing was
69/// skipped, left unresolved, or reported unsupported.
70#[derive(Debug, Clone, Default, serde::Serialize)]
71pub struct ImportFidelityReport {
72    /// `"xtdb"` or `"datomic"`.
73    pub source: &'static str,
74    /// Number of source entities read (history maps / distinct `:e`).
75    pub entities_read: usize,
76    /// Total source versions read (history segments / datom transactions).
77    pub versions_read: usize,
78    /// Distinct nodes created.
79    pub nodes_created: usize,
80    /// Node versions written (creates + updates).
81    pub node_versions_written: usize,
82    /// Edges created.
83    pub edges_created: usize,
84    /// Node + edge valid-interval closes (retractions).
85    pub retractions: usize,
86    /// Property values written.
87    pub properties_imported: usize,
88    /// Source → AletheiaDB mapping table with counts.
89    pub mapping: Vec<MappingEntry>,
90    /// Malformed source items skipped (populated in skip-and-report mode).
91    pub skipped: Vec<RowError>,
92    /// Ref endpoints that did not resolve (populated in skip-and-report mode).
93    pub unresolved_endpoints: Vec<UnresolvedEndpoint>,
94    /// Source constructs reported as unsupported (never silently dropped).
95    pub unsupported: Vec<UnsupportedNote2>,
96    /// Values accepted but transformed.
97    pub coerced: Vec<CoercionNote2>,
98    /// `true` iff `skipped`, `unresolved_endpoints`, and `unsupported` are all
99    /// empty — a lossless import.
100    pub zero_loss: bool,
101}
102
103impl ImportFidelityReport {
104    /// Create an empty report tagged with its `source`.
105    pub(crate) fn new(source: &'static str) -> Self {
106        ImportFidelityReport {
107            source,
108            ..Default::default()
109        }
110    }
111
112    /// Recompute [`zero_loss`](Self::zero_loss) from the accumulated notes.
113    pub(crate) fn finalize(&mut self) {
114        self.zero_loss = self.skipped.is_empty()
115            && self.unresolved_endpoints.is_empty()
116            && self.unsupported.is_empty();
117    }
118
119    /// Fold one occurrence into the mapping table, deduping by
120    /// `(source_kind, source_name, aletheia_kind, aletheia_label)`.
121    pub(crate) fn note_mapping(
122        &mut self,
123        source_kind: &str,
124        source_name: &str,
125        aletheia_kind: &str,
126        aletheia_label: &str,
127    ) {
128        if let Some(entry) = self.mapping.iter_mut().find(|e| {
129            e.source_kind == source_kind
130                && e.source_name == source_name
131                && e.aletheia_kind == aletheia_kind
132                && e.aletheia_label == aletheia_label
133        }) {
134            entry.count += 1;
135        } else {
136            self.mapping.push(MappingEntry {
137                source_kind: source_kind.to_string(),
138                source_name: source_name.to_string(),
139                aletheia_kind: aletheia_kind.to_string(),
140                aletheia_label: aletheia_label.to_string(),
141                count: 1,
142            });
143        }
144    }
145
146    /// Fold one occurrence into the unsupported table, deduping by `(kind, detail)`.
147    pub(crate) fn note_unsupported(&mut self, kind: &str, detail: &str) {
148        if let Some(entry) = self
149            .unsupported
150            .iter_mut()
151            .find(|e| e.kind == kind && e.detail == detail)
152        {
153            entry.count += 1;
154        } else {
155            self.unsupported.push(UnsupportedNote2 {
156                kind: kind.to_string(),
157                detail: detail.to_string(),
158                count: 1,
159            });
160        }
161    }
162
163    /// Fold one occurrence into the coercion table, deduping by `(kind, detail)`.
164    pub(crate) fn note_coerced(&mut self, kind: &str, detail: &str) {
165        if let Some(entry) = self
166            .coerced
167            .iter_mut()
168            .find(|e| e.kind == kind && e.detail == detail)
169        {
170            entry.count += 1;
171        } else {
172            self.coerced.push(CoercionNote2 {
173                kind: kind.to_string(),
174                detail: detail.to_string(),
175                count: 1,
176            });
177        }
178    }
179}

181/// One node version to write during replay: a create (first) or an update.
182pub(crate) struct PreparedVersion {
183    /// The node label for this version.
184    pub label: String,
185    /// The properties written at this version.
186    pub properties: PropertyMap,
187    /// When this version became valid in reality.
188    pub valid_from: Timestamp,
189    /// Optional write-time provenance for this version.
190    pub provenance: Option<Provenance>,
191    /// `true` for the entity's first version (a create); `false` for updates.
192    pub is_create: bool,
193    /// Number of property values in this version (for the fidelity count).
194    pub property_count: usize,
195}

197/// A node-level valid-interval close (a delete / whole-entity retraction).
198pub(crate) struct PreparedRetraction {
199    /// The valid-time at which the entity stops being valid.
200    pub valid_to: Timestamp,
201    /// Optional write-time provenance for the retraction version.
202    pub provenance: Option<Provenance>,
203    /// Whether to co-retract connected edges (`retract_node_detach`).
204    pub detach: bool,
205}

207/// A ref field / ref attribute that becomes an edge sourced at this entity.
208pub(crate) struct PreparedRefEdge {
209    /// The business key of the edge target.
210    pub target_key: String,
211    /// The edge label.
212    pub label: String,
213    /// The field/attribute name the edge came from (for the mapping table).
214    pub source_field: String,
215    /// When the edge became valid.
216    pub valid_from: Timestamp,
217    /// Optional write-time provenance.
218    pub provenance: Option<Provenance>,
219    /// If set, close the edge's valid interval at this time after creation.
220    pub retract_at: Option<Timestamp>,
221}

223/// A fully-prepared source entity ready for replay.
224pub(crate) struct PreparedEntity {
225    /// The entity's business key (`:xt/id` / `:e`).
226    pub key: String,
227    /// The exact value stored under the id property — a `String` for XTDB, an
228    /// `Int` for Datomic. The durable re-import guard probes with this so the
229    /// property-value type matches how the importer stamped it.
230    pub id_value: PropertyValue,
231    /// Node versions, ascending by valid-time (create first, then updates).
232    pub versions: Vec<PreparedVersion>,
233    /// Ref edges sourced at this entity.
234    pub ref_edges: Vec<PreparedRefEdge>,
235    /// Optional whole-entity retraction (delete / interval close).
236    pub retraction: Option<PreparedRetraction>,
237}

239/// Replay prepared entities into the database in a two-phase pass: **all nodes
240/// first** (populating `key_to_id`), then ref edges, then retractions (so a
241/// detach retraction can co-retract the edges just created).
242///
243/// Refuses the whole run with an `AlreadyImported` error if any entity key is
244/// already present in this importer session (import into a fresh database — the
245/// documented idempotency contract, AC5).
246pub(crate) fn replay(
247    importer: &mut Importer<'_>,
248    entities: &[PreparedEntity],
249    report: &mut ImportFidelityReport,
250    id_property: &str,
251) -> Result<()> {
252    // Guard (AC5 — no silent duplication). Checked up front, before any write,
253    // so a re-import is refused atomically rather than partially duplicating.
254    //
255    // Two layers:
256    //   1. In-memory session guard — a second call on the same `Importer`.
257    //   2. **Durable** guard — probe the target database's current state for a
258    //      node already carrying this importer's business-key property, so a
259    //      fresh-process re-run against a persistent target is refused too (not
260    //      just a same-session repeat). The probe is a current-state
261    //      label+property lookup keyed on the exact `id_property` value the
262    //      importer stamps.
263    for entity in entities {
264        if importer.key_to_id.contains_key(&entity.key) {
265            return Err(Error::other(format!(
266                "AlreadyImported: business key '{}' is already present in this session; \
267                 history imports must run against a fresh database",
268                entity.key
269            )));
270        }
271        if let Some(v0) = entity.versions.first() {
272            let existing =
273                importer
274                    .db
275                    .find_nodes_by_property(&v0.label, id_property, &entity.id_value);
276            if !existing.is_empty() {
277                return Err(Error::other(format!(
278                    "AlreadyImported: the target database already contains a {} node with \
279                     {id_property}='{}'; history imports must run against a fresh database \
280                     (no silent duplication)",
281                    v0.label, entity.key
282                )));
283            }
284        }
285    }
286
287    // Phase 1: node creates + updates. Each version commits in its **own**
288    // transaction so it lands as a distinct bi-temporal version — replaying all
289    // versions of an entity in one transaction would collapse the intermediate
290    // valid-time segments into a single final version (supersession needs
291    // separate commits, exactly like a normal sequence of `update_node` calls).
292    for entity in entities {
293        if entity.versions.is_empty() {
294            continue;
295        }
296        let mut node_id: Option<NodeId> = None;
297        for version in &entity.versions {
298            let label = version.label.clone();
299            let properties = version.properties.clone();
300            let valid_from = version.valid_from;
301            let provenance = version.provenance.clone();
302
303            // The first version of an entity is a create; the rest are updates.
304            // `is_create` is authoritative (the importers mark exactly the head
305            // version), with `node_id.is_none()` as a defensive fallback.
306            if version.is_create || node_id.is_none() {
307                let nid = importer.db.write(move |tx| {
308                    use crate::api::transaction::{WriteOps, WriteRequestOptions};
309                    let mut options = WriteRequestOptions::new().with_valid_from(valid_from);
310                    if let Some(provenance) = provenance {
311                        options = options.with_provenance(provenance);
312                    }
313                    tx.create_node_with_options(&label, properties, options)
314                })?;
315                node_id = Some(nid);
316            } else if let Some(nid) = node_id {
317                // Subsequent version → update (supersession) in its own commit.
318                importer.db.write(move |tx| {
319                    use crate::api::transaction::{WriteOps, WriteRequestOptions};
320                    let mut options = WriteRequestOptions::new().with_valid_from(valid_from);
321                    if let Some(provenance) = provenance {
322                        options = options.with_provenance(provenance);
323                    }
324                    tx.update_node_with_options(nid, properties, options)?;
325                    Ok::<(), Error>(())
326                })?;
327            }
328        }
329
330        if let Some(nid) = node_id {
331            importer.key_to_id.insert(entity.key.clone(), nid);
332            report.nodes_created += 1;
333            report.node_versions_written += entity.versions.len();
334            report.properties_imported += entity
335                .versions
336                .iter()
337                .map(|v| v.property_count)
338                .sum::<usize>();
339        }
340    }
341
342    // Phase 2: ref edges (endpoints resolve against the fully-populated map).
343    let mut edge_seq = 0usize;
344    for entity in entities {
345        let Some(source) = importer.key_to_id.get(&entity.key).copied() else {
346            continue;
347        };
348        for edge in &entity.ref_edges {
349            edge_seq += 1;
350            let Some(target) = importer.key_to_id.get(&edge.target_key).copied() else {
351                let unresolved = UnresolvedEndpoint {
352                    row: edge_seq,
353                    key: edge.target_key.clone(),
354                    side: Endpoint::Target,
355                    file: None,
356                };
357                match importer.config.failure_mode {
358                    super::FailureMode::Abort => {
359                        return Err(super::ImportError::UnresolvedEndpoint(unresolved).into());
360                    }
361                    super::FailureMode::SkipAndReport => {
362                        report.unresolved_endpoints.push(unresolved);
363                        continue;
364                    }
365                }
366            };
367
368            let label = edge.label.clone();
369            let valid_from = edge.valid_from;
370            let provenance = edge.provenance.clone();
371            let props = PropertyMap::new();
372            let edge_id = importer.db.write(move |tx| {
373                use crate::api::transaction::{WriteOps, WriteRequestOptions};
374                let mut options = WriteRequestOptions::new().with_valid_from(valid_from);
375                if let Some(provenance) = provenance {
376                    options = options.with_provenance(provenance);
377                }
378                tx.create_edge_with_options(source, target, &label, props, options)
379            })?;
380            report.edges_created += 1;
381            report.note_mapping("ref", &edge.source_field, "edge", &edge.label);
382
383            if let Some(retract_at) = edge.retract_at {
384                importer.db.retract_edge_with_provenance(
385                    edge_id,
386                    retract_at,
387                    edge.provenance.clone(),
388                )?;
389                report.retractions += 1;
390            }
391        }
392    }
393
394    // Phase 3: node retractions (after edges, so detach co-retracts them).
395    for entity in entities {
396        let Some(retraction) = &entity.retraction else {
397            continue;
398        };
399        let Some(node_id) = importer.key_to_id.get(&entity.key).copied() else {
400            continue;
401        };
402        let provenance = retraction.provenance.clone();
403        let result = if retraction.detach {
404            importer.db.retract_node_detach_with_provenance(
405                node_id,
406                retraction.valid_to,
407                provenance,
408            )?
409        } else {
410            importer
411                .db
412                .retract_node_with_provenance(node_id, retraction.valid_to, provenance)?
413        };
414        if !result.already_retracted {
415            report.retractions += 1;
416            report.retractions += result.edges_retracted;
417        }
418    }
419
420    Ok(())
