1//! Streaming Parquet row reader for the bulk importer (Issue #3364).
2//!
3//! Produces the same `Result<Row, ImportError>` stream the CSV/JSONL readers in
4//! [`super::parse`] yield, so the format-agnostic import loop
5//! ([`Importer::import_nodes`](super::Importer) / `import_edges`) is reused verbatim
6//! — inheriting the #3211 abort/skip semantics, precise `row N:` errors,
7//! chunk-boundary partial-commit, null-as-absent, and per-row `valid_time` backfill.
8//!
9//! Columns are decoded **physically** from their Arrow array to a native
10//! [`PropertyValue`] (numbers, booleans, timestamps, and `list<float32>` embeddings
11//! never round-trip through a string), wrapped in [`Cell::Native`]. A SQL null cell
12//! decodes to [`PropertyValue::Null`], which the property builder treats as an absent
13//! property — matching the CSV blank-cell contract.
14//!
15//! Reading is chunked in Arrow `RecordBatch`es of [`PARQUET_BATCH_SIZE`] rows, so a
16//! multi-row-group file is streamed with bounded memory rather than fully
17//! materialized. A stable 1-based logical row index runs across every row group, so
18//! `row N:` error messages stay precise.
19

40/// Rows per Arrow `RecordBatch` decoded from the Parquet file. Bounds peak import
41/// memory to roughly one batch of decoded cells rather than the whole file.
42const PARQUET_BATCH_SIZE: usize = 8192;
43
44/// Microseconds in one day, used to widen `Date32` (days since epoch) to micros.
45const MICROS_PER_DAY: i64 = 86_400_000_000;
46
47/// File-metadata key stamped by the AletheiaDB exporter. Its presence marks a file as
48/// our own export, enabling auto-expansion of the `properties_json` overflow column.
49const EXPORT_SCHEMA_VERSION_KEY: &str = "aletheiadb.schema_version";
50
51/// The overflow column name written by the exporter (bytes / heterogeneous arrays /
52/// sparse vectors / type-conflicting keys, as reversible tagged JSON).
53const OVERFLOW_COLUMN: &str = "properties_json";
54
55/// Build a streaming row reader over a Parquet file.
56///
57/// The file is opened up front (open / metadata-parse failure is fatal, matching the
58/// CSV/JSONL readers); rows are then decoded lazily one `RecordBatch` at a time.
59pub(crate) fn parquet_rows(path: &Path) -> Result<RowIter, ImportError> {
60    let file = File::open(path)
61        .map_err(|e| ImportError::Io(format!("opening {}: {e}", path.display())))?;
62    let builder = ParquetRecordBatchReaderBuilder::try_new(file)
63        .map_err(|e| ImportError::Io(format!("opening parquet {}: {e}", path.display())))?;
64    let schema = builder.schema();
65    let field_names: Vec<String> = schema.fields().iter().map(|f| f.name().clone()).collect();
66    // Auto-expand the overflow column only for our own exports (identified by the
67    // schema-version file metadata) that actually carry a `properties_json` column, so a
68    // foreign file with a same-named column is never reinterpreted.
69    let expand_overflow = schema.metadata().contains_key(EXPORT_SCHEMA_VERSION_KEY)
70        && field_names.iter().any(|n| n == OVERFLOW_COLUMN);
71    let reader = builder
72        .with_batch_size(PARQUET_BATCH_SIZE)
73        .build()
74        .map_err(|e| ImportError::Io(format!("reading parquet {}: {e}", path.display())))?;
75
76    Ok(Box::new(ParquetRowIter {
77        reader,
78        field_names,
79        expand_overflow,
80        batch: None,
81        row_in_batch: 0,
82        logical_row: 0,
83        stopped: false,
84    }))
85}

87/// A streaming iterator over the rows of a Parquet file, one `RecordBatch` at a time.
88struct ParquetRowIter {
89    reader: ParquetRecordBatchReader,
90    /// Column names in schema order; the cell map is keyed by these.
91    field_names: Vec<String>,
92    /// Whether to auto-expand the `properties_json` overflow column into native
93    /// properties (true only for an AletheiaDB export carrying that column).
94    expand_overflow: bool,
95    /// The batch currently being drained, if any.
96    batch: Option<RecordBatch>,
97    /// Next row to emit within `batch`.
98    row_in_batch: usize,
99    /// 1-based logical row number across every row group (for `row N:` errors).
100    logical_row: usize,
101    /// Set once a fatal reader error has been surfaced, to stop iteration.
102    stopped: bool,
103}
104
105impl Iterator for ParquetRowIter {
106    type Item = Result<Row, ImportError>;
107
108    fn next(&mut self) -> Option<Self::Item> {
109        if self.stopped {
110            return None;
111        }
112
113        // Advance to a batch that still has rows to emit, pulling new batches as needed.
114        loop {
115            let needs_batch = match &self.batch {
116                Some(batch) => self.row_in_batch >= batch.num_rows(),
117                None => true,
118            };
119            if needs_batch {
120                match self.reader.next() {
121                    None => return None,
122                    Some(Ok(batch)) => {
123                        if batch.num_rows() == 0 {
124                            // Empty batch (possible for an empty row group): skip it.
125                            self.batch = None;
126                            continue;
127                        }
128                        self.batch = Some(batch);
129                        self.row_in_batch = 0;
130                    }
131                    Some(Err(e)) => {
132                        // A mid-stream reader error is fatal even in skip mode.
133                        self.stopped = true;
134                        return Some(Err(ImportError::Io(format!("reading parquet batch: {e}"))));
135                    }
136                }
137            } else {
138                break;
139            }
140        }
141
142        let batch = self.batch.as_ref()?;
143        let i = self.row_in_batch;
144        self.row_in_batch += 1;
145        self.logical_row += 1;
146        let row_num = self.logical_row;
147
148        let mut cells = HashMap::with_capacity(self.field_names.len());
149        for (col_idx, name) in self.field_names.iter().enumerate() {
150            let array = batch.column(col_idx);
151            match decode_value(array.as_ref(), i) {
152                Ok(value) => {
153                    cells.insert(name.clone(), Cell::Native(value));
154                }
155                Err(message) => {
156                    return Some(Err(ImportError::Row(RowError::new(
157                        row_num,
158                        format!("column '{name}': {message}"),
159                    ))));
160                }
161            }
162        }
163
164        let mut row = Row::new(row_num, cells);
165
166        // Auto-expand the overflow column of an AletheiaDB export back into native
167        // properties, so callers need not map the exotic keys explicitly.
168        if self.expand_overflow {
169            match decode_overflow(row.cells.get(OVERFLOW_COLUMN)) {
170                Ok(overflow) => row.overflow = overflow,
171                Err(message) => {
172                    return Some(Err(ImportError::Row(RowError::new(
173                        row_num,
174                        format!("column '{OVERFLOW_COLUMN}': {message}"),
175                    ))));
176                }
177            }
178        }
179
180        Some(Ok(row))
181    }
182}

184/// Decode the `properties_json` overflow cell of an AletheiaDB export into native
185/// key → [`PropertyValue`] pairs (the inverse of the exporter's tagged-JSON encoding).
186///
187/// An absent or SQL-null cell yields no properties. A malformed cell (non-string, or a
188/// string that is not a JSON object of tagged values) is a clean decode error carried up
189/// as a `row N:` [`RowError`], never a panic.
190fn decode_overflow(cell: Option<&Cell>) -> Result<Vec<(String, PropertyValue)>, String> {
191    let json = match cell {
192        None | Some(Cell::Native(PropertyValue::Null)) => return Ok(Vec::new()),
193        Some(Cell::Native(PropertyValue::String(s))) => s.to_string(),
194        Some(_) => return Err("expected a JSON string overflow column".to_string()),
195    };
196    let value: serde_json::Value =
197        serde_json::from_str(&json).map_err(|e| format!("invalid overflow JSON: {e}"))?;
198    let obj = value
199        .as_object()
200        .ok_or_else(|| "overflow column must be a JSON object".to_string())?;
201    let mut out = Vec::with_capacity(obj.len());
202    for (key, tagged) in obj {
203        let decoded = crate::api::export::overflow_json_to_value(tagged)?;
204        out.push((key.clone(), decoded));
205    }
206    Ok(out)
207}

209/// Downcast an Arrow array to a concrete typed array, erroring on mismatch.
210fn downcast<A: Array + 'static>(array: &dyn Array) -> Result<&A, String> {
211    array
212        .as_any()
213        .downcast_ref::<A>()
214        .ok_or_else(|| "internal error: unexpected Arrow array layout".to_string())
215}

217/// Physically decode the cell at `row` of an Arrow column into a [`PropertyValue`].
218///
219/// A null slot decodes to [`PropertyValue::Null`] (treated as an absent property by
220/// the caller). Timestamps and dates widen to microseconds since the Unix epoch and
221/// are stored as [`PropertyValue::Int`]; `list<float32>`/`list<float64>` become a
222/// [`PropertyValue::Vector`] embedding.
223fn decode_value(array: &dyn Array, row: usize) -> Result<PropertyValue, String> {
224    if array.is_null(row) {
225        return Ok(PropertyValue::Null);
226    }
227    match array.data_type() {
228        DataType::Boolean => Ok(PropertyValue::Bool(
229            downcast::<BooleanArray>(array)?.value(row),
230        )),
231        DataType::Int8 => Ok(PropertyValue::Int(
232            downcast::<Int8Array>(array)?.value(row) as i64
233        )),
234        DataType::Int16 => Ok(PropertyValue::Int(
235            downcast::<Int16Array>(array)?.value(row) as i64,
236        )),
237        DataType::Int32 => Ok(PropertyValue::Int(
238            downcast::<Int32Array>(array)?.value(row) as i64,
239        )),
240        DataType::Int64 => Ok(PropertyValue::Int(
241            downcast::<Int64Array>(array)?.value(row),
242        )),
243        DataType::UInt8 => Ok(PropertyValue::Int(
244            downcast::<UInt8Array>(array)?.value(row) as i64,
245        )),
246        DataType::UInt16 => Ok(PropertyValue::Int(
247            downcast::<UInt16Array>(array)?.value(row) as i64,
248        )),
249        DataType::UInt32 => Ok(PropertyValue::Int(
250            downcast::<UInt32Array>(array)?.value(row) as i64,
251        )),
252        DataType::UInt64 => {
253            let v = downcast::<UInt64Array>(array)?.value(row);
254            // A UInt64 above i64::MAX cannot be represented losslessly; reject rather
255            // than silently wrapping to a negative i64.
256            let as_i64 =
257                i64::try_from(v).map_err(|_| format!("UInt64 value {v} exceeds the i64 range"))?;
258            Ok(PropertyValue::Int(as_i64))
259        }
260        DataType::Float32 => Ok(PropertyValue::Float(
261            downcast::<Float32Array>(array)?.value(row) as f64,
262        )),
263        DataType::Float64 => Ok(PropertyValue::Float(
264            downcast::<Float64Array>(array)?.value(row),
265        )),
266        DataType::Utf8 => Ok(PropertyValue::string(
267            downcast::<StringArray>(array)?.value(row),
268        )),
269        DataType::LargeUtf8 => Ok(PropertyValue::string(
270            downcast::<LargeStringArray>(array)?.value(row),
271        )),
272        DataType::Timestamp(unit, _) => {
273            let micros = timestamp_to_micros(array, row, unit)?;
274            Ok(PropertyValue::Int(micros))
275        }
276        DataType::Date32 => {
277            let days = downcast::<Date32Array>(array)?.value(row) as i64;
278            // A hostile far-future/past date must not panic (debug) or wrap (release).
279            let micros = days.checked_mul(MICROS_PER_DAY).ok_or_else(|| {
280                format!(
281                    "date value {days} (days since epoch) out of representable microsecond range"
282                )
283            })?;
284            Ok(PropertyValue::Int(micros))
285        }
286        DataType::Date64 => {
287            let millis = downcast::<Date64Array>(array)?.value(row);
288            let micros = millis.checked_mul(1_000).ok_or_else(|| {
289                format!("date value {millis}ms out of representable microsecond range")
290            })?;
291            Ok(PropertyValue::Int(micros))
292        }
293        DataType::Binary => Ok(PropertyValue::bytes(
294            downcast::<BinaryArray>(array)?.value(row),
295        )),
296        DataType::LargeBinary => Ok(PropertyValue::bytes(
297            downcast::<LargeBinaryArray>(array)?.value(row),
298        )),
299        DataType::List(field) => decode_embedding::<i32>(array, row, field.data_type()),
300        DataType::LargeList(field) => decode_embedding::<i64>(array, row, field.data_type()),
301        other => Err(format!("unsupported Parquet column type {other}")),
302    }
303}

305/// Convert a timestamp value at `row` to microseconds since the Unix epoch.
306fn timestamp_to_micros(array: &dyn Array, row: usize, unit: &TimeUnit) -> Result<i64, String> {
307    let micros = match unit {
308        TimeUnit::Second => {
309            let secs = downcast::<TimestampSecondArray>(array)?.value(row);
310            secs.checked_mul(1_000_000).ok_or_else(|| {
311                format!("timestamp value {secs}s out of representable microsecond range")
312            })?
313        }
314        TimeUnit::Millisecond => {
315            let millis = downcast::<TimestampMillisecondArray>(array)?.value(row);
316            millis.checked_mul(1_000).ok_or_else(|| {
317                format!("timestamp value {millis}ms out of representable microsecond range")
318            })?
319        }
320        TimeUnit::Microsecond => downcast::<TimestampMicrosecondArray>(array)?.value(row),
321        TimeUnit::Nanosecond => downcast::<TimestampNanosecondArray>(array)?.value(row) / 1_000,
322    };
323    Ok(micros)
324}

326/// Decode a list cell into a [`PropertyValue::Vector`] embedding, requiring a
327/// float element type.
328fn decode_embedding<O: OffsetSizeTrait>(
329    array: &dyn Array,
330    row: usize,
331    element_type: &DataType,
332) -> Result<PropertyValue, String> {
333    match element_type {
334        DataType::Float32 | DataType::Float64 => {}
335        other => {
336            return Err(format!(
337                "unsupported list element type {other} (only float lists map to embeddings)"
338            ));
339        }
340    }
341    let list = downcast::<GenericListArray<O>>(array)?;
342    let values: ArrayRef = list.value(row);
343    let len = values.len();
344    let mut floats = Vec::with_capacity(len);
345    if let Some(f32s) = values.as_any().downcast_ref::<Float32Array>() {
346        for i in 0..len {
347            if f32s.is_null(i) {
348                return Err("embedding contains null element".to_string());
349            }
350            floats.push(f32s.value(i));
351        }
352    } else if let Some(f64s) = values.as_any().downcast_ref::<Float64Array>() {
353        for i in 0..len {
354            if f64s.is_null(i) {
355                return Err("embedding contains null element".to_string());
356            }
357            floats.push(f64s.value(i) as f32);
358        }
359    } else {
360        return Err("unsupported embedding element layout".to_string());
361    }
362    PropertyValue::try_vector(&floats).map_err(|e| e.to_string())
363}
