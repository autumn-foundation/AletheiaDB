1//! Bulk graph importer for CSV/JSONL with per-row valid-time backfill (Issue #3211).
2//!
3//! The importer loads a file of nodes and a file of edges into the graph in one pass.
4//! Compared to a hand-written per-record `create_node` / `create_edge` loop it:
5//!
6//! - commits **chunks** of rows in a single ACID transaction (never one commit per row),
7//!   routing the writes through the WAL `append_batch` path for higher throughput;
8//! - supports an **optional per-row `valid_time` column**, so historical facts are
9//!   backfilled and immediately queryable with `AS OF VALID_TIME`. When the column is
10//!   absent (or blank for a row) the row defaults to import (transaction) time;
11//! - resolves edges by a user-supplied **business key** (not internal `NodeId`) in one
12//!   pass; unresolved endpoints are **reported, not silently dropped**;
13//! - reports malformed rows with a precise `row N:` message, with a caller-selectable
14//!   [`FailureMode`] (abort on the first bad row, or skip-and-report).
15//!
16//! # Atomicity
17//!
18//! Each chunk of [`ImportConfig::batch_size`] rows commits as one ACID transaction. A
19//! multi-file import (nodes then edges) is therefore **not** a single global transaction:
20//! in [`FailureMode::Abort`] the first bad row returns an `Err` and no further chunks are
21//! committed, but any *already-committed* chunks persist. Callers needing all-or-nothing
22//! semantics should validate input first or run against a disposable database.
23//!
24//! # Example
25//!
26//! ```rust,no_run
27//! use aletheiadb::AletheiaDB;
28//! use aletheiadb::api::import::{ColumnType, EdgeMapping, LabelSource, NodeMapping};
29//!
30//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
31//! let db = AletheiaDB::new()?;
32//! let mut importer = db.import();
33//!
34//! let nodes = NodeMapping::new(LabelSource::fixed("Person"), "id")
35//!     .property("name", "name", ColumnType::String)
36//!     .valid_time_column("known_since");
37//! importer.nodes_from_csv("people.csv", nodes)?;
38//!
39//! let edges = EdgeMapping::new(LabelSource::fixed("KNOWS"), "src", "dst");
40//! importer.edges_from_csv("knows.csv", edges)?;
41//! # Ok(())
42//! # }
43//! ```

101/// How the importer reacts to a malformed row or an edge whose endpoint key cannot
102/// be resolved.
103#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
104pub enum FailureMode {
105    /// Abort the entire import on the first bad row, returning an `Err`.
106    #[default]
107    Abort,
108    /// Skip bad rows, collect them in the [`ImportReport`], and keep importing.
109    SkipAndReport,
110}

112/// Tunables for an import run.
113#[derive(Debug, Clone)]
114pub struct ImportConfig {
115    /// Number of rows committed per WAL batch transaction (default `1000`).
116    pub batch_size: usize,
117    /// How to react to malformed rows / unresolved endpoints (default [`FailureMode::Abort`]).
118    pub failure_mode: FailureMode,
119    /// Optional write-time [`Provenance`] bundle stamped on every imported
120    /// node/edge version (Issue #3224). When `None` (the default) the import
121    /// records no provenance, reproducing the historical behavior of the
122    /// CSV/JSONL importers exactly. The Neo4j importer (Issue #3356) sets this
123    /// to `neo4j-import::<file>`.
124    pub provenance: Option<Provenance>,
125}
126
127impl Default for ImportConfig {
128    fn default() -> Self {
129        ImportConfig {
130            batch_size: 1000,
131            failure_mode: FailureMode::Abort,
132            provenance: None,
133        }
134    }
135}

155/// A row that could not be imported, with its 1-based row/line number.
156#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize)]
157pub struct RowError {
158    /// The 1-based row (CSV) or line (JSONL) number.
159    pub row: usize,
160    /// A precise, human-readable reason.
161    pub message: String,
162    /// The source file the row came from, when a multi-file import needs to
163    /// disambiguate per-file row numbering (Issue #3356). `None` for the
164    /// single-file CSV/JSONL importers, which render as `row N: ...`.
165    #[serde(skip_serializing_if = "Option::is_none")]
166    pub file: Option<String>,
167}
168
169impl RowError {
170    /// Build a `RowError` with no source-file attribution (the single-file
171    /// importers' behavior).
172    pub(crate) fn new(row: usize, message: String) -> Self {
173        RowError {
174            row,
175            message,
176            file: None,
177        }
178    }
179}
180
181impl fmt::Display for RowError {
182    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
183        match &self.file {
184            Some(file) => write!(f, "{} row {}: {}", file, self.row, self.message),
185            None => write!(f, "row {}: {}", self.row, self.message),
186        }
187    }

225pub struct ImportReport {

239/// Errors raised by the bulk importer.
240#[derive(Debug)]
241pub enum ImportError {
242    /// A malformed row.
243    Row(RowError),
244    /// An edge endpoint that did not resolve to a node.
245    UnresolvedEndpoint(UnresolvedEndpoint),
246    /// An I/O error opening or reading the source file.
247    Io(String),
248}
249
250impl fmt::Display for ImportError {
251    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
252        match self {
253            ImportError::Row(e) => write!(f, "{e}"),
254            ImportError::UnresolvedEndpoint(e) => write!(f, "{e}"),
255            ImportError::Io(msg) => write!(f, "import I/O error: {msg}"),
256        }
257    }

292/// A stateful bulk importer obtained via [`AletheiaDB::import`].
293///
294/// The importer remembers the business-key → [`NodeId`] mapping built while importing
295/// nodes, so a subsequent `edges_from_*` call can resolve edge endpoints by key.
296pub struct Importer<'a> {
297    db: &'a AletheiaDB,
298    config: ImportConfig,
299    key_to_id: HashMap<String, NodeId>,
300}

302impl AletheiaDB {

304    ///
305    /// Returns a stateful [`Importer`] that resolves edge endpoints against nodes
306    /// imported earlier in the same session. See the [module docs](self) for details.
307    pub fn import(&self) -> Importer<'_> {
308        Importer {
309            db: self,
310            config: ImportConfig::default(),
311            key_to_id: HashMap::new(),
312        }
313    }

395    /// Shared node-import driver: chunked ACID commit + failure-mode handling,
396    /// parameterized by a per-row `prepare` closure that turns a raw [`Row`]
397    /// into a [`PreparedNode`]. Both the CSV/JSONL importers (Issue #3211) and
398    /// the Neo4j importer (Issue #3356) route through this so the commit path,
399    /// batching, provenance, and `row N:` error contract stay identical.
400    fn import_nodes_with<F>(&mut self, rows: RowIter, mut prepare: F) -> Result<ImportReport>
401    where
402        F: FnMut(&Row) -> std::result::Result<PreparedNode, RowError>,
403    {
404        let mut report = ImportReport::default();
405        let mut chunk: Vec<PreparedNode> = Vec::with_capacity(self.config.batch_size);
406
407        for row_result in rows {
408            report.rows_read += 1;
409            let row = match row_result {
410                Ok(row) => row,
411                Err(err) => {
412                    self.handle_row_failure(err, &mut report)?;
413                    continue;
414                }
415            };
416
417            match prepare(&row) {
418                Ok(prepared) => chunk.push(prepared),
419                Err(row_err) => match self.config.failure_mode {
420                    FailureMode::Abort => return Err(ImportError::Row(row_err).into()),
421                    FailureMode::SkipAndReport => report.skipped.push(row_err),
422                },
423            }
424
425            if chunk.len() >= self.config.batch_size {
426                self.commit_node_chunk(&mut chunk, &mut report)?;
427            }
428        }
429
430        self.commit_node_chunk(&mut chunk, &mut report)?;
431        Ok(report)
432    }
433
434    fn commit_node_chunk(
435        &mut self,
436        chunk: &mut Vec<PreparedNode>,
437        report: &mut ImportReport,
438    ) -> Result<()> {
439        if chunk.is_empty() {
440            return Ok(());
441        }
442        let prepared = std::mem::take(chunk);
443        let keys: Vec<String> = prepared.iter().map(|p| p.key.clone()).collect();
444        let provenance = self.config.provenance.clone();
445
446        // One ACID transaction for the whole chunk: writes flow through the WAL
447        // `append_batch` path (see `api::transaction::write::wal`).
448        let ids = self.db.write(move |tx| {
449            use crate::api::transaction::{WriteOps, WriteRequestOptions};
450            let mut ids = Vec::with_capacity(prepared.len());
451            for node in prepared {
452                let mut options = WriteRequestOptions::new();
453                if let Some(valid_from) = node.valid_from {
454                    options = options.with_valid_from(valid_from);
455                }
456                if let Some(provenance) = provenance.clone() {
457                    options = options.with_provenance(provenance);
458                }
459                let id = tx.create_node_with_options(&node.label, node.properties, options)?;
460                ids.push(id);
461            }
462            Ok::<Vec<NodeId>, Error>(ids)
463        })?;
464
465        for (key, id) in keys.into_iter().zip(ids) {
466            self.key_to_id.insert(key, id);
467            report.nodes_imported += 1;
468        }
469        Ok(())
470    }

509    fn import_edges(&mut self, rows: RowIter, mapping: &EdgeMapping) -> Result<ImportReport> {
510        self.import_edges_with(rows, |key_to_id, row| prepare_edge(key_to_id, row, mapping))

513    /// Shared edge-import driver: the edge counterpart to [`import_nodes_with`].
514    /// The `prepare` closure receives the accumulated business-key → [`NodeId`]
515    /// map so endpoints resolve against nodes imported earlier in the session.
516    fn import_edges_with<F>(&mut self, rows: RowIter, mut prepare: F) -> Result<ImportReport>
517    where
518        F: FnMut(
519            &HashMap<String, NodeId>,
520            &Row,
521        ) -> std::result::Result<PreparedEdge, EdgePrepError>,
522    {
523        let mut report = ImportReport::default();
524        let mut chunk: Vec<PreparedEdge> = Vec::with_capacity(self.config.batch_size);
525
526        for row_result in rows {
527            report.rows_read += 1;
528            let row = match row_result {
529                Ok(row) => row,
530                Err(err) => {
531                    self.handle_row_failure(err, &mut report)?;
532                    continue;
533                }
534            };
535
536            match prepare(&self.key_to_id, &row) {
537                Ok(prepared) => chunk.push(prepared),
538                Err(EdgePrepError::Row(row_err)) => match self.config.failure_mode {
539                    FailureMode::Abort => return Err(ImportError::Row(row_err).into()),
540                    FailureMode::SkipAndReport => report.skipped.push(row_err),
541                },
542                Err(EdgePrepError::Unresolved(endpoint)) => match self.config.failure_mode {
543                    FailureMode::Abort => {
544                        return Err(ImportError::UnresolvedEndpoint(endpoint).into());
545                    }
546                    FailureMode::SkipAndReport => report.unresolved_endpoints.push(endpoint),
547                },
548            }
549
550            if chunk.len() >= self.config.batch_size {
551                self.commit_edge_chunk(&mut chunk, &mut report)?;
552            }
553        }
554
555        self.commit_edge_chunk(&mut chunk, &mut report)?;
556        Ok(report)
557    }
558
559    fn commit_edge_chunk(
560        &mut self,
561        chunk: &mut Vec<PreparedEdge>,
562        report: &mut ImportReport,
563    ) -> Result<()> {
564        if chunk.is_empty() {
565            return Ok(());
566        }
567        let prepared = std::mem::take(chunk);
568        let count = prepared.len();
569        let provenance = self.config.provenance.clone();
570
571        self.db.write(move |tx| {
572            use crate::api::transaction::{WriteOps, WriteRequestOptions};
573            for edge in prepared {
574                let mut options = WriteRequestOptions::new();
575                if let Some(valid_from) = edge.valid_from {
576                    options = options.with_valid_from(valid_from);
577                }
578                if let Some(provenance) = provenance.clone() {
579                    options = options.with_provenance(provenance);
580                }
581                tx.create_edge_with_options(
582                    edge.source,
583                    edge.target,
584                    &edge.label,
585                    edge.properties,
586                    options,
587                )?;
588            }
589            Ok::<(), Error>(())
590        })?;
591
592        report.edges_imported += count;
593        Ok(())
594    }
595
596    /// Apply the failure mode to a row-level reader/parse error.
597    fn handle_row_failure(&self, err: ImportError, report: &mut ImportReport) -> Result<()> {
598        match self.config.failure_mode {
599            FailureMode::Abort => Err(err.into()),
600            FailureMode::SkipAndReport => {
601                match err {
602                    ImportError::Row(row_err) => report.skipped.push(row_err),
603                    ImportError::UnresolvedEndpoint(endpoint) => {
604                        report.unresolved_endpoints.push(endpoint)
605                    }
606                    // An I/O error mid-stream is fatal even in skip mode.
607                    ImportError::Io(_) => return Err(err.into()),
608                }
609                Ok(())
610            }
611        }
612    }
613}

615/// Resolve a label from a row per the [`LabelSource`].
616fn resolve_label(row: &Row, source: &LabelSource) -> std::result::Result<String, RowError> {
617    match source {
618        LabelSource::Fixed(label) => Ok(label.clone()),
619        LabelSource::Column(column) => {
620            let value = row
621                .get_str(column)
622                .map_err(|message| RowError::new(row.index, message))?;
623            if value.trim().is_empty() {
624                return Err(RowError::new(
625                    row.index,
626                    format!("label column '{column}' is empty"),
627                ));
628            }
629            Ok(value)
630        }
631    }
632}

634/// Build a [`PropertyMap`] from the mapping's property columns.
635fn build_properties(
636    row: &Row,
637    properties: &[PropertyMapping],
638) -> std::result::Result<PropertyMap, RowError> {
639    let mut builder = PropertyMapBuilder::new();
640    // Names inserted from explicit mapping columns; they take precedence over any
641    // auto-expanded overflow key of the same name (Parquet import only).
642    #[cfg(feature = "parquet")]
643    let mut explicit: std::collections::HashSet<String> = std::collections::HashSet::new();
644    for mapping in properties {
645        let cell = row.cells.get(&mapping.column).ok_or_else(|| {
646            RowError::new(row.index, format!("missing column '{}'", mapping.column))
647        })?;
648        let value = parse::coerce(cell, mapping.ty).map_err(|message| {
649            RowError::new(row.index, format!("column '{}': {message}", mapping.column))
650        })?;
651        #[cfg(feature = "parquet")]
652        explicit.insert(mapping.name.clone());
653        // Skip nulls so blank cells become absent properties rather than stored nulls.
654        if matches!(value, PropertyValue::Null) {
655            continue;
656        }
657        builder = builder
658            .try_insert(&mapping.name, value)
659            .map_err(|e| RowError::new(row.index, e.to_string()))?;
660    }
661    // Merge overflow properties auto-expanded from a `properties_json` column of an
662    // AletheiaDB export, so the caller need not map the exotic keys explicitly.
663    #[cfg(feature = "parquet")]
664    for (key, value) in &row.overflow {
665        if explicit.contains(key) || matches!(value, PropertyValue::Null) {
666            continue;
667        }
668        builder = builder
669            .try_insert(key, value.clone())
670            .map_err(|e| RowError::new(row.index, e.to_string()))?;
671    }
672    Ok(builder.build())
673}
674
675/// Extract an optional valid time from the named column.
676fn extract_valid_time(
677    row: &Row,
678    column: &Option<String>,
679) -> std::result::Result<Option<Timestamp>, RowError> {
680    let Some(column) = column else {
681        return Ok(None);
682    };
683    let raw = row
684        .get_str(column)
685        .map_err(|message| RowError::new(row.index, message))?;
686    if raw.trim().is_empty() {
687        return Ok(None);
688    }
689    let ts = parse::parse_valid_time(&raw).map_err(|message| RowError::new(row.index, message))?;
690    Ok(Some(ts))
691}
692
693fn prepare_node(row: &Row, mapping: &NodeMapping) -> std::result::Result<PreparedNode, RowError> {
694    let label = resolve_label(row, &mapping.label)?;
695    let key = row
696        .get_str(&mapping.key_column)
697        .map_err(|message| RowError::new(row.index, message))?;
698    if key.trim().is_empty() {
699        return Err(RowError::new(
700            row.index,
701            format!("key column '{}' is empty", mapping.key_column),
702        ));
703    }
704    let properties = build_properties(row, &mapping.properties)?;
705    let valid_from = extract_valid_time(row, &mapping.valid_time_column)?;
706    Ok(PreparedNode {
707        key,
708        label,
709        properties,
710        valid_from,
711    })
712}

714fn prepare_edge(
715    key_to_id: &HashMap<String, NodeId>,
716    row: &Row,
717    mapping: &EdgeMapping,
718) -> std::result::Result<PreparedEdge, EdgePrepError> {
719    let label = resolve_label(row, &mapping.label).map_err(EdgePrepError::Row)?;
720
721    let source_key = row
722        .get_str(&mapping.source_key_column)
723        .map_err(|message| EdgePrepError::Row(RowError::new(row.index, message)))?;
724    let target_key = row
725        .get_str(&mapping.target_key_column)
726        .map_err(|message| EdgePrepError::Row(RowError::new(row.index, message)))?;
727
728    let source = key_to_id.get(&source_key).copied().ok_or_else(|| {
729        EdgePrepError::Unresolved(UnresolvedEndpoint {
730            row: row.index,
731            key: source_key.clone(),
732            side: Endpoint::Source,
733            file: None,
734        })
735    })?;
736    let target = key_to_id.get(&target_key).copied().ok_or_else(|| {
737        EdgePrepError::Unresolved(UnresolvedEndpoint {
738            row: row.index,
739            key: target_key.clone(),
740            side: Endpoint::Target,
741            file: None,
742        })
743    })?;
744
745    let properties = build_properties(row, &mapping.properties).map_err(EdgePrepError::Row)?;
746    let valid_from =
747        extract_valid_time(row, &mapping.valid_time_column).map_err(EdgePrepError::Row)?;
748
749    Ok(PreparedEdge {
750        source,
751        target,
752        label,
753        properties,
754        valid_from,
755    })
756}
