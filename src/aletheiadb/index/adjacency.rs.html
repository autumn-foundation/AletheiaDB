1//! CSR (Compressed Sparse Row) adjacency index for graph traversals.
2//!
3//! This module implements a cache-friendly adjacency list representation using
4//! the Compressed Sparse Row format. This enables fast sequential access during
5//! graph traversals with minimal cache misses.
6//!
7//! # CSR Format
8//!
9//! The CSR format stores the graph as two arrays:
10//! - `offsets` where `offsets[i]` is the starting position in `edges` for node i's adjacency list
11//! - `edges`: Flat array of (target_node, edge_id, edge_label) tuples
12//!
13//! This layout is cache-friendly because traversing from a node requires
14//! sequential access to a contiguous region of memory.
15

19use rayon::prelude::*;

22/// A single entry in the adjacency list.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub struct AdjacencyEntry {
25    /// Target node ID.
26    pub target: NodeId,
27    /// Edge ID connecting source to target.
28    pub edge_id: EdgeId,
29    /// Edge label (interned for memory efficiency).
30    pub label: InternedString,
31}

35    #[inline]
36    pub const fn new(target: NodeId, edge_id: EdgeId, label: InternedString) -> Self {
37        AdjacencyEntry {
38            target,
39            edge_id,
40            label,
41        }
42    }

45/// Compressed Sparse Row adjacency index.
46///
47/// This structure provides O(log n) access to a node's adjacency list with
48/// excellent cache locality and memory efficiency for sparse node IDs.
49///
50/// The index uses a sparse representation where only nodes with outgoing edges
51/// are stored, making it efficient even with large gaps in node IDs (e.g., after deletions).
52#[derive(Debug, Clone)]
53pub struct AdjacencyIndex {
54    /// Sorted list of node IDs that have outgoing edges.
55    /// Used for binary search to map node_id -> index in offsets array.
56    node_ids: Vec<NodeId>,
57    /// Offsets into the edges array for each node in node_ids.
58    /// `offsets[i]` = start index in edges array for `node_ids[i]`
59    /// `offsets[i + 1]` = end index (exclusive)
60    offsets: Vec<usize>,
61    /// Flat array of adjacency entries, sorted by source node.
62    edges: Vec<AdjacencyEntry>,
63    /// Maximum node ID (for bounds checking).
64    max_node_id: u64,
65}
66
67impl AdjacencyIndex {
68    /// Export CSR data for persistence.
69    ///
70    /// The CSR structure is decomposed into three raw arrays suitable for fast
71    /// binary serialization to disk. This is heavily utilized by the persistence
72    /// engine to avoid serializing rust-specific enum wrappers or iterating over
73    /// complex graphs.
74    ///
75    /// Returns a tuple `(node_ids, offsets, edge_ids)` where:
76    /// - `node_ids`: Sorted array of `NodeId`s that have outgoing edges.
77    /// - `offsets`: The CSR offset array defining edge boundaries per node.
78    /// - `edge_ids`: Flat array of all outgoing `EdgeId`s in traversal order.
79    ///
80    /// ## Examples
81    ///
82    /// ```rust
83    /// use aletheiadb::core::id::{NodeId, EdgeId};
84    /// use aletheiadb::index::AdjacencyIndex;
85    /// use aletheiadb::core::interning::GLOBAL_INTERNER;
86    ///
87    /// let label = GLOBAL_INTERNER.intern("KNOWS").unwrap();
88    /// let edges = vec![
89    ///     (NodeId::new(1).unwrap(), NodeId::new(2).unwrap(), EdgeId::new(100).unwrap(), label)
90    /// ];
91    ///
92    /// let index = AdjacencyIndex::build(edges);
93    /// let (nodes, offsets, edges_out) = index.export_csr();
94    ///
95    /// assert_eq!(nodes, vec![1]);
96    /// assert_eq!(offsets, vec![0, 1]);
97    /// assert_eq!(edges_out, vec![100]);
98    /// ```
99    pub fn export_csr(&self) -> (Vec<u64>, Vec<u64>, Vec<u64>) {
100        let node_ids = self.node_ids.iter().map(|n| n.as_u64()).collect();
101        let offsets = self.offsets.iter().map(|&x| x as u64).collect();
102        let edge_ids = self.edges.iter().map(|e| e.edge_id.as_u64()).collect();
103        (node_ids, offsets, edge_ids)
104    }
105
106    /// Import CSR data from persistence, reconstructing adjacency entries from edges.
107    ///
108    /// Re-hydrates a CSR structure from its raw binary components. This reconstructs
109    /// the full `AdjacencyEntry` data by looking up the edge metadata (target, label)
110    /// in the provided `edges_map`.
111    ///
112    /// This method is highly optimized and performs zero-copy vector transmutations
113    /// where possible (such as on 64-bit systems converting `u64` to `usize`).
114    ///
115    /// # Arguments
116    /// * `node_ids` - Sorted array of node IDs that have outgoing edges.
117    /// * `offsets` - CSR offset array defining edge boundaries per node.
118    /// * `edge_ids` - Flat array of edge IDs corresponding to the offsets.
119    /// * `edges_map` - Map from `EdgeId` to `(target, label)` for full reconstruction.
120    ///
121    /// ## Panics
122    ///
123    /// Panics if the provided CSR invariants are violated (e.g., offsets array length mismatch,
124    /// non-monotonic sequences, or invalid bounds) to prevent corrupted database state.
125    ///
126    /// ## Examples
127    ///
128    /// ```rust
129    /// use aletheiadb::core::id::{NodeId, EdgeId};
130    /// use aletheiadb::index::AdjacencyIndex;
131    /// use aletheiadb::core::interning::GLOBAL_INTERNER;
132    /// use aletheiadb::core::hasher::IdentityHasher;
133    /// use std::collections::HashMap;
134    /// use std::hash::BuildHasherDefault;
135    ///
136    /// let nodes = vec![1];
137    /// let offsets = vec![0, 1];
138    /// let edge_ids = vec![100];
139    ///
140    /// let mut edge_map = HashMap::with_hasher(BuildHasherDefault::<IdentityHasher>::default());
141    /// let label = GLOBAL_INTERNER.intern("KNOWS").unwrap();
142    /// edge_map.insert(EdgeId::new(100).unwrap(), (NodeId::new(2).unwrap(), label));
143    ///
144    /// let index = AdjacencyIndex::import_csr(nodes, offsets, edge_ids, &edge_map);
145    /// assert_eq!(index.edge_count(), 1);
146    /// ```
147    pub fn import_csr(
148        node_ids: Vec<u64>,
149        offsets: Vec<u64>,
150        edge_ids: Vec<u64>,
151        edges_map: &std::collections::HashMap<
152            EdgeId,
153            (NodeId, InternedString),
154            BuildHasherDefault<IdentityHasher>,
155        >,
156    ) -> Self {
157        if offsets.is_empty() || edge_ids.is_empty() {
158            return Self::new();
159        }
160
161        // Validate CSR invariants
162        Self::validate_csr_invariants(&node_ids, &offsets, &edge_ids).unwrap();
163
164        let max_node_id = node_ids.iter().max().copied().unwrap_or(0);
165
166        // Zero-copy conversion: NodeId(u64) has same layout as u64
167        let node_ids_typed: Vec<NodeId> = bytemuck::cast_vec(node_ids);
168
169        // Convert offsets (zero-copy on 64-bit, allocating on 32-bit)
170        let offsets_usize = Self::convert_offsets(offsets);
171
172        let mut adjacency_entries = Vec::with_capacity(edge_ids.len());
173
174        for &edge_id_u64 in &edge_ids {
175            let edge_id = EdgeId::new_unchecked(edge_id_u64);
176            if let Some((target, label)) = edges_map.get(&edge_id) {
177                adjacency_entries.push(AdjacencyEntry::new(*target, edge_id, *label));
178            } else {
179                // Edge not found - this shouldn't happen with valid data
180                // Use a placeholder to maintain CSR structure integrity
181                adjacency_entries.push(AdjacencyEntry::new(
182                    NodeId::new(0).unwrap(),
183                    edge_id,
184                    InternedString::from_raw(0),
185                ));
186            }
187        }
188
189        Self {
190            node_ids: node_ids_typed,
191            offsets: offsets_usize,
192            edges: adjacency_entries,
193            max_node_id,
194        }
195    }
196}

199    /// Create a new empty adjacency index.
200    ///
201    /// Initializes an empty CSR structure that allocates no heap memory until
202    /// edges are explicitly added via building or importing.
203    ///
204    /// ## Examples
205    ///
206    /// ```rust
207    /// use aletheiadb::index::AdjacencyIndex;
208    ///
209    /// let index = AdjacencyIndex::new();
210    /// assert_eq!(index.node_count(), 0);
211    /// assert_eq!(index.edge_count(), 0);
212    /// ```
213    pub fn new() -> Self {
214        AdjacencyIndex {
215            node_ids: Vec::new(),
216            offsets: vec![0],
217            edges: Vec::new(),
218            max_node_id: 0,
219        }
220    }

222    /// Build an adjacency index from a list of edges.
223    ///
224    /// Accepts a flat list of edges and dynamically constructs the sparse CSR representation.
225    /// The input is automatically sorted in parallel by `(source, target, edge_id)` to ensure
226    /// deterministic adjacency lists and correct offset calculation.
227    ///
228    /// Edges should be provided as `(source, target, edge_id, label)` tuples.
229    ///
230    /// This uses a sparse representation: only nodes with outgoing edges are stored. This makes
231    /// it extremely memory efficient even with large gaps in node IDs (O(num_nodes) instead of O(max_node_id)).
232    ///
233    /// ## Examples
234    ///
235    /// ```rust
236    /// use aletheiadb::core::id::{NodeId, EdgeId};
237    /// use aletheiadb::index::AdjacencyIndex;
238    /// use aletheiadb::core::interning::GLOBAL_INTERNER;
239    ///
240    /// let label = GLOBAL_INTERNER.intern("KNOWS").unwrap();
241    /// let edges = vec![
242    ///     (NodeId::new(1).unwrap(), NodeId::new(2).unwrap(), EdgeId::new(100).unwrap(), label),
243    ///     (NodeId::new(1).unwrap(), NodeId::new(3).unwrap(), EdgeId::new(101).unwrap(), label),
244    /// ];
245    ///
246    /// let index = AdjacencyIndex::build(edges);
247    /// assert_eq!(index.degree(NodeId::new(1).unwrap()), 2);
248    /// ```
249    pub fn build(mut edges: Vec<(NodeId, NodeId, EdgeId, InternedString)>) -> Self {
250        if edges.is_empty() {
251            return Self::new();
252        }
253
254        let edge_count = edges.len();
255
256        // Sort by source node, then target node for deterministic ordering.
257        // We use parallel sort for performance on large graphs.
258        // We include edge_id for canonical deterministic ordering.
259        edges.par_sort_unstable_by_key(|(src, target, edge_id, _)| (*src, *target, *edge_id));
260
261        // Pre-allocate assuming some average degree > 1 to avoid resizing
262        let estimated_nodes = (edge_count / 4).max(16);
263        let mut node_ids = Vec::with_capacity(estimated_nodes);
264        let mut offsets = Vec::with_capacity(estimated_nodes + 1);
265        let mut flat_edges = Vec::with_capacity(edge_count);
266        let mut max_node_id = 0;
267
268        offsets.push(0);
269
270        if !edges.is_empty() {
271            let mut current_source = edges[0].0;
272            node_ids.push(current_source);
273
274            for (source, target, edge_id, label) in edges {
275                let src_val = source.as_u64();
276                let tgt_val = target.as_u64();
277                if src_val > max_node_id {
278                    max_node_id = src_val;
279                }
280                if tgt_val > max_node_id {
281                    max_node_id = tgt_val;
282                }
283
284                if source != current_source {
285                    offsets.push(flat_edges.len());
286                    current_source = source;
287                    node_ids.push(current_source);
288                }
289                flat_edges.push(AdjacencyEntry::new(target, edge_id, label));
290            }
291            offsets.push(flat_edges.len());
292        }
293
294        // Optimize memory usage by releasing unused capacity
295        node_ids.shrink_to_fit();
296        offsets.shrink_to_fit();
297
298        AdjacencyIndex {
299            node_ids,
300            offsets,
301            edges: flat_edges,
302            max_node_id,
303        }
304    }

306    /// Get the adjacency list for a node.
307    ///
308    /// Returns a sequential, cache-friendly slice of all outgoing edges originating
309    /// from the specified node. Because the CSR edges are stored contiguously, iterating
310    /// through this slice ensures near-zero cache misses during graph traversal.
311    ///
312    /// Returns an empty slice if the node has no outgoing edges.
313    ///
314    /// This uses a fast binary search over the sparse `node_ids` array to locate the node,
315    /// providing O(log n) lookup time where `n` is the number of nodes with outgoing edges.
316    ///
317    /// ## Examples
318    ///
319    /// ```rust
320    /// use aletheiadb::core::id::{NodeId, EdgeId};
321    /// use aletheiadb::index::AdjacencyIndex;
322    /// use aletheiadb::core::interning::GLOBAL_INTERNER;
323    ///
324    /// let label = GLOBAL_INTERNER.intern("KNOWS").unwrap();
325    /// let edges = vec![
326    ///     (NodeId::new(1).unwrap(), NodeId::new(2).unwrap(), EdgeId::new(100).unwrap(), label)
327    /// ];
328    ///
329    /// let index = AdjacencyIndex::build(edges);
330    /// let adj = index.get_adjacency(NodeId::new(1).unwrap());
331    ///
332    /// assert_eq!(adj.len(), 1);
333    /// assert_eq!(adj[0].target, NodeId::new(2).unwrap());
334    /// ```
335    #[inline]
336    pub fn get_adjacency(&self, node: NodeId) -> &[AdjacencyEntry] {
337        // Binary search to find the node's index in node_ids
338        match self.node_ids.binary_search(&node) {
339            Ok(idx) => {
340                let start = self.offsets[idx];
341                let end = self.offsets[idx + 1];
342                &self.edges[start..end]
343            }
344            Err(_) => {
345                // Node not found (no outgoing edges)
346                &[]
347            }
348        }
349    }

351    /// Get outgoing edges for a node with a specific label.
352    ///
353    /// Performs an `O(log N) + O(E)` traversal where `N` is the number of nodes with
354    /// outgoing edges, and `E` is the degree of the given node. It yields an iterator
355    /// over only the adjacency entries that possess the specified `InternedString` label.
356    ///
357    /// ## Examples
358    ///
359    /// ```rust
360    /// use aletheiadb::core::id::{NodeId, EdgeId};
361    /// use aletheiadb::index::AdjacencyIndex;
362    /// use aletheiadb::core::interning::GLOBAL_INTERNER;
363    ///
364    /// let knows = GLOBAL_INTERNER.intern("KNOWS").unwrap();
365    /// let follows = GLOBAL_INTERNER.intern("FOLLOWS").unwrap();
366    /// let edges = vec![
367    ///     (NodeId::new(1).unwrap(), NodeId::new(2).unwrap(), EdgeId::new(100).unwrap(), knows),
368    ///     (NodeId::new(1).unwrap(), NodeId::new(3).unwrap(), EdgeId::new(101).unwrap(), follows),
369    /// ];
370    ///
371    /// let index = AdjacencyIndex::build(edges);
372    /// let knows_edges: Vec<_> = index.get_adjacency_with_label(NodeId::new(1).unwrap(), knows).collect();
373    ///
374    /// assert_eq!(knows_edges.len(), 1);
375    /// assert_eq!(knows_edges[0].target, NodeId::new(2).unwrap());
376    /// ```
377    pub fn get_adjacency_with_label(
378        &self,
379        node: NodeId,
380        label: InternedString,
381    ) -> impl Iterator<Item = &AdjacencyEntry> {
382        self.get_adjacency(node)
383            .iter()
384            .filter(move |entry| entry.label == label)
385    }

544    /// Validate CSR invariants.
545    fn validate_csr_invariants(
546        node_ids: &[u64],
547        offsets: &[u64],
548        edge_ids: &[u64],
549    ) -> Result<(), String> {
550        if offsets.len() != node_ids.len() + 1 {
551            return Err(format!(
552                "CSR offsets length mismatch: expected {}, got {}",
553                node_ids.len() + 1,
554                offsets.len()
555            ));
556        }
557
558        #[allow(clippy::collapsible_if)]
559        if let Some(&first_offset) = offsets.first() {
560            if first_offset != 0 {
561                return Err(format!(
562                    "CSR first offset mismatch: expected 0, got {}",
563                    first_offset
564                ));
565            }
566        }
567
568        for window in offsets.windows(2) {
569            if window[0] > window[1] {
570                return Err(format!(
571                    "CSR offsets are not monotonically increasing: {} > {}",
572                    window[0], window[1]
573                ));
574            }
575        }
576
577        for window in node_ids.windows(2) {
578            if window[0] >= window[1] {
579                return Err(format!(
580                    "CSR node_ids are not strictly monotonically increasing: {} >= {}",
581                    window[0], window[1]
582                ));
583            }
584        }
585
586        #[allow(clippy::collapsible_if)]
587        if let Some(&last_offset) = offsets.last() {
588            if last_offset != edge_ids.len() as u64 {
589                return Err(format!(
590                    "CSR last offset mismatch: expected {}, got {}",
591                    edge_ids.len(),
592                    last_offset
593                ));
594            }
595        }
596
597        Ok(())
598    }

600    /// Convert offsets to usize vector.
601    ///
602    /// On 64-bit systems, this is a zero-copy operation because usize == u64.
603    /// On 32-bit systems, this allocates a new vector because usize == u32 != u64.
604    fn convert_offsets(offsets: Vec<u64>) -> Vec<usize> {
605        #[cfg(target_pointer_width = "64")]
606        {
607            bytemuck::cast_vec(offsets)
608        }
609
610        #[cfg(not(target_pointer_width = "64"))]
611        {
612            offsets.iter().map(|&x| x as usize).collect()
613        }
614    }

963            let edge_id = EdgeId::new(i as u64).unwrap();
