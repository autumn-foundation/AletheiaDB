1//! Vector index abstraction for approximate nearest neighbor search.
2//!
3//! This module provides a trait-based abstraction for vector indexes, enabling multiple
4//! implementation strategies (HNSW, IVF, etc.) while maintaining a consistent interface
5//! for the query engine.
6//!
7//! For the complete vector search architecture and implementation plan, see
8//! [`docs/VECTOR_SEARCH_DESIGN.md`](../../docs/VECTOR_SEARCH_DESIGN.md).
9//!
10//! # Overview
11//!
12//! The `VectorIndex` trait defines the core operations for managing and querying vector
13//! embeddings:
14//! - **Adding vectors**: Associate embeddings with node IDs
15//! - **Removing vectors**: Delete embeddings from the index
16//! - **Searching**: Find k-nearest neighbors by similarity
17//! - **Filtered search**: Search with custom predicates
18//! - **Distance metric**: Query which similarity metric is used
19//!
20//! # Implementation Strategies
21//!
22//! Different index implementations offer various trade-offs:
23//!
24//! | Strategy | Build Time | Query Time (avg) | Memory | Use Case |
25//! |----------|-----------|------------------|---------|----------|
26//! | HNSW | O(n log n) | O(log n) | High | General purpose, high recall |
27//! | IVF | O(n) | O(√n) | Medium | Large datasets, approximate |
28//! | Flat | O(1) | O(n) | Low | Small datasets, exact search |
29//!
30//! **Note**: HNSW worst-case query time is O(n), though average case is O(log n).
31//! Performance also depends on vector dimensionality.
32//!
33//! # Input Validation Requirements
34//!
35//! Implementations must validate all inputs to prevent invalid state and DoS attacks:
36//!
37//! - **Vector validation**: Check for NaN/Infinity using `validate_vector()` from `core::vector`
38//! - **Dimension matching**: Ensure all vectors match index dimensionality
39//! - **k bounds**: Limit `k` parameter (e.g., max 10,000) to prevent excessive memory allocation
40//! - **Empty vectors**: Reject zero-length vectors
41//!
42//! Violations should return specific errors (see Error Handling below).
43//!
44//! # Examples
45//!
46//! ```rust
47//! use aletheiadb::index::VectorIndex;
48//! use aletheiadb::core::id::NodeId;
49//!
50//! fn search_similar_documents(
51//!     index: &impl VectorIndex,
52//!     query_embedding: &[f32],
53//!     limit: usize
54//! ) -> aletheiadb::core::error::Result<Vec<(NodeId, f32)>> {
55//!     // Find top-k most similar documents
56//!     let results = index.search(query_embedding, limit)?;
57//!
58//!     // Results are sorted by similarity (highest first)
59//!     for (node_id, similarity) in &results {
60//!         println!("Node {:?}: similarity = {}", node_id, similarity);
61//!     }
62//!
63//!     Ok(results)
64//! }
65//!
66//! fn search_with_constraint(
67//!     index: &impl VectorIndex,
68//!     query: &[f32],
69//!     allowed_ids: &[NodeId],
70//!     k: usize
71//! ) -> aletheiadb::core::error::Result<Vec<(NodeId, f32)>> {
72//!     // Search only within a subset of nodes
73//!     index.search_with_filter(query, k, |id| allowed_ids.contains(id))
74//! }
75//! ```
76//!
77//! # Implementation
78//!
79//! AletheiaDB implements this trait using HNSW (Hierarchical Navigable Small World)
80//! via the `usearch` crate. This implementation provides:
81//!
82//! - **High Performance**: 100µs-1ms typical query latency
83//! - **High Recall**: >95% for typical configurations
84//! - **Memory Efficiency**: Memory-efficient graph structure with optional quantization
85//! - **Concurrency**: Thread-safe insertions and searches via internal locking
86//!
87//! See [`docs/VECTOR_SEARCH_DESIGN.md`](../../docs/VECTOR_SEARCH_DESIGN.md) for complete architecture.

95/// Quantization level for vector storage.
96///
97/// Lower precision reduces memory usage but may impact recall slightly.
98/// - F32: Full precision (default), no recall impact
99/// - F16: Half precision, ~2x memory savings, <1% recall impact typical
100/// - I8: Quarter precision, ~4x memory savings, 1-3% recall impact typical
101#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
102pub enum Quantization {
103    /// 32-bit floating point (default, full precision)
104    #[default]
105    F32,
106    /// 16-bit floating point (half precision, ~2x memory savings)
107    F16,
108    /// 8-bit signed integer (quarter precision, ~4x memory savings)
109    I8,
110}
111
112impl Quantization {
113    /// Encode quantization level as a byte for serialization.
114    ///
115    /// Encoding:
116    /// - 0 = F32
117    /// - 1 = F16
118    /// - 2 = I8
119    pub fn to_u8(self) -> u8 {
120        match self {
121            Quantization::F32 => 0,
122            Quantization::F16 => 1,
123            Quantization::I8 => 2,
124        }
125    }
126
127    /// Decode quantization level from a byte.
128    ///
129    /// # Errors
130    ///
131    /// Returns an error if the byte value is not a valid quantization encoding (>= 3).
132    pub fn from_u8(value: u8) -> Result<Self> {
133        match value {
134            0 => Ok(Quantization::F32),
135            1 => Ok(Quantization::F16),
136            2 => Ok(Quantization::I8),
137            _ => Err(crate::core::error::StorageError::CorruptedData(format!(
138                "Invalid quantization encoding: {}",
139                value
140            ))
141            .into()),
142        }
143    }
144}

201/// Distance metric used for similarity computation.
202///
203/// Different metrics are suitable for different use cases:
204/// - **Cosine**: Measures angle between vectors, ignores magnitude (semantic similarity)
205/// - **Euclidean**: Measures straight-line distance (spatial data, clustering)
206/// - **DotProduct**: Inner product, preserves magnitude (MaxSim, ColBERT)
207#[derive(Debug, Clone, Copy, PartialEq, Eq)]
208pub enum DistanceMetric {
209    /// Cosine similarity: measures angle between vectors, range [-1, 1]
210    Cosine,
211    /// Euclidean distance (L2): measures straight-line distance, range [0, ∞)
212    Euclidean,
213    /// Dot product: inner product of vectors, range (-∞, ∞)
214    DotProduct,
215    /// Haversine: great circle distance for geographic coordinates
216    Haversine,
217    /// Hamming: bit-level distance for binary vectors
218    Hamming,
219    /// Tanimoto: bit-level Jaccard similarity for chemical fingerprints
220    Tanimoto,
221}
222impl DistanceMetric {
223    /// Encode distance metric as a byte for serialization.
224    ///
225    /// Encoding:
226    /// - 0 = Cosine
227    /// - 1 = Euclidean
228    /// - 2 = DotProduct
229    /// - 3 = Haversine
230    /// - 4 = Hamming
231    /// - 5 = Tanimoto
232    ///
233    /// # Example
234    ///
235    /// ```
236    /// use aletheiadb::index::vector::DistanceMetric;
237    ///
238    /// assert_eq!(DistanceMetric::Cosine.to_u8(), 0);
239    /// assert_eq!(DistanceMetric::Euclidean.to_u8(), 1);
240    /// assert_eq!(DistanceMetric::DotProduct.to_u8(), 2);
241    /// assert_eq!(DistanceMetric::Haversine.to_u8(), 3);
242    /// assert_eq!(DistanceMetric::Hamming.to_u8(), 4);
243    /// assert_eq!(DistanceMetric::Tanimoto.to_u8(), 5);
244    /// ```
245    pub fn to_u8(self) -> u8 {
246        match self {
247            DistanceMetric::Cosine => 0,
248            DistanceMetric::Euclidean => 1,
249            DistanceMetric::DotProduct => 2,
250            DistanceMetric::Haversine => 3,
251            DistanceMetric::Hamming => 4,
252            DistanceMetric::Tanimoto => 5,
253        }
254    }
255
256    /// Decode distance metric from a byte.
257    ///
258    /// # Errors
259    ///
260    /// Returns an error if the byte value is not a valid metric encoding (>= 6).
261    ///
262    /// # Example
263    ///
264    /// ```
265    /// use aletheiadb::index::vector::DistanceMetric;
266    ///
267    /// assert_eq!(DistanceMetric::from_u8(0).unwrap(), DistanceMetric::Cosine);
268    /// assert_eq!(DistanceMetric::from_u8(1).unwrap(), DistanceMetric::Euclidean);
269    /// assert_eq!(DistanceMetric::from_u8(2).unwrap(), DistanceMetric::DotProduct);
270    /// assert_eq!(DistanceMetric::from_u8(3).unwrap(), DistanceMetric::Haversine);
271    /// assert_eq!(DistanceMetric::from_u8(4).unwrap(), DistanceMetric::Hamming);
272    /// assert_eq!(DistanceMetric::from_u8(5).unwrap(), DistanceMetric::Tanimoto);
273    /// assert!(DistanceMetric::from_u8(6).is_err());
274    /// ```
275    pub fn from_u8(value: u8) -> Result<Self> {
276        match value {
277            0 => Ok(DistanceMetric::Cosine),
278            1 => Ok(DistanceMetric::Euclidean),
279            2 => Ok(DistanceMetric::DotProduct),
280            3 => Ok(DistanceMetric::Haversine),
281            4 => Ok(DistanceMetric::Hamming),
282            5 => Ok(DistanceMetric::Tanimoto),
283            _ => Err(crate::core::error::StorageError::CorruptedData(format!(
284                "Invalid distance metric encoding: {}",
285                value
286            ))
287            .into()),
288        }
289    }
290}

292/// Trait for vector indexes supporting approximate k-nearest neighbor search.
293///
294/// This trait abstracts over different vector index implementations, allowing
295/// AletheiaDB to support multiple ANN (Approximate Nearest Neighbor) algorithms
296/// while maintaining a consistent query interface.
297///
298/// # Invariants
299///
300/// Implementations must maintain these invariants:
301/// - All vectors in the index have the same dimensionality (returned by `dimensions()`)
302/// - `search()` returns at most `k` results, sorted by similarity (descending)
303/// - `search_with_filter()` only returns results where the predicate returns `true`
304/// - `len()` returns the exact number of vectors currently in the index
305/// - Adding the same NodeId twice replaces the previous vector
306/// - The distance metric never changes after index creation
307///
308/// # Thread Safety
309///
310/// Implementations must be thread-safe for both concurrent reads and writes.
311/// The trait methods take `&self` (not `&mut self`) to enable concurrent operations.
312/// Implementations should use interior mutability (e.g., internal locks, atomics) to
313/// coordinate concurrent access. For example, usearch supports concurrent insertions
314/// through internal locking while maintaining `&self` semantics.
315///
316/// # Performance Expectations
317///
318/// For HNSW implementation (Phase 2):
319/// - **Insert**: O(log n) average case with configurable ef_construction parameter
320/// - **Search**: O(log n) average case, O(n) worst case, with configurable ef_search
321/// - **Memory**: O(n * M * d) where M is connections per node, d is dimensions
322/// - **Query latency**: 100µs-1ms for typical datasets (<10M vectors, 384-1536 dims)
323/// - **Dimensionality impact**: Linear scaling with dimensions for both time and memory
324///
325/// # Error Handling
326///
327/// Methods return specific error variants for validation failures:
328/// - `Error::Vector(VectorError::DimensionMismatch { expected, actual })` - Wrong dimensions
329/// - `Error::Vector(VectorError::ContainsNaN { count })` - Vector contains NaN values
330/// - `Error::Vector(VectorError::ContainsInfinity { count })` - Vector contains Infinity
331/// - `Error::Vector(VectorError::DimensionTooLarge { ... })` - Vector too large
332pub trait VectorIndex: Send + Sync {
333    /// Adds a vector to the index, associating it with the given node ID.
334    ///
335    /// If a vector with the same `id` already exists, it will be replaced.
336    /// The vector dimensions must match the index's configured dimensionality.
337    ///
338    /// # Arguments
339    ///
340    /// * `id` - The node ID to associate with this vector
341    /// * `vector` - The embedding vector (must match index dimensions)
342    ///
343    /// # Returns
344    ///
345    /// - `Ok(())` if the vector was successfully added
346    /// - `Err(Error::Vector(VectorError::DimensionMismatch))` if dimensions don't match
347    /// - `Err(Error::Vector(VectorError::ContainsNaN))` if vector contains NaN
348    /// - `Err(Error::Vector(VectorError::ContainsInfinity))` if vector contains Infinity
349    ///
350    /// # Examples
351    ///
352    /// ```rust
353    /// # use aletheiadb::index::VectorIndex;
354    /// # use aletheiadb::core::id::NodeId;
355    /// # use aletheiadb::index::vector::{HnswIndexBuilder, DistanceMetric};
356    /// # fn main() -> aletheiadb::core::error::Result<()> {
357    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
358    /// let node_id = NodeId::new(123).unwrap();
359    /// let embedding = vec![0.1, 0.2, 0.3, 0.4];
360    /// index.add(node_id, &embedding)?;
361    /// # Ok(())
362    /// # }
363    /// ```
364    fn add(&self, id: NodeId, vector: &[f32]) -> Result<()>;
365
366    /// Removes a vector from the index by node ID.
367    ///
368    /// If the ID does not exist in the index, this is a no-op (returns Ok).
369    ///
370    /// # Arguments
371    ///
372    /// * `id` - The node ID to remove
373    ///
374    /// # Returns
375    ///
376    /// - `Ok(())` if the vector was removed or didn't exist
377    /// - `Err(_)` if the underlying index encounters an error
378    ///
379    /// # Examples
380    ///
381    /// ```rust
382    /// # use aletheiadb::index::VectorIndex;
383    /// # use aletheiadb::core::id::NodeId;
384    /// # use aletheiadb::index::vector::{HnswIndexBuilder, DistanceMetric};
385    /// # fn main() -> aletheiadb::core::error::Result<()> {
386    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
387    /// let node_id = NodeId::new(123).unwrap();
388    /// index.remove(node_id)?;
389    /// # Ok(())
390    /// # }
391    /// ```
392    fn remove(&self, id: NodeId) -> Result<()>;
393
394    /// Searches for the k-nearest neighbors of the query vector.
395    ///
396    /// Returns up to `k` results sorted by similarity in descending order
397    /// (highest similarity first). The similarity score interpretation depends on the
398    /// configured distance metric (query via `distance_metric()`).
399    ///
400    /// # Arguments
401    ///
402    /// * `query` - The query embedding vector (must match index dimensions)
403    /// * `k` - Maximum number of results to return (implementations may cap this,
404    ///   e.g., at 10,000 to prevent DoS)
405    ///
406    /// # Returns
407    ///
408    /// A vector of (NodeId, similarity) pairs, sorted by similarity (descending).
409    /// May contain fewer than `k` results if the index has fewer vectors.
410    ///
411    /// # Errors
412    ///
413    /// - `Err(Error::Vector(VectorError::DimensionMismatch))` if query dimensions don't match
414    /// - `Err(Error::Vector(VectorError::ContainsNaN))` if query contains NaN
415    /// - `Err(Error::Vector(VectorError::ContainsInfinity))` if query contains Infinity
416    ///
417    /// # Examples
418    ///
419    /// ```rust
420    /// # use aletheiadb::index::VectorIndex;
421    /// # use aletheiadb::index::vector::{HnswIndexBuilder, DistanceMetric};
422    /// # fn main() -> aletheiadb::core::error::Result<()> {
423    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
424    /// let query = vec![0.5, 0.3, 0.1, 0.9];
425    /// let results = index.search(&query, 10)?;
426    ///
427    /// for (node_id, similarity) in results {
428    ///     println!("Found node {:?} with similarity {}", node_id, similarity);
429    /// }
430    /// # Ok(())
431    /// # }
432    /// ```
433    fn search(&self, query: &[f32], k: usize) -> Result<Vec<(NodeId, f32)>>;
434
435    /// Searches for k-nearest neighbors with a filter predicate.
436    ///
437    /// Like `search()`, but only returns results where `predicate(node_id)` returns `true`.
438    /// This enables filtered search without materializing the full result set.
439    ///
440    /// # Arguments
441    ///
442    /// * `query` - The query embedding vector (must match index dimensions)
443    /// * `k` - Maximum number of results to return
444    /// * `predicate` - Filter function that returns true for nodes to include
445    ///
446    /// # Returns
447    ///
448    /// A vector of (NodeId, similarity) pairs where the predicate returned true,
449    /// sorted by similarity (descending).
450    ///
451    /// # Errors
452    ///
453    /// Same as `search()`.
454    ///
455    /// # Performance
456    ///
457    /// **WARNING**: Low-selectivity filters (<1%) can degrade to nearly linear scan performance.
458    /// The implementation must examine many candidates to find `k` results that satisfy
459    /// the predicate. For example, with 1% selectivity, finding 100 results may require
460    /// examining ~10,000 candidates.
461    ///
462    /// **Alternatives for better performance**:
463    /// - Maintain separate indexes per category if filtering is common
464    /// - Use hybrid graph+vector queries to pre-filter via graph traversal
465    /// - Consider the `search()` method and filter results in-memory if selectivity is very low
466    ///
467    /// # Examples
468    ///
469    /// ```rust
470    /// # use aletheiadb::index::VectorIndex;
471    /// # use aletheiadb::core::id::NodeId;
472    /// # use std::collections::HashSet;
473    /// # use aletheiadb::index::vector::{HnswIndexBuilder, DistanceMetric};
474    /// # fn main() -> aletheiadb::core::error::Result<()> {
475    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
476    /// let query = vec![0.5, 0.3, 0.1, 0.9];
477    /// let allowed = HashSet::from([NodeId::new(1).unwrap(), NodeId::new(5).unwrap(), NodeId::new(10).unwrap()]);
478    ///
479    /// // Only search within allowed nodes
480    /// let results = index.search_with_filter(&query, 5, |id| allowed.contains(id))?;
481    /// # Ok(())
482    /// # }
483    /// ```
484    fn search_with_filter<F>(
485        &self,
486        query: &[f32],
487        k: usize,
488        predicate: F,
489    ) -> Result<Vec<(NodeId, f32)>>
490    where
491        F: Fn(&NodeId) -> bool + Send + Sync;
492
493    /// Returns the number of vectors currently in the index.
494    ///
495    /// # Examples
496    ///
497    /// ```rust
498    /// # use aletheiadb::index::VectorIndex;
499    /// # use aletheiadb::index::vector::{HnswIndexBuilder, DistanceMetric};
500    /// # fn main() -> aletheiadb::core::error::Result<()> {
501    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
502    /// println!("Index contains {} vectors", index.len());
503    /// # Ok(())
504    /// # }
505    /// ```
506    #[must_use]
507    fn len(&self) -> usize;
508
509    /// Returns the dimensionality of vectors in this index.
510    ///
511    /// All vectors added to the index must have this many dimensions.
512    ///
513    /// # Examples
514    ///
515    /// ```rust
516    /// # use aletheiadb::index::VectorIndex;
517    /// # use aletheiadb::index::vector::{HnswIndexBuilder, DistanceMetric};
518    /// # fn main() -> aletheiadb::core::error::Result<()> {
519    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
520    /// let dims = index.dimensions();
521    /// println!("This index accepts {}-dimensional vectors", dims);
522    /// # Ok(())
523    /// # }
524    /// ```
525    #[must_use]
526    fn dimensions(&self) -> usize;
527
528    /// Returns the distance metric used by this index.
529    ///
530    /// The distance metric determines how similarity scores are computed and
531    /// how to interpret the `f32` values returned by `search()` methods.
532    ///
533    /// # Metric Interpretation
534    ///
535    /// - **Cosine**: Returns cosine similarity in range [-1, 1]. Higher is more similar.
536    ///   Typical threshold for "similar": >0.7
537    /// - **Euclidean**: Returns negative squared L2 distance in range (-∞, 0]. Higher is more similar.
538    ///   (More negative = more distant, closer to 0 = more similar)
539    ///   Threshold depends on embedding magnitude and dimensionality.
540    /// - **DotProduct**: Returns inner product in range (-∞, ∞). Higher is more similar.
541    ///   Threshold depends on embedding normalization.
542    ///
543    /// # Examples
544    ///
545    /// ```rust
546    /// # use aletheiadb::index::{VectorIndex, DistanceMetric};
547    /// # use aletheiadb::index::vector::HnswIndexBuilder;
548    /// # fn main() -> aletheiadb::core::error::Result<()> {
549    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
550    /// match index.distance_metric() {
551    ///     DistanceMetric::Cosine => println!("Using cosine similarity"),
552    ///     DistanceMetric::Euclidean => println!("Using Euclidean distance"),
553    ///     DistanceMetric::DotProduct => println!("Using dot product"),
554    ///     DistanceMetric::Haversine => println!("Using Haversine distance"),
555    ///     DistanceMetric::Hamming => println!("Using Hamming distance"),
556    ///     DistanceMetric::Tanimoto => println!("Using Tanimoto similarity"),
557    /// }
558    /// # Ok(())
559    /// # }
560    /// ```
561    #[must_use]
562    fn distance_metric(&self) -> DistanceMetric;
563
564    /// Returns true if the index is empty.
565    ///
566    /// Equivalent to `self.len() == 0`.
567    ///
568    /// # Examples
569    ///
570    /// ```rust
571    /// # use aletheiadb::index::VectorIndex;
572    /// # use aletheiadb::index::vector::{HnswIndexBuilder, DistanceMetric};
573    /// # fn main() -> aletheiadb::core::error::Result<()> {
574    /// # let index = HnswIndexBuilder::new(4, DistanceMetric::Cosine).build()?;
575    /// if index.is_empty() {
576    ///     println!("No vectors indexed yet");
577    /// }
578    /// # Ok(())
579    /// # }
580    /// ```
581    #[must_use]
582    fn is_empty(&self) -> bool {
583        self.len() == 0
584    }

586    /// Adds multiple vectors in a batch operation.
587    ///
588    /// More efficient than calling `add()` repeatedly for bulk insertions.
589    /// Default implementation calls `add()` sequentially.
590    fn add_batch(&self, items: &[(NodeId, Vec<f32>)]) -> Result<()> {
591        for (id, vec) in items {
592            self.add(*id, vec)?;
593        }
594        Ok(())
595    }
596
597    /// Adds multiple vectors in a batch operation using references.
598    ///
599    /// This allows adding vectors without transferring ownership, avoiding clones.
600    /// Default implementation calls `add()` sequentially.
601    fn add_batch_ref(&self, items: &[(NodeId, &[f32])]) -> Result<()> {
602        for (id, vec) in items {
603            self.add(*id, vec)?;
604        }
605        Ok(())
606    }

608    /// Removes multiple vectors in a batch operation.
609    ///
610    /// Default implementation calls `remove()` sequentially.
611    fn remove_batch(&self, ids: &[NodeId]) -> Result<()> {
612        for id in ids {
613            self.remove(*id)?;
614        }
615        Ok(())
616    }

629    /// Returns the approximate memory usage of this index in bytes.

753/// Wrapper for f32 that implements Ord for use in BinaryHeap.
754///
755/// NaN values are treated as less than all other values for consistent ordering.
756#[derive(Debug, Clone, Copy, PartialEq)]
757pub(crate) struct OrderedFloat(pub f32);
758
759impl Eq for OrderedFloat {}
760
761impl PartialOrd for OrderedFloat {
762    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
763        Some(self.cmp(other))
764    }
765}
766
767impl Ord for OrderedFloat {
768    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
769        self.0
770            .partial_cmp(&other.0)
771            .unwrap_or_else(|| match (self.0.is_nan(), other.0.is_nan()) {
772                (true, true) => std::cmp::Ordering::Equal,
773                (true, false) => std::cmp::Ordering::Less,
774                (false, true) => std::cmp::Ordering::Greater,
775                (false, false) => unreachable!(),
776            })
777    }
778}

780/// Merge search results from multiple sources using a min-heap for top-k efficiency.
781///
782/// O(n log k) where n is total results and k is the desired count.
783pub(crate) fn merge_top_k_results(
784    all_results: Vec<Vec<(crate::core::id::NodeId, f32)>>,
785    k: usize,
786) -> Vec<(crate::core::id::NodeId, f32)> {
787    use std::cmp::Reverse;
788    use std::collections::BinaryHeap;
789
790    if k == 0 {
791        return Vec::new();
792    }
793
794    let mut heap: BinaryHeap<(Reverse<OrderedFloat>, crate::core::id::NodeId)> =
795        BinaryHeap::with_capacity(k + 1);
796
797    for results in all_results {
798        for (id, score) in results {
799            let ordered_score = OrderedFloat(score);
800
801            if heap.len() < k {
802                heap.push((Reverse(ordered_score), id));
803            } else if let Some(&(Reverse(min_score), _)) = heap.peek()
804                && ordered_score > min_score
805            {
806                heap.pop();
807                heap.push((Reverse(ordered_score), id));
808            }
809        }
810    }
811
812    let mut results: Vec<(crate::core::id::NodeId, f32)> = heap
813        .into_iter()
814        .map(|(Reverse(score), id)| (id, score.0))
815        .collect();
816
817    results.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
818
819    results
820}
