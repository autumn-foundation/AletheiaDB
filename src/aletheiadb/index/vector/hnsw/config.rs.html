9/// Configuration for HNSW (Hierarchical Navigable Small World) index.
10///
11/// This struct encapsulates all parameters needed to configure an HNSW index
12/// for approximate nearest neighbor search. It provides sensible defaults
13/// optimized for a balance between accuracy, speed, and memory usage.
14#[derive(Debug, Clone)]
15pub struct HnswConfig {
16    /// Vector dimensionality (must be > 0)
17    pub dimensions: usize,
18    /// Distance metric for similarity computation
19    pub metric: DistanceMetric,
20    /// Maximum bidirectional connections per node (default: 16)
21    pub m: usize,
22    /// Build-time candidate list size (default: 128)
23    pub ef_construction: usize,
24    /// Query-time candidate list size (default: 64)
25    pub ef_search: usize,
26    /// Initial capacity for pre-allocation (default: 0)
27    pub capacity: usize,
28    /// Quantization level (default: F32)
29    pub quantization: Quantization,
30    /// Storage mode (default: InMemory)
31    pub storage: StorageMode,
32    /// Custom distance metric (overrides `metric` if set)
33    pub custom_metric: Option<CustomMetric>,
34}
35
36impl PartialEq for HnswConfig {
37    fn eq(&self, other: &Self) -> bool {
38        self.dimensions == other.dimensions
39            && self.metric == other.metric
40            && self.m == other.m
41            && self.ef_construction == other.ef_construction
42            && self.ef_search == other.ef_search
43            && self.capacity == other.capacity
44            && self.quantization == other.quantization
45            && self.storage == other.storage
46            && self.custom_metric == other.custom_metric
47    }
48}

136    /// Serialize configuration to a writer in little-endian binary format.
137    pub fn serialize_into<W: Write>(&self, writer: &mut W) -> Result<()> {
138        writer.write_all(&(self.dimensions as u64).to_le_bytes())?;
139        writer.write_all(&[self.metric.to_u8()])?;
140        writer.write_all(&(self.m as u64).to_le_bytes())?;
141        writer.write_all(&(self.ef_construction as u64).to_le_bytes())?;
142        writer.write_all(&(self.ef_search as u64).to_le_bytes())?;
143        writer.write_all(&(self.capacity as u64).to_le_bytes())?;
144        writer.write_all(&[self.quantization.to_u8()])?;
145        Ok(())
146    }
147
148    /// Deserialize configuration from a reader.
149    pub fn deserialize_from<R: Read>(reader: &mut R) -> Result<Self> {
150        let mut buf_u64 = [0u8; 8];
151        let mut buf_u8 = [0u8; 1];
152
153        reader.read_exact(&mut buf_u64)?;
154        let dimensions = u64::from_le_bytes(buf_u64) as usize;
155
156        if dimensions > MAX_VECTOR_DIMENSIONS {
157            return Err(Error::Vector(VectorError::InvalidVector {
158                reason: format!(
159                    "dimensions {} exceeds maximum allowed {}",
160                    dimensions, MAX_VECTOR_DIMENSIONS
161                ),
162            }));
163        }
164
165        reader.read_exact(&mut buf_u8)?;
166        let metric = DistanceMetric::from_u8(buf_u8[0])?;
167
168        reader.read_exact(&mut buf_u64)?;
169        let m = u64::from_le_bytes(buf_u64) as usize;
170
171        reader.read_exact(&mut buf_u64)?;
172        let ef_construction = u64::from_le_bytes(buf_u64) as usize;
173
174        reader.read_exact(&mut buf_u64)?;
175        let ef_search = u64::from_le_bytes(buf_u64) as usize;
176
177        reader.read_exact(&mut buf_u64)?;
178        let capacity = u64::from_le_bytes(buf_u64) as usize;
179
180        // Try read quantization (for backward compatibility)
181        let quantization = match reader.read_exact(&mut buf_u8) {
182            Ok(_) => Quantization::from_u8(buf_u8[0])?,
183            Err(ref e) if e.kind() == std::io::ErrorKind::UnexpectedEof => {
184                // Old format (v1) didn't have quantization, assume F32 default
185                Quantization::default()
186            }
187            Err(e) => return Err(e.into()),
188        };
189
190        Ok(HnswConfig {
191            dimensions,
192            metric,
193            m,
194            ef_construction,
195            ef_search,
196            capacity,
197            quantization,
198            ..Default::default()
199        })
200    }
