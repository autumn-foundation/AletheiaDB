19/// Type alias for vector snapshot: map of NodeId to vector data
20/// Represents vector data in a snapshot, supporting both full and delta formats.
21///
22/// Full snapshots store all vectors, while delta snapshots store only changes
23/// relative to a base snapshot, significantly reducing memory usage for incremental updates.
24#[derive(Clone)]
25pub(crate) enum VectorSnapshot {
26    /// Full snapshot containing all vectors
27    Full(Arc<HashMap<NodeId, Arc<[f32]>, IdentityBuildHasher>>),
28
29    /// Delta snapshot storing only differences
30    Delta {
31        /// Timestamp of the base snapshot this delta is relative to
32        base_time: Timestamp,
33        /// Vectors added or updated since the base snapshot
34        added: Arc<HashMap<NodeId, Arc<[f32]>, IdentityBuildHasher>>,
35        /// NodeIds removed since the base snapshot
36        removed: Arc<HashSet<NodeId, IdentityBuildHasher>>,
37    },
38}

40impl VectorSnapshot {
41    /// Retrieves a vector for a given node ID, traversing the delta chain if necessary.
42    ///
43    /// # Complexity
44    /// - Full: O(1)
45    /// - Delta: O(depth) where depth is the number of delta layers
46    ///
47    /// # Safety
48    /// Enforces `MAX_DELTA_CHAIN_DEPTH` to prevent stack overflow or excessive latency.
49    pub(crate) fn get_vector(
50        &self,
51        node_id: &NodeId,
52        all_snapshots: &BTreeMap<Timestamp, VectorSnapshot>,
53    ) -> Result<Option<Arc<[f32]>>> {
54        let mut current = self;
55        let mut depth = 0;
56
57        loop {
58            // SAFETY: Check depth limit to prevent unbounded traversal
59            // If chain exceeds MAX_DELTA_CHAIN_DEPTH, return error instead of silently failing
60            if depth >= MAX_DELTA_CHAIN_DEPTH {
61                return Err(VectorError::IndexError(format!(
62                    "Delta chain depth exceeded {} for node {:?}. \
63                     This indicates corrupted snapshot state or misconfiguration. \
64                     Reduce full_snapshot_interval or check snapshot integrity.",
65                    MAX_DELTA_CHAIN_DEPTH, node_id
66                ))
67                .into());
68            }
69
70            match current {
71                VectorSnapshot::Full(vectors) => {
72                    return Ok(vectors.get(node_id).cloned());
73                }
74                VectorSnapshot::Delta {
75                    base_time,
76                    added,
77                    removed,
78                } => {
79                    // First check if removed
80                    if removed.contains(node_id) {
81                        return Ok(None);
82                    }
83
84                    // Then check if in added/updated
85                    if let Some(vec) = added.get(node_id) {
86                        return Ok(Some(Arc::clone(vec)));
87                    }
88
89                    // If not found, traverse to base
90                    if let Some(base) = all_snapshots.get(base_time) {
91                        current = base;
92                        depth += 1;
93                    } else {
94                        // Base snapshot missing - this is a serious integrity error
95                        return Err(VectorError::IndexError(format!(
96                            "Base snapshot at {} missing for delta snapshot",
97                            base_time
98                        ))
99                        .into());
100                    }
101                }
102            }
103        }
104    }

106    /// Reconstructs the full state of vectors at this snapshot.
107    ///
108    /// Useful for operations that need the complete dataset (e.g., drift analysis).
109    ///
110    /// # Performance
111    /// This operation can be expensive for deep delta chains. Consider caching results
112    /// if called frequently.
113    pub(crate) fn to_hashmap(
114        &self,
115        all_snapshots: &BTreeMap<Timestamp, VectorSnapshot>,
116    ) -> Result<HashMap<NodeId, Arc<[f32]>, IdentityBuildHasher>> {
117        // Collect layers
118        let mut current = self;
119        let mut delta_layers = Vec::new();
120        let mut depth = 0;
121
122        // Walk backwards through delta chain to find the Full snapshot base
123        let base_vectors: HashMap<NodeId, Arc<[f32]>, IdentityBuildHasher> = loop {
124            if depth >= MAX_DELTA_CHAIN_DEPTH {
125                // Return error instead of partial results to prevent silent data loss
126                return Err(VectorError::IndexError(format!(
127                    "Delta chain depth exceeded {} in to_hashmap(). \
128                     This indicates corrupted snapshot state or misconfiguration. \
129                     Reduce full_snapshot_interval or check snapshot integrity.",
130                    MAX_DELTA_CHAIN_DEPTH
131                ))
132                .into());
133            }
134
135            match current {
136                VectorSnapshot::Full(vectors) => {
137                    // Found the base Full snapshot
138                    // Clone the inner HashMap (O(N)) - unavoidable for reconstruction
139                    break vectors.as_ref().clone();
140                }
141                VectorSnapshot::Delta {
142                    base_time,
143                    added,
144                    removed,
145                } => {
146                    // Store delta layer to apply later (in reverse order)
147                    struct DeltaLayer<'a> {
148                        added: &'a HashMap<NodeId, Arc<[f32]>, IdentityBuildHasher>,
149                        removed: &'a HashSet<NodeId, IdentityBuildHasher>,
150                    }
151                    delta_layers.push(DeltaLayer {
152                        added: added.as_ref(),
153                        removed: removed.as_ref(),
154                    });
155
156                    // Move to previous snapshot
157                    if let Some(base) = all_snapshots.get(base_time) {
158                        current = base;
159                        depth += 1;
160                    } else {
161                        return Err(VectorError::IndexError(format!(
162                            "Base snapshot at {} missing during reconstruction",
163                            base_time
164                        ))
165                        .into());
166                    }
167                }
168            }
169        };
170
171        // Apply deltas forward (from oldest to newest)
172        // Base is already cloned, so we can modify it directly
173        let mut result = base_vectors;
174
175        for layer in delta_layers.iter().rev() {
176            // Apply removals
177            for node_id in layer.removed.iter() {
178                result.remove(node_id);
179            }
180            // Apply additions/updates
181            for (node_id, vector) in layer.added.iter() {
182                result.insert(*node_id, Arc::clone(vector));
183            }
184        }
185
186        Ok(result)
187    }

189    /// Optimized version of to_hashmap that collects ALL vectors from a set of snapshots.
190    ///
191    /// This is more efficient than calling to_hashmap on each snapshot individually
192    /// when iterating over a time range.
193    pub(crate) fn collect_all(
194        &self,
195        all_snapshots: &BTreeMap<Timestamp, VectorSnapshot>,
196    ) -> Result<Vec<(NodeId, Arc<[f32]>)>> {
197        let map = self.to_hashmap(all_snapshots)?;
198        Ok(map.into_iter().collect())
199    }
200
201    /// Returns the number of vectors in this snapshot.
202    ///
203    /// # Accuracy Note
204    /// For Full snapshots, this is exact.
205    /// For Delta snapshots, this is an **estimate** based on the added vectors count.
206    /// It does not account for removals or base vectors, so it returns a safe
207    /// underestimate used for capacity estimation during index construction.
208    ///
209    /// **For exact counts**, use `to_hashmap().len()` which reconstructs the full snapshot
210    /// by applying all deltas to the base. Note that reconstruction has O(depth) cost and
211    /// should be avoided in hot paths unless necessary.
212    ///
213    /// # Example
214    /// ```rust,ignore
215    /// // Fast, O(1), but approximate for Deltas
216    /// let approx_len = snapshot.len();
217    ///
218    /// // Slow, O(depth), but exact
219    /// let exact_len = snapshot.to_hashmap(&all_snapshots)?.len();
220    /// ```
221    pub(crate) fn len(&self) -> usize {
222        match self {
223            VectorSnapshot::Full(vectors) => vectors.len(),
224            VectorSnapshot::Delta { added, .. } => {
225                // Approximation: just return added size
226                // This is used only for capacity estimation during index construction
227                // and intentionally underestimates to avoid excessive memory allocation
228                added.len().max(MIN_CAPACITY_ESTIMATE)
229            }
230        }
231    }
232
233    /// Collect all vectors in this snapshot into a Vec.
234    ///
235    /// For delta snapshots, this reconstructs the full set.
236    #[allow(dead_code)]
237    pub(crate) fn to_vec(
238        &self,
239        all_snapshots: &BTreeMap<Timestamp, VectorSnapshot>,
240    ) -> Result<Vec<(NodeId, Arc<[f32]>)>> {
241        let map = self.to_hashmap(all_snapshots)?;
242        Ok(map.into_iter().collect())
243    }
244}

246/// Helper struct for `SnapshotIndex` enum.
247/// Wraps either a full HNSW index or a delta index.
248#[derive(Clone)]
249pub(crate) enum SnapshotIndex {
250    Full(Arc<HnswIndex>),
251    Delta(Arc<DeltaIndex>),
252}

254impl std::fmt::Debug for SnapshotIndex {
255    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
256        match self {
257            SnapshotIndex::Full(index) => f
258                .debug_struct("SnapshotIndex::Full")
259                .field("len", &index.len())
260                .field("dimensions", &index.dimensions())
261                .finish(),
262            SnapshotIndex::Delta(delta) => f
263                .debug_struct("SnapshotIndex::Delta")
264                .field("base_len", &delta.base.len())
265                .field("added_len", &delta.added.len())
266                .field("removed_len", &delta.removed.len())
267                .field(
268                    "total_len",
269                    &(delta.base.len() + delta.added.len() - delta.removed.len()),
270                )
271                .finish(),
272        }
273    }

276impl SnapshotIndex {
277    /// Searches the snapshot for the top `k` nearest neighbors to the `query` vector.
278    ///
279    /// # Why?
280    /// This provides a point-in-time semantic search. It routes the query to either
281    /// the fully consolidated HNSW index, or a delta-index depending on the snapshot's state.
282    pub fn search(&self, query: &[f32], k: usize) -> Result<Vec<(NodeId, f32)>> {
283        match self {
284            SnapshotIndex::Full(index) => index.search(query, k),
285            SnapshotIndex::Delta(delta) => delta.search(query, k),
286        }
287    }
288
289    /// Searches the snapshot for the top `k` nearest neighbors, applying a custom predicate filter.
290    ///
291    /// # Why?
292    /// Pure vector search often returns irrelevant results (e.g., matching a document from
293    /// the wrong user). This allows pre-filtering the candidate list during the HNSW traversal
294    /// to guarantee both semantic relevance and logical correctness.
295    pub fn search_with_filter(
296        &self,
297        query: &[f32],
298        k: usize,
299        predicate: &(dyn Fn(&NodeId) -> bool + Send + Sync),
300    ) -> Result<Vec<(NodeId, f32)>> {
301        match self {
302            SnapshotIndex::Full(index) => index.search_with_filter(query, k, predicate),
303            SnapshotIndex::Delta(delta) => delta.search_with_filter(query, k, predicate),
304        }
305    }
306
307    pub(crate) fn len(&self) -> usize {
308        match self {
309            SnapshotIndex::Full(index) => index.len(),
310            SnapshotIndex::Delta(delta) => {
311                // Estimate length for delta
312                delta.base.len() + delta.added.len() - delta.removed.len()
313            }
314        }
315    }
316
317    pub(crate) fn dimensions(&self) -> usize {
318        match self {
319            SnapshotIndex::Full(index) => index.dimensions(),
320            SnapshotIndex::Delta(delta) => delta.added.dimensions(),
321        }
322    }
323}

325/// Internal delta index implementation.
326///
327/// Stores changes relative to a base index.
328pub(crate) struct DeltaIndex {
329    pub(crate) base: Arc<SnapshotIndex>,
330    pub(crate) added: Arc<HnswIndex>,
331    pub(crate) removed: Arc<HashSet<NodeId, IdentityBuildHasher>>,
332}

336        f.debug_struct("DeltaIndex")
337            .field("base", &self.base)
338            .field("added_len", &self.added.len())
339            .field("added_dimensions", &self.added.dimensions())
340            .field("removed_count", &self.removed.len())
341            .finish()
342    }
343}

345impl DeltaIndex {
346    fn search(&self, query: &[f32], k: usize) -> Result<Vec<(NodeId, f32)>> {
347        // Search for k*2 candidates to ensure global top-k
348        // We need more candidates because some might be in the 'removed' set
349        // or shadowed by newer versions in 'added'
350        let search_k = k.saturating_mul(2).max(k + 10);
351
352        // Search added (small index, fast)
353        let mut results = self.added.search(query, search_k)?;
354
355        // Filter out results that are in 'removed' set (unlikely for added, but possible if
356        // a node was added then removed in same snapshot window - though usually that's handled at construction)
357        // More importantly, this handles the case where we merge results.
358
359        // Search base
360        // We must filter out nodes that are in 'removed' set OR in 'added' set (updates shadow old versions)
361        let removed = &self.removed;
362        let added_ids: HashSet<NodeId> = results.iter().map(|(id, _)| *id).collect();
363
364        let predicate = |id: &NodeId| !removed.contains(id) && !added_ids.contains(id);
365
366        let base_results = self.base.search_with_filter(query, search_k, &predicate)?;
367
368        // Merge results
369        results.extend(base_results);
370
371        // Sort and truncate to k
372        results.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
373        results.truncate(k);
374
375        Ok(results)
376    }
377
378    fn search_with_filter(
379        &self,
380        query: &[f32],
381        k: usize,
382        predicate: &(dyn Fn(&NodeId) -> bool + Send + Sync),
383    ) -> Result<Vec<(NodeId, f32)>> {
384        // Search for k*2 candidates to ensure global top-k (same strategy as search())
385        let search_k = k.saturating_mul(2).max(k + 10);
386
387        // Combine user predicate with our removed set
388        let removed = &self.removed;
389        let combined_predicate = |id: &NodeId| predicate(id) && !removed.contains(id);
390
391        // Search added (using user predicate only)
392        let mut results = self.added.search_with_filter(query, search_k, predicate)?;
393
394        // Search base (using combined predicate to filter out removed/updated nodes)
395        let base_results = self
396            .base
397            .search_with_filter(query, search_k, &combined_predicate)?;
398
399        // Merge results
400        results.extend(base_results);
401
402        // Deduplicate: Although the combined predicate should prevent duplicates,
403        // we deduplicate as a safety measure to ensure correctness.
404        // We keep the first occurrence (which has the better score after sorting).
405        use std::collections::HashSet;
406        let mut seen = HashSet::new();
407        results.retain(|(id, _)| seen.insert(*id));
408
409        // Sort and truncate to k
410        results.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
411        results.truncate(k);
412
413        Ok(results)
414    }
415}

417/// Snapshot data protected by a single lock.
418pub(crate) struct SnapshotData {
419    /// Maps timestamp -> (snapshot_id, snapshot_index)
420    pub(crate) snapshots: BTreeMap<Timestamp, (usize, SnapshotIndex)>,
421
422    /// Maps timestamp -> vector_data (for reconstruction)
423    pub(crate) vector_history: BTreeMap<Timestamp, VectorSnapshot>,
424}
425
426impl SnapshotData {
427    pub(crate) fn new() -> Self {
428        Self {
429            snapshots: BTreeMap::new(),
430            vector_history: BTreeMap::new(),
431        }
432    }
433
434    pub(crate) fn insert(
435        &mut self,
436        timestamp: Timestamp,
437        id: usize,
438        index: SnapshotIndex,
439        vectors: VectorSnapshot,
440    ) {
441        self.snapshots.insert(timestamp, (id, index));
442        self.vector_history.insert(timestamp, vectors);
443    }
444
445    pub(crate) fn len(&self) -> usize {
446        self.snapshots.len()
447    }
448
449    pub(crate) fn remove_oldest(&mut self) {
450        if let Some(key) = self.snapshots.keys().next().copied() {
451            self.snapshots.remove(&key);
452            self.vector_history.remove(&key);
453        }
454    }
455}

457/// Snapshot metadata tracking state.
458pub(crate) struct SnapshotMetadata {
459    /// Total number of snapshots created (ever)
460    pub(crate) total_snapshots: usize,
461
462    /// Number of transactions since last snapshot
463    pub(crate) transactions_since_snapshot: usize,
464
465    /// Last snapshot creation time
466    pub(crate) last_snapshot_time: Timestamp,
467
468    /// Set of NodeIds changed since last snapshot (for ChangeThreshold)
469    pub(crate) vectors_changed_since_snapshot: HashSet<NodeId, IdentityBuildHasher>,
470
471    /// Last FULL snapshot time (for delta calculation)
472    pub(crate) last_full_snapshot_time: Timestamp,
473
474    /// Number of snapshots since last FULL snapshot
475    pub(crate) snapshots_since_full: usize,
476
477    /// Accumulator for ALL changes since last FULL snapshot
478    /// Used to build the delta index.
479    pub(crate) changes_accumulated: HashSet<NodeId, IdentityBuildHasher>,
480}
481
482impl SnapshotMetadata {
483    pub(crate) fn new(initial_time: Timestamp) -> Self {
484        Self {
485            total_snapshots: 0,
486            transactions_since_snapshot: 0,
487            last_snapshot_time: initial_time,
488            vectors_changed_since_snapshot: HashSet::with_hasher(BuildHasherDefault::default()),
489            last_full_snapshot_time: initial_time,
490            snapshots_since_full: 0,
491            changes_accumulated: HashSet::with_hasher(BuildHasherDefault::default()),
492        }
493    }
494
495    pub(crate) fn record_change(&mut self, id: NodeId) {
496        self.vectors_changed_since_snapshot.insert(id);
497        self.changes_accumulated.insert(id);
498    }
499
500    pub(crate) fn record_transaction(&mut self) {
501        self.transactions_since_snapshot += 1;
502    }
503
504    pub(crate) fn reset(&mut self, current_time: Timestamp, is_full: bool) {
505        self.transactions_since_snapshot = 0;
506        self.last_snapshot_time = current_time;
507        self.vectors_changed_since_snapshot.clear();
508        self.total_snapshots += 1;
509
510        if is_full {
511            self.last_full_snapshot_time = current_time;
512            self.snapshots_since_full = 0;
513            self.changes_accumulated.clear();
514        } else {
515            self.snapshots_since_full += 1;
516        }
517    }
518}

520/// **Issue #233 Optimization**: This struct combines vector storage and metadata
521/// into a single structure protected by one RwLock, reducing lock acquisitions
