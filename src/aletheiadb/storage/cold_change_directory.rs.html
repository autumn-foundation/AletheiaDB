1//! In-memory, transaction-time-ordered directory of cold-resident change versions (Issue #3677).
2//!
3//! # Why
4//!
5//! `AletheiaDB::list_changes` serves a bi-temporal changefeed window by merging a hot-tier scan
6//! with a cold-tier scan. The cold tier (redb) is keyed by `version_id`, which is **not**
7//! transaction-time ordered (a version's id is allocated at transaction *build* time while its
8//! commit timestamp is assigned later under the commit-clock lock — two concurrent commits can
9//! invert the two orders). So the naive fix — `range()` the ascending `version_id` key and
10//! early-stop — is unsound: it would silently drop or misorder rows.
11//!
12//! This directory is the sound alternative (Approach B): an in-memory `BTreeSet<ChangeCursor>`
13//! that mirrors the *membership* of the cold tier, ordered by the changefeed's real
14//! [`ChangeCursor`] (transaction-time first). A windowed `list_changes` then ranges the set over
15//! the query's `[start, end)` transaction-time window and point-reads only the versions that name
16//! candidate cold rows, decoding `O(window)` versions instead of the whole cold store.
17//!
18//! # Correctness
19//!
20//! - Each entry is *just* a `ChangeCursor` (≈5 machine words) — a self-sufficient pointer that
21//!   already carries `kind_ord` + `version_id`, enough to point-read the exact cold row. The
22//!   materialization decodes the selected rows through the **same** `consider_version` the
23//!   full-scan path uses, so parity is byte-identical by construction.
24//! - The directory is **never authoritative** in two senses. First, it must be **successfully
25//!   seeded** from the cold tier before it may serve any query: an as-yet-unseeded directory, or
26//!   one whose seed scan *failed* over a populated cold store, returns `None` from
27//!   [`ColdChangeDirectory::eligible_candidates`] so the query **degrades to the full cold scan**
28//!   — it never serves an empty-but-untrusted set as if it were a complete answer (which would
29//!   silently drop every cold row). A failed seed latches a permanent `Degraded` state for the
30//!   process lifetime rather than thrashing a re-scan of a broken store on every query. Second,
31//!   even once seeded, a memory budget may evict its oldest entries; an eligibility watermark then
32//!   forces any query whose window could touch an evicted region to degrade to the full cold scan.
33//!   Correctness never depends on the directory being complete — or even present.
34//! - Its `RwLock` is a **leaf**: it is only ever acquired after the `historical` lock is released,
35//!   and it never calls back into `historical` / `wal` / `current_timestamp` (see the CLAUDE.md
36//!   lock order). No redb I/O is performed while it is held.
37
38use crate::core::changefeed::ChangeCursor;
39use crate::core::temporal::TimeRange;
40use parking_lot::RwLock;
41use std::collections::BTreeSet;
42use std::ops::Bound;

44/// In-memory, `ChangeCursor`-ordered directory of the cold tier's change-version membership.
45pub(crate) struct ColdChangeDirectory {
46    inner: RwLock<ColdDirInner>,
47    /// Maximum retained entries. `0` disables the directory entirely (every query degrades to the
48    /// full cold scan).
49    max_entries: usize,
50}
51
52/// Whether the directory has been authoritatively seeded from the cold tier.
53///
54/// This is distinct from [`ColdDirInner::complete`] (the *budget* flag). `complete` answers "has
55/// nothing been evicted?"; `DirectoryState` answers the prior question "may the directory be
56/// trusted to speak for the cold tier at all?". A directory that has never been seeded, or whose
57/// seed scan failed, must **not** serve queries even though its (empty) entry set is trivially
58/// `complete` — otherwise a failed seed over a populated cold store would answer every query with
59/// an empty set, silently dropping all cold rows.
60#[derive(Clone, Copy, PartialEq, Eq, Debug)]
61enum DirectoryState {
62    /// No successful seed has happened yet. Every query degrades to the full cold scan.
63    Unseeded,
64    /// A full seed scan completed successfully; the directory is authoritative (subject only to
65    /// the budget watermark below).
66    Seeded,
67    /// A seed scan failed (cold I/O / decode error) over the cold tier. The directory can never be
68    /// trusted for the remainder of the process lifetime, so every query permanently degrades to
69    /// the (always-correct) full cold scan. Latched — never re-attempted.
70    Degraded,
71}
72
73struct ColdDirInner {
74    /// Transaction-time-ordered membership of cold-resident versions.
75    entries: BTreeSet<ChangeCursor>,
76    /// `false` once anything has been evicted for budget — the directory is then only a partial
77    /// (newest-region) view, guarded by `coverage_watermark`.
78    complete: bool,
79    /// The largest cursor ever evicted for budget. A query whose scan lower bound is at or below
80    /// this watermark could be missing an evicted in-window row, so it must degrade to a scan.
81    coverage_watermark: Option<ChangeCursor>,
82    /// Whether the directory has been authoritatively seeded (see [`DirectoryState`]). Gates
83    /// eligibility independently of `complete`.
84    state: DirectoryState,
85}
86
87impl ColdChangeDirectory {
88    /// Create an empty directory retaining at most `max_entries` cursors (`0` disables it).
89    pub(crate) fn new(max_entries: usize) -> Self {
90        Self {
91            inner: RwLock::new(ColdDirInner {
92                entries: BTreeSet::new(),
93                complete: true,
94                coverage_watermark: None,
95                // Starts NOT seeded: a directory that has not yet scanned the cold tier must not be
96                // trusted, so `eligible_candidates` degrades to the full scan until a successful
97                // seed (`mark_seeded`) makes it authoritative.
98                state: DirectoryState::Unseeded,
99            }),
100            max_entries,
101        }
102    }
103
104    /// Mark the directory as authoritatively seeded after a **successful** full seed scan of the
105    /// cold tier. Only after this does [`eligible_candidates`](Self::eligible_candidates) serve
106    /// queries (a successfully-scanned *empty* cold store counts as seeded, so a fresh empty
107    /// database stays on the fast path). A directory already latched `Degraded` stays degraded.
108    pub(crate) fn mark_seeded(&self) {
109        let mut inner = self.inner.write();
110        if inner.state != DirectoryState::Degraded {
111            inner.state = DirectoryState::Seeded;
112        }
113    }
114
115    /// Latch the directory permanently `Degraded` after a **failed** seed scan (cold I/O / decode
116    /// error). Every subsequent query degrades to the full cold scan for the process lifetime; the
117    /// seed is never re-attempted (no thrash on a broken store).
118    pub(crate) fn mark_degraded(&self) {
119        self.inner.write().state = DirectoryState::Degraded;
120    }
121
122    /// Insert a single cold-resident version's cursor.
123    #[cfg(test)]
124    pub(crate) fn insert(&self, cursor: ChangeCursor) {
125        if self.max_entries == 0 {
126            return;
127        }
128        let mut inner = self.inner.write();
129        inner.entries.insert(cursor);
130        inner.enforce_cap(self.max_entries);
131    }
132
133    /// Insert many cold-resident versions' cursors, enforcing the cap once at the end.
134    pub(crate) fn insert_many(&self, cursors: impl IntoIterator<Item = ChangeCursor>) {
135        if self.max_entries == 0 {
136            return;
137        }
138        let mut inner = self.inner.write();
139        for cursor in cursors {
140            inner.entries.insert(cursor);
141        }
142        inner.enforce_cap(self.max_entries);
143    }
144
145    /// Return the in-window candidate cursors (ascending) the caller may point-read, or `None` to
146    /// signal the query must **degrade to the full cold scan**.
147    ///
148    /// `tx_window` is the half-open `[start, end)` transaction-time window; `resume_after` is the
149    /// strict `> cursor` pagination anchor (already validated by `list_changes`). `bound` is the
150    /// per-tier limit and `has_filter` is `true` iff a valid-time window and/or label filter is
151    /// active for this query.
152    ///
153    /// Eligibility: the directory may serve the window iff (a) it is enabled (`max_entries > 0`),
154    /// (b) it has been authoritatively **seeded** ([`DirectoryState::Seeded`] — an unseeded or
155    /// permanently-degraded directory always degrades, so a failed/absent seed can never serve an
156    /// empty set as a complete answer), and (c) either it is `complete` or its `coverage_watermark`
157    /// is strictly below the query's effective scan lower bound (so no evicted cursor can fall
158    /// inside the window). Otherwise it returns `None` and the caller falls back to the sound full
159    /// scan.
160    ///
161    /// Bound-aware allocation: when no valid-window/label filter is active (`has_filter == false`),
162    /// every in-window candidate survives `consider_version`, so the `bound`-smallest survivors are
163    /// exactly the ascending prefix — the collected candidate `Vec` is capped at `bound`, restoring
164    /// `O(bound)` allocation for the common unfiltered windowed query. When a filter is active
165    /// survival can't be predicted, so the full in-window range is collected (the point-read walk
166    /// still early-stops after `bound` survivors). This changes neither output: the point-read path
167    /// already retains only the `bound`-smallest.
168    pub(crate) fn eligible_candidates(
169        &self,
170        tx_window: &TimeRange,
171        resume_after: Option<ChangeCursor>,
172        bound: usize,
173        has_filter: bool,
174    ) -> Option<Vec<ChangeCursor>> {
175        if self.max_entries == 0 {
176            return None;
177        }
178
179        // Half-open window bounds as sentinel cursors: `min_at(start)` (inclusive lower) and
180        // `min_at(end)` (exclusive upper) exactly select versions whose commit lies in
181        // `[start, end)`, mirroring `TimeRange::contains`.
182        let start_cursor = ChangeCursor::min_at(tx_window.start());
183        let hi = ChangeCursor::min_at(tx_window.end());
184
185        // Effective lower bound of the scan: the max of the window start and (exclusively) the
186        // resume anchor. This is the smallest cursor the directory could be asked to serve.
187        let lo = match resume_after {
188            Some(r) if r > start_cursor => r,
189            _ => start_cursor,
190        };
191
192        let inner = self.inner.read();
193
194        // Authority guard: an unseeded or permanently-degraded directory must never be trusted —
195        // degrade to the (always-correct) full cold scan regardless of `complete`.
196        if inner.state != DirectoryState::Seeded {
197            return None;
198        }
199
200        // Partial-coverage guard: if anything was evicted and the largest evicted cursor is at or
201        // above the query's lower bound, an in-window row might be missing → degrade to scan.
202        if !inner.complete
203            && let Some(watermark) = inner.coverage_watermark
204            && watermark >= lo
205        {
206            return None;
207        }
208
209        // Retained set fully covers the window → collect the in-range cursors (ascending). The
210        // lower bound is exclusive of `resume_after` when it dominates the window start (strict
211        // `> cursor` resume), else inclusive of the window start.
212        let lo_bound = match resume_after {
213            Some(r) if r > start_cursor => Bound::Excluded(r),
214            _ => Bound::Included(start_cursor),
215        };
216        let range = inner
217            .entries
218            .range((lo_bound, Bound::Excluded(hi)))
219            .copied();
220        // Unfiltered windowed query: only the `bound`-smallest can reach the page, so cap the
221        // candidate allocation at the ascending prefix. A filtered query keeps the whole range.
222        let candidates: Vec<ChangeCursor> = if !has_filter && bound != usize::MAX {
223            range.take(bound).collect()
224        } else {
225            range.collect()
226        };
227        Some(candidates)
228    }
229
230    /// Test-only: number of retained cursors.
231    #[cfg(test)]
232    pub(crate) fn len(&self) -> usize {
233        self.inner.read().entries.len()
234    }
235
236    /// Test-only: whether the directory still holds every cold cursor (nothing evicted).
237    #[cfg(test)]
238    pub(crate) fn is_complete(&self) -> bool {
239        self.inner.read().complete
240    }
241
242    /// Test-only: whether the directory has been authoritatively seeded (may serve queries).
243    #[cfg(test)]
244    pub(crate) fn is_seeded(&self) -> bool {
245        self.inner.read().state == DirectoryState::Seeded
246    }
247
248    /// Test-only: whether the directory has latched the permanent `Degraded` state.
249    #[cfg(test)]
250    pub(crate) fn is_degraded(&self) -> bool {
251        self.inner.read().state == DirectoryState::Degraded
252    }
253}
254
255impl ColdDirInner {
256    /// Evict the smallest (oldest tx-time) cursors until within `max_entries`, advancing the
257    /// coverage watermark and clearing the completeness flag.
258    fn enforce_cap(&mut self, max_entries: usize) {
259        while self.entries.len() > max_entries {
260            let Some(evicted) = self.entries.pop_first() else {
261                break;
262            };
263            self.complete = false;
264            self.coverage_watermark = Some(match self.coverage_watermark {
265                Some(w) => w.max(evicted),
266                None => evicted,
267            });
268        }
269    }
270}

278    fn tr(from: i64, to: i64) -> TimeRange {
279        TimeRange::new(Timestamp::from(from), Timestamp::from(to)).unwrap()
280    }
281
282    fn node_cursor(tx: i64, id: u64) -> ChangeCursor {
283        ChangeCursor::for_version(Timestamp::from(tx), EntityKind::Node, id, id)
284    }

286    #[test]
287    fn unseeded_directory_degrades_even_when_complete() {
288        // A never-seeded directory is trivially `complete` (empty), but must still degrade — a
289        // failed/absent seed must never serve an empty set as if it were a complete answer.
290        let dir = ColdChangeDirectory::new(1000);
291        assert!(dir.is_complete());
292        assert!(!dir.is_seeded());
293        assert!(
294            dir.eligible_candidates(&tr(0, 1_000_000), None, usize::MAX, false)
295                .is_none(),
296            "unseeded directory must degrade to scan"
297        );
298    }

300    #[test]
301    fn degraded_directory_never_serves() {
302        let dir = ColdChangeDirectory::new(1000);
303        dir.mark_degraded();
304        // A subsequent successful seed can never un-degrade it.
305        dir.mark_seeded();
306        assert!(dir.is_degraded());
307        assert!(
308            dir.eligible_candidates(&tr(0, 1_000_000), None, usize::MAX, false)
309                .is_none()
310        );
311    }

324    #[test]
325    fn ranges_half_open_window() {
326        let dir = ColdChangeDirectory::new(1000);
327        for i in 1..=10u64 {
328            dir.insert(node_cursor((i as i64) * 100, i));
329        }
330        dir.mark_seeded();
331        // Window [300, 700): ticks 300,400,500,600 -> ids 3,4,5,6.
332        let got = dir
333            .eligible_candidates(&tr(300, 700), None, usize::MAX, false)
334            .unwrap();
335        let ids: Vec<u64> = got.iter().map(|c| c.version_id).collect();
336        assert_eq!(ids, vec![3, 4, 5, 6]);
337    }

340    fn resume_after_is_exclusive() {
341        let dir = ColdChangeDirectory::new(1000);

358    #[test]
359    fn bound_caps_unfiltered_candidate_prefix() {
360        let dir = ColdChangeDirectory::new(1000);
361        for i in 1..=10u64 {
362            dir.insert(node_cursor((i as i64) * 100, i));
363        }
364        dir.mark_seeded();
365        // Unfiltered, bound 3 over window [0, 1100): only the ascending prefix 1,2,3 is collected.
366        let got = dir
367            .eligible_candidates(&tr(0, 1100), None, 3, false)
368            .unwrap();
369        let ids: Vec<u64> = got.iter().map(|c| c.version_id).collect();
370        assert_eq!(ids, vec![1, 2, 3], "unfiltered bound caps the prefix");
371        // Filtered: the full in-window range is collected (survival can't be predicted).
372        let got_filtered = dir
373            .eligible_candidates(&tr(0, 1100), None, 3, true)
374            .unwrap();
375        assert_eq!(
376            got_filtered.len(),
377            10,
378            "filtered query keeps the full range"
379        );
380    }

382    #[test]
383    fn eviction_sets_watermark_and_degrades_old_window() {
384        let dir = ColdChangeDirectory::new(3);
385        for i in 1..=10u64 {
386            dir.insert(node_cursor((i as i64) * 100, i));
387        }
388        dir.mark_seeded();
389        assert!(!dir.is_complete());
390        assert_eq!(dir.len(), 3, "capped to max_entries");
391        // Recent window above the watermark is eligible.
392        assert!(
393            dir.eligible_candidates(&tr(850, 1100), None, usize::MAX, false)
394                .is_some()
395        );
396        // Old window below the watermark degrades.
397        assert!(
398            dir.eligible_candidates(&tr(0, 500), None, usize::MAX, false)
399                .is_none()
400        );
401    }
