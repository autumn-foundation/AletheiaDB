1//! WAL recovery logic.
2//!
3//! This module provides the shared logic for replaying Write-Ahead Log (WAL) entries
4//! back into the database's storage engines. It acts as the bridge between durable
5//! disk storage and the in-memory/disk-backed data structures.
6//!
7//! # The "Resurrection" Process
8//!
9//! When AletheiaDB experiences an ungraceful shutdown (a crash or power failure),
10//! the data in memory is lost. To ensure data durability (the 'D' in ACID), every
11//! committed transaction is first written to the WAL on disk.
12//!
13//! Upon restart, this module reads those persisted WAL entries and "replays" them.
14//! It reconstructs the exact state of the database by systematically applying every
15//! recorded `Create`, `Update`, and `Delete` operation to both the `CurrentStorage`
16//! (the hot, current state of the graph) and the `HistoricalStorage` (the temporal,
17//! versioned state of the graph).
18//!
19//! # Why is it needed?
20//!
21//! Replaying the WAL is necessary because building the database's optimized read
22//! structures (like the CSR adjacency matrices or vector indexes) is expensive and
23//! typically done in memory. Instead of syncing these complex structures to disk on
24//! every write, we sync a simple, sequential log of operations. The recovery process
25//! uses this log to rebuild the complex structures exactly as they were.
26

38/// Replay WAL entries starting from a given LSN into existing storage instances.
39///
40/// # Arguments
41///
42/// * `wal` - The concurrent WAL system to read entries from
43/// * `current` - Current storage to apply updates to
44/// * `historical` - Historical storage to apply updates to
45/// * `start_lsn` - The LSN to start replaying from (inclusive)
46/// * `next_version_id` - The starting version ID for new versions created during replay
47///
48/// # Returns
49///
50/// A tuple containing:
51/// - The final LSN reached after replay (current LSN of the WAL)
52/// - The maximum node ID observed (if any)
53/// - The maximum edge ID observed (if any)
54/// - The next version ID to use (updated after replay)
55///
56/// # Examples
57///
58/// ```rust,ignore
59/// # use tempfile::tempdir;
60/// # use std::sync::Arc;
61/// # use aletheiadb::WalConfigBuilder;
62/// # use aletheiadb::storage::wal::{LSN, WalOperation};
63/// # use aletheiadb::storage::wal::concurrent_system::ConcurrentWalSystem;
64/// # use aletheiadb::storage::current::CurrentStorage;
65/// # use aletheiadb::storage::historical::HistoricalStorage;
66/// # // Note: This function is pub(crate) and generally called internally by
67/// # // AletheiaDB::new() or CheckpointManager::recover().
68/// # use aletheiadb::core::id::{NodeId, VersionId, TxId};
69/// # use aletheiadb::core::interning::InternedString;
70/// # use aletheiadb::core::property::PropertyMap;
71/// # use aletheiadb::core::temporal::Timestamp;
72/// # use aletheiadb::core::GLOBAL_INTERNER;
73/// # fn main() -> aletheiadb::core::error::Result<()> {
74/// // 1. Set up a temporary directory and WAL system
75/// let dir = tempdir().unwrap();
76/// let mut config = WalConfigBuilder::new().build();
77/// config.wal_dir = dir.path().to_path_buf();
78/// let wal = ConcurrentWalSystem::new(&config)?;
79///
80/// // 2. Simulate a crash: Write a raw entry directly to the WAL
81/// let node_id = NodeId::new(1).unwrap();
82/// let label = GLOBAL_INTERNER.intern("User").unwrap();
83/// let ts = Timestamp::from(100);
84///
85/// let op = WalOperation::CreateNode {
86///     node_id,
87///     label,
88///     properties: PropertyMap::default(),
89///     valid_from: ts,
90/// };
91/// wal.append(op)?; // Append takes just the operation, timestamp is assigned by WAL
92/// wal.flush()?;
93///
94/// // 3. The Resurrection: Create empty storage instances
95/// let current = CurrentStorage::new();
96/// let mut historical = HistoricalStorage::new();
97///
98/// // 4. Replay the WAL into storage
99/// let start_lsn = LSN::initial();
100/// let initial_version_id = 1;
101///
102/// // Internally, AletheiaDB calls this during initialization
103/// let (final_lsn, max_node, _, next_vid) = replay_wal_into_storage(
104///     &wal,
105///     &current,
106///     &mut historical,
107///     start_lsn,
108///     initial_version_id
109/// )?;
110///
111/// // 5. Verify the state was reconstructed
112/// assert_eq!(max_node, Some(1));
113/// assert!(current.get_node(node_id).is_ok(), "Node was recovered!");
114/// # Ok(())
115/// # }
116/// ```
117pub(crate) fn replay_wal_into_storage(
118    wal: &ConcurrentWalSystem,
119    current: &CurrentStorage,
120    historical: &mut HistoricalStorage,
121    start_lsn: LSN,
122    next_version_id: u64,
123) -> Result<(LSN, Option<u64>, Option<u64>, u64)> {
124    replay_wal_into_storage_with_constraints(
125        wal,
126        current,
127        historical,
128        start_lsn,
129        next_version_id,
130        None,
131    )
132}

134pub(crate) fn replay_wal_into_storage_with_constraints(
135    wal: &ConcurrentWalSystem,
136    current: &CurrentStorage,
137    historical: &mut HistoricalStorage,
138    start_lsn: LSN,
139    next_version_id: u64,
140    constraint_registry: Option<&ConstraintRegistry>,
141) -> Result<(LSN, Option<u64>, Option<u64>, u64)> {
142    // Read the segment directory once, then apply. Callers that have already
143    // read the WAL for startup (Issue #3429) skip this read entirely and call
144    // `replay_entries_into_storage_with_constraints` directly with the
145    // pre-decoded entries, avoiding a redundant full decode of the WAL.
146    let wal_entries = wal.read_from(start_lsn)?;
147    replay_entries_into_storage_with_constraints(
148        wal,
149        wal_entries,
150        current,
151        historical,
152        next_version_id,
153        constraint_registry,
154    )
155}

157/// Apply an already-decoded WAL entry stream to storage (Issue #3429).
158///
159/// This is the body of [`replay_wal_into_storage_with_constraints`] with the
160/// segment read hoisted out. The startup path decodes the WAL exactly once and
161/// feeds the resulting entries here (filtered to `>= start_lsn`) so recovery no
162/// longer re-reads the segment directory for the differential replay. `wal` is
163/// retained only to report the post-replay `current_lsn()`; every operation
164/// applied comes from `wal_entries`, never from a fresh read.
165///
166/// The caller MUST pass exactly the entries with `LSN >= start_lsn` (in the
167/// same LSN-sorted order [`ConcurrentWalSystem::read_from`] produces): the
168/// transaction-framing resolver below is order-sensitive and expects the same
169/// contiguous suffix a direct `read_from(start_lsn)` would return.
170pub(crate) fn replay_entries_into_storage_with_constraints(
171    wal: &ConcurrentWalSystem,
172    wal_entries: Vec<WalEntry>,
173    current: &CurrentStorage,
174    historical: &mut HistoricalStorage,
175    mut next_version_id: u64,
176    constraint_registry: Option<&ConstraintRegistry>,
177) -> Result<(LSN, Option<u64>, Option<u64>, u64)> {
178    const RECOVERY_TX_ID: u64 = 0;
179
180    let mut max_node_id: Option<u64> = None;
181    let mut max_edge_id: Option<u64> = None;
182
183    if !wal_entries.is_empty() {
184        #[cfg(feature = "observability")]
185        tracing::info!("Replaying {} WAL entries", wal_entries.len());
186        #[cfg(not(feature = "observability"))]
187        eprintln!("Replaying {} WAL entries", wal_entries.len());
188    }
189
190    // Resolve transaction framing (Issue #3413) BEFORE applying: pair every
191    // operation with the timestamp it must be recorded at, and drop any
192    // uncommitted transaction tail. Framed data ops committed by a durable
193    // `CommitTx` are paired with that marker's authoritative commit timestamp
194    // (so an atomic batch's versions all share ONE transaction time); legacy
195    // (pre-v7) and non-transactional entries keep their own logged timestamp.
196    let resolved = resolve_transaction_frames(wal_entries);
197
198    for (operation, commit_timestamp) in resolved {
199        match operation {
200            WalOperation::CreateNode {
201                node_id,
202                label,
203                properties,
204                valid_from,
205                provenance,
206            } => {
207                max_node_id = Some(match max_node_id {
208                    Some(current_max) => current_max.max(node_id.as_u64()),
209                    None => node_id.as_u64(),
210                });
211
212                // Idempotent re-application guard (Issue #3419). Replay starts
213                // AT the manifest LSN (inclusive) because the snapshot may or
214                // may not contain effects of entries at/after that LSN. An
215                // existing node whose historical head is a LIVE (open valid
216                // interval) version means this exact create was already
217                // applied; re-applying it blindly would append a duplicate
218                // version to the node's bi-temporal history. The two stores
219                // are guarded independently because background persistence
220                // may snapshot them at different LSNs.
221                //
222                // Review round 2 (#3428): a head with a CLOSED valid interval
223                // (delete tombstone or #3230 retraction) does NOT reflect this
224                // create — the id was deleted/retracted and this entry
225                // RE-CREATES it (a legal in-WAL sequence: create → delete →
226                // create with the same id). Treating that head as "already
227                // applied" left current pointing at the tombstone's version
228                // and lost the new incarnation's history entirely.
229                let current_node = current.get_node(node_id).ok();
230                let historical_head = historical.get_current_node_version(node_id);
231                let live_head = historical_head.filter(|vid| {
232                    historical
233                        .get_node_version(*vid)
234                        .is_some_and(|v| v.temporal.valid_time().is_current())
235                });
236                if current_node.is_some() && live_head.is_some() {
237                    continue;
238                }
239
240                let interned_label = label;
241
242                // Transaction time is `commit_timestamp` (Issue #3413): for a
243                // framed transaction this is the CommitTx marker's single
244                // authoritative commit time (shared by every op in the batch);
245                // for legacy/non-transactional entries it is the entry's own
246                // logged timestamp. Either way it is supplied by the caller.
247                let metadata = VersionMetadata::new(TxId::new(RECOVERY_TX_ID), commit_timestamp);
248                // Reuse whichever store already applied this create so that
249                // `current.current_version == historical head id` stays
250                // aligned; only allocate a fresh recovery version id when
251                // neither store has one to align to.
252                let version_id = match (live_head, &current_node) {
253                    // Historical already holds this create: align current.
254                    (Some(vid), _) => vid,
255                    // Current already holds this create: align history to its
256                    // version id — unless that id already exists in history
257                    // (inconsistent pre-fix state), in which case a fresh id
258                    // is the only safe choice.
259                    (None, Some(node))
260                        if historical.get_node_version(node.current_version).is_none() =>
261                    {
262                        node.current_version
263                    }
264                    _ => VersionId::new(next_version_id)?,
265                };
266                next_version_id = next_version_id.max(version_id.as_u64() + 1);
267
268                if current_node.is_none() {
269                    let node = Node::with_metadata(
270                        node_id,
271                        interned_label,
272                        properties.clone(),
273                        version_id,
274                        metadata,
275                    );
276                    current.insert_node_direct(node, commit_timestamp)?;
277                }
278
279                if live_head.is_none() {
280                    // Re-creation after a tombstone/retraction: supersede the
281                    // closed head in transaction time (mirroring the
282                    // UpdateNode arm) before appending the new incarnation.
283                    //
284                    // DELIBERATELY KEPT — NOT redundant (Issue #3407). The
285                    // following `add_node_version_with_provenance` also closes
286                    // the prior head via `close_previous_version_intervals`,
287                    // BUT that helper guards the tx-time close with a STRICT
288                    // `new_tx_start > prev_tx_start`. When a delete/retract and
289                    // this re-creation land in the SAME transaction they share
290                    // one framed `commit_timestamp`, so the strict `>` SKIPS the
291                    // close and the tombstone head would be left open — leaving
292                    // two versions with overlapping open tx-intervals after
293                    // replay. The explicit close is UNCONDITIONAL (matching the
294                    // live write path in `api/transaction/write/apply.rs`),
295                    // producing the empty `[T, T)` tx-interval live and replay
296                    // alike. See the reachability test in
297                    // `tests/recovery/replay_update_tests.rs`.
298                    if let Some(prev) = historical_head {
299                        historical.close_node_version_transaction_time(prev, commit_timestamp)?;
300                    }
301                    historical.add_node_version_with_provenance(
302                        node_id,
303                        version_id,
304                        valid_from,
305                        commit_timestamp,
306                        interned_label,
307                        properties,
308                        false, // not a tombstone
309                        provenance.map(std::sync::Arc::new),
310                    )?;
311                }
312            }
313            WalOperation::CreateEdge {
314                edge_id,
315                source,
316                target,
317                label,
318                properties,
319                valid_from,
320                provenance,
321            } => {
322                max_edge_id = Some(match max_edge_id {
323                    Some(current_max) => current_max.max(edge_id.as_u64()),
324                    None => edge_id.as_u64(),
325                });
326
327                // Idempotent re-application guard (Issue #3419) — see the
328                // CreateNode arm above. Review round 2 (#3428): a tombstone/
329                // retraction head likewise means this entry RE-CREATES the
330                // edge id, not that the create was already applied.
331                let current_edge = current.get_edge(edge_id).ok();
332                let historical_head = historical.get_current_edge_version(edge_id);
333                let live_head = historical_head.filter(|vid| {
334                    historical
335                        .get_edge_version(*vid)
336                        .is_some_and(|v| v.temporal.valid_time().is_current())
337                });
338                if current_edge.is_some() && live_head.is_some() {
339                    continue;
340                }
341
342                let interned_label = label;
343
344                let metadata = VersionMetadata::new(TxId::new(RECOVERY_TX_ID), commit_timestamp);
345                let version_id = match (live_head, &current_edge) {
346                    (Some(vid), _) => vid,
347                    (None, Some(edge))
348                        if historical.get_edge_version(edge.current_version).is_none() =>
349                    {
350                        edge.current_version
351                    }
352                    _ => VersionId::new(next_version_id)?,
353                };
354                next_version_id = next_version_id.max(version_id.as_u64() + 1);
355
356                if current_edge.is_none() {
357                    let edge = Edge::with_metadata(
358                        edge_id,
359                        interned_label,
360                        source,
361                        target,
362                        properties.clone(),
363                        version_id,
364                        metadata,
365                    );
366                    current.insert_edge_direct(edge)?;
367                }
368
369                if live_head.is_none() {
370                    // DELIBERATELY KEPT — NOT redundant (Issue #3407); the
371                    // helper's strict `>` tx-time guard skips the close when a
372                    // same-transaction delete/retract + re-create share one
373                    // framed `commit_timestamp`. See the CreateNode arm above.
374                    if let Some(prev) = historical_head {
375                        historical.close_edge_version_transaction_time(prev, commit_timestamp)?;
376                    }
377                    historical.add_edge_version_with_provenance(
378                        edge_id,
379                        version_id,
380                        valid_from,
381                        commit_timestamp,
382                        interned_label,
383                        source,
384                        target,
385                        properties,
386                        false, // not a tombstone
387                        provenance.map(std::sync::Arc::new),
388                    )?;
389                }
390            }
391            WalOperation::UpdateNode {
392                node_id,
393                version_id,
394                label,
395                properties,
396                valid_from,
397                provenance,
398            } => {
399                next_version_id = next_version_id.max(version_id.as_u64() + 1);
400
401                let interned_label = label;
402
403                let metadata = VersionMetadata::new(TxId::new(RECOVERY_TX_ID), commit_timestamp);
404
405                let node = Node::with_metadata(
406                    node_id,
407                    interned_label,
408                    properties.clone(),
409                    version_id,
410                    metadata,
411                );
412
413                // Overwriting current state with the logged version is
414                // idempotent; always apply so a lagging graph snapshot
415                // converges to the logged state.
416                current.update_node_direct(node, commit_timestamp)?;
417
418                // Idempotent re-application guard (Issue #3419): updates log
419                // their version_id, so "this exact version already exists in
420                // history" means the update was already applied. Re-applying
421                // it blindly would append the version AGAIN with ITSELF as
422                // the previous head — a self-referential delta chain that
423                // makes history reconstruction loop forever (observed
424                // empirically: get_node_history OOMs after a double replay).
425                if historical.get_node_version(version_id).is_none() {
426                    // DELIBERATELY KEPT — NOT redundant (Issue #3407). The
427                    // `add_node_version_with_provenance` below ALSO closes the
428                    // prior head via `close_previous_version_intervals`, but
429                    // that helper guards the tx-time close with a STRICT
430                    // `new_tx_start > prev_tx_start`. Two updates to the SAME
431                    // node in ONE transaction share a single framed
432                    // `commit_timestamp`, so the successor's tx start EQUALS the
433                    // intermediate version's tx start and the strict `>` SKIPS
434                    // the close — the intermediate would be left OPEN on replay
435                    // while the live path (unconditional close in
436                    // `api/transaction/write/apply.rs`) closes it to the empty
437                    // `[T, T)` interval. Leaving it open produces two versions
438                    // with overlapping open tx-intervals, so an
439                    // `AS OF SYSTEM_TIME = T` read could surface the superseded
440                    // version. This UNCONDITIONAL close keeps replay bit-for-bit
441                    // consistent with the live write path. Regression coverage:
442                    // `replay_double_update_same_node_in_one_tx_closes_intermediate`.
443                    if let Some(prev_version_id) = historical.get_current_node_version(node_id) {
444                        historical.close_node_version_transaction_time(
445                            prev_version_id,
446                            commit_timestamp,
447                        )?;
448                    }
449
450                    historical.add_node_version_with_provenance(
451                        node_id,
452                        version_id,
453                        valid_from,
454                        commit_timestamp,
455                        interned_label,
456                        properties,
457                        false, // not a tombstone
458                        provenance.map(std::sync::Arc::new),
459                    )?;
460                }
461            }
462            WalOperation::UpdateEdge {
463                edge_id,
464                version_id,
465                label,
466                properties,
467                valid_from,
468                provenance,
469            } => {
470                next_version_id = next_version_id.max(version_id.as_u64() + 1);
471
472                let current_edge = current.get_edge(edge_id)?;
473
474                let interned_label = label;
475
476                let metadata = VersionMetadata::new(TxId::new(RECOVERY_TX_ID), commit_timestamp);
477
478                let edge = Edge::with_metadata(
479                    edge_id,
480                    interned_label,
481                    current_edge.source,
482                    current_edge.target,
483                    properties.clone(),
484                    version_id,
485                    metadata,
486                );
487
488                current.update_edge_direct(edge)?;
489
490                // Idempotent re-application guard (Issue #3419) — see the
491                // UpdateNode arm above for why an already-present version_id
492                // must not be appended to history a second time.
493                if historical.get_edge_version(version_id).is_none() {
494                    // DELIBERATELY KEPT — NOT redundant (Issue #3407); the
495                    // helper's strict `>` tx-time guard skips the close when two
496                    // updates to the same edge in ONE transaction share a framed
497                    // `commit_timestamp`. See the UpdateNode arm above.
498                    if let Some(prev_version_id) = historical.get_current_edge_version(edge_id) {
499                        historical.close_edge_version_transaction_time(
500                            prev_version_id,
501                            commit_timestamp,
502                        )?;
503                    }
504
505                    historical.add_edge_version_with_provenance(
506                        edge_id,
507                        version_id,
508                        valid_from,
509                        commit_timestamp,
510                        interned_label,
511                        current_edge.source,
512                        current_edge.target,
513                        properties,
514                        false, // not a tombstone
515                        provenance.map(std::sync::Arc::new),
516                    )?;
517                }
518            }
519            WalOperation::DeleteNode {
520                node_id,
521                valid_from,
522                version_id: logged_tombstone_id,
523                provenance,
524            } => {
525                // Idempotent re-application guard, per store (review round 2,
526                // #3428). Background persistence may snapshot current and
527                // historical at different LSNs, so a boundary delete can be
528                // reflected in one store but not the other:
529                //   (a) node still in current + tombstone already in
530                //       historical → only the current-side removal is owed;
531                //       re-applying the historical side would append a SECOND
532                //       tombstone.
533                //   (b) node already gone from current + head still live in
534                //       historical → only the historical closure/tombstone is
535                //       owed; gating it on current presence (the pre-fix
536                //       behavior) left the head open forever.
537                // The historical side applies iff the head is still live; the
538                // current side applies iff the node is present.
539                let current_node = current.get_node(node_id).ok();
540                let historical_head = historical.get_current_node_version(node_id);
541                let live_head = historical_head.filter(|vid| {
542                    historical
543                        .get_node_version(*vid)
544                        .is_some_and(|v| v.temporal.valid_time().is_current())
545                });
546
547                // Tombstone payload: prefer the current node, else reconstruct
548                // from the live historical head (case (b) above).
549                let tombstone_payload = if let Some(node) = &current_node {
550                    if live_head.is_some() || historical_head.is_none() {
551                        Some((node.label, node.properties.clone()))
552                    } else {
553                        None // head already tombstoned/retracted: case (a)
554                    }
555                } else if let Some(vid) = live_head {
556                    let label = historical.get_node_version(vid).map(|v| v.label);
557                    match label {
558                        Some(label) => Some((label, historical.reconstruct_node_properties(vid)?)),
559                        None => None,
560                    }
561                } else {
562                    None
563                };
564
565                if let Some((label, properties)) = tombstone_payload {
566                    // DELIBERATELY KEPT — NOT redundant (Issue #3407). The
567                    // tombstone `add_node_version` below also closes the prior
568                    // head via `close_previous_version_intervals`, but that
569                    // helper's STRICT `new_tx_start > prev_tx_start` guard SKIPS
570                    // the tx-time close when an update-then-delete of the same
571                    // node in ONE transaction shares a framed `commit_timestamp`
572                    // — leaving the superseded head open on replay while the
573                    // live path (unconditional close) closes it. This
574                    // UNCONDITIONAL close keeps replay consistent with the live
575                    // write path. See the UpdateNode arm above.
576                    if let Some(current_version_id) = historical_head {
577                        historical.close_node_version_transaction_time(
578                            current_version_id,
579                            commit_timestamp,
580                        )?;
581                    }
582
583                    // Honor the LOGGED tombstone version_id (Issue #3406) so the
584                    // recovered version chain is bit-identical to the live one
585                    // and replay is idempotent; segments predating
586                    // WAL_VERSION_DELETE_VERSION_ID carry no id, so synthesize as
587                    // before. Either way, advance next_version_id past it to keep
588                    // future synthesized ids collision-free.
589                    let tombstone_version_id = match logged_tombstone_id {
590                        Some(vid) => vid,
591                        None => VersionId::new(next_version_id)?,
592                    };
593                    next_version_id = next_version_id.max(tombstone_version_id.as_u64() + 1);
594
595                    // Honor the LOGGED valid_from (possibly backdated, Issues
596                    // #3221/#3400) — mirroring the live path's
597                    // `apply_node_delete` — while tx_time comes from the WAL
598                    // entry. The is_tombstone=true flag closes the valid_time
599                    // immediately at valid_from (empty interval).
600                    historical.add_node_version_with_provenance(
601                        node_id,
602                        tombstone_version_id,
603                        valid_from,
604                        commit_timestamp,
605                        label,
606                        properties,
607                        true, // is_tombstone
608                        // Issue #3427: thread the acting-principal provenance
609                        // logged with the delete so recovery reproduces the
610                        // attribution instead of dropping it.
611                        provenance.map(std::sync::Arc::new),
612                    )?;
613                }
614
615                if current_node.is_some() {
616                    current.delete_node_direct(node_id, commit_timestamp)?;
617                }
618            }
619            WalOperation::DeleteEdge {
620                edge_id,
621                valid_from,
622                version_id: logged_tombstone_id,
623                provenance,
624            } => {
625                // Per-store re-application guard — see the DeleteNode arm
626                // above (review round 2, #3428).
627                let current_edge = current.get_edge(edge_id).ok();
628                let historical_head = historical.get_current_edge_version(edge_id);
629                let live_head = historical_head.filter(|vid| {
630                    historical
631                        .get_edge_version(*vid)
632                        .is_some_and(|v| v.temporal.valid_time().is_current())
633                });
634
635                let tombstone_payload = if let Some(edge) = &current_edge {
636                    if live_head.is_some() || historical_head.is_none() {
637                        Some((
638                            edge.label,
639                            edge.source,
640                            edge.target,
641                            edge.properties.clone(),
642                        ))
643                    } else {
644                        None // head already tombstoned/retracted
645                    }
646                } else if let Some(vid) = live_head {
647                    let head = historical
648                        .get_edge_version(vid)
649                        .map(|v| (v.label, v.source, v.target));
650                    match head {
651                        Some((label, source, target)) => Some((
652                            label,
653                            source,
654                            target,
655                            historical.reconstruct_edge_properties(vid)?,
656                        )),
657                        None => None,
658                    }
659                } else {
660                    None
661                };
662
663                if let Some((label, source, target, properties)) = tombstone_payload {
664                    // DELIBERATELY KEPT — NOT redundant (Issue #3407); the
665                    // helper's strict `>` tx-time guard skips the close when an
666                    // update-then-delete of the same edge in ONE transaction
667                    // shares a framed `commit_timestamp`. See the DeleteNode /
668                    // UpdateNode arms above.
669                    if let Some(current_version_id) = historical_head {
670                        historical.close_edge_version_transaction_time(
671                            current_version_id,
672                            commit_timestamp,
673                        )?;
674                    }
675
676                    // Honor the LOGGED tombstone version_id (Issue #3406); see
677                    // the DeleteNode arm above for rationale and the synthesis
678                    // fallback for pre-v9 segments.
679                    let tombstone_version_id = match logged_tombstone_id {
680                        Some(vid) => vid,
681                        None => VersionId::new(next_version_id)?,
682                    };
683                    next_version_id = next_version_id.max(tombstone_version_id.as_u64() + 1);
684
685                    // Honor the LOGGED valid_from (possibly backdated, Issues
686                    // #3221/#3400) — mirroring the live path's
687                    // `apply_edge_delete` — while tx_time comes from the WAL
688                    // entry. The is_tombstone=true flag closes the valid_time
689                    // immediately at valid_from (empty interval).
690                    historical.add_edge_version_with_provenance(
691                        edge_id,
692                        tombstone_version_id,
693                        valid_from,
694                        commit_timestamp,
695                        label,
696                        source,
697                        target,
698                        properties,
699                        true, // is_tombstone
700                        // Issue #3427: thread the acting-principal provenance.
701                        provenance.map(std::sync::Arc::new),
702                    )?;
703                }
704
705                if current_edge.is_some() {
706                    current.delete_edge_direct(edge_id)?;
707                }
708            }
709            WalOperation::RetractNode {
710                node_id,
711                valid_to,
712                version_id: logged_retraction_id,
713                provenance,
714            } => {
715                // Valid-time retraction (Issue #3230). Reconstruct exactly what
716                // the original commit did:
717                //   (a) close the head version's TRANSACTION time (its valid
718                //       interval stays untouched — append-only, never rewrite);
719                //   (b) append a new version with the same properties whose
720                //       valid interval is closed at the logged `valid_to`
721                //       (honored faithfully — NOT the replay/commit time);
722                //   (c) remove the node from current storage.
723                //
724                // Guarded per store (review round 2, #3428), like DeleteNode:
725                // the historical side applies iff the head is still live (a
726                // head already retracted/tombstoned means this retraction's
727                // historical effect was applied); the current-side removal
728                // applies iff the node is present.
729                let current_node = current.get_node(node_id).ok();
730                let head_version_id = historical.get_current_node_version(node_id);
731                let live_head = head_version_id.filter(|vid| {
732                    historical
733                        .get_node_version(*vid)
734                        .is_some_and(|v| v.temporal.valid_time().is_current())
735                });
736
737                // Retraction payload: prefer the current node, else
738                // reconstruct from the live historical head.
739                let payload = if let Some(node) = &current_node {
740                    if live_head.is_some() || head_version_id.is_none() {
741                        Some((node.label, node.properties.clone()))
742                    } else {
743                        None // head already retracted/tombstoned
744                    }
745                } else if let Some(vid) = live_head {
746                    let label = historical.get_node_version(vid).map(|v| v.label);
747                    match label {
748                        Some(label) => Some((label, historical.reconstruct_node_properties(vid)?)),
749                        None => None,
750                    }
751                } else {
752                    None
753                };
754
755                if let Some((label, properties)) = payload {
756                    // Capture the head's valid_from BEFORE appending, so the
757                    // retraction version reproduces the same interval start.
758                    let valid_from = head_version_id
759                        .and_then(|vid| historical.get_node_version(vid))
760                        .map(|v| v.temporal.valid_time().start())
761                        .unwrap_or(commit_timestamp);
762
763                    // Step (a): close the head version's TRANSACTION time (its
764                    // valid interval stays untouched — append-only).
765                    //
766                    // DELIBERATELY KEPT — NOT redundant (Issue #3407). The
767                    // `add_retracted_node_version` below also closes the prior
768                    // head via `close_previous_version_intervals`, but that
769                    // helper's tx-time close is guarded by a STRICT
770                    // `new_tx_start > prev_tx_start`. An update-then-retract of
771                    // the same node in ONE transaction shares a framed
772                    // `commit_timestamp`, so the strict `>` SKIPS the close and
773                    // the superseded head would be left open on replay — while
774                    // the live path (unconditional close) closes it. This
775                    // UNCONDITIONAL close keeps replay consistent with live. (The
776                    // helper additionally leaves the head's valid interval
777                    // untouched here, since the retraction reuses the head's
778                    // `valid_from`.) See the UpdateNode arm above.
779                    if let Some(current_version_id) = head_version_id {
780                        historical.close_node_version_transaction_time(
781                            current_version_id,
782                            commit_timestamp,
783                        )?;
784                    }
785
786                    // Honor the LOGGED retraction version_id (Issue #3406); see
787                    // the DeleteNode arm for rationale and the pre-v9 synthesis
788                    // fallback.
789                    let retraction_version_id = match logged_retraction_id {
790                        Some(vid) => vid,
791                        None => VersionId::new(next_version_id)?,
792                    };
793                    next_version_id = next_version_id.max(retraction_version_id.as_u64() + 1);
794
795                    historical.add_retracted_node_version_with_provenance(
796                        node_id,
797                        retraction_version_id,
798                        valid_from,
799                        valid_to,
800                        commit_timestamp,
801                        label,
802                        properties,
803                        // Issue #3427: thread the acting-principal provenance.
804                        provenance.map(std::sync::Arc::new),
805                    )?;
806                }
807
808                if current_node.is_some() {
809                    current.delete_node_direct(node_id, commit_timestamp)?;
810                }
811            }
812            WalOperation::RetractEdge {
813                edge_id,
814                valid_to,
815                version_id: logged_retraction_id,
816                provenance,
817            } => {
818                // Valid-time retraction of an edge (Issue #3230); mirrors the
819                // RetractNode arm above, honoring the logged `valid_to`, with
820                // the same per-store re-application guards (#3428).
821                let current_edge = current.get_edge(edge_id).ok();
822                let head_version_id = historical.get_current_edge_version(edge_id);
823                let live_head = head_version_id.filter(|vid| {
824                    historical
825                        .get_edge_version(*vid)
826                        .is_some_and(|v| v.temporal.valid_time().is_current())
827                });
828
829                let payload = if let Some(edge) = &current_edge {
830                    if live_head.is_some() || head_version_id.is_none() {
831                        Some((
832                            edge.label,
833                            edge.source,
834                            edge.target,
835                            edge.properties.clone(),
836                        ))
837                    } else {
838                        None // head already retracted/tombstoned
839                    }
840                } else if let Some(vid) = live_head {
841                    let head = historical
842                        .get_edge_version(vid)
843                        .map(|v| (v.label, v.source, v.target));
844                    match head {
845                        Some((label, source, target)) => Some((
846                            label,
847                            source,
848                            target,
849                            historical.reconstruct_edge_properties(vid)?,
850                        )),
851                        None => None,
852                    }
853                } else {
854                    None
855                };
856
857                if let Some((label, source, target, properties)) = payload {
858                    let valid_from = head_version_id
859                        .and_then(|vid| historical.get_edge_version(vid))
860                        .map(|v| v.temporal.valid_time().start())
861                        .unwrap_or(commit_timestamp);
862
863                    // Step (a): close the head version's TRANSACTION time.
864                    // DELIBERATELY KEPT — NOT redundant (Issue #3407); the
865                    // helper's strict `>` tx-time guard skips the close when an
866                    // update-then-retract of the same edge in ONE transaction
867                    // shares a framed `commit_timestamp`. See the RetractNode /
868                    // UpdateNode arms above.
869                    if let Some(current_version_id) = head_version_id {
870                        historical.close_edge_version_transaction_time(
871                            current_version_id,
872                            commit_timestamp,
873                        )?;
874                    }
875
876                    // Honor the LOGGED retraction version_id (Issue #3406); see
877                    // the DeleteNode arm for rationale and the pre-v9 synthesis
878                    // fallback.
879                    let retraction_version_id = match logged_retraction_id {
880                        Some(vid) => vid,
881                        None => VersionId::new(next_version_id)?,
882                    };
883                    next_version_id = next_version_id.max(retraction_version_id.as_u64() + 1);
884
885                    historical.add_retracted_edge_version_with_provenance(
886                        edge_id,
887                        retraction_version_id,
888                        valid_from,
889                        valid_to,
890                        commit_timestamp,
891                        label,
892                        source,
893                        target,
894                        properties,
895                        // Issue #3427: thread the acting-principal provenance.
896                        provenance.map(std::sync::Arc::new),
897                    )?;
898                }
899
900                if current_edge.is_some() {
901                    current.delete_edge_direct(edge_id)?;
902                }
903            }
904            WalOperation::Checkpoint { .. } => {
905                // Checkpoint markers are informational only during replay
906            }
907            WalOperation::DeclareUniqueConstraint { label, property } => {
908                if let Some(reg) = constraint_registry {
909                    reg.declare(label, property);
910                }
911            }
912            WalOperation::DropUniqueConstraint { label, property } => {
913                if let Some(reg) = constraint_registry {
914                    reg.undeclare(label, property);
915                }
916            }
917            // Transaction framing markers (Issue #3413) are consumed by
918            // `resolve_transaction_frames` above and never reach this apply
919            // loop; the arm exists only for match exhaustiveness.
920            WalOperation::BeginTx { .. } | WalOperation::CommitTx { .. } => {}
921        }
922    }
923
924    let final_lsn = wal.current_lsn();
925
926    Ok((final_lsn, max_node_id, max_edge_id, next_version_id))
927}

944/// Are the buffered ops' LSNs strictly consecutive (`l, l+1, l+2, …`)?
945///
946/// A committing transaction owns one contiguous LSN band (single atomic
947/// `allocate_batch`), so a gap means the buffer does not correspond to an
948/// intact batch and the frame must be rejected (Issue #3413, integrity check
949/// R4).
950fn lsns_contiguous(pending: &[(LSN, WalOperation)]) -> bool {
951    pending
952        .windows(2)
953        .all(|w| w[1].0.0 == w[0].0.0.wrapping_add(1))
954}

956/// Resolve transaction framing (Issue #3413), turning the raw, LSN-ordered WAL
957/// entry stream into the ordered list of `(operation, transaction_timestamp)`
958/// pairs that replay must apply — with uncommitted transaction tails removed.
959///
960/// # Load-bearing invariant: replay begins at a transaction-frame boundary
961///
962/// This resolver assumes the entry stream it is handed **never begins in the
963/// middle of a `[BeginTx .. CommitTx]` band** — i.e. `start_lsn` is always a
964/// band start (or a legacy/unframed boundary), never a data op or `CommitTx`
965/// whose `BeginTx` was already skipped. Production guarantees this: startup
966/// replays from an inclusive `start_lsn = manifest.lsn` (see `db::config`), and
967/// the manifest LSN is the observable next-to-allocate LSN, which
968/// `allocate_batch` only ever advances by a WHOLE `[BeginTx .. CommitTx]` band
969/// in one atomic step — so it can never fall mid-band. If that invariant were
970/// ever violated (see the note on `checkpoint.rs`'s exclusive `.next()`
971/// convention, currently test-only and unwired), a `CommitTx` could arrive with
972/// no open frame; this resolver treats that as a **benign skip** (its `BeginTx`
973/// lies below `start_lsn`, so there are no buffered ops to apply) rather than a
974/// "failed validation" corruption warning.
975///
976/// # Framed-replay invariant
977///
978/// A committing `WriteTransaction` writes `[BeginTx, ..data ops.., CommitTx]`
979/// as one atomic, contiguous LSN band inside a single flush epoch (see
980/// `api::transaction::write::wal::log_operations_to_wal`). Because the durable
981/// suffix of the WAL is always an LSN-ordered *prefix* of what was appended,
982/// the terminal `CommitTx` marker is present on disk **iff** every op of its
983/// batch is — so:
984///
985/// * A `BeginTx … data … CommitTx` frame whose marker validated (matching
986///   `tx_id`, `entry_count` equal to the buffered op count, contiguous LSNs)
987///   is applied **atomically**, and every one of its ops is stamped with the
988///   marker's single `commit_timestamp`. This is what makes an atomic batch's
989///   versions share one transaction time so `AS OF SYSTEM_TIME` can never
990///   observe a half-batch.
991/// * A frame whose marker never reached disk (a crash tail), or that fails the
992///   integrity checks, is **discarded whole** — never applied as a torn prefix.
993/// * Entries from pre-v7 segments (`!framed`), and any op appended outside a
994///   frame (raw `append`/`append_async`, control ops such as checkpoints and
995///   constraint declarations), keep the legacy **immediate-apply** path using
996///   their own logged timestamp. This preserves the behavior of every existing
997///   non-transactional WAL producer unchanged.
998fn resolve_transaction_frames(entries: Vec<WalEntry>) -> Vec<(WalOperation, Timestamp)> {
999    let mut resolved: Vec<(WalOperation, Timestamp)> = Vec::with_capacity(entries.len());
1000    // Buffered data ops of the currently-open frame, with their LSNs.
1001    let mut pending: Vec<(LSN, WalOperation)> = Vec::new();
1002    // `(tx_id, begin_lsn)` of the currently-open frame, if any. The `BeginTx`
1003    // LSN is retained so the terminal `CommitTx` can bracket-check the band
1004    // (Issue #3413 integrity check R5): the whole `[BeginTx .. CommitTx]` band
1005    // must be gap-free at BOTH ends, not just internally contiguous.
1006    let mut open_tx: Option<(u64, LSN)> = None;
1007
1008    for entry in entries {
1009        // Legacy / non-transactional entries apply immediately with their own
1010        // logged timestamp (pre-v7 segments never carry framing markers).
1011        if !entry.framed {
1012            resolved.push((entry.operation, entry.timestamp));
1013            continue;
1014        }
1015
1016        // Copy the header fields we need before moving `operation` out.
1017        let entry_lsn = entry.lsn;
1018        let entry_ts = entry.timestamp;
1019
1020        match entry.operation {
1021            WalOperation::BeginTx { tx_id } => {
1022                if !pending.is_empty() {
1023                    warn_discarded_frame(
1024                        "new BeginTx encountered before the previous frame committed",
1025                        pending.len(),
1026                    );
1027                    pending.clear();
1028                }
1029                open_tx = Some((tx_id, entry_lsn));
1030            }
1031            WalOperation::CommitTx {
1032                tx_id,
1033                entry_count,
1034                commit_timestamp,
1035            } => {
1036                match open_tx {
1037                    // Benign skip (see the load-bearing invariant on this fn): a
1038                    // `CommitTx` with no open frame is a band whose `BeginTx`
1039                    // lies below `start_lsn`. Under the production invariant this
1040                    // never happens; if it ever did, `pending` is empty and there
1041                    // is nothing to apply — quietly skip rather than emit a
1042                    // spurious corruption warning.
1043                    None => {
1044                        debug_assert!(
1045                            pending.is_empty(),
1046                            "a CommitTx with no open frame must have no buffered ops"
1047                        );
1048                    }
1049                    Some((open_id, begin_lsn)) => {
1050                        // Bracket check (R5): the band must be gap-free at both
1051                        // ends — the first data op immediately follows BeginTx
1052                        // and CommitTx immediately follows the last data op — in
1053                        // addition to being internally contiguous (R4) and
1054                        // matching the marker's tx_id / entry_count.
1055                        let brackets_ok = pending.first().map(|(l, _)| l.0)
1056                            == Some(begin_lsn.0.wrapping_add(1))
1057                            && pending.last().map(|(l, _)| l.0.wrapping_add(1))
1058                                == Some(entry_lsn.0);
1059                        let valid = open_id == tx_id
1060                            && pending.len() == entry_count as usize
1061                            && lsns_contiguous(&pending)
1062                            && brackets_ok;
1063                        if valid {
1064                            // Commit the frame atomically: every op takes the
1065                            // marker's authoritative commit timestamp.
1066                            for (_lsn, op) in pending.drain(..) {
1067                                resolved.push((op, commit_timestamp));
1068                            }
1069                        } else {
1070                            warn_discarded_frame(
1071                                "CommitTx failed validation (tx_id / entry_count / LSN contiguity / band brackets)",
1072                                pending.len(),
1073                            );
1074                            pending.clear();
1075                        }
1076                    }
1077                }
1078                open_tx = None;
1079            }
1080            // Self-committing control ops are never written inside a
1081            // transaction batch; if a frame is somehow open it is torn, so
1082            // discard it, then apply the control op immediately.
1083            op @ (WalOperation::Checkpoint { .. }
1084            | WalOperation::DeclareUniqueConstraint { .. }
1085            | WalOperation::DropUniqueConstraint { .. }) => {
1086                if !pending.is_empty() {
1087                    warn_discarded_frame(
1088                        "control op encountered inside an open transaction frame",
1089                        pending.len(),
1090                    );
1091                    pending.clear();
1092                }
1093                open_tx = None;
1094                resolved.push((op, entry_ts));
1095            }
1096            // A framed data op.
1097            op => {
1098                if open_tx.is_some() {
1099                    pending.push((entry_lsn, op));
1100                } else {
1101                    // ⚠️ LANDMINE (Issue #3413): a framed data op with NO open
1102                    // frame falls through to immediate-apply with its OWN
1103                    // per-entry timestamp — exactly the per-op timestamp
1104                    // bisection transaction framing exists to prevent. This is
1105                    // deliberately reachable ONLY for raw `append`/`append_async`
1106                    // into a v7+ segment (test/tooling paths), which carry the
1107                    // `framed` flag from the segment version yet legitimately
1108                    // have no `BeginTx`; for them immediate-apply is correct.
1109                    // The real committing producer (`log_operations_to_wal`)
1110                    // ALWAYS brackets its data ops in a `[BeginTx .. CommitTx]`
1111                    // band, so a *transactional* op never lands here. If any
1112                    // future producer raw-appends framed data ops that were
1113                    // meant to share a commit timestamp, this branch would
1114                    // silently re-bisect them — audit this path before adding
1115                    // one.
1116                    resolved.push((op, entry_ts));
1117                }
1118            }
1119        }
1120    }
1121
1122    // Any buffer still open at end-of-stream is an uncommitted crash tail.
1123    if !pending.is_empty() {
1124        warn_discarded_frame(
1125            "WAL ended with an uncommitted transaction frame (crash tail)",
1126            pending.len(),
1127        );
1128    }
1129
1130    resolved
1131}

1133/// Replay ONLY constraint declaration/drop entries from the full WAL history.
1134///
1135/// Because constraint declarations are written to the WAL before the
1136/// corresponding node data, they may lie at LSNs below the persisted snapshot
1137/// LSN and therefore be skipped by the normal differential replay. This reads
1138/// every WAL entry from LSN 0 and replays only the two constraint-related
1139/// operations, giving the net constraint state.
1140///
1141/// As of Issue #3429 the startup path no longer calls this: it decodes the WAL
1142/// once and folds the constraint net-state out of the already-read entries via
1143/// [`apply_constraint_declarations`], avoiding a dedicated full read. This
1144/// wrapper (a thin `read_from` + `apply_constraint_declarations`) is retained
1145/// for the recovery unit tests that exercise the read-and-apply path directly.
1146#[cfg(test)]
1147pub(crate) fn replay_constraint_declarations_from_wal(
1148    wal: &ConcurrentWalSystem,
1149    registry: &ConstraintRegistry,
1150) -> Result<()> {
1151    let all_entries = wal.read_from(LSN::initial())?;
1152    apply_constraint_declarations(&all_entries, registry);
1153    Ok(())
1154}

1156/// Fold the constraint declare/drop net-state out of an already-decoded WAL
1157/// entry stream (Issue #3429).
1158///
1159/// Same net effect as [`replay_constraint_declarations_from_wal`] but operates
1160/// on entries the startup path has *already* read from disk, so recovery does
1161/// not decode the segment directory a second time just to recover constraint
1162/// state. `entries` is expected to span the full WAL history (from
1163/// [`LSN::initial()`]) so declarations that predate the index snapshot LSN --
1164/// and would be skipped by the differential replay -- are still applied.
1165///
1166/// Declares and drops are replayed in stream order; the net set is
1167/// order-dependent (declare then drop == dropped), matching the prior full-read
1168/// pass exactly. Borrows the entries (does not consume them) so the same vector
1169/// can then be filtered and handed to the differential replay.
1170pub(crate) fn apply_constraint_declarations(entries: &[WalEntry], registry: &ConstraintRegistry) {
1171    for entry in entries {
1172        match entry.operation {
1173            WalOperation::DeclareUniqueConstraint { label, property } => {
1174                registry.declare(label, property);
1175            }
1176            WalOperation::DropUniqueConstraint { label, property } => {
1177                registry.undeclare(label, property);
1178            }
1179            _ => {}
1180        }
1181    }
1182}
