14/// Read WAL entries from a directory, starting from the specified LSN.
15///
16/// This is a standalone function that can be used for recovery without
17/// requiring an active WAL writer. It reads all segment files in the directory
18/// and parses entries that have LSN >= start_lsn.
19///
20/// # Arguments
21///
22/// * `wal_dir` - Path to the WAL directory containing segment files
23/// * `start_lsn` - Only entries with LSN >= this value are returned
24///
25/// # Returns
26///
27/// A vector of WAL entries sorted by LSN.
28pub fn read_wal_entries(wal_dir: &Path, start_lsn: LSN) -> Result<Vec<WalEntry>> {
29    segment_reader::read_entries_from_dir(wal_dir, start_lsn)
30}
31
32/// Read WAL entries from a directory with an explicit crash-torn-tail recovery
33/// policy (Issue #3433).
34///
35/// When `tolerate_torn_tail` is `true` (the default recovery behavior), an
36/// undecodable trailing entry in the final segment stops replay there and the
37/// intact prefix is applied; when `false`, any parse failure hard-errors
38/// (fail-stop recovery). See
39/// [`segment_reader::read_entries_from_dir_with_options`].
40pub fn read_wal_entries_with_options(
41    wal_dir: &Path,
42    start_lsn: LSN,
43    tolerate_torn_tail: bool,
44) -> Result<Vec<WalEntry>> {
45    read_wal_entries_with_cipher_and_options(wal_dir, start_lsn, None, tolerate_torn_tail)
46}
47
48/// Read WAL entries from a directory, decrypting encrypted segments with the
49/// supplied `cipher`, and honoring the crash-torn-tail recovery policy.
50///
51/// This is the cipher-aware counterpart to [`read_wal_entries_with_options`].
52/// Recovery replay for an encryption-at-rest database MUST route through here
53/// with the configured WAL cipher: encrypted segments (versions 2/4/6/8/10)
54/// cannot be decoded without it, so a cipher-less replay of an encrypted WAL
55/// tail hard-errors (`Cannot read encrypted WAL segment ... without a cipher`)
56/// and would otherwise brick startup after a crash that left acknowledged
57/// writes in the WAL past the last index snapshot. Passing `None` reproduces
58/// the plaintext behavior exactly.
59pub fn read_wal_entries_with_cipher_and_options(
60    wal_dir: &Path,
61    start_lsn: LSN,
62    cipher: Option<&Arc<dyn Cipher>>,
63    tolerate_torn_tail: bool,
64) -> Result<Vec<WalEntry>> {
65    segment_reader::read_entries_from_dir_with_options(
66        wal_dir,
67        start_lsn,
68        cipher,
69        tolerate_torn_tail,
70    )
71}

73#[cfg(test)]
74mod tests {
75    use super::*;
76    use crate::GLOBAL_INTERNER;
77    use crate::core::id::NodeId;
78    use crate::core::property::PropertyMap;
79    use crate::core::temporal::time;
80    use crate::storage::wal::DurabilityMode;
81    use crate::storage::wal::WalOperation;
82    use crate::storage::wal::concurrent_system::{ConcurrentWalSystem, ConcurrentWalSystemConfig};
83    use tempfile::tempdir;
84
85    fn create_op(id: u64) -> WalOperation {
86        WalOperation::CreateNode {
87            node_id: NodeId::new(id).unwrap(),
88            label: GLOBAL_INTERNER.intern(format!("ReaderNode{id}")).unwrap(),
89            properties: PropertyMap::new(),
90            valid_from: time::now(),
91            provenance: None,
92        }
93    }
94
95    /// Plaintext directory: `read_wal_entries_with_cipher_and_options` with
96    /// `None` (and its `read_wal_entries_with_options` delegator) must return
97    /// every durable entry. Exercises the plaintext-delegation branch of the
98    /// new cipher-aware reader directly at the unit level (integration tests
99    /// that also hit it live under `tests/`, which coverage ignores).
100    #[test]
101    fn cipher_and_options_reads_plaintext_with_none() {
102        let dir = tempdir().unwrap();
103        {
104            let config = ConcurrentWalSystemConfig::new(dir.path())
105                .with_durability_mode(DurabilityMode::Synchronous);
106            let wal = ConcurrentWalSystem::new(config).unwrap();
107            wal.append(create_op(1)).unwrap();
108            wal.append(create_op(2)).unwrap();
109            // drop → flush any remainder
110        }
111
112        let via_cipher_fn =
113            read_wal_entries_with_cipher_and_options(dir.path(), LSN(1), None, true).unwrap();
114        assert_eq!(
115            via_cipher_fn.len(),
116            2,
117            "plaintext read via the cipher-aware function (None) must return both entries"
118        );
119
120        // The thin delegator must return the identical set.
121        let via_delegator = read_wal_entries_with_options(dir.path(), LSN(1), true).unwrap();
122        let lsns_a: Vec<u64> = via_cipher_fn.iter().map(|e| e.lsn.0).collect();
123        let lsns_b: Vec<u64> = via_delegator.iter().map(|e| e.lsn.0).collect();
124        assert_eq!(
125            lsns_a, lsns_b,
126            "delegator must match the cipher-aware reader"
127        );
128        assert_eq!(lsns_a, vec![1, 2]);
129    }

131    /// Encrypted directory: the reader MUST thread the supplied cipher through
132    /// (decrypt path) — and MUST fail without it. Guards the fix that
133    /// `ConcurrentWalSystem::read_from` no longer drops the cipher, and kills
134    /// the mutant that would ignore the `cipher` argument.
135    #[cfg(feature = "encryption")]
136    #[test]
137    fn cipher_and_options_decrypts_encrypted_and_requires_cipher() {
138        use crate::encryption::key_provider::FileKeyProvider;
139        use crate::encryption::{EncryptionConfig, EncryptionManager};
140
141        let dir = tempdir().unwrap();
142        let key_path = dir.path().join("mek.key");
143        FileKeyProvider::generate_key_file(&key_path).unwrap();
144        let manager =
145            EncryptionManager::from_config(&EncryptionConfig::file_based(&key_path)).unwrap();
146        let cipher = Arc::clone(manager.wal_cipher());
147
148        {
149            let mut config = ConcurrentWalSystemConfig::new(dir.path())
150                .with_durability_mode(DurabilityMode::Synchronous);
151            config.wal_cipher = Some(Arc::clone(&cipher));
152            let wal = ConcurrentWalSystem::new(config).unwrap();
153            wal.append(create_op(1)).unwrap();
154            wal.append(create_op(2)).unwrap();
155        }
156
157        // With the cipher: entries decode.
158        let decrypted =
159            read_wal_entries_with_cipher_and_options(dir.path(), LSN(1), Some(&cipher), true)
160                .unwrap();
161        assert_eq!(
162            decrypted.iter().map(|e| e.lsn.0).collect::<Vec<_>>(),
163            vec![1, 2],
164            "encrypted segments must decode when the cipher is threaded through"
165        );
166
167        // Without the cipher: encrypted segments cannot be read (proves the
168        // cipher argument is load-bearing, not ignored).
169        assert!(
170            read_wal_entries_with_cipher_and_options(dir.path(), LSN(1), None, true).is_err(),
171            "reading an encrypted WAL without a cipher must hard-error"
172        );
173    }
