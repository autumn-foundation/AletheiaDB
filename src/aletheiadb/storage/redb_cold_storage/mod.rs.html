1//! Redb-based cold storage backend for tiered storage architecture.
2//!
3//! This module implements the "Cold" tier of AletheiaDB's storage hierarchy using [Redb](https://github.com/cberner/redb),
4//! a pure Rust embedded database. It provides durable, disk-based storage for historical data that has
5//! aged out of the "Warm" memory tier.
6//!
7//! # Role in Architecture
8//!
9//! 1.  **Historical Storage**: Persists `NodeVersion` and `EdgeVersion` objects that represent the
10//!     state of the graph at past points in time.
11//! 2.  **WAL Truncation Coordination**: Tracks the "Flushed LSN" (Log Sequence Number). The Write-Ahead
12//!     Log (WAL) can only be truncated up to the point that has been safely persisted here.
13//! 3.  **Compression**: Automatically compresses data using Zstd or LZ4 (via `compression` module)
14//!     to minimize disk usage.
15//!
16//! # Schema
17//!
18//! The storage uses three internal tables:
19//! - **`node_versions`**: Maps `VersionId` (u64) -> Compressed `NodeVersion` (bytes).
20//! - **`edge_versions`**: Maps `VersionId` (u64) -> Compressed `EdgeVersion` (bytes).
21//! - **`metadata`**: Singleton table storing system state, primarily the `flushed_lsn`.
22//!
23//! # Concurrency
24//!
25//! Redb provides ACID guarantees. Read transactions are snapshot-isolated and non-blocking.
26//! Write transactions are serialized (one active writer at a time). This module handles
27//! the transaction management internally.
28//!
29//! # Example
30//!
31//! ```no_run
32//! use aletheiadb::storage::redb_cold_storage::{RedbColdStorage, RedbConfig};
33//! use aletheiadb::storage::wal::LSN;
34//! use aletheiadb::core::version::{NodeVersion, EdgeVersion};
35//!
36//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
37//! let config = RedbConfig::default();
38//! let storage = RedbColdStorage::new("data/cold.redb", config)?;
39//!
40//! // Store versions with LSN tracking (atomic batch)
41//! let node_versions: Vec<NodeVersion> = vec![]; // ... populate
42//! let edge_versions: Vec<EdgeVersion> = vec![]; // ... populate
43//! let lsn = LSN(1000);
44//!
45//! storage.store_batch_with_lsn(&node_versions, &edge_versions, lsn)?;
46//!
47//! // Get flushed LSN for WAL truncation
48//! let flushed_lsn = storage.get_flushed_lsn()?;
49//! assert_eq!(flushed_lsn, Some(lsn));
50//! # Ok(())
51//! # }
52//! ```

186/// Durable resume cursor for the cold bulk re-encrypt pass (Issue #3617 PR3).
187///
188/// Hand-rolled fixed-width bytes (no serde, so the Feature Matrix stays clean):
189/// `[version:u8][target_version:u32 LE][table:u8][last_completed_key:u64 LE]`.
190/// `last_completed_key` is the highest `VersionId` already re-wrapped in
191/// `table`; the pass resumes at `> last_completed_key`.
192#[derive(Debug, Clone, Copy, PartialEq, Eq)]
193struct ColdRotationCursor {
194    target_version: u32,
195    table: ColdTableKind,
196    last_completed_key: u64,
197}
198
199impl ColdRotationCursor {
200    const LEN: usize = 1 + 4 + 1 + 8;
201
202    fn to_bytes(self) -> [u8; Self::LEN] {
203        let mut out = [0u8; Self::LEN];
204        out[0] = COLD_ROTATION_RECORD_VERSION;
205        out[1..5].copy_from_slice(&self.target_version.to_le_bytes());
206        out[5] = self.table.as_u8();
207        out[6..14].copy_from_slice(&self.last_completed_key.to_le_bytes());
208        out
209    }
210
211    fn from_bytes(bytes: &[u8]) -> Option<Self> {
212        if bytes.len() != Self::LEN || bytes[0] != COLD_ROTATION_RECORD_VERSION {
213            return None;
214        }
215        let target_version = u32::from_le_bytes([bytes[1], bytes[2], bytes[3], bytes[4]]);
216        let table = ColdTableKind::from_u8(bytes[5])?;
217        let last_completed_key = u64::from_le_bytes([
218            bytes[6], bytes[7], bytes[8], bytes[9], bytes[10], bytes[11], bytes[12], bytes[13],
219        ]);
220        Some(Self {
221            target_version,
222            table,
223            last_completed_key,
224        })
225    }
226}

279/// Persisted bi-temporal extent bounds for the cold tier (Issue #3389).
280///
281/// Cold-migrated versions are removed from the hot-tier temporal indexes, so
282/// after a restart the write-time extent aggregate — which is rebuilt from the
283/// hot tier only — would silently omit them, wrongly narrowing the reported
284/// [`temporal_extent`](crate::AletheiaDB::temporal_extent). To close that gap
285/// the cold store maintains these bounds incrementally as versions are written
286/// to it, persisted durably in the `metadata` table;
287/// [`RedbColdStorage::get_temporal_extent_bounds`] returns them at startup so
288/// they can be merged back into the aggregate.
289///
290/// Bounds follow the exact convention of the in-memory extent aggregate:
291/// `earliest` is the minimum interval start; `latest` is the maximum of
292/// interval starts and *closed* interval ends — an open (`TIMESTAMP_MAX`) end
293/// contributes only its start, so the open-interval sentinel never leaks. All
294/// four fields are `None` on an empty cold tier and become `Some` together
295/// once any version is stored. Merging is monotonic (bounds only ever widen).
296#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
297pub struct ColdTemporalExtentBounds {
298    /// Minimum valid-time interval start observed across cold versions.
299    pub valid_earliest: Option<Timestamp>,
300    /// Maximum finite valid-time coordinate observed across cold versions.
301    pub valid_latest: Option<Timestamp>,
302    /// Minimum transaction-time interval start observed across cold versions.
303    pub tx_earliest: Option<Timestamp>,
304    /// Maximum finite transaction-time coordinate observed across cold versions.
305    pub tx_latest: Option<Timestamp>,
306}
307
308impl ColdTemporalExtentBounds {
309    /// `true` when no bounds have been observed (all fields `None`).
310    fn is_empty(&self) -> bool {
311        self.valid_earliest.is_none()
312            && self.valid_latest.is_none()
313            && self.tx_earliest.is_none()
314            && self.tx_latest.is_none()
315    }
316
317    /// Fold one `[start, end)` range into a single dimension's bounds, applying
318    /// the extent convention: `earliest` = min start; `latest` = max of starts
319    /// and *closed* ends (an open `TIMESTAMP_MAX` end contributes only its
320    /// start, so the sentinel never leaks).
321    fn observe_range(
322        earliest: &mut Option<Timestamp>,
323        latest: &mut Option<Timestamp>,
324        start: Timestamp,
325        end: Timestamp,
326    ) {
327        *earliest = Some(earliest.map_or(start, |e| e.min(start)));
328        // TimeRange guarantees end >= start, so a closed end is itself the
329        // latest candidate; open ends fall back to the start.
330        let candidate = if end == TIMESTAMP_MAX { start } else { end };
331        *latest = Some(latest.map_or(candidate, |l| l.max(candidate)));
332    }
333
334    /// Fold one bi-temporal interval (both dimensions) into these bounds.
335    fn observe_interval(&mut self, temporal: &BiTemporalInterval) {
336        let valid = temporal.valid_time();
337        Self::observe_range(
338            &mut self.valid_earliest,
339            &mut self.valid_latest,
340            valid.start(),
341            valid.end(),
342        );
343        let tx = temporal.transaction_time();
344        Self::observe_range(
345            &mut self.tx_earliest,
346            &mut self.tx_latest,
347            tx.start(),
348            tx.end(),
349        );
350    }
351
352    /// Accumulate the bounds contributed by a batch of versions.
353    fn from_versions<V: TemporalVersion>(versions: &[V]) -> Self {
354        let mut bounds = Self::default();
355        for version in versions {
356            bounds.observe_interval(version.temporal());
357        }
358        bounds
359    }
360
361    /// Merge another set of bounds into these (monotonic widen-only union).
362    fn merge(&mut self, other: &Self) {
363        Self::merge_dim(
364            &mut self.valid_earliest,
365            &mut self.valid_latest,
366            other.valid_earliest,
367            other.valid_latest,
368        );
369        Self::merge_dim(
370            &mut self.tx_earliest,
371            &mut self.tx_latest,
372            other.tx_earliest,
373            other.tx_latest,
374        );
375    }
376
377    fn merge_dim(
378        earliest: &mut Option<Timestamp>,
379        latest: &mut Option<Timestamp>,
380        other_earliest: Option<Timestamp>,
381        other_latest: Option<Timestamp>,
382    ) {
383        if let Some(e) = other_earliest {
384            *earliest = Some(earliest.map_or(e, |cur| cur.min(e)));
385        }
386        if let Some(l) = other_latest {
387            *latest = Some(latest.map_or(l, |cur| cur.max(l)));
388        }
389    }
390
391    /// Serialize to the fixed-length metadata record layout.
392    fn to_bytes(self) -> [u8; EXTENT_RECORD_LEN] {
393        let mut buf = [0u8; EXTENT_RECORD_LEN];
394        buf[0] = EXTENT_RECORD_VERSION;
395        let mut offset = 1;
396        for field in [
397            self.valid_earliest,
398            self.valid_latest,
399            self.tx_earliest,
400            self.tx_latest,
401        ] {
402            if let Some(ts) = field {
403                buf[offset] = 1;
404                buf[offset + 1..offset + 9].copy_from_slice(&ts.wallclock().to_le_bytes());
405                buf[offset + 9..offset + 13].copy_from_slice(&ts.logical().to_le_bytes());
406            }
407            offset += 13;
408        }
409        buf
410    }
411
412    /// Deserialize from the metadata record layout. Returns `Ok(None)` for an
413    /// unrecognized version tag or wrong length (treated as absent), never an
414    /// error, so a forward-incompatible record can never panic startup.
415    fn from_bytes(bytes: &[u8]) -> Option<Self> {
416        if bytes.len() != EXTENT_RECORD_LEN || bytes[0] != EXTENT_RECORD_VERSION {
417            return None;
418        }
419        let read_field = |offset: usize| -> Option<Timestamp> {
420            if bytes[offset] == 0 {
421                return None;
422            }
423            let mut wall = [0u8; 8];
424            wall.copy_from_slice(&bytes[offset + 1..offset + 9]);
425            let mut logical = [0u8; 4];
426            logical.copy_from_slice(&bytes[offset + 9..offset + 13]);
427            // Values were validated when the source versions were created, so
428            // reconstructing them from trusted internal storage is sound.
429            Some(Timestamp::new_unchecked(
430                i64::from_le_bytes(wall),
431                u32::from_le_bytes(logical),
432            ))
433        };
434        Some(Self {
435            valid_earliest: read_field(1),
436            valid_latest: read_field(14),
437            tx_earliest: read_field(27),
438            tx_latest: read_field(40),
439        })
440    }
441}

482/// Compression algorithm for cold storage.
483#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
484#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
485pub enum CompressionAlgorithm {
486    /// No compression (fastest, but uses more disk space)
487    None,
488    /// Zstd compression (ratio-optimized, default)
489    #[default]
490    Zstd,
491    /// LZ4-compatible fast compression using Zstd level 1
492    /// Provides similar speed characteristics to LZ4 with better compatibility
493    Fast,
494}
495
496impl CompressionAlgorithm {
497    /// Get the `zstd` compression level for this algorithm.
498    ///
499    /// The compression level balances speed and ratio. This method translates
500    /// the abstract `CompressionAlgorithm` enum into the specific integer level
501    /// expected by the `zstd` crate.
502    ///
503    /// ## Examples
504    ///
505    /// ```rust
506    /// use aletheiadb::storage::redb_cold_storage::CompressionAlgorithm;
507    ///
508    /// assert_eq!(CompressionAlgorithm::Zstd.zstd_level(), Some(3));
509    /// assert_eq!(CompressionAlgorithm::Fast.zstd_level(), Some(1));
510    /// assert_eq!(CompressionAlgorithm::None.zstd_level(), None);
511    /// ```
512    pub fn zstd_level(&self) -> Option<i32> {
513        match self {
514            CompressionAlgorithm::None => None,
515            CompressionAlgorithm::Zstd => Some(3), // Balanced ratio/speed
516            CompressionAlgorithm::Fast => Some(1), // Speed-optimized
517        }
518    }
519}

607    /// in the cold storage tier. A higher ratio indicates better compression.
608    ///
609    /// The underlying byte counters are not persisted, so this ratio covers
610    /// writes made since the storage was opened (1.0 when nothing has been
611    /// written yet by this process).
612    ///
613    /// ## Examples
614    ///
615    /// ```rust
616    /// use aletheiadb::storage::redb_cold_storage::ColdStorageStats;
617    ///
618    /// let mut stats = ColdStorageStats::default();
619    /// stats.bytes_written_raw = 1000;
620    /// stats.bytes_written_compressed = 250;
621    /// assert_eq!(stats.compression_ratio(), 4.0); // 4x compression!
622    /// ```
623    pub fn compression_ratio(&self) -> f64 {
624        if self.bytes_written_compressed == 0 {
625            1.0
626        } else {
627            self.bytes_written_raw as f64 / self.bytes_written_compressed as f64
628        }
629    }
630}

919/// Redb-based cold storage implementation.
920///
921/// This struct provides a durable, disk-based storage backend for historical version data.
922/// It uses Redb for ACID-compliant storage and supports compression (Zstd/LZ4) and
923/// checksum validation.
924///
925/// # Key Features
926///
927/// - **Atomic Batches**: Store multiple node and edge versions atomically.
928/// - **LSN Tracking**: Tracks the highest LSN flushed to disk to coordinate WAL truncation.
929/// - **Compression**: Transparent compression of version data (Zstd by default).
930/// - **Integrity**: Optional CRC32 checksums on top of Redb's guarantees.
931pub struct RedbColdStorage {
932    /// Path to the database file.
933    path: PathBuf,
934    /// Redb database instance.
935    db: redb::Database,
936    /// Configuration.
937    config: RedbConfig,
938    /// Statistics tracker.
939    stats: AtomicColdStorageStats,
940    /// Optional cold key ring for encrypting data at rest (Issue #3617 PR3),
941    /// held in a runtime-swappable presence cell (Issue #3708).
942    ///
943    /// Empty (`None`) when encryption is disabled (values stored as bare
944    /// compressed bytes, unchanged from the unencrypted path). When present it
945    /// holds one generation in the steady state and two during a full-MEK key
946    /// rotation, so a half-rotated store reads every value under its own
947    /// generation via the `ACV1` wrapper dispatch.
948    ///
949    /// The cell is an [`ArcSwapOption`] so a keyring can be **installed at
950    /// runtime** (`None` -> `Some`, Issue #3708) into a live plaintext store —
951    /// the cold-tier mirror of the WAL's
952    /// [`install_wal_keyring`](crate::storage::wal::concurrent_system::ConcurrentWalSystem::install_wal_keyring)
953    /// seam (#3669) — without reopening the store. Every read path
954    /// (`is_encrypted`, `encrypt_if_needed`, `decrypt_if_needed`,
955    /// `prepare_batch`, ...) `load()`s it lock-free, so an install is observed
956    /// atomically and never as a torn read; the install itself is serialized by
957    /// [`install_lock`](Self::install_lock).
958    keyring: ArcSwapOption<ColdKeyring>,
959    /// Serializes runtime keyring installs (Issue #3708) so two concurrent
960    /// installers cannot both observe the cell as `None`, both pass the presence
961    /// check, and both store -- the second silently replacing the first keyring.
962    /// Held for the entire [`install_cold_keyring`](Self::install_cold_keyring)
963    /// body (presence check + store) so a second concurrent installer blocks,
964    /// then observes `Some`, and returns the existing rejection `Err`. A private
965    /// leaf taken only at the top of install; the hot read/write paths never
966    /// touch it.
967    install_lock: Mutex<()>,
968    /// Fault injection flag for testing.
969    #[cfg(test)]
970    fail_writes: AtomicBool,
971    #[cfg(test)]
972    writes_attempted: AtomicBool,
973    /// Fault injection flag for the cold-change-cursor seed scan (Issue #3677). When set, the
974    /// streaming seed [`stream_change_cursors`](Self::stream_change_cursors) returns an error, so a
975    /// test can force the cold-change directory to fail its seed over a populated cold store and
976    /// assert the query path degrades to the full scan instead of dropping cold rows.
977    #[cfg(test)]
978    fail_change_cursor_scan: AtomicBool,
979}

981impl RedbColdStorage {
982    /// Create a new Redb cold storage at the given path.
983    ///
984    /// This initializes the database and creates the necessary internal tables
985    /// (`node_versions`, `edge_versions`, `metadata`) if they do not exist.
986    ///
987    /// # Usage
988    ///
989    /// Use this to explicitly configure the cold storage backend. If you don't need
990    /// custom configuration, use [`with_default_config`](Self::with_default_config).
991    ///
992    /// # Details
993    ///
994    /// If the parent directories do not exist, this method will attempt to create them.
995    ///
996    /// When opening an existing database, the `node_versions_stored` and
997    /// `edge_versions_stored` statistics counters are seeded from the
998    /// persisted table lengths (an O(1) metadata read), so
999    /// [`stats`](Self::stats) reflects the full on-disk history rather than
1000    /// only writes made by the current process. Byte and read counters are
1001    /// not persisted and always start at zero.
1002    ///
1003    /// # Examples
1004    ///
1005    /// ```rust
1006    /// # use aletheiadb::storage::redb_cold_storage::{RedbColdStorage, RedbConfig};
1007    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
1008    /// let temp_dir = tempfile::tempdir()?;
1009    /// let path = temp_dir.path().join("my_cold_data.redb");
1010    ///
1011    /// let config = RedbConfig::new();
1012    /// let storage = RedbColdStorage::new(&path, config)?;
1013    /// # Ok(())
1014    /// # }
1015    /// ```
1016    pub fn new<P: AsRef<Path>>(path: P, config: RedbConfig) -> Result<Self> {
1017        let path = path.as_ref().to_path_buf();
1018
1019        // Ensure parent directory exists
1020        if let Some(parent) = path.parent() {
1021            std::fs::create_dir_all(parent).map_err(map_io_error("Failed to create directory"))?;
1022        }
1023
1024        // Open or create the database
1025        let db =
1026            redb::Database::create(&path).map_err(map_db_error("Failed to open Redb database"))?;
1027
1028        // Initialize tables by creating them if they don't exist
1029        let write_txn = db
1030            .begin_write()
1031            .map_err(map_transaction_error("Failed to begin write transaction"))?;
1032
1033        // Open tables to create them, and read their persisted entry counts so
1034        // the version counters can be seeded below. `Table::len()` is an O(1)
1035        // read of the B-tree root's cached length — no scan.
1036        let (persisted_node_versions, persisted_edge_versions) = {
1037            let node_table = write_txn
1038                .open_table(NODE_VERSIONS_TABLE)
1039                .map_err(map_table_error("Failed to create node_versions table"))?;
1040            let edge_table = write_txn
1041                .open_table(EDGE_VERSIONS_TABLE)
1042                .map_err(map_table_error("Failed to create edge_versions table"))?;
1043            write_txn
1044                .open_table(METADATA_TABLE)
1045                .map_err(map_table_error("Failed to create metadata table"))?;
1046            (
1047                node_table
1048                    .len()
1049                    .map_err(map_storage_error("Failed to read node_versions length"))?,
1050                edge_table
1051                    .len()
1052                    .map_err(map_storage_error("Failed to read edge_versions length"))?,
1053            )
1054        };
1055
1056        write_txn
1057            .commit()
1058            .map_err(map_commit_error("Failed to commit table creation"))?;
1059
1060        // Seed the version counters from the persisted tables so a reopened
1061        // database reports its full cold history instead of misleading zeros
1062        // (Issue #3222). Byte counters (raw/compressed) are not recoverable
1063        // from table metadata, so `compression_ratio()` and the read counters
1064        // remain since-process-start.
1065        let stats = AtomicColdStorageStats::new();
1066        stats
1067            .node_versions_stored
1068            .store(persisted_node_versions, Ordering::Relaxed);
1069        stats
1070            .edge_versions_stored
1071            .store(persisted_edge_versions, Ordering::Relaxed);
1072
1073        Ok(Self {
1074            path,
1075            db,
1076            config,
1077            stats,
1078            keyring: ArcSwapOption::empty(),
1079            install_lock: Mutex::new(()),
1080            #[cfg(test)]
1081            fail_writes: AtomicBool::new(false),
1082            #[cfg(test)]
1083            writes_attempted: AtomicBool::new(false),
1084            #[cfg(test)]
1085            fail_change_cursor_scan: AtomicBool::new(false),
1086        })
1087    }

1191    /// [`install_wal_keyring`](crate::storage::wal::concurrent_system::ConcurrentWalSystem::install_wal_keyring)
1192    /// seam (#3669): the structural primitive a hot-live `encryption enable`
1193    /// driver installs so the cold tier is re-keyed without reopening the store.
1194    ///
1195    /// # Transition
1196    ///
1197    /// Only the `None` -> `Some` transition is supported. Installing when a
1198    /// keyring is already present is **rejected** with a structured error (never
1199    /// a silent replace, so no generation is lost): rotation (`Some` -> `Some'`)
1200    /// is the job of [`install_cold_generation`](Self::install_cold_generation),
1201    /// not this seam.
1202    ///
1203    /// # Concurrency
1204    ///
1205    /// The whole body -- presence check and store -- runs under a dedicated leaf
1206    /// [`install_lock`](Self::install_lock), so two concurrent installers cannot
1207    /// both observe the cell as `None` and both store (the second silently
1208    /// replacing the first): the loser blocks, then observes `Some`, and takes
1209    /// the rejection path. Readers `load()` the cell lock-free, so an install is
1210    /// observed atomically and never as a torn read.
1211    ///
1212    /// # Data at rest
1213    ///
1214    /// Unlike the WAL, no segment seal/reopen is required: cold values are
1215    /// individually self-describing (`ACV1`), so values written **after** the
1216    /// install are wrapped under the new keyring while values written before it
1217    /// stay bare. Wrapping the pre-install plaintext corpus (via
1218    /// [`wrap_plaintext_cold_values`](Self::wrap_plaintext_cold_values)) is the
1219    /// enable **driver's** responsibility, exactly as the WAL seam leaves its
1220    /// downstream migration to its driver -- installing the keyring alone does
1221    /// NOT retroactively encrypt already-stored values.
1222    ///
1223    /// # Errors
1224    ///
1225    /// Returns an error if a keyring is already installed (a double-install).
1226    // The production consumer is the hot-live plaintext -> encrypted enable
1227    // engine (Issue #3708 follow-up), which drives this from `enable_encryption`
1228    // after the WAL's `install_wal_keyring` and a bare -> `ACV1` wrap pass. A
1229    // dedicated `StorageError` variant for the double-install rejection (so the
1230    // enable engine can map ONLY it to `FAILED_PRECONDITION`, as the WAL seam's
1231    // `WalKeyringAlreadyInstalled` allows) is a trivial follow-up for the
1232    // enable lane; reusing `InconsistentState` here keeps the change within the
1233    // cold-storage module.
1234    pub fn install_cold_keyring(&self, keyring: ColdKeyring) -> Result<()> {
1235        // Serialize the ENTIRE install -- presence check + store -- under a
1236        // dedicated leaf mutex. Without it two concurrent installers could both
1237        // observe `None`, both pass the check, and both store, the second
1238        // silently replacing the first keyring.
1239        let _install_guard = self.install_lock.lock().unwrap_or_else(|e| e.into_inner());
1240
1241        // Reject a double-install: presence is a one-way None -> Some transition.
1242        if self.keyring.load().is_some() {
1243            return Err(StorageError::InconsistentState {
1244                reason: "a cold keyring is already installed; runtime install only \
1245                         supports the plaintext -> encrypted (None -> Some) transition \
1246                         (key rotation is install_cold_generation)"
1247                    .to_string(),
1248            }
1249            .into());
1250        }
1251
1252        self.keyring.store(Some(Arc::new(keyring)));
1253        Ok(())
1254    }

1298    /// Encrypt a compressed value for storage, applying the self-describing
1299    /// `ACV1` wrapper when encryption is configured (Issue #3617 PR3).
1300    ///
1301    /// The value is stamped with the cold keyring's CURRENT `key_version` so a
1302    /// half-rotated store distinguishes old- from new-generation values on read.
1303    /// A no-op (returns the input) when encryption is disabled.
1304    fn encrypt_if_needed(&self, data: Vec<u8>) -> Result<Vec<u8>> {
1305        let guard = self.keyring.load();
1306        match guard.as_deref() {
1307            Some(ring) => {
1308                let (cipher, key_version) = ring.current().ok_or_else(|| {
1309                    Into::<crate::core::error::Error>::into(StorageError::Encryption(
1310                        "Cold storage keyring holds no generation".to_string(),
1311                    ))
1312                })?;
1313                let ciphertext =
1314                    cipher
1315                        .encrypt(&data, &[])
1316                        .map_err(|e| -> crate::core::error::Error {
1317                            StorageError::Encryption(format!("Cold storage encryption failed: {e}"))
1318                                .into()
1319                        })?;
1320                Ok(wrap_cold_value(key_version, &ciphertext))
1321            }
1322            None => Ok(data),
1323        }

1326    /// Decrypt a stored value, transparently handling both `ACV1`-wrapped
1327    /// (post-rotation) and legacy bare-ciphertext values (Issue #3617 PR3).
1328    ///
1329    /// Dispatch: a value is treated as wrapped ONLY IF it begins with the `ACV1`
1330    /// magic AND names a `key_version` the keyring holds; otherwise it is a legacy
1331    /// value decrypted under the oldest (pre-rotation) cold generation. See
1332    /// [`keyring`](crate::storage::redb_cold_storage::keyring) for the collision
1333    /// argument. A no-op (returns a copy) when encryption is disabled.
1334    fn decrypt_if_needed(&self, data: &[u8]) -> Result<Vec<u8>> {
1335        let guard = self.keyring.load();
1336        let Some(ring) = guard.as_deref() else {
1337            return Ok(data.to_vec());
1338        };
1339        // Wrapped iff ACV1 magic AND the stamped version names a held generation.
1340        if let Some((key_version, ciphertext)) = parse_cold_wrapper(data)
1341            && ring.has_version(key_version)
1342            && let Some(cipher) = ring.cipher_for_value(Some(key_version))
1343        {
1344            return cipher
1345                .decrypt(ciphertext, &[])
1346                .map_err(|e| -> crate::core::error::Error {
1347                    StorageError::Encryption(format!("Cold storage decryption failed: {e}")).into()
1348                });
1349        }
1350        // Legacy bare ciphertext → oldest (pre-rotation) / sole generation.
1351        let cipher = ring.cipher_for_value(None).ok_or_else(|| {
1352            Into::<crate::core::error::Error>::into(StorageError::Encryption(
1353                "Cold storage keyring holds no generation".to_string(),
1354            ))
1355        })?;
1356        cipher
1357            .decrypt(data, &[])
1358            .map_err(|e| -> crate::core::error::Error {
1359                StorageError::Encryption(format!("Cold storage decryption failed: {e}")).into()
1360            })
1361    }

1401    fn prepare_batch<V, EncodeFn>(
1402        &self,
1403        versions: &[V],
1404        encode_version: EncodeFn,
1405    ) -> Result<PreparedVersionBatch>
1406    where
1407        V: EntityVersion + Sync,
1408        EncodeFn: Fn(&V) -> Vec<u8> + Sync + Send,
1409    {
1410        let cold_config = self.config.to_cold_storage_config();
1411        // Resolve the current cold generation ONCE up front so the parallel
1412        // compression closure is `Sync` (no borrow of `self.keyring`'s lock) and
1413        // every value in the batch is wrapped under the same `key_version`.
1414        let keyring_guard = self.keyring.load();
1415        let cold_generation: Option<(Arc<dyn crate::encryption::cipher::Cipher>, u32)> =
1416            match keyring_guard.as_deref() {
1417                Some(ring) => Some(ring.current().ok_or_else(|| {
1418                    Into::<crate::core::error::Error>::into(StorageError::Encryption(
1419                        "Cold storage keyring holds no generation".to_string(),
1420                    ))
1421                })?),
1422                None => None,
1423            };
1424        let cold_generation_ref = &cold_generation;
1425
1426        // Helper closure: compress then optionally encrypt+`ACV1`-wrap.
1427        let compress_and_encrypt = |data: &[u8]| -> Result<Vec<u8>> {
1428            let compressed = crate::storage::compression::compress(data, &cold_config)?;
1429            match cold_generation_ref {
1430                Some((cipher, key_version)) => {
1431                    let ciphertext = cipher.encrypt(&compressed, &[]).map_err(
1432                        |e| -> crate::core::error::Error {
1433                            StorageError::Encryption(format!("Cold storage encryption failed: {e}"))
1434                                .into()
1435                        },
1436                    )?;
1437                    Ok(wrap_cold_value(*key_version, &ciphertext))
1438                }
1439                None => Ok(compressed),
1440            }
1441        };
1442
1443        if self.should_parallel_compress(versions.len()) {
1444            versions
1445                .par_iter()
1446                .try_fold(PreparedVersionBatch::default, |mut prepared, version| {
1447                    let encoded = encode_version(version);
1448                    let raw_size_bytes = encoded.len() as u64;
1449                    let to_store = compress_and_encrypt(&encoded)?;
1450                    prepared.add_entry(version.version_id(), to_store, raw_size_bytes);
1451                    Ok::<_, crate::core::error::Error>(prepared)
1452                })
1453                .try_reduce(PreparedVersionBatch::default, |mut left, right| {
1454                    left.merge(right);
1455                    Ok::<_, crate::core::error::Error>(left)
1456                })
1457        } else {
1458            let mut prepared = PreparedVersionBatch::with_capacity(versions.len());
1459            for version in versions {
1460                let encoded = encode_version(version);
1461                let raw_size_bytes = encoded.len() as u64;
1462                let to_store = compress_and_encrypt(&encoded)?;
1463                prepared.add_entry(version.version_id(), to_store, raw_size_bytes);
1464            }
1465
1466            Ok(prepared)
1467        }
1468    }

1532    /// Get the Log Sequence Number (LSN) of the last safely flushed transaction.
1533    ///
1534    /// The Write-Ahead Log uses this value to determine which segments can be
1535    /// safely truncated and deleted. If a transaction's LSN is less than or equal
1536    /// to this `flushed_lsn`, it is durably persisted in cold storage.
1537    ///
1538    /// ## Examples
1539    ///
1540    /// ```rust
1541    /// # use aletheiadb::storage::redb_cold_storage::RedbColdStorage;
1542    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
1543    /// let temp_dir = tempfile::tempdir()?;
1544    /// let path = temp_dir.path().join("db.redb");
1545    /// let storage = RedbColdStorage::with_default_config(&path)?;
1546    ///
1547    /// if let Some(lsn) = storage.get_flushed_lsn()? {
1548    ///     println!("Safe to truncate WAL up to LSN: {:?}", lsn);
1549    /// } else {
1550    ///     println!("No data flushed yet.");
1551    /// }
1552    /// # Ok(())
1553    /// # }
1554    /// ```
1555    pub fn get_flushed_lsn(&self) -> Result<Option<LSN>> {
1556        let read_txn = self
1557            .db
1558            .begin_read()
1559            .map_err(map_transaction_error("Failed to begin read transaction"))?;
1560        let table = read_txn
1561            .open_table(METADATA_TABLE)
1562            .map_err(map_table_error("Failed to open metadata table"))?;
1563
1564        match table.get(FLUSHED_LSN_KEY) {
1565            Ok(Some(value)) => {
1566                let bytes: &[u8] = value.value();
1567                if bytes.len() != 8 {
1568                    return Err(
1569                        StorageError::corruption("Invalid flushed_lsn format".to_string()).into(),
1570                    );
1571                }
1572                let lsn_value = u64::from_le_bytes([
1573                    bytes[0], bytes[1], bytes[2], bytes[3], bytes[4], bytes[5], bytes[6], bytes[7],
1574                ]);
1575                Ok(Some(LSN(lsn_value)))
1576            }
1577            Ok(None) => Ok(None),
1578            Err(e) => {
1579                Err(StorageError::io_error(format!("Failed to read flushed_lsn: {}", e)).into())
1580            }
1581        }
1582    }

1584    /// Set the flushed LSN in the metadata table (internal helper).
1585    fn set_flushed_lsn_internal(
1586        table: &mut redb::Table<'_, &'static str, &'static [u8]>,
1587        lsn: LSN,
1588    ) -> Result<()> {
1589        // Read current LSN to ensure we only increase it
1590        let current_lsn = if let Ok(Some(value)) = table.get(FLUSHED_LSN_KEY) {
1591            let bytes = value.value();
1592            if bytes.len() == 8 {
1593                let lsn_value = u64::from_le_bytes([
1594                    bytes[0], bytes[1], bytes[2], bytes[3], bytes[4], bytes[5], bytes[6], bytes[7],
1595                ]);
1596                Some(LSN(lsn_value))
1597            } else {
1598                None
1599            }
1600        } else {
1601            None
1602        };
1603
1604        // Only update if new LSN is higher (prevents race condition)
1605        let final_lsn = match current_lsn {
1606            Some(current) if lsn.0 <= current.0 => {
1607                // LSN is not higher, skip update
1608                return Ok(());
1609            }
1610            _ => lsn,
1611        };
1612
1613        let lsn_bytes = final_lsn.0.to_le_bytes();
1614        table
1615            .insert(FLUSHED_LSN_KEY, lsn_bytes.as_slice())
1616            .map_err(|e| -> crate::core::error::Error {
1617                StorageError::io_error(format!("Failed to write flushed_lsn: {}", e)).into()
1618            })?;
1619        Ok(())
1620    }

1622    /// Read the persisted cold-tier bi-temporal extent bounds (Issue #3389).
1623    ///
1624    /// Returns `Ok(None)` when the cold tier has never stored a version (no
1625    /// extent record yet) or when the persisted record uses an unrecognized
1626    /// layout version — both mean "no cold bounds to contribute", so callers
1627    /// leave the hot-tier extent untouched. Otherwise returns the bounds
1628    /// accumulated across every version ever written to this cold store,
1629    /// following the same `earliest`/`latest` convention as the in-memory
1630    /// extent aggregate (the open-interval sentinel never appears).
1631    ///
1632    /// This is an O(1) single metadata read; it is the startup counterpart to
1633    /// the incremental bounds maintained on every cold write, letting
1634    /// [`temporal_extent`](crate::AletheiaDB::temporal_extent) span history
1635    /// migrated to cold before the current process started.
1636    pub fn get_temporal_extent_bounds(&self) -> Result<Option<ColdTemporalExtentBounds>> {
1637        let read_txn = self
1638            .db
1639            .begin_read()
1640            .map_err(map_transaction_error("Failed to begin read transaction"))?;
1641        let table = match read_txn.open_table(METADATA_TABLE) {
1642            Ok(table) => table,
1643            // A cold store that never persisted metadata has no such table.
1644            Err(redb::TableError::TableDoesNotExist(_)) => return Ok(None),
1645            Err(e) => {
1646                return Err(StorageError::io_error(format!(
1647                    "Failed to open metadata table: {}",
1648                    e
1649                ))
1650                .into());
1651            }
1652        };
1653
1654        match table.get(TEMPORAL_EXTENT_KEY) {
1655            Ok(Some(value)) => Ok(ColdTemporalExtentBounds::from_bytes(value.value())),
1656            Ok(None) => Ok(None),
1657            Err(e) => {
1658                Err(StorageError::io_error(format!("Failed to read temporal_extent: {}", e)).into())
1659            }
1660        }
1661    }

1663    /// Merge a batch of freshly-written versions' bounds into the persisted
1664    /// cold-tier extent, within the caller's write transaction (Issue #3389).
1665    ///
1666    /// A no-op when the delta is empty (nothing observed). The read-modify-write
1667    /// happens on the same `metadata` table the caller already opens, so the
1668    /// extent update is atomic with the version writes: either the versions and
1669    /// their widened bounds both commit, or neither does.
1670    fn merge_extent_into_metadata(
1671        table: &mut redb::Table<'_, &'static str, &'static [u8]>,
1672        delta: &ColdTemporalExtentBounds,
1673    ) -> Result<()> {
1674        if delta.is_empty() {
1675            return Ok(());
1676        }
1677        let mut current = match table.get(TEMPORAL_EXTENT_KEY) {
1678            Ok(Some(value)) => {
1679                // An existing record with an UNRECOGNIZED version tag decodes to
1680                // `None` and is clobbered here: `unwrap_or_default()` starts from
1681                // empty bounds and rebuilds them widen-only from post-write
1682                // deltas. That makes a downgrade across a future
1683                // `EXTENT_RECORD_VERSION` bump lossy (bounds a newer binary
1684                // persisted are dropped, then re-accumulated only from writes the
1685                // older binary performs) — acceptable because the extent is
1686                // advisory (under-reporting is safe) and only ever widens.
1687                ColdTemporalExtentBounds::from_bytes(value.value()).unwrap_or_default()
1688            }
1689            Ok(None) => ColdTemporalExtentBounds::default(),
1690            Err(e) => {
1691                return Err(StorageError::io_error(format!(
1692                    "Failed to read temporal_extent: {}",
1693                    e
1694                ))
1695                .into());
1696            }
1697        };
1698        current.merge(delta);
1699        table
1700            .insert(TEMPORAL_EXTENT_KEY, current.to_bytes().as_slice())
1701            .map_err(|e| -> crate::core::error::Error {
1702                StorageError::io_error(format!("Failed to write temporal_extent: {}", e)).into()
1703            })?;
1704        Ok(())
1705    }

1776    /// Transactionally re-encrypt every stored cold value to the `target_version`
1777    /// generation (Issue #3617 PR3), decrypting each under its own (old/legacy)
1778    /// generation and re-encrypting under the target cold DEK, in bounded
1779    /// per-transaction batches, resumable via a durable redb cursor.
1780    ///
1781    /// # Preconditions
1782    ///
1783    /// The store must be encrypted and its keyring must hold BOTH the target
1784    /// generation (installed by the driver) and every source generation needed to
1785    /// decrypt existing values. New appends interleaved with the pass already
1786    /// stamp the current (target) generation and are simply skipped as
1787    /// already-wrapped.
1788    ///
1789    /// # Crash-consistency
1790    ///
1791    /// Each batch's value rewrites and the cursor advance commit in ONE redb write
1792    /// transaction, so the cursor never runs ahead of (or behind) the durable
1793    /// rewrites — a crash resumes from exactly the last committed key. The `ACV1`
1794    /// wrapper is the idempotency backstop: a value already wrapped at
1795    /// `target_version` is skipped, so a re-run after a stale cursor never
1796    /// double-encrypts. Only `node_versions`/`edge_versions` VALUES are touched;
1797    /// `flushed_lsn` and `temporal_extent` (incl. the #3389 per-dimension bounds)
1798    /// are never read or rewritten, so they are preserved untouched.
1799    ///
1800    /// On completion the durable [`COLD_VALUE_FORMAT_KEY`] marker is set to
1801    /// `wrapped@target_version` and the resume cursor is deleted.
1802    ///
1803    /// # Errors
1804    ///
1805    /// * encryption is not configured, or the keyring does not hold
1806    ///   `target_version`;
1807    /// * a value fails to decrypt (a genuine wrong/absent key — surfaced loudly,
1808    ///   never silent wrong data) or a redb transaction fails.
1809    pub fn reencrypt_cold_values(&self, target_version: u32) -> Result<ColdReencryptStats> {
1810        let guard = self.keyring.load();
1811        let ring = guard.as_deref().ok_or_else(|| {
1812            Into::<crate::core::error::Error>::into(StorageError::InconsistentState {
1813                reason: "cold storage is not encrypted; cannot rotate cold keys".to_string(),
1814            })
1815        })?;
1816        if !ring.has_version(target_version) {
1817            return Err(StorageError::InconsistentState {
1818                reason: "cold keyring does not hold the target generation for rotation".to_string(),
1819            }
1820            .into());
1821        }
1822        // The cipher every value is re-encrypted UNDER (the target generation).
1823        let target_cipher = ring.cipher_for_value(Some(target_version)).ok_or_else(|| {
1824            Into::<crate::core::error::Error>::into(StorageError::InconsistentState {
1825                reason: "cold keyring target generation missing a cipher".to_string(),
1826            })
1827        })?;
1828
1829        self.migrate_cold_values(target_version, &target_cipher, ColdSourceMode::Rekey)
1830    }

1832    /// Wrap every **bare (plaintext, never-encrypted)** cold value into the `ACV1`
1833    /// format under `target_cipher`, stamping `target_version` (Issue #3616 PR3 —
1834    /// the plaintext → encrypted *enable* migration).
1835    ///
1836    /// This is the cold counterpart the enable engine needs but
1837    /// [`reencrypt_cold_values`](Self::reencrypt_cold_values) cannot provide: that
1838    /// pass *decrypts* each source value first (it rekeys an already-encrypted
1839    /// store old-gen → new-gen), and the cold reader treats a bare value as legacy
1840    /// **ciphertext** — so feeding a bare plaintext value to the rekey pass would
1841    /// fail AEAD authentication. This pass instead treats each un-`ACV1` value as
1842    /// the plaintext to encrypt, so a plaintext cold store becomes a uniformly
1843    /// `ACV1`-wrapped one whose values the cold reader dispatches correctly (never
1844    /// mis-reading a bare value as legacy ciphertext once wrapped).
1845    ///
1846    /// The cipher is passed explicitly rather than sourced from the keyring, so the
1847    /// pass runs on a store opened with **no** keyring (the live plaintext handle
1848    /// during `enable`) as well as one opened under the enable cold DEK (a resuming
1849    /// `open()`). Only `node_versions`/`edge_versions` VALUES are rewritten;
1850    /// `flushed_lsn` / `temporal_extent` are never touched.
1851    ///
1852    /// **Idempotent / resumable / crash-safe** exactly like
1853    /// [`reencrypt_cold_values`](Self::reencrypt_cold_values): a value already
1854    /// wrapped at `target_version` is skipped, and each batch's rewrites + cursor
1855    /// advance commit in one redb transaction, so a crash mid-pass resumes with no
1856    /// double-encrypt. On completion the [`COLD_VALUE_FORMAT_KEY`] marker is set to
1857    /// `wrapped@target_version` and the cursor is cleared.
1858    ///
1859    /// # Errors
1860    ///
1861    /// A value fails to encrypt, or a redb transaction fails.
1862    pub fn wrap_plaintext_cold_values(
1863        &self,
1864        target_cipher: &Arc<dyn crate::encryption::cipher::Cipher>,
1865        target_version: u32,
1866    ) -> Result<ColdReencryptStats> {
1867        self.migrate_cold_values(target_version, target_cipher, ColdSourceMode::WrapPlaintext)
1868    }

1870    /// Unwrap every **`ACV1`-wrapped** cold value back to bare plaintext (the
1871    /// exact inverse of [`wrap_plaintext_cold_values`]; Issue #3616 PR4 disable).
1872    ///
1873    /// Each `ACV1` value is decrypted under its stamped generation (via the live
1874    /// keyring, exactly as a normal read does) and rewritten BARE — the same
1875    /// compressed bytes a never-encrypted value carries — so a uniformly-encrypted
1876    /// cold store becomes a plaintext one the cold reader dispatches as bare. This
1877    /// is the cold counterpart the disable engine needs: the enable engine wrapped
1878    /// bare → `ACV1`, so a disable takes each `ACV1` value and produces the
1879    /// compressed plaintext it wraps. No record is ever decoded (the same
1880    /// compressed bytes are unwrapped in place), so temporal bounds cannot change;
1881    /// only `node_versions`/`edge_versions` VALUES are touched, never `flushed_lsn`
1882    /// / `temporal_extent`. On completion the durable [`COLD_VALUE_FORMAT_KEY`]
1883    /// marker is cleared (the store is plaintext, `cold_value_format_version`
1884    /// reports `None`).
1885    ///
1886    /// **Idempotent / resumable / crash-safe** WITHOUT a version cursor: each
1887    /// batch's rewrites commit in one redb transaction, and a re-run after a crash
1888    /// skips the already-bare values a prior partial run produced (a bare value is
1889    /// its own completion marker — the mirror of the wrap pass's already-wrapped
1890    /// skip). A value carrying no `ACV1` wrapper is treated as already-bare
1891    /// plaintext (the post-enable invariant this pass inverts; the enable wrap
1892    /// produces uniformly `ACV1`-wrapped values, so no pre-#3617 legacy
1893    /// bare-ciphertext value is in scope).
1894    ///
1895    /// # Errors
1896    ///
1897    /// * the store is not encrypted (no keyring to decrypt with);
1898    /// * a value fails to decrypt (a genuine wrong/absent key — surfaced loudly,
1899    ///   never silently left encrypted) or a redb transaction fails.
1900    pub fn unwrap_encrypted_cold_values(&self) -> Result<ColdUnwrapStats> {
1901        if self.keyring.load().is_none() {
1902            return Err(StorageError::InconsistentState {
1903                reason: "cold storage is not encrypted; nothing to unwrap".to_string(),
1904            }
1905            .into());
1906        }
1907
1908        let mut stats = ColdUnwrapStats::default();
1909        for table in [ColdTableKind::Node, ColdTableKind::Edge] {
1910            self.unwrap_one_table(table, &mut stats)?;
1911        }
1912
1913        // Terminal marker: the whole store is now plaintext; drop the format
1914        // marker (so `cold_value_format_version` reports `None`) and any stale
1915        // rotation cursor, in one transaction.
1916        let write_txn = self
1917            .db
1918            .begin_write()
1919            .map_err(map_transaction_error("Failed to begin write transaction"))?;
1920        {
1921            let mut meta = write_txn
1922                .open_table(METADATA_TABLE)
1923                .map_err(map_table_error("Failed to open metadata table"))?;
1924            meta.remove(COLD_VALUE_FORMAT_KEY)
1925                .map_err(map_storage_error("Failed to clear cold_value_format"))?;
1926            meta.remove(COLD_ROTATION_CURSOR_KEY)
1927                .map_err(map_storage_error("Failed to clear cold_rotation cursor"))?;

1936    /// Unwrap one value-bearing table's `ACV1` values back to bare, in bounded
1937    /// per-transaction batches (Issue #3616 PR4 disable). Mirrors
1938    /// [`reencrypt_one_table`](Self::reencrypt_one_table) but writes the decrypted
1939    /// compressed bytes BARE (no wrapper) instead of re-wrapping, and relies on
1940    /// the natural "bare == done" idempotency rather than a version cursor.
1941    fn unwrap_one_table(
1942        &self,
1943        table_kind: ColdTableKind,
1944        stats: &mut ColdUnwrapStats,
1945    ) -> Result<()> {
1946        use std::ops::Bound;
1947        let table_def = table_kind.table();
1948        let batch_size = self.config.reencrypt_batch_size.max(1);
1949        let mut resume_after: Option<u64> = None;
1950        loop {
1951            let write_txn = self
1952                .db
1953                .begin_write()
1954                .map_err(map_transaction_error("Failed to begin write transaction"))?;
1955
1956            let batch_len;
1957            let mut last_key = resume_after;
1958            {
1959                let mut table = write_txn
1960                    .open_table(table_def)
1961                    .map_err(map_table_error("Failed to open versions table"))?;
1962
1963                // Collect a bounded slice into owned buffers FIRST — the redb
1964                // iterator borrows the table immutably and cannot be held across
1965                // the `insert` calls below (which need `&mut table`).
1966                let collected: Vec<(u64, Vec<u8>)> = {
1967                    let lower = match resume_after {
1968                        Some(k) => Bound::Excluded(k),
1969                        None => Bound::Unbounded,
1970                    };
1971                    let range = table
1972                        .range::<u64>((lower, Bound::Unbounded))
1973                        .map_err(map_storage_error("Failed to range cold table"))?;
1974                    let mut v = Vec::with_capacity(batch_size);
1975                    for entry in range.take(batch_size) {
1976                        let (k, val) =
1977                            entry.map_err(map_storage_error("Failed to read cold entry"))?;
1978                        v.push((k.value(), val.value().to_vec()));
1979                    }
1980                    v
1981                };
1982
1983                batch_len = collected.len();
1984                if batch_len == 0 {
1985                    drop(table);
1986                    drop(write_txn);
1987                    break;
1988                }
1989
1990                for (key, value) in &collected {
1991                    last_key = Some(*key);
1992                    // Idempotency backstop: a value with no `ACV1` wrapper is
1993                    // already bare plaintext → skip (a prior partial run, or the
1994                    // post-enable invariant this pass inverts).
1995                    //
1996                    // FAIL-CLOSED INVARIANT (Issue #3616 PR4 review): treating a
1997                    // no-wrapper value as bare plaintext is only correct because
1998                    // this pass runs EXCLUSIVELY over an ENABLE-engine-created
1999                    // store, where every encrypted cold value carries an `ACV1`
2000                    // wrapper. That invariant is enforced upstream by construction,
2001                    // not merely assumed here:
2002                    //   * the normal encrypted write path (`encrypt_if_needed`)
2003                    //     ALWAYS `ACV1`-wraps (`wrap_cold_value`) when a keyring is
2004                    //     present, and
2005                    //   * `enable`'s `wrap_plaintext_cold_values` converts every
2006                    //     pre-existing bare value bare → `ACV1` and unconditionally
2007                    //     stamps the terminal `COLD_VALUE_FORMAT_KEY` marker (even
2008                    //     for an empty store).
2009                    // So a disable-able store's encrypted cold values are uniformly
2010                    // `ACV1`-wrapped, and the ONLY no-wrapper values reaching here
2011                    // are genuinely bare (a completed prior batch/run). LEGACY
2012                    // pre-#3617 bare-ciphertext (encrypted, no `ACV1` wrapper), which
2013                    // `decrypt_if_needed` still supports on the read path, is OUT OF
2014                    // SCOPE: it predates both the `ACV1` scheme and the durable
2015                    // encryption authority the disable engine requires, so it never
2016                    // reaches this pass.
2017                    //
2018                    // A marker-based fail-closed guard was evaluated and deliberately
2019                    // NOT added: the `COLD_VALUE_FORMAT_KEY` marker cannot represent
2020                    // the legacy-encrypted state distinctly — it is ABSENT for a
2021                    // legacy store, a never-encrypted plaintext store, AND a store
2022                    // whose disable already cleared it and is being re-driven by the
2023                    // post-clear resume window (a crash between
2024                    // `unwrap_encrypted_cold_values` clearing the marker and
2025                    // `mark_cold_complete` legitimately re-runs this pass over an
2026                    // all-bare, marker-absent, keyring-present store). Refusing on
2027                    // "marker absent" would therefore break that legitimate
2028                    // idempotent resume rather than catch a real defect, and refusing
2029                    // for enable-engine stores is impossible (their marker is always
2030                    // set). The safe guard is the upstream construction invariant
2031                    // above, so this skip stays.
2032                    if parse_cold_wrapper(value).is_none() {
2033                        stats.values_skipped += 1;
2034                        continue;
2035                    }
2036                    // Decrypt the wrapped value under its stamped generation,
2037                    // yielding the compressed plaintext bytes a bare value holds.
2038                    let bare = self.decrypt_if_needed(value)?;
2039                    table
2040                        .insert(*key, bare.as_slice())
2041                        .map_err(map_storage_error("Failed to rewrite cold value"))?;
2042                    stats.values_unwrapped += 1;
2043                }
2044            }
2045
2046            write_txn
2047                .commit()
2048                .map_err(map_commit_error("Failed to commit cold unwrap batch"))?;
2049            stats.batches_committed += 1;
2050
2051            resume_after = last_key;
2052            if batch_len < batch_size {
2053                break;
2054            }
2055        }
2056        Ok(())
2057    }

2059    /// Shared driver for the cold value-migration passes (Issue #3617 PR3 rekey /
2060    /// Issue #3616 PR3 enable-wrap): resume from the durable cursor, process the
2061    /// Node then Edge value tables in bounded batches, and write the terminal
2062    /// format marker. The `source_mode` selects how each source value's plaintext
2063    /// is obtained (decrypt-under-own-generation vs treat-bare-as-plaintext).
2064    fn migrate_cold_values(
2065        &self,
2066        target_version: u32,
2067        target_cipher: &Arc<dyn crate::encryption::cipher::Cipher>,
2068        source_mode: ColdSourceMode,
2069    ) -> Result<ColdReencryptStats> {
2070        let mut stats = ColdReencryptStats::default();
2071
2072        // Resume point: a cursor for THIS target resumes it; any other cursor
2073        // (stale, from an aborted earlier rotation) is ignored and the pass
2074        // restarts from the node table (idempotent — already-wrapped values are
2075        // skipped). Tables are always processed Node then Edge.
2076        let (start_table, start_after) = match self.read_cold_rotation_cursor()? {
2077            Some(c) if c.target_version == target_version => (c.table, Some(c.last_completed_key)),
2078            _ => (ColdTableKind::Node, None),
2079        };
2080
2081        let tables = [ColdTableKind::Node, ColdTableKind::Edge];
2082        let start_idx = tables.iter().position(|t| *t == start_table).unwrap_or(0);
2083        for (i, &table) in tables.iter().enumerate() {
2084            if i < start_idx {
2085                // A table before the cursor's table is already fully processed.
2086                continue;
2087            }
2088            let resume_after = if i == start_idx { start_after } else { None };
2089            self.reencrypt_one_table(
2090                table,
2091                target_version,
2092                target_cipher,
2093                source_mode,
2094                resume_after,
2095                &mut stats,
2096            )?;
2097        }
2098
2099        // Terminal marker: whole store is wrapped at the target; drop the cursor.
2100        let write_txn = self
2101            .db
2102            .begin_write()
2103            .map_err(map_transaction_error("Failed to begin write transaction"))?;
2104        {
2105            let mut meta = write_txn
2106                .open_table(METADATA_TABLE)
2107                .map_err(map_table_error("Failed to open metadata table"))?;
2108            let mut format_bytes = [0u8; 5];
2109            format_bytes[0] = COLD_ROTATION_RECORD_VERSION;
2110            format_bytes[1..5].copy_from_slice(&target_version.to_le_bytes());
2111            meta.insert(COLD_VALUE_FORMAT_KEY, format_bytes.as_slice())
2112                .map_err(map_storage_error("Failed to write cold_value_format"))?;
2113            meta.remove(COLD_ROTATION_CURSOR_KEY)
2114                .map_err(map_storage_error("Failed to clear cold_rotation cursor"))?;
2115        }
2116        write_txn.commit().map_err(map_commit_error(
2117            "Failed to commit cold rotation completion",
2118        ))?;
2119
2120        Ok(stats)
2121    }

2500    /// Filter-during-decode changefeed scan of the cold tier (Issue #3216, PR 2).
2501    ///
2502    /// Runs the shared [`consider_version`] predicate (transaction-time window, optional
2503    /// valid-time window, optional label filter, and the strict `> resume_after` cursor) inline
2504    /// as each node then edge version is decoded, retaining only the `bound`-smallest survivors
2505    /// by [`ChangeCursor`] order. It returns `Vec<RawChange>` directly — the caller no longer
2506    /// materializes a full `Vec<NodeVersion>` / `Vec<EdgeVersion>` and re-filters it.
2507    ///
2508    /// # Correctness note — no `version_id` early-stop
2509    ///
2510    /// Cold storage is keyed by `VersionId`, but a version's id is allocated at transaction-build
2511    /// time while its commit timestamp is assigned later, so `version_id` is **not** monotonic
2512    /// with transaction time — two concurrent commits can invert the two orders. The changefeed
2513    /// is ordered by transaction time (`ChangeCursor`), so an early-stop / `range()` on the
2514    /// ascending `version_id` key would silently drop or misorder rows. This scan therefore
2515    /// **decodes every entry** (I/O stays O(N)) and only the in-memory retention is bounded. A
2516    /// true sub-linear cold pushdown needs a transaction-time-ordered directory (tracked
2517    /// follow-up), not a `version_id` range.
2518    #[allow(clippy::too_many_arguments)]
2519    pub(crate) fn collect_changes_filtered(
2520        &self,
2521        tx_window: &TimeRange,
2522        valid_window: Option<&TimeRange>,
2523        label_filter: Option<&str>,
2524        resume_after: Option<ChangeCursor>,
2525        bound: usize,
2526    ) -> Result<Vec<RawChange>> {
2527        let mut acc = BoundedChanges::new(bound);
2528
2529        // Namespace derivation for the cold tier (Issue #3349, PR3c). The immutable
2530        // ride-along namespace key is carried only on **anchor** versions (a delta
2531        // never diffs the immutable key). Cold storage cannot cheaply reconstruct a
2532        // delta's property chain during a scan, but it does not need to *when the
2533        // covering anchor is also cold*: the table is keyed by `VersionId` and
2534        // iterated ascending, and an entity's create version (its lowest `VersionId`)
2535        // is always an anchor carrying the key — so an in-cold anchor is decoded
2536        // *before* any of its deltas. We record **every** anchor's namespace
2537        // (default included) in a running map as it is decoded and read a delta's
2538        // namespace back from it (fast path).
2539        //
2540        // # LRU anchor-split (the fail-closed miss)
2541        //
2542        // The "anchor always precedes delta" invariant only holds *within* the cold
2543        // scan. Under `MigrationPolicy::aggressive()` / `enable_lru`, a
2544        // frequently-accessed anchor can stay HOT while an older delta migrates COLD
2545        // — so a cold delta's covering anchor is absent from this scan. A map MISS is
2546        // therefore **not** a `default` entity; it is an unresolvable-here delta. We
2547        // stamp such a delta with the reserved
2548        // [`crate::core::namespace::UNRESOLVED_NAMESPACE`] sentinel — **never**
2549        // `default` — and the [`HistoricalStorage`] layer, which sees both tiers,
2550        // re-derives its real namespace via tier-aware reconstruction (see
2551        // `resolve_unresolved_namespaces`). Failing open to `default` here would leak
2552        // the change to `default`-scoped subscribers and hide it from its own.
2553        let unresolved_ns_id = unresolved_namespace_id();
2554        let mut node_ns: std::collections::HashMap<u64, NamespaceId> =
2555            std::collections::HashMap::new();
2556        let mut edge_ns: std::collections::HashMap<u64, NamespaceId> =
2557            std::collections::HashMap::new();
2558
2559        self.scan_versions_into(
2560            NODE_VERSIONS_TABLE,
2561            decode_node_version,
2562            |v: NodeVersion| {
2563                // Count this as a query-time cold decode (the full-scan changefeed path); the
2564                // one-time directory seed decodes off this counter, see `scan_versions_into`.
2565                #[cfg(test)]
2566                COLD_VERSIONS_DECODED.with(|c| c.set(c.get() + 1));
2567                let ns_id = Self::version_namespace_id(
2568                    &v.data,
2569                    v.node_id.as_u64(),
2570                    &mut node_ns,
2571                    unresolved_ns_id,
2572                );
2573                consider_version(
2574                    &mut acc,
2575                    resume_after,
2576                    v.id.as_u64(),
2577                    v.node_id.as_u64(),
2578                    EntityKind::Node,
2579                    &v.temporal,
2580                    v.label,
2581                    v.prev_version.is_none(),
2582                    tx_window,
2583                    valid_window,
2584                    label_filter,
2585                    move || ns_id,
2586                );
2587            },
2588        )?;
2589
2590        self.scan_versions_into(
2591            EDGE_VERSIONS_TABLE,
2592            decode_edge_version,
2593            |v: EdgeVersion| {
2594                #[cfg(test)]
2595                COLD_VERSIONS_DECODED.with(|c| c.set(c.get() + 1));
2596                let ns_id = Self::version_namespace_id(
2597                    &v.data,
2598                    v.edge_id.as_u64(),
2599                    &mut edge_ns,
2600                    unresolved_ns_id,
2601                );
2602                consider_version(
2603                    &mut acc,
2604                    resume_after,
2605                    v.id.as_u64(),
2606                    v.edge_id.as_u64(),
2607                    EntityKind::Edge,
2608                    &v.temporal,
2609                    v.label,
2610                    v.prev_version.is_none(),
2611                    tx_window,
2612                    valid_window,
2613                    label_filter,
2614                    move || ns_id,
2615                );
2616            },
2617        )?;
2618
2619        Ok(acc.into_vec())
2620    }

2622    /// Materialize the changefeed rows for an explicit, ascending list of candidate cursors
2623    /// (Issue #3677 — the cold-tier directory pushdown).
2624    ///
2625    /// Each candidate names one cold row (`kind_ord` + `version_id`). This point-reads that row
2626    /// (decoding it — I/O is `O(candidates)`, not `O(N_cold)`) and feeds the decoded version
2627    /// through the **same** [`consider_version`] the full scan uses, so the retained page is
2628    /// byte-identical to [`collect_changes_filtered`](Self::collect_changes_filtered) for the same
2629    /// arguments. Because candidates arrive in ascending [`ChangeCursor`] order and
2630    /// `consider_version` keeps only the `bound`-smallest survivors, once the accumulator holds
2631    /// `bound` rows no later candidate can displace one, so the walk early-stops.
2632    ///
2633    /// A candidate whose version_id is absent (raced eviction between directory read and point
2634    /// read) is skipped, never an error.

2642        bound: usize,

2645
2646        // Namespace resolution mirrors the full scan (`collect_changes_filtered`): an anchor's
2647        // namespace is read from its own properties and recorded per entity; a delta reads its
2648        // entity's namespace back from the map when its covering anchor was also point-read on this
2649        // walk (in ascending-cursor order the anchor's earlier tx-time usually precedes the delta),
2650        // else it is stamped with the fail-closed `UNRESOLVED_NAMESPACE` sentinel — never `default`.
2651        // Any residual sentinel is re-derived tier-aware by the `HistoricalStorage` layer's

2672                            &v.data,
2673                            v.node_id.as_u64(),
2674                            &mut node_ns,
2675                            unresolved_ns_id,
2676                        );
2677                        consider_version(
2678                            &mut acc,
2679                            resume_after,
2680                            v.id.as_u64(),
2681                            v.node_id.as_u64(),
2682                            EntityKind::Node,
2683                            &v.temporal,
2684                            v.label,
2685                            v.prev_version.is_none(),
2686                            tx_window,
2687                            valid_window,
2688                            label_filter,
2689                            move || ns_id,
2690                        );
2691                    }
2692                }
2693                EntityKind::Edge => {
2694                    if let Some(v) = self.get_edge_version(version_id)? {
2695                        let ns_id = Self::version_namespace_id(
2696                            &v.data,
2697                            v.edge_id.as_u64(),
2698                            &mut edge_ns,
2699                            unresolved_ns_id,
2700                        );
2701                        consider_version(
2702                            &mut acc,
2703                            resume_after,
2704                            v.id.as_u64(),
2705                            v.edge_id.as_u64(),
2706                            EntityKind::Edge,

2708                            v.label,

2711                            valid_window,
2712                            label_filter,
2713                            move || ns_id,

2803            VersionData::Delta { .. } => seen.get(&entity_id).copied().unwrap_or(unresolved_ns_id),

3385    /// storage.store_batch_with_lsn(&nodes, &edges, lsn)?;
3386    ///
3387    /// // Now safe to truncate WAL < 500
3388    /// # Ok(())
3389    /// # }
3390    /// ```
3391    pub fn store_batch_with_lsn(
3392        &self,
3393        nodes: &[NodeVersion],
3394        edges: &[EdgeVersion],
3395        lsn: LSN,
3396    ) -> Result<()> {
3397        self.check_fail_writes()?;
3398
3399        let prepared_nodes = self.prepare_node_versions_batch(nodes)?;
3400        let prepared_edges = self.prepare_edge_versions_batch(edges)?;
3401        let node_count = prepared_nodes.entries.len() as u64;
3402        let edge_count = prepared_edges.entries.len() as u64;
3403
3404        let write_txn = self
3405            .db
3406            .begin_write()
3407            .map_err(map_transaction_error("Failed to begin write transaction"))?;
3408
3409        // Store node versions
3410        Self::write_prepared_batch_to_table(&write_txn, NODE_VERSIONS_TABLE, &prepared_nodes)?;
3411
3412        // Store edge versions
3413        Self::write_prepared_batch_to_table(&write_txn, EDGE_VERSIONS_TABLE, &prepared_edges)?;
3414
3415        // Update flushed_lsn and widen the persisted cold-tier extent
3416        // atomically with the batch (Issue #3389).
3417        {
3418            let mut table = write_txn
3419                .open_table(METADATA_TABLE)
3420                .map_err(map_table_error("Failed to open metadata table"))?;
3421            Self::set_flushed_lsn_internal(&mut table, lsn)?;
3422
3423            let mut extent_delta = ColdTemporalExtentBounds::from_versions(nodes);
3424            extent_delta.merge(&ColdTemporalExtentBounds::from_versions(edges));
3425            Self::merge_extent_into_metadata(&mut table, &extent_delta)?;
3426        }
3427
3428        // Commit atomically
3429        write_txn
3430            .commit()
3431            .map_err(map_commit_error("Failed to commit batch"))?;
3432
3433        self.stats
3434            .node_versions_stored
3435            .fetch_add(node_count, Ordering::Relaxed);
3436        self.stats
3437            .edge_versions_stored
3438            .fetch_add(edge_count, Ordering::Relaxed);
3439        self.stats.bytes_written_raw.fetch_add(
3440            prepared_nodes.raw_size_bytes + prepared_edges.raw_size_bytes,
3441            Ordering::Relaxed,
3442        );
3443        self.stats.bytes_written_compressed.fetch_add(
3444            prepared_nodes.compressed_size_bytes + prepared_edges.compressed_size_bytes,
3445            Ordering::Relaxed,
3446        );
3447
3448        Ok(())
3449    }

3451    /// Compact the database file to reclaim free space.
3452    ///
3453    /// As records are overwritten or deleted, Redb accumulates fragmented free space.
3454    /// This method copies all active data into a fresh, tightly-packed file, swapping
3455    /// it atomically with the old one.
3456    ///
3457    /// # Performance
3458    ///
3459    /// This operation is I/O heavy and blocks. It requires an exclusive `&mut self`
3460    /// reference, ensuring no other threads can access the storage during compaction.
3461    ///
3462    /// ## Examples
3463    ///
3464    /// ```rust
3465    /// # use aletheiadb::storage::redb_cold_storage::RedbColdStorage;
3466    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
3467    /// # let temp_dir = tempfile::tempdir()?;
3468    /// # let mut storage = RedbColdStorage::with_default_config(temp_dir.path().join("db.redb"))?;
3469    /// // ... after many deletes ...
3470    /// storage.compact()?; // Space is now reclaimed
3471    /// # Ok(())
3472    /// # }
3473    /// ```
3474    pub fn compact(&mut self) -> Result<()> {
3475        self.db
3476            .compact()
3477            .map_err(map_compaction_error("Failed to compact database"))?;
3478        Ok(())
3479    }

3752/// Decode a `NodeVersion` from a previously serialized byte payload.
3753///
3754/// Reconstructs the internal references (like `InternedString`) alongside
3755/// the core node temporal data.
3756///
3757/// #[doc(hidden)]
3758pub fn decode_node_version(data: &[u8]) -> Result<NodeVersion> {
3759    use crate::core::hlc::HybridTimestamp;
3760    use crate::core::id::NodeId;
3761    use crate::core::interning::GLOBAL_INTERNER;
3762    use crate::core::temporal::{BiTemporalInterval, TimeRange};
3763
3764    // Tagged records carry provenance: V3 (Issue #3350) bundles include the
3765    // authenticated principal, V2 (Issue #3224) bundles do not. Untagged
3766    // records are pre-provenance and are decoded via the frozen `..V1`
3767    // shape instead. There is no version marker on legacy records at all,
3768    // so the magic sequences themselves are the only signal -- see
3769    // `COLD_RECORD_MAGIC_V2` / `COLD_RECORD_MAGIC_V3`.
3770    //
3771    // A magic-byte match means this is unambiguously a record of that tagged
3772    // generation (legacy records never carry either prefix), so a decode
3773    // failure past the magic indicates real corruption -- it must be
3774    // surfaced directly rather than falling through to an older decoder,
3775    // which would otherwise misinterpret the remaining bytes
3776    // (magic-matched-but-corrupt is not "maybe legacy").
3777    let (
3778        label,
3779        temporal_valid_start,
3780        temporal_valid_end,
3781        temporal_tx_start,
3782        temporal_tx_end,
3783        id,
3784        node_id,
3785        data_field,
3786        next_version,
3787        prev_version,
3788        provenance,
3789    ) = if data.starts_with(&COLD_RECORD_MAGIC_V3) {
3790        let s: SerializableNodeVersion = bitcode::decode(&data[COLD_RECORD_MAGIC_V3.len()..])
3791            .map_err(|e| {
3792                StorageError::corruption(format!("Failed to decode node version V3: {}", e))
3793            })?;
3794        (
3795            s.label,
3796            s.temporal_valid_start,
3797            s.temporal_valid_end,
3798            s.temporal_tx_start,
3799            s.temporal_tx_end,
3800            s.id,
3801            s.node_id,
3802            s.data,
3803            s.next_version,
3804            s.prev_version,
3805            decode_provenance(s.provenance)?,
3806        )
3807    } else if data.starts_with(&COLD_RECORD_MAGIC_V2) {
3808        let s: SerializableNodeVersionV2 = bitcode::decode(&data[COLD_RECORD_MAGIC_V2.len()..])
3809            .map_err(|e| {
3810                StorageError::corruption(format!("Failed to decode node version V2: {}", e))
3811            })?;
3812        (
3813            s.label,
3814            s.temporal_valid_start,
3815            s.temporal_valid_end,
3816            s.temporal_tx_start,
3817            s.temporal_tx_end,
3818            s.id,
3819            s.node_id,
3820            s.data,
3821            s.next_version,
3822            s.prev_version,
3823            decode_provenance_v2(s.provenance)?,
3824        )
3825    } else {
3826        let s: SerializableNodeVersionV1 = bitcode::decode(data).map_err(|e| {
3827            StorageError::corruption(format!("Failed to decode node version V1: {}", e))
3828        })?;
3829        (
3830            s.label,
3831            s.temporal_valid_start,
3832            s.temporal_valid_end,
3833            s.temporal_tx_start,
3834            s.temporal_tx_end,
3835            s.id,
3836            s.node_id,
3837            s.data,
3838            s.next_version,
3839            s.prev_version,
3840            None,
3841        )
3842    };
3843
3844    let valid_time = TimeRange::new(
3845        HybridTimestamp::new_unchecked(temporal_valid_start, 0),
3846        HybridTimestamp::new_unchecked(temporal_valid_end, 0),
3847    )
3848    .map_err(|e| StorageError::corruption(format!("Invalid valid time range: {}", e)))?;
3849    let tx_time = TimeRange::new(
3850        HybridTimestamp::new_unchecked(temporal_tx_start, 0),
3851        HybridTimestamp::new_unchecked(temporal_tx_end, 0),
3852    )
3853    .map_err(|e| StorageError::corruption(format!("Invalid transaction time range: {}", e)))?;
3854
3855    Ok(NodeVersion {
3856        id: VersionId::new(id)
3857            .map_err(|e| StorageError::corruption(format!("Invalid version ID: {}", e)))?,
3858        node_id: NodeId::new(node_id)
3859            .map_err(|e| StorageError::corruption(format!("Invalid node ID: {}", e)))?,
3860        commit_timestamp: tx_time.start(),
3861        temporal: BiTemporalInterval::new(valid_time, tx_time),
3862        label: GLOBAL_INTERNER
3863            .intern(&label)
3864            .map_err(|e| StorageError::corruption(format!("Failed to intern label: {}", e)))?,
3865        data: decode_version_data(data_field)?,
3866        next_version: next_version
3867            .map(VersionId::new)
3868            .transpose()
3869            .map_err(|e| StorageError::corruption(format!("Invalid next version ID: {}", e)))?,
3870        prev_version: prev_version
3871            .map(VersionId::new)
3872            .transpose()
3873            .map_err(|e| StorageError::corruption(format!("Invalid prev version ID: {}", e)))?,
3874        provenance,
3875    })
3876}

4091    use crate::core::interning::GLOBAL_INTERNER;
4092    use crate::core::property::PropertyMapBuilder;

4094

4096        SerializableVersionData::Anchor {
4097            properties,
4098            vector_snapshot_id,
4099        } => {
4100            let mut builder = PropertyMapBuilder::new();
4101            for (key, value) in properties {
4102                builder = builder.insert(&key, decode_property_value(value)?);

4105                properties: builder.build(),

4107            })
4108        }
4109        SerializableVersionData::Delta { changed, removed } => {
4110            let mut delta = PropertyDelta::new();
4111            for (key, value) in changed {
4112                let interned_key = GLOBAL_INTERNER.intern(&key).map_err(|e| {
4113                    StorageError::corruption(format!("Failed to intern key: {}", e))
4114                })?;
4115                delta
4116                    .changed
4117                    .insert(interned_key, decode_property_value(value)?);
4118            }
4119            for key in removed {
4120                let interned_key = GLOBAL_INTERNER.intern(&key).map_err(|e| {
4121                    StorageError::corruption(format!("Failed to intern key: {}", e))
4122                })?;
4123                delta.removed.insert(interned_key);

4146
4147    match value {
