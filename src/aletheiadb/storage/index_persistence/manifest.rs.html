1//! Index manifest persistence.
2//!
3//! The [`IndexManifest`] acts as the "root of trust" for a database checkpoint. It catalogs
4//! all other persisted components (vector indexes, string interner, etc.) and tracks metadata
5//! like the Last Sequence Number (LSN) and creation timestamps.
6//!
7//! # Format
8//!
9//! The manifest is serialized using Bitcode and protected by a CRC32 checksum.
10//!
11//! ```text
12//! [ Bitcode Encoded Manifest ] [ CRC32 (4 bytes) ]
13//! ```
14//!
15//! # Atomic Persistence
16//!
17//! The manifest is saved using a "write-temp-then-rename" strategy. This ensures that the
18//! `manifest.idx` file is updated atomically. If a crash occurs during writing, the original
19//! manifest remains untouched, preventing database corruption.
20

30impl IndexManifest {
31    /// Create a new empty manifest.
32    ///
33    /// # Examples
34    ///
35    /// ```
36    /// use aletheiadb::storage::index_persistence::formats::IndexManifest;
37    ///
38    /// let lsn = 12345;
39    /// let manifest = IndexManifest::new(lsn);
40    ///
41    /// assert_eq!(manifest.lsn, lsn);
42    /// assert!(manifest.vector_indexes.is_empty());
43    /// ```
44    pub fn new(lsn: u64) -> Self {
45        let now = SystemTime::now()
46            .duration_since(UNIX_EPOCH)
47            .unwrap_or_else(|_| Duration::from_secs(0))
48            .as_secs() as i64;
49
50        Self {
51            magic: MANIFEST_MAGIC,
52            version: MANIFEST_VERSION,
53            created_at: now,
54            last_modified: now,
55            lsn,
56            vector_indexes: Vec::new(),
57            graph_index: None,
58            temporal_index: None,
59            temporal_adjacency_index: None,
60            string_interner: None,
61        }
62    }
63
64    /// Update the last_modified timestamp.
65    pub fn touch(&mut self) {
66        self.last_modified = SystemTime::now()
67            .duration_since(UNIX_EPOCH)
68            .unwrap_or_else(|_| Duration::from_secs(0))
69            .as_secs() as i64;
70    }
71
72    /// Update the LSN.
73    pub fn set_lsn(&mut self, lsn: u64) {
74        self.lsn = lsn;
75        self.touch();
76    }
77}

79/// Save manifest to disk with CRC32 checksum using atomic write.
80///
81/// Format: `[bitcode_data][crc32_checksum_4_bytes]`
82///
83/// Uses write-temp-then-rename to prevent corruption on crash.
84///
85/// # Examples
86///
87/// ```
88/// use aletheiadb::storage::index_persistence::manifest::save_manifest;
89/// use aletheiadb::storage::index_persistence::formats::IndexManifest;
90/// use tempfile::tempdir;
91///
92/// let dir = tempdir().unwrap();
93/// let path = dir.path().join("manifest.idx");
94/// let manifest = IndexManifest::new(100);
95///
96/// save_manifest(&manifest, &path).unwrap();
97/// assert!(path.exists());
98/// ```
99///
100/// # Errors
101///
102/// Returns an error if:
103/// - The file cannot be written (e.g., permission denied, disk full).
104/// - The atomic rename operation fails.
105pub fn save_manifest(manifest: &IndexManifest, path: &Path) -> Result<()> {
106    save_manifest_with_cipher(manifest, path, None)
107}
108
109/// Save the manifest, optionally encrypting it at rest (Issue #481).
110///
111/// With `cipher == None` this is byte-for-byte identical to [`save_manifest`].
112pub fn save_manifest_with_cipher(
113    manifest: &IndexManifest,
114    path: &Path,
115    cipher: Option<&Arc<dyn Cipher>>,
116) -> Result<()> {
117    super::common::save_encoded_maybe_encrypted(manifest, path, cipher)
118}
119
120/// Save the manifest, encrypting with the current generation of an
121/// [`IndexKeyring`](super::common::IndexKeyring) (Issue #488 key rotation).
122pub(crate) fn save_manifest_with_keyring(
123    manifest: &IndexManifest,
124    path: &Path,
125    keyring: Option<&super::common::IndexKeyring>,
126) -> Result<()> {
127    super::common::save_encoded_maybe_encrypted_with_keyring(manifest, path, keyring)
128}

130/// Load manifest from disk and validate CRC32 checksum.
131///
132/// # Examples
133///
134/// ```
135/// use aletheiadb::storage::index_persistence::manifest::{save_manifest, load_manifest};
136/// use aletheiadb::storage::index_persistence::formats::IndexManifest;
137/// use tempfile::tempdir;
138///
139/// let dir = tempdir().unwrap();
140/// let path = dir.path().join("manifest.idx");
141/// let original = IndexManifest::new(100);
142///
143/// save_manifest(&original, &path).unwrap();
144///
145/// let loaded = load_manifest(&path).unwrap();
146/// assert_eq!(loaded.lsn, 100);
147/// ```
148///
149/// # Errors
150///
151/// Returns an error if:
152/// - The file does not exist or cannot be read.
153/// - The file size exceeds `MAX_MANIFEST_FILE_SIZE`.
154/// - The CRC32 checksum verification fails (indicating corruption).
155/// - The file has invalid magic bytes or an unsupported version.
156pub fn load_manifest(path: &Path) -> Result<IndexManifest> {
157    load_manifest_with_cipher(path, None)
158}
159
160/// Load the manifest, transparently decrypting it if it was written encrypted
161/// (Issue #481). A legacy plaintext manifest is loaded even when a cipher is
162/// supplied (header sniffing). With `cipher == None` and a plaintext file this
163/// is identical to [`load_manifest`].
164///
165/// # Errors
166///
167/// Returns an error if:
168/// - The file does not exist or cannot be read.
169/// - The file size exceeds `MAX_MANIFEST_FILE_SIZE`.
170/// - The file is encrypted but no cipher is supplied, or decryption fails.
171/// - The CRC32 checksum verification fails (indicating corruption).
172/// - The file has invalid magic bytes or an unsupported version.
173pub fn load_manifest_with_cipher(
174    path: &Path,
175    cipher: Option<&Arc<dyn Cipher>>,
176) -> Result<IndexManifest> {
177    // Reads the file (decrypting if it carries the encrypted-index header),
178    // verifies + strips the trailing CRC32, and returns the bitcode payload.
179    let data = super::common::read_and_verify_crc_maybe_encrypted(
180        path,
181        super::MAX_MANIFEST_FILE_SIZE,
182        "Manifest",
183        cipher,
184    )?;
185    decode_and_validate_manifest(data, path)
186}
187
188/// Load the manifest, decrypting via an [`IndexKeyring`](super::common::IndexKeyring)
189/// that dispatches on the header `key_version` (Issue #488 key rotation).
190pub(crate) fn load_manifest_with_keyring(
191    path: &Path,
192    keyring: Option<&super::common::IndexKeyring>,
193) -> Result<IndexManifest> {
194    let data = super::common::read_and_verify_crc_maybe_encrypted_with_keyring(
195        path,
196        super::MAX_MANIFEST_FILE_SIZE,
197        "Manifest",
198        keyring,
199    )?;
200    decode_and_validate_manifest(data, path)
201}
202
203/// Decode + validate the bitcode manifest payload (magic + version check).
204fn decode_and_validate_manifest(data: Vec<u8>, path: &Path) -> Result<IndexManifest> {
205    // Decode and validate
206    let manifest: IndexManifest = bitcode::decode(&data)?;
207
208    // Validate magic bytes
209    if manifest.magic != MANIFEST_MAGIC {
210        return Err(IndexPersistenceError::InvalidMagic {
211            path: path.to_path_buf(),
212            expected: MANIFEST_MAGIC,
213            got: manifest.magic,
214        });
215    }
216
217    // Validate version
218    if manifest.version > MANIFEST_VERSION {
219        return Err(IndexPersistenceError::UnsupportedVersion {
220            found: manifest.version,
221            supported: MANIFEST_VERSION,
222        });
223    }
224
225    Ok(manifest)
226}

235    #[test]
236    fn test_manifest_round_trip() {
237        let dir = tempdir().unwrap();
238        let path = dir.path().join("manifest.idx");
239
240        let mut manifest = IndexManifest::new(42);
241        manifest.string_interner = Some(StringInternerManifestEntry {
242            interner_file: "strings/interner.idx".to_string(),
243            string_count: 100,
244        });
245        manifest.vector_indexes.push(VectorIndexManifestEntry {
246            property_name: "embedding".to_string(),
247            dimensions: 384,
248            metric: 0,
249            current_file: "vector/embedding/current.usearch".to_string(),
250            mappings_file: "vector/embedding/current.mappings".to_string(),
251            snapshot_count: 5,
252            temporal_enabled: true,
253        });
254
255        save_manifest(&manifest, &path).unwrap();
256        let loaded = load_manifest(&path).unwrap();
257
258        assert_eq!(loaded.magic, MANIFEST_MAGIC);
259        assert_eq!(loaded.lsn, 42);
260        assert_eq!(loaded.vector_indexes.len(), 1);
261        assert_eq!(loaded.vector_indexes[0].property_name, "embedding");
262        assert!(loaded.string_interner.is_some());
263    }

265    #[test]
266    fn test_manifest_encrypted_round_trip() {
267        use crate::encryption::Aes256GcmCipher;
268        use zeroize::Zeroizing;
269
270        let dir = tempdir().unwrap();
271        let path = dir.path().join("manifest.idx");
272        let cipher: Arc<dyn Cipher> = {
273            let mut key = Zeroizing::new([0u8; 32]);
274            key[4] = 0x88;
275            Arc::new(Aes256GcmCipher::new(&key))
276        };
277
278        let manifest = IndexManifest::new(4242);
279        save_manifest_with_cipher(&manifest, &path, Some(&cipher)).unwrap();
280        assert!(super::super::common::is_encrypted_index(
281            &fs::read(&path).unwrap()
282        ));
283
284        let loaded = load_manifest_with_cipher(&path, Some(&cipher)).unwrap();
285        assert_eq!(loaded.lsn, 4242);
286        assert_eq!(loaded.magic, MANIFEST_MAGIC);
287
288        // Fails closed without a cipher; a legacy plaintext manifest still loads.
289        assert!(load_manifest_with_cipher(&path, None).is_err());
290        save_manifest(&manifest, &path).unwrap();
291        assert_eq!(
292            load_manifest_with_cipher(&path, Some(&cipher)).unwrap().lsn,
293            4242
294        );
295    }

308    #[test]
309    fn test_manifest_crc_corruption_detected() {
310        let dir = tempdir().unwrap();
311        let path = dir.path().join("manifest.idx");
312
313        // Save a valid manifest
314        let manifest = IndexManifest::new(42);
315        save_manifest(&manifest, &path).unwrap();
316
317        // Corrupt the data (change a byte in the middle)
318        let mut bytes = fs::read(&path).unwrap();
319        bytes[10] ^= 0xFF; // Flip all bits in one byte
320        fs::write(&path, bytes).unwrap();
321
322        // Loading should fail with corruption error
323        let result = load_manifest(&path);
324        assert!(result.is_err());
325        let err = result.unwrap_err();
326        assert!(err.to_string().contains("Index file corrupted"));
327    }

409    fn test_manifest_crc_covers_all_data() {
410        // Ensure CRC calculation includes all fields
411        let dir = tempdir().unwrap();
412        let path = dir.path().join("manifest.idx");
413
414        let mut manifest = IndexManifest::new(1);
415        manifest.lsn = 100;
416        save_manifest(&manifest, &path).unwrap();
417
418        // Read file
419        let mut bytes = fs::read(&path).unwrap();
420        // Modify LSN in the serialized data
421        // LSN is near the beginning. Bitcode is variable length, but we can just flip bytes.
422        // Flipping ANY byte in the data section (except last 4 CRC bytes) should trigger checksum mismatch.
423        let len = bytes.len();
424        bytes[len - 5] ^= 0xFF; // Flip last byte of data
425
426        fs::write(&path, bytes).unwrap();
427
428        let result = load_manifest(&path);
429        assert!(result.is_err());
430        assert!(matches!(
431            result.unwrap_err(),
432            IndexPersistenceError::Corrupted { .. }
433        ));
