1//! Index-layer key-rotation re-encryption engine (Issue #488).
2//!
3//! Rebuilds every encrypted index file under a data directory from an old key
4//! generation to a new one, in place, crash-safely, and resumably. It is the
5//! real work behind key rotation: the counter-only
6//! [`KeyRotationManager`](crate::encryption::KeyRotationManager) tracks the
7//! version state machine, while this engine actually rewrites persisted bytes.
8//!
9//! # How it works
10//!
11//! Every encrypted index file is self-describing: Issue #481's 10-byte plaintext
12//! header carries the `key_version` that wrote it. During a rotation the data
13//! dir holds a MIX of old- and new-key files; the engine:
14//!
15//! 1. Holds BOTH generations in an [`IndexKeyring`](super::common::IndexKeyring)
16//!    (so live reads dispatch on the header `key_version`).
17//! 2. Walks the `indexes/` tree and, for each file still tagged with the OLD
18//!    `key_version`, decrypts it with the old generation and re-encrypts it with
19//!    the NEW generation, publishing atomically via temp-write + rename
20//!    (a crash mid-file leaves either the intact old or the intact new file,
21//!    never a torn one).
22//! 3. Skips files already at the new `key_version` — the per-file header IS the
23//!    progress marker, so a resumed pass converges idempotently.
24//!
25//! `complete` refuses while any old-key file remains; only then is the old
26//! generation retired. `cancel` re-encrypts any already-migrated file back to
27//! the old generation (a reverse pass) and retires the new generation, leaving
28//! a uniform old-key dataset identical to the pre-`begin` state. Both
29//! generations stay live for the whole operation, so reads never fail.
30

41/// Errors from the index key-rotation engine.
42#[derive(Debug, thiserror::Error)]
43pub enum RotationError {
44    /// Encryption is not configured (no cipher), so there is nothing to rotate.
45    #[error("index encryption is not configured; cannot rotate keys")]
46    NotConfigured,
47    /// The new key derives the same index DEK as the old key.
48    #[error("new key equals the current key; rotation would be a no-op and risk nonce reuse")]
49    SameKey,
50    /// Index-only key rotation was requested while one or more OTHER layers
51    /// (WAL / checkpoint / cold storage) are still encrypted under the current
52    /// MEK. Rotating only the index tree to a new MEK and then switching the key
53    /// provider would leave those layers undecryptable — catastrophic loss — so
54    /// v1 refuses (Issue #488 P0.1). Full-MEK rotation across every layer is a
55    /// documented follow-up.
56    #[error(
57        "index-only key rotation is unsupported while other encrypted-at-rest \
58         layers are present ({layers}); rotating the index alone to a new key \
59         would leave those layers undecryptable"
60    )]
61    UnsupportedWhileEncryptedLayersPresent {
62        /// Comma-separated names of the still-encrypted layers (e.g. "wal").
63        layers: String,
64    },
65    /// A rotation is already in progress.
66    #[error("a key rotation is already in progress")]
67    AlreadyInProgress,
68    /// No rotation is in progress.
69    #[error("no key rotation is in progress")]
70    NotInProgress,
71    /// A file is stamped with the current (new) `key_version` but does NOT
72    /// authenticate under the current key — evidence of a foreign key
73    /// generation (e.g. an earlier interrupted rotation stamped the same version
74    /// number with a *different* key). Treating it as already-migrated would let
75    /// `complete` retire the old key over a file only the old key can read, so
76    /// the rotation wedges here instead (Issue #488 P0.3).
77    #[error(
78        "index file {path} is stamped with the current key_version but does not \
79         decrypt under the current key (foreign key generation); manual \
80         intervention required"
81    )]
82    ForeignKeyVersionFile {
83        /// The offending file.
84        path: PathBuf,
85    },
86    /// `complete` was called while old-key files remain.
87    #[error("cannot complete rotation: {remaining} index file(s) still hold the old key")]
88    OldKeyFilesRemain {
89        /// How many files are still tagged with a non-current key_version.
90        remaining: usize,
91    },
92    /// A key provider failed while sourcing an MEK.
93    #[error("key provider error: {0}")]
94    KeyProvider(String),
95    /// Filesystem error.
96    #[error("io error: {0}")]
97    Io(#[from] std::io::Error),
98    /// Index persistence error (decrypt/encrypt/serialize).
99    #[error("index error: {0}")]
100    Index(#[from] super::error::IndexPersistenceError),
101}

116/// Read-only classification of the on-disk index files by key generation.
117#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
118pub struct RotationStatus {
119    /// Encrypted files stamped with the current (new) `key_version`.
120    pub at_current: usize,
121    /// Encrypted files stamped with a non-current (old) `key_version`.
122    pub at_old: usize,
123    /// Encrypted files whose `key_version` matches no held generation.
124    pub unknown: usize,
125    /// Plaintext (non-encrypted) files encountered.
126    pub plaintext: usize,
127}
128
129impl RotationStatus {
130    /// True when no old-key or unknown-key encrypted files remain.
131    #[must_use]
132    pub fn is_fully_rotated(&self) -> bool {
133        self.at_old == 0 && self.unknown == 0
134    }
135}

137/// Result of the manifest-/index-body decrypt probe (Issue #3618).
138///
139/// [`RotationStatus`] classifies files by their 10-byte `AEIX` header
140/// `key_version` ONLY; it never AEAD-decrypts the body. A wrong key that happens
141/// to share the same `key_version` number therefore passes header classification
142/// (`unknown == 0`) even though it cannot actually decrypt anything — a
143/// false-PASS for `encryption verify`. This report captures an *active* decrypt
144/// probe: it attempts to AEAD-decrypt representative index bodies with the live
145/// keyring and records what actually authenticated.
146#[derive(Debug, Clone, Default, PartialEq, Eq)]
147pub struct DecryptProbeReport {
148    /// Number of encrypted files whose body the probe actually attempted to
149    /// decrypt (bounded — see [`IndexKeyRotation::verify_decryptable`]).
150    pub probed: usize,
151    /// Probed files whose body AEAD-decrypted successfully under the keyring.
152    pub decrypted_ok: usize,
153    /// Plaintext (non-`AEIX`) files encountered — skipped, NOT a failure.
154    pub plaintext: usize,
155    /// Paths of probed files whose body did NOT decrypt (wrong key material or
156    /// corruption). Path only — never any key bytes.
157    pub decrypt_failed: Vec<PathBuf>,
158}
159
160impl DecryptProbeReport {
161    /// True when the probe attempted at least one decrypt and none failed.
162    #[must_use]
163    pub fn all_decrypted(&self) -> bool {
164        self.decrypt_failed.is_empty()
165    }
166}

176/// The index re-encryption engine. Holds both key generations for the duration
177/// of a rotation and rewrites files from the old generation to the new.
178pub struct IndexKeyRotation {
179    /// The `indexes/` directory whose encrypted files are rotated.
180    indexes_dir: PathBuf,
181    /// Shared keyring holding both generations (dispatches live reads/writes).
182    keyring: IndexKeyring,
183    /// Old (retiring) generation.
184    old_version: u32,
185    old_cipher: Arc<dyn Cipher>,
186    /// New (current) generation.
187    new_version: u32,
188    new_cipher: Arc<dyn Cipher>,
189}
190
191impl IndexKeyRotation {
192    /// Construct an engine over `indexes_dir` for a rotation from `old` to
193    /// `new`. The `keyring` MUST already hold both generations (the new one
194    /// current); the engine mutates it only at `complete`/`cancel` to retire a
195    /// generation.
196    pub(crate) fn new(
197        indexes_dir: impl Into<PathBuf>,
198        keyring: IndexKeyring,
199        old_version: u32,
200        old_cipher: Arc<dyn Cipher>,
201        new_version: u32,
202        new_cipher: Arc<dyn Cipher>,
203    ) -> Self {
204        Self {
205            indexes_dir: indexes_dir.into(),
206            keyring,
207            old_version,
208            old_cipher,
209            new_version,
210            new_cipher,
211        }
212    }
213
214    /// The `key_version` new files are stamped with.
215    #[must_use]
216    pub fn new_version(&self) -> u32 {
217        self.new_version
218    }
219
220    /// The retiring `key_version`.
221    #[must_use]
222    pub fn old_version(&self) -> u32 {
223        self.old_version
224    }

226    /// Scan the index files and classify them by key generation without
227    /// rewriting anything (dry-run / verify probe).
228    pub fn status(&self) -> Result<RotationStatus, RotationError> {
229        let mut status = RotationStatus::default();
230        for path in self.enumerate_index_files()? {
231            match peek_key_version(&path)? {
232                None => status.plaintext += 1,
233                Some(v) if v == self.new_version => status.at_current += 1,
234                Some(v) if v == self.old_version => status.at_old += 1,
235                Some(_) => status.unknown += 1,
236            }
237        }
238        Ok(status)
239    }
240
241    /// Actively probe that index bodies decrypt under the live keyring
242    /// (Issue #3618).
243    ///
244    /// [`status`](Self::status) classifies files by their `AEIX` header
245    /// `key_version` alone and cannot catch a wrong key that shares the same
246    /// version number (the body is never touched). This probe closes that hole:
247    /// it AEAD-decrypts representative index *bodies* through the keyring and
248    /// reports which authenticated. A wrong key (or a corrupted body) fails the
249    /// AEAD auth tag — the header is AAD — so `decrypt_failed` becomes non-empty.
250    ///
251    /// **Bounded cost.** `verify` is an operator command, not a hot path, and
252    /// decrypting every file of a huge database would be wasteful. The probe
253    /// therefore decrypts the manifest body (the small canonical file every
254    /// encrypted database writes) plus ONE representative encrypted file per
255    /// distinct `key_version` actually present on disk. That fully catches a
256    /// uniformly-wrong configured key — the Issue #3618 scenario — since a wrong
257    /// key fails on the very first file of each generation.
258    ///
259    /// Plaintext (non-`AEIX`) files are counted under `plaintext` and skipped;
260    /// they are not decrypt failures. Recorded failure paths never carry key
261    /// material.
262    pub fn verify_decryptable(&self) -> Result<DecryptProbeReport, RotationError> {
263        use std::collections::HashSet;
264
265        let mut report = DecryptProbeReport::default();
266        let mut probed_versions: HashSet<u32> = HashSet::new();
267        let manifest_path = self.indexes_dir.join("manifest.idx");
268
269        // Walk EVERY non-scratch file (not just encrypted ones) so plaintext
270        // files are counted rather than silently invisible. Sorted for a
271        // deterministic representative per key-version.
272        let mut files = Vec::new();
273        if self.indexes_dir.exists() {
274            collect_all_files(&self.indexes_dir, &mut files)?;
275        }
276        files.sort();
277
278        for path in files {
279            // verify may run against a LIVE db: a file listed a moment ago can be
280            // removed/renamed by a concurrent writer's atomic_write between the
281            // scan and this read. A vanished file is benign — skip it rather than
282            // fail the probe; any OTHER io error still propagates. Peek only the
283            // 10-byte header first (index files can be up to ~100GB, so reading a
284            // whole body just to read its key_version would risk OOM); the full
285            // body is read ONLY for a file we actually decrypt-probe.
286            let key_version = match peek_key_version(&path) {
287                Ok(Some(v)) => v,
288                // Not an AEIX-encrypted file: plaintext, skip (not a failure).
289                Ok(None) => {
290                    report.plaintext += 1;
291                    continue;
292                }
293                Err(RotationError::Io(e)) if e.kind() == std::io::ErrorKind::NotFound => continue,
294                Err(e) => return Err(e),
295            };
296            // Bounded cost: always probe the manifest body; otherwise probe only
297            // the first file seen for each distinct key_version.
298            let is_manifest = path == manifest_path;
299            let first_of_version = probed_versions.insert(key_version);
300            if !is_manifest && !first_of_version {
301                continue;
302            }
303            // Only now read the full body — the file we will actually
304            // decrypt-probe (bounded to the manifest + one file per key_version).
305            let bytes = match std::fs::read(&path) {
306                Ok(b) => b,
307                Err(e) if e.kind() == std::io::ErrorKind::NotFound => continue,
308                Err(e) => return Err(e.into()),
309            };
310            report.probed += 1;
311            match decrypt_index_bytes_with_keyring(&bytes, &path, Some(&self.keyring)) {
312                Ok(_) => report.decrypted_ok += 1,
313                // Path only — never any key bytes.
314                Err(_) => report.decrypt_failed.push(path),
315            }
316        }
317        Ok(report)
318    }
319
320    /// Re-encrypt every old-key file to the new key (forward pass).
321    ///
322    /// Idempotent and resumable: files already at the new key are skipped, so a
323    /// crash-interrupted pass converges when re-run. `progress` is invoked after
324    /// each file; returning `false` stops the pass early (used to simulate a
325    /// crash mid-rotation in tests), leaving a valid mix that a later call
326    /// finishes.
327    pub fn re_encrypt(
328        &self,
329        progress: &mut dyn FnMut(RotationProgress) -> bool,
330    ) -> Result<RotationProgress, RotationError> {
331        let files = self.enumerate_index_files()?;
332        let mut p = RotationProgress {
333            files_total: files.len(),
334            ..Default::default()
335        };
336        for path in files {
337            match self.reencrypt_one(&path, self.new_version, &self.new_cipher)? {
338                FileOutcome::Reencrypted => p.files_reencrypted += 1,
339                FileOutcome::SkippedAlreadyNew | FileOutcome::SkippedPlaintext => {
340                    p.files_skipped += 1
341                }
342            }
343            p.files_done += 1;
344            if !progress(p) {
345                break;
346            }
347        }
348        Ok(p)
349    }
350
351    /// Verify a full pass completed (zero old/unknown encrypted files remain)
352    /// AND that every file claiming the new `key_version` actually authenticates
353    /// under the new key.
354    ///
355    /// Binding verification to key IDENTITY, not just the version NUMBER, closes
356    /// the double-rotation hole (Issue #488 P0.3): a file stamped with
357    /// `new_version` by a *different* key (from an earlier interrupted rotation)
358    /// is not silently counted as migrated — it wedges the rotation with
359    /// [`RotationError::ForeignKeyVersionFile`] so [`complete`](Self::complete)
360    /// can never retire the old key over a file only the old key can read.
361    pub fn verify_complete(&self) -> Result<(), RotationError> {
362        let mut remaining = 0;
363        for path in self.enumerate_index_files()? {
364            let bytes = std::fs::read(&path)?;
365            match index_file_key_version(&bytes) {
366                // Not an encrypted-index file (enumeration already excludes
367                // these, but stay defensive).
368                None => {}
369                Some(v) if v == self.new_version => {
370                    // Version says "new" — prove it with an AEAD auth check under
371                    // the current key, not the bare number.
372                    if decrypt_index_bytes(&bytes, &path, Some(&self.new_cipher)).is_err() {
373                        return Err(RotationError::ForeignKeyVersionFile { path });
374                    }
375                }
376                // Old or unknown key generation still present.
377                Some(_) => remaining += 1,
378            }
379        }
380        if remaining > 0 {
381            return Err(RotationError::OldKeyFilesRemain { remaining });
382        }
383        Ok(())
384    }
385
386    /// Finish the rotation: require a fully-rotated dataset, then retire the old
387    /// generation from the keyring so the old key can be dropped.
388    pub fn complete(&self) -> Result<(), RotationError> {
389        self.verify_complete()?;
390        self.keyring.retain_only(self.new_version);
391        Ok(())
392    }
393
394    /// Roll back: re-encrypt any already-migrated files back to the OLD key,
395    /// then retire the new generation. Leaves a uniform old-key dataset. Both
396    /// generations remain live until the reverse pass finishes, so reads stay
397    /// correct throughout.
398    pub fn cancel(&self) -> Result<(), RotationError> {
399        for path in self.enumerate_index_files()? {
400            // Reverse: rewrite files currently at the new key back to old.
401            self.reencrypt_one(&path, self.old_version, &self.old_cipher)?;
402        }
403        self.keyring.retain_only(self.old_version);
404        Ok(())
405    }

407    /// Byte-level, format-agnostic re-encryption of a single file to
408    /// `target_version` using `target_cipher`. Reads the whole file, dispatches
409    /// the decrypt cipher on the header `key_version` via the keyring, and
410    /// publishes atomically. Files already at `target_version`, and plaintext
411    /// files, are left untouched.
412    fn reencrypt_one(
413        &self,
414        path: &Path,
415        target_version: u32,
416        target_cipher: &Arc<dyn Cipher>,
417    ) -> Result<FileOutcome, RotationError> {
418        let bytes = std::fs::read(path)?;
419        if !is_encrypted_index(&bytes) {
420            return Ok(FileOutcome::SkippedPlaintext);
421        }
422        let current = index_file_key_version(&bytes).unwrap_or(0);
423        if current == target_version {
424            // Key-identity binding (Issue #488 P0.3): a matching version NUMBER
425            // is not proof this file was written by `target_cipher`. A prior
426            // interrupted rotation could have stamped `target_version` with a
427            // DIFFERENT key. Confirm the file authenticates under the target key
428            // before trusting it as already-migrated; otherwise wedge rather
429            // than skip, so a later `complete` never retires the old key over a
430            // foreign-key file.
431            return match decrypt_index_bytes(&bytes, path, Some(target_cipher)) {
432                Ok(_) => Ok(FileOutcome::SkippedAlreadyNew),
433                Err(_) => Err(RotationError::ForeignKeyVersionFile {
434                    path: path.to_path_buf(),
435                }),
436            };
437        }
438        // Decrypt with whichever generation wrote this file (keyring dispatch).
439        let plaintext = decrypt_index_bytes_with_keyring(&bytes, path, Some(&self.keyring))?;
440        let reencrypted = encrypt_index_bytes_versioned(&plaintext, target_cipher, target_version)?;
441        atomic_write(path, &reencrypted)?;
442        Ok(FileOutcome::Reencrypted)
443    }

445    /// Recursively collect every regular file under `indexes_dir` that carries
446    /// the encrypted-index header. Temp files (atomic-write scratch and the
447    /// native-index decrypt scratch) are skipped by name.
448    fn enumerate_index_files(&self) -> Result<Vec<PathBuf>, RotationError> {
449        let mut out = Vec::new();
450        if self.indexes_dir.exists() {
451            collect_index_files(&self.indexes_dir, &mut out)?;
452        }
453        out.sort();
454        Ok(out)
455    }
456}

458/// Read only the header bytes of `path` and return its `key_version`, or `None`
459/// for a plaintext file or a file too short to carry a header.
460fn peek_key_version(path: &Path) -> Result<Option<u32>, RotationError> {
461    use std::io::Read;
462    let mut header = [0u8; ENC_HEADER_LEN];
463    let mut file = std::fs::File::open(path)?;
464    let mut filled = 0usize;
465    loop {
466        match file.read(&mut header[filled..]) {
467            Ok(0) => break,
468            Ok(n) => {
469                filled += n;
470                if filled == header.len() {
471                    break;
472                }
473            }
474            Err(ref e) if e.kind() == std::io::ErrorKind::Interrupted => continue,
475            Err(e) => return Err(e.into()),
476        }
477    }
478    Ok(index_file_key_version(&header[..filled]))
479}

481/// The maximum `key_version` stamped across the persisted index files under
482/// `indexes_dir` (Issue #488 version-provisioning).
483///
484/// Reads only file *headers* via [`peek_key_version`]; returns `None` when the
485/// directory holds no encrypted-index files (plaintext or empty). The durable
486/// `open()` path folds this with the WAL segment version to provision the index
487/// keyring's current version, so a rotated-then-reopened database reports the
488/// real version instead of a hard-coded 1.
489pub(crate) fn max_index_key_version_in_dir(
490    indexes_dir: &Path,
491) -> Result<Option<u32>, RotationError> {
492    let mut files = Vec::new();
493    if indexes_dir.exists() {
494        collect_index_files(indexes_dir, &mut files)?;
495    }
496    let mut max: Option<u32> = None;
497    for path in files {
498        if let Some(v) = peek_key_version(&path)? {
499            max = Some(max.map_or(v, |m| m.max(v)));
500        }
501    }
502    Ok(max)
503}

505/// Progress of a plaintext → `AEIX` wrap pass (Issue #3616 PR3 enable).
506#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
507pub struct WrapProgress {
508    /// Total files considered under the indexes dir (plaintext + already-`AEIX`).
509    pub files_total: usize,
510    /// Bare plaintext files freshly wrapped into `AEIX` this pass.
511    pub files_wrapped: usize,
512    /// Files already carrying the `AEIX` header, left untouched (idempotency).
513    pub files_skipped: usize,
514}
515
516/// Wrap every **bare plaintext** index file under `indexes_dir` into the
517/// encrypted `AEIX` format under `cipher`, stamping `key_version` (Issue #3616
518/// PR3 — the plaintext → encrypted *enable* migration).
519///
520/// This is the counterpart the `enable` engine needs but
521/// [`IndexKeyRotation::re_encrypt`] cannot provide: `re_encrypt` walks only files
522/// that already carry the `AEIX` header and returns
523/// [`FileOutcome::SkippedPlaintext`] for a bare file (it rekeys old-gen → new-gen
524/// *encrypted* files). Enabling encryption over a plaintext database must do the
525/// opposite — take each bare file and produce an `AEIX` file.
526///
527/// A plaintext index file on disk is exactly the AEAD *plaintext* an encrypted
528/// index file wraps (both are `[bitcode_body][crc32:4]` before the header — see
529/// [`save_encoded_encrypted`](super::common::save_encoded_encrypted)), so this
530/// pass reads the whole file and re-publishes it as
531/// `[header][AEAD(whole_file, aad=header)]` atomically (temp-write + rename). The
532/// transform is byte-preserving: the encrypted reader (`read_index_file`) decrypts
533/// back to the identical original bytes, so a checkpoint / native-`usearch` / any
534/// index file round-trips regardless of its internal shape.
535///
536/// **Idempotent / resumable / crash-safe.** A file already carrying the `AEIX`
537/// header is skipped, so a re-run after a crash mid-pass converges (each file is
538/// published atomically, so a crash leaves either the intact plaintext or the
539/// intact `AEIX` file, never a torn one). Checkpoint files ride the index DEK and
540/// format and so are covered by this same pass. Every non-scratch regular file
541/// under `indexes_dir` participates (control-plane files such as
542/// `rotation.state` / `encryption.state` live in the manager's *base* dir, one
543/// level up, and are never enumerated here).
544///
545/// # Errors
546///
547/// Returns an error if a file cannot be read, encrypted, or atomically written.
548pub(crate) fn wrap_plaintext_index_dir(
549    indexes_dir: &Path,
550    cipher: &Arc<dyn Cipher>,
551    key_version: u32,
552) -> Result<WrapProgress, RotationError> {
553    let mut files = Vec::new();
554    if indexes_dir.exists() {
555        collect_all_files(indexes_dir, &mut files)?;
556    }
557    files.sort();
558
559    let mut progress = WrapProgress {
560        files_total: files.len(),
561        ..Default::default()
562    };
563    for path in files {
564        let bytes = std::fs::read(&path)?;
565        if is_encrypted_index(&bytes) {
566            // Already wrapped (a resumed pass, or an interleaved encrypted write).
567            progress.files_skipped += 1;
568            continue;
569        }
570        let wrapped = encrypt_index_bytes_versioned(&bytes, cipher, key_version)?;
571        atomic_write(&path, &wrapped)?;
572        progress.files_wrapped += 1;
573    }
574    Ok(progress)
575}

577/// Progress of an `AEIX` → plaintext unwrap pass (Issue #3616 PR4 disable).
578///
579/// Consumed in non-test builds by the disable-engine driver (via
580/// `unwrap_disable_index_files` in `db/rotation.rs`).
581#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
582pub struct UnwrapProgress {
583    /// Total files considered under the indexes dir (`AEIX` + already-plaintext).
584    pub files_total: usize,
585    /// `AEIX` files freshly decrypted back to bare plaintext this pass.
586    pub files_unwrapped: usize,
587    /// Files already bare (no `AEIX` header), left untouched (idempotency).
588    pub files_skipped: usize,
589}
590
591/// Unwrap every **`AEIX`-encrypted** index file under `indexes_dir` back to bare
592/// plaintext, decrypting under `cipher` (Issue #3616 PR4 — the encrypted →
593/// plaintext *disable* migration; the exact inverse of
594/// [`wrap_plaintext_index_dir`]).
595///
596/// This is the counterpart the `disable` engine needs and the mirror of the
597/// enable engine's [`wrap_plaintext_index_dir`]: enable takes each bare file and
598/// produces an `AEIX` file; disable takes each `AEIX` file and produces the bare
599/// plaintext body it wraps. The wrap is byte-preserving
600/// (`[header][AEAD(whole_file, aad=header)]` decrypts back to the identical
601/// original `[bitcode_body][crc32:4]` bytes — see
602/// [`save_encoded_encrypted`](super::common::save_encoded_encrypted)), so a
603/// checkpoint / native-`usearch` / any index file round-trips regardless of its
604/// internal shape and the resulting file is byte-identical to the pre-enable
605/// plaintext. Every non-header index file is republished atomically (temp-write +
606/// rename), exactly as the wrap pass does.
607///
608/// The decrypt `cipher` is passed explicitly (mirroring the wrap's explicit
609/// encrypt cipher): the disable driver derives it once from the recorded key
610/// source. A single cipher decrypts every file the enable wrap stamped, since the
611/// stamped `key_version` is authenticated as AAD but the key itself is the sole
612/// index DEK a disable retires.
613///
614/// **Idempotent / resumable / crash-safe.** A file already bare (no `AEIX`
615/// header) is skipped, so a re-run after a crash mid-pass converges (each file is
616/// published atomically, so a crash leaves either the intact `AEIX` file or the
617/// intact plaintext file, never a torn one). Control-plane files
618/// (`rotation.state` / `encryption.state`) live one level up in the manager's
619/// base dir and are never enumerated here.
620///
621/// # Errors
622///
623/// Returns an error if a file cannot be read, decrypted (a genuine wrong/absent
624/// key — surfaced loudly, never silently left encrypted), or atomically written.
625pub(crate) fn unwrap_encrypted_index_dir(
626    indexes_dir: &Path,
627    cipher: &Arc<dyn Cipher>,
628) -> Result<UnwrapProgress, RotationError> {
629    let mut files = Vec::new();
630    if indexes_dir.exists() {
631        collect_all_files(indexes_dir, &mut files)?;
632    }
633    files.sort();
634
635    let mut progress = UnwrapProgress {
636        files_total: files.len(),
637        ..Default::default()
638    };
639    for path in files {
640        let bytes = std::fs::read(&path)?;
641        if !is_encrypted_index(&bytes) {
642            // Already bare (a resumed pass, or a never-encrypted file).
643            progress.files_skipped += 1;
644            continue;
645        }
646        let plaintext = decrypt_index_bytes(&bytes, &path, Some(cipher))?;
647        atomic_write(&path, &plaintext)?;
648        progress.files_unwrapped += 1;
649    }
650    Ok(progress)
651}

653/// A temp/scratch file the rotation engine must never treat as an index file.
654fn is_scratch_file(name: &str) -> bool {
655    name.ends_with(".tmp") || name.contains(".tmp.") || name.starts_with(".aeix-usearch-tmp-")
656}
657
658/// Recursively collect every regular, non-scratch file under `dir` — encrypted
659/// AND plaintext. Unlike [`collect_index_files`], this does NOT filter to files
660/// carrying the `AEIX` header, so the decrypt probe (Issue #3618) can count
661/// plaintext files instead of treating them as absent.
662fn collect_all_files(dir: &Path, out: &mut Vec<PathBuf>) -> Result<(), RotationError> {
663    for entry in std::fs::read_dir(dir)? {
664        let entry = entry?;
665        let path = entry.path();
666        let file_type = entry.file_type()?;
667        if file_type.is_dir() {
668            collect_all_files(&path, out)?;
669            continue;
670        }
671        if !file_type.is_file() {
672            continue;
673        }
674        let name = entry.file_name();
675        let name = name.to_string_lossy();
676        if is_scratch_file(&name) {
677            continue;
678        }
679        out.push(path);
680    }
681    Ok(())
682}
683
684fn collect_index_files(dir: &Path, out: &mut Vec<PathBuf>) -> Result<(), RotationError> {
685    for entry in std::fs::read_dir(dir)? {
686        let entry = entry?;
687        let path = entry.path();
688        let file_type = entry.file_type()?;
689        if file_type.is_dir() {
690            collect_index_files(&path, out)?;
691            continue;
692        }
693        if !file_type.is_file() {
694            continue;
695        }
696        let name = entry.file_name();
697        let name = name.to_string_lossy();
698        if is_scratch_file(&name) {
699            continue;
700        }
701        // Only files that are actually encrypted-index files participate.
702        if peek_key_version(&path)?.is_some() {
703            out.push(path);
704        }
705    }
706    Ok(())
707}
