1//! Bitcode-serializable format structs for index persistence.
2//!
3//! This module defines the schema for all persisted index files. These structs are
4//! serialized using [bitcode](https://github.com/llogiq/bitcode), a fast and compact
5//! binary serialization format.
6//!
7//! # Schema Overview
8//!
9//! | Struct | Corresponding File | Description |
10//! |--------|-------------------|-------------|
11//! | [`IndexManifest`] | `manifest.idx` | Root registry of all indexes |
12//! | [`StringInternerData`] | `strings/interner.idx` | Interned strings table |
13//! | [`GraphIndexData`] | `graph/adjacency.idx` | Graph structure and properties |
14//! | [`GraphIndexDelta`] | `graph/delta.idx` | Incremental graph changes |
15//! | [`TemporalIndexData`] | `temporal/versions.idx` | Historical version chains |
16//! | [`VectorIndexMeta`] | `vector/{prop}/meta.idx` | Vector index metadata |
17//! | [`VectorMappingsData`] | `vector/{prop}/mappings.idx` | Vector ID mappings |

25/// Root manifest - entry point for index loading.
26#[derive(Debug, Clone, Encode, Decode)]
27pub struct IndexManifest {
28    /// Magic bytes: "GIDX"
29    pub magic: [u8; 4],
30    /// Format version
31    pub version: u16,
32    /// Unix timestamp when created
33    pub created_at: i64,
34    /// Unix timestamp of last modification
35    pub last_modified: i64,
36    /// WAL position this manifest is consistent with.
37    ///
38    /// # Semantics (Issue #3419)
39    ///
40    /// This is the **applied-frontier watermark** =
41    /// `min(in-flight commit base LSN, wal.current_lsn())` captured *before*
42    /// the snapshot was taken — i.e. the minimum LSN of any commit that is
43    /// durable (fsynced) but not yet applied, or the WAL allocation frontier
44    /// when no commit is in flight. It is NOT the last-allocated LSN, and it
45    /// equals the allocation frontier (`wal.current_lsn()`) ONLY when no commit
46    /// is in flight. The resulting contract is:
47    ///
48    /// - Every WAL entry with LSN **< lsn** is guaranteed to be applied AND
49    ///   reflected in the persisted snapshot.
50    /// - Entries with LSN **>= lsn** may or may not be reflected (a write can
51    ///   race a background persist, or be durable-but-unapplied at capture
52    ///   time), and are re-applied idempotently by inclusive replay.
53    ///
54    /// Startup replay must therefore begin **AT `lsn` (inclusive)** — never
55    /// at `lsn + 1`, which silently drops the first post-persist write — and
56    /// WAL replay must be idempotent for entries whose effects are already in
57    /// the snapshot (see the re-application guards in
58    /// `storage::recovery::replay_wal_into_storage_with_constraints`).
59    ///
60    /// The allocator must also never allocate below this value after a
61    /// restart (Issue #3420); startup seeds it from
62    /// `max(max LSN in WAL segments + 1, manifest lsn)`.
63    pub lsn: u64,
64
65    /// Vector index entries (one per property)
66    pub vector_indexes: Vec<VectorIndexManifestEntry>,
67    /// Graph index entry
68    pub graph_index: Option<GraphIndexManifestEntry>,
69    /// Temporal index entry
70    pub temporal_index: Option<TemporalIndexManifestEntry>,
71    /// Temporal adjacency index entry
72    pub temporal_adjacency_index: Option<TemporalAdjacencyIndexManifestEntry>,
73    /// String interner entry
74    pub string_interner: Option<StringInternerManifestEntry>,
75}

300/// Persisted write-time attributive provenance bundle (Issue #3224).
301///
302/// Mirrors [`crate::core::provenance::Provenance`]'s fields exactly; kept as
303/// a separate bitcode-encodable type since `Provenance` itself has private
304/// fields and validates on construction (not on persistence).
305///
306/// FORMAT-FROZEN: reused verbatim by `legacy_v3` and `BackupPayloadV3`
307/// (the pre-#3350 shape is separately frozen as
308/// `legacy_v2::PersistedProvenanceV2`). Changing its wire layout requires
309/// freezing a copy for those shapes first.
310#[derive(Debug, Clone, Encode, Decode)]
311pub struct PersistedProvenance {
312    /// Source system/identifier that produced the write, if any.
313    pub source: Option<String>,
314    /// Confidence in `[0.0, 1.0]`, if any.
315    pub confidence: Option<f64>,
316    /// Free-text note, if any.
317    pub note: Option<String>,
318    /// Correlation ID grouping co-committed writes, if any.
319    pub correlation_id: Option<String>,
320    /// Authenticated principal that made the write (Issue #3350), if any.
321    pub principal: Option<String>,
322}
