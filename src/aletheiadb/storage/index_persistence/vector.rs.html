66/// Load vector index metadata, transparently decrypting it if written
67/// encrypted (Issue #481).
68///
69/// # Errors
70///
71/// Same as [`load_vector_meta`], plus a structured error if the file is
72/// encrypted but no cipher is supplied or decryption fails.

81/// Load vector index metadata, decrypting via an
82/// [`IndexKeyring`](super::common::IndexKeyring) that dispatches on the header
83/// `key_version` (Issue #488 key rotation).
84pub(crate) fn load_vector_meta_with_keyring(
85    path: &Path,
86    keyring: Option<&super::common::IndexKeyring>,
87) -> Result<VectorIndexMeta> {
88    // Metadata should be small, but use standard limit for consistency
89    let meta: VectorIndexMeta = super::common::load_encoded_maybe_encrypted_with_keyring(
90        path,
91        super::MAX_VECTOR_INDEX_FILE_SIZE,
92        "Vector index",
93        keyring,
94    )?;
95
96    if meta.magic != VECTOR_META_MAGIC {
97        return Err(IndexPersistenceError::InvalidMagic {
98            path: path.to_path_buf(),
99            expected: VECTOR_META_MAGIC,
100            got: meta.magic,
101        });
102    }
103
104    if meta.version > MANIFEST_VERSION {
105        return Err(IndexPersistenceError::UnsupportedVersion {
106            found: meta.version,
107            supported: MANIFEST_VERSION,
108        });
109    }
110
111    Ok(meta)
112}

230/// Load a complete vector index (meta + mappings + index path) from a
231/// directory, transparently decrypting the `meta.idx` / `mappings.idx` files if
232/// they were written encrypted (Issue #481).
233///
234/// # Errors
235///
236/// Returns an error if any required file (meta.idx, mappings.idx) is missing or
237/// corrupted, or is encrypted but no cipher is supplied / decryption fails.
238pub fn load_vector_index_with_cipher(
239    path: &Path,
240    cipher: Option<&Arc<dyn Cipher>>,
241) -> Result<VectorIndexData> {
242    let meta_path = path.join("meta.idx");
243    let mappings_path = path.join("mappings.idx");
244    // usearch index is just a file path, we don't load it into memory here generally
245    let index_path = path.join("current.usearch");
246
247    let meta = load_vector_meta_with_cipher(&meta_path, cipher)?;
248    let mappings = load_vector_mappings_with_cipher(&mappings_path, cipher)?;
249
250    Ok(VectorIndexData {
251        meta,
252        mappings,
253        index_path,
254    })
255}

257/// Create new vector index metadata.
258///
259/// Initializes a new configuration object for an HNSW vector index, capturing the
260/// creation timestamp, index dimensions, distance metric, and the `usearch`
261/// hyper-parameters (`m`, `ef_construction`, `ef_search`).
262pub fn new_vector_meta(
263    property_name: &str,
264    dimensions: u32,
265    metric: u8,
266    config: PersistedHnswConfig,
267) -> VectorIndexMeta {
268    let now = SystemTime::now()
269        .duration_since(UNIX_EPOCH)
270        .unwrap_or_else(|_| Duration::from_secs(0))
271        .as_secs() as i64;
272
273    VectorIndexMeta {
274        magic: VECTOR_META_MAGIC,
275        version: MANIFEST_VERSION,
276        property_name: property_name.to_string(),
277        dimensions,
278        metric,
279        hnsw_config: config,
280        vector_count: 0,
281        created_at: now,
282        last_modified: now,
283    }
284}

299// ── Native HNSW (`current.usearch`) encryption-at-rest (Issue #481) ───────
300//
301// The native usearch index is written and memory-mapped directly by the
302// bundled `usearch` C++ library through a filesystem PATH — it has no
303// in-memory-buffer API we can route through our AEAD cipher. To encrypt it at
304// rest anyway, we let usearch produce its two plaintext files at a temporary
305// path (`<base>` + `<base>.usearch.mappings`), then whole-file-encrypt each
306// (with the same `AEIX` plaintext header used for every other index file, fed
307// as AAD) and atomically write the ciphertext to the real paths. On load, if
308// the real file carries the `AEIX` header we decrypt both files to a temp
309// path in the same directory, hand THAT path to usearch, and remove the temp
310// files afterwards (a drop guard cleans them up on every exit path). When no
311// cipher is configured the plaintext path is byte-identical to a plain
312// `index.save(path)` / `HnswIndex::load(path)` — a pure no-op — and a legacy
313// plaintext `current.usearch` (no header) still loads when a cipher IS
314// configured (the upgrade scenario), via header sniffing.
315
316/// RAII guard that best-effort removes a set of temporary files when dropped,
317/// so a decrypt-to-temp (or a usearch save-to-temp) never leaks partial
318/// plaintext on disk — on success OR on any error/early-return path.
319struct TempFileGuard {
320    paths: Vec<PathBuf>,
321}
322
323impl TempFileGuard {
324    fn new(paths: Vec<PathBuf>) -> Self {
325        Self { paths }
326    }
327}
328
329impl Drop for TempFileGuard {
330    fn drop(&mut self) {
331        for path in &self.paths {
332            // Best-effort: a missing temp file (never created / already gone)
333            // is fine; we deliberately ignore errors so cleanup never panics.
334            let _ = std::fs::remove_file(path);
335        }
336    }
337}

339/// Build a unique temporary sibling path for the native usearch shuffle,
340/// living in the SAME directory as `real_path` so the subsequent
341/// `atomic_write` rename stays on one filesystem. The `.usearch` extension is
342/// kept so usearch's own `path.with_extension("usearch.mappings")` sidecar
343/// naming stays consistent between the save and load temp bases.
344fn native_temp_base(real_path: &Path) -> PathBuf {
345    use rand::Rng;
346    let suffix: u64 = rand::thread_rng().r#gen();
347    let dir = real_path.parent().unwrap_or_else(|| Path::new("."));
348    dir.join(format!(".aeix-usearch-tmp-{suffix}.usearch"))
349}

351/// Whole-file-encrypt the plaintext bytes at `src`, stamping `key_version` into
352/// the `[AEIX header]`, and atomically write the ciphertext blob to `dst`
353/// (Issue #488 key rotation; the native usearch write path).
354fn encrypt_file_whole_versioned(
355    src: &Path,
356    dst: &Path,
357    cipher: &Arc<dyn Cipher>,
358    key_version: u32,
359) -> Result<()> {
360    let plaintext = std::fs::read(src)?;
361    let encrypted = super::common::encrypt_index_bytes_versioned(&plaintext, cipher, key_version)?;
362    super::atomic_write(dst, &encrypted)
363}

365/// Read `src`, transparently decrypting if it carries the `AEIX` header (a
366/// legacy plaintext file is copied through unchanged, covering the upgrade /
367/// torn-write case) via an [`IndexKeyring`](super::common::IndexKeyring) that
368/// dispatches on the header `key_version`, and atomically write the plaintext to
369/// `dst` (Issue #488 key rotation; the native usearch read path).
370fn decrypt_file_to_with_keyring(
371    src: &Path,
372    dst: &Path,
373    max_size: u64,
374    context: &str,
375    keyring: Option<&super::common::IndexKeyring>,
376) -> Result<()> {
377    let plaintext = super::common::read_index_file_with_keyring(src, max_size, context, keyring)?;
378    super::atomic_write(dst, &plaintext)
379}

381/// Save the native HNSW index, encrypting both files with the current
382/// generation of an [`IndexKeyring`](super::common::IndexKeyring) and stamping
383/// its `key_version` (Issue #488 key rotation). A `None`/empty keyring writes
384/// plaintext (identical to `index.save`).
385pub(crate) fn save_hnsw_index_with_keyring(
386    index: &crate::index::vector::HnswIndex,
387    real_path: &Path,
388    keyring: Option<&super::common::IndexKeyring>,
389) -> crate::core::error::Result<()> {
390    use crate::index::vector::VectorIndex;
391
392    let Some((cipher, key_version)) = keyring.and_then(super::common::IndexKeyring::current) else {
393        return index.save(real_path);
394    };
395
396    let temp_base = native_temp_base(real_path);
397    let temp_mappings = temp_base.with_extension("usearch.mappings");
398    let _guard = TempFileGuard::new(vec![temp_base.clone(), temp_mappings.clone()]);
399
400    index.save(&temp_base)?;
401
402    let real_mappings = real_path.with_extension("usearch.mappings");
403    encrypt_file_whole_versioned(&temp_base, real_path, &cipher, key_version)
404        .map_err(native_enc_err)?;
405    encrypt_file_whole_versioned(&temp_mappings, &real_mappings, &cipher, key_version)
406        .map_err(native_enc_err)?;
407    Ok(())
408}

410/// Load the native HNSW index, transparently decrypting `current.usearch` and
411/// its sidecar first if they carry the `AEIX` header (Issue #481).
412///
413/// A legacy plaintext file (no header) is handed straight to usearch with no
414/// temp shuffle, so the plaintext path is unchanged.
415/// Load the native HNSW index, decrypting via an
416/// [`IndexKeyring`](super::common::IndexKeyring) that dispatches on the header
417/// `key_version` (Issue #488 key rotation). A legacy plaintext file loads
418/// directly.
419pub(crate) fn load_hnsw_index_with_keyring(
420    real_path: &Path,
421    config: crate::index::vector::HnswConfig,
422    keyring: Option<&super::common::IndexKeyring>,
423) -> crate::core::error::Result<crate::index::vector::HnswIndex> {
424    use crate::index::vector::HnswIndex;
425
426    if !native_file_is_encrypted(real_path) {
427        return HnswIndex::load(real_path, config);
428    }
429
430    if keyring.is_none() {
431        return Err(crate::core::error::Error::Vector(
432            crate::core::error::VectorError::IndexError(
433                "usearch index is encrypted but no encryption key is configured".to_string(),
434            ),
435        ));
436    }
437
438    let temp_base = native_temp_base(real_path);
439    let temp_mappings = temp_base.with_extension("usearch.mappings");
440    let real_mappings = real_path.with_extension("usearch.mappings");
441    let _guard = TempFileGuard::new(vec![temp_base.clone(), temp_mappings.clone()]);
442
443    decrypt_file_to_with_keyring(
444        real_path,
445        &temp_base,
446        super::MAX_MMAP_FILE_SIZE,
447        "usearch index",
448        keyring,
449    )
450    .map_err(native_enc_err)?;
451    decrypt_file_to_with_keyring(
452        &real_mappings,
453        &temp_mappings,
454        super::MAX_VECTOR_INDEX_FILE_SIZE,
455        "usearch mappings",
456        keyring,
457    )
458    .map_err(native_enc_err)?;
459
460    HnswIndex::load(&temp_base, config)
461}

463/// Sniff the first bytes of `path` for the `AEIX` encrypted-index header
464/// without reading the whole (potentially huge) native index file.
465fn native_file_is_encrypted(path: &Path) -> bool {
466    use std::io::Read;
467    let mut header = [0u8; super::common::ENC_HEADER_LEN];
468    match std::fs::File::open(path).and_then(|mut f| f.read_exact(&mut header)) {
469        Ok(()) => super::common::is_encrypted_index(&header),
470        Err(_) => false,
471    }
472}

474/// Map an index-persistence error from the native encrypt/decrypt shuffle into
475/// the vector-index error channel used by `HnswIndex::save`/`load`. The message
476/// never contains key material (it is derived from `IndexPersistenceError`,
477/// which carries only paths and cipher-opaque failure strings).
478fn native_enc_err(e: IndexPersistenceError) -> crate::core::error::Error {
479    crate::core::error::Error::Vector(crate::core::error::VectorError::IndexError(format!(
480        "native usearch index encryption/decryption failed: {e}"
481    )))
482}

640    #[test]
641    fn test_native_usearch_encrypted_round_trip_and_fail_closed() {
642        use crate::core::id::NodeId;
643        use crate::index::vector::{DistanceMetric, HnswConfig, HnswIndex, VectorIndex};
644
645        let dir = tempdir().unwrap();
646        let path = dir.path().join("current.usearch");
647        let cipher = enc_test_cipher();
648
649        let config = HnswConfig::new(4, DistanceMetric::Cosine);
650        let index = HnswIndex::new(config.clone()).unwrap();
651        index
652            .add(NodeId::new(1).unwrap(), &[1.0, 0.0, 0.0, 0.0])
653            .unwrap();
654        index
655            .add(NodeId::new(2).unwrap(), &[0.0, 1.0, 0.0, 0.0])
656            .unwrap();
657
658        let ring = super::super::common::IndexKeyring::single(cipher.clone());
659        save_hnsw_index_with_keyring(&index, &path, Some(&ring)).unwrap();
660
661        // Both the native index file and its mappings sidecar are encrypted at
662        // rest (carry the AEIX header).
663        assert!(
664            native_file_is_encrypted(&path),
665            "native index not encrypted on disk"
666        );
667        let sidecar = path.with_extension("usearch.mappings");
668        assert!(
669            native_file_is_encrypted(&sidecar),
670            "mappings sidecar not encrypted on disk"
671        );
672        // No decrypted-plaintext temp files leaked after save.
673        assert!(
674            !any_aeix_temp_present(dir.path()),
675            "plaintext temp file leaked after save"
676        );
677
678        // Correct key => decrypts to a temp file, loads, and the vectors survive.
679        let loaded = load_hnsw_index_with_keyring(&path, config.clone(), Some(&ring)).unwrap();
680        assert_eq!(loaded.len(), 2);
681        assert!(
682            !any_aeix_temp_present(dir.path()),
683            "plaintext temp file leaked after load"
684        );
685
686        // Fails closed: encrypted on disk but no cipher supplied. The error
687        // message names the missing key and leaks no plaintext / no temp file.
688        let err = load_hnsw_index_with_keyring(&path, config.clone(), None).unwrap_err();
689        assert!(
690            format!("{err}").contains("no encryption key"),
691            "expected fail-closed missing-key error, got: {err}"
692        );
693        assert!(
694            !any_aeix_temp_present(dir.path()),
695            "temp leaked on fail-closed load"
696        );
697
698        // Wrong key => authentication failure, never the original data, and the
699        // decrypted-plaintext temp files are cleaned up on the error path.
700        let wrong = {
701            use crate::encryption::Aes256GcmCipher;
702            use zeroize::Zeroizing;
703            let mut k = Zeroizing::new([0u8; 32]);
704            k[7] = 0x5A;
705            Arc::new(Aes256GcmCipher::new(&k)) as Arc<dyn Cipher>
706        };
707        let wrong_ring = super::super::common::IndexKeyring::single(wrong);
708        assert!(load_hnsw_index_with_keyring(&path, config, Some(&wrong_ring)).is_err());
709        assert!(
710            !any_aeix_temp_present(dir.path()),
711            "temp leaked on wrong-key load"
712        );
713    }
