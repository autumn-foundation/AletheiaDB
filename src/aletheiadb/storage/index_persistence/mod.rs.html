1//! Comprehensive index persistence layer for AletheiaDB.
2//!
3//! This module provides persistence for all index types:
4//! - Vector indexes (HNSW via usearch)
5//! - Graph indexes (CSR adjacency)
6//! - Temporal indexes (version chains)
7//! - String interner
8//!
9//! # Architecture
10//!
11//! ```text
12//! indexes/
13//! ├── manifest.idx          # Index registry
14//! ├── strings/interner.idx  # String interning table
15//! ├── graph/adjacency.idx   # CSR adjacency data
16//! ├── temporal/versions.idx # Version chains
17//! └── vector/{prop}/        # Per-property vector indexes
18//! ```
19//!
20//! # Load Order
21//!
22//! 1. String interner (others depend on string indices)
23//! 2. Manifest (tells us what indexes exist)
24//! 3. Graph, Temporal, Vector (parallel)
25//!
26//! # Usage
27//!
28//! ```rust,no_run
29//! use aletheiadb::storage::index_persistence::{
30//!     IndexPersistenceManager, PersistenceConfig, IndexManifest
31//! };
32//!
33//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
34//! // Use a valid path
35//! let manager = IndexPersistenceManager::new("data");
36//! manager.ensure_directories()?;
37//!
38//! // Save
39//! manager.save_string_interner()?;
40//! let manifest = IndexManifest::new(100);
41//! manager.save_manifest(&manifest)?;
42//!
43//! // Load (respects load order)
44//! if manager.indexes_exist() {
45//!     let manifest = manager.load_manifest_and_strings()?;
46//!     // String interner is now restored
47//!     // Ready to load other indexes
48//! }
49//! # Ok(())
50//! # }
51//! ```
52//!
53//! # Format Details
54//!
55//! All index files (except the native HNSW index) use [bitcode](https://github.com/llogiq/bitcode)
56//! serialization with a consistent header format: `[MAGIC:4][VERSION:1][DATA...]`.
57//!
58//! | File Type | Magic Bytes | Rust Struct | Description |
59//! |-----------|-------------|-------------|-------------|
60//! | **Manifest** | `GIDX` | [`formats::IndexManifest`] | Registry of all indexes, LSN tracking, and timestamps. |
61//! | **String Interner** | `GSTR` | [`formats::StringInternerData`] | Ordered list of interned strings for ID restoration. |
62//! | **Graph Index** | `GGRP` | [`formats::GraphIndexData`] | Nodes, edges, CSR adjacency, and current properties. |
63//! | **Graph Delta** | `GDLT` | [`formats::GraphIndexDelta`] | Incremental changes (added/modified/deleted) since base snapshot. |
64//! | **Temporal Index** | `GTMP` | [`formats::TemporalIndexData`] | Version chains, anchors, and deltas for time-travel. |
65//! | **Vector Meta** | `GVEC` | [`formats::VectorIndexMeta`] | Metadata for a vector index (dimensions, metric, etc.). |
66//!
67//! ## Vector Index Hybrid Format
68//!
69//! Vector indexes use a hybrid format for performance:
70//! 1. **Metadata** (`meta.idx`): Bitcode-serialized [`formats::VectorIndexMeta`].
71//! 2. **Mappings** (`mappings.idx`): Bitcode-serialized [`formats::VectorMappingsData`] mapping NodeIDs to usearch keys.
72//! 3. **Index** (`current.usearch`): Native binary format produced by the `usearch` C++ library (HNSW graph).
73//!
74//! # Safety and Integrity
75//!
76//! - **Atomic Writes**: All files are written using a write-temp-then-rename strategy (`atomic_write`) to prevent corruption during crashes.
77//! - **Checksums**: Many formats (Graph, Vector Meta/Mappings) include CRC32 checksums for integrity verification.
78//! - **Magic Bytes**: All files start with 4 magic bytes to prevent parsing invalid file types.
79//! - **Versioning**: All files include a version byte to support future schema evolution.

266/// Atomically write data to a file using write-temp-then-rename pattern.
267///
268/// This prevents corruption if the process crashes mid-write:
269/// 1. Write to `{path}.tmp.{random_suffix}`
270/// 2. Sync to disk
271/// 3. Rename temp → target (atomic on POSIX, nearly-atomic on Windows)
272///
273/// # Thread Safety
274///
275/// Uses a random suffix for the temporary file to allow multiple threads to attempt
276/// atomic writes to the same target concurrently (though last writer wins).
277/// This prevents race conditions where one thread truncates another thread's
278/// temporary file.
279///
280/// # Errors
281///
282/// Returns an error if:
283/// - Failed to write temp file
284/// - Failed to sync to disk
285/// - Failed to rename temp to target
286pub(crate) fn atomic_write(path: &std::path::Path, data: &[u8]) -> Result<()> {
287    use rand::Rng;
288    use std::fs;
289    use std::io::Write;
290
291    // Generate a random suffix to prevent collisions between concurrent writers
292    let suffix: u32 = rand::thread_rng().r#gen();
293    let extension = match path.extension() {
294        Some(ext) => format!("{}.{}.tmp", ext.to_string_lossy(), suffix),
295        None => format!("{}.tmp", suffix),
296    };
297
298    // Write to temporary file with unique name
299    let temp_path = path.with_extension(extension);
300    let mut file = fs::File::create(&temp_path)?;
301    file.write_all(data)?;
302    file.sync_all()?; // Ensure data is on disk
303
304    // Atomically replace target with temp
305    fs::rename(&temp_path, path)?;
306
307    // Durably persist the rename itself: without an fsync of the containing
308    // directory, a crash after the rename can lose the new directory entry even
309    // though the file's data was fsync'd, resurrecting the old target or leaving
310    // no target at all (Issue #488 P0.2). Best-effort: a directory that cannot
311    // be fsync'd (e.g. some network filesystems) must not fail the write.
312    if let Some(parent) = path.parent() {
313        fsync_dir(parent);
314    }
315
316    Ok(())
317}
318
319/// Best-effort fsync of a directory so a preceding `rename`/`create` in it is
320/// durable across a crash. Errors are swallowed: not every filesystem supports
321/// directory fsync, and a failed dir-sync must never fail the enclosing write.
322pub(crate) fn fsync_dir(dir: &std::path::Path) {
323    if let Ok(handle) = std::fs::File::open(dir) {
324        let _ = handle.sync_all();
325    }
326}

328/// Load graph, temporal, and vector indexes in parallel for faster startup.
329///
330/// This function spawns threads to load all three index types concurrently,
331/// reducing startup time for databases with large indexes.
332///
333/// # Arguments
334///
335/// * `graph_path` - Path to the graph index file
336/// * `temporal_path` - Optional path to the temporal index file
337/// * `vector_paths` - Optional vector of vector index paths (meta, mappings, snapshots)
338///
339/// # Returns
340///
341/// A tuple of (graph_data, temporal_data_option, vector_data_vec)
342///
343/// # ⚠️ Interner remap required (Issue #3490)
344///
345/// This function returns **raw, un-remapped** `GraphIndexData` /
346/// `TemporalIndexData`: every embedded `u32` interner id (labels, property
347/// keys, string values, delta `removed_keys`) is still in the saved file's
348/// id space. It does **not** restore the string interner. Any caller that goes
349/// on to resolve those ids MUST first restore the interner (via
350/// [`strings::restore_string_interner`], obtaining an
351/// [`InternerRemap`](strings::InternerRemap)) and then translate the returned
352/// data through [`InternerRemap::remap_graph_index_data`] /
353/// [`InternerRemap::remap_temporal_index_data`] before resolving — otherwise
354/// persisted ids silently resolve to the wrong strings when the live interner
355/// order diverges from the saved file (the #3490 corruption). This is a latent
356/// trap: there are currently no production callers, and the startup path
357/// (`load_indexes_startup`) deliberately loads sequentially with the remap
358/// applied instead of using this helper.
359///
360/// # Errors
361///
362/// Returns an error if any of the index files fail to load.
363///
364/// # Examples
365///
366/// ```ignore
367/// use aletheiadb::storage::index_persistence::load_indexes_parallel;
368///
369/// let (graph, temporal, vector) = load_indexes_parallel(
370///     &graph_path,
371///     Some(&temporal_path),
372///     vec![],
373/// )?;
374/// ```
375pub fn load_indexes_parallel(
376    graph_path: &std::path::Path,
377    temporal_path: Option<&std::path::Path>,
378    vector_paths: Vec<&std::path::Path>,
379) -> Result<(
380    formats::GraphIndexData,
381    Option<formats::TemporalIndexData>,
382    Vec<formats::VectorIndexData>,
383)> {
384    load_indexes_parallel_with_cipher(graph_path, temporal_path, vector_paths, None)
385}

387/// Like [`load_indexes_parallel`], but transparently decrypts any index file
388/// written encrypted at rest (Issue #481). Passing `cipher == None` is
389/// byte-for-byte equivalent to [`load_indexes_parallel`] (header sniffing means
390/// a plaintext file still loads with a cipher present, and a plaintext-only
391/// directory is unaffected). Kept in lockstep with the cipher-aware loaders so
392/// this alternate parallel entry point never fails closed on an encrypted
393/// directory.
394pub fn load_indexes_parallel_with_cipher(
395    graph_path: &std::path::Path,
396    temporal_path: Option<&std::path::Path>,
397    vector_paths: Vec<&std::path::Path>,
398    cipher: Option<&std::sync::Arc<dyn crate::encryption::cipher::Cipher>>,
399) -> Result<(
400    formats::GraphIndexData,
401    Option<formats::TemporalIndexData>,
402    Vec<formats::VectorIndexData>,
403)> {
404    use std::thread;
405
406    // Convert paths to owned PathBufs for thread safety
407    let graph_path = graph_path.to_path_buf();
408    let temporal_path_opt = temporal_path.map(|p| p.to_path_buf());
409    let vector_paths: Vec<_> = vector_paths.into_iter().map(|p| p.to_path_buf()).collect();
410
411    // Clone the cipher Arc into owned Options so each thread/closure can decrypt
412    // independently (a `&Arc` cannot cross the thread boundary).
413    let graph_cipher = cipher.cloned();
414    let temporal_cipher = cipher.cloned();
415    let vector_cipher = cipher.cloned();
416
417    // Spawn thread for graph loading
418    let graph_handle = thread::spawn(move || {
419        graph::load_graph_index_with_cipher(&graph_path, graph_cipher.as_ref())
420    });
421
422    // Spawn thread for temporal loading if path provided
423    let temporal_handle = temporal_path_opt.map(|path| {
424        thread::spawn(move || {
425            temporal::load_temporal_index_with_cipher(&path, temporal_cipher.as_ref())
426        })
427    });
428
429    // Load vector indexes in parallel using Rayon (blocks current thread)
430    // We do this while graph and temporal indexes are loading in background threads
431    let vector_data: Result<Vec<_>> = vector_paths
432        .par_iter()
433        .map(|path| vector::load_vector_index_with_cipher(path, vector_cipher.as_ref()))
434        .collect();
435    let vector_data = vector_data?;
436
437    // Join threads and collect results
438    let graph_data = graph_handle
439        .join()
440        .expect("Graph loading thread panicked")?;
441
442    let temporal_data = if let Some(handle) = temporal_handle {
443        Some(handle.join().expect("Temporal loading thread panicked")?)
444    } else {
445        None
446    };
447
448    Ok((graph_data, temporal_data, vector_data))
449}
