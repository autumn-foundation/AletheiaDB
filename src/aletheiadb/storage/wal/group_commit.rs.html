1//! Group commit coordination for batched WAL flushes.
2//!
3//! This module provides [`GroupCommitCoordinator`], which manages the epoch-based
4//! waiting mechanism for [`GroupCommit`](super::DurabilityMode::GroupCommit) mode.
5//!
6//! # Error Handling Strategy
7//!
8//! All public methods return `Result` types to handle lock poisoning gracefully.
9//! Lock poisoning occurs when a thread panics while holding the coordinator's mutex.
10//!
11//! ## For Callers
12//!
13//! When a `StorageError::LockPoisoned` error is returned:
14//! - **Flush thread**: Should panic immediately. Continuing would leave waiting
15//!   transactions hanging indefinitely. This is an unrecoverable state.
16//! - **Transaction threads**: Should propagate the error to the caller. The
17//!   transaction cannot complete and must be rolled back.
18//!
19//! ## Rationale
20//!
21//! Lock poisoning in the coordinator indicates severe corruption. The alternatives
22//! are:
23//! 1. **Panic everywhere** (too aggressive for transaction threads)
24//! 2. **Continue silently** (leaves transactions hanging - worse than panicking)
25//! 3. **Return Result** (chosen approach - lets callers decide appropriate action)
26//!
27//! The flush thread uses `.expect()` to panic on lock poisoning because silent
28//! degradation is worse than fail-fast behavior for background infrastructure.
29

36/// Coordinates group commit batching and waiting.
37///
38/// The coordinator uses an epoch-based system where:
39/// 1. Each transaction registers and receives the current epoch number
40/// 2. Multiple transactions accumulate in the same epoch
41/// 3. When the batch is flushed, the epoch advances
42/// 4. All waiting transactions for that epoch are notified
43///
44/// This allows ACID durability with amortized fsync cost across many transactions.
45///
46/// # Why?
47///
48/// Group commit is a crucial performance optimization for a database's Write-Ahead Log.
49/// Fsync operations on modern NVMe drives can still take ~1-2ms. If every transaction
50/// synchronously blocks on an fsync, throughput is capped at ~500-1000 tx/sec. By
51/// batching multiple concurrent transactions into a single "epoch" and performing one
52/// fsync for all of them, throughput can scale to hundreds of thousands of tx/sec while
53/// still maintaining strict ACID durability.
54///
55/// # Epoch Model
56///
57/// ```text
58/// Epoch 0: [tx1, tx2, tx3] → flush → advance to epoch 1
59/// Epoch 1: [tx4, tx5] → flush → advance to epoch 2
60/// ...
61/// ```
62///
63/// Each transaction waits on its epoch number. When `mark_flushed()` is called,
64/// all transactions waiting on that epoch are released.
65///
66/// # Error Propagation
67///
68/// If a flush fails, the error is stored and propagated to all waiting transactions.
69/// This ensures no transaction incorrectly believes its data is durable.
70///
71/// # Examples
72///
73/// ```
74/// use aletheiadb::storage::wal::group_commit::GroupCommitCoordinator;
75/// use std::sync::Arc;
76/// use std::thread;
77///
78/// let coordinator = Arc::new(GroupCommitCoordinator::new(10, 100));
79///
80/// // Transaction 1
81/// let (epoch1, _) = coordinator.register_transaction().unwrap();
82///
83/// // Background Flush Thread
84/// let flush_coord = Arc::clone(&coordinator);
85/// thread::spawn(move || {
86///     let epoch = flush_coord.start_flush().unwrap();
87///     // Flush writes to disk here...
88///     // Then notify waiting transactions
89///     flush_coord.finish_flush(epoch, Ok(())).unwrap();
90/// });
91///
92/// // Transaction 1 waits for flush
93/// coordinator.wait_for_flush(epoch1).unwrap();
94/// ```
95pub struct GroupCommitCoordinator {
96    /// State protected by mutex
97    state: Mutex<GroupCommitState>,
98    /// Condition variable for flush completion
99    flush_complete: Condvar,
100    /// Configuration
101    config: GroupCommitConfig,
102}

104/// Configuration for group commit behavior.
105#[derive(Debug, Clone)]
106pub struct GroupCommitConfig {
107    /// Maximum time to wait for more transactions before flushing.
108    pub max_delay_ms: u64,
109    /// Maximum transactions to batch before forcing a flush.
110    pub max_batch_size: usize,
111
112    /// Multiplier for max_delay_ms to calculate base timeout.

120    /// Maximum number of recent errors to keep in history.
121    pub recent_errors_capacity: usize,
122}

138struct GroupCommitState {
139    /// Current epoch (increments on each flush)
140    current_epoch: u64,
141    /// Number of transactions in current batch
142    batch_count: usize,
143    /// Epoch that has been durably flushed
144    flushed_epoch: u64,
145    /// Recent flush errors, stored as (epoch, error_message).
146    /// Used to verify that a specific epoch was successfully flushed.
147    recent_errors: VecDeque<(u64, String)>,
148    /// The oldest epoch in the recent_errors list (or older if evicted).
149    /// Used to detect if we've lost history.
150    oldest_error_epoch: u64,
151    /// Set of completed epochs that are ahead of flushed_epoch
152    completed_epochs: BTreeSet<u64>,

223    /// Register a transaction for group commit.
224    ///
225    /// Returns the epoch number that this transaction should wait for.
226    /// If the batch is full, returns `true` in the second element to signal
227    /// that an immediate flush should be triggered.
228    ///
229    /// # Returns
230    ///
231    /// A tuple of (epoch_to_wait_for, should_trigger_flush)
232    ///
233    /// # Errors
234    ///
235    /// Returns `StorageError::LockPoisoned` if the coordinator lock is poisoned.
236    pub fn register_transaction(&self) -> Result<(u64, bool), Error> {
237        let mut state = self.state.lock().map_err(|_| {
238            Error::Storage(StorageError::LockPoisoned {
239                resource: "group_commit_state".to_string(),
240            })
241        })?;
242
243        state.batch_count += 1;
244        let epoch = state.current_epoch;
245
246        // Check if we should trigger immediate flush (batch full)
247        let should_flush = state.batch_count >= self.config.max_batch_size;
248
249        Ok((epoch, should_flush))
250    }

252    /// Wait for the specified epoch to be flushed.
253    ///
254    /// This blocks until the epoch has been durably flushed. If the flush
255    /// failed, the error is propagated to all waiting transactions.
256    ///
257    /// # Timeout Calculation
258    ///
259    /// The timeout is a **deadlock detection mechanism**, not a performance target.
260    /// It's designed to catch stuck flush threads, not to enforce timing.
261    ///
262    /// Formula: `clamp(max_delay_ms * multiplier + base, min, max)`
263    /// - multiplier: Allows for thread scheduling overhead
264    /// - base: Fixed overhead for thread startup
265    /// - min: Handles very fast configs in slow CI
266    /// - max: Prevents indefinite waiting on stuck threads
267    ///
268    /// The timeout accounts for:
269    /// - Environments with unpredictable thread scheduling (e.g., CI runners)
270    /// - Systems under heavy load causing thread starvation
271    /// - Variable I/O latency for disk flushes
272    ///
273    /// # Errors
274    ///
275    /// Returns an error if:
276    /// - The flush for this epoch failed
277    /// - The wait times out (indicates a stuck flush thread or excessive system load)
278    ///
279    /// # Examples
280    ///
281    /// ```rust,no_run
282    /// # use aletheiadb::storage::wal::group_commit::GroupCommitCoordinator;
283    /// # let coordinator = GroupCommitCoordinator::new(10, 100);
284    /// # let (epoch, _) = coordinator.register_transaction().unwrap();
285    /// coordinator.wait_for_flush(epoch).unwrap();
286    /// ```
287    pub fn wait_for_flush(&self, epoch: u64) -> Result<(), Error> {
288        let mut state = self.state.lock().map_err(|_| {
289            Error::Storage(StorageError::LockPoisoned {
290                resource: "group_commit_state".to_string(),
291            })
292        })?;
293
294        // Deadlock detection timeout (NOT a performance SLA)
295        let base_timeout =
296            Duration::from_millis(self.config.max_delay_ms * self.config.timeout_multiplier as u64)
297                + Duration::from_millis(self.config.timeout_base_ms);
298        let timeout = base_timeout
299            .max(Duration::from_millis(self.config.timeout_min_ms))
300            .min(Duration::from_millis(self.config.timeout_max_ms));
301
302        // Use a deadline to prevent spurious wakeups from resetting the timeout clock
303        let deadline = std::time::Instant::now() + timeout;
304
305        // RACE CONDITION SAFETY: If the epoch was already flushed between register_transaction()
306        // and this wait (rare but possible on fast systems), this loop exits immediately since
307        // flushed_epoch >= epoch, and we return Ok(()) without waiting.
308        //
309        // EPOCH SEMANTICS: flushed_epoch = N means "epoch N has been flushed".
310        // Transaction at epoch E waits while flushed_epoch < E (i.e., E has not been flushed yet).
311        while state.flushed_epoch < epoch {
312            let now = std::time::Instant::now();
313            let remaining = if now >= deadline {
314                Duration::from_secs(0)
315            } else {
316                deadline - now
317            };
318
319            if remaining.as_nanos() == 0 {
320                return Err(Error::Storage(StorageError::WalError {
321                    reason: format!(
322                        "Group commit timeout waiting for epoch {} (current flushed: {})",
323                        epoch, state.flushed_epoch
324                    ),
325                }));
326            }
327
328            let (new_state, timeout_result) = self
329                .flush_complete
330                .wait_timeout(state, remaining)
331                .map_err(|_| {
332                    Error::Storage(StorageError::LockPoisoned {
333                        resource: "group_commit_state".to_string(),
334                    })
335                })?;
336
337            state = new_state;
338
339            // Check if we timed out (either by Condvar result OR by deadline)
340            if (timeout_result.timed_out() || std::time::Instant::now() >= deadline)
341                && state.flushed_epoch < epoch
342            {
343                return Err(Error::Storage(StorageError::WalError {
344                    reason: format!(
345                        "Group commit timeout waiting for epoch {} (current flushed: {})",
346                        epoch, state.flushed_epoch
347                    ),
348                }));
349            }
350        }
351
352        // Check for flush errors specifically for this epoch
353        for (failed_epoch, error_msg) in &state.recent_errors {
354            if *failed_epoch == epoch {
355                return Err(Error::Storage(StorageError::WalError {
356                    reason: format!("Group commit flush failed: {}", error_msg),
357                }));
358            }
359        }
360
361        // Check for history eviction (False Success protection)
362        // We only lose certainty about an epoch's status if its error record was EVICTED.
363        // state.oldest_error_epoch tracks the threshold of lost history.
364        // If epoch < state.oldest_error_epoch, it might have failed and been forgotten.
365        //
366        // Note: We do NOT check against recent_errors.front() because a sparse error list
367        // is valid. If epoch 1 succeeded and epoch 2 failed, recent_errors = [(2, ...)].
368        // epoch 1 < 2, but it shouldn't fail unless 1 < oldest_error_epoch.
369
370        if epoch < state.oldest_error_epoch {
371            return Err(Error::Storage(StorageError::WalError {
372                reason: format!(
373                    "Group commit status unknown: epoch {} evicted from error history (history starts at {})",
374                    epoch, state.oldest_error_epoch
375                ),
376            }));
377        }
378
379        Ok(())
380    }

382    /// Start a flush operation.
383    ///
384    /// This method MUST be called before the actual flush begins. It advances
385    /// the `current_epoch` so that any new transactions registering during the
386    /// flush operation are assigned to the *next* epoch, preventing race conditions
387    /// where late-arriving transactions are marked as flushed without actually being written.
388    ///
389    /// # Returns
390    ///
391    /// The epoch number that is being flushed.
392    pub fn start_flush(&self) -> Result<u64, Error> {
393        let mut state = self.state.lock().map_err(|_| {
394            Error::Storage(StorageError::LockPoisoned {
395                resource: "group_commit_state".to_string(),
396            })
397        })?;
398
399        let epoch_to_flush = state.current_epoch;
400
401        // Advance current epoch immediately so new transactions go to the next one
402        state.current_epoch += 1;
403        state.batch_count = 0;
404
405        Ok(epoch_to_flush)
406    }

408    /// Finish a flush operation and notify waiters.
409    ///
410    /// Called by the flush thread after completing a flush.
411    ///
412    /// # Arguments
413    ///
414    /// * `epoch` - The epoch that was flushed (returned by `start_flush`).
415    /// * `result` - The result of the flush operation.
416    pub fn finish_flush(&self, epoch: u64, result: Result<(), Error>) -> Result<(), Error> {
417        let mut state = self.state.lock().map_err(|_| {
418            Error::Storage(StorageError::LockPoisoned {
419                resource: "group_commit_state".to_string(),
420            })
421        })?;
422
423        // Store error if any
424        if let Err(e) = result {
425            state.recent_errors.push_back((epoch, e.to_string()));
426
427            // Keep history limited
428            while state.recent_errors.len() > self.config.recent_errors_capacity {
429                if let Some((evicted_epoch, _)) = state.recent_errors.pop_front() {
430                    // Track the newest evicted epoch to know what we've lost
431                    // We set oldest_error_epoch to evicted_epoch + 1 because if
432                    // evicted_epoch is gone, any check for it (or older) is invalid.
433                    state.oldest_error_epoch = evicted_epoch + 1;
434                }
435            }
436        }
437
438        // Mark this epoch as completed
439        if epoch > state.flushed_epoch {
440            state.completed_epochs.insert(epoch);
441        }
442
443        // Advance flushed_epoch contiguously to wake up waiters
444        // We only advance if we have a contiguous sequence of completed epochs.
445        // This prevents data loss scenarios where a later successful flush
446        // could mask an earlier failed (or still pending) flush.
447        let mut next_epoch = state.flushed_epoch + 1;
448        while state.completed_epochs.contains(&next_epoch) {
449            state.completed_epochs.remove(&next_epoch);
450            state.flushed_epoch = next_epoch;
451            next_epoch += 1;
452        }
453
454        // Wake all waiting transactions
455        self.flush_complete.notify_all();
456
457        Ok(())
458    }

460    /// Mark the current batch as flushed (Legacy/Combined Helper).
461    ///
462    /// WARNING: This method combines `start_flush` and `finish_flush` but IS NOT SAFE
463    /// against the race condition described in security review (transactions registering
464    /// during the flush).
465    ///
466    /// It is kept for backward compatibility with existing tests but `start_flush`
467    /// and `finish_flush` should be used in production.
468    pub fn mark_flushed(&self, result: Result<(), Error>) -> Result<(), Error> {
469        let epoch = self.start_flush()?;
470        self.finish_flush(epoch, result)
471    }

904    #[test]
905    fn test_flush_race_condition() {
906        let coord = GroupCommitCoordinator::new(100, 100);
907
908        // 1. Transaction A registers (Epoch 1)
909        let (epoch_a, _) = coord.register_transaction().unwrap();
910        assert_eq!(epoch_a, 1);
911
912        // 2. Flush starts (Epoch 1 is being flushed)
913        // This advances current_epoch to 2
914        let flushing_epoch = coord.start_flush().unwrap();
915        assert_eq!(flushing_epoch, 1);
916        assert_eq!(coord.current_epoch().unwrap(), 2);
917
918        // 3. Transaction B registers (Should be Epoch 2)
919        // Because start_flush advanced the epoch, B is not part of the current flush
920        let (epoch_b, _) = coord.register_transaction().unwrap();
921        assert_eq!(epoch_b, 2);
922
923        // 4. Flush finishes successfully
924        coord.finish_flush(flushing_epoch, Ok(())).unwrap();
925
926        // 5. Transaction A should be done
927        assert!(coord.wait_for_flush(epoch_a).is_ok());
928
929        // 6. Transaction B should NOT be done (it needs Epoch 2 flush)
930        // We expect it to timeout if we wait, but we can just check flushed_epoch
931        assert_eq!(coord.flushed_epoch().unwrap(), 1);
932
933        // 7. Flush Epoch 2
934        let flushing_epoch_2 = coord.start_flush().unwrap();
935        assert_eq!(flushing_epoch_2, 2);
936        coord.finish_flush(flushing_epoch_2, Ok(())).unwrap();
937
938        // 8. Transaction B should be done
939        assert!(coord.wait_for_flush(epoch_b).is_ok());
940    }

947            max_batch_size: 100,
948            timeout_multiplier: 1, // 10ms
949            timeout_base_ms: 10,   // + 10ms = 20ms

951            timeout_max_ms: 200,   // clamp max -> 200ms
952            recent_errors_capacity: 1024,
953        };
954
955        let coord = Arc::new(GroupCommitCoordinator::with_config(config));
956        let coord_clone = Arc::clone(&coord);
957
958        // Register a transaction for Epoch 1
959        let (epoch, _) = coord.register_transaction().unwrap();
960
961        // Spawn a thread that keeps triggering "spurious" wakeups every 10ms
962        thread::spawn(move || {
963            let start = std::time::Instant::now();
964            // Run for 500ms (10x the timeout)
965            while start.elapsed() < Duration::from_millis(500) {
966                thread::sleep(Duration::from_millis(10));
967                // Finish a future epoch (100) triggers notify_all but doesn't advance flushed_epoch
968                let _ = coord_clone.finish_flush(100, Ok(()));
969            }
970        });
971
972        let start = std::time::Instant::now();
973        let result = coord.wait_for_flush(epoch);
974        let elapsed = start.elapsed();
975
976        // Should fail fast (~50ms) with timeout error
977        assert!(result.is_err());
978        assert!(result.unwrap_err().to_string().contains("timeout"));
979
980        // If bug exists, it will wait ~500ms (or whatever the thread runs for)
981        // If fix works, it will timeout around 50ms.
982        // We set threshold at 150ms to be safe for CI.
983        assert!(
984            elapsed < Duration::from_millis(150),
985            "Wait took {:?}, expected < 150ms",
986            elapsed
987        );
