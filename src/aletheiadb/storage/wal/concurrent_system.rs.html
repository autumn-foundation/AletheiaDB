1//! Unified Concurrent WAL System.
2//!
3//! This module provides [`ConcurrentWalSystem`], which combines the concurrent
4//! WAL striped architecture with the flush coordinator into a single, cohesive
5//! component that can be used as a drop-in replacement for the old `WriteAheadLog`.
6//!
7//! # Architecture
8//!
9//! ```text
10//! ┌─────────────────────────────────────────────────────────────────┐
11//! │                    ConcurrentWalSystem                           │
12//! │                                                                  │
13//! │  ┌──────────────────────┐    ┌─────────────────────────────┐   │
14//! │  │    ConcurrentWal     │    │     FlushCoordinator        │   │
15//! │  │  (Striped Buffers)   │───▶│   (Segment Management)      │   │
16//! │  └──────────────────────┘    └─────────────────────────────┘   │
17//! │                                                                  │
18//! │  ┌──────────────────────────────────────────────────────────┐  │
19//! │  │                   Background Flush Thread                  │  │
20//! │  │  - Drains stripes periodically                            │  │
21//! │  │  - Writes to segment files                                │  │
22//! │  │  - Notifies completion handles                            │  │
23//! │  └──────────────────────────────────────────────────────────┘  │
24//! └─────────────────────────────────────────────────────────────────┘
25//! ```
26//!
27//! # Usage
28//!
29//! ```ignore
30//! use aletheiadb::storage::wal::concurrent_system::{ConcurrentWalSystem, ConcurrentWalSystemConfig};
31//!
32//! let config = ConcurrentWalSystemConfig::new("data/wal");
33//! let wal = ConcurrentWalSystem::new(config)?;
34//!
35//! // Async append (returns immediately)
36//! let lsn = wal.append_async(operation)?;
37//!
38//! // Sync append (waits for durability)
39//! let lsn = wal.append_sync(operation)?;
40//!
41//! // Shutdown gracefully
42//! wal.shutdown();
43//! ```

170/// Signal for waking up the flush thread when batch is full.
171struct FlushNotifier {
172    /// Lock for condvar.
173    lock: Mutex<bool>,
174    /// Condvar to signal immediate flush.
175    condvar: Condvar,
176}
177
178impl FlushNotifier {
179    fn new() -> Self {
180        Self {
181            lock: Mutex::new(false),
182            condvar: Condvar::new(),
183        }
184    }
185
186    /// Signal the flush thread to wake up immediately.
187    fn notify(&self) {
188        let mut guard = self.lock.lock().unwrap_or_else(|e| e.into_inner());
189        *guard = true;
190        self.condvar.notify_one();
191    }
192
193    /// Wait for a signal or timeout, returns true if signaled.
194    fn wait_timeout(&self, duration: Duration) -> bool {
195        let mut guard = self.lock.lock().unwrap_or_else(|e| e.into_inner());
196
197        // Check if already signaled before waiting.
198        // This handles the race where notify() is called before we enter wait_timeout().
199        if *guard {
200            *guard = false; // Reset signal
201            return true;
202        }
203
204        let (new_guard, result) = self
205            .condvar
206            .wait_timeout(guard, duration)
207            .unwrap_or_else(|e| e.into_inner());
208        guard = new_guard;
209
210        let was_signaled = *guard && !result.timed_out();
211        *guard = false; // Reset signal
212        was_signaled
213    }
214}

216/// Threshold for consecutive flush errors before logging a critical warning.
217const FLUSH_ERROR_WARNING_THRESHOLD: u64 = 3;

219/// Helper struct to encapsulate background flush logic.
220struct BackgroundFlusher {
221    wal: Arc<ConcurrentWal>,
222    coordinator: Arc<FlushCoordinator>,
223    shutdown: Arc<AtomicBool>,
224    flush_notifier: Arc<FlushNotifier>,
225    group_commit: Option<Arc<GroupCommitCoordinator>>,
226    error_counter: Arc<AtomicU64>,
227    interval: Duration,
228    sync_on_flush: bool,
229}
230
231impl BackgroundFlusher {
232    fn run(&self) {
233        while !self.shutdown.load(Ordering::Relaxed) {
234            self.perform_flush_cycle();
235            // Wait for flush interval OR immediate signal (batch full)
236            self.flush_notifier.wait_timeout(self.interval);
237        }
238        self.perform_final_flush();
239    }
240
241    fn perform_flush_cycle(&self) {
242        let entries = self.wal.drain_all();
243
244        // Always try to advance the epoch when there are entries OR when
245        // group commit has pending transactions.
246        //
247        // LOCK POISONING: If current_batch_size() fails, the coordinator lock is
248        // poisoned and the system is in an unrecoverable state. Panicking is correct
249        // here - continuing would leave waiting transactions hanging indefinitely.
250        let should_mark_flushed = !entries.is_empty()
251            || self.group_commit.as_ref().is_some_and(|gc| {
252                gc.current_batch_size()
253                    .expect("GroupCommitCoordinator lock poisoned - flush thread cannot continue")
254                    > 0
255            });
256
257        if !entries.is_empty() {
258            // Flush to coordinator
259            let result = self.coordinator.flush(entries, self.sync_on_flush);
260            self.handle_flush_result(result.map(|_| ()));
261        } else if should_mark_flushed {
262            // No entries but there are pending transactions - advance epoch anyway
263            self.handle_flush_result(Ok(()));
264        }
265    }
266
267    fn perform_final_flush(&self) {
268        let entries = self.wal.drain_all();
269        if !entries.is_empty() {
270            let result = self.coordinator.flush(entries, true);
271            self.handle_flush_result(result.map(|_| ()));
272        }
273    }
274
275    fn handle_flush_result(&self, result: Result<()>) {
276        match result {
277            Ok(_) => {
278                // Reset error counter on success
279                self.error_counter.store(0, Ordering::Relaxed);
280                if let Some(ref gc) = self.group_commit {
281                    gc.mark_flushed(Ok(())).expect(
282                        "GroupCommitCoordinator lock poisoned - flush thread cannot continue",
283                    );
284                }
285            }
286            Err(e) => {
287                // Track consecutive errors for health monitoring
288                let errors = self.error_counter.fetch_add(1, Ordering::Relaxed) + 1;
289                if errors == FLUSH_ERROR_WARNING_THRESHOLD {
290                    eprintln!(
291                        "CRITICAL: WAL flush failed {} consecutive times. \
292                         Data durability may be compromised. Last error: {}",
293                        errors, e
294                    );
295                } else {
296                    eprintln!("WAL flush error: {}", e);
297                }
298
299                if let Some(ref gc) = self.group_commit {
300                    // Create a new error from the string representation
301                    gc.mark_flushed(Err(crate::core::error::Error::other(e.to_string())))
302                        .expect(
303                            "GroupCommitCoordinator lock poisoned - flush thread cannot continue",
304                        );
305                }
306            }
307        }
308    }
309}

311/// Unified concurrent WAL system.
312///
313/// This combines the striped concurrent WAL with the flush coordinator
314/// and a background flush thread to provide a complete WAL solution.
315pub struct ConcurrentWalSystem {
316    /// The concurrent WAL with striped buffers.
317    wal: Arc<ConcurrentWal>,
318    /// The flush coordinator for segment management.
319    coordinator: Arc<FlushCoordinator>,
320    /// Handle to the background flush thread.
321    flush_thread: Option<JoinHandle<()>>,
322    /// Signal to stop the flush thread.
323    shutdown_signal: Arc<AtomicBool>,
324    /// Signal to wake up flush thread immediately (batch full).
325    flush_notifier: Arc<FlushNotifier>,
326    /// Durability mode.
327    durability_mode: DurabilityMode,
328    /// Group commit coordinator for epoch-based waiting (GroupCommit mode only).
329    group_commit: Option<Arc<GroupCommitCoordinator>>,
330    /// Counter for consecutive flush errors (for health monitoring).
331    consecutive_flush_errors: Arc<AtomicU64>,
332    /// Crash-torn-tail recovery policy applied by [`Self::read_from`]
333    /// (Issue #3433).
334    tolerate_torn_tail: bool,
335    /// Optional WAL DEK keyring for encryption at rest, held in a runtime-swappable
336    /// presence cell (Issues #3617, #3616). Retained so [`Self::read_from`] can
337    /// decrypt encrypted segments during recovery replay — dispatching per segment
338    /// on its `key_version` so a mixed old-DEK/new-DEK directory (an in-flight
339    /// full-MEK rotation) replays correctly.
340    ///
341    /// The cell is a single `Arc<ArcSwapOption<..>>` **shared** (Arc-cloned) with
342    /// the flush coordinator's config, so a runtime install (`None` → `Some`, Issue
343    /// #3616 PR2) or a rotation's inner-generation advance (Issue #3617) is observed
344    /// by both the write path (flush) and the recovery/`is_encrypted` path at once.
345    /// Off-lock readers (`read_from`, `is_encrypted`, `wal_keyring`) `load()` it
346    /// lock-free; the write path's reads stay serialized under the coordinator
347    /// `writer` mutex, and the install stores into it inside the seal→reopen
348    /// hand-off (see [`Self::install_wal_keyring`]).
349    wal_keyring: Arc<ArcSwapOption<crate::encryption::wal_encryption::WalKeyring>>,
350    /// Serializes runtime keyring installs (Issue #3616 PR2) so two concurrent
351    /// installers cannot both observe the cell as `None`, both pass the presence
352    /// check, and both run seal->store->reopen -- the second silently replacing
353    /// the first keyring and producing an undecryptable segment. This mutex is
354    /// held for the entire [`Self::install_wal_keyring`] body (presence check +
355    /// seal + store + reopen) so a second concurrent installer blocks, then
356    /// observes `Some`, and returns the existing rejection `Err`.
357    ///
358    /// Why a dedicated field: it is a private LEAF, taken only at the very top of
359    /// install. The hot path (`append`/`flush`) never touches it, so it adds zero
360    /// steady-state overhead; and because install itself never calls into
361    /// `historical`/`current_timestamp`/cold, it is never held across any lock
362    /// ordered after `wal` -- it merely wraps the presence check and the
363    /// coordinator-`writer`-guarded seal->store->reopen hand-off.
364    install_lock: Mutex<()>,
365}

367impl ConcurrentWalSystem {
368    /// Create a new concurrent WAL system.
369    pub fn new(config: ConcurrentWalSystemConfig) -> Result<Self> {
370        // Create ConcurrentWal config
371        let wal_config = ConcurrentWalConfig {
372            wal_dir: config.wal_dir.clone(),
373            num_stripes: config.num_stripes,
374            stripe_capacity: config.stripe_capacity,
375            segment_size: config.segment_size,
376            segments_to_retain: config.segments_to_retain,
377        };
378
379        // Build the WAL DEK keyring from the configured cipher (Issue #3617). A
380        // never-rotated encrypted WAL starts as a single-generation keyring
381        // (stamps INITIAL_WAL_KEY_VERSION, decrypts any segment); a full-MEK
382        // rotation later advances it to a second generation. Plaintext WALs have
383        // no keyring.
384        //
385        // Issue #488 version-provisioning: when the durable `open()` path
386        // resolves a provisioned key version from durable on-disk state, build
387        // the keyring at that version (still `match_any`, so mixed/legacy
388        // segments decrypt exactly as before) so new segments stamp the real
389        // version and `current_version` reports it. `None` reproduces prior
390        // behavior exactly.
391        let initial_keyring = config.wal_cipher.clone().map(|cipher| {
392            use crate::encryption::wal_encryption::WalKeyring;
393            match config.wal_key_version {
394                Some(v) => WalKeyring::single_versioned(cipher, v),
395                None => WalKeyring::single(cipher),
396            }
397        });
398        // One shared presence cell, cloned (Arc-cloned, same underlying cell) into
399        // both the system and the coordinator so a runtime install or rotation is
400        // observed by both the flush and recovery paths (Issues #3616, #3617).
401        let wal_keyring = Arc::new(ArcSwapOption::from(initial_keyring.map(Arc::new)));
402
403        // Create FlushCoordinator config
404        let coordinator_config = FlushCoordinatorConfig {
405            wal_dir: config.wal_dir,
406            segment_size: config.segment_size,
407            segments_to_retain: config.segments_to_retain,
408            flush_interval_ms: config.flush_interval_ms,
409            sync_on_flush: matches!(
410                config.durability_mode,
411                DurabilityMode::Synchronous | DurabilityMode::GroupCommit { .. }
412            ),
413            write_buffer_size: config.write_buffer_size,
414            wal_keyring: Arc::clone(&wal_keyring),
415        };
416
417        let wal = Arc::new(ConcurrentWal::new(wal_config)?);
418        let coordinator = Arc::new(FlushCoordinator::new(coordinator_config)?);
419        let shutdown_signal = Arc::new(AtomicBool::new(false));
420
421        // Create group commit coordinator for modes that need epoch tracking
422        let group_commit = match config.durability_mode {
423            DurabilityMode::GroupCommit {
424                max_batch_size,
425                max_delay_ms,
426            } => Some(Arc::new(GroupCommitCoordinator::new(
427                max_delay_ms,
428                max_batch_size,
429            ))),
430            DurabilityMode::AsyncBatched {
431                max_batch_size,
432                max_delay_ms,
433                ..
434            } => Some(Arc::new(GroupCommitCoordinator::new(
435                max_delay_ms,
436                max_batch_size,
437            ))),
438            _ => None,
439        };
440
441        // Create flush notifier for batch-size-triggered flushes
442        let flush_notifier = Arc::new(FlushNotifier::new());
443
444        // Create error counter for health monitoring
445        let consecutive_flush_errors = Arc::new(AtomicU64::new(0));
446
447        // Start background flush thread for async/group-commit modes
448        let flush_thread = if matches!(
449            config.durability_mode,
450            DurabilityMode::Async { .. }
451                | DurabilityMode::GroupCommit { .. }
452                | DurabilityMode::AsyncBatched { .. }
453        ) {
454            let wal_clone = Arc::clone(&wal);
455            let coordinator_clone = Arc::clone(&coordinator);
456            let shutdown_clone = Arc::clone(&shutdown_signal);
457            let flush_notifier_clone = Arc::clone(&flush_notifier);
458            let group_commit_clone = group_commit.clone();
459            let error_counter_clone = Arc::clone(&consecutive_flush_errors);
460            let flush_interval = Duration::from_millis(config.flush_interval_ms);
461            let sync_on_flush =
462                matches!(config.durability_mode, DurabilityMode::GroupCommit { .. });
463
464            Some(thread::spawn(move || {
465                Self::flush_loop(
466                    wal_clone,
467                    coordinator_clone,
468                    shutdown_clone,
469                    flush_notifier_clone,
470                    group_commit_clone,
471                    error_counter_clone,
472                    flush_interval,
473                    sync_on_flush,
474                );
475            }))
476        } else {
477            None
478        };
479
480        Ok(Self {
481            wal,
482            coordinator,
483            flush_thread,
484            shutdown_signal,
485            flush_notifier,
486            durability_mode: config.durability_mode,
487            group_commit,
488            consecutive_flush_errors,
489            tolerate_torn_tail: config.tolerate_torn_tail,
490            wal_keyring,
491            install_lock: Mutex::new(()),
492        })
493    }
494
495    /// Background flush loop.
496    ///
497    /// Wakes up either when:
498    /// - The flush interval expires (normal periodic flush)
499    /// - The flush_notifier is signaled (batch size reached)
500    /// - Shutdown is requested
501    #[allow(clippy::too_many_arguments)]
502    fn flush_loop(
503        wal: Arc<ConcurrentWal>,
504        coordinator: Arc<FlushCoordinator>,
505        shutdown: Arc<AtomicBool>,
506        flush_notifier: Arc<FlushNotifier>,
507        group_commit: Option<Arc<GroupCommitCoordinator>>,
508        error_counter: Arc<AtomicU64>,
509        interval: Duration,
510        sync_on_flush: bool,
511    ) {
512        let flusher = BackgroundFlusher {
513            wal,
514            coordinator,
515            shutdown,
516            flush_notifier,
517            group_commit,
518            error_counter,
519            interval,
520            sync_on_flush,
521        };
522        flusher.run();
523    }

525    /// Append an operation asynchronously (fire and forget).
526    ///
527    /// For `DurabilityMode::Synchronous`, this blocks until durable.
528    /// For other modes, this returns immediately.
529    pub fn append(&self, operation: WalOperation) -> Result<LSN> {
530        match self.durability_mode {
531            DurabilityMode::Synchronous => self.append_sync(operation),
532            DurabilityMode::Async { .. }
533            | DurabilityMode::GroupCommit { .. }
534            | DurabilityMode::AsyncBatched { .. } => self.append_async(operation),
535        }
536    }

540    /// The entry is buffered and will be flushed by the background thread.
541    pub fn append_async(&self, operation: WalOperation) -> Result<LSN> {

545    /// Append a batch of operations to the buffer without flushing (returns immediately).
546    ///
547    /// This is the batch counterpart of [`append_async`](Self::append_async): it buffers
548    /// every operation under a single atomic LSN allocation (the Issue #219 win) and lets
549    /// the configured durability path flush them later. Like `append_async`, it performs
550    /// **no** durability-mode branching — callers that need a flush still call
551    /// [`commit`](Self::commit) afterwards, which honors the active `DurabilityMode`. This
552    /// keeps the transaction commit path's behavior identical across Sync/Async/GroupCommit
553    /// while routing multi-operation transactions through the efficient batch append.
554    pub fn append_batch_async(&self, operations: Vec<WalOperation>) -> Result<Vec<LSN>> {
555        self.wal.append_batch(operations)
556    }

558    /// Append an operation synchronously (waits for durability).
559    ///
560    /// This flushes immediately and waits for fsync.
561    pub fn append_sync(&self, operation: WalOperation) -> Result<LSN> {
562        let (lsn, handle) = self.wal.append_with_handle(operation)?;
563
564        // Drain and flush immediately for sync mode
565        let entries = self.wal.drain_all();
566        if !entries.is_empty() {
567            self.coordinator.flush(entries, true)?;
568        }
569
570        // Wait for durability
571        handle.wait().map_err(|e| {
572            Error::Storage(StorageError::WalError {
573                reason: format!("WAL flush failed: {}", e),
574            })
575        })?;
576
577        Ok(lsn)
578    }

580    /// Append a batch of operations efficiently.
581    ///
582    /// This method provides significant performance improvements for high-throughput
583    /// workloads by batching multiple operations into fewer I/O operations.
584    ///
585    /// # Performance Benefits
586    ///
587    /// Compared to calling `append()` multiple times:
588    /// - Single atomic LSN allocation for all operations (vs N atomic operations)
589    /// - Better CPU cache locality during serialization
590    /// - Reduced stripe buffer contention
591    ///
592    /// # Durability Behavior
593    ///
594    /// The durability semantics follow the configured `DurabilityMode`:
595    /// - **Synchronous**: All operations are flushed and synced before returning
596    /// - **Async**: Operations are buffered and flushed by background thread (eventual consistency)
597    /// - **GroupCommit**: Operations are buffered and flushed by background thread (caller must wait on epoch)
598    /// - **AsyncBatched**: Same as GroupCommit (operations buffered, background flush)
599    ///
600    /// # Arguments
601    ///
602    /// * `operations` - Vector of operations to append
603    ///
604    /// # Returns
605    ///
606    /// Vector of allocated LSNs in the same order as the operations.
607    /// Returns an empty vector if `operations` is empty.
608    ///
609    /// # Example
610    ///
611    /// ```ignore
612    /// use aletheiadb::storage::wal::{WalOperation, ConcurrentWalSystem};
613    ///
614    /// let ops = vec![
615    ///     WalOperation::CreateNode { /* ... */ },
616    ///     WalOperation::CreateEdge { /* ... */ },
617    ///     WalOperation::UpdateNode { /* ... */ },
618    /// ];
619    ///
620    /// // Efficient batch append
621    /// let lsns = wal.append_batch(ops)?;
622    /// assert_eq!(lsns.len(), 3);
623    ///
624    /// // For GroupCommit mode, commit and wait
625    /// if let Some(epoch) = wal.commit()? {
626    ///     wal.group_commit_coordinator().unwrap().wait_for_flush(epoch)?;
627    /// }
628    /// ```
629    pub fn append_batch(&self, operations: Vec<WalOperation>) -> Result<Vec<LSN>> {
630        // Handle empty batch early
631        if operations.is_empty() {
632            return Ok(Vec::new());
633        }
634
635        // Use the underlying WAL's batch append for async modes
636        match self.durability_mode {
637            DurabilityMode::Synchronous => {
638                // For synchronous mode, append batch then flush all
639                let (lsns, handles) = self.wal.append_batch_with_handles(operations)?;
640
641                // Drain and flush immediately for sync mode
642                let entries = self.wal.drain_all();
643                if !entries.is_empty() {
644                    self.coordinator.flush(entries, true).map_err(|e| {
645                        Error::Storage(StorageError::WalError {
646                            reason: format!("Failed to flush batch after drain: {}", e),
647                        })
648                    })?;
649                }
650
651                // Wait for all handles to ensure durability.
652                // Note: Since flush coordinator preserves LSN order, waiting for the last one
653                // technically implies all previous ones are done, but waiting for all is safer
654                // against future changes and handles errors correctly.
655                if let Some(last_handle) = handles.into_iter().last() {
656                    last_handle.wait().map_err(|e| {
657                        Error::Storage(StorageError::WalError {
658                            reason: format!("WAL flush failed: {}", e),
659                        })
660                    })?;
661                }
662
663                Ok(lsns)
664            }
665            DurabilityMode::Async { .. }
666            | DurabilityMode::GroupCommit { .. }
667            | DurabilityMode::AsyncBatched { .. } => {
668                // For async modes, just batch append (background thread handles flush)
669                self.wal.append_batch(operations)
670            }
671        }
672    }

674    /// Force a flush of all pending entries.
675    pub fn flush(&self) -> Result<FlushStats> {
676        let entries = self.wal.drain_all();
677        if entries.is_empty() {
678            return Ok(FlushStats::default());
679        }
680
681        let should_sync = !matches!(self.durability_mode, DurabilityMode::Async { .. });
682        self.coordinator.flush(entries, should_sync)
683    }

685    /// Commit with the configured durability mode.
686    ///
687    /// # Usage
688    ///
689    /// **Important**: All `append_async()` calls for a transaction MUST complete
690    /// before calling `commit()`. The typical pattern is:
691    ///
692    /// ```ignore
693    /// wal.append_async(op1)?;
694    /// wal.append_async(op2)?;
695    /// let epoch = wal.commit()?;  // Register for durability
696    /// if let Some(epoch) = epoch {
697    ///     wal.group_commit_coordinator().unwrap().wait_for_flush(epoch)?;
698    /// }
699    /// ```
700    ///
701    /// # Returns
702    ///
703    /// Returns an epoch number for GroupCommit/AsyncBatched modes that the
704    /// caller should wait on using `group_commit_coordinator().wait_for_flush(epoch)`.
705    ///
706    /// For other modes, returns `None`:
707    /// - Synchronous: Data is already durable when this returns
708    /// - Async: No waiting needed (fire-and-forget)
709    ///
710    /// # Race Condition Handling
711    ///
712    /// In GroupCommit mode, there's an intentional race between:
713    /// 1. The flush thread draining entries
714    /// 2. Transactions calling `register_transaction()`
715    ///
716    /// This is handled safely: if entries are drained before registration,
717    /// the epoch will still advance (with no entries), ensuring waiters
718    /// are notified. The data durability is guaranteed because entries
719    /// must be in the ring buffer before this method is called.
720    pub fn commit(&self) -> Result<Option<u64>> {
721        match self.durability_mode {
722            DurabilityMode::Synchronous => {
723                // Drain and flush immediately with fsync
724                let entries = self.wal.drain_all();
725                if !entries.is_empty() {
726                    self.coordinator.flush(entries, true)?;
727                }
728                Ok(None)
729            }
730            DurabilityMode::Async { .. } => {
731                // Just let background thread handle it
732                Ok(None)
733            }
734            DurabilityMode::GroupCommit { .. } | DurabilityMode::AsyncBatched { .. } => {
735                // Register with coordinator and return epoch to wait for
736                if let Some(ref gc) = self.group_commit {
737                    let (epoch, should_trigger) = gc.register_transaction()?;
738
739                    // If batch is full, signal flush thread to wake up immediately
740                    if should_trigger {
741                        self.flush_notifier.notify();
742                    }
743
744                    Ok(Some(epoch))
745                } else {
746                    // Fallback to sync if no coordinator (shouldn't happen)
747                    let entries = self.wal.drain_all();
748                    if !entries.is_empty() {
749                        self.coordinator.flush(entries, true)?;
750                    }
751                    Ok(None)
752                }
753            }
754        }
755    }

852    /// Install a WAL DEK keyring at runtime, flipping a live plaintext WAL to
853    /// encrypted (Issue #3616 PR2). This is the structural seam the plaintext →
854    /// encrypted enable engine drives.
855    ///
856    /// # Transition
857    ///
858    /// Only the `None` → `Some` transition is supported here. Installing when a
859    /// keyring is already present is **rejected** with a structured error (never
860    /// a silent replace, so no data is lost): rotation (`Some` → `Some'`) is the
861    /// job of the keyring's own [`add_generation`], not this seam.
862    ///
863    /// # Crash-consistency / concurrency
864    ///
865    /// You cannot append encrypted (v16) frames into an already-open plaintext
866    /// (v13) segment, so the install runs as the `advance` closure inside the
867    /// existing seal→reopen hand-off ([`Self::seal_active_segment_for_rotation`]):
868    /// it drains + fsyncs in-flight ring-buffer entries into the current plaintext
869    /// segment, seals it, stores the keyring into the shared presence cell, then
870    /// opens a fresh segment which is now written in the encrypted keyversioned
871    /// format. Because the store happens **between** seal and reopen while holding
872    /// the coordinator `writer` mutex, no `flush()` can interleave: every segment's
873    /// header and frames use one consistent keyring state, and every acknowledged
874    /// append is preserved across the flip.
875    ///
876    /// The store closure obeys the same lock-order contract as a rotation
877    /// `advance`: it only mutates the in-memory presence cell and must not acquire
878    /// any lock ordered after `wal` (no cold flush, no `historical`,
879    /// no `current_timestamp`).
880    ///
881    /// [`add_generation`]: crate::encryption::wal_encryption::WalKeyring::add_generation
882    // PR2 (#3616) ships this structural install seam; its production consumer is
883    // the plaintext → encrypted enable engine (#3616 PR3), which drives it from
884    // `enable_encryption` and the startup `install_pending_enable_wal_keyring` hook.
885    pub(crate) fn install_wal_keyring(
886        &self,
887        keyring: crate::encryption::wal_encryption::WalKeyring,
888    ) -> Result<()> {
889        // Serialize the ENTIRE install — presence check, seal, store, reopen —
890        // under a dedicated leaf mutex. Without it, the presence check below runs
891        // outside the coordinator `writer` mutex while the store happens inside the
892        // seal→reopen hand-off, so two concurrent installers could both observe
893        // `None`, both pass the check, and both run seal→store→reopen — the second
894        // silently replacing the first keyring and producing an undecryptable
895        // segment. Holding this guard for the whole body makes a second concurrent
896        // installer block here, then observe `Some`, and take the rejection path.
897        let _install_guard = self.install_lock.lock().unwrap_or_else(|e| e.into_inner());
898
899        // Reject a double-install: presence is a one-way None → Some transition.
900        // Checked (under the install lock) before the seal so an already-encrypted
901        // WAL is never re-rolled.
902        if self.wal_keyring.load().is_some() {
903            // A DISTINGUISHABLE precondition variant (not a generic `WalError`) so
904            // the enable engine's `map_wal_install_err` maps ONLY this to
905            // FAILED_PRECONDITION and leaves genuine WAL I/O / seal faults as
906            // INTERNAL (Issue #3616 PR3). MCP classifies it as FAILED_PRECONDITION.
907            return Err(Error::Storage(StorageError::WalKeyringAlreadyInstalled {
908                reason: "a WAL keyring is already installed; runtime install only \
909                         supports the plaintext → encrypted (None → Some) transition \
910                         (key rotation is a separate operation)"
911                    .to_string(),
912            }));
913        }
914
915        let cell = Arc::clone(&self.wal_keyring);
916        let new_keyring = Arc::new(keyring);
917        // Store into the shared cell strictly between sealing the plaintext
918        // segment and opening the fresh (now encrypted) one, under the writer
919        // mutex — the atomic hand-off point.
920        self.seal_active_segment_for_rotation(move || {
921            cell.store(Some(new_keyring));
922        })
923    }

925    /// Uninstall the WAL DEK keyring at runtime, flipping a live encrypted WAL
926    /// back to plaintext (Issue #3616 PR4). This is the structural seam the
927    /// encrypted → plaintext DISABLE engine drives — the exact inverse of
928    /// [`Self::install_wal_keyring`].
929    ///
930    /// # Transition
931    ///
932    /// Only the `Some` → `None` transition is supported here. Uninstalling when
933    /// NO keyring is present is **rejected** with a structured error (never a
934    /// silent no-op or a spurious segment roll): a caller cannot disable
935    /// encryption on an already-plaintext WAL.
936    ///
937    /// # Crash-consistency / concurrency
938    ///
939    /// Symmetric to install: you cannot append plaintext (v13) frames into an
940    /// already-open encrypted (v16) segment, so the uninstall runs as the
941    /// `advance` closure inside the same seal→reopen hand-off
942    /// ([`Self::seal_active_segment_for_rotation`]): it drains + fsyncs in-flight
943    /// ring-buffer entries into the current encrypted segment, seals it, stores
944    /// `None` into the shared presence cell, then opens a fresh segment which is
945    /// now written in the plaintext string-label format. Because the store
946    /// happens **between** seal and reopen while holding the coordinator `writer`
947    /// mutex, no `flush()` can interleave: every segment's header and frames use
948    /// one consistent keyring state, and every acknowledged append is preserved
949    /// across the flip.
950    ///
951    /// # Read capability after uninstall
952    ///
953    /// Dropping the keyring removes read-decrypt capability from the live cell,
954    /// so [`Self::read_from`] (which snapshots the cell) can no longer decode the
955    /// pre-uninstall encrypted segments still on disk. This is expected and
956    /// mirrors the enable engine's inverse concern (its retired plaintext
957    /// segments must be *deleted* for security): the DISABLE driver captures a
958    /// plaintext index snapshot of all pre-uninstall state, then RETIRES those
959    /// encrypted segments, so a reopen reconstructs from the plaintext snapshot
960    /// and replays only the fresh plaintext segment. The retire is a separate
961    /// driver step (mirroring [`install_wal_keyring`]'s retire being separate),
962    /// not part of this seam.
963    ///
964    /// The store closure obeys the same lock-order contract as a rotation
965    /// `advance`: it only mutates the in-memory presence cell and must not
966    /// acquire any lock ordered after `wal`.
967    ///
968    /// [`install_wal_keyring`]: Self::install_wal_keyring
969    // PR4 (#3616) ships this structural uninstall seam; its production consumer is
970    // the encrypted → plaintext disable engine (#3616 PR4), which drives it from
971    // `disable_encryption` and the startup disable-resume hook.
972    pub(crate) fn uninstall_wal_keyring(&self) -> Result<()> {
973        // Serialize the ENTIRE uninstall — presence check, seal, store, reopen —
974        // under the same dedicated leaf mutex install uses. Without it, two
975        // concurrent uninstallers could both observe `Some`, both pass the check,
976        // and both run seal→store→reopen — the second sealing an already-plaintext
977        // segment spuriously. Holding this guard for the whole body makes a second
978        // concurrent uninstaller block here, then observe `None`, and take the
979        // rejection path. It also serializes install against uninstall (both take
980        // this same leaf), so the presence transition is never torn.
981        let _install_guard = self.install_lock.lock().unwrap_or_else(|e| e.into_inner());
982
983        // Reject when there is nothing to uninstall: presence must be `Some` for
984        // this one-way Some → None transition. Checked (under the install lock)
985        // before the seal so an already-plaintext WAL is never re-rolled.
986        if self.wal_keyring.load().is_none() {
987            // A DISTINGUISHABLE precondition variant (not a generic `WalError`) so
988            // the disable engine maps ONLY this to FAILED_PRECONDITION and leaves
989            // genuine WAL I/O / seal faults as INTERNAL (Issue #3616 PR4). MCP
990            // classifies it as FAILED_PRECONDITION.
991            return Err(Error::Storage(StorageError::WalKeyringNotInstalled {
992                reason: "no WAL keyring is installed; runtime uninstall only supports the \
993                         encrypted → plaintext (Some → None) transition (a plaintext WAL is \
994                         already un-encrypted)"
995                    .to_string(),
996            }));
997        }
998
999        let cell = Arc::clone(&self.wal_keyring);
1000        // Store `None` into the shared cell strictly between sealing the encrypted
1001        // segment and opening the fresh (now plaintext) one, under the writer
1002        // mutex — the atomic hand-off point, the exact inverse of install.
1003        self.seal_active_segment_for_rotation(move || {
1004            cell.store(None);
1005        })
1006    }

1023    /// Force-roll the active WAL segment for a full-MEK key rotation (Issue
1024    /// #3617): drain and durably flush every in-flight ring-buffer entry into
1025    /// the CURRENT (old-generation) segment, then atomically seal it, run
1026    /// `advance` (which flips the WAL keyring to the new generation), and open a
1027    /// fresh segment stamped with — and encrypting under — the new generation.
1028    ///
1029    /// # Quiesce / correctness
1030    ///
1031    /// Appends land in the lock-free ring buffer BEFORE any flush. This method
1032    /// first drains and flushes them (with fsync) into the old segment, so no
1033    /// acknowledged entry is lost across the roll. The subsequent seal + keyring
1034    /// flip + reopen all run under the coordinator `writer` mutex, which every
1035    /// flush also holds while it reads the write cipher and writes the segment
1036    /// header — so the (segment header generation, frame-encrypting generation)
1037    /// pair is always consistent and a batch can never straddle two generations.
1038    /// Any append racing the roll is simply picked up by whichever flush runs
1039    /// next and written, consistently, to whatever segment is current at that
1040    /// flush. `advance` runs while holding only the `writer` mutex, so it must
1041    /// not acquire any lock ordered after `wal` (no cold flush inside it).
1042    pub(crate) fn seal_active_segment_for_rotation<F: FnOnce()>(&self, advance: F) -> Result<()> {
1043        // 1. Drain + flush all pending entries into the current (old) segment,
1044        //    with fsync, so nothing acknowledged is left only in the ring buffer.
1045        let entries = self.wal.drain_all();
1046        if !entries.is_empty() {
1047            self.coordinator.flush(entries, true)?;
1048        }
1049        // 2. Seal old + flip keyring + open new, atomically under the writer mutex.
1050        self.coordinator.seal_active_segment_and_reopen(advance)
1051    }

1058    /// Honors the configured crash-torn-tail recovery policy (Issue #3433): by

1071            self.wal_dir(),
1072            start_lsn,
1073            keyring_guard.as_ref().map(|k| k.as_ref()),
1074            self.tolerate_torn_tail,
1075        )
1076    }

1078    /// Shutdown the WAL system gracefully.
1079    ///
1080    /// This signals the background thread to stop, waits for it to finish,
1081    /// and performs a final flush of all pending entries.
1082    pub fn shutdown(&mut self) {
1083        // Gracefully shutdown the WAL.
1084        // This stops accepting new writes, waits for active batches to complete,
1085        // and then closes the ring buffers.
1086        self.wal.shutdown_graceful();
1087
1088        // Signal shutdown
1089        self.shutdown_signal.store(true, Ordering::Relaxed);
1090
1091        // Wake up flush thread so it can see the shutdown signal
1092        self.flush_notifier.notify();
1093
1094        // Wait for flush thread to finish
1095        if let Some(handle) = self.flush_thread.take() {
1096            let _ = handle.join();
1097        }
1098    }

1739    //
1740    // The exact inverse of the PR2 install seam: seal the encrypted (v16) active
1741    // segment, store `None` into the shared presence cell, and reopen a fresh
1742    // PLAINTEXT (v13) segment — the structural seam the encrypted → plaintext
1743    // DISABLE engine drives. Because dropping the keyring removes read-decrypt
1744    // capability, the pre-uninstall encrypted segments become undecryptable
1745    // through the live cell (`read_from` now snapshots `None`); the disable
1746    // DRIVER retires them after capturing a plaintext snapshot (a later slice).
1747    // These seam tests model that retire by deleting the pre-uninstall segment
1748    // files, then prove the post-uninstall cohort is plaintext at rest.

1750    /// D1 — Some→None transition: an encrypted WAL reports `is_encrypted() ==
1751    /// true`; after `uninstall_wal_keyring` it reports `false`; records appended
1752    /// after the uninstall land in a PLAINTEXT (v13) segment readable with NO
1753    /// keyring, and a full round-trip with the retired keyring still recovers
1754    /// every cohort (encrypted pre-uninstall + plaintext post-uninstall) in order.
1755    #[test]
1756    fn uninstall_keyring_some_to_none_transition() {
1757        use crate::storage::wal::segment_reader::read_entries_from_dir_with_keyring;
1758        use std::collections::HashSet;
1759
1760        let dir = tempdir().unwrap();
1761        let mut config = ConcurrentWalSystemConfig::new(dir.path());
1762        config.durability_mode = DurabilityMode::Synchronous;
1763        let wal = ConcurrentWalSystem::new(config).unwrap();
1764
1765        // Reach the ENCRYPTED state the disable engine strips (None → Some).
1766        wal.install_wal_keyring(wal_keyring(7)).unwrap();
1767        assert!(
1768            wal.is_encrypted(),
1769            "the WAL must be encrypted before uninstall"
1770        );
1771
1772        // Pre-uninstall cohort → encrypted v16 segment(s).
1773        for id in 1..=4 {
1774            wal.append(create_test_operation(id)).unwrap();
1775        }
1776        wal.flush().unwrap();
1777
1778        // Snapshot the encrypted segment files (the driver retires these after
1779        // capturing a plaintext snapshot); retain a keyring clone to model the
1780        // driver's decrypt capability over them until then.
1781        let pre_uninstall_stems: HashSet<String> = std::fs::read_dir(dir.path())
1782            .unwrap()
1783            .flatten()
1784            .filter_map(|e| {
1785                let name = e.file_name().to_string_lossy().into_owned();
1786                name.strip_suffix(".log").map(str::to_string)
1787            })
1788            .collect();
1789        let retired = wal
1790            .wal_keyring()
1791            .expect("an encrypted WAL must expose its keyring");
1792
1793        // Reverse seam.
1794        wal.uninstall_wal_keyring().unwrap();
1795        assert!(
1796            !wal.is_encrypted(),
1797            "uninstalling the keyring must flip the WAL to plaintext"
1798        );
1799
1800        // Post-uninstall cohort → plaintext v13 segment(s).
1801        for id in 5..=8 {
1802            wal.append(create_test_operation(id)).unwrap();
1803        }
1804        wal.flush().unwrap();
1805
1806        // (a) ROUND-TRIP: with the retired keyring the encrypted pre-uninstall
1807        // cohort and the plaintext post-uninstall cohort both recover in order
1808        // (plaintext segments parse regardless of a present keyring).
1809        let all =
1810            read_entries_from_dir_with_keyring(dir.path(), LSN::initial(), Some(&retired), true)
1811                .unwrap();
1812        assert_eq!(
1813            recovered_ids(&all),
1814            (1..=8).collect::<Vec<_>>(),
1815            "with the retired keyring both cohorts recover in order across the uninstall"
1816        );
1817
1818        // (b) PLAINTEXT-AT-REST: model the driver's retire of the encrypted
1819        // segments, then a read with NO keyring recovers the post-uninstall
1820        // cohort as plaintext — positive proof the reopened segment is v13, not
1821        // an in-memory-flag flip that kept writing encrypted.
1822        for entry in std::fs::read_dir(dir.path()).unwrap().flatten() {
1823            let name = entry.file_name().to_string_lossy().into_owned();
1824            let stem = name
1825                .strip_suffix(".log")
1826                .or_else(|| name.strip_suffix(".log.meta"));
1827            if let Some(stem) = stem
1828                && pre_uninstall_stems.contains(stem)
1829            {
1830                std::fs::remove_file(entry.path()).unwrap();
1831            }
1832        }
1833        let plaintext_only =
1834            read_entries_from_dir_with_keyring(dir.path(), LSN::initial(), None, true).unwrap();
1835        assert_eq!(
1836            recovered_ids(&plaintext_only),
1837            (5..=8).collect::<Vec<_>>(),
1838            "after retiring the encrypted segments, the post-uninstall cohort reads back \
1839             as plaintext with NO keyring (the reopened segment is v13 at rest)"
1840        );
1841    }
