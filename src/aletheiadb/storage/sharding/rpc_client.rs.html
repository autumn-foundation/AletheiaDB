43/// Configuration for RPC client.
44///
45/// # The Spark
46/// Shard-to-shard communication over a network is inherently unreliable. The
47/// `RpcConfig` allows operators to tune how resilient the system should be to
48/// transient network partitions and latency spikes during distributed transactions.
49///
50/// # The Details
51/// Includes settings for request timeouts and retry logic with exponential backoff.
52/// In production, `timeout` should typically be set lower than the 2PC coordinator
53/// timeout to prevent cascading failures.
54///
55/// # Examples
56/// ```
57/// use aletheiadb::storage::sharding::rpc_client::RpcConfig;
58/// use std::time::Duration;
59///
60/// let config = RpcConfig {
61///     endpoint: "http://shard1:9000".to_string(),
62///     timeout: Duration::from_secs(5),
63///     max_retries: 3,
64///     ..Default::default()
65/// };
66/// assert_eq!(config.max_retries, 3);
67/// ```
68#[derive(Debug, Clone)]
69pub struct RpcConfig {
70    /// Base endpoint URL (e.g., "http://shard0:9000").
71    pub endpoint: String,
72    /// Request timeout.
73    pub timeout: Duration,
74    /// Number of retries for transient failures.
75    pub max_retries: usize,
76    /// Base delay for exponential backoff.
77    pub retry_base_delay: Duration,
78    /// Maximum delay for exponential backoff.
79    pub retry_max_delay: Duration,
80    /// Enable TLS/HTTPS.
81    pub use_tls: bool,
82    /// HTTP connection pool size.
83    pub pool_size: usize,
84    /// Idle connection timeout.
85    pub idle_timeout: Duration,
86}
87
88impl Default for RpcConfig {
89    fn default() -> Self {
90        Self {
91            endpoint: "http://localhost:9000".to_string(),
92            timeout: Duration::from_secs(30),
93            max_retries: 3,
94            retry_base_delay: Duration::from_millis(100),
95            retry_max_delay: Duration::from_secs(10),
96            use_tls: false,
97            pool_size: 10,
98            idle_timeout: Duration::from_secs(90),
99        }
100    }

103/// HTTP-based RPC client for shard communication.
104///
105/// This client implements the `ShardClient` trait using HTTP/JSON as the
106/// transport protocol. It includes automatic retries with exponential backoff,
107/// connection health tracking, and comprehensive error handling.
108///
109/// # Thread Safety
110/// HTTP-based implementation of the ShardClient trait.
111///
112/// # The Spark
113/// In a sharded environment, nodes must coordinate to execute distributed queries
114/// and Two-Phase Commit (2PC) transactions. The `HttpShardClient` abstracts away
115/// the network complexity, making a remote shard look like a local resource.
116///
117/// # The Details
118/// This client uses `reqwest` internally and implements the `ShardClient` trait
119/// to send `Prepare`, `Commit`, and `Abort` signals. It handles network retries
120/// automatically based on its `RpcConfig`. It is thread-safe and designed to be
121/// shared via `Arc`. Internal state is protected by atomic operations and RwLocks.
122///
123/// # Protocol
124///
125/// The HTTP API endpoints are:
126///
127/// - `POST /api/v1/2pc/prepare` - Prepare phase of 2PC
128/// - `POST /api/v1/2pc/commit` - Commit phase of 2PC
129/// - `POST /api/v1/2pc/abort` - Abort phase of 2PC
130/// - `POST /api/v1/query` - Execute a query
131/// - `GET /api/v1/state` - Get shard state
132/// - `POST /api/v1/migration/receive` - Receive migration batch
133/// - `POST /api/v1/migration/extract` - Extract migration batch
134/// - `GET /api/v1/health` - Health check
135///
136/// # Examples
137/// ```ignore
138/// use aletheiadb::storage::sharding::types::ShardId;
139/// use aletheiadb::storage::sharding::rpc_client::{HttpShardClient, RpcConfig};
140///
141/// let shard_id = ShardId::new(1).unwrap();
142/// let config = RpcConfig::default();
143///
144/// // In a real app, this connects to the remote shard's HTTP API
145/// let client = HttpShardClient::new(shard_id, config).unwrap();
146/// let state = client.get_state().unwrap();
147/// ```
148pub struct HttpShardClient {
149    /// Shard ID this client connects to.
150    shard_id: ShardId,
151    /// Configuration.
152    config: RpcConfig,
153    /// Health status.
154    healthy: AtomicBool,
155    /// Last successful request time.
156    last_success: RwLock<Option<Instant>>,
157    /// Total requests made.
158    request_count: AtomicU64,
159    /// Failed request count.
160    failure_count: AtomicU64,
161    /// HTTP client (placeholder - requires reqwest feature).
162    #[cfg(feature = "sharding-rpc")]
163    http_client: reqwest::blocking::Client,
164}

236    /// Execute a request with retry logic.
237    #[cfg(feature = "sharding-rpc")]
238    fn execute_with_retry<T, F>(&self, operation: &str, f: F) -> NetworkResult<T>
239    where
240        F: Fn() -> NetworkResult<T>,
241    {
242        self.request_count.fetch_add(1, Ordering::Relaxed);
243
244        let mut last_error = None;
245        let mut delay = self.config.retry_base_delay;
246
247        for attempt in 0..=self.config.max_retries {
248            match f() {
249                Ok(result) => {
250                    self.healthy.store(true, Ordering::SeqCst);
251                    if let Ok(mut last) = self.last_success.write() {
252                        *last = Some(Instant::now());
253                    }
254                    return Ok(result);
255                }
256                Err(e) => {
257                    last_error = Some(e.clone());
258
259                    // Check if error is retryable
260                    if !Self::is_retryable(&e) || attempt == self.config.max_retries {
261                        break;
262                    }
263
264                    // Exponential backoff with jitter
265                    let jitter = Duration::from_millis(rand_jitter());
266                    std::thread::sleep(delay + jitter);
267                    delay = std::cmp::min(delay * 2, self.config.retry_max_delay);
268                }
269            }
270        }
271
272        self.failure_count.fetch_add(1, Ordering::Relaxed);
273        self.healthy.store(false, Ordering::SeqCst);
274
275        Err(
276            last_error.unwrap_or_else(|| NetworkError::ConnectionFailed {
277                shard_id: self.shard_id,
278                reason: format!("Unknown error in {}", operation),
279            }),
280        )
281    }

310    #[cfg(feature = "sharding-rpc")]
311    fn post_json<Req: serde::Serialize, Resp: serde::de::DeserializeOwned>(
312        &self,
313        path: &str,
314        body: &Req,
315    ) -> NetworkResult<Resp> {
316        let url = format!("{}{}", self.config.endpoint, path);
317
318        let response = self.http_client.post(&url).json(body).send().map_err(|e| {
319            if e.is_timeout() {
320                NetworkError::Timeout {
321                    shard_id: self.shard_id,
322                    operation: path.to_string(),
323                    duration: self.config.timeout,
324                }
325            } else if e.is_connect() {
326                NetworkError::ConnectionFailed {
327                    shard_id: self.shard_id,
328                    reason: e.to_string(),
329                }
330            } else {
331                NetworkError::ProtocolError(e.to_string())
332            }
333        })?;
334
335        if !response.status().is_success() {
336            return Err(NetworkError::ProtocolError(format!(
337                "HTTP {}: {}",
338                response.status(),
339                response.text().unwrap_or_default()
340            )));
341        }
342
343        response.json().map_err(|e| {
344            NetworkError::SerializationError(format!("Failed to deserialize response: {}", e))
345        })
346    }

371                "HTTP {}: {}",
372                response.status(),

376
377        response.json().map_err(|e| {
378            NetworkError::SerializationError(format!("Failed to deserialize response: {}", e))
379        })
380    }

383/// Simple jitter function (returns 0-100ms).
384#[allow(dead_code)]
385fn rand_jitter() -> u64 {
386    // Simple pseudo-random using time
387    let nanos = std::time::SystemTime::now()
388        .duration_since(std::time::UNIX_EPOCH)
389        .map(|d| d.subsec_nanos())
390        .unwrap_or(0);
391    (nanos % 100) as u64
392}

403    #[cfg(feature = "sharding-rpc")]
404    fn prepare(
405        &self,
406        tx_id: TxId,
407        operations: &[u8],
408        timestamp: Option<HybridTimestamp>,
409    ) -> NetworkResult<PrepareResponse> {
410        #[derive(serde::Serialize)]
411        struct PrepareRequest {
412            tx_id: u64,
413            operations: Vec<u8>,
414            timestamp: Option<HybridTimestampDto>,
415        }
416
417        #[derive(serde::Serialize)]
418        struct HybridTimestampDto {
419            wallclock: i64,
420            logical: u32,
421        }
422
423        #[derive(serde::Deserialize)]
424        struct PrepareResponseDto {
425            ready: bool,
426            reason: Option<String>,
427        }
428
429        self.execute_with_retry("prepare", || {
430            let timestamp = timestamp.map(|ts| HybridTimestampDto {
431                wallclock: ts.wallclock(),
432                logical: ts.logical(),
433            });
434            let req = PrepareRequest {
435                tx_id: tx_id.as_u64(),
436                operations: operations.to_vec(),
437                timestamp,
438            };
439
440            let resp: PrepareResponseDto = self.post_json("/api/v1/2pc/prepare", &req)?;
441
442            Ok(PrepareResponse {
443                ready: resp.ready,
444                reason: resp.reason,
445            })
446        })
447    }

707        _labels: &[String],
708        _batch_size: usize,
709        _offset: u64,
710    ) -> NetworkResult<MigrationBatch> {
711        Err(NetworkError::ConnectionFailed {
712            shard_id: self.shard_id,
713            reason: "RPC not available: enable 'sharding-rpc' feature".to_string(),
714        })
715    }
716
717    #[cfg(feature = "sharding-rpc")]
718    fn health_check(&self) -> NetworkResult<()> {
719        #[derive(serde::Deserialize)]
720        struct HealthResponse {
721            status: String,
722        }
723
724        self.execute_with_retry("health_check", || {
725            let resp: HealthResponse = self.get_json("/api/v1/health")?;
726
727            if resp.status == "healthy" {
728                Ok(())
729            } else {
730                Err(NetworkError::ShardUnavailable(self.shard_id))
731            }
732        })
733    }
734
735    #[cfg(not(feature = "sharding-rpc"))]
736    fn health_check(&self) -> NetworkResult<()> {
737        Err(NetworkError::ConnectionFailed {
738            shard_id: self.shard_id,
739            reason: "RPC not available: enable 'sharding-rpc' feature".to_string(),
740        })
741    }
742}

744/// Statistics for an HTTP shard client.
745#[derive(Debug, Clone)]
746pub struct ClientStats {
747    /// Shard ID.
748    pub shard_id: ShardId,
749    /// Endpoint URL.
750    pub endpoint: String,
751    /// Health status.
752    pub healthy: bool,
753    /// Total request count.
754    pub request_count: u64,
755    /// Failed request count.
756    pub failure_count: u64,
757    /// Last successful request time.
758    pub last_success: Option<Instant>,
759}
760
761impl ClientStats {
762    /// Calculate success rate (0.0 to 1.0).
763    pub fn success_rate(&self) -> f64 {
764        if self.request_count == 0 {
765            1.0
766        } else {
767            1.0 - (self.failure_count as f64 / self.request_count as f64)
768        }
769    }
770}
