1//! Optional embedding generation via [`embed_anything`].
2//!
3//! AletheiaDB stores and indexes vector properties; it does not maintain
4//! provider-specific embedding clients. Enable the `embeddings` feature to use
5//! and re-export `embed_anything`, then store the resulting dense vectors with
6//! `PropertyMapBuilder::insert_vector()`.

23/// Dense embedding data with chunk text and metadata preserved.
24#[derive(Debug, Clone, PartialEq)]
25pub struct DenseEmbedData {
26    /// Text associated with the embedding chunk, when provided upstream.
27    pub text: Option<String>,
28    /// Metadata associated with the embedding chunk, when provided upstream.
29    pub metadata: Option<HashMap<String, String>>,
30    /// Dense vector representation suitable for AletheiaDB vector storage.
31    pub embedding: Vec<f32>,
32}
33
34/// Error returned when an upstream embedding result is not dense.
35#[derive(Debug, Clone, Copy, Eq, PartialEq)]
36pub enum DenseEmbeddingError {
37    /// A multi-vector embedding cannot be represented as a single dense vector.
38    NotDense,
39}
40
41impl fmt::Display for DenseEmbeddingError {
42    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
43        match self {
44            Self::NotDense => formatter.write_str("embedding result is not a dense vector"),
45        }
46    }
47}
48
49impl Error for DenseEmbeddingError {}
50
51/// Converts upstream embedding results into dense vectors lazily.
52///
53/// When `limit` is provided, at most that many results are converted.
54pub fn to_dense_iter<I>(
55    results: I,
56    limit: Option<usize>,
57) -> impl Iterator<Item = Result<Vec<f32>, DenseEmbeddingError>>
58where
59    I: IntoIterator<Item = EmbeddingResult>,
60{
61    results
62        .into_iter()
63        .take(limit.unwrap_or(usize::MAX))
64        .map(embedding_result_to_dense)
65}
66
67/// Converts upstream `EmbedData` values into dense vectors while preserving
68/// chunk text and metadata.
69///
70/// When `limit` is provided, at most that many chunks are converted.
71pub fn embed_data_to_dense_iter<I>(
72    data: I,
73    limit: Option<usize>,
74) -> impl Iterator<Item = Result<DenseEmbedData, DenseEmbeddingError>>
75where
76    I: IntoIterator<Item = EmbedData>,
77{
78    data.into_iter()
79        .take(limit.unwrap_or(usize::MAX))
80        .map(embed_data_to_dense)
81}
82
83fn embed_data_to_dense(data: EmbedData) -> Result<DenseEmbedData, DenseEmbeddingError> {
84    let embedding = embedding_result_to_dense(data.embedding)?;
85
86    Ok(DenseEmbedData {
87        text: data.text,
88        metadata: data.metadata,
89        embedding,
90    })
91}
92
93fn embedding_result_to_dense(result: EmbeddingResult) -> Result<Vec<f32>, DenseEmbeddingError> {
94    match result {
95        EmbeddingResult::DenseVector(vector) => Ok(vector),
96        EmbeddingResult::MultiVector(_) => Err(DenseEmbeddingError::NotDense),
97    }
