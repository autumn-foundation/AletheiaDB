1//! WAL entry payload encryption/decryption.
2//!
3//! Pure functions that encrypt/decrypt the **payload portion** of a serialized
4//! WAL entry while leaving the header and op-type byte in plaintext.
5//!
6//! # WAL Entry Layout
7//!
8//! ```text
9//! Offset 0..8:   LSN (u64 LE)
10//! Offset 8..20:  Timestamp (12 bytes, HybridTimestamp)
11//! Offset 20..24: Checksum (u32 LE, CRC32)
12//! Offset 24:     Op Type (u8 discriminant)
13//! Offset 25+:    Operation-specific payload (variable length)
14//! ```
15//!
16//! The first 25 bytes (header + op type) stay plaintext and are passed as AAD
17//! (Additional Authenticated Data) so any tampering with them is detected during
18//! decryption.
19
20use std::sync::{Arc, RwLock};
21
22use crate::encryption::Cipher;
23use crate::encryption::error::EncryptionError;
24
25/// Byte offset where the encrypted payload begins (24-byte header + 1-byte op type).
26const WAL_HEADER_SIZE: usize = 25;
27
28/// The `key_version` a fresh (never-rotated) encrypted WAL stamps into its
29/// keyversioned segment headers (Issue #3617). Mirrors the index keyring's
30/// `ENC_INDEX_KEY_VERSION_V1` starting point so the two layers number their
31/// first generation identically.
32pub const INITIAL_WAL_KEY_VERSION: u32 = 1;
33
34// ── WAL key ring (dual-generation rotation, Issue #3617) ─────────────
35//
36// A full-MEK key rotation re-keys the WAL layer WITHOUT a bulk rewrite: new
37// appends are written under the NEW WAL DEK (in `KEYVERSIONED` segments
38// stamping the new `key_version`), while legacy/old-DEK segments still on disk
39// replay under the OLD DEK until they are truncated. To decrypt correctly the
40// recovery reader must therefore be able to hold BOTH generations at once and
41// select per segment. The `WalKeyring` is that holder — the WAL analogue of the
42// index [`IndexKeyring`](crate::storage::index_persistence::common::IndexKeyring),
43// deliberately mirroring its shape (small `Vec` of generations, a `current`
44// stamped into new segments, and a single-cipher `match_any` back-compat mode).
45
46#[derive(Clone)]
47struct WalKeyGeneration {
48    key_version: u32,
49    cipher: Arc<dyn Cipher>,
50}
51
52struct WalKeyringInner {
53    /// All live generations (1 before rotation, 2 during, 1 after). Small.
54    generations: Vec<WalKeyGeneration>,
55    /// Version stamped into freshly written segments (the newest generation).
56    current_version: u32,
57    /// Back-compat mode: a lone generation created from a single cipher
58    /// decrypts ANY segment (any `key_version`, or a legacy header carrying
59    /// none) with that one cipher. This is the never-rotated steady state and,
60    /// crucially, the post-provider-switch state: after a completed rotation the
61    /// operator reopens under the new key alone, so the single new-DEK keyring
62    /// must still read the new-DEK segments regardless of the `key_version` they
63    /// were stamped with during the rotation.
64    match_any: bool,
65}
66
67/// A cheaply-cloneable, shared-mutable set of WAL DEK ciphers addressed by
68/// `key_version` (Issue #3617).
69///
70/// Clones share the same underlying state, so the rotation driver can advance
71/// the current generation (making new appends stamp + encrypt under the new
72/// DEK) while the flush coordinator and recovery reader holding clones observe
73/// the change immediately. Never exposes key material; `Debug` is redacted.
74#[derive(Clone)]
75pub struct WalKeyring {
76    inner: Arc<RwLock<WalKeyringInner>>,
77}
78
79impl std::fmt::Debug for WalKeyring {
80    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
81        // Never render key material — only opaque generation metadata.
82        let inner = self.inner.read().unwrap_or_else(|e| e.into_inner());
83        f.debug_struct("WalKeyring")
84            .field("generations", &inner.generations.len())
85            .field("current_version", &inner.current_version)
86            .field("match_any", &inner.match_any)
87            .finish()
88    }
89}

91impl WalKeyring {
92    /// A single-generation keyring from one cipher (the non-rotation path).
93    ///
94    /// Reads decrypt any segment (any header `key_version`, or a legacy header
95    /// with none) with this cipher and writes stamp [`INITIAL_WAL_KEY_VERSION`].
96    pub fn single(cipher: Arc<dyn Cipher>) -> Self {
97        Self {
98            inner: Arc::new(RwLock::new(WalKeyringInner {
99                generations: vec![WalKeyGeneration {
100                    key_version: INITIAL_WAL_KEY_VERSION,
101                    cipher,
102                }],
103                current_version: INITIAL_WAL_KEY_VERSION,
104                match_any: true,
105            })),
106        }
107    }
108
109    /// A single-generation WAL keyring pinned to an explicit `key_version`
110    /// (Issue #488 version-provisioning). Reads decrypt every segment with this
111    /// one cipher (`match_any`, byte-identical to [`Self::single`]); only the
112    /// write-stamp / reported [`current_version`](Self::current_version) is
113    /// pinned to `key_version`. The durable `open()` path builds the WAL keyring
114    /// at the max on-disk segment `key_version` so new segments stamp the real
115    /// version instead of a stale [`INITIAL_WAL_KEY_VERSION`], keeping the WAL
116    /// and index layers in lockstep after a rotate-then-reopen.
117    pub fn single_versioned(cipher: Arc<dyn Cipher>, key_version: u32) -> Self {
118        Self {
119            inner: Arc::new(RwLock::new(WalKeyringInner {
120                generations: vec![WalKeyGeneration {
121                    key_version,
122                    cipher,
123                }],
124                current_version: key_version,
125                match_any: true,
126            })),
127        }
128    }
129
130    /// The current (write) cipher and the `key_version` it stamps into new
131    /// segment headers, or `None` if the keyring somehow holds no generation.
132    pub fn current(&self) -> Option<(Arc<dyn Cipher>, u32)> {
133        let inner = self.inner.read().unwrap_or_else(|e| e.into_inner());
134        let v = inner.current_version;
135        inner
136            .generations
137            .iter()
138            .find(|g| g.key_version == v)
139            .map(|g| (g.cipher.clone(), v))
140    }
141
142    /// The version freshly written segments are stamped with.
143    pub fn current_version(&self) -> u32 {
144        self.inner
145            .read()
146            .unwrap_or_else(|e| e.into_inner())
147            .current_version
148    }
149
150    /// Resolve the decryption cipher for a segment, given the `key_version` read
151    /// from its header (`Some` for a `KEYVERSIONED` segment, `None` for a legacy
152    /// segment that carries no key-version field).
153    ///
154    /// * `match_any` (single-cipher) keyring → the sole cipher for every
155    ///   segment (never-rotated / post-switch steady state).
156    /// * strict keyring, legacy segment (`None`) → the OLDEST (minimum-version,
157    ///   i.e. pre-rotation) generation — the DEK legacy segments were written
158    ///   under before the rotation advanced the generation.
159    /// * strict keyring, `Some(kv)` → the generation stamped `kv`, or `None`
160    ///   when no such generation is held (a genuine wrong/absent key surfaces as
161    ///   a loud AEAD failure downstream rather than silent wrong data).
162    pub fn cipher_for_segment(&self, key_version: Option<u32>) -> Option<Arc<dyn Cipher>> {
163        let inner = self.inner.read().unwrap_or_else(|e| e.into_inner());
164        if inner.match_any {
165            return inner.generations.first().map(|g| g.cipher.clone());
166        }
167        match key_version {
168            Some(kv) => inner
169                .generations
170                .iter()
171                .find(|g| g.key_version == kv)
172                .map(|g| g.cipher.clone()),
173            None => inner
174                .generations
175                .iter()
176                .min_by_key(|g| g.key_version)
177                .map(|g| g.cipher.clone()),
178        }
179    }
180
181    /// Whether a generation with `key_version` is currently held (a `match_any`
182    /// single keyring holds every version).
183    ///
184    /// Test-only for now (Issue #3617 PR2): no production path consults it —
185    /// segment dispatch goes through [`cipher_for_segment`](Self::cipher_for_segment).
186    /// `#[cfg(test)]`-gated rather than removed so PR3 can re-expose it if a
187    /// production caller lands.
188    #[cfg(test)]
189    pub fn has_version(&self, key_version: u32) -> bool {
190        let inner = self.inner.read().unwrap_or_else(|e| e.into_inner());
191        inner.match_any
192            || inner
193                .generations
194                .iter()
195                .any(|g| g.key_version == key_version)
196    }
197
198    /// Add a new generation and make it current. Switches the keyring to strict
199    /// per-version dispatch (leaving `match_any`). Idempotent for a version
200    /// already present (its cipher is replaced). Used by the rotation driver to
201    /// install the new WAL DEK before force-rolling the active segment.
202    pub fn add_generation(&self, key_version: u32, cipher: Arc<dyn Cipher>) {
203        let mut inner = self.inner.write().unwrap_or_else(|e| e.into_inner());
204        inner.match_any = false;
205        inner.generations.retain(|g| g.key_version != key_version);
206        inner.generations.push(WalKeyGeneration {
207            key_version,
208            cipher,
209        });
210        inner.current_version = key_version;
211    }
212
213    /// Number of live generations (test/introspection helper; never leaks keys).
214    #[cfg(test)]
215    pub fn generation_count(&self) -> usize {
216        self.inner
217            .read()
218            .unwrap_or_else(|e| e.into_inner())
219            .generations
220            .len()
221    }
222}

224/// Encrypt the payload portion of a serialized WAL entry.
225///
226/// # Arguments
227///
228/// * `serialized` - Full serialized WAL entry: `[header:24][op_type:1][payload:N]`
229/// * `cipher` - AEAD cipher to use for encryption
230///
231/// # Returns
232///
233/// A new buffer: `[header:24][op_type:1][encrypted_payload]` where
234/// `encrypted_payload` is `N + cipher.overhead()` bytes.
235///
236/// # Errors
237///
238/// Returns [`EncryptionError::InvalidWalEntry`] if `serialized` is shorter than
239/// 25 bytes (the minimum for header + op type with an empty payload).
240pub fn encrypt_wal_payload(
241    serialized: &[u8],
242    cipher: &Arc<dyn Cipher>,
243) -> Result<Vec<u8>, EncryptionError> {
244    if serialized.len() < WAL_HEADER_SIZE {
245        return Err(EncryptionError::InvalidWalEntry {
246            expected: WAL_HEADER_SIZE,
247            actual: serialized.len(),
248        });
249    }
250
251    let header_and_op = &serialized[..WAL_HEADER_SIZE];
252    let payload = &serialized[WAL_HEADER_SIZE..];
253
254    // Use header + op type as AAD so tampering with them is detected.
255    let encrypted_payload = cipher.encrypt(payload, header_and_op)?;
256
257    let mut output = Vec::with_capacity(WAL_HEADER_SIZE + encrypted_payload.len());
258    output.extend_from_slice(header_and_op);
259    output.extend_from_slice(&encrypted_payload);
260    Ok(output)
261}

263/// Decrypt the payload portion of a WAL entry encrypted by [`encrypt_wal_payload`].
264///
265/// # Arguments
266///
267/// * `encrypted_entry` - Encrypted WAL entry: `[header:24][op_type:1][encrypted_payload]`
268/// * `cipher` - Same AEAD cipher used for encryption
269///
270/// # Returns
271///
272/// A new buffer: `[header:24][op_type:1][plaintext_payload]`.
273///
274/// # Errors
275///
276/// - [`EncryptionError::InvalidWalEntry`] if the entry is too short for the
277///   header plus cipher overhead.
278/// - [`EncryptionError::DecryptFailed`] if AAD verification fails (e.g. header
279///   was tampered with) or the ciphertext is corrupted.
280pub fn decrypt_wal_payload(
281    encrypted_entry: &[u8],
282    cipher: &Arc<dyn Cipher>,
283) -> Result<Vec<u8>, EncryptionError> {
284    let min_len = WAL_HEADER_SIZE + cipher.overhead();
285    if encrypted_entry.len() < min_len {
286        return Err(EncryptionError::InvalidWalEntry {
287            expected: min_len,
288            actual: encrypted_entry.len(),
289        });
290    }
291
292    let header_and_op = &encrypted_entry[..WAL_HEADER_SIZE];
293    let encrypted_payload = &encrypted_entry[WAL_HEADER_SIZE..];
294
295    let plaintext = cipher.decrypt(encrypted_payload, header_and_op)?;
296
297    let mut output = Vec::with_capacity(WAL_HEADER_SIZE + plaintext.len());
298    output.extend_from_slice(header_and_op);
299    output.extend_from_slice(&plaintext);
300    Ok(output)
301}

310    /// Build a fake serialized WAL entry with a recognizable payload.
311    fn fake_entry(payload_len: usize) -> Vec<u8> {
312        let mut buf = vec![0u8; WAL_HEADER_SIZE + payload_len];
313        // Fake LSN = 42
314        buf[..8].copy_from_slice(&42u64.to_le_bytes());
315        // Op type = 1 (CreateNode)
316        buf[24] = 1;
317        // Fill payload with a recognizable pattern
318        for (i, b) in buf[WAL_HEADER_SIZE..].iter_mut().enumerate() {
319            *b = (i % 256) as u8;
320        }
321        buf
322    }

324    fn random_key() -> Zeroizing<[u8; 32]> {
325        let mut key = Zeroizing::new([0u8; 32]);
326        rand::thread_rng().fill_bytes(key.as_mut());
327        key
328    }

439    #[test]
440    fn keyring_single_matches_any_version() {
441        let cipher = aes_cipher();
442        let ring = WalKeyring::single(Arc::clone(&cipher));
443        assert_eq!(ring.current_version(), INITIAL_WAL_KEY_VERSION);
444        assert_eq!(ring.generation_count(), 1);
445        // A single-cipher keyring decrypts any segment: a legacy header (None),
446        // its own version, or an unrelated version stamped during a since-
447        // completed rotation (post-provider-switch state).
448        assert!(ring.cipher_for_segment(None).is_some());
449        assert!(
450            ring.cipher_for_segment(Some(INITIAL_WAL_KEY_VERSION))
451                .is_some()
452        );
453        assert!(ring.cipher_for_segment(Some(999)).is_some());
454        assert!(ring.has_version(7));
455    }

457    #[test]
458    fn keyring_dual_generation_dispatch() {
459        let old = aes_cipher();
460        let new = aes_cipher();
461        let ring = WalKeyring::single(Arc::clone(&old));
462        ring.add_generation(2, Arc::clone(&new));
463        // Now strict: current stamps the new version.
464        assert_eq!(ring.current_version(), 2);
465        assert_eq!(ring.generation_count(), 2);
466        let (cur, v) = ring.current().unwrap();
467        assert_eq!(v, 2);
468        assert!(Arc::ptr_eq(&cur, &new));
469        // Legacy segment (no key_version) resolves to the pre-rotation (oldest)
470        // generation.
471        assert!(Arc::ptr_eq(&ring.cipher_for_segment(None).unwrap(), &old));
472        // Keyversioned segments resolve to their exact generation.
473        assert!(Arc::ptr_eq(
474            &ring.cipher_for_segment(Some(1)).unwrap(),
475            &old
476        ));
477        assert!(Arc::ptr_eq(
478            &ring.cipher_for_segment(Some(2)).unwrap(),
479            &new
480        ));
481        // An unheld version is a loud miss (no silent wrong-key), not a fallback.
482        assert!(ring.cipher_for_segment(Some(3)).is_none());
483    }
