1//! Key derivation via HKDF-SHA256.
2//!
3//! Derives per-component Data Encryption Keys (DEKs) from the Master Encryption
4//! Key (MEK) using HKDF-SHA256. Each storage component (WAL, indexes, cold
5//! storage, checkpoints) gets a unique DEK to limit blast radius if any single
6//! key is compromised.
7
8use hkdf::Hkdf;
9use sha2::Sha256;
10use zeroize::Zeroizing;
11
12use crate::encryption::KeyDerivationError;
13
14/// HKDF info-string context for WAL encryption keys.
15pub const WAL_DEK_CONTEXT: &str = "wal";
16
17/// HKDF info-string context for index encryption keys.
18pub const INDEX_DEK_CONTEXT: &str = "index";
19
20/// HKDF info-string context for cold storage encryption keys.
21pub const COLD_DEK_CONTEXT: &str = "cold";
22
23/// HKDF info-string context for checkpoint encryption keys.
24pub const CHECKPOINT_DEK_CONTEXT: &str = "checkpoint";
25
26/// Derives per-component DEKs from a master encryption key using HKDF-SHA256.
27///
28/// The info string format is `aletheiadb-{component}-dek-v1` which binds each
29/// derived key to a specific component and allows future key-schedule versioning.
30pub struct KeyDerivation {
31    mek: Zeroizing<[u8; 32]>,
32}
33
34impl KeyDerivation {
35    /// Create a new key derivation context from a master encryption key.
36    pub fn new(mek: Zeroizing<[u8; 32]>) -> Self {
37        Self { mek }
38    }
39
40    /// Derive a 32-byte DEK for the given component name.
41    ///
42    /// The component name is embedded in the HKDF info string as
43    /// `aletheiadb-{component}-dek-v1`.
44    pub fn derive_dek(&self, component: &str) -> Result<Zeroizing<[u8; 32]>, KeyDerivationError> {
45        let hkdf = Hkdf::<Sha256>::new(None, self.mek.as_ref());
46        let info = format!("aletheiadb-{component}-dek-v1");
47
48        let mut okm = Zeroizing::new([0u8; 32]);
49        hkdf.expand(info.as_bytes(), okm.as_mut())
50            .map_err(|e| KeyDerivationError::DerivationFailed(e.to_string()))?;
51
52        Ok(okm)
53    }
54
55    /// Derive the DEK for WAL encryption.
56    pub fn derive_wal_dek(&self) -> Result<Zeroizing<[u8; 32]>, KeyDerivationError> {
57        self.derive_dek(WAL_DEK_CONTEXT)
58    }
59
60    /// Derive the DEK for index encryption.
61    pub fn derive_index_dek(&self) -> Result<Zeroizing<[u8; 32]>, KeyDerivationError> {
62        self.derive_dek(INDEX_DEK_CONTEXT)
63    }
64
65    /// Derive the DEK for cold storage encryption.
66    pub fn derive_cold_dek(&self) -> Result<Zeroizing<[u8; 32]>, KeyDerivationError> {
67        self.derive_dek(COLD_DEK_CONTEXT)
68    }
69
70    /// Derive the DEK for checkpoint encryption.
71    pub fn derive_checkpoint_dek(&self) -> Result<Zeroizing<[u8; 32]>, KeyDerivationError> {
72        self.derive_dek(CHECKPOINT_DEK_CONTEXT)
73    }
74}

76#[cfg(test)]
77mod tests {
78    use super::*;
79    use rand::RngCore;
80
81    fn random_mek() -> Zeroizing<[u8; 32]> {
82        let mut key = Zeroizing::new([0u8; 32]);
83        rand::thread_rng().fill_bytes(key.as_mut());
84        key
85    }
86
87    #[test]
88    fn derive_dek_deterministic() {
89        let mek = random_mek();
90        let kd = KeyDerivation::new(mek.clone());
91
92        let dek1 = kd.derive_dek("wal").unwrap();
93        let dek2 = kd.derive_dek("wal").unwrap();
94        assert_eq!(dek1.as_ref(), dek2.as_ref());
95    }
96
97    #[test]
98    fn different_components_different_deks() {
99        let mek = random_mek();
100        let kd = KeyDerivation::new(mek);
101
102        let wal = kd.derive_wal_dek().unwrap();
103        let index = kd.derive_index_dek().unwrap();
104        let cold = kd.derive_cold_dek().unwrap();
105        let checkpoint = kd.derive_checkpoint_dek().unwrap();
106
107        // All four must be distinct
108        let deks: Vec<&[u8; 32]> = vec![&*wal, &*index, &*cold, &*checkpoint];
109        for i in 0..deks.len() {
110            for j in (i + 1)..deks.len() {
111                assert_ne!(deks[i], deks[j], "DEK {i} and {j} must differ");
112            }
113        }
114    }
115
116    #[test]
117    fn different_mek_different_deks() {
118        let mek1 = random_mek();
119        let mek2 = random_mek();
120        let kd1 = KeyDerivation::new(mek1);
121        let kd2 = KeyDerivation::new(mek2);
122
123        let dek1 = kd1.derive_wal_dek().unwrap();
124        let dek2 = kd2.derive_wal_dek().unwrap();
125        assert_ne!(dek1.as_ref(), dek2.as_ref());
126    }
127
128    #[test]
129    fn custom_component_works() {
130        let mek = random_mek();
131        let kd = KeyDerivation::new(mek);
132
133        let dek = kd.derive_dek("my-custom-component").unwrap();
134        assert_eq!(dek.as_ref().len(), 32);
135    }
136
137    #[test]
138    fn dek_is_32_bytes() {
139        let mek = random_mek();
140        let kd = KeyDerivation::new(mek);
141
142        assert_eq!(kd.derive_wal_dek().unwrap().as_ref().len(), 32);
143        assert_eq!(kd.derive_index_dek().unwrap().as_ref().len(), 32);
144        assert_eq!(kd.derive_cold_dek().unwrap().as_ref().len(), 32);
145        assert_eq!(kd.derive_checkpoint_dek().unwrap().as_ref().len(), 32);
146    }
