1//! Key rotation mechanism for encryption at rest.
2//!
3//! Supports atomic key version switching without downtime.
4//! Background re-encryption of existing data is planned for a future phase.
5
6use crate::encryption::KeyProviderError;
7use std::sync::RwLock;
8
9/// Tracks key version information.
10#[derive(Debug, Clone)]
11pub struct KeyVersion {
12    /// Monotonically increasing version number.
13    pub version: u32,
14    /// UUID for this key version (for file headers).
15    pub version_id: [u8; 16],
16}
17
18/// State of a key rotation operation.
19#[derive(Debug, Clone, PartialEq, Eq)]
20pub enum RotationState {
21    /// No rotation in progress.
22    Idle,
23    /// Rotation initiated, new key active for writes.
24    InProgress {
25        /// Old key version.
26        old_version: u32,
27        /// New key version.
28        new_version: u32,
29    },
30    /// Rotation complete, old key removed.
31    Complete,
32}
33
34/// Manages key rotation for the encryption system.
35///
36/// Holds the current key version and supports atomic key switching.
37/// Background re-encryption of existing data is planned for a future phase.
38pub struct KeyRotationManager {
39    /// Current key version counter.
40    current_version: RwLock<u32>,
41    /// Current rotation state.
42    state: RwLock<RotationState>,
43}
44
45impl KeyRotationManager {
46    /// Create a new rotation manager starting at version 1.
47    #[must_use]
48    pub fn new() -> Self {
49        Self {
50            current_version: RwLock::new(1),
51            state: RwLock::new(RotationState::Idle),
52        }
53    }
54
55    /// Get the current key version.
56    #[must_use]
57    pub fn current_version(&self) -> u32 {
58        *self
59            .current_version
60            .read()
61            .unwrap_or_else(|e| e.into_inner())
62    }
63
64    /// Get the current rotation state.
65    #[must_use]
66    pub fn state(&self) -> RotationState {
67        self.state.read().unwrap_or_else(|e| e.into_inner()).clone()
68    }
69
70    /// Begin a key rotation.
71    ///
72    /// Returns the new version number on success.
73    ///
74    /// # Errors
75    ///
76    /// Returns [`KeyProviderError::Unavailable`] if a rotation is already in progress.
77    pub fn begin_rotation(&self) -> Result<u32, KeyProviderError> {
78        let mut state = self.state.write().unwrap_or_else(|e| e.into_inner());
79        if *state != RotationState::Idle {
80            return Err(KeyProviderError::Unavailable(
81                "Key rotation already in progress".into(),
82            ));
83        }
84        let old = self.current_version();
85        let new = old + 1;
86        *state = RotationState::InProgress {
87            old_version: old,
88            new_version: new,
89        };
90        Ok(new)
91    }
92
93    /// Complete the current rotation, advancing the version counter.
94    ///
95    /// # Errors
96    ///
97    /// Returns [`KeyProviderError::Unavailable`] if no rotation is in progress.
98    pub fn complete_rotation(&self) -> Result<(), KeyProviderError> {
99        let mut state = self.state.write().unwrap_or_else(|e| e.into_inner());
100        match *state {
101            RotationState::InProgress { new_version, .. } => {
102                let mut ver = self
103                    .current_version
104                    .write()
105                    .unwrap_or_else(|e| e.into_inner());
106                *ver = new_version;
107                *state = RotationState::Idle;
108                Ok(())
109            }
110            _ => Err(KeyProviderError::Unavailable(
111                "No rotation in progress".into(),
112            )),
113        }
114    }
115
116    /// Cancel the current rotation (rollback).
117    ///
118    /// # Errors
119    ///
120    /// Returns [`KeyProviderError::Unavailable`] if no rotation is in progress.
121    pub fn cancel_rotation(&self) -> Result<(), KeyProviderError> {
122        let mut state = self.state.write().unwrap_or_else(|e| e.into_inner());
123        match *state {
124            RotationState::InProgress { .. } => {
125                *state = RotationState::Idle;
126                Ok(())
127            }
128            _ => Err(KeyProviderError::Unavailable(
129                "No rotation in progress".into(),
130            )),
131        }
132    }
133}
134
135impl Default for KeyRotationManager {
136    fn default() -> Self {
137        Self::new()
138    }
139}

141#[cfg(test)]
142mod tests {
143    use super::*;
144
145    #[test]
146    fn new_starts_at_version_1() {
147        let mgr = KeyRotationManager::new();
148        assert_eq!(mgr.current_version(), 1);
149        assert_eq!(mgr.state(), RotationState::Idle);
150    }
151
152    #[test]
153    fn begin_rotation_transitions_state() {
154        let mgr = KeyRotationManager::new();
155        let new_ver = mgr.begin_rotation().unwrap();
156        assert_eq!(new_ver, 2);
157        assert_eq!(
158            mgr.state(),
159            RotationState::InProgress {
160                old_version: 1,
161                new_version: 2,
162            }
163        );
164        // Version counter is not advanced until complete.
165        assert_eq!(mgr.current_version(), 1);
166    }
167
168    #[test]
169    fn complete_rotation_advances_version() {
170        let mgr = KeyRotationManager::new();
171        mgr.begin_rotation().unwrap();
172        mgr.complete_rotation().unwrap();
173
174        assert_eq!(mgr.current_version(), 2);
175        assert_eq!(mgr.state(), RotationState::Idle);
176    }
177
178    #[test]
179    fn cancel_rotation_returns_to_idle() {
180        let mgr = KeyRotationManager::new();
181        mgr.begin_rotation().unwrap();
182        mgr.cancel_rotation().unwrap();
183
184        assert_eq!(mgr.current_version(), 1);
185        assert_eq!(mgr.state(), RotationState::Idle);
186    }
187
188    #[test]
189    fn double_begin_fails() {
190        let mgr = KeyRotationManager::new();
191        mgr.begin_rotation().unwrap();
192        let err = mgr.begin_rotation().unwrap_err();
193        assert!(
194            err.to_string().contains("already in progress"),
195            "expected 'already in progress', got: {err}"
196        );
197    }
198
199    #[test]
200    fn complete_without_begin_fails() {
201        let mgr = KeyRotationManager::new();
202        let err = mgr.complete_rotation().unwrap_err();
203        assert!(
204            err.to_string().contains("No rotation in progress"),
205            "expected 'No rotation in progress', got: {err}"
206        );
207    }
208
209    #[test]
210    fn cancel_without_begin_fails() {
211        let mgr = KeyRotationManager::new();
212        let err = mgr.cancel_rotation().unwrap_err();
213        assert!(
214            err.to_string().contains("No rotation in progress"),
215            "expected 'No rotation in progress', got: {err}"
216        );
217    }
218
219    #[test]
220    fn multiple_rotation_cycles() {
221        let mgr = KeyRotationManager::new();
222
223        // Cycle 1
224        mgr.begin_rotation().unwrap();
225        mgr.complete_rotation().unwrap();
226        assert_eq!(mgr.current_version(), 2);
227
228        // Cycle 2
229        mgr.begin_rotation().unwrap();
230        mgr.complete_rotation().unwrap();
231        assert_eq!(mgr.current_version(), 3);
232
233        // Cycle 3 with cancel
234        mgr.begin_rotation().unwrap();
235        mgr.cancel_rotation().unwrap();
236        assert_eq!(mgr.current_version(), 3);
237
238        // Cycle 4 after cancel
239        mgr.begin_rotation().unwrap();
240        mgr.complete_rotation().unwrap();
241        assert_eq!(mgr.current_version(), 4);
242    }
