1//! AWS KMS key provider (Issue #3587), gated behind `encryption-aws-kms`.
2//!
3//! # Envelope model
4//!
5//! The Master Encryption Key (MEK) is never stored in plaintext. Instead the
6//! configuration carries:
7//! - a KMS key id / ARN, and
8//! - a **base64-encoded KMS-encrypted data-key ciphertext blob** (the wrapped
9//!   MEK).
10//!
11//! At startup [`get_mek`](KmsKeyProvider::get_mek) calls KMS `Decrypt` on the
12//! blob and returns the recovered 32-byte plaintext as a
13//! [`Zeroizing<[u8; 32]>`]. Nothing sensitive is logged or exposed via `Debug`.
14//!
15//! # Async bridge
16//!
17//! `aws-sdk-kms` is async but the [`KeyProvider`] trait is synchronous, so this
18//! provider owns an internal current-thread Tokio runtime and blocks on the
19//! `Decrypt` future. If a call happens to occur *inside* an ambient Tokio
20//! runtime (which would make a nested `block_on` panic), the future is run on a
21//! dedicated OS thread instead, detected via
22//! [`Handle::try_current`](tokio::runtime::Handle::try_current).
23//!
24//! # Credentials
25//!
26//! Credentials are read from the standard `AWS_ACCESS_KEY_ID` /
27//! `AWS_SECRET_ACCESS_KEY` / `AWS_SESSION_TOKEN` environment variables. A full
28//! default credential provider chain (instance metadata, SSO, config files) via
29//! `aws-config` is a deliberate follow-up; the envelope-decrypt flow is
30//! identical either way.
31

50/// Configuration for the AWS KMS key provider.
51#[derive(Clone)]
52pub struct KmsConfig {
53    /// KMS key id or ARN used to decrypt the data key.
54    pub key_id: String,
55    /// Base64-encoded KMS-encrypted data-key ciphertext blob (the wrapped MEK).
56    pub encrypted_data_key: String,
57    /// AWS region (defaults to `us-east-1` if unset).
58    pub region: Option<String>,
59    /// Optional custom endpoint URL (e.g. LocalStack or a VPC endpoint).
60    pub endpoint_url: Option<String>,
61}

63impl fmt::Debug for KmsConfig {
64    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
65        // Redact the wrapped-key blob; key_id/region/endpoint are not secret but
66        // the ciphertext blob is treated as sensitive material.
67        f.debug_struct("KmsConfig")
68            .field("key_id", &self.key_id)
69            .field("encrypted_data_key", &"<redacted>")
70            .field("region", &self.region)
71            .field("endpoint_url", &self.endpoint_url)
72            .finish()
73    }
74}

76/// Decrypts a KMS-wrapped data key into the MEK on demand.
77pub struct KmsKeyProvider {

95impl KmsKeyProvider {
96    /// Build a KMS key provider from config.
97    ///
98    /// # Errors
99    ///
100    /// Returns [`KeyProviderError::InvalidKeyFormat`] if the ciphertext blob is
101    /// not valid base64, or [`KeyProviderError::Unavailable`] if the internal
102    /// runtime cannot be constructed.
103    pub fn new(config: &KmsConfig) -> Result<Self, KeyProviderError> {
104        let blob = BASE64
105            .decode(config.encrypted_data_key.as_bytes())
106            .map_err(|_| {
107                KeyProviderError::InvalidKeyFormat(
108                    "encrypted_data_key is not valid base64".to_string(),
109                )
110            })?;
111
112        let runtime = tokio::runtime::Builder::new_current_thread()
113            .enable_all()
114            .build()
115            .map_err(|e| {
116                KeyProviderError::Unavailable(format!("failed to build KMS runtime: {e}"))
117            })?;
118
119        // Credentials from the standard AWS_* environment variables. Missing
120        // credentials simply yield an auth failure at Decrypt time (typed
121        // AccessDenied), never a panic.
122        let access = std::env::var("AWS_ACCESS_KEY_ID").unwrap_or_default();
123        let secret = std::env::var("AWS_SECRET_ACCESS_KEY").unwrap_or_default();
124        let session = std::env::var("AWS_SESSION_TOKEN").ok();
125        let creds = Credentials::new(access, secret, session, None, "aletheia-env");
126
127        let region = config
128            .region
129            .clone()
130            .or_else(|| std::env::var("AWS_REGION").ok())
131            .unwrap_or_else(|| "us-east-1".to_string());
132
133        let mut builder = aws_sdk_kms::config::Builder::default()
134            .behavior_version(BehaviorVersion::latest())
135            .region(Region::new(region))
136            .credentials_provider(creds)
137            // Fail fast rather than retrying against an unreachable endpoint.
138            .retry_config(aws_sdk_kms::config::retry::RetryConfig::disabled())
139            .timeout_config(
140                aws_sdk_kms::config::timeout::TimeoutConfig::builder()
141                    .operation_timeout(std::time::Duration::from_secs(10))
142                    .operation_attempt_timeout(std::time::Duration::from_secs(5))
143                    .build(),
144            );
145        if let Some(ep) = &config.endpoint_url {
146            builder = builder.endpoint_url(ep.clone());
147        }
148        let client = Client::from_conf(builder.build());
149
150        Ok(Self {
151            client,
152            key_id: config.key_id.clone(),
153            blob,
154            runtime,
155        })
156    }
157
158    /// Test-only constructor injecting an already-built KMS `Client` (e.g. one
159    /// backed by a mocked transport) plus the raw ciphertext blob, bypassing the
160    /// real AWS credential/endpoint config. Not part of the public API; lets the
161    /// Decrypt success path be exercised without a live KMS.
162    #[cfg(test)]
163    fn with_client_for_test(client: Client, blob: Vec<u8>) -> Self {
164        let runtime = tokio::runtime::Builder::new_current_thread()
165            .enable_all()
166            .build()
167            .expect("test runtime");
168        Self {
169            client,
170            key_id: "alias/test".to_string(),
171            blob,
172            runtime,
173        }
174    }
175
176    /// Run the (owned, `'static`) decrypt future to completion, bridging the
177    /// async client into the synchronous trait without panicking inside an
178    /// ambient runtime.
179    fn block_on_decrypt(&self) -> Result<Zeroizing<[u8; 32]>, KeyProviderError> {
180        let fut = decrypt(self.client.clone(), self.key_id.clone(), self.blob.clone());
181
182        if Handle::try_current().is_ok() {
183            // We are inside another Tokio runtime; a nested block_on would
184            // panic. Run on a dedicated OS thread with our runtime's handle.
185            let handle = self.runtime.handle().clone();
186            let jh = std::thread::spawn(move || handle.block_on(fut));
187            jh.join().unwrap_or_else(|_| {
188                Err(KeyProviderError::Unavailable(
189                    "KMS decrypt worker thread panicked".to_string(),
190                ))
191            })
192        } else {
193            self.runtime.block_on(fut)
194        }
195    }
196}

198/// The owned, `'static` decrypt future body.
199async fn decrypt(
200    client: Client,
201    key_id: String,
202    blob: Vec<u8>,
203) -> Result<Zeroizing<[u8; 32]>, KeyProviderError> {
204    let out = client
205        .decrypt()
206        .key_id(key_id)
207        .ciphertext_blob(Blob::new(blob))
208        .send()
209        .await
210        .map_err(map_decrypt_err)?;
211
212    let plaintext = out.plaintext().ok_or_else(|| {
213        KeyProviderError::InvalidKeyFormat("KMS returned no plaintext".to_string())
214    })?;
215    plaintext_to_key(plaintext.as_ref())
216}
217
218/// Convert a decrypted plaintext byte slice into a 32-byte MEK.
219fn plaintext_to_key(bytes: &[u8]) -> Result<Zeroizing<[u8; 32]>, KeyProviderError> {
220    if bytes.len() != 32 {
221        return Err(KeyProviderError::InvalidKeyFormat(format!(
222            "KMS plaintext data key is {} bytes, expected 32",
223            bytes.len()
224        )));
225    }
226    let mut key = Zeroizing::new([0u8; 32]);
227    key.copy_from_slice(bytes);
228    Ok(key)
229}

231/// Map a KMS `Decrypt` SDK error into a typed [`KeyProviderError`] without
232/// leaking any request/response body.
233fn map_decrypt_err<R>(e: SdkError<DecryptError, R>) -> KeyProviderError {
234    match &e {
235        SdkError::TimeoutError(_) => {
236            KeyProviderError::Unavailable("KMS request timed out".to_string())
237        }
238        SdkError::DispatchFailure(_) => {
239            KeyProviderError::Unavailable("KMS transport/dispatch failure".to_string())
240        }
241        SdkError::ResponseError(_) => {
242            KeyProviderError::Unavailable("KMS returned an unparseable response".to_string())
243        }
244        SdkError::ServiceError(se) => {
245            let code = se.err().code().unwrap_or("Unknown");
246            if code.contains("AccessDenied")
247                || code.contains("NotAuthorized")
248                || code.contains("Unauthorized")
249                || code.contains("Disabled")
250                || code.contains("InvalidState")
251            {
252                KeyProviderError::AccessDenied(format!("KMS denied decrypt ({code})"))
253            } else if code.contains("Throttling")
254                || code.contains("Internal")
255                || code.contains("Unavailable")
256                || code.contains("DependencyTimeout")
257            {
258                KeyProviderError::Unavailable(format!("KMS transient error ({code})"))
259            } else {
260                KeyProviderError::Provider(Box::new(std::io::Error::other(format!(
261                    "KMS decrypt error ({code})"
262                ))))
263            }
264        }
265        _ => KeyProviderError::Provider(Box::new(std::io::Error::other(
266            "KMS decrypt failed".to_string(),
267        ))),
268    }
269}
270
271impl KeyProvider for KmsKeyProvider {
272    fn get_mek(&self) -> Result<Zeroizing<[u8; 32]>, KeyProviderError> {
273        self.block_on_decrypt()
274    }
275
276    fn get_key_version(&self, version: u32) -> Result<Zeroizing<[u8; 32]>, KeyProviderError> {
277        // The envelope model wraps a single data key. Only the current version
278        // is retrievable; any other version is a typed KeyNotFound (never a
279        // panic). Native KMS key-rotation history is a follow-up.
280        if version == CURRENT_VERSION || version == 0 {
281            self.get_mek()
282        } else {
283            Err(KeyProviderError::KeyNotFound)
284        }
285    }
286
287    fn provider_name(&self) -> &str {
288        "kms"
289    }
290
291    fn health_check(&self) -> Result<(), KeyProviderError> {
292        self.get_mek().map(|_| ())
293    }
294}
