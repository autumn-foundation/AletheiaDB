1//! Cipher abstraction and implementations for authenticated encryption.
2//!
3//! All ciphers produce output in the wire format: `[nonce][ciphertext][tag]`.
4
5use crate::encryption::EncryptionError;
6use rand::RngCore;
7use zeroize::Zeroizing;
8
9/// AEAD cipher for encrypting/decrypting data at rest.
10///
11/// All implementations prepend the nonce and append the auth tag to the output.
12pub trait Cipher: Send + Sync {
13    /// Encrypt `plaintext`, returning `[nonce || ciphertext || tag]`.
14    ///
15    /// `aad` is Additional Authenticated Data -- authenticated but not encrypted.
16    /// Pass `&[]` if no AAD is needed.
17    fn encrypt(&self, plaintext: &[u8], aad: &[u8]) -> Result<Vec<u8>, EncryptionError>;
18
19    /// Decrypt ciphertext produced by [`encrypt`](Cipher::encrypt).
20    ///
21    /// `aad` must match what was passed to `encrypt`, or decryption will fail.
22    fn decrypt(&self, ciphertext: &[u8], aad: &[u8]) -> Result<Vec<u8>, EncryptionError>;
23
24    /// Byte overhead added by encryption: nonce size + tag size.
25    fn overhead(&self) -> usize;
26
27    /// Numeric algorithm identifier for file headers / WAL entries.
28    /// - 0 = None/plaintext
29    /// - 1 = AES-256-GCM
30    /// - 2 = ChaCha20-Poly1305
31    fn algorithm_id(&self) -> u8;
32
33    /// Human-readable algorithm name for logging.
34    fn algorithm_name(&self) -> &'static str;
35}

39use aes_gcm::aead::Aead;
40use aes_gcm::{Aes256Gcm, KeyInit, Nonce as AesNonce};
41
42/// AES-256-GCM authenticated encryption cipher.
43///
44/// Nonce: 12 bytes (96 bits), Tag: 16 bytes (128 bits).
45/// Total overhead: 28 bytes per encrypted payload.
46///
47/// # Key hygiene (Issue #488 P2.1)
48///
49/// The `aes-gcm` dependency is compiled with its `zeroize` feature enabled (see
50/// `Cargo.toml`), so the expanded AES key schedule held by `inner` is wiped on
51/// drop. This matters for key rotation: when the old generation is retired
52/// (`IndexKeyring::retain_only`) its `Arc<dyn Cipher>` is dropped, and the
53/// retired key schedule must not linger un-zeroized in memory.
54pub struct Aes256GcmCipher {
55    inner: Aes256Gcm,
56}
57
58/// AES-256-GCM algorithm identifier.
59pub const AES_256_GCM_ID: u8 = 1;
60
61/// AES-256-GCM nonce size in bytes.
62const AES_NONCE_SIZE: usize = 12;
63
64/// AES-256-GCM tag size in bytes.
65const AES_TAG_SIZE: usize = 16;
66
67impl Aes256GcmCipher {
68    /// Create a new AES-256-GCM cipher from a 32-byte key.
69    ///
70    /// # Why?
71    /// Automatically leverages hardware acceleration (AES-NI) on x86_64 architectures,
72    /// providing line-rate encryption for high-throughput components like the WAL.
73    ///
74    /// ## Examples
75    /// ```
76    /// use aletheiadb::encryption::cipher::{Aes256GcmCipher, Cipher};
77    /// use zeroize::Zeroizing;
78    ///
79    /// let key = Zeroizing::new([0u8; 32]);
80    /// let cipher = Aes256GcmCipher::new(&key);
81    /// assert_eq!(cipher.algorithm_name(), "AES-256-GCM");
82    /// ```
83    pub fn new(key: &Zeroizing<[u8; 32]>) -> Self {
84        let inner = Aes256Gcm::new_from_slice(key.as_ref())
85            .expect("32-byte key is always valid for AES-256");
86        Self { inner }
87    }
88}
89
90impl Cipher for Aes256GcmCipher {
91    fn encrypt(&self, plaintext: &[u8], aad: &[u8]) -> Result<Vec<u8>, EncryptionError> {
92        // Generate random 12-byte nonce
93        let mut nonce_bytes = [0u8; AES_NONCE_SIZE];
94        rand::thread_rng().fill_bytes(&mut nonce_bytes);
95        let nonce = AesNonce::from_slice(&nonce_bytes);
96
97        // Build AAD payload
98        let payload = aes_gcm::aead::Payload {
99            msg: plaintext,
100            aad,
101        };
102
103        // Encrypt (produces ciphertext || tag)
104        let ct_with_tag = self
105            .inner
106            .encrypt(nonce, payload)
107            .map_err(|e| EncryptionError::EncryptFailed(e.to_string()))?;
108
109        // Wire format: [nonce:12][ciphertext:N][tag:16]
110        let mut output = Vec::with_capacity(AES_NONCE_SIZE + ct_with_tag.len());
111        output.extend_from_slice(&nonce_bytes);
112        output.extend_from_slice(&ct_with_tag);
113        Ok(output)
114    }
115
116    fn decrypt(&self, ciphertext: &[u8], aad: &[u8]) -> Result<Vec<u8>, EncryptionError> {
117        let min_len = AES_NONCE_SIZE + AES_TAG_SIZE;
118        if ciphertext.len() < min_len {
119            return Err(EncryptionError::InvalidCiphertext {
120                expected: min_len,
121                actual: ciphertext.len(),
122            });
123        }
124
125        let nonce = AesNonce::from_slice(&ciphertext[..AES_NONCE_SIZE]);
126        let ct_and_tag = &ciphertext[AES_NONCE_SIZE..];
127        let payload = aes_gcm::aead::Payload {
128            msg: ct_and_tag,
129            aad,
130        };
131
132        self.inner
133            .decrypt(nonce, payload)
134            .map_err(|e| EncryptionError::DecryptFailed(e.to_string()))
135    }
136
137    fn overhead(&self) -> usize {
138        AES_NONCE_SIZE + AES_TAG_SIZE // 28
139    }
140
141    fn algorithm_id(&self) -> u8 {
142        AES_256_GCM_ID
143    }
144
145    fn algorithm_name(&self) -> &'static str {
146        "AES-256-GCM"
147    }
148}

155/// ChaCha20-Poly1305 authenticated encryption cipher.
156///
157/// Nonce: 12 bytes, Tag: 16 bytes. Total overhead: 28 bytes.
158/// Preferred on platforms without AES-NI hardware acceleration.
159pub struct ChaCha20Poly1305Cipher {
160    inner: ChaChaInner,

163/// ChaCha20-Poly1305 algorithm identifier.

167const CHACHA_NONCE_SIZE: usize = 12;
168
169/// Poly1305 tag size in bytes.
170const CHACHA_TAG_SIZE: usize = 16;
171
172impl ChaCha20Poly1305Cipher {
173    /// Create a new ChaCha20-Poly1305 cipher from a 32-byte key.
174    ///
175    /// # Why?
176    /// Serves as the high-performance fallback for ARM/Edge deployments where AES-NI
177    /// hardware acceleration might not be available, ensuring predictable performance.
178    ///
179    /// ## Examples
180    /// ```
181    /// use aletheiadb::encryption::cipher::{ChaCha20Poly1305Cipher, Cipher};
182    /// use zeroize::Zeroizing;

189        let inner = ChaChaInner::new_from_slice(key.as_ref())
190            .expect("32-byte key is always valid for ChaCha20");
191        Self { inner }
192    }
193}

263    /// Issue #488 P2.1: the retired-cipher key schedule is zeroized on drop.
264    /// There is no public API to observe the wipe, so this test's real job is to
265    /// keep the `zeroize` code path COMPILED IN: it constructs and drops both
266    /// AEAD ciphers, which only links when `aes-gcm`'s `zeroize` feature (and
267    /// ChaCha's unconditional zeroize) are present. A regression that dropped
268    /// the feature from `Cargo.toml` would not fail here, but the accompanying
269    /// `Cargo.toml` note and this smoke test document the requirement.
270    #[test]
271    fn ciphers_construct_and_drop_with_zeroize_compiled_in() {
272        let key = test_key();
273        {
274            let c = Aes256GcmCipher::new(&key);
275            let _ = c.encrypt(b"x", b"aad").unwrap();
276        } // AES key schedule zeroized here.
277        {
278            let c = ChaCha20Poly1305Cipher::new(&key);
279            let _ = c.encrypt(b"x", b"aad").unwrap();
280        } // ChaCha key schedule zeroized here.
281    }
