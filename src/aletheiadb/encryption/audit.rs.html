1//! Structured audit logging for encryption operations (Issue #489).
2//!
3//! Emits compliance-friendly, structured **JSON-lines** for key-management
4//! events and (optionally) individual encrypt/decrypt operations. Each line
5//! matches the schema:
6//!
7//! ```json
8//! {
9//!   "timestamp": "2024-01-15T10:30:00.123Z",
10//!   "event": "key.loaded",
11//!   "level": "info",
12//!   "data": { "key_version": 1, "provider_type": "file" },
13//!   "instance_id": "node-1",
14//!   "correlation_id": "evt-1a2b3c4d"
15//! }
16//! ```
17//!
18//! # Security invariants
19//!
20//! The audit path **never** logs key material (the MEK, DEKs, `Zeroizing`
21//! buffers, or raw key bytes). Only opaque descriptors (provider name, key
22//! version, error *category*) are emitted. Error strings are reduced to a
23//! stable category token so that key file paths or other secrets present in a
24//! raw error `Display` can never leak into the audit log.
25//!
26//! # Fail-safe
27//!
28//! An audit-logging failure (for example, the destination log file cannot be
29//! opened) must never crash or block the database. Write failures degrade
30//! gracefully: a one-line warning is printed to stderr and execution
31//! continues.
32

38/// Level of audit logging.
39#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, PartialOrd, Ord)]
40#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
41#[cfg_attr(feature = "serde", serde(rename_all = "snake_case"))]
42pub enum AuditLevel {
43    /// No audit logging.
44    None,
45    /// Log only key management events (load, rotate, access-denied).
46    #[default]
47    KeyEvents,
48    /// Log all encryption operations (high volume).
49    AllOperations,
50}
51
52/// An encryption audit event.
53#[derive(Debug, Clone)]
54pub enum AuditEvent {
55    /// Master key loaded at startup.
56    KeyLoaded {
57        /// Name of the key provider (e.g., "file", "env").
58        provider: String,
59        /// Key version that was loaded.
60        key_version: u32,
61    },
62    /// Key rotation started.
63    RotationStarted {
64        /// Previous key version.
65        old_version: u32,
66        /// New key version being rotated to.
67        new_version: u32,
68    },
69    /// Key rotation completed successfully.
70    RotationCompleted {
71        /// The new active key version.
72        new_version: u32,
73        /// Time taken for the rotation in milliseconds.
74        duration_ms: u64,
75    },
76    /// Key rotation failed.
77    RotationFailed {
78        /// Key version that failed to rotate.
79        version: u32,
80        /// Error description.
81        error: String,
82    },
83    /// Key access denied / key load failed by the provider.
84    KeyAccessDenied {
85        /// Name of the key provider.
86        provider: String,
87        /// Stable, key-safe error category (never raw key material or paths).
88        error: String,
89    },
90    // TODO(#489 follow-up): `EncryptOperation` / `DecryptOperation` are
91    // forward-scaffolding for the deferred high-volume `all_operations` audit
92    // path. They are only emitted once #481's storage encrypt/decrypt sites
93    // call `log(..)` with these variants; until then this surface is
94    // intentionally unwired (dead) rather than silently absent.
95    /// Encryption operation performed (high volume, only at `AllOperations` level).
96    EncryptOperation {
97        /// Component that performed the operation (e.g., "wal", "index").
98        component: String,
99        /// Key version used.
100        key_version: u32,
101    },
102    /// Decryption operation performed (high volume, only at `AllOperations` level).
103    DecryptOperation {
104        /// Component that performed the operation.
105        component: String,
106        /// Key version used.
107        key_version: u32,
108    },
109}
110
111impl AuditEvent {
112    /// Minimum [`AuditLevel`] at which this event is emitted.
113    fn required_level(&self) -> AuditLevel {
114        match self {
115            AuditEvent::KeyLoaded { .. }
116            | AuditEvent::RotationStarted { .. }
117            | AuditEvent::RotationCompleted { .. }
118            | AuditEvent::RotationFailed { .. }
119            | AuditEvent::KeyAccessDenied { .. } => AuditLevel::KeyEvents,
120            AuditEvent::EncryptOperation { .. } | AuditEvent::DecryptOperation { .. } => {
121                AuditLevel::AllOperations
122            }
123        }
124    }
125
126    /// Stable dotted event name for the `event` field.
127    fn event_name(&self) -> String {
128        match self {
129            AuditEvent::KeyLoaded { .. } => "key.loaded".to_string(),
130            AuditEvent::RotationStarted { .. } => "key.rotation.started".to_string(),
131            AuditEvent::RotationCompleted { .. } => "key.rotation.completed".to_string(),
132            AuditEvent::RotationFailed { .. } => "key.rotation.failed".to_string(),
133            AuditEvent::KeyAccessDenied { .. } => "key.access.denied".to_string(),
134            AuditEvent::EncryptOperation { component, .. } => format!("encrypt.{component}"),
135            AuditEvent::DecryptOperation { component, .. } => format!("decrypt.{component}"),
136        }
137    }
138
139    /// Severity string for the `level` field (info/error).
140    fn severity(&self) -> &'static str {
141        match self {
142            AuditEvent::RotationFailed { .. } | AuditEvent::KeyAccessDenied { .. } => "error",
143            _ => "info",
144        }
145    }
146
147    /// Structured `data` payload. Contains only opaque descriptors -- never key
148    /// material.
149    fn data(&self) -> serde_json::Value {
150        use serde_json::json;
151        match self {
152            AuditEvent::KeyLoaded {
153                provider,
154                key_version,
155            } => json!({ "key_version": key_version, "provider_type": provider }),
156            AuditEvent::RotationStarted {
157                old_version,
158                new_version,
159            } => json!({ "old_version": old_version, "new_version": new_version }),
160            AuditEvent::RotationCompleted {
161                new_version,
162                duration_ms,
163            } => json!({ "version": new_version, "duration_ms": duration_ms }),
164            AuditEvent::RotationFailed { version, error } => {
165                json!({ "version": version, "error": error })
166            }
167            AuditEvent::KeyAccessDenied { provider, error } => {
168                json!({ "provider": provider, "error": error })
169            }
170            AuditEvent::EncryptOperation {
171                component,
172                key_version,
173            }
174            | AuditEvent::DecryptOperation {
175                component,
176                key_version,
177            } => json!({ "component": component, "key_version": key_version }),
178        }
179    }
180}

182/// Where audit lines are written.
183#[derive(Debug, Clone)]
184enum AuditSink {
185    /// Write to standard output.
186    Stdout,
187    /// Write to standard error (backwards-compatible default).
188    Stderr,
189    /// Append to a file at the given path. The second field lazily caches the
190    /// open file handle (created + appended, `0600` on Unix) so we do not
191    /// reopen the file on every write. The `Arc<Mutex<..>>` keeps the sink
192    /// `Clone` and shares one handle across clones/threads.
193    File(PathBuf, Arc<Mutex<Option<std::fs::File>>>),
194    /// Syslog destination -- not yet implemented; falls back to stderr with a
195    /// note (documented follow-up).
196    Syslog,
197}

199/// Audit logger for encryption operations.
200///
201/// Filters events based on the configured [`AuditLevel`] and emits structured
202/// JSON lines to the configured destination.
203pub struct EncryptionAuditLogger {
204    level: AuditLevel,
205    instance_id: String,
206    sink: AuditSink,
207    /// When set, emitted lines are captured here instead of written to `sink`.
208    /// Only populated by test helpers; always `None` in production.
209    capture: Option<Arc<Mutex<Vec<String>>>>,
210}
211
212impl EncryptionAuditLogger {
213    /// Create a new audit logger with the given level and instance identifier.
214    ///
215    /// Writes to stderr (backwards-compatible default). Use
216    /// [`from_audit_config`](Self::from_audit_config) to honor a destination.
217    ///
218    /// # Why?
219    /// The `instance_id` allows centralized logging platforms to differentiate
220    /// encryption events coming from multiple database nodes in a clustered environment.
221    ///
222    /// ## Examples
223    /// ```
224    /// use aletheiadb::encryption::audit::{EncryptionAuditLogger, AuditLevel};
225    ///
226    /// let logger = EncryptionAuditLogger::new(AuditLevel::KeyEvents, "node-1");
227    /// assert_eq!(logger.level(), AuditLevel::KeyEvents);
228    /// ```
229    #[must_use]
230    pub fn new(level: AuditLevel, instance_id: impl Into<String>) -> Self {
231        Self {
232            level,
233            instance_id: instance_id.into(),
234            sink: AuditSink::Stderr,
235            capture: None,
236        }
237    }
238
239    /// Create a disabled audit logger that swallows all events.
240    ///
241    /// # Why?
242    /// Provides a zero-cost abstraction for configurations where audit logging
243    /// is disabled, avoiding `Option` unwrapping on every database operation.
244    ///
245    /// ## Examples
246    /// ```
247    /// use aletheiadb::encryption::audit::{EncryptionAuditLogger, AuditLevel};
248    ///
249    /// let logger = EncryptionAuditLogger::disabled();
250    /// assert_eq!(logger.level(), AuditLevel::None);
251    /// ```
252    #[must_use]
253    pub fn disabled() -> Self {
254        Self::new(AuditLevel::None, "")
255    }
256
257    /// Build a logger from an [`AuditConfig`].
258    ///
259    /// When `config.enabled` is `false`, the returned logger is
260    /// [`disabled`](Self::disabled) (level `None`) and produces **zero** output,
261    /// preserving backwards-compatible behavior for databases without an
262    /// `[encryption.audit]` block.
263    ///
264    /// # Why?
265    /// Centralizes translation from declarative config (destination, file path,
266    /// level) into a ready-to-use logger, so callers (the encryption manager)
267    /// don't reimplement destination resolution.
268    #[must_use]
269    pub fn from_audit_config(config: &AuditConfig) -> Self {
270        if !config.enabled {
271            return Self::disabled();
272        }
273
274        let sink = match config.destination {
275            AuditDestination::Stdout => AuditSink::Stdout,
276            AuditDestination::File => match &config.file_path {
277                Some(path) => AuditSink::File(path.clone(), Arc::new(Mutex::new(None))),
278                None => {
279                    eprintln!(
280                        "[AUDIT] warning: destination=file but no file_path configured; falling back to stdout"
281                    );
282                    AuditSink::Stdout
283                }
284            },
285            AuditDestination::Syslog => AuditSink::Syslog,
286        };
287
288        let instance_id = config
289            .instance_id
290            .clone()
291            .unwrap_or_else(default_instance_id);
292
293        Self {
294            level: config.level,
295            instance_id,
296            sink,
297            capture: None,
298        }
299    }

301    /// Get the configured audit level.
302    ///
303    /// # Why?
304    /// Allows external components (like the query planner) to check if detailed
305    /// logging is active, potentially optimizing away expensive telemetry data generation.
306    ///
307    /// ## Examples
308    /// ```
309    /// use aletheiadb::encryption::audit::{EncryptionAuditLogger, AuditLevel};
310    ///
311    /// let logger = EncryptionAuditLogger::disabled();
312    /// assert_eq!(logger.level(), AuditLevel::None);
313    /// ```
314    #[must_use]
315    pub fn level(&self) -> AuditLevel {
316        self.level
317    }
318
319    /// Returns `true` if the given event would be emitted at the current level.
320    #[must_use]
321    pub fn is_enabled_for(&self, event: &AuditEvent) -> bool {
322        self.level != AuditLevel::None && self.level >= event.required_level()
323    }
324
325    /// Format an event as a structured JSON line, or `None` if the current
326    /// [`AuditLevel`] suppresses it.
327    ///
328    /// # Why?
329    /// Separating formatting from emission lets callers (and tests) inspect the
330    /// exact serialized payload -- crucial for verifying that no key material
331    /// ever appears in the output.
332    #[must_use]
333    pub fn format_entry(&self, event: &AuditEvent) -> Option<String> {
334        if !self.is_enabled_for(event) {
335            return None;
336        }
337
338        let entry = serde_json::json!({
339            "timestamp": now_rfc3339_millis(),
340            "event": event.event_name(),
341            "level": event.severity(),
342            "data": event.data(),
343            "instance_id": self.instance_id,
344            "correlation_id": new_correlation_id(),
345        });
346
347        Some(entry.to_string())
348    }
349
350    /// Log an audit event if the configured level permits it.
351    ///
352    /// Emission is **fail-safe**: a write error (e.g. an unopenable log file)
353    /// is reported to stderr and swallowed; it never panics or propagates.
354    ///
355    /// # Why?
356    /// Centralizes all the filtering and formatting logic, ensuring that sensitive
357    /// keys are never accidentally logged, and that log structures remain consistent.
358    ///
359    /// ## Examples
360    /// ```
361    /// use aletheiadb::encryption::audit::{EncryptionAuditLogger, AuditLevel, AuditEvent};
362    ///
363    /// let logger = EncryptionAuditLogger::new(AuditLevel::KeyEvents, "node-1");
364    /// let event = AuditEvent::KeyLoaded {
365    ///     provider: "file".to_string(),
366    ///     key_version: 1,
367    /// };
368    ///
369    /// logger.log(&event);
370    /// ```
371    pub fn log(&self, event: &AuditEvent) {
372        if let Some(line) = self.format_entry(event) {
373            self.emit(&line);
374        }
375    }
376
377    /// Write a formatted line to the configured sink, degrading gracefully on
378    /// failure.
379    fn emit(&self, line: &str) {
380        if let Some(buf) = &self.capture {
381            if let Ok(mut guard) = buf.lock() {
382                guard.push(line.to_string());
383            }
384            return;
385        }
386
387        use std::io::Write;
388        match &self.sink {
389            // Fail-safe: `writeln!` onto a locked handle swallows I/O errors
390            // (e.g. EPIPE / broken pipe) instead of panicking like the
391            // `println!`/`eprintln!` macros would -- an audit write must never
392            // crash the DB, and stdout is the default destination.
393            AuditSink::Stdout => {
394                let _ = writeln!(std::io::stdout().lock(), "{line}");
395            }
396            AuditSink::Stderr => {
397                let _ = writeln!(std::io::stderr().lock(), "{line}");
398            }
399            AuditSink::Syslog => {
400                // Not yet implemented -- documented follow-up (#489). Degrade to
401                // stderr so events are never silently dropped (fail-safe write).
402                let _ = writeln!(std::io::stderr().lock(), "{line}");
403            }
404            AuditSink::File(path, handle) => {
405                // Lazily open + cache the handle on first write, reuse
406                // thereafter. All I/O errors degrade to a fail-safe stderr
407                // warning; the DB keeps running.
408                let mut guard = match handle.lock() {
409                    Ok(g) => g,
410                    // A prior panic could poison the lock; recover the handle
411                    // rather than propagate the panic through the audit path.
412                    Err(poisoned) => poisoned.into_inner(),
413                };
414                if guard.is_none() {
415                    match open_audit_file(path) {
416                        Ok(f) => *guard = Some(f),
417                        Err(e) => {
418                            audit_warn(&format!(
419                                "failed to open audit log {}: {e}",
420                                path.display()
421                            ));
422                            return;
423                        }
424                    }
425                }
426                let write_res = guard.as_mut().map(|f| writeln!(f, "{line}"));
427                if let Some(Err(e)) = write_res {
428                    // Drop the cached handle so the next write retries a fresh
429                    // open instead of reusing a broken descriptor.
430                    *guard = None;
431                    audit_warn(&format!(
432                        "failed to write audit log to {}: {e}",
433                        path.display()
434                    ));
435                }
436            }
437        }
438    }
439}

441/// Open (create + append) the audit log file, restricting permissions to
442/// `0600` on Unix so audit records are not world-readable. Mirrors the posture
443/// of `FileKeyProvider::generate_key_file`. The mode is applied only when the
444/// file is created; a pre-existing file keeps its permissions.
445fn open_audit_file(path: &std::path::Path) -> std::io::Result<std::fs::File> {
446    let mut opts = std::fs::OpenOptions::new();
447    opts.create(true).append(true);
448    #[cfg(unix)]
449    {
450        use std::os::unix::fs::OpenOptionsExt;
451        opts.mode(0o600);
452    }
453    opts.open(path)
454}
455
456/// Emit a fail-safe `[AUDIT]` warning to stderr. Uses a locked `writeln!` so a
457/// broken stderr pipe cannot panic (unlike `eprintln!`) -- and is never called
458/// while holding the file-handle lock across a panic path.
459fn audit_warn(msg: &str) {
460    use std::io::Write;
461    let _ = writeln!(std::io::stderr().lock(), "[AUDIT] warning: {msg}");
462}

464/// Default per-process instance identifier when none is configured.
465fn default_instance_id() -> String {
466    format!("aletheiadb-{}", std::process::id())
467}

469/// Current UTC time as an RFC 3339 string with millisecond precision and a `Z`
470/// suffix (e.g. `2024-01-15T10:30:00.123Z`).
471fn now_rfc3339_millis() -> String {
472    chrono::Utc::now().to_rfc3339_opts(chrono::SecondsFormat::Millis, true)
473}

475/// Generate a per-event correlation id (`evt-<8 hex>`).
476fn new_correlation_id() -> String {
477    let n: u32 = rand::random();
478    format!("evt-{n:08x}")
479}

645    /// Security assertion: audit lines must never contain raw key material, and
646    /// the check is *meaningful* -- the events below are constructed from data
647    /// DERIVED from a recognizable secret, so a regression that routed
648    /// key-derived bytes into a log line would reintroduce `secret_hex` and
649    /// fail this test.
650    #[test]
651    fn no_key_material_in_output() {
652        // A recognizable 32-byte "key". Its full hex rendering is the pattern
653        // that must NEVER surface in an audit line.
654        let secret: [u8; 32] = [
655            0xaa, 0xbb, 0xcc, 0xdd, 0xee, 0xff, 0x00, 0x11, 0x22, 0x33, 0x44, 0x55, 0x66, 0x77,
656            0x88, 0x99, 0xaa, 0xbb, 0xcc, 0xdd, 0xee, 0xff, 0x00, 0x11, 0x22, 0x33, 0x44, 0x55,
657            0x66, 0x77, 0x88, 0x99,
658        ];
659        let secret_hex: String = secret.iter().map(|b| format!("{b:02x}")).collect();
660
661        // Derive the events' *opaque* inputs from the secret: the key version is
662        // projected from a key byte (a safe, non-sensitive descriptor) while the
663        // raw key bytes themselves must be dropped on the audit path. If a
664        // regression logged the source key buffer instead of the projected
665        // descriptor, `secret_hex` would appear below.
666        let key_version = u32::from(secret[0]);
667        let (logger, buf) = EncryptionAuditLogger::with_capture(AuditLevel::KeyEvents, "node-1");
668        logger.log(&AuditEvent::KeyLoaded {
669            provider: "file".into(),
670            key_version,
671        });
672        // The denied path carries only a stable category token -- never the raw,
673        // secret-bearing provider error.
674        logger.log(&AuditEvent::KeyAccessDenied {
675            provider: "file".into(),
676            error: "key_decrypt_failed".into(),
677        });
678
679        let lines = buf.lock().unwrap();
680        assert_eq!(lines.len(), 2, "both key events emitted");
681        for line in lines.iter() {
682            assert!(
683                !line.contains(&secret_hex),
684                "audit line must not contain key material: {line}"
685            );
686        }
687        // The only data fields are opaque descriptors.
688        assert!(lines[0].contains("provider_type"));
689        assert!(lines[0].contains("key_version"));
690    }

742    #[test]
743    fn file_destination_caches_handle_and_appends() {
744        // Write two events to a real file sink sharing one lazily-opened,
745        // cached handle; both lines must be appended (handle reused, not
746        // truncated on the second write).
747        let dir = std::env::temp_dir().join(format!("aletheiadb-audit-{}", std::process::id()));
748        std::fs::create_dir_all(&dir).expect("mkdir");
749        let path = dir.join("cached-audit.log");
750        let _ = std::fs::remove_file(&path);
751
752        let logger = EncryptionAuditLogger {
753            level: AuditLevel::KeyEvents,
754            instance_id: "node-1".into(),
755            sink: AuditSink::File(path.clone(), Arc::new(Mutex::new(None))),
756            capture: None,
757        };
758        logger.log(&key_loaded());
759        logger.log(&AuditEvent::RotationStarted {
760            old_version: 1,
761            new_version: 2,
762        });
763
764        let contents = std::fs::read_to_string(&path).expect("read audit log");
765        let lines: Vec<&str> = contents.lines().collect();
766        assert_eq!(lines.len(), 2, "both events appended to the cached handle");
767        assert!(lines[0].contains("key.loaded"));
768        assert!(lines[1].contains("key.rotation.started"));
769
770        // On Unix the file is created with 0600 (owner-only) permissions.
771        #[cfg(unix)]
772        {
773            use std::os::unix::fs::PermissionsExt;
774            let mode = std::fs::metadata(&path).expect("stat").permissions().mode();
775            assert_eq!(mode & 0o777, 0o600, "audit log must be owner-only (0600)");
776        }
777
778        let _ = std::fs::remove_file(&path);
779        let _ = std::fs::remove_dir(&dir);
780    }
