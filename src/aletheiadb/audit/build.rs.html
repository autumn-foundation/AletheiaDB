1//! Building a signed audit export from live database history (Issue #3358).
2//!
3//! This is a read-only path: it consumes the existing history reads
4//! (`get_node_history` / `get_edge_history`) plus current adjacency for the
5//! neighborhood scope, and never mutates the database — so it takes no
6//! write-path locks.
7

26/// A reference to a single node or edge for export.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub enum EntityRef {
29    /// A node, by id.
30    Node(NodeId),
31    /// An edge, by id.
32    Edge(EdgeId),
33}
34
35/// What to export (AC5): a single entity, an explicit set, or a neighborhood.
36#[derive(Debug, Clone)]
37pub enum AuditScope {
38    /// A single node or edge.
39    SingleEntity(EntityRef),
40    /// An explicit list of entities.
41    EntitySet(Vec<EntityRef>),
42    /// A node plus its N-hop neighborhood at a bi-temporal coordinate.
43    Neighborhood {
44        /// The entity to start from.
45        start: NodeId,
46        /// Hop radius.
47        hops: u32,
48        /// Valid-time coordinate (defaults to now when omitted).
49        valid_time: Option<Timestamp>,
50        /// Transaction-time coordinate (defaults to now when omitted).
51        transaction_time: Option<Timestamp>,
52    },
53}
54
55impl AuditScope {
56    /// Convenience: export a single node.
57    #[must_use]
58    pub fn node(id: NodeId) -> Self {
59        AuditScope::SingleEntity(EntityRef::Node(id))
60    }
61
62    /// Convenience: export a single edge.
63    #[must_use]
64    pub fn edge(id: EdgeId) -> Self {
65        AuditScope::SingleEntity(EntityRef::Edge(id))
66    }
67}

69/// Options controlling an export: database identity + redaction (AC6/AC8).
70#[derive(Debug, Clone)]
71pub struct ExportOptions {
72    database_id: String,
73    redact_keys: Vec<String>,
74}
75
76impl ExportOptions {
77    /// New options with the given operator-supplied database identity.
78    #[must_use]
79    pub fn new(database_id: impl Into<String>) -> Self {
80        Self {
81            database_id: database_id.into(),
82            redact_keys: Vec::new(),
83        }
84    }
85
86    /// Redact the given property keys at export (their values are omitted, but
87    /// the redaction is recorded and verifiable — never silent).
88    #[must_use]
89    pub fn redact<I, S>(mut self, keys: I) -> Self
90    where
91        I: IntoIterator<Item = S>,
92        S: Into<String>,
93    {
94        self.redact_keys
95            .extend(keys.into_iter().map(std::convert::Into::into));
96        self
97    }
98}

101    /// Produce a signed, self-contained audit export of an entity's (or a set
102    /// or neighborhood's) complete bi-temporal history (Issue #3358).
103    ///
104    /// The result verifies OFFLINE with only the signer's public key. See the
105    /// [`crate::audit`] module docs and `docs/guides/audit-export.md` for the
106    /// artifact format and verification contract.
107    ///
108    /// # Errors
109    /// Returns [`AuditError::NoHistory`] if a single requested entity has no
110    /// retrievable history, or [`AuditError::DatabaseRead`] on a read failure.
111    pub fn audit_export(
112        &self,
113        scope: AuditScope,
114        signing_key: &AuditSigningKey,
115        options: &ExportOptions,
116    ) -> AuditResult<AuditExport> {
117        let redact: BTreeSet<String> = options.redact_keys.iter().cloned().collect();
118
119        let (kind_str, requested, entities, not_included, coordinate, hops) =
120            self.resolve_scope(&scope, &redact)?;
121
122        let included: Vec<EntityIdRecord> = entities
123            .iter()
124            .map(|e| EntityIdRecord {
125                kind: e.kind.clone(),
126                id: e.id,
127            })
128            .collect();
129
130        let version_count: usize = entities.iter().map(|e| e.versions.len()).sum();
131        let mut redacted_keys: Vec<String> = redact.into_iter().collect();
132        redacted_keys.sort();
133
134        let metadata = ExportMetadata {
135            database_id: options.database_id.clone(),
136            exported_at: format_timestamp(time::now()),
137            tool_version: env!("CARGO_PKG_VERSION").to_string(),
138            scope_description: describe_scope(&scope),
139            scope: ScopeRecord {
140                kind: kind_str,
141                requested,
142                included,
143                not_included,
144                coordinate,
145                hops,
146            },
147            chain_anchor: ChainAnchor {
148                source_lsn: self.stats().wal.current_lsn,
149                description: "WAL LSN at export time (chain-head position, #3351 anchor)"
150                    .to_string(),
151            },
152            redacted_keys,
153            entity_count: entities.len(),
154            version_count,
155            hash_algorithm: HASH_ALGORITHM.to_string(),
156            signature_algorithm: SIGNATURE_ALGORITHM.to_string(),
157        };
158
159        // Integrity: per-version leaves + folded chain root, then sign the root.
160        let (leaves, root) = canonical::compute_chain(&metadata, &entities);
161        let signature = signing_key.sign_root(&root);
162        let public_key = signing_key.public_key();
163
164        Ok(AuditExport {
165            format_version: AUDIT_FORMAT_VERSION,
166            metadata,
167            entities,
168            chain: ChainProof {
169                algorithm: HASH_ALGORITHM.to_string(),
170                leaves: leaves.iter().map(canonical::to_hex).collect(),
171                root: canonical::to_hex(&root),
172            },
173            signature: SignatureBlock {
174                algorithm: SIGNATURE_ALGORITHM.to_string(),
175                public_key: public_key.to_hex(),
176                signature: crate::core::hex::encode(&signature),
177                signed: SIGNED_DESCRIPTION.to_string(),
178            },
179        })
180    }

182    /// Resolve a scope into the concrete entities to export plus scope metadata.
183    #[allow(clippy::type_complexity)]
184    fn resolve_scope(
185        &self,
186        scope: &AuditScope,
187        redact: &BTreeSet<String>,
188    ) -> AuditResult<(
189        String,
190        Vec<EntityIdRecord>,
191        Vec<ExportedEntity>,
192        Vec<NotIncludedRecord>,
193        Option<CoordinateRecord>,
194        Option<u32>,
195    )> {
196        match scope {
197            AuditScope::SingleEntity(entity_ref) => {
198                let kind = match entity_ref {
199                    EntityRef::Node(_) => "single_node",
200                    EntityRef::Edge(_) => "single_edge",
201                };
202                let requested = vec![entity_id_record(*entity_ref)];
203                let exported = self
204                    .export_one(*entity_ref, redact)?
205                    .ok_or_else(|| AuditError::NoHistory(describe_entity(*entity_ref)))?;
206                Ok((
207                    kind.to_string(),
208                    requested,
209                    vec![exported],
210                    Vec::new(),
211                    None,
212                    None,
213                ))
214            }
215            AuditScope::EntitySet(refs) => {
216                let requested: Vec<EntityIdRecord> =
217                    refs.iter().map(|r| entity_id_record(*r)).collect();
218                let mut entities = Vec::new();
219                let mut not_included = Vec::new();
220                for r in refs {
221                    match self.export_one(*r, redact)? {
222                        Some(e) => entities.push(e),
223                        None => {
224                            let rec = entity_id_record(*r);
225                            not_included.push(NotIncludedRecord {
226                                kind: rec.kind,
227                                id: rec.id,
228                                reason: "no_history".to_string(),
229                            });
230                        }
231                    }
232                }
233                Ok((
234                    "entity_set".to_string(),
235                    requested,
236                    entities,
237                    not_included,
238                    None,
239                    None,
240                ))
241            }
242            AuditScope::Neighborhood {
243                start,
244                hops,
245                valid_time,
246                transaction_time,
247            } => {
248                let vt = valid_time.unwrap_or_else(time::now);
249                let tt = transaction_time.unwrap_or_else(time::now);
250                let (node_ids, edge_ids) = self.collect_neighborhood(*start, *hops, vt, tt);
251
252                let mut entities = Vec::new();
253                for n in node_ids {
254                    if let Some(e) = self.export_one(EntityRef::Node(n), redact)? {
255                        entities.push(e);
256                    }
257                }
258                for ed in edge_ids {
259                    if let Some(e) = self.export_one(EntityRef::Edge(ed), redact)? {
260                        entities.push(e);
261                    }
262                }
263                let coordinate = Some(CoordinateRecord {
264                    valid_time: Some(format_timestamp(vt)),
265                    transaction_time: Some(format_timestamp(tt)),
266                });
267                Ok((
268                    "neighborhood".to_string(),
269                    vec![entity_id_record(EntityRef::Node(*start))],
270                    entities,
271                    Vec::new(),
272                    coordinate,
273                    Some(*hops),
274                ))
275            }
276        }
277    }

279    /// Export a single entity's full history, or `None` if it has no history.
280    fn export_one(
281        &self,
282        entity_ref: EntityRef,
283        redact: &BTreeSet<String>,
284    ) -> AuditResult<Option<ExportedEntity>> {
285        match entity_ref {
286            EntityRef::Node(id) => {
287                let history = match self.get_node_history(id) {
288                    Ok(h) => h,
289                    Err(_) => return Ok(None),
290                };
291                if history.version_count() == 0 {
292                    return Ok(None);
293                }
294                let label = history
295                    .current_version()
296                    .or_else(|| history.first_version())
297                    .map(|v| v.label.clone());
298                Ok(Some(ExportedEntity {
299                    kind: "node".to_string(),
300                    id: id.as_u64(),
301                    label,
302                    source: None,
303                    target: None,
304                    versions: convert_versions(&history, redact),
305                }))
306            }
307            EntityRef::Edge(id) => {
308                let history = match self.get_edge_history(id) {
309                    Ok(h) => h,
310                    Err(_) => return Ok(None),
311                };
312                if history.version_count() == 0 {
313                    return Ok(None);
314                }
315                let label = history
316                    .current_version()
317                    .or_else(|| history.first_version())
318                    .map(|v| v.label.clone());
319                // Endpoints are informational; recover them if the edge still
320                // resolves (deleted edges leave them unknown → None).
321                let (source, target) = match self.get_edge(id) {
322                    Ok(edge) => (Some(edge.source.as_u64()), Some(edge.target.as_u64())),
323                    Err(_) => (None, None),
324                };
325                Ok(Some(ExportedEntity {
326                    kind: "edge".to_string(),
327                    id: id.as_u64(),
328                    label,
329                    source,
330                    target,
331                    versions: convert_versions(&history, redact),
332                }))
333            }
334        }
335    }

337    /// BFS the current graph from `start` up to `hops`, keeping only nodes/edges
338    /// valid at the given bi-temporal coordinate. Returns `(nodes, edges)`.
339    fn collect_neighborhood(
340        &self,
341        start: NodeId,
342        hops: u32,
343        vt: Timestamp,
344        tt: Timestamp,
345    ) -> (Vec<NodeId>, Vec<EdgeId>) {
346        let mut nodes: BTreeSet<u64> = BTreeSet::new();
347        let mut edges: BTreeSet<u64> = BTreeSet::new();
348        let mut frontier: Vec<NodeId> = Vec::new();
349
350        if self.get_node_at_time(start, vt, tt).is_ok() {
351            nodes.insert(start.as_u64());
352            frontier.push(start);
353        }
354
355        for _ in 0..hops {
356            let mut next = Vec::new();
357            for node in std::mem::take(&mut frontier) {
358                for edge_id in self
359                    .get_outgoing_edges(node)
360                    .into_iter()
361                    .chain(self.get_incoming_edges(node))
362                {
363                    if self.get_edge_at_time(edge_id, vt, tt).is_err() {
364                        continue;
365                    }
366                    edges.insert(edge_id.as_u64());
367                    for endpoint in [
368                        self.get_edge_source(edge_id).ok(),
369                        self.get_edge_target(edge_id).ok(),
370                    ]
371                    .into_iter()
372                    .flatten()
373                    {
374                        if self.get_node_at_time(endpoint, vt, tt).is_ok()
375                            && nodes.insert(endpoint.as_u64())
376                        {
377                            next.push(endpoint);
378                        }
379                    }
380                }
381            }
382            frontier = next;
383        }
384
385        let node_ids = nodes
386            .into_iter()
387            .filter_map(|id| NodeId::new(id).ok())
388            .collect();
389        let edge_ids = edges
390            .into_iter()
391            .filter_map(|id| EdgeId::new(id).ok())
392            .collect();
393        (node_ids, edge_ids)
394    }
395}

397/// Convert an entity's history into exported versions (oldest first).
398fn convert_versions(history: &EntityHistory, redact: &BTreeSet<String>) -> Vec<ExportedVersion> {
399    let now = time::now();
400    history
401        .versions
402        .iter()
403        .map(|v| convert_version(v, redact, now))
404        .collect()
405}

407fn convert_version(v: &VersionInfo, redact: &BTreeSet<String>, now: Timestamp) -> ExportedVersion {
408    let valid = v.temporal.valid_time();
409    let tx = v.temporal.transaction_time();
410
411    let (valid_to_micros, valid_to_logical) = closed_bound(valid.end());
412    let (tx_to_micros, tx_to_logical) = closed_bound(tx.end());
413    let is_current = tx.is_current() && valid.contains(now);
414
415    let mut properties: Vec<ExportedProperty> = v
416        .properties
417        .iter()
418        .map(|(key, value)| {
419            let key_str = GLOBAL_INTERNER
420                .resolve_with(*key, std::string::ToString::to_string)
421                .unwrap_or_else(|| "<unknown-key>".to_string());
422            if redact.contains(&key_str) {
423                ExportedProperty {
424                    key: key_str,
425                    value: None,
426                    redacted: true,
427                }
428            } else {
429                ExportedProperty {
430                    key: key_str,
431                    value: Some(property_value_to_exported(value)),
432                    redacted: false,
433                }
434            }
435        })
436        .collect();
437    properties.sort_by(|a, b| a.key.cmp(&b.key));
438
439    ExportedVersion {
440        version_number: v.version_number,
441        version_id: v.version_id.as_u64(),
442        label: v.label.clone(),
443        valid_from_micros: valid.start().wallclock(),
444        valid_from_logical: valid.start().logical(),
445        valid_to_micros,
446        valid_to_logical,
447        transaction_from_micros: tx.start().wallclock(),
448        transaction_from_logical: tx.start().logical(),
449        transaction_to_micros: tx_to_micros,
450        transaction_to_logical: tx_to_logical,
451        is_current,
452        provenance: v.provenance.as_ref().map(convert_provenance),
453        properties,
454    }
455}

457/// `None` for an open-ended (`TIMESTAMP_MAX`) bound, else its `(micros, logical)`.
458fn closed_bound(end: Timestamp) -> (Option<i64>, Option<u32>) {
459    if end == TIMESTAMP_MAX {
460        (None, None)
461    } else {
462        (Some(end.wallclock()), Some(end.logical()))
463    }
464}

466fn convert_provenance(p: &crate::core::provenance::Provenance) -> ExportedProvenance {
467    ExportedProvenance {
468        source: p.source().map(std::string::ToString::to_string),
469        confidence: p.confidence(),
470        note: p.note().map(std::string::ToString::to_string),
471        correlation_id: p.correlation_id().map(std::string::ToString::to_string),
472        principal: p.principal().map(std::string::ToString::to_string),
473    }
474}

476fn property_value_to_exported(value: &PropertyValue) -> ExportedValue {
477    match value {
478        PropertyValue::Null => ExportedValue::Null,
479        PropertyValue::Bool(b) => ExportedValue::Bool { value: *b },
480        PropertyValue::Int(i) => ExportedValue::Int { value: *i },
481        PropertyValue::Float(f) => ExportedValue::Float { value: *f },
482        PropertyValue::String(s) => ExportedValue::String {
483            value: s.to_string(),
484        },
485        PropertyValue::Bytes(b) => ExportedValue::Bytes {
486            hex: crate::core::hex::encode(b),
487        },
488        PropertyValue::Array(values) => ExportedValue::Array {
489            values: values.iter().map(property_value_to_exported).collect(),
490        },
491        PropertyValue::Vector(v) => ExportedValue::Vector {
492            values: v.iter().map(|f| f64_to_token(f64::from(*f))).collect(),
493        },
494        PropertyValue::SparseVector(sv) => ExportedValue::SparseVector {
495            dimension: sv.dimension(),
496            indices: sv.indices().to_vec(),
497            values: sv
498                .values()
499                .iter()
500                .map(|f| f64_to_token(f64::from(*f)))
501                .collect(),
502        },
503    }
504}

506fn entity_id_record(r: EntityRef) -> EntityIdRecord {
507    match r {
508        EntityRef::Node(id) => EntityIdRecord {
509            kind: "node".to_string(),
510            id: id.as_u64(),
511        },
512        EntityRef::Edge(id) => EntityIdRecord {
513            kind: "edge".to_string(),
514            id: id.as_u64(),
515        },
516    }
517}

519fn describe_entity(r: EntityRef) -> String {
520    match r {
521        EntityRef::Node(id) => format!("node {}", id.as_u64()),
522        EntityRef::Edge(id) => format!("edge {}", id.as_u64()),
523    }
524}
525
526fn describe_scope(scope: &AuditScope) -> String {
527    match scope {
528        AuditScope::SingleEntity(EntityRef::Node(id)) => {
529            format!("single_node: node {}", id.as_u64())
530        }
531        AuditScope::SingleEntity(EntityRef::Edge(id)) => {
532            format!("single_edge: edge {}", id.as_u64())
533        }
534        AuditScope::EntitySet(refs) => format!("entity_set: {} entities requested", refs.len()),
535        AuditScope::Neighborhood { start, hops, .. } => {
536            format!("neighborhood: {} hops around node {}", hops, start.as_u64())
537        }
538    }
539}

542/// outside chrono's representable range.

544    crate::audit::model::rfc3339_micros(ts.wallclock())
