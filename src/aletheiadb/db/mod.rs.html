116/// Main AletheiaDB database.
117///
118/// This is the primary entry point for interacting with the database.
119/// It coordinates between current storage (for fast current-state queries)
120/// and historical storage (for temporal queries).
121///
122/// # Durability Modes
123///
124/// AletheiaDB supports three durability modes for write transactions:
125///
126/// - **Synchronous** (default): Each commit waits for fsync. Maximum durability.
127/// - **Async**: Commits return immediately, background thread syncs. Fast but risk of data loss.
128/// - **GroupCommit**: Multiple commits share one fsync. ACID durability with high throughput.
129///
130/// Use [`with_unified_config`](Self::with_unified_config) to configure the database settings,
131/// including durability modes. You can also use [`write_with_options`](Self::write_with_options)
132/// for per-transaction overrides.
133///
134/// # Examples
135///
136/// ```rust,no_run
137/// use aletheiadb::{AletheiaDB, PropertyMapBuilder};
138/// use aletheiadb::index::vector::{HnswConfig, DistanceMetric};
139///
140/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
141/// // 1. Initialize the database
142/// let db = AletheiaDB::new().expect("Failed to open database");
143///
144/// // 2. Enable vector indexing (optional)
145/// db.vector_index("embedding")
146///     .hnsw(HnswConfig::new(384, DistanceMetric::Cosine))
147///     .enable()?;
148///
149/// // 3. Create nodes
150/// let alice_id = db.create_node(
151///     "Person",
152///     PropertyMapBuilder::new()
153///         .insert("name", "Alice")
154///         .insert("age", 30)
155///         .build()
156/// )?;
157///
158/// let bob_id = db.create_node(
159///     "Person",
160///     PropertyMapBuilder::new()
161///         .insert("name", "Bob")
162///         .insert_vector("embedding", &[0.1, 0.2, 0.3])
163///         .build()
164/// )?;
165///
166/// // 4. Create an edge
167/// db.create_edge(
168///     alice_id,
169///     bob_id,
170///     "KNOWS",
171///     PropertyMapBuilder::new().insert("since", 2024).build()
172/// )?;
173///
174/// // 5. Query
175/// let alice = db.get_node(alice_id)?;
176/// println!("Found: {:?}", alice.properties.get("name"));
177/// # Ok(())
178/// # }
179/// ```
180pub struct AletheiaDB {
181    /// Current state storage (hot path) - Arc-wrapped for sharing across transactions
182    pub(crate) current: Arc<CurrentStorage>,
183    // Lock ordering for write-path primitives:
184    // 1. `current_timestamp`
185    // 2. `wal`
186    // 3. `historical`
187    // 4. `temporal_indexes`
188    // 5. `id generators`
189    // 6. `outgoing`
190    // 7. `incoming`
191    //
192    // Some entries are currently lock-free or internally sharded, but future
193    // explicit locks must preserve this order to avoid deadlocks.
194    /// Historical version storage (temporal path) - RwLock-protected for concurrent reads
195    pub(crate) historical: Arc<RwLock<HistoricalStorage>>,
196    /// Temporal indexes for efficient time-based queries - Uses DashMap internally for fine-grained locking
197    pub(crate) temporal_indexes: Arc<TemporalIndexes>,
198    /// Concurrent Write-Ahead Log for durability - lock-free striped architecture
199    pub(crate) wal: Arc<ConcurrentWalSystem>,
200    /// In-flight (durable-but-not-yet-applied) LSN tracker (lost-write persist
201    /// race fix). Records the base LSN of every commit that is fsynced but not
202    /// yet applied to current/historical storage. Index persistence stamps the
203    /// manifest with `min(in_flight)` — the applied watermark — instead of the
204    /// WAL allocation frontier, so replay always re-covers a durable-but-
205    /// unapplied write instead of dropping it. Its internal mutex is a leaf
206    /// (never held while acquiring another write-path primitive).
207    pub(crate) in_flight: Arc<crate::api::transaction::write::InFlightLsns>,
208    /// Current logical timestamp for transaction time - Mutex-protected for thread-safe increment
209    pub(crate) current_timestamp: Arc<Mutex<Timestamp>>,
210    /// Monotonic observation time for adaptive forward clock-skew limits.
211    pub(crate) commit_clock_observed_at: Arc<Mutex<Instant>>,
212    /// Transaction ID generator for MVCC
213    pub(crate) tx_id_gen: Arc<TxIdGenerator>,
214    /// Transaction visibility manager for Snapshot Isolation
215    pub(crate) visibility_manager: Arc<TxVisibilityManager>,
216    /// ID generators for nodes, edges, and versions (shared with transactions)
217    /// IdGenerator uses AtomicU64 internally, so no external Mutex is needed.
218    pub(crate) node_id_gen: Arc<IdGenerator>,
219    pub(crate) edge_id_gen: Arc<IdGenerator>,
220    pub(crate) version_id_gen: Arc<IdGenerator>,
221    /// Default durability mode for write transactions
222    pub(crate) default_durability: DurabilityMode,
223    /// Query optimization statistics - cached across queries for effective cost-based optimization
224    pub(crate) stats: Arc<Statistics>,
225    /// Index persistence configuration (stored for potential future use)
226    #[allow(dead_code)]
227    pub(crate) persistence_config: crate::storage::index_persistence::PersistenceConfig,
228    /// Index persistence manager (if enabled)
229    pub(crate) persistence_manager:
230        Option<Arc<crate::storage::index_persistence::IndexPersistenceManager>>,
231    /// Persistence mutation tracking
232    pub(crate) persistence_tracker: Option<Arc<PersistenceTracker>>,
233    /// Background persistence thread health flag - set to true if thread panics or stops
234    pub(crate) persistence_thread_stopped: Arc<std::sync::atomic::AtomicBool>,
235    /// Background persistence thread handle (if enabled) - used to join thread on shutdown
236    pub(crate) persistence_thread_handle: Option<std::thread::JoinHandle<()>>,
237    /// Encryption manager (if encryption at rest is enabled)
238    pub(crate) encryption_manager: Option<Arc<crate::encryption::EncryptionManager>>,
239    /// Encryption configuration used at startup (retained for key rotation,
240    /// Issue #488: re-sourcing the current MEK to guard against rotating to the
241    /// same key). `None` when encryption is disabled.
242    pub(crate) encryption_config: Option<crate::encryption::config::EncryptionConfig>,
243    /// The AEAD algorithm from `config.encryption.algorithm`, retained REGARDLESS
244    /// of whether encryption is currently enabled (Issue #3616 PR3). A plaintext
245    /// database has no `encryption_config`, but `enable_encryption` must still write
246    /// its at-rest bytes under the operator's configured algorithm (and pin the
247    /// resolved concrete form into the authority) so the reopen — which builds its
248    /// ciphers from `config.encryption.algorithm` — reads them back under the
249    /// identical cipher. On the interrupted-enable resume path this also keeps the
250    /// wrap passes in lockstep with the ciphers `open()` built from the same config
251    /// algorithm.
252    pub(crate) configured_encryption_algorithm: crate::encryption::factory::Algorithm,
253    /// Uniqueness constraint registry (declarations + reservation index) plus
254    /// property-type / required-key schema constraints (Issue #3378).
255    pub(crate) constraint_registry: Arc<crate::core::constraint::ConstraintRegistry>,
256    /// On-disk path for the schema-constraint sidecar (Issue #3378), or `None`
257    /// for an ephemeral (in-memory) database. When `Some`, declaring/dropping
258    /// schema constraints atomically rewrites this file, and it is loaded at
259    /// startup. Lives under the data dir (the parent of the WAL dir), mirroring
260    /// the provenance-chain sidecar's placement.
261    pub(crate) schema_constraint_path: Option<std::path::PathBuf>,
262    /// GDPR crypto-shred state (Issue #3359): the per-subject key registry, its
263    /// durable paths, and the attestation signing key. In-memory-only for an
264    /// ephemeral database (no index persistence). Loaded fail-closed at startup
265    /// with breadcrumb crash-recovery.
266    #[cfg(feature = "audit-export")]
267    pub(crate) crypto_shred: Arc<crate::db::crypto_shred::CryptoShredState>,
268    /// Fact-to-fact derivation lineage index (Issue #3371).
269    ///
270    /// Records that a fact version was derived from a set of source fact
271    /// versions and answers upstream/downstream closure queries. Immutable,
272    /// append-only, and independent of the graph/version stores so retracting
273    /// or superseding a fact never disturbs lineage pointing at it. v1 is
274    /// in-memory (does not survive restart) to keep the WAL format untouched
275    /// (Issue #3413); see [`crate::core::lineage`].
276    pub(crate) lineage: Arc<crate::core::lineage::LineageStore>,
277    /// Push-changefeed broadcaster (Issue #3375).
278    ///
279    /// Fan-out hub for [`AletheiaDB::subscribe_changes`]. Each committed write transaction
280    /// broadcasts its change set (byte-identical to what `list_changes` would return for
281    /// that window) to matching subscribers, after the write is durable + applied + visible
282    /// and outside every write-path lock. Best-effort at-least-once: the durable ground
283    /// truth is `list_changes`, and a lagged/reconnecting/crash-surviving consumer resumes
284    /// losslessly from its last cursor. Its internal locks are leaves (never held while
285    /// acquiring a write-path primitive).
286    pub(crate) changefeed: Arc<crate::core::changefeed_subscription::ChangefeedBroadcaster>,
287    /// Named-snapshot registry for reproducible reads (Issue #3370).
288    ///
289    /// Maps a human-readable name to a pinned bi-temporal coordinate. Entirely
290    /// off the data write path (creating a snapshot never touches
291    /// current/historical storage, the WAL, or `current_timestamp`), so it
292    /// adds zero overhead to `create_node`/`create_edge`. Durably persisted to
293    /// a `{data_dir}/snapshots.json` sidecar when index persistence is enabled;
294    /// in-memory-only for ephemeral databases. See [`crate::db::snapshot`].
295    pub(crate) snapshots: Arc<snapshot::SnapshotRegistry>,
296    /// Agent-scoped namespace registry (Issue #3349).
297    ///
298    /// Maps a namespace name to its `{description, created_at}` metadata so
299    /// namespaces are creatable / listable / describable even when empty.
300    /// Auto-registered on first write to an unknown namespace. Entirely off the
301    /// data write path (a leaf, like [`snapshots`](Self::snapshots)). Durably
302    /// persisted to a `{data_dir}/namespaces.json` sidecar when index
303    /// persistence is enabled; in-memory-only for ephemeral databases. See
304    /// [`crate::db::namespace`].
305    pub(crate) namespaces: Arc<namespace::NamespaceRegistry>,
306    /// Opt-in tamper-evident provenance hash chain (Issue #3351).
307    ///
308    /// `None` unless `config.chain.enabled`. When present, each committed
309    /// transaction's version refs are enqueued to a background sealer that
310    /// folds them into a domain-separated SHA-256 hash chain over recorded
311    /// history. Declared before `_tempdir` so it is dropped (flushing the
312    /// sealer) before the tempdir is removed.
313    pub(crate) chain: Option<Arc<crate::provenance_chain::ProvenanceChain>>,
314    /// Backing tempdir for ephemeral databases created via [`AletheiaDB::new`].
315    /// Declared last so it is dropped last (Rust drops struct fields in
316    /// declaration order); this guarantees the WAL/persistence file handles
317    /// above have already been closed before the directory is removed.
318    pub(crate) _tempdir: Option<tempfile::TempDir>,
319}

321impl std::fmt::Debug for AletheiaDB {
322    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
323        let current_ts = self
324            .current_timestamp
325            .try_lock()
326            .map(|ts| format!("{:?}", ts))
327            .unwrap_or_else(|_| "<locked>".to_string());
328
329        f.debug_struct("AletheiaDB")
330            .field("current_timestamp", &current_ts)
331            .field("default_durability", &self.default_durability)
332            .field("persistence_enabled", &self.persistence_manager.is_some())
333            .field("encryption_enabled", &self.encryption_manager.is_some())
334            .field("stats", &self.stats)
335            .finish_non_exhaustive()
336    }
337}

339impl Drop for AletheiaDB {
340    fn drop(&mut self) {
341        // Flush and stop the provenance-chain sealer first so its final records
342        // and head checkpoint are durable before storage handles are torn down
343        // (Issue #3351). Idempotent with the chain's own Drop.
344        if let Some(ref chain) = self.chain {
345            chain.shutdown();
346        }
347
348        // Signal shutdown to background persistence thread
349        if let Some(ref tracker) = self.persistence_tracker {
350            tracker.signal_shutdown();
351
352            // Wait for the background thread to fully exit and release all resources
353            // This ensures the thread drops its Arc references to TieredStorage/RedbColdStorage
354            // before Drop returns, preventing file locking issues when reopening Redb.
355            if let Some(handle) = self.persistence_thread_handle.take() {
356                let _ = handle.join();
357            }
358        }
359    }
