1//! Builder pattern for creating vector indexes.
2//!
3//! Provides `VectorIndexBuilder` to configure HNSW and temporal vector indexes.
4use crate::core::error::{Error, Result};
5use crate::db::AletheiaDB;
6use crate::index::vector::hnsw::HnswConfig;
7use crate::index::vector::temporal::TemporalVectorConfig;
8
9/// Builder for configuring and enabling a vector index on a property.
10///
11/// This builder provides a fluent API for setting up vector indexes with
12/// HNSW configuration and optional temporal indexing. The builder pattern
13/// ensures that required configuration (HNSW) is provided before enabling.
14///
15/// # Example
16///
17/// ```rust,no_run
18/// use aletheiadb::AletheiaDB;
19/// use aletheiadb::index::vector::{HnswConfig, DistanceMetric};
20/// use aletheiadb::index::vector::temporal::TemporalVectorConfig;
21///
22/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
23/// let db = AletheiaDB::new()?;
24///
25/// // Create a basic vector index
26/// db.vector_index("embedding")
27///     .hnsw(HnswConfig::new(384, DistanceMetric::Cosine))
28///     .enable()?;
29///
30/// // Create a vector index with temporal support
31/// // Note: Using a different property since "embedding" is already used
32/// db.vector_index("content_embedding")
33///     .hnsw(HnswConfig::new(384, DistanceMetric::Cosine))
34///     .temporal(TemporalVectorConfig::default())
35///     .enable()?;
36/// # Ok(())
37/// # }
38/// ```
39#[must_use = "VectorIndexBuilder does nothing unless .enable() is called"]
40pub struct VectorIndexBuilder<'a> {
41    db: &'a AletheiaDB,
42    property_name: String,
43    hnsw_config: Option<HnswConfig>,
44    temporal_config: Option<TemporalVectorConfig>,
45}
46
47impl<'a> VectorIndexBuilder<'a> {
48    /// Create a new builder for the specified property.
49    pub fn new(db: &'a AletheiaDB, property_name: String) -> Self {
50        VectorIndexBuilder {
51            db,
52            property_name,
53            hnsw_config: None,
54            temporal_config: None,
55        }
56    }
57
58    /// Set the HNSW index configuration.
59    ///
60    /// This is required before calling `enable()`. The configuration specifies
61    /// the vector dimensions, distance metric, and HNSW parameters.
62    ///
63    /// # Arguments
64    ///
65    /// * `config` - The HNSW index configuration
66    ///
67    /// # Example
68    ///
69    /// ```rust,no_run
70    /// # use aletheiadb::AletheiaDB;
71    /// # use aletheiadb::index::vector::{HnswConfig, DistanceMetric};
72    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
73    /// # let db = AletheiaDB::new()?;
74    /// db.vector_index("embedding")
75    ///     .hnsw(HnswConfig::new(384, DistanceMetric::Cosine).with_capacity(10000))
76    ///     .enable()?;
77    /// # Ok(())
78    /// # }
79    /// ```
80    pub fn hnsw(mut self, config: HnswConfig) -> Self {
81        self.hnsw_config = Some(config);
82        self
83    }
84
85    /// Enable temporal vector indexing.
86    ///
87    /// When temporal indexing is enabled, vector snapshots are created
88    /// periodically to enable point-in-time vector queries and semantic
89    /// drift tracking. This automatically enables the current index as well.
90    ///
91    /// # Arguments
92    ///
93    /// * `config` - The temporal vector index configuration
94    ///
95    /// # Example
96    ///
97    /// ```rust,no_run
98    /// # use aletheiadb::AletheiaDB;
99    /// # use aletheiadb::index::vector::{HnswConfig, DistanceMetric};
100    /// use aletheiadb::index::vector::temporal::{TemporalVectorConfig, SnapshotStrategy};
101    ///
102    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
103    /// # let db = AletheiaDB::new()?;
104    /// db.vector_index("embedding")
105    ///     .hnsw(HnswConfig::new(384, DistanceMetric::Cosine))
106    ///     .temporal(TemporalVectorConfig {
107    ///         snapshot_strategy: SnapshotStrategy::TransactionInterval(100),
108    ///         ..Default::default()
109    ///     })
110    ///     .enable()?;
111    /// # Ok(())
112    /// # }
113    /// ```
114    pub fn temporal(mut self, config: TemporalVectorConfig) -> Self {
115        self.temporal_config = Some(config);
116        self
117    }
118
119    /// Enable the vector index with the configured settings.
120    ///
121    /// This method validates the configuration and enables the index.
122    /// If temporal configuration was provided, both the current and
123    /// temporal indexes are enabled.
124    ///
125    /// # Errors
126    ///
127    /// Returns an error if:
128    /// - HNSW configuration was not provided (call `.hnsw()` first)
129    /// - A vector index already exists for this property
130    /// - The temporal index configuration is invalid
131    ///
132    /// # Example
133    ///
134    /// ```rust,no_run
135    /// # use aletheiadb::AletheiaDB;
136    /// # use aletheiadb::index::vector::{HnswConfig, DistanceMetric};
137    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
138    /// # let db = AletheiaDB::new()?;
139    /// // This will fail - no HNSW config
140    /// let result = db.vector_index("embedding").enable();
141    /// assert!(result.is_err());
142    ///
143    /// // This will succeed
144    /// db.vector_index("embedding")
145    ///     .hnsw(HnswConfig::new(384, DistanceMetric::Cosine))
146    ///     .enable()?;
147    /// # Ok(())
148    /// # }
149    /// ```
150    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
151    pub fn enable(self) -> Result<()> {
152        let hnsw_config = self
153            .hnsw_config
154            .ok_or_else(|| {
155                Error::Vector(crate::core::error::VectorError::IndexError(
156                    "HNSW configuration is required. Call .hnsw() before .enable()".to_string(),
157                ))
158            })
159            .inspect_err(|err| {
160                err.record_metric();
161            })?;
162
163        // If temporal config is provided, use enable_temporal_vector_index which
164        // automatically enables the current index as well (fixing #386)
165        if let Some(temporal_config) = self.temporal_config {
166            // Merge HNSW config into temporal config
167            let temporal_config_with_hnsw = TemporalVectorConfig {
168                hnsw_config: Some(hnsw_config),
169                ..temporal_config
170            };
171            self.db
172                .enable_temporal_vector_index(&self.property_name, temporal_config_with_hnsw)
173        } else {
174            // Just enable the current index
175            self.db
176                .enable_vector_index(&self.property_name, hnsw_config)
177        }
178    }
179}
