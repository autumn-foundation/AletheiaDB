1//! Durable encryption-state authority — `{data_dir}/encryption.state` (Issue #3616).
2//!
3//! A tiny, versioned, line-based file that records the **authoritative**
4//! encryption-at-rest posture of a durable database: whether encryption is on,
5//! and — as a non-secret *reference*, never key bytes — where the Master
6//! Encryption Key is sourced from. It is the durable authority the
7//! plaintext↔encrypted migration engine (Issue #3616) flips so the *next*
8//! [`open()`](crate::AletheiaDB::open) uses the correct mode, independent of the
9//! operator's TOML/env config.
10//!
11//! # Why a separate file (not the TOML)?
12//!
13//! The operator's `ALETHEIADB_CONFIG` TOML (or `ALETHEIADB_DATA_DIR`, which is
14//! hard-wired plaintext) is *input*, not durable database state. After an
15//! `encryption enable` migration rewrites every at-rest byte through the cipher,
16//! something inside the data dir must tell the next `open()` "this database is
17//! now encrypted, under this key source" — otherwise the operator's unchanged
18//! plaintext TOML would try to open encrypted bytes as plaintext and fail. This
19//! file is that durable record.
20//!
21//! # Precedence (the authority WINS)
22//!
23//! [`read_encryption_state`] is consulted by `open()` (see
24//! [`with_unified_config`](crate::AletheiaDB::with_unified_config)) as the
25//! **authority** for the encryption on/off decision — it overrides the
26//! TOML/env-derived [`EncryptionConfig`]. On divergence (the authority says one
27//! thing, the config another) a LOUD warning naming BOTH sources is logged so an
28//! operator is never silently overridden.
29//!
30//! # Fail-closed
31//!
32//! This file *guards encryption*, so it never guesses:
33//! * **absent** → `Ok(None)` — not enabled via the authority; fall through to
34//!   TOML/env (today's behavior, fully backward compatible).
35//! * **present & well-formed** → `Ok(Some(_))`.
36//! * **present but corrupt** → `Err(_)` — startup aborts loudly for manual
37//!   intervention; a garbled authority is NEVER treated as "not enabled".
38//!
39//! When the authority says `enabled` but the referenced key source is
40//! missing/unreadable, `open()` refuses (the encryption manager fails to source
41//! the MEK) rather than falling back to plaintext.
42//!
43//! # Security
44//!
45//! Only a key-source **reference** is ever written — a File path or an Env var
46//! NAME — exactly like the rotation ledger. Key bytes, passphrases, and tokens
47//! are never persisted here; passphrase/KMS/Vault sources (which carry secrets
48//! or multi-field config the line format cannot round-trip) are refused
49//! fail-closed by [`write_encryption_state_durable`], mirroring the rotation
50//! ledger's File/Env-only constraint.
51

58/// Current on-disk format version of the authority file.
59///
60/// `version=2` (Issue #3616 PR3) additively records the pinned concrete AEAD
61/// `algorithm` an enable migration wrote under. The reader still accepts a legacy
62/// `version=1` file (no `algorithm` line → `algorithm: None`, reopen falls back to
63/// the config algorithm, exactly the prior behavior). An unknown version fails
64/// closed (never guessed).
65const ENCRYPTION_STATE_VERSION: u32 = 2;
66
67/// Serialize a concrete AEAD [`Algorithm`] to its stable on-disk token. `Auto` is
68/// resolved to its concrete form first so the pinned value is never ambiguous
69/// (the whole point of pinning is cross-host portability).
70fn algorithm_token(algorithm: Algorithm) -> &'static str {
71    match algorithm.resolve() {
72        Algorithm::Aes256Gcm => "aes256gcm",
73        Algorithm::ChaCha20Poly1305 => "chacha20poly1305",
74        // `resolve()` never returns `Auto`; be explicit rather than `unreachable!`.
75        Algorithm::Auto => "aes256gcm",
76    }
77}
78
79/// Parse a pinned-algorithm token written by [`algorithm_token`]. Returns `None`
80/// for an unrecognized token (treated as corrupt by the caller).
81fn algorithm_from_token(token: &str) -> Option<Algorithm> {
82    match token {
83        "aes256gcm" => Some(Algorithm::Aes256Gcm),
84        "chacha20poly1305" => Some(Algorithm::ChaCha20Poly1305),
85        _ => None,
86    }
87}

89/// The durable encryption-state authority parsed from `{data_dir}/encryption.state`.
90///
91/// `enabled` is the authoritative on/off decision. `key_source` is present iff
92/// `enabled` and is restricted to File/Env references (never key bytes).
93#[derive(Debug, Clone, PartialEq, Eq)]
94pub(crate) struct EncryptionState {
95    /// Whether encryption at rest is authoritatively ON for this database.
96    pub enabled: bool,
97    /// The non-secret key-source reference (File path / Env var name). `Some`
98    /// iff `enabled`; `None` when the authority records "disabled".
99    pub key_source: Option<KeyProviderConfig>,
100    /// The pinned concrete AEAD algorithm the enable migration wrote under
101    /// (Issue #3616 PR3). `Some` for a `version=2` enabled authority; `None` for a
102    /// legacy `version=1` authority or a disabled state. On reopen it OVERRIDES the
103    /// operator's `config.encryption.algorithm`, so an enable performed under a
104    /// concrete (or host-resolved `Auto`) algorithm always reopens under the exact
105    /// same cipher, regardless of TOML edits or a cross-CPU host migration.
106    pub algorithm: Option<Algorithm>,
107}
108
109impl EncryptionState {
110    /// Construct the "encryption enabled under `key_source`" authority WITHOUT a
111    /// pinned algorithm (legacy shape). Prefer [`Self::enabled_with_algorithm`] on
112    /// the production enable path so the concrete cipher is pinned durably.
113    ///
114    /// `key_source` must be a File or Env reference; other backends are refused
115    /// at write time (they carry secrets the line format cannot persist).
116    #[cfg(test)]
117    pub(crate) fn enabled(key_source: KeyProviderConfig) -> Self {
118        Self {
119            enabled: true,
120            key_source: Some(key_source),
121            algorithm: None,
122        }
123    }
124
125    /// Construct the "encryption enabled under `key_source`, pinned to
126    /// `algorithm`" authority (Issue #3616 PR3). The algorithm is resolved to its
127    /// concrete form so the pinned value is host-independent.
128    ///
129    /// The producer is the `encryption enable` migration engine
130    /// (`crate::db::encryption_enable`), which flips this authority durably as the
131    /// binding step BEFORE clearing the migration ledger.
132    pub(crate) fn enabled_with_algorithm(
133        key_source: KeyProviderConfig,
134        algorithm: Algorithm,
135    ) -> Self {
136        Self {
137            enabled: true,
138            key_source: Some(key_source),
139            algorithm: Some(algorithm.resolve()),
140        }
141    }
142}

149/// Reduce a corrupt-file detail to a fail-closed error (never fabricates state).
150fn corrupt(detail: &str) -> crate::core::error::Error {
151    StorageError::InconsistentState {
152        reason: format!(
153            "encryption.state present but corrupt ({detail}); manual intervention required"
154        ),
155    }
156    .into()
157}

159/// Read + parse the durable encryption-state authority for `data_dir`.
160///
161/// Fail-closed: absent → `Ok(None)`, well-formed → `Ok(Some(_))`, present but
162/// corrupt / unreadable → `Err(_)`. A malformed authority is NEVER downgraded to
163/// "not enabled" — that would silently open an encrypted database as plaintext.
164///
165/// Accepted key-source kinds are `file` / `env` only; any other kind (or an
166/// `enabled=true` record with no key source) is corrupt.
167pub(crate) fn read_encryption_state(data_dir: &Path) -> Result<Option<EncryptionState>> {
168    read_encryption_state_at(&encryption_state_path(data_dir))
169}
170
171/// Path-based core of [`read_encryption_state`], so tests and future startup
172/// wiring can target an exact file. Same fail-closed contract.
173pub(crate) fn read_encryption_state_at(path: &Path) -> Result<Option<EncryptionState>> {
174    let body = match std::fs::read_to_string(path) {
175        Ok(b) => b,
176        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(None),
177        Err(e) => {
178            return Err(StorageError::io_error(format!(
179                "encryption.state present but unreadable ({e}); manual intervention required"
180            ))
181            .into());
182        }
183    };
184
185    let mut version = None;
186    let mut enabled = None;
187    let mut kind = None;
188    let mut value = None;
189    let mut algorithm_line = None;
190    for line in body.lines() {
191        if line.trim().is_empty() {
192            continue;
193        }
194        let Some((k, v)) = line.split_once('=') else {
195            return Err(corrupt("malformed line"));
196        };
197        match k {
198            "version" => version = Some(v.parse::<u32>().map_err(|_| corrupt("bad version"))?),
199            "enabled" => {
200                enabled = Some(match v {
201                    "true" => true,
202                    "false" => false,
203                    other => return Err(corrupt(&format!("bad enabled {other:?}"))),
204                })
205            }
206            "key_source_kind" => kind = Some(v.to_string()),
207            "key_source_value" => value = Some(v.to_string()),
208            // Pinned concrete AEAD algorithm (Issue #3616 PR3, `version=2`).
209            "algorithm" => algorithm_line = Some(v.to_string()),
210            // Unknown keys are ignored for forward-compatibility, mirroring the
211            // rotation ledger reader.
212            _ => {}
213        }
214    }
215
216    // `version=` is MANDATORY — a well-formed writer always emits it, so its
217    // absence means a truncated/garbled file, not a legacy artifact.
218    let version = version.ok_or_else(|| corrupt("missing version"))?;
219    // Accept every version this build understands. A `version=1` file predates the
220    // pinned `algorithm` line (Issue #3616 PR3) and loads with `algorithm: None` —
221    // reopen then falls back to the config algorithm, exactly the prior behavior.
222    if version != 1 && version != ENCRYPTION_STATE_VERSION {
223        return Err(corrupt(&format!("unsupported version {version}")));
224    }
225    let enabled = enabled.ok_or_else(|| corrupt("missing enabled"))?;
226
227    if !enabled {
228        return Ok(Some(EncryptionState {
229            enabled: false,
230            key_source: None,
231            algorithm: None,
232        }));
233    }
234
235    // enabled → a File/Env key-source reference is required (fail-closed: an
236    // enabled authority with no resolvable key source is corrupt, never a
237    // silent plaintext fallback).
238    let value = value.ok_or_else(|| corrupt("missing key_source_value"))?;
239    let key_source = match kind.as_deref() {
240        Some("file") => KeyProviderConfig::File { path: value.into() },
241        Some("env") => KeyProviderConfig::Env { variable: value },
242        Some(other) => return Err(corrupt(&format!("unsupported key_source_kind {other:?}"))),
243        None => return Err(corrupt("missing key_source_kind")),
244    };
245    // A present `algorithm` line must parse to a known concrete algorithm; an
246    // unrecognized token is corrupt (fail-closed, never a silent wrong cipher).
247    // Absent (legacy `version=1`) → `None`.
248    let algorithm = match algorithm_line {
249        Some(token) => Some(
250            algorithm_from_token(&token)
251                .ok_or_else(|| corrupt(&format!("unsupported algorithm {token:?}")))?,
252        ),
253        None => None,
254    };
255    Ok(Some(EncryptionState {
256        enabled: true,
257        key_source: Some(key_source),
258        algorithm,
259    }))
260}

262/// Durably write the encryption-state authority for `data_dir`.
263///
264/// Uses the SAME ordered, crash-safe write as the rotation ledger — temp →
265/// `sync_all` → atomic rename → parent-directory fsync
266/// ([`write_durable`](crate::db::rotation::write_durable)) — so a power loss can
267/// never leave a torn authority file. No key material is written: only a
268/// non-secret File/Env reference.
269///
270/// # Errors
271///
272/// * The key source is a passphrase/KMS/Vault backend (refused fail-closed —
273///   those carry secrets or multi-field config the line format cannot persist,
274///   mirroring the rotation ledger).
275/// * An enabled state carries no key source, or a disabled state carries one
276///   (a caller bug — the authority must be internally consistent).
277/// * The directory cannot be created or the file cannot be written durably.
278///
279/// The production caller is the `encryption enable` migration engine (Issue
280/// #3616 PR3, `crate::db::encryption_enable`): `enable_encryption` flips the
281/// authority to `enabled` as the binding step before clearing the ledger, and
282/// `resume_pending_enable` performs the same flip when finishing an interrupted
283/// enable.
284pub(crate) fn write_encryption_state_durable(
285    data_dir: &Path,
286    state: &EncryptionState,
287) -> Result<()> {
288    let body = match (state.enabled, &state.key_source) {
289        (true, Some(source)) => {
290            let (kind, value) = match source {
291                KeyProviderConfig::File { path } => ("file", path.to_string_lossy().into_owned()),
292                KeyProviderConfig::Env { variable } => ("env", variable.clone()),
293                KeyProviderConfig::PassphraseFile { .. }
294                | KeyProviderConfig::Kms { .. }
295                | KeyProviderConfig::Vault { .. } => {
296                    let (provider_type, _) = source.describe();
297                    return Err(StorageError::PersistenceError(format!(
298                        "recording a {provider_type} key source in the encryption-state authority \
299                         is not supported (only file/env references can be persisted without \
300                         leaking a secret)"
301                    ))
302                    .into());
303                }
304            };
305            // The pinned concrete algorithm line is additive (`version=2`); a state
306            // constructed without one (legacy `enabled`) omits it and reads back as
307            // `algorithm: None`.
308            let algorithm_line = match state.algorithm {
309                Some(algorithm) => format!("algorithm={}\n", algorithm_token(algorithm)),
310                None => String::new(),
311            };
312            format!(
313                "version={ENCRYPTION_STATE_VERSION}\nenabled=true\nkey_source_kind={kind}\nkey_source_value={value}\n{algorithm_line}"
314            )
315        }
316        (true, None) => {
317            return Err(StorageError::PersistenceError(
318                "cannot record an enabled encryption-state authority with no key source"
319                    .to_string(),
320            )
321            .into());
322        }
323        (false, Some(_)) => {
324            return Err(StorageError::PersistenceError(
325                "cannot record a disabled encryption-state authority with a key source".to_string(),
326            )
327            .into());
328        }
329        (false, None) => format!("version={ENCRYPTION_STATE_VERSION}\nenabled=false\n"),
330    };
331
332    std::fs::create_dir_all(data_dir)
333        .map_err(|e| StorageError::io_error(format!("Failed to create data dir: {e}")))?;
334    crate::db::rotation::write_durable(&encryption_state_path(data_dir), body.as_bytes())
335}

337/// Emit a LOUD warning under either logging configuration, matching the crate's
338/// `observability`-gated logging style (mirrors `log_registry_warning` in
339/// `src/db/snapshot.rs`).
340pub(crate) fn log_authority_warning(message: &str) {
341    #[cfg(feature = "observability")]
342    tracing::warn!("{}", message);
343    #[cfg(not(feature = "observability"))]
344    eprintln!("WARNING: {}", message);
345}
