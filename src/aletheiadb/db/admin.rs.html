26    /// Persist all indexes to disk.
27    ///
28    /// This saves the current state of all indexes (graph, temporal, vector, strings)
29    /// to disk in the configured persistence directory.
30    ///
31    /// # Errors
32    ///
33    /// Returns an error if:
34    /// - Index persistence is not enabled in configuration
35    /// - Writing index files fails due to I/O errors
36    /// - Index serialization fails
37    ///
38    /// # Example
39    ///
40    /// ```rust,no_run
41    /// # use aletheiadb::AletheiaDB;
42    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
43    /// let db = AletheiaDB::new()?;
44    /// // ... add data ...
45    /// db.persist_indexes()?; // Save indexes to disk
46    /// # Ok(())
47    /// # }
48    /// ```
49    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
50    pub fn persist_indexes(&self) -> Result<()> {
51        let result = (|| {
52            use crate::storage::index_persistence::formats::IndexManifest;
53
54            let manager = self.persistence_manager.as_ref().ok_or_else(|| {
55                StorageError::InconsistentState {
56                    reason: "Index persistence not enabled".to_string(),
57                }
58            })?;
59
60            // Fail-closed guard for the QUIESCED post-`enable_encryption` handle
61            // (Issue #3616 PR3). After `enable_encryption` the live WAL is encrypted
62            // but this handle's index manager still carries a PLAINTEXT keyring
63            // (there is no live `None → Some` index-keyring install — see the loud
64            // reopen contract on `enable_encryption`). Persisting now would write
65            // PLAINTEXT index files OVER the freshly-wrapped `AEIX` snapshot — the
66            // exact corruption the enable engine exists to prevent. Refuse loudly and
67            // direct the caller to reopen; the resume path (which DOES build the
68            // manager under the enable index DEK, so its keyring is `Some`) is
69            // unaffected, as is a normally-encrypted reopen.
70            if self.wal.is_encrypted() && manager.keyring().is_none() {
71                return Err(crate::core::error::Error::FailedPrecondition(
72                    "cannot persist indexes on a post-enable quiesced handle: the WAL is \
73                     encrypted but this handle's index manager is still plaintext, so a \
74                     persist would write plaintext index files over the encrypted (AEIX) \
75                     snapshot. You MUST reopen the database (drop this handle and call \
76                     AletheiaDB::open) to get a fully-encrypted, persistable instance."
77                        .to_string(),
78                ));
79            }
80
81            // Fail-closed guard for the QUIESCED post-`disable_encryption` handle
82            // (Issue #3616 PR4) — the mirror of the enable guard above. After
83            // `disable_encryption` the live WAL is plaintext (keyring uninstalled)
84            // but this handle's index manager still carries the ENCRYPTED keyring
85            // (there is no live `Some → None` index-keyring uninstall — see the loud
86            // reopen contract on `disable_encryption`). Persisting now would write
87            // `AEIX` index files OVER the freshly-unwrapped plaintext snapshot — the
88            // exact corruption the disable engine exists to prevent.
89            //
90            // Unlike the enable guard, the raw state (WAL plaintext + index keyring
91            // `Some`) is NOT unique to a post-disable handle: an index-only-encrypted
92            // database (a plaintext WAL over an encrypted index — the sole config in
93            // which an index-only key rotation is safe) has the identical shape and
94            // MUST keep persisting normally. So this rare ambiguous branch is
95            // disambiguated by TWO durable signals, EITHER of which marks a
96            // post-disable quiesced handle:
97            //
98            //   (a) the authority records `disabled` — the terminal state a
99            //       SUCCESSFUL `disable_encryption` flips it to; and
100            //   (b) a pending `direction=disable` rotation ledger is present — the
101            //       breadcrumb a disable writes BEFORE it flips the authority.
102            //
103            // Signal (b) closes a real gap: a disable that FAILS at the cold-unwrap
104            // step (index already unwrapped to plaintext, WAL already plaintext, but
105            // the authority not yet flipped) returns Err leaving WAL plaintext +
106            // keyring `Some` + authority STILL `enabled` + a pending disable ledger.
107            // On that errored handle, gating on the authority alone would read
108            // `enabled` and WRONGLY allow a persist that rewrites `AEIX` over the
109            // already-plaintext index snapshot. Also firing on the disable ledger
110            // fails closed in that window. This does NOT reintroduce the
111            // index-only-encrypted false positive: that config runs an index key
112            // ROTATION whose ledger is `direction=rotate/enable`, and
113            // `read_disable_ledger` filters to `direction=disable` only (returning
114            // `None` for a rotation/enable ledger or no ledger), so the legitimate
115            // index-only case still persists normally. The extra reads happen ONLY in
116            // this branch (never entered by a normal plaintext DB — keyring `None`, or
117            // a normal encrypted DB — WAL encrypted), so they add no hot-path cost.
118            if !self.wal.is_encrypted()
119                && manager.keyring().is_some()
120                && (crate::db::encryption_state::read_encryption_state(manager.base_path())?
121                    .is_some_and(|state| !state.enabled)
122                    || crate::db::rotation::read_disable_ledger(manager)?.is_some())
123            {
124                return Err(crate::core::error::Error::FailedPrecondition(
125                    "cannot persist indexes on a post-disable quiesced handle: the WAL is \
126                     plaintext but this handle's index manager is still encrypted, so a \
127                     persist would write encrypted (AEIX) index files over the plaintext \
128                     snapshot. You MUST reopen the database (drop this handle and call \
129                     AletheiaDB::open) to get a fully-plaintext, persistable instance."
130                        .to_string(),
131                ));
132            }
133
134            // Warn if background persistence thread has stopped
135            if self
136                .persistence_thread_stopped
137                .load(std::sync::atomic::Ordering::Acquire)
138            {
139                eprintln!(
140                    "Warning: Background persistence thread has stopped. \
141                     Automatic persistence is disabled. Manual persist_indexes() calls will still work."
142                );
143            }
144
145            let tracker = self.persistence_tracker.as_ref();
146
147            // ── Coherence barrier (lost-write persist race fix) ─────────────────
148            //
149            // The manifest LSN is the APPLIED WATERMARK: the minimum LSN of any
150            // commit that is durable (fsynced) but not yet applied. Stamping the
151            // manifest with this — rather than the WAL allocation frontier —
152            // guarantees replay re-covers any durable-but-unapplied write instead
153            // of dropping it. When nothing is in flight it equals today's frontier
154            // (idle persist keeps the identical manifest LSN). Re-applying a write
155            // that DID make it into the snapshot is safe: replay is idempotent
156            // (keyed by version_id).
157            //
158            // We read the watermark under a BRIEF `current_timestamp` hold
159            // (lock-order class 1, acquired FIRST), then RELEASE it before the two
160            // O(N) snapshot clones. This closes a narrow window: a commit's WAL LSN
161            // band is allocated (bumping the frontier) INSIDE `append_batch`, but
162            // its in-flight registration happens just AFTER the append returns —
163            // both under the commit's own `current_timestamp` hold. Without taking
164            // `current_timestamp` here, we could read a frontier already advanced
165            // past such a commit while `in_flight.min()` does not yet see it, so the
166            // manifest would sit ABOVE that soon-to-be-durable write and replay
167            // would drop it. Holding `current_timestamp` only for this O(log n) read
168            // guarantees the frontier and the in-flight set are mutually consistent,
169            // WITHOUT stalling every writer (class 1 is the top of every commit)
170            // across the deep clones below.
171            //
172            // CORRECTNESS INVARIANT (why releasing before the clone is safe): with
173            // T0 = the instant of this watermark read and T1 > T0 the instant of the
174            // snapshot clone, `manifest_lsn = min(in-flight bases at T0)` and
175            // `manifest_lsn <= frontier(T0) <= any later LSN allocation`. Every write
176            // with `lsn < manifest_lsn` was NOT in-flight at T0 (min is the smallest
177            // in-flight), hence already applied at T0, hence present in the coherent
178            // snapshot taken at T1 (storage is monotonic; the snapshot is coherent
179            // under `historical.read()`). Every durable write NOT in the snapshot has
180            // `lsn >= manifest_lsn` and is re-applied by inclusive replay (idempotent
181            // by version_id). So releasing `current_timestamp` before the clone does
182            // NOT reintroduce a lost write, and snapshot entries with
183            // `lsn >= manifest_lsn` are harmless (replay is idempotent). We do NOT
184            // re-read the frontier or `in_flight.min()` after releasing the lock.
185            let manifest_lsn = {
186                let _ts =
187                    self.current_timestamp
188                        .lock()
189                        .map_err(|_| StorageError::InconsistentState {
190                            reason: "current_timestamp lock poisoned during persist_indexes"
191                                .to_string(),
192                        })?;
193                let frontier = self.wal.current_lsn().0;
194                self.in_flight.min().unwrap_or(frontier)
195            }; // current_timestamp released here — clones run WITHOUT the class-1 lock
196
197            // Now take the coherent in-memory snapshot WITHOUT holding
198            // `current_timestamp`, so the graph and temporal indexes cannot observe
199            // different instants (the pre-fix torn snapshot could restore e.g. 85
200            // graph nodes against 86 temporal versions). We mirror the PROVEN
201            // backup.rs lock order EXACTLY — `historical.read()` THEN
202            // `current.snapshot_lock.write()` — which is consistent with the commit
203            // path's `historical.write()` → `snapshot_lock.read()` ordering, so no
204            // AB-BA inversion. Locks are released BEFORE any disk I/O; serialization
205            // below runs OFF-LOCK from the immutable snapshots, like checkpoint /
206            // backup.
207            let hist = self.historical.read();
208            let current_snapshot = {
209                let _snap_lock = self.current.snapshot_lock.write();
210                self.current
211                    .create_snapshot(crate::storage::wal::LSN(manifest_lsn))
212            };
213            let historical_snapshot = hist.create_snapshot(crate::storage::wal::LSN(manifest_lsn));
214            drop(hist);
215
216            // String interner must be saved first (dependency for all others).
217            // Update the string LSN tracker to manifest_lsn BEFORE calculating
218            // safe LSN so even if no new strings were added, the tracker reflects
219            // the watermark.
220            if let Some(tracker) = tracker {
221                crate::storage::index_persistence::operations::persist_string_interner(
222                    manager,
223                    tracker,
224                    manifest_lsn,
225                )?;
226            } else {
227                manager.save_string_interner().map_err(|e| {
228                    StorageError::persistence_with_kind(
229                        PersistenceErrorKind::from(&e),
230                        format!("Failed to save string interner: {}", e),
231                    )
232                })?;
233            }
234
235            // Graph index from the COHERENT current snapshot (off-lock).
236            persist_graph_index_from_snapshot(&current_snapshot, manager, tracker, manifest_lsn)?;
237
238            if let Some(tracker) = tracker {
239                // NOTE (v1 scope, F7 follow-up): vector indexes are persisted from
240                // LIVE current storage here — AFTER the coherence barrier above was
241                // released — rather than from the coherent snapshot. Consequently a
242                // graph-vs-vector torn snapshot is possible on recovery. This is NOT
243                // a lost write: replay re-indexes every entry with lsn >= manifest_lsn,
244                // so any node missing from the vector file is re-covered. But an entry
245                // that IS in the vector file AND also replayed can get a double HNSW
246                // insert. The reported torn-snapshot symptom is graph-vs-temporal
247                // (both fixed above); a snapshot-coherent vector persist (snapshot the
248                // vectors under the same barrier, or gate loaded vector entries by
249                // lsn <= manifest on restore) is tracked follow-up F7. See design doc.
250                persist_vector_indexes(&self.current, manager, Some(tracker), manifest_lsn)?;
251                // Temporal index from the COHERENT historical snapshot (off-lock).
252                persist_temporal_index_from_snapshot(
253                    &historical_snapshot,
254                    manager,
255                    tracker,
256                    manifest_lsn,
257                )?;
258            }
259
260            // Persist the temporal-adjacency index too, so a manual force-persist
261            // (e.g. re-encrypting a dataset after enabling encryption) rewrites
262            // EVERY on-disk index file — matching the shutdown path
263            // `persist_all_indexes`. Without this, `temporal_adjacency/adjacency.idx`
264            // could remain plaintext after a manual re-encrypt (Issue #481).
265            //
266            // Persisted from LIVE historical storage AFTER the coherence barrier was
267            // released (same treatment as `persist_vector_indexes` above): the
268            // temporal-adjacency index shares the vector index's documented
269            // torn-snapshot follow-up (F7). This is NOT a lost write — replay
270            // reconstructs adjacency for every entry with lsn >= manifest_lsn, so the
271            // on-disk file is re-covered on restore.
272            persist_temporal_adjacency_index(&self.historical, manager)?;
273
274            // Record WAL position for future replay coordination. Safe LSN = min
275            // of all components; since every component was just persisted at
276            // `manifest_lsn`, this resolves to the applied watermark.
277            let safe_lsn = tracker
278                .map(|t| t.get_safe_manifest_lsn())
279                .unwrap_or(manifest_lsn);
280
281            let manifest = IndexManifest::new(safe_lsn);
282            manager.save_manifest(&manifest).map_err(|e| {
283                StorageError::persistence_with_kind(
284                    PersistenceErrorKind::from(&e),
285                    format!("Failed to save manifest: {}", e),
286                )
287            })?;
288
289            Ok(())
290        })();
291        result.record_error_metric()
292    }

416    /// Refresh query optimization statistics from current storage.
417    ///
418    /// This collects fresh statistics about node counts, edge counts, label
419    /// cardinalities, and other metrics used for cost-based query optimization.
420    /// Call this method after significant schema changes or data modifications
421    /// to ensure the query planner has accurate information.
422    ///
423    /// Statistics are automatically refreshed lazily on first query, so this
424    /// method is typically only needed for benchmarking or after bulk imports.
425    ///
426    /// # Example
427    ///
428    /// ```rust,no_run
429    /// # use aletheiadb::{AletheiaDB, PropertyMap};
430    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
431    /// # let db = AletheiaDB::new()?;
432    /// # let documents: Vec<PropertyMap> = vec![];
433    /// // After bulk import
434    /// for props in documents {
435    ///     db.create_node("Document", props)?;
436    /// }
437    ///
438    /// // Refresh statistics for optimal query planning
439    /// db.refresh_statistics();
440    ///
441    /// // Now queries will use accurate statistics
442    /// # let query = db.query().build();
443    /// let results = db.execute_query(query)?;
444    /// # Ok(())
445    /// # }
446    /// ```
447    pub fn refresh_statistics(&self) {
448        // Collect statistics from current storage
449        let node_count = self.current.node_count();
450        let edge_count = self.current.edge_count();
451        let vector_count = self.current.vector_count();
452
453        // Collect label counts from current storage
454        let label_counts = self.current.label_counts();
455
456        // Calculate the average delta chain length from historical storage
457        // (Issue #366). This feeds the query planner's temporal-lookup cost
458        // model. O(1): reads incrementally-maintained anchor/delta counters
459        // (Issue #212) and falls back to the default estimate of 5.0 when the
460        // historical storage is empty.
461        let avg_delta_chain = self.historical.read().calculate_avg_delta_chain();
462
463        self.stats.refresh(
464            node_count,
465            edge_count,
466            vector_count,
467            label_counts,
468            avg_delta_chain,
469        );
470    }
