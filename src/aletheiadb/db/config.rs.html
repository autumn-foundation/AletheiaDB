25/// Validate and clamp a configured `persistence.max_interned_strings` value
26/// before it drives the process-global interner cap (configurable interner cap).
27///
28/// - `0` is **rejected** with a clear error: a zero cap would refuse every
29///   string intern, bricking all writes and `open()` itself. It must never be
30///   pushed into `GLOBAL_INTERNER.set_max_capacity`.
31/// - Any value at or above `u32::MAX` is **clamped** to `u32::MAX`: interner ids
32///   are `AtomicU32`, so a higher cap is unreachable and would only mislead.
33fn validate_interner_cap(configured: usize) -> Result<usize> {
34    if configured == 0 {
35        return Err(crate::core::error::Error::FailedPrecondition(
36            "persistence.max_interned_strings must be at least 1; a cap of 0 would reject all \
37             string interning and make every write (and open) fail"
38                .to_string(),
39        ));
40    }
41    Ok(configured.min(u32::MAX as usize))
42}

44/// The outcome of reconciling a newly-opened database's requested interner cap
45/// against the process-global cap already in effect (Issue #3724).
46///
47/// The string interner is a single process-wide static, so there is exactly one
48/// effective cap shared by every `AletheiaDB` opened in the process. This enum is
49/// the pure decision of the "max-of-all-opens high-water mark" policy; see
50/// [`decide_interner_cap`].
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52enum InternerCapDecision {
53    /// The first open in the process establishes the baseline cap (honoring an
54    /// intentionally low DoS bound for the common single-DB deployment, even below
55    /// the default seed).
56    Established(usize),
57    /// A later open requested MORE headroom than the current effective cap; the
58    /// shared cap is raised to it. Legitimate and silent.
59    Raised(usize),
60    /// A later open requested a LOWER cap than the current effective cap. The lower
61    /// value is NOT applied — lowering the shared cap could refuse writes on an
62    /// already-open database that may already hold more strings than `requested`.
63    /// The higher `effective` cap is kept and a loud warning is emitted.
64    KeptHigher { requested: usize, effective: usize },
65    /// A later open requested exactly the current effective cap; nothing changes.
66    Unchanged(usize),
67}

69/// Decide how a newly-opened database's `requested` interner cap reconciles with
70/// the process-global cap already in effect, under the "max-of-all-opens
71/// high-water mark" policy (Issue #3724).
72///
73/// Pure and side-effect-free so it can be unit-tested exhaustively without
74/// touching the process-global interner:
75///
76/// - The **first** open (`established == false`) establishes the baseline = its
77///   requested cap, so a single DB can still set a deliberately low DoS bound.
78/// - Every **subsequent** open may only ever ADD headroom: the effective cap
79///   becomes `max(current, requested)` and is never lowered. A subsequently-opened
80///   lower cap therefore can no longer refuse writes on an already-open database.
81fn decide_interner_cap(established: bool, current: usize, requested: usize) -> InternerCapDecision {
82    if !established {
83        return InternerCapDecision::Established(requested);
84    }
85    match requested.cmp(&current) {
86        std::cmp::Ordering::Greater => InternerCapDecision::Raised(requested),
87        std::cmp::Ordering::Less => InternerCapDecision::KeptHigher {
88            requested,
89            effective: current,
90        },
91        std::cmp::Ordering::Equal => InternerCapDecision::Unchanged(current),
92    }
93}

95/// Whether any database open in this process has already established the shared
96/// interner cap. Guards the first-open-establishes vs subsequent-raise-only arms
97/// of [`decide_interner_cap`].
98static INTERNER_CAP_ESTABLISHED: std::sync::atomic::AtomicBool =
99    std::sync::atomic::AtomicBool::new(false);
100
101/// Serializes concurrent `open()`s racing to establish/raise the process-global
102/// interner cap, so the first-open-establishes / raise-only decision is atomic
103/// with respect to the read-then-write of `GLOBAL_INTERNER.max_capacity()`.
104static INTERNER_CAP_LOCK: Mutex<()> = Mutex::new(());
105
106/// Apply a validated `requested` interner cap to the process-global interner under
107/// the "max-of-all-opens high-water mark" policy (Issue #3724).
108///
109/// A later open may only ever raise the shared cap, never lower it, so a
110/// subsequently-opened lower cap cannot break writes on an already-open database.
111/// When a lower request is clamped up (not honored), a loud warning is emitted.
112///
113/// The high-water-mark invariant is enforced **only** on this open seam. The
114/// `pub(crate)` [`StringInterner::set_max_capacity`] primitive still stores
115/// exactly what it is given (a handful of in-crate/test callers rely on lowering
116/// it directly); production databases all funnel their cap through here.
117fn apply_interner_cap(requested: usize) {
118    // Poisoning only happens if a prior holder panicked; the guarded region is a
119    // couple of atomic ops with no panic path, so recover the guard either way.
120    let _guard = INTERNER_CAP_LOCK
121        .lock()
122        .unwrap_or_else(|poisoned| poisoned.into_inner());
123    let interner = &*crate::core::GLOBAL_INTERNER;
124    let established = INTERNER_CAP_ESTABLISHED.swap(true, std::sync::atomic::Ordering::AcqRel);
125    let decision = decide_interner_cap(established, interner.max_capacity(), requested);
126    match decision {
127        InternerCapDecision::Established(cap) | InternerCapDecision::Raised(cap) => {
128            interner.set_max_capacity(cap);
129        }
130        InternerCapDecision::Unchanged(_) => {}
131        InternerCapDecision::KeptHigher {
132            requested,
133            effective,
134        } => {
135            warn_interner_cap_kept_higher(requested, effective);
136        }
137    }
138}

140/// Emit the loud, one-line warning for the case where a later open requested a
141/// LOWER interner cap than the process-global cap already in effect and the lower
142/// value was therefore not applied (Issue #3724).
143///
144/// Uses `tracing` when the `observability` feature is compiled and falls back to
145/// `eprintln!` (the persistence layer's best-effort-stderr convention) otherwise,
146/// so the warning is loud in every build configuration, including
147/// `--no-default-features`.
148fn warn_interner_cap_kept_higher(requested: usize, effective: usize) {
149    #[cfg(feature = "observability")]
150    tracing::warn!(
151        requested,
152        effective,
153        "persistence.max_interned_strings is lower than the process-global \
154         string-interner cap already in effect; the interner is shared across all \
155         databases opened in this process, so the lower cap is NOT applied \
156         (keeping the higher effective cap) to avoid refusing writes on an \
157         already-open database. See Issue #3724."
158    );
159    #[cfg(not(feature = "observability"))]
160    eprintln!(
161        "WARNING: persistence.max_interned_strings={requested} is lower than the \
162         process-global string-interner cap already in effect ({effective}); the \
163         interner is shared across all databases opened in this process, so the \
164         lower cap is NOT applied (keeping {effective}) to avoid refusing writes on \
165         an already-open database. See Issue #3724."
166    );
167}

169fn bootstrap_timestamp(
170    current: &CurrentStorage,
171    historical: &RwLock<HistoricalStorage>,
172) -> crate::core::temporal::Timestamp {
173    let mut max_timestamp = time::now();
174
175    for node in current.iter_nodes() {
176        if let Some(commit_ts) = node.metadata.commit_timestamp
177            && commit_ts > max_timestamp
178        {
179            max_timestamp = commit_ts;
180        }
181    }
182
183    for edge in current.iter_edges() {
184        if let Some(commit_ts) = edge.metadata.commit_timestamp
185            && commit_ts > max_timestamp
186        {
187            max_timestamp = commit_ts;
188        }
189    }
190
191    let historical = historical.read();
192    for node_version in historical.get_node_versions().values() {
193        let tx_time = node_version.temporal.transaction_time();
194        if tx_time.start() > max_timestamp {
195            max_timestamp = tx_time.start();
196        }
197        // Issue #3387: restored versions carry CLOSED tx ends too. A closure
198        // stamped by a superseding version that was since cold-migrated may
199        // exceed every restored tx start; fold it in so the HLC seed stays
200        // monotonic under clock skew.
201        if !tx_time.is_current() && tx_time.end() > max_timestamp {
202            max_timestamp = tx_time.end();
203        }
204    }
205
206    for edge_version in historical.get_edge_versions().values() {
207        let tx_time = edge_version.temporal.transaction_time();
208        if tx_time.start() > max_timestamp {
209            max_timestamp = tx_time.start();
210        }
211        if !tx_time.is_current() && tx_time.end() > max_timestamp {
212            max_timestamp = tx_time.end();
213        }
214    }
215
216    max_timestamp
217}

219pub(crate) fn seed_startup_current_timestamp(db: &AletheiaDB) -> Result<()> {
220    let startup_timestamp = bootstrap_timestamp(&db.current, &db.historical);
221    let mut current_timestamp = db.current_timestamp.lock().map_err(|_| {
222        crate::core::error::Error::Storage(StorageError::LockPoisoned {
223            resource: "current_timestamp".to_string(),
224        })
225    })?;
226    *current_timestamp = startup_timestamp;
227    Ok(())
228}

230/// Seed the WAL LSN allocator from durable state at startup (Issue #3420).
231///
232/// Scans the WAL directory for the maximum LSN present in existing segments
233/// and moves the allocator to `max + 1` (never backwards). Must run after WAL
234/// construction and **before any write is accepted**; otherwise a restarted
235/// process starts allocating at LSN 1 again, producing duplicate LSNs across
236/// segments and writes that land below the index manifest LSN — which the
237/// next startup's differential replay then silently skips.
238///
239/// Seeding policy lives here (in the database startup path), not inside the
240/// WAL constructor, so WAL-crate users keep full control over allocator state.
241fn seed_lsn_allocator_from_segments(
242    wal: &ConcurrentWalSystem,
243    cipher: Option<&Arc<dyn crate::encryption::cipher::Cipher>>,
244) -> Result<()> {
245    let max_lsn = crate::storage::wal::segment_reader::max_lsn_in_dir(wal.wal_dir(), cipher)?;
246    seed_lsn_allocator_from_max(wal, max_lsn);
247    Ok(())
248}
249
250/// Move the allocator to `max_lsn + 1` (never backwards), given a max LSN that
251/// the caller has already determined (Issue #3429).
252///
253/// Split out of [`seed_lsn_allocator_from_segments`] so the startup path can
254/// seed from the max LSN of the single full WAL read it performs, instead of
255/// scanning the segment directory a second time purely to compute it. See
256/// [`seed_lsn_allocator_from_segments`] for the seeding-policy rationale.
257fn seed_lsn_allocator_from_max(
258    wal: &ConcurrentWalSystem,
259    max_lsn: Option<crate::storage::wal::LSN>,
260) {
261    if let Some(max_lsn) = max_lsn {
262        let next = crate::storage::wal::LSN(max_lsn.0.saturating_add(1));
263        if next > wal.current_lsn() {
264            wal.set_next_lsn(next);
265        }
266    }
267}

293    hist.rebuild_temporal_index_from_versions();
294}

296/// Reconcile the namespace registry with the entities actually present after a
297/// load (Issue #3349, PR2). Every namespace that holds at least one node or edge
298/// in the rebuilt membership index is ensured-registered, so a scoped read
299/// naming it validates (rather than returning `NOT_FOUND`) after a load path
300/// that populates entities without the `namespaces.json` sidecar — notably
301/// `.albk` restore, whose payload carries the ride-along namespace on each
302/// entity but not the registry sidecar.
303///
304/// Idempotent and off the hot path: on a durable reopen (sidecar already loaded)
305/// every name is already registered, so this is a cheap no-op scan. A per-name
306/// persist failure never bricks startup — the entity data is already loaded and
307/// the membership index is authoritative for isolation.
308///
309/// v1 caveat: an *empty* explicitly-created namespace (registered but holding no
310/// entities) is not carried across `.albk` restore, because the registry sidecar
311/// is not part of the backup payload; folding the registry into the payload is a
312/// deliberate follow-up (`// TODO(#3349)`).
313fn reconcile_namespace_registry(db: &AletheiaDB) {
314    for ns in db.current.populated_namespaces() {
315        if let Err(_e) = db.namespaces.ensure_registered(&ns) {
316            #[cfg(feature = "observability")]
317            tracing::warn!(
318                namespace = %ns,
319                error = %_e,
320                "failed to auto-register namespace during load reconciliation"
321            );
322        }
323    }
324}

328    ///
329    /// The WAL is backed by a freshly created temporary directory that is
330    /// removed automatically when the returned [`AletheiaDB`] is dropped. No
331    /// state survives the process; nothing is loaded from prior runs.
332    ///
333    /// This is the right constructor for tests, scratch sessions, and quick
334    /// experiments. For durable storage that persists across restarts, use
335    /// [`Self::open`] — the one-line durable counterpart to this
336    /// constructor. Power users needing full control can call
337    /// [`Self::with_unified_config`] with a config built from
338    /// [`crate::config::durable_config_for_data_dir`], or
339    /// [`Self::open_from_env`] to honor the `ALETHEIADB_DATA_DIR`
340    /// environment variable.
341    ///
342    /// # Errors
343    ///
344    /// Returns an error if the temporary directory cannot be created or WAL
345    /// initialization fails inside it.
346    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
347    pub fn new() -> Result<Self> {
348        let tempdir = tempfile::Builder::new()
349            .prefix("aletheiadb-")
350            .tempdir()
351            .map_err(crate::core::error::Error::Io)?;
352        let wal_config = crate::config::WalConfigBuilder::new()
353            .wal_dir(tempdir.path().join("wal"))
354            .build();
355        let mut db = Self::with_full_config(AnchorConfig::default(), wal_config)?;
356        db._tempdir = Some(tempdir);
357        Ok(db)
358    }

391    /// Open (or create) a **durable** database rooted at `path`.
392    ///
393    /// This is the one-line entry point for embedding a durable AletheiaDB:
394    /// it creates the directory tree at `path` if absent, and opens an
395    /// existing one otherwise, replaying any prior state so calls are
396    /// idempotent across process restarts. Internally it is exactly
397    /// [`Self::with_unified_config`] with a config built by
398    /// [`crate::config::durable_config_for_data_dir`] — WAL + index
399    /// persistence with `load_on_startup`, group-commit durability — so
400    /// behavior stays in one canonical place and does not fork config
401    /// defaults. For an ephemeral, tempdir-backed database, use
402    /// [`Self::new`] instead.
403    ///
404    /// # Errors
405    ///
406    /// Returns an error if `path` is not writable, WAL initialization
407    /// fails, or index loading fails. Never falls back to an ephemeral
408    /// database on failure.
409    ///
410    /// # Example
411    ///
412    /// ```rust
413    /// use aletheiadb::{AletheiaDB, PropertyMapBuilder};
414    ///
415    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
416    /// let dir = tempfile::tempdir()?;
417    ///

419    ///     let db = AletheiaDB::open(dir.path())?;
420    ///     db.create_node(

427    /// // Reopening the same path replays the prior state.
428    /// let db = AletheiaDB::open(dir.path())?;
429    /// let node = db.get_node(node_id)?;
430    /// assert_eq!(
431    ///     node.properties.get("name").and_then(|v| v.as_str()),
432    ///     Some("Alice")

439        Self::with_unified_config(crate::config::durable_config_for_data_dir(

505    /// Create a new database with unified configuration.
506    ///
507    /// This method accepts a [`AletheiaDBConfig`] which consolidates all configuration
508    /// settings for the database, including WAL, historical storage, and vector indexes.
509    ///
510    /// # Errors
511    ///
512    /// Returns an error if WAL initialization fails (e.g., cannot create WAL directory).
513    ///
514    /// # Example
515    ///
516    /// ```ignore
517    /// use aletheiadb::{AletheiaDB, config::AletheiaDBConfig, config::WalConfigBuilder};
518    ///
519    /// let config = AletheiaDBConfig::builder()
520    ///     .wal(WalConfigBuilder::new()
521    ///         .with_validated(32, 2048, 64 * 1024, 64 * 1024 * 1024, 10, 10).unwrap()
522    ///         .build())
523    ///     .build();
524    ///
525    /// let db = AletheiaDB::with_unified_config(config)?;
526    /// ```
527    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
528    pub fn with_unified_config(config: AletheiaDBConfig) -> Result<Self> {
529        let result = (|| {
530            // Rebound mutable so the durable encryption-state authority
531            // (Issue #3616) can override `config.encryption` below.
532            let mut config = config;
533            let durability_mode = config.wal.durability_mode;
534
535            // Reconfigure the process-global string-interner cap from
536            // `persistence.max_interned_strings` BEFORE any WAL bootstrap
537            // deserialization (which interns property keys) or index load (which
538            // re-interns persisted strings and validates the persisted count
539            // against this cap). Doing it here guarantees both the runtime
540            // intern cap and the load-validation cap reflect the configured
541            // value for the rest of open(). Configurable interner cap fix.
542            //
543            // A cap of 0 would reject ALL interning and brick every write and
544            // open(); reject it with a clear error instead of silently bricking
545            // the DB. A cap above `u32::MAX` is unreachable (ids are `AtomicU32`),
546            // so it is clamped so the effective cap is honest.
547            //
548            // The interner is process-global (Issue #3724): rather than
549            // last-open-wins (which would let a later, LOWER cap refuse writes on
550            // an already-open database), `apply_interner_cap` applies a
551            // max-of-all-opens high-water mark — the first open establishes the
552            // baseline and later opens may only RAISE the shared cap, never lower
553            // it, warning loudly when a lower request is clamped up.
554            let effective_interner_cap =
555                validate_interner_cap(config.persistence.max_interned_strings)?;

558            // Capture provenance-chain settings before `config.wal.wal_dir` is

560            // dir (the parent of the WAL dir) unless an explicit override is set.
561            let chain_config = config.chain.clone();

563                .wal
564                .wal_dir
565                .parent()
566                .map(std::path::Path::to_path_buf)
567                .unwrap_or_else(|| std::path::PathBuf::from("."));

569            // Issue #3616: the durable encryption-state authority
570            // (`{persistence.data_dir}/encryption.state`) is the AUTHORITY for
571            // the encryption on/off decision — it WINS over the TOML/env-derived
572            // `EncryptionConfig`. This is what lets an `encryption enable`
573            // migration flip a database to encrypted-at-rest so the NEXT open()
574            // uses the cipher even though the operator's plaintext config is
575            // unchanged. Precedence & fail-closed contract (see
576            // `db::encryption_state`):
577            //   * absent           -> no override; today's TOML/env behavior.
578            //   * enabled + source -> force encryption ON under that key source.
579            //   * disabled         -> force encryption OFF.
580            //   * corrupt / enabled-without-source -> read returns Err -> open
581            //     aborts loudly (fail-closed; NEVER a silent plaintext fallback).
582            // On DIVERGENCE a LOUD warning naming BOTH sources is logged so an
583            // operator is never silently overridden.
584            if config.persistence.enabled
585                && let Some(authority) = crate::db::encryption_state::read_encryption_state(
586                    &config.persistence.data_dir,
587                )?

589                if authority.enabled != config.encryption.enabled {
590                    crate::db::encryption_state::log_authority_warning(&format!(
591                        "encryption-state authority at {} records enabled={}, OVERRIDING the \
592                         loaded config (encryption.enabled={}). The durable authority wins; \
593                         opening with enabled={}.",
594                        crate::db::encryption_state::encryption_state_path(
595                            &config.persistence.data_dir
596                        )
597                        .display(),
598                        authority.enabled,
599                        config.encryption.enabled,
600                        authority.enabled,
601                    ));

603                if authority.enabled {
604                    // The reader guarantees an enabled authority carries a key
605                    // source. Compose that key SOURCE with the config's
606                    // algorithm/audit; a missing/unreadable key then fails
607                    // `EncryptionManager::from_config` below, so open() refuses
608                    // rather than falling back to plaintext (fail-closed).
609                    let key_source =
610                        authority
611                            .key_source
612                            .ok_or_else(|| -> crate::core::error::Error {
613                                crate::core::error::StorageError::InconsistentState {
614                                    reason:
615                                        "encryption-state authority is enabled but resolved no \
616                                         key source"
617                                            .to_string(),
618                                }
619                                .into()
620                            })?;

622                    config.encryption.key_provider = key_source;

624                    // algorithm the enable migration wrote under. Override the
625                    // operator's `config.encryption.algorithm` with it so the DB
626                    // reopens under the EXACT cipher it was written with — immune to
627                    // a later TOML edit (e.g. a concrete `chacha20` that Auto did not
628                    // resolve to) and to a cross-CPU host migration (`Auto` resolving
629                    // AES on one host, ChaCha on another). A legacy `version=1`
630                    // authority pins nothing (`None`) → the config algorithm stands,
631                    // exactly the prior behavior.

633                        config.encryption.algorithm = pinned;
634                    }
635                } else {
636                    config.encryption.enabled = false;
637                }
638            }

640            // Issue #3616 PR4: an INTERRUPTED encrypted → plaintext disable leaves
641            // the durable authority STILL `enabled` (the flip to `disabled` is the
642            // commit point, done LAST), so `config.encryption.enabled` is `true`
643            // here even though the on-disk bytes are being unwrapped to plaintext.
644            // But the disable END STATE is plaintext, and a live encrypted index
645            // manager / cold store would RE-ENCRYPT over the freshly-unwrapped bytes
646            // on the next background persist / cold migration. So on a pending-disable
647            // startup we build every at-rest layer PLAINTEXT (the end state) by
648            // FORCING `enabled=false`, and resolve the DECRYPT ciphers from the
649            // pending disable ledger for the transient resume read passes: the
650            // pre-read WAL hook installs the decrypt keyring so the replay decrypts
651            // the still-encrypted segments, the index unwrap decrypts each `AEIX`
652            // file explicitly, and the cold store is built under the decrypt cold DEK
653            // so its `ACV1` → bare unwrap can decrypt. `None` on the common path (no
654            // pending disable) reproduces prior behavior exactly. A ledger is either
655            // an enable OR a disable ledger, never both, so this never collides with
656            // the enable-resume path below.
657            let disable_resume = if config.persistence.enabled && config.encryption.enabled {
658                crate::db::rotation::disable_resume_ciphers(
659                    &config.persistence.data_dir,
660                    config.encryption.algorithm,
661                )?
662            } else {
663                None
664            };

667                // the decrypt ciphers resolved above, and the authority is flipped to
668                // `disabled` only once every layer is unwrapped.
669                config.encryption.enabled = false;
670            }

673            // the durable authority NOT yet flipped, so `config.encryption.enabled`
674            // is still `false` here even though the on-disk WAL/index/cold bytes may
675            // already have been (partially) migrated. On that branch, resolve the
676            // enable DEKs from the pending ENABLE ledger so the at-rest layers below
677            // (index manager, cold store) are built UNDER those DEKs — otherwise
678            // they would be built plaintext and then fail to read (or over-write
679            // with plaintext) the freshly-wrapped `AEIX`/`ACV1` bytes the resume
680            // passes produce. The WAL is handled separately by the pre-read hook.
681            // `None` on the common path (no pending enable) reproduces prior
682            // behavior exactly.

684                crate::db::rotation::enable_resume_ciphers(
685                    &config.persistence.data_dir,
686                    config.encryption.algorithm,
687                )?

689                None

697                    })?;

708            } else {
709                None

722            // otherwise `None` reproduces prior behavior exactly. Computed here,
723            // before the WAL is built, so both keyrings are provisioned in

726            {

728                    &config.persistence.data_dir.join("indexes"),
729                    &config.wal.wal_dir,
730                    &config.persistence.data_dir,
731                )?)

733                None

738                num_stripes: config.wal.num_stripes,
739                stripe_capacity: config.wal.stripe_capacity,
740                segment_size: config.wal.segment_size,
741                segments_to_retain: config.wal.segments_to_retain,

746                durability_mode,

748                wal_cipher: wal_cipher.clone(),

756            // single decode for all three startup passes that historically each
757            // re-read every segment: (1) the LSN-allocator seed below, (2) the
758            // constraint declaration net-state, and (3) the differential
759            // node/edge replay. `read_from` mirrors the recovery reader's cipher
760            // behavior (plaintext today), so for an encrypted WAL its max LSN
761            // would omit encrypted entries; the cipher-aware directory scan is
762            // used as a targeted fallback below only when a cipher is
763            // configured, preserving the #3420/#3430 encrypted seed exactly.
764            //
765            // Issue #3617 PR2 (crash-consistency): if a WAL rotation ledger is
766            // PENDING, install BOTH the old and new WAL key generations into the
767            // keyring BEFORE the replay read, so a half-rotated WAL (old-DEK +
768            // new-DEK segments both on disk) decrypts regardless of generation.
769            // Without this the read below fails on new-DEK frames before
770            // `resume_pending_rotation` (further down) can install the new
771            // generation. Additive: a no-op when no rotation is pending.
772            if config.encryption.enabled && config.persistence.enabled {
773                crate::db::rotation::install_pending_wal_generations(
774                    &wal,
775                    &config.encryption,
776                    &config.persistence.data_dir,
777                )?;

779                // Issue #3616 PR3: an INTERRUPTED plaintext→encrypted enable leaves
780                // the authority NOT yet flipped, so `config.encryption.enabled` is
781                // false here even though the WAL may already have been rolled to
782                // encrypted. Install the WAL keyring from the pending enable ledger
783                // BEFORE the replay read so those encrypted segments decrypt
784                // (otherwise the read below mis-reads ciphertext as undecodable
785                // plaintext and fails). `resume_pending_enable` (further down)
786                // finishes the migration. A no-op when no enable ledger is pending

789                    &wal,
790                    &config.encryption,
791                    &config.persistence.data_dir,
792                )?;
793                // Issue #3616 PR4: mirror for an INTERRUPTED disable — the WAL was
794                // built plaintext (end state) above, but the on-disk segments may
795                // still be encrypted (uninstall/retire not finished). Install the
796                // DECRYPT keyring from the pending disable ledger BEFORE the replay
797                // read so those encrypted segments decrypt; `resume_pending_disable`
798                // (further down) then uninstalls it back to plaintext and retires the
799                // encrypted tail. A no-op when no disable ledger is pending or the
800                // encrypted WAL tail is already retired.

802                    &wal,
803                    &config.encryption,
804                    &config.persistence.data_dir,
805                )?;

809            // index load so no later pass re-reads the segment directory.
810            let mut startup_wal_entries = wal.read_from(crate::storage::wal::LSN::initial())?;

812            // Issue #3420: seed the LSN allocator past every LSN already durable
813            // in existing WAL segments, BEFORE any write is accepted. Without
814            // this, a restarted process re-allocates LSNs starting at 1,
815            // breaking LSN total ordering across segments and placing new
816            // writes below the index manifest LSN (so the next startup's
817            // differential replay silently skips them).

819                // Encrypted segments are invisible to the cipher-less read
820                // above; fall back to the cipher-aware directory scan so the
821                // seed still accounts for encrypted entries (unchanged #3420
822                // behavior for encrypted WALs; the constraint/replay passes
823                // stay cipher-less exactly as before, per #3430).
824                crate::storage::wal::segment_reader::max_lsn_in_dir(
825                    wal.wal_dir(),
826                    wal_cipher.as_ref(),
827                )?
828            } else {
829                // `read_from` returns entries globally sorted by LSN
830                // (segment_reader: `entries.sort_by_key(|e| e.lsn)`), so the
831                // last element carries the max LSN in O(1). Empty WAL -> `None`
832                // -> the seed is a no-op.
833                startup_wal_entries.last().map(|e| e.lsn)
834            };
835            seed_lsn_allocator_from_max(&wal, seed_max_lsn);

837            // Create persistence manager if enabled. When encryption is
838            // enabled, thread the index cipher through so index files are
839            // encrypted at rest (Issue #481); mirrors the WAL cipher wiring
840            // above. Encryption disabled => None => plaintext, unchanged.
841            let persistence_manager = if config.persistence.enabled {
842                // Issue #3616 PR3: on an interrupted-enable resume the manager must
843                // read/write under the enable index DEK (checkpoints ride it), so a
844                // half-wrapped index dir loads and the resume wrap pass publishes
845                // `AEIX` the same open() then reads back. Otherwise the index cipher
846                // comes from the (enabled) encryption manager as before.
847                let index_cipher = enable_resume
848                    .as_ref()
849                    .map(|e| Arc::clone(&e.index))
850                    .or_else(|| {
851                        encryption_manager
852                            .as_ref()
853                            .map(|mgr| Arc::clone(mgr.index_cipher()))
854                    });
855                // Issue #488 version-provisioning: build the index keyring at the
856                // resolved on-disk version so `index_rotation_status` /
857                // `encryption verify` classify rotated files correctly after a
858                // cold reopen and the next rotation increments from the real
859                // version. Reads stay `match_any` (identical to `with_cipher`);
860                // `None` falls back to `with_cipher` (unchanged behavior).
861                let manager = match provisioned_key_version {
862                    Some(v) => crate::storage::index_persistence::IndexPersistenceManager::with_cipher_versioned(
863                        &config.persistence.data_dir,
864                        index_cipher,
865                        v,
866                    ),
867                    None => crate::storage::index_persistence::IndexPersistenceManager::with_cipher(
868                        &config.persistence.data_dir,
869                        index_cipher,
870                    ),
871                };
872                Some(Arc::new(manager))
873            } else {
874                None
875            };

877            // Create persistence tracker if persistence is enabled
878            let persistence_tracker = if config.persistence.enabled {
879                Some(Arc::new(PersistenceTracker::new()))
880            } else {
881                None
882            };

885            let enable_cold_storage = config.historical.enable_cold_storage;
886            let cold_storage_path = config.historical.cold_storage_path.clone();

889            // quota) before `config` fields are moved, so the broadcaster is
890            // constructed from the unified config rather than defaults.
891            let changefeed_config = config.changefeed.clone();

895            // in-memory-only otherwise. Loaded here so pins survive restart.
896            let snapshots = Arc::new(crate::db::snapshot::SnapshotRegistry::open(
897                crate::db::snapshot::registry_path_for(&config.persistence),
898            )?);

901            // `{data_dir}/namespaces.json` when index persistence is enabled,
902            // in-memory-only otherwise. Loaded here so registrations survive restart.
903            let namespaces = Arc::new(crate::db::namespace::NamespaceRegistry::open(
904                crate::db::namespace::registry_path_for(&config.persistence),
905            )?);

908            // keyring fail-closed and run breadcrumb crash-recovery. Co-located

910            // in-memory-only when index persistence is disabled.

944                persistence_thread_handle: None,

975            // generation by the time it is read back (a crash mid-rotation left

980            // old-generation WAL segments (indexes are not loaded yet, so a

991                    &config.encryption,
992                    Some(&db.wal),
993                    None,

998            // migration BEFORE loading indexes, so any half-wrapped index/cold
999            // bytes are completed before the index directory is read back. This is
1000            // DISTINCT from `resume_pending_rotation` above: an interrupted enable
1001            // has NOT yet flipped the encryption.state authority, so
1002            // `config.encryption.enabled` is still false and that gated path never
1003            // fires. A guarded no-op when no pending-enable ledger exists.
1004            db.resume_pending_enable()?;
1005
1006            // Issue #3616 PR4: resume an interrupted encrypted → plaintext disable
1007            // migration BEFORE loading indexes, so any half-unwrapped index bytes are
1008            // completed (and the WAL finished uninstalling + retiring its encrypted
1009            // tail) before the index directory is read back by the PLAINTEXT manager.
1010            // Distinct from the rotation/enable paths: a pending disable ledger forced
1011            // `config.encryption.enabled=false` above, so those gated paths never
1012            // fire. A guarded no-op when no pending-disable ledger exists.

1015            // Load indexes on startup if enabled
1016            if let Some(ref manager) = persistence_manager
1017                && config.persistence.load_on_startup
1018            {
1019                let loaded_lsn =
1020                    crate::storage::index_persistence::operations::load_indexes_startup(
1021                        manager,
1022                        &db.current,
1023                        &db.historical,
1024                        &db.node_id_gen,
1025                        &db.edge_id_gen,
1026                        &db.version_id_gen,
1027                    );

1029                // Issue #3420: the manifest LSN is a second durability floor for
1030                // the allocator. Normally the segment scan above already seeded
1031                // the allocator higher, but if the WAL was truncated below the
1032                // manifest LSN (e.g. LSN-based truncation after cold-storage
1033                // migration), the segments alone under-seed it. The manifest
1034                // stores the next-to-allocate LSN captured at snapshot time
1035                // (see `IndexManifest::lsn`), so it is itself a valid "next".
1036                if let Some(lsn) = loaded_lsn {
1037                    let manifest_floor = crate::storage::wal::LSN(lsn);
1038                    if manifest_floor > db.wal.current_lsn() {
1039                        db.wal.set_next_lsn(manifest_floor);
1040                    }
1041                }
1042
1043                // Initialize tracker LSNs from the loaded manifest
1044                if let Some(ref tracker) = persistence_tracker
1045                    && let Some(lsn) = loaded_lsn
1046                {
1047                    tracker.set_start_lsn(lsn);
1048                    tracker.update_last_persisted_counts(
1049                        db.current.node_count() as u64,
1050                        db.current.edge_count() as u64,
1051                    );
1052                    // Also initialize string count from current state
1053                    tracker.update_last_persisted_string_count(
1054                        crate::core::GLOBAL_INTERNER.len() as u64
1055                    );
1056                }

1058                // Replay constraint declarations from the full WAL history before the
1059                // differential node/edge replay.  Constraint WAL entries may predate
1060                // the snapshot LSN and would be skipped by the differential replay.
1061                //
1062                // Issue #3429: fold the constraint net-state out of the single
1063                // full read taken at startup (`startup_wal_entries` spans the
1064                // whole history from LSN 0), so this no longer re-reads the
1065                // segment directory. Below-manifest declarations are still
1066                // applied because the borrowed slice includes them.
1067                crate::storage::recovery::apply_constraint_declarations(
1068                    &startup_wal_entries,
1069                    &db.constraint_registry,
1070                );

1072                // Replay WAL entries that occurred after the persisted snapshot
1073                // This ensures no data loss if the WAL is ahead of the indexes (e.g. crash before persist)
1074                //
1075                // Issue #3419: the manifest LSN is the NEXT-to-allocate LSN
1076                // captured *before* the snapshot was taken (see
1077                // `IndexManifest::lsn`). Entries with LSN < manifest.lsn are
1078                // guaranteed to be in the snapshot; entries with LSN >=
1079                // manifest.lsn may or may not be. Replay therefore starts AT
1080                // the manifest LSN (inclusive) — the previous `.next()` here
1081                // skipped the first post-persist write entirely — and the
1082                // replay itself is idempotent for already-applied entries
1083                // (see the re-application guards in
1084                // `replay_wal_into_storage_with_constraints`).
1085                let start_lsn = match loaded_lsn {
1086                    Some(lsn) => crate::storage::wal::LSN(lsn),
1087                    None => {
1088                        // Safety check: if we have data but no LSN, replaying from initial is dangerous
1089                        // as it might overwrite existing data with old WAL entries or duplicate IDs.
1090                        if db.current.node_count() > 0 {
1091                            #[cfg(feature = "observability")]
1092                            tracing::error!(
1093                                "Database contains data ({} nodes) but no persistence LSN found. \
1094                             Skipping WAL replay to prevent potential data corruption. \
1095                             This may indicate a missing or corrupted manifest file.",
1096                                db.current.node_count()
1097                            );
1098                            #[cfg(not(feature = "observability"))]
1099                            eprintln!(
1100                                "ERROR: Database contains data ({} nodes) but no persistence LSN found. \
1101                             Skipping WAL replay to prevent potential data corruption.",
1102                                db.current.node_count()
1103                            );
1104                            // Skip replay by setting start_lsn to current WAL LSN (effectively no-op)
1105                            db.wal.current_lsn()
1106                        } else {
1107                            crate::storage::wal::LSN::initial()
1108                        }
1109                    }
1110                };
1111
1112                // Capture initial version ID before replay
1113                let initial_version_id = db.version_id_gen.current();
1114
1115                // Issue #3429: feed the differential replay the suffix of the
1116                // single startup read (entries with LSN >= start_lsn), rather
1117                // than re-reading the segment directory a third time.
1118                //
1119                // CORRECTNESS: `read_from` returns entries globally sorted by
1120                // LSN (segment_reader: `entries.sort_by_key(|e| e.lsn)`), so
1121                // `partition_point` finds the first index with LSN >= start_lsn
1122                // and `drain(..idx)` discards exactly the entries below it — a
1123                // contiguous, in-place split with no extra allocation. This is
1124                // byte-identical to what `read_from(start_lsn)` would have
1125                // produced (the same LSN-ordered tail the framing resolver
1126                // expects). If the sort is ever removed upstream, this
1127                // partition (and the O(1) seed above) breaks — keep them
1128                // together.
1129                let split = startup_wal_entries.partition_point(|entry| entry.lsn < start_lsn);
1130                startup_wal_entries.drain(..split);
1131                let replay_entries = std::mem::take(&mut startup_wal_entries);
1132
1133                let mut historical_guard = db.historical.write();
1134                let (_final_lsn, max_node_id, max_edge_id, next_version_id) =
1135                    crate::storage::recovery::replay_entries_into_storage_with_constraints(
1136                        &db.wal,
1137                        replay_entries,
1138                        &db.current,
1139                        &mut historical_guard,
1140                        initial_version_id,
1141                        Some(&db.constraint_registry),
1142                    )?;
1143                drop(historical_guard);
1144
1145                // Rebuild reservation index from currently-valid nodes for each declared constraint.
1146                for (label_str, property_str) in db.constraint_registry.list() {
1147                    if let (Some(label_id), Some(property_id)) = (
1148                        crate::core::interning::GLOBAL_INTERNER.get_id(&label_str),
1149                        crate::core::interning::GLOBAL_INTERNER.get_id(&property_str),
1150                    ) {
1151                        let nodes = db.current.get_nodes_by_label(&label_str);
1152                        db.constraint_registry
1153                            .rebuild_from_nodes(&nodes, label_id, property_id);
1154                    }
1155                }
1156
1157                // Update ID generators to account for replayed entities.
1158                // This ensures that subsequent writes use IDs that don't collide with replayed data.
1159                if let Some(max_nid) = max_node_id {
1160                    db.node_id_gen.ensure_at_least(max_nid + 1);
1161                }
1162                if let Some(max_eid) = max_edge_id {
1163                    db.edge_id_gen.ensure_at_least(max_eid + 1);
1164                }
1165                db.version_id_gen.ensure_at_least(next_version_id);
1166            }
1167
1168            // When only the WAL is configured (no index persistence), replay the full
1169            // WAL from the beginning to restore node/edge data AND constraint declarations.
1170            if persistence_manager.is_none() {
1171                let initial_version_id = db.version_id_gen.current();
1172
1173                // Issue #3429: replay directly from the single startup read
1174                // instead of re-decoding the segment directory. This branch
1175                // replays from LSN 0, so it consumes every entry; the inline
1176                // constraint arms in the replay loop recover constraint state
1177                // (no separate constraint pass is needed here). `std::mem::take`
1178                // moves the entries out of the (persistence-branch-shared)
1179                // binding; only one of the two branches runs.
1180                let replay_entries = std::mem::take(&mut startup_wal_entries);
1181
1182                let mut historical_guard = db.historical.write();
1183                let (_final_lsn, max_node_id, max_edge_id, next_version_id) =
1184                    crate::storage::recovery::replay_entries_into_storage_with_constraints(
1185                        &db.wal,
1186                        replay_entries,
1187                        &db.current,
1188                        &mut historical_guard,
1189                        initial_version_id,
1190                        Some(&db.constraint_registry),
1191                    )?;
1192                drop(historical_guard);
1193
1194                // Rebuild reservation index.
1195                for (label_str, property_str) in db.constraint_registry.list() {
1196                    if let (Some(label_id), Some(property_id)) = (
1197                        crate::core::interning::GLOBAL_INTERNER.get_id(&label_str),
1198                        crate::core::interning::GLOBAL_INTERNER.get_id(&property_str),
1199                    ) {
1200                        let nodes = db.current.get_nodes_by_label(&label_str);
1201                        db.constraint_registry
1202                            .rebuild_from_nodes(&nodes, label_id, property_id);
1203                    }
1204                }
1205
1206                if let Some(max_nid) = max_node_id {
1207                    db.node_id_gen.ensure_at_least(max_nid + 1);
1208                }
1209                if let Some(max_eid) = max_edge_id {
1210                    db.edge_id_gen.ensure_at_least(max_eid + 1);
1211                }
1212                db.version_id_gen.ensure_at_least(next_version_id);
1213            }
1214
1215            // Issue #3617 PR2 (resume-path double-crash data-loss fix): PHASE 2 of
1216            // a startup-resumed WAL key rotation. `resume_pending_rotation` above
1217            // installed both WAL generations but DEFERRED retiring the
1218            // old-generation segments (indexes were not loaded yet, so a persist
1219            // then would have snapshotted empty state). Now that the snapshot is
1220            // loaded and the WAL has been replayed into current state, retire the
1221            // old-generation segments — but only AFTER a SYNCHRONOUS
1222            // `persist_indexes()` captures the replayed `[checkpoint, seal)` writes
1223            // (run inside `finalize_resumed_wal_rotation`, after the seal and
1224            // before the retire). So no old-generation WAL segment is deleted until
1225            // its entries are durable in the index snapshot, closing the window
1226            // where a second crash before the async checkpoint lost them. A no-op
1227            // when no WAL rotation is pending. MUST run BEFORE the background
1228            // persistence thread starts, so the synchronous persist is the one
1229            // that closes the window (not a racy async checkpoint).
1230            if let Some(ref manager) = persistence_manager
1231                && config.encryption.enabled
1232                && config.persistence.load_on_startup
1233            {
1234                crate::db::rotation::finalize_resumed_wal_rotation(
1235                    manager,
1236                    &config.encryption,
1237                    &db.wal,
1238                    || db.persist_indexes(),
1239                )?;
1240            }
1241
1242            // Start background persistence thread if enabled.
1243            //
1244            // Issue #3616 PR3 (NIT-2) load-bearing invariant: on an interrupted-enable
1245            // resume this worker is spawned HERE, BEFORE `resume_pending_enable_cold`
1246            // runs the cold wrap (below). That is safe ONLY because both at-rest layers
1247            // were built under the enable-resume ciphers on this path — the index
1248            // manager under `enable_resume.index` (so a persist writes `AEIX`, never
1249            // plaintext over the freshly-wrapped files) and the cold store under
1250            // `enable_resume.cold` (so a hot→cold migration writes `ACV1`, and the wrap
1251            // pass is idempotent). A future refactor that built either layer PLAINTEXT
1252            // on the resume path would reintroduce a plaintext/bare-over-encrypted race
1253            // with this running worker: the worker must NEVER run with a plaintext
1254            // manager while an enable ledger is pending. The `index_cipher`
1255            // (config.rs) and cold `with_cipher_versioned` (below) wiring enforce this.
1256            if let Some(ref tracker) = persistence_tracker
1257                && let Some(ref manager) = persistence_manager
1258            {
1259                debug_assert!(
1260                    enable_resume.is_none() || manager.keyring().is_some(),
1261                    "enable-resume must build the index manager under the enable index \
1262                     DEK; a plaintext manager with a pending enable ledger would let this \
1263                     worker persist plaintext over encrypted index files (Issue #3616 PR3)"
1264                );
1265                // Issue #3616 PR4: the inverse invariant — a pending disable ledger
1266                // MUST build the index manager PLAINTEXT (the end state). An encrypted
1267                // manager with a pending disable ledger would let this worker persist
1268                // `AEIX` over the freshly-unwrapped plaintext index files. The
1269                // `disable_resume` force-`enabled=false` wiring above enforces this.
1270                debug_assert!(
1271                    disable_resume.is_none() || manager.keyring().is_none(),
1272                    "disable-resume must build the index manager PLAINTEXT; an encrypted \
1273                     manager with a pending disable ledger would let this worker persist \
1274                     encrypted index files over the unwrapped plaintext snapshot (Issue #3616 PR4)"
1275                );
1276                let handle = spawn_background_persistence_thread(
1277                    Arc::clone(&db.current),
1278                    Arc::clone(&db.historical),
1279                    Arc::clone(&db.temporal_indexes),
1280                    Arc::clone(&db.wal),
1281                    Arc::clone(manager),
1282                    Arc::clone(tracker),
1283                    config.persistence.policies.clone(),
1284                    Arc::clone(&db.persistence_thread_stopped),
1285                );
1286                db.persistence_thread_handle = Some(handle);
1287            }
1288
1289            wire_temporal_indexes(&db);
1290            reconcile_namespace_registry(&db);
1291
1292            // Initialize cold storage if enabled
1293            if enable_cold_storage && let Some(cold_storage_path) = cold_storage_path {
1294                // Create Redb cold storage backend, with optional encryption cipher
1295                let mut cold_storage = RedbColdStorage::new(&cold_storage_path, RedbConfig::new())?;
1296
1297                if let Some(ref enc_mgr) = encryption_manager {
1298                    // Issue #3617 PR3 version-provisioning: build the cold keyring
1299                    // at the provisioned key version (the max on-disk key version,
1300                    // or `ledger.new_version - 1` mid-rotation) so freshly written
1301                    // cold values stamp the real version and stay in lockstep with
1302                    // index/WAL after a rotate-then-reopen. Reads stay `match_any`
1303                    // (identical to `with_cipher`); `None` falls back to
1304                    // `with_cipher` (unchanged behavior for a never-rotated DB).
1305                    cold_storage = match provisioned_key_version {
1306                        Some(v) => {
1307                            cold_storage.with_cipher_versioned(Arc::clone(enc_mgr.cold_cipher()), v)
1308                        }
1309                        None => cold_storage.with_cipher(Arc::clone(enc_mgr.cold_cipher())),
1310                    };
1311                } else if let Some(ref e) = enable_resume {
1312                    // Issue #3616 PR3: on an interrupted-enable resume the authority
1313                    // is not yet flipped (no encryption manager), but the cold store
1314                    // must be built under the enable cold DEK so `ACV1` values
1315                    // written by the resume wrap pass (below) are read back this same
1316                    // open(). Provisioned at the enable version.
1317                    cold_storage = cold_storage.with_cipher_versioned(
1318                        Arc::clone(&e.cold),
1319                        crate::db::rotation::ENABLE_KEY_VERSION,
1320                    );
1321                } else if let Some(ref d) = disable_resume {
1322                    // Issue #3616 PR4: on an interrupted-disable resume the authority
1323                    // is not yet flipped (no encryption manager), but the cold store
1324                    // must be built under the DECRYPT cold DEK — at the recorded
1325                    // (to-be-retired) version — so `resume_pending_disable_cold`'s
1326                    // `ACV1` → bare unwrap pass (below) can decrypt each wrapped value.
1327                    cold_storage =
1328                        cold_storage.with_cipher_versioned(Arc::clone(&d.cold), d.version);
1329                }
1330
1331                let cold_storage = Arc::new(cold_storage);
1332
1333                // Create tiered storage with warm cache configuration
1334                let tiered_config = TieredStorageConfig::default();
1335                let tiered_storage = TieredStorage::new(tiered_config, cold_storage);
1336
1337                // Issue #3617 PR3: finish a startup-resumed COLD key rotation. The
1338                // cold store is built here — AFTER index/WAL resume — so it is the
1339                // last layer to settle; this completes the bulk re-encrypt from the
1340                // durable redb cursor (idempotent, skips already-wrapped values),
1341                // retires the old cold generation, and clears the rotation ledger.
1342                // A no-op when no rotation is pending or cold is not in scope.
1343                if let Some(ref manager) = persistence_manager
1344                    && config.encryption.enabled
1345                {
1346                    crate::db::rotation::finalize_resumed_cold_rotation(
1347                        manager,
1348                        &config.encryption,
1349                        tiered_storage.cold_storage(),
1350                    )?;
1351                }
1352
1353                // Merge the cold tier's persisted extent bounds into the
1354                // temporal index so `temporal_extent` spans history migrated to
1355                // cold before this restart (Issue #3389). `wire_temporal_indexes`
1356                // above rebuilt the extent aggregate from the hot tier only;
1357                // absent/empty cold metadata leaves it untouched, and merging
1358                // only ever widens (never narrows) the reported extent.
1359                if let Some(bounds) = tiered_storage.cold_storage().get_temporal_extent_bounds()? {
1360                    db.temporal_indexes.merge_extent_bounds(
1361                        bounds.valid_earliest,
1362                        bounds.valid_latest,
1363                        bounds.tx_earliest,
1364                        bounds.tx_latest,
1365                    );
1366                }
1367
1368                // Wire tiered storage to historical storage
1369                // Note: migration_age_threshold and max_hot_versions from config.historical
1370                // are used by HistoricalStorage's migration logic, not by TieredStorage
1371                db.historical
1372                    .write()
1373                    .set_tiered_storage(Arc::new(tiered_storage));
1374
1375                // Issue #3616 PR3: finish an interrupted plaintext → encrypted
1376                // enable migration's COLD layer, now that the cold store is wired.
1377                // `resume_pending_enable` (run before index load) deferred the cold
1378                // wrap + the authority flip + the ledger clear to here — this wraps
1379                // the bare cold values to `ACV1`, then (all layers settled) flips
1380                // the authority and clears the ledger, preserving the binding order.
1381                // A guarded no-op when no enable migration with a cold layer is
1382                // pending.
1383                db.resume_pending_enable_cold()?;
1384
1385                // Issue #3616 PR4: finish an interrupted encrypted → plaintext
1386                // disable migration's COLD layer, now that the cold store is wired
1387                // under the decrypt cold DEK. `resume_pending_disable` (run before
1388                // index load) deferred the cold unwrap + the authority flip + the
1389                // ledger clear to here — this unwraps each `ACV1` value back to bare,
1390                // then (all layers settled) flips the authority to `disabled` and
1391                // clears the ledger, preserving the binding order. A guarded no-op
1392                // when no disable migration with a cold layer is pending.
1393                db.resume_pending_disable_cold()?;
1394            } else {
1395                // Issue #3616 PR3: this open() did NOT build a cold tier
1396                // (enable_cold_storage disabled or no cold path). If an interrupted
1397                // enable recorded a cold layer as Pending, `resume_pending_enable_cold`
1398                // (with its fail-closed guard) never runs — so the ledger would wedge
1399                // forever (authority never flips, no diagnostic). Fail LOUDLY here
1400                // with the same actionable "reopen with the cold tier enabled" error
1401                // instead of silently stranding the migration.
1402                db.fail_if_pending_enable_cold_without_tier()?;
1403                // Issue #3616 PR4: the same guard for an interrupted disable whose
1404                // ledger records a cold layer but this open() built no cold tier.
1405                db.fail_if_pending_disable_cold_without_tier()?;
1406            }
1407
1408            // Crypto-shred subject-keyring re-wrap (Slice 5): the LAST layer of a
1409            // full-MEK rotation to settle. Runs AFTER the index/WAL/cold resume
1410            // passes above, unconditionally (whether or not a cold tier was built),
1411            // so a rotation that crashed after the ledger recorded
1412            // `layer.subject_keyring=pending` finishes here — re-wrapping every
1413            // per-subject DEK under the new MEK and clearing the ledger. Idempotent
1414            // and a guarded no-op when the layer is not in flight; requires the OLD
1415            // MEK to still be the configured key source (reopen-under-old-key
1416            // contract). Gated on `audit-export` (the crypto-shred feature).
1417            #[cfg(feature = "audit-export")]
1418            if let Some(manager) = persistence_manager
1419                .as_ref()
1420                .filter(|_| config.encryption.enabled)
1421            {
1422                crate::db::rotation::finalize_resumed_subject_keyring_rotation(
1423                    manager,
1424                    &config.encryption,
1425                    db.crypto_shred.as_ref(),
1426                )?;
1427            }
1428
1429            // Fail-closed counterpart (Slice 5 review 4b): when THIS binary was
1430            // built WITHOUT the crypto-shred (`audit-export`) feature there is no
1431            // finalizer for a subject-keyring re-wrap left `Pending` by a crash.
1432            // Rather than wedge the rotation ledger `Pending` forever with no
1433            // diagnostic, surface an actionable `FailedPrecondition` telling the
1434            // operator to reopen with the crypto-shred feature enabled. A guarded
1435            // no-op when no such pending re-wrap exists.
1436            #[cfg(not(feature = "audit-export"))]
1437            if let Some(manager) = persistence_manager
1438                .as_ref()
1439                .filter(|_| config.encryption.enabled)
1440            {
1441                crate::db::rotation::fail_if_pending_subject_keyring_without_crypto_shred(manager)?;
1442            }
1443
1444            seed_startup_current_timestamp(&db)?;
1445
1446            // Wire the opt-in provenance hash chain (Issue #3351). Constructed
1447            // after WAL replay + state restore so the genesis anchors the
1448            // post-recovery LSN and the tail rebuild can fold every restored
1449            // transaction. Nothing here runs (and no chain dir is created) when
1450            // the chain is disabled — the default — preserving byte-identical
1451            // behavior.
1452            if chain_config.enabled {
1453                let source: Arc<dyn crate::provenance_chain::VersionSource + Send + Sync> =
1454                    Arc::new(crate::db::chain_source::DbVersionSource::new(Arc::clone(
1455                        &db.historical,
1456                    )));
1457                let genesis_lsn = db.wal.current_lsn().0;
1458                let genesis_ts = time::now().wallclock();
1459                let chain = crate::provenance_chain::ProvenanceChain::open(
1460                    &chain_config,
1461                    &chain_data_dir,
1462                    genesis_lsn,
1463                    genesis_ts,
1464                    source,
1465                )
1466                .map_err(|e| {
1467                    crate::core::error::Error::Other(format!(
1468                        "provenance hash chain open failed: {e}"
1469                    ))
1470                })?;
1471                // Rebuild the unsealed tail from replayed history, then start the
1472                // background sealer for live commits.
1473                db.rebuild_chain_tail(&chain);
1474                chain.start();
1475                db.chain = Some(chain);
1476            }
1477
1478            // Schema constraints (Issue #3378): load the durable sidecar LAST —
1479            // after WAL replay + index load have fully restored the string
1480            // interner — so the label/property strings re-intern to the same
1481            // ids that live writes use (loading earlier would key constraints by
1482            // pre-restore interner ids that no longer match). Tolerant load
1483            // (missing = empty; corrupt = warn + quarantine + empty) so a bad
1484            // sidecar never bricks startup. Ephemeral databases have no path.
1485            if let Some(ref path) = db.schema_constraint_path {
1486                crate::db::schema_constraint::load_sidecar(path, &db.constraint_registry);
1487            }
1488
1489            Ok(db)
1490        })();
1491        result.record_error_metric()
1492    }
