3//! Thin wrappers that build the subject-wrapping cipher from the database's

19/// HKDF component label for the subject-wrapping DEK. Yields HKDF info string
20/// `aletheiadb-subject-wrap-dek-v1` (see `KeyDerivation::derive_dek`).
21const SUBJECT_WRAP_COMPONENT: &str = "subject-wrap";
22
23impl AletheiaDB {
24    /// Build the subject-wrapping cipher by re-sourcing the MEK and deriving a
25    /// dedicated `subject-wrap` DEK. Requires encryption to be configured.
26    ///
27    /// This re-sources the MEK via the public `KeyProvider` seam and derives an
28    /// independent component DEK; it never touches the `EncryptionManager`'s own
29    /// private component ciphers.
30    fn subject_wrap_cipher(&self) -> Result<Box<dyn Cipher>, CryptoShredError> {
31        let cfg = self
32            .encryption_config
33            .as_ref()
34            .ok_or(CryptoShredError::EncryptionNotConfigured)?;
35        let provider = cfg
36            .key_provider
37            .build_provider()
38            .map_err(|e| CryptoShredError::Crypto(e.to_string()))?;
39        let mek = provider
40            .get_mek()
41            .map_err(|e| CryptoShredError::Crypto(e.to_string()))?;
42        let wrap_dek = KeyDerivation::new(mek)
43            .derive_dek(SUBJECT_WRAP_COMPONENT)
44            .map_err(|e| CryptoShredError::Crypto(e.to_string()))?;
45        Ok(create_cipher(cfg.algorithm, &wrap_dek))
46    }

48    /// Designate an erasure subject over one or more targets (whole entities
49    /// and/or specific property keys).
50    ///
51    /// A new subject gets a fresh random DEK, wrapped under the MEK-derived
52    /// subject-wrapping key and recorded `Active`; adding targets to an existing
53    /// active subject merges them. The keyring is persisted durably when index
54    /// persistence is enabled (in-memory only otherwise).
55    ///
56    /// # Errors
57    /// - [`CryptoShredError::EncryptionNotConfigured`] if encryption is not set up.
58    /// - [`CryptoShredError::InvalidArgument`] if the subject id is invalid or
59    ///   `targets` is empty.
60    /// - [`CryptoShredError::SubjectErased`] if the subject was already erased.
61    pub fn designate_subject(
62        &self,
63        subject_id: impl Into<String>,
64        targets: Vec<DesignationTarget>,
65    ) -> Result<(), CryptoShredError> {
66        let subject_id = SubjectId::new(subject_id)?;
67        if targets.is_empty() {
68            return Err(CryptoShredError::InvalidArgument(
69                "designation requires at least one target".to_string(),
70            ));
71        }
72        let wrap_cipher = self.subject_wrap_cipher()?;
73        self.crypto_shred
74            .designate(&subject_id, targets, wrap_cipher.as_ref())
75    }

77    /// Erase a subject: destroy its key material (unrecoverable) and return a
78    /// signed [`ErasureAttestation`]. Re-erase is an idempotent no-op returning
79    /// the recorded attestation.
80    ///
81    /// The erasure-tombstone write transaction is deferred to PR-1b; this slice
82    /// proves key destruction at the cryptographic-core level.
83    ///
84    /// # Errors
85    /// [`CryptoShredError::NotDesignated`] (`FAILED_PRECONDITION`) if the subject
86    /// was never designated (AC6).
87    pub fn erase_subject(
88        &self,
89        subject_id: impl Into<String>,
90    ) -> Result<ErasureAttestation, CryptoShredError> {
91        let subject_id = SubjectId::new(subject_id)?;
92        self.crypto_shred.erase(&subject_id)
93    }

95    /// Unwrap and return a subject's DEK (for the PR-1b seal/unseal path).
96    ///
97    /// # Errors
98    /// - [`CryptoShredError::EncryptionNotConfigured`] if encryption is not set up.
99    /// - [`CryptoShredError::NotDesignated`] if the subject is unknown.
100    /// - [`CryptoShredError::SubjectErased`] if the subject was erased (key gone).
101    pub fn subject_key(
102        &self,
103        subject_id: impl Into<String>,
104    ) -> Result<SubjectKey, CryptoShredError> {
105        let subject_id = SubjectId::new(subject_id)?;
106        let wrap_cipher = self.subject_wrap_cipher()?;
107        self.crypto_shred
108            .subject_key(&subject_id, wrap_cipher.as_ref())
109    }

119    /// Build a prepared [`SealingContext`] for a write transaction, or `None`
120    /// when there is nothing to seal (no active designation) — the cheap common
121    /// case. Fail-closed: when active designations exist but the subject-wrapping
122    /// cipher cannot be built (e.g. encryption misconfigured), returns `Err` so
123    /// the write aborts rather than silently persisting plaintext.
124    pub(crate) fn sealing_context(&self) -> Result<Option<SealingContext>, CryptoShredError> {
125        if !self.crypto_shred.has_active_designations() {
126            return Ok(None);
127        }
128        let cfg = self
129            .encryption_config
130            .as_ref()
131            .ok_or(CryptoShredError::EncryptionNotConfigured)?;
132        let wrap = self
133            .crypto_shred
134            .cached_wrap_cipher(|| self.subject_wrap_cipher())?;
135        Ok(Some(SealingContext::new(
136            Arc::clone(&self.crypto_shred),
137            wrap,
138            cfg.algorithm,
139        )))
140    }

142    /// Materialize sealed property values on read (PR-1b read hook).
143    ///
144    /// Unseal active-subject envelopes to plaintext in place; leave erased ones
145    /// as opaque ciphertext and report them via the returned [`ShredStatusMap`].
146    /// Cheap no-op (returns the map untouched) when the database has never had
147    /// any designation. Building the subject-wrapping cipher (needed only to
148    /// unwrap active DEKs on a cache miss) is memoized; a failure to build it
149    /// leaves values opaque rather than fabricating plaintext.
150    pub(crate) fn materialize_shred(
151        &self,
152        entity_id: u64,
153        props: PropertyMap,
154    ) -> (PropertyMap, ShredStatusMap) {
155        if !self.crypto_shred.has_any_designations() {
156            return (props, ShredStatusMap::new());
157        }
158        let Some(cfg) = self.encryption_config.as_ref() else {
159            return (props, ShredStatusMap::new());
160        };
161        let wrap = match self
162            .crypto_shred
163            .cached_wrap_cipher(|| self.subject_wrap_cipher())
164        {
165            Ok(w) => w,
166            Err(_) => return (props, ShredStatusMap::new()),
167        };
168        self.crypto_shred
169            .unseal_map(entity_id, props, wrap.as_ref(), cfg.algorithm)
170    }

172    /// Unseal a node's designated properties for a read boundary (PR-1b).
173    #[must_use]
174    pub(crate) fn unseal_node_view(&self, mut node: Node) -> Node {
175        if !self.crypto_shred.has_any_designations() {
176            return node;
177        }
178        let (props, _status) = self.materialize_shred(node.id.as_u64(), node.properties);
179        node.properties = props;
180        node
181    }

194    /// The crypto-shred status of a node's current-state properties (PR-1b).
195    ///
196    /// Returns a typed [`ShredStatusMap`] keyed by property key: an entry is
197    /// present only for a property whose subject key has been **destroyed**
198    /// ([`super::ShredStatus::Erased`]); an absent key is
199    /// [`super::ShredStatus::Plaintext`] (never sealed, or sealed under a still
200    /// active subject and thus readable). The map is empty for a node with no
201    /// erased designated property, and empty (allocation-free) when the database
202    /// has never used crypto-shred. This is the **public Rust** surface for the
203    /// erased marker — the complement to [`AletheiaDB::get_node`], which returns
204    /// the (possibly opaque) values but discards the status side-channel.
205    ///
206    /// # Errors
207    /// Propagates the underlying node read error (e.g. the node does not exist).
208    pub fn node_shred_status(
209        &self,
210        node_id: crate::core::NodeId,
211    ) -> crate::core::error::Result<ShredStatusMap> {
212        // Read the RAW stored node (sealed bytes intact) directly from the
213        // current tier, bypassing the `get_node` unseal hook, so the status
214        // reflects the stored envelopes rather than already-unsealed plaintext.
215        let node = self.current.get_node(node_id)?;
216        let (_props, status) = self.materialize_shred(node.id.as_u64(), node.properties);
217        Ok(status)
218    }

220    /// The crypto-shred status of an edge's current-state properties (PR-1b).
221    ///
222    /// See [`AletheiaDB::node_shred_status`] — the edge analog.
223    ///
224    /// # Errors
225    /// Propagates the underlying edge read error (e.g. the edge does not exist).
226    pub fn edge_shred_status(
227        &self,
228        edge_id: crate::core::EdgeId,
229    ) -> crate::core::error::Result<ShredStatusMap> {
230        let edge = self.current.get_edge(edge_id)?;
231        let (_props, status) = self.materialize_shred(edge.id.as_u64(), edge.properties);
232        Ok(status)
233    }
