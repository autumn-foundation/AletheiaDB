1//! Signed erasure attestation (Issue #3359, slice PR-1a).
2//!
3//! An [`ErasureAttestation`] is the auditor-facing proof that a subject was
4//! erased. It carries **only** the subject id, affected entity count, and a
5//! timestamp — **never any property content** — canonicalized deterministically,
6//! SHA-256'd, and signed with the audit Ed25519 key
7//! ([`crate::audit::AuditSigningKey::sign_root`]). The signer's public key is
8//! embedded so a third party can verify the signature independently.
9
10use bitcode::{Decode, Encode};
11use sha2::{Digest, Sha256};
12
13use crate::audit::{AuditPublicKey, AuditSigningKey};
14
15use super::error::CryptoShredError;
16
17/// Domain-separation tag for the attestation canonical root.
18const ATTESTATION_DOMAIN: &[u8] = b"aletheiadb-erasure-attestation-v1";
19
20/// Deterministically canonicalize the attestation fields and SHA-256 them into a
21/// 32-byte signing root.
22///
23/// The variable-length `subject_id` is length-prefixed (injective encoding) so
24/// distinct field tuples never collide.
25fn canonical_root(subject_id: &str, entity_count: u32, timestamp_micros: i64) -> [u8; 32] {
26    let subj = subject_id.as_bytes();
27    let mut h = Sha256::new();
28    h.update(ATTESTATION_DOMAIN);
29    h.update((subj.len() as u64).to_le_bytes());
30    h.update(subj);
31    h.update(entity_count.to_le_bytes());
32    h.update(timestamp_micros.to_le_bytes());
33    h.finalize().into()
34}
35
36/// A signed proof that a subject's key was destroyed.
37#[derive(Debug, Clone)]
38pub struct ErasureAttestation {
39    /// The erased subject's identifier.
40    pub subject_id: String,
41    /// Number of designation targets (entities) covered by the subject.
42    pub entity_count: u32,
43    /// Erasure timestamp, microseconds since epoch.
44    pub timestamp_micros: i64,
45    /// Raw 64-byte Ed25519 signature over the canonical root.
46    pub signature: [u8; 64],
47    /// The signer's public key (for independent verification).
48    pub signer_public_key: AuditPublicKey,
49}
50
51impl ErasureAttestation {
52    /// Build and sign a fresh attestation.
53    pub(crate) fn sign(
54        signing_key: &AuditSigningKey,
55        subject_id: &str,
56        entity_count: u32,
57        timestamp_micros: i64,
58    ) -> Self {
59        let root = canonical_root(subject_id, entity_count, timestamp_micros);
60        let signature = signing_key.sign_root(&root);
61        Self {
62            subject_id: subject_id.to_string(),
63            entity_count,
64            timestamp_micros,
65            signature,
66            signer_public_key: signing_key.public_key(),
67        }
68    }
69
70    /// Verify the embedded signature against the embedded public key.
71    ///
72    /// Returns `true` iff the signature is valid for this attestation's
73    /// canonical root. (An auditor additionally compares `signer_public_key`
74    /// against a known-good key out of band.)
75    #[must_use]
76    pub fn verify(&self) -> bool {
77        let root = canonical_root(&self.subject_id, self.entity_count, self.timestamp_micros);
78        self.signer_public_key
79            .verify_root(&root, &self.signature)
80            .is_ok()
81    }
82
83    /// Convert to the durable (bitcode) record form.
84    #[must_use]
85    pub fn to_record(&self) -> AttestationRecord {
86        AttestationRecord {
87            subject_id: self.subject_id.clone(),
88            entity_count: self.entity_count,
89            timestamp_micros: self.timestamp_micros,
90            signature: self.signature.to_vec(),
91            signer_public_key_hex: self.signer_public_key.to_hex(),
92        }
93    }
94
95    /// Reconstruct from a durable record.
96    ///
97    /// # Errors
98    /// [`CryptoShredError::KeyringCorrupt`] if the stored signature or public key
99    /// is malformed (a corrupt keyring file).
100    pub fn from_record(record: &AttestationRecord) -> Result<Self, CryptoShredError> {
101        let signature: [u8; 64] = record.signature.as_slice().try_into().map_err(|_| {
102            CryptoShredError::KeyringCorrupt("attestation signature is not 64 bytes".to_string())
103        })?;
104        let signer_public_key =
105            AuditPublicKey::from_hex(&record.signer_public_key_hex).map_err(|_| {
106                CryptoShredError::KeyringCorrupt(
107                    "attestation signer public key is malformed".to_string(),
108                )
109            })?;
110        Ok(Self {
111            subject_id: record.subject_id.clone(),
112            entity_count: record.entity_count,
113            timestamp_micros: record.timestamp_micros,
114            signature,
115            signer_public_key,
116        })
117    }
118}

120/// Durable (bitcode) form of an [`ErasureAttestation`], stored in the keyring so
121/// a re-erase returns the original attestation across restarts.
122#[derive(Debug, Clone, PartialEq, Eq, Encode, Decode)]
123pub struct AttestationRecord {
124    /// The erased subject's identifier.
125    pub subject_id: String,
126    /// Number of designation targets covered.
127    pub entity_count: u32,
128    /// Erasure timestamp, microseconds since epoch.
129    pub timestamp_micros: i64,
130    /// Raw 64-byte Ed25519 signature.
131    pub signature: Vec<u8>,
132    /// Lowercase-hex of the signer's 32-byte public key.
133    pub signer_public_key_hex: String,
134}
