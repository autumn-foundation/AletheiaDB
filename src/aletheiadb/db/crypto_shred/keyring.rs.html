1//! Durable per-subject key registry (Issue #3359, slice PR-1a).
2//!
3//! The keyring maps `subject_id -> SubjectEntry`, holding each subject's DEK in
4//! **wrapped** (MEK-derived-cipher-encrypted) form. Erasing a subject sets
5//! `wrapped_key = None` and rewrites the file so the wrapped blob is physically
6//! removed from the live file — the random DEK is then unrecoverable.
7//!
8//! ## Durability & fail-closed load
9//!
10//! The registry persists as a bitcode+CRC [`SubjectKeyringSidecar`] written
11//! through [`crate::db::rotation::write_durable`] (temp -> `sync_all` -> rename
12//! -> parent-dir fsync). Load is **fail-closed**: a present-but-corrupt file
13//! returns [`CryptoShredError::KeyringCorrupt`] (startup aborts) rather than
14//! silently starting empty — the opposite of the tolerant snapshot-quarantine
15//! policy, so an erased subject can never be silently re-exposed.
16

48/// Lifecycle state of a subject.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Encode, Decode)]
50pub enum SubjectState {
51    /// Designated; key present and usable.
52    Active,
53    /// Erased; key destroyed, payload permanently unrecoverable.
54    Erased,
55}
56
57/// A subject's DEK in wrapped (encrypted) form.
58///
59/// Only ever the ciphertext of the DEK is persisted — never the raw key.
60#[derive(Clone, PartialEq, Eq, Encode, Decode)]
61pub struct WrappedKey {
62    /// Version of the subject-wrapping key used to wrap this DEK.
63    pub key_version: u32,
64    /// AEAD wrap blob `[nonce || ciphertext || tag]` of the 32-byte DEK.
65    pub wrapped: Vec<u8>,
66}
67
68impl std::fmt::Debug for WrappedKey {
69    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
70        // Defense-in-depth: never render the (key-adjacent) wrapped blob bytes,
71        // only its length + the key version.
72        f.debug_struct("WrappedKey")
73            .field("key_version", &self.key_version)
74            .field("wrapped_len", &self.wrapped.len())
75            .finish()
76    }
77}

79/// A durable registry entry for one subject.
80#[derive(Debug, Clone, PartialEq, Eq, Encode, Decode)]
81pub struct SubjectEntry {
82    /// The subject identifier.
83    pub subject_id: String,
84    /// The wrapped DEK — `None` once erased (physically removed).
85    pub wrapped_key: Option<WrappedKey>,
86    /// The designation targets (entities / property keys) under this subject.
87    pub designation: Vec<DesignationTarget>,
88    /// Lifecycle state.
89    pub state: SubjectState,
90    /// Microseconds-since-epoch when the subject was first designated.
91    pub created_at_micros: i64,
92    /// Microseconds-since-epoch when the subject was erased (`None` if active).
93    pub erased_at_micros: Option<i64>,
94    /// The signed erasure attestation, recorded at erase time so a re-erase is
95    /// an idempotent no-op returning the original attestation.
96    pub attestation: Option<AttestationRecord>,
97}

99/// On-disk keyring wrapper (bitcode + CRC).
100#[derive(Debug, Clone, PartialEq, Eq, Encode, Decode)]
101pub struct SubjectKeyringSidecar {
102    /// Format version for forward compatibility.
103    pub version: u16,
104    /// All subject entries.
105    pub entries: Vec<SubjectEntry>,
106}

108/// In-memory per-subject key registry.
109///
110/// Holds the durable entries plus a small cache of unwrapped DEKs (cleared and
111/// zeroized on erase). The cache is a pure optimization — every unwrapped key is
112/// reconstructible from `entries` while the subject is `Active`.
113#[derive(Default)]
114pub struct SubjectKeyring {
115    entries: BTreeMap<String, SubjectEntry>,
116    /// Unwrapped-DEK cache; entries dropped (and thus zeroized) on erase.
117    cache: HashMap<String, SubjectKey>,
118    /// Reverse designation index: `(is_node, entity_id) -> subject ids` (PR-1b).
119    ///
120    /// Lets the write/read hooks answer "which subject(s) seal this entity?"
121    /// in O(1) instead of scanning every subject's designation set. In-memory
122    /// only (rebuilt from `entries` on load). Erased subjects are **retained**
123    /// in this index so the read hook can still report an erased marker for a
124    /// value whose key was destroyed.
125    designation_index: HashMap<(bool, u64), Vec<String>>,
126}
127
128impl std::fmt::Debug for SubjectKeyring {
129    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
130        // Never expose cached key material; report only counts + ids.
131        f.debug_struct("SubjectKeyring")
132            .field("subjects", &self.entries.len())
133            .field("cached_keys", &self.cache.len())
134            .finish()
135    }
136}

138impl SubjectKeyring {
139    /// An empty in-memory keyring.
140    #[must_use]
141    pub fn new() -> Self {
142        Self::default()
143    }
144
145    /// Number of registered subjects.
146    #[must_use]
147    pub fn len(&self) -> usize {
148        self.entries.len()
149    }
150
151    /// Whether the keyring has no subjects.
152    #[must_use]
153    pub fn is_empty(&self) -> bool {
154        self.entries.is_empty()
155    }
156
157    /// Borrow a subject entry, if present.
158    #[must_use]
159    pub fn get(&self, subject_id: &str) -> Option<&SubjectEntry> {
160        self.entries.get(subject_id)
161    }
162
163    /// Insert or replace a subject entry.
164    pub fn upsert(&mut self, entry: SubjectEntry) {
165        self.index_entry(&entry);
166        self.entries.insert(entry.subject_id.clone(), entry);
167    }
168
169    /// Fold one entry's designation targets into the reverse index (idempotent).
170    ///
171    /// Designations only ever grow in v1 (merge adds targets; erase retains
172    /// them), so a plain additive re-index with per-key dedup is sufficient and
173    /// never leaves a stale mapping.
174    fn index_entry(&mut self, entry: &SubjectEntry) {
175        for target in &entry.designation {
176            let key = (target.is_node(), target.entity_id());
177            let bucket = self.designation_index.entry(key).or_default();
178            if !bucket.iter().any(|s| s == &entry.subject_id) {
179                bucket.push(entry.subject_id.clone());
180            }
181        }
182    }
183
184    /// The subject ids that designate `(is_node, entity_id)`, in insertion order.
185    #[must_use]
186    pub fn subjects_for_entity(&self, is_node: bool, entity_id: u64) -> &[String] {
187        self.designation_index
188            .get(&(is_node, entity_id))
189            .map_or(&[], Vec::as_slice)
190    }
191
192    /// Iterate all subject entries (read-only), in subject-id order.
193    ///
194    /// Used by the subject-keyring re-wrap pass (Slice 5) to plan which entries
195    /// hold a stale-generation wrapped DEK; never exposes key material (the
196    /// wrapped blob is AEAD ciphertext, and `SubjectEntry`'s `Debug` is redacted).
197    pub fn iter_entries(&self) -> impl Iterator<Item = &SubjectEntry> {
198        self.entries.values()
199    }
200
201    /// Replace an existing **active** subject's wrapped DEK blob (subject-keyring
202    /// re-wrap on full-MEK rotation, Slice 5).
203    ///
204    /// A no-op when the subject is absent. Fail-closed on the erased invariant:
205    /// an `Erased` subject (or one whose `wrapped_key` is already `None`) is
206    /// **never** given a wrapped key back — a re-wrap must never resurrect a
207    /// destroyed key. The caller (`CryptoShredState::rewrap_keyring`) already
208    /// skips such entries; this guard is defense in depth.
209    pub fn set_wrapped_key(&mut self, subject_id: &str, wrapped: WrappedKey) {
210        if let Some(entry) = self
211            .entries
212            .get_mut(subject_id)
213            .filter(|e| e.state == SubjectState::Active && e.wrapped_key.is_some())
214        {
215            entry.wrapped_key = Some(wrapped);
216        }
217    }
218
219    /// Whether any subject is currently `Active`.
220    #[must_use]
221    pub fn any_active(&self) -> bool {
222        self.entries
223            .values()
224            .any(|e| e.state == SubjectState::Active)
225    }
226
227    /// Cache an unwrapped DEK for a subject.
228    pub fn cache_key(&mut self, subject_id: &str, key: SubjectKey) {
229        self.cache.insert(subject_id.to_string(), key);
230    }
231
232    /// Fetch a cached unwrapped DEK, if present.
233    #[must_use]
234    pub fn cached_key(&self, subject_id: &str) -> Option<&SubjectKey> {
235        self.cache.get(subject_id)
236    }
237
238    /// Erase a subject in memory: drop (zeroize) any cached DEK, remove the
239    /// wrapped blob, and flip the state to `Erased`, recording the attestation.
240    ///
241    /// The dropped `SubjectKey` zeroizes its bytes on drop.
242    pub fn erase_in_memory(
243        &mut self,
244        subject_id: &str,
245        erased_at_micros: i64,
246        attestation: AttestationRecord,
247    ) {
248        // Dropping the cached key zeroizes it (SubjectKey wraps Zeroizing).
249        self.cache.remove(subject_id);
250        if let Some(entry) = self.entries.get_mut(subject_id) {
251            entry.wrapped_key = None;
252            entry.state = SubjectState::Erased;
253            entry.erased_at_micros = Some(erased_at_micros);
254            entry.attestation = Some(attestation);
255        }
256    }
257
258    /// Force a subject to the erased state during crash recovery, creating a
259    /// tombstone entry if the subject is not present. Fail-closed: a breadcrumbed
260    /// subject is treated as erased no matter the on-disk keyring state.
261    pub fn force_erased(&mut self, subject_id: &str, erased_at_micros: i64) {
262        self.cache.remove(subject_id);
263        match self.entries.get_mut(subject_id) {
264            Some(entry) => {
265                entry.wrapped_key = None;
266                entry.state = SubjectState::Erased;
267                if entry.erased_at_micros.is_none() {
268                    entry.erased_at_micros = Some(erased_at_micros);
269                }
270            }
271            None => {
272                // Recovered-orphan tombstone: the breadcrumb named a subject with
273                // no keyring entry (crashed before the entry was ever persisted, or
274                // designation itself was interrupted). We fail-closed by minting an
275                // empty tombstone. It necessarily carries `designation: Vec::new()`,
276                // so any later attestation over it reports `entity_count == 0` — the
277                // recorded designation facts were lost with the un-persisted entry.
278                self.entries.insert(
279                    subject_id.to_string(),
280                    SubjectEntry {
281                        subject_id: subject_id.to_string(),
282                        wrapped_key: None,
283                        designation: Vec::new(),
284                        state: SubjectState::Erased,
285                        created_at_micros: erased_at_micros,
286                        erased_at_micros: Some(erased_at_micros),
287                        attestation: None,
288                    },
289                );
290            }
291        }
292    }
293
294    /// Build the durable sidecar snapshot from the current entries.
295    #[must_use]
296    pub fn to_sidecar(&self) -> SubjectKeyringSidecar {
297        SubjectKeyringSidecar {
298            version: KEYRING_SIDECAR_VERSION,
299            entries: self.entries.values().cloned().collect(),
300        }
301    }
302
303    /// Replace all entries from a loaded sidecar (cache is left empty), and
304    /// rebuild the in-memory reverse designation index from scratch.
305    fn load_from_sidecar(&mut self, sidecar: SubjectKeyringSidecar) {
306        self.entries.clear();
307        self.designation_index.clear();
308        for entry in sidecar.entries {
309            self.index_entry(&entry);
310            self.entries.insert(entry.subject_id.clone(), entry);
311        }
312    }
313}

315/// Encode a sidecar to `[bitcode || crc32 LE]` — the exact wire shape
316/// [`crate::storage::index_persistence::common::load_encoded_with_crc`] reads,
317/// so encoding here + writing through `write_durable` round-trips through the
318/// standard loader.
319pub(crate) fn encode_sidecar_with_crc(sidecar: &SubjectKeyringSidecar) -> Vec<u8> {
320    let encoded = bitcode::encode(sidecar);
321    let mut hasher = crc32fast::Hasher::new();
322    hasher.update(&encoded);
323    let checksum = hasher.finalize();
324    let mut out = encoded;
325    out.extend_from_slice(&checksum.to_le_bytes());
326    out
327}

329/// Persist a keyring durably to `path` (temp -> `sync_all` -> rename ->
330/// parent-dir fsync via `write_durable`).
331///
332/// # Errors
333/// [`CryptoShredError::Io`] on any durable-write failure.
334pub fn save_keyring(path: &Path, keyring: &SubjectKeyring) -> Result<(), CryptoShredError> {
335    let bytes = encode_sidecar_with_crc(&keyring.to_sidecar());
336    crate::db::rotation::write_durable(path, &bytes)
337        .map_err(|e| CryptoShredError::Io(format!("keyring write failed: {e}")))
338}

340/// Write already-CRC-wrapped keyring sidecar `bytes` verbatim to `path`
341/// durably (temp -> `sync_all` -> rename -> parent-dir fsync via
342/// `write_durable`).
343///
344/// Used by the `.albk` restore path (Issue #3665): the payload's
345/// `keyring_sidecar` is the exact [`encode_sidecar_with_crc`] wire form, so it
346/// is written byte-for-byte to `{data_dir}/subject_keyring.dat` and then loaded
347/// by the fail-closed [`load_keyring`] on the subsequent reopen. The caller
348/// no-ops on empty `bytes` (nothing to restore).
349///
350/// # Errors
351/// [`CryptoShredError::Io`] on any durable-write failure.
352pub(crate) fn save_sidecar_bytes(path: &Path, bytes: &[u8]) -> Result<(), CryptoShredError> {
353    crate::db::rotation::write_durable(path, bytes)
354        .map_err(|e| CryptoShredError::Io(format!("keyring sidecar write failed: {e}")))
355}

357/// Load a keyring from `path`, **fail-closed**.
358///
359/// - Missing file -> empty keyring (`Ok`).
360/// - Present + well-formed -> loaded (`Ok`).
361/// - Present + corrupt / unknown version -> [`CryptoShredError::KeyringCorrupt`]
362///   (`Err`) so startup aborts rather than silently re-exposing erased subjects.
363pub fn load_keyring(path: &Path) -> Result<SubjectKeyring, CryptoShredError> {
364    if !path.exists() {
365        return Ok(SubjectKeyring::new());
366    }
367    let sidecar = crate::storage::index_persistence::common::load_encoded_with_crc::<
368        SubjectKeyringSidecar,
369    >(path, MAX_KEYRING_BYTES, "subject keyring")
370    .map_err(|e| CryptoShredError::KeyringCorrupt(e.to_string()))?;
371
372    if sidecar.version != KEYRING_SIDECAR_VERSION {
373        return Err(CryptoShredError::KeyringCorrupt(format!(
374            "unsupported keyring version {} (supported {})",
375            sidecar.version, KEYRING_SIDECAR_VERSION
376        )));
377    }
378
379    let mut keyring = SubjectKeyring::new();
380    keyring.load_from_sidecar(sidecar);
381    Ok(keyring)
382}

384/// Write the in-flight-erase breadcrumb naming `subject_id`, durably.
385///
386/// # Errors
387/// [`CryptoShredError::Io`] on durable-write failure.
388pub fn write_breadcrumb(path: &Path, subject_id: &str) -> Result<(), CryptoShredError> {
389    crate::db::rotation::write_durable(path, subject_id.as_bytes())
390        .map_err(|e| CryptoShredError::Io(format!("breadcrumb write failed: {e}")))
391}

393/// Read the breadcrumb's subject id, if the breadcrumb exists.
394///
395/// # Errors
396/// [`CryptoShredError::Io`] if the breadcrumb is present but unreadable / not
397/// valid utf-8 (fail-closed — do not silently ignore a present breadcrumb).
398pub fn read_breadcrumb(path: &Path) -> Result<Option<String>, CryptoShredError> {
399    match std::fs::read(path) {
400        Ok(bytes) => {
401            let s = String::from_utf8(bytes)
402                .map_err(|_| CryptoShredError::Io("breadcrumb is not valid utf-8".to_string()))?;
403            Ok(Some(s))
404        }
405        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
406        Err(e) => Err(CryptoShredError::Io(format!("breadcrumb unreadable: {e}"))),
407    }
408}

410/// Remove the breadcrumb and fsync its parent directory so the removal is
411/// durable (a crash right after must not resurrect the breadcrumb).
412///
413/// Return-`Ok` behavior is preserved (this never surfaces an error), but a
414/// removal failure for a reason **other** than "already gone" is logged so a
415/// stuck breadcrumb — which would force the subject to be re-erased on the next
416/// restart — is observable rather than silent.
417pub fn clear_breadcrumb(path: &Path) {
418    match std::fs::remove_file(path) {
419        Ok(()) => {
420            if let Some(parent) = path.parent() {
421                crate::storage::index_persistence::fsync_dir(parent);
422            }
423        }
424        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {}
425        Err(e) => {
426            // Do not include the subject id here (it lives in the breadcrumb),
427            // only the path + error, so nothing sensitive is logged. Uses the
428            // crate's `observability`-gated logging style (mirrors
429            // `log_authority_warning` in `src/db/encryption_state.rs`).
430            let message = format!(
431                "crypto-shred: failed to clear erase breadcrumb at {}: {e}",
432                path.display()
433            );
434            #[cfg(feature = "observability")]
435            tracing::warn!("{}", message);
436            #[cfg(not(feature = "observability"))]
437            eprintln!("WARNING: {}", message);
438        }
439    }
