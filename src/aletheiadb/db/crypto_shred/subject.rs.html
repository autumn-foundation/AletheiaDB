1//! Subject identifiers and per-subject key material (Issue #3359, slice PR-1a).
2//!
3//! A [`SubjectId`] is a caller-chosen **identifier** grouping the entities /
4//! property keys that share one erasure fate. It is not a secret — its `Display`
5//! and `Debug` are allowed to print the id.
6//!
7//! A [`SubjectKey`] is the random 32-byte per-subject data-encryption key (DEK).
8//! It **is** a secret: the bytes live in a [`Zeroizing`] buffer, its `Debug`
9//! redacts, and it exposes no `Serialize`/`Display` of the raw bytes.
10
11use zeroize::Zeroizing;
12
13use super::error::CryptoShredError;
14
15/// Maximum accepted length (in bytes) of a subject identifier.
16///
17/// Bounded to keep the durable keyring compact and to reject accidental
18/// large-blob ids; 256 bytes is far more than any real subject id needs.
19pub const MAX_SUBJECT_ID_LEN: usize = 256;
20
21/// A validated erasure-subject identifier.
22///
23/// Construction validates the id: non-empty, `<= MAX_SUBJECT_ID_LEN` bytes, and
24/// free of control characters (so it never corrupts logs / breadcrumb files).
25/// The identifier is **not** secret — it appears in attestations, breadcrumbs,
26/// and error `details` so callers can act on it.
27#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
28pub struct SubjectId(String);
29
30impl SubjectId {
31    /// Validate and wrap a subject identifier.
32    ///
33    /// # Errors
34    /// [`CryptoShredError::InvalidArgument`] if the id is empty, too long, or
35    /// contains control characters.
36    pub fn new(id: impl Into<String>) -> Result<Self, CryptoShredError> {
37        let id = id.into();
38        if id.is_empty() {
39            return Err(CryptoShredError::InvalidArgument(
40                "subject id must not be empty".to_string(),
41            ));
42        }
43        if id.len() > MAX_SUBJECT_ID_LEN {
44            return Err(CryptoShredError::InvalidArgument(format!(
45                "subject id must be <= {MAX_SUBJECT_ID_LEN} bytes"
46            )));
47        }
48        if id.chars().any(|c| c.is_control()) {
49            return Err(CryptoShredError::InvalidArgument(
50                "subject id must not contain control characters".to_string(),
51            ));
52        }
53        Ok(Self(id))
54    }
55
56    /// The identifier as a string slice.
57    #[must_use]
58    pub fn as_str(&self) -> &str {
59        &self.0
60    }
61
62    /// The identifier's bytes (used as AEAD AAD when wrapping the subject DEK).
63    #[must_use]
64    pub fn as_bytes(&self) -> &[u8] {
65        self.0.as_bytes()
66    }
67
68    /// Consume into the inner `String`.
69    #[must_use]
70    pub fn into_inner(self) -> String {
71        self.0
72    }
73}
74
75impl std::fmt::Display for SubjectId {
76    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
77        f.write_str(&self.0)
78    }
79}

81/// Length of a subject data-encryption key in bytes.
82pub const SUBJECT_KEY_LEN: usize = 32;
83
84/// A random per-subject data-encryption key (DEK).
85///
86/// # Security
87/// The 32 raw key bytes are held in a [`Zeroizing`] buffer (wiped on drop). The
88/// [`Debug`] impl **redacts** — it never prints the bytes or a hex encoding —
89/// and the type intentionally implements neither `Serialize` nor `Display`, so
90/// no code path can accidentally emit the key into a log, error, or file. Only
91/// the wrapped (encrypted) form is ever persisted (see
92/// [`super::keyring::WrappedKey`]).
93#[derive(Clone)]
94pub struct SubjectKey(Zeroizing<[u8; SUBJECT_KEY_LEN]>);
95
96impl SubjectKey {
97    /// Generate a fresh random subject key from the OS CSPRNG.
98    ///
99    /// This is a **random independent** key, not derived from the MEK — that is
100    /// the crux of crypto-shred: destroying it is irreversible even for a holder
101    /// of the MEK (see the design doc's rejection of `HKDF(MEK, subject_id)`).
102    #[must_use]
103    pub fn generate() -> Self {
104        use rand::RngCore;
105        let mut bytes = Zeroizing::new([0u8; SUBJECT_KEY_LEN]);
106        rand::rngs::OsRng.fill_bytes(bytes.as_mut());
107        Self(bytes)
108    }
109
110    /// Wrap already-zeroizing key bytes (used after unwrapping from the keyring).
111    ///
112    /// Takes a [`Zeroizing`] buffer (not a bare array) so every caller is forced
113    /// onto the wiped path — the transit copy of the unwrapped DEK is guaranteed
114    /// to be zeroized on drop, never left in a plain stack array.
115    #[must_use]
116    pub fn from_bytes(bytes: Zeroizing<[u8; SUBJECT_KEY_LEN]>) -> Self {
117        Self(bytes)
118    }
119
120    /// Borrow the raw key bytes (crate-internal; for wrapping/cipher build only).
121    #[must_use]
122    pub(crate) fn expose_bytes(&self) -> &[u8; SUBJECT_KEY_LEN] {
123        &self.0
124    }
125}
126
127impl std::fmt::Debug for SubjectKey {
128    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
129        // NEVER print key material.
130        f.write_str("SubjectKey(<redacted>)")
131    }
