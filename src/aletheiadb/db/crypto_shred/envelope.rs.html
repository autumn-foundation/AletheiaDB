1//! Self-describing sealed-envelope codec (Issue #3359, slice PR-1a).
2//!
3//! Wire format (modeled on the cold-tier `ACV1` loud-fail wrapper):
4//!
5//! ```text
6//! SUBJ (4) | fmt_ver:u8 | alg_id:u8 | key_version:u32 LE | subj_len:u16 LE | subject_id | AEAD_blob
7//! ```
8//!
9//! The `AEAD_blob` is `[nonce || ciphertext || tag]` produced by the subject
10//! cipher. The AEAD **AAD binds the subject id + key version + an opaque
11//! write-site context**, so an envelope only decrypts under its own subject's
12//! key at its own key version *and* at the same write-site context — a
13//! cross-entity swap (moving an envelope onto another subject, or relocating it
14//! to a different entity/property within the same subject) is a loud AEAD auth
15//! failure, never fabricated plaintext.
16//!
17//! ## Write-site context
18//!
19//! `seal`/`unseal` take an opaque `context: &[u8]` folded into the AAD. PR-1a
20//! call sites pass an empty context (`b""`); PR-1b will pass the write-site
21//! binding (e.g. `entity_id || property_key`) so ciphertext cannot be relocated
22//! within a subject. The parameter exists now, while the API is private, so
23//! PR-1b binds context without an API break.
24

32/// Magic prefix identifying a sealed subject envelope.
33pub const ENVELOPE_MAGIC: &[u8; 4] = b"SUBJ";
34
35/// Current envelope format version.
36pub const ENVELOPE_FORMAT_VERSION: u8 = 1;
37
38/// Fixed-size header length preceding the variable subject id + AEAD blob:
39/// magic(4) + fmt_ver(1) + alg_id(1) + key_version(4) + subj_len(2) = 12.
40pub const ENVELOPE_HEADER_LEN: usize = 4 + 1 + 1 + 4 + 2;
41
42/// AAD domain-separation prefix (bound alongside key version + subject id).
43const AAD_DOMAIN: &[u8] = b"aletheiadb-subject-envelope-v1";
44
45/// Build the AEAD AAD binding this envelope to its subject id + key version +
46/// opaque write-site context.
47///
48/// Every variable-length field is length-prefixed so distinct
49/// `(subject_id, context)` splits can never collide (injective encoding).
50fn envelope_aad(subject_id: &[u8], key_version: u32, context: &[u8]) -> Vec<u8> {
51    let mut aad =
52        Vec::with_capacity(AAD_DOMAIN.len() + 4 + 8 + subject_id.len() + 8 + context.len());
53    aad.extend_from_slice(AAD_DOMAIN);
54    aad.extend_from_slice(&key_version.to_le_bytes());
55    aad.extend_from_slice(&(subject_id.len() as u64).to_le_bytes());
56    aad.extend_from_slice(subject_id);
57    aad.extend_from_slice(&(context.len() as u64).to_le_bytes());
58    aad.extend_from_slice(context);
59    aad
60}

62/// Seal `plaintext` into a self-describing envelope under `cipher`.
63///
64/// `subject_id` and `key_version` are stamped into the header AND bound as AEAD
65/// AAD, together with the opaque `context` (a write-site binding — pass `b""`
66/// when there is none). `cipher` must be the subject's own cipher (built from
67/// the unwrapped subject DEK).
68///
69/// # Errors
70/// [`CryptoShredError::InvalidArgument`] if `subject_id` is longer than `u16`
71/// can express, or [`CryptoShredError::Crypto`] on AEAD failure.
72pub fn seal(
73    plaintext: &[u8],
74    subject_id: &str,
75    key_version: u32,
76    context: &[u8],
77    cipher: &dyn Cipher,
78) -> Result<Vec<u8>, CryptoShredError> {
79    let subj = subject_id.as_bytes();
80    let subj_len: u16 = subj.len().try_into().map_err(|_| {
81        CryptoShredError::InvalidArgument("subject id too long to seal".to_string())
82    })?;
83
84    let aad = envelope_aad(subj, key_version, context);
85    let blob = cipher
86        .encrypt(plaintext, &aad)
87        .map_err(|e| CryptoShredError::Crypto(e.to_string()))?;
88
89    let mut out = Vec::with_capacity(ENVELOPE_HEADER_LEN + subj.len() + blob.len());
90    out.extend_from_slice(ENVELOPE_MAGIC);
91    out.push(ENVELOPE_FORMAT_VERSION);
92    out.push(cipher.algorithm_id());
93    out.extend_from_slice(&key_version.to_le_bytes());
94    out.extend_from_slice(&subj_len.to_le_bytes());
95    out.extend_from_slice(subj);
96    out.extend_from_slice(&blob);
97    Ok(out)
98}

100/// Parsed header of a sealed envelope (excludes the AEAD blob).
101#[derive(Debug, Clone, PartialEq, Eq)]
102pub struct EnvelopeHeader {
103    /// Format version.
104    pub format_version: u8,
105    /// AEAD algorithm id the envelope was sealed with.
106    pub algorithm_id: u8,
107    /// Key version of the subject key that sealed this envelope.
108    pub key_version: u32,
109    /// The subject id this envelope belongs to.
110    pub subject_id: String,
111}
112
113/// `true` if `bytes` begins with the sealed-envelope magic.
114#[must_use]
115pub fn is_envelope(bytes: &[u8]) -> bool {
116    bytes.len() >= 4 && &bytes[0..4] == ENVELOPE_MAGIC
117}
118
119/// Parse the header and return it plus the trailing AEAD blob slice.
120fn parse(envelope: &[u8]) -> Result<(EnvelopeHeader, &[u8]), CryptoShredError> {
121    if envelope.len() < ENVELOPE_HEADER_LEN {
122        return Err(CryptoShredError::Crypto(
123            "sealed envelope truncated".to_string(),
124        ));
125    }
126    if &envelope[0..4] != ENVELOPE_MAGIC {
127        return Err(CryptoShredError::Crypto(
128            "not a sealed envelope".to_string(),
129        ));
130    }
131    let format_version = envelope[4];
132    if format_version != ENVELOPE_FORMAT_VERSION {
133        return Err(CryptoShredError::Crypto(format!(
134            "unsupported sealed-envelope format version {format_version}"
135        )));
136    }
137    let algorithm_id = envelope[5];
138    let key_version = u32::from_le_bytes([envelope[6], envelope[7], envelope[8], envelope[9]]);
139    let subj_len = u16::from_le_bytes([envelope[10], envelope[11]]) as usize;
140
141    let subj_start = ENVELOPE_HEADER_LEN;
142    let subj_end = subj_start
143        .checked_add(subj_len)
144        .ok_or_else(|| CryptoShredError::Crypto("sealed envelope length overflow".to_string()))?;
145    if envelope.len() < subj_end {
146        return Err(CryptoShredError::Crypto(
147            "sealed envelope truncated".to_string(),
148        ));
149    }
150    let subject_id = std::str::from_utf8(&envelope[subj_start..subj_end])
151        .map_err(|_| CryptoShredError::Crypto("sealed envelope subject id not utf-8".to_string()))?
152        .to_string();
153    let blob = &envelope[subj_end..];
154    Ok((
155        EnvelopeHeader {
156            format_version,
157            algorithm_id,
158            key_version,
159            subject_id,
160        },
161        blob,
162    ))
163}

165/// Parse a sealed envelope's header without decrypting (read-path routing).
166///
167/// # Errors
168/// [`CryptoShredError::Crypto`] if the bytes are not a well-formed envelope.
169pub fn parse_header(envelope: &[u8]) -> Result<EnvelopeHeader, CryptoShredError> {
170    parse(envelope).map(|(h, _)| h)
171}

173/// Unseal an envelope produced by [`seal`], returning the original plaintext.
174///
175/// The AAD (subject id + key version + `context`) is reconstructed from the
176/// parsed header and the caller-supplied `context`, so a mismatch — a tampered
177/// header, a swapped subject, the wrong key, or the wrong write-site context —
178/// fails the AEAD auth check loudly rather than fabricating a value. `context`
179/// must match the value passed to [`seal`] (`b""` when there was none).
180///
181/// # Errors
182/// [`CryptoShredError::Crypto`] if the envelope is malformed or AEAD auth fails
183/// (including "wrong / destroyed key" and "wrong context").
184pub fn unseal(
185    envelope: &[u8],
186    context: &[u8],
187    cipher: &dyn Cipher,
188) -> Result<Vec<u8>, CryptoShredError> {
189    let (header, blob) = parse(envelope)?;
190    let aad = envelope_aad(header.subject_id.as_bytes(), header.key_version, context);
191    cipher
192        .decrypt(blob, &aad)
193        .map_err(|e| CryptoShredError::Crypto(e.to_string()))
194}

196/// Seal a [`PropertyValue`] by encoding it to bytes then sealing.
197///
198/// The serialized plaintext is held in a [`Zeroizing`] buffer so the cleartext
199/// property bytes are wiped after encryption rather than left on the heap.
200/// `context` is the opaque write-site binding (pass `b""` when there is none).
201///
202/// # Errors
203/// [`CryptoShredError::Crypto`] on encode failure, or the errors of [`seal`].
204pub fn seal_property_value(
205    value: &PropertyValue,
206    subject_id: &str,
207    key_version: u32,
208    context: &[u8],
209    cipher: &dyn Cipher,
210) -> Result<Vec<u8>, CryptoShredError> {
211    let bytes = Zeroizing::new(
212        value
213            .serialize()
214            .map_err(|e| CryptoShredError::Crypto(format!("property encode failed: {e}")))?,
215    );
216    seal(&bytes, subject_id, key_version, context, cipher)
217}
218
219/// Unseal an envelope back into a [`PropertyValue`].
220///
221/// `context` must match the value passed to [`seal_property_value`] (`b""` when
222/// there was none).
223///
224/// # Errors
225/// The errors of [`unseal`], plus [`CryptoShredError::Crypto`] if the decrypted
226/// bytes do not decode to a `PropertyValue`.
227pub fn unseal_property_value(
228    envelope: &[u8],
229    context: &[u8],
230    cipher: &dyn Cipher,
231) -> Result<PropertyValue, CryptoShredError> {
232    let bytes = unseal(envelope, context, cipher)?;
233    let (value, _) = PropertyValue::deserialize(&bytes)
234        .map_err(|e| CryptoShredError::Crypto(format!("property decode failed: {e}")))?;
235    Ok(value)
236}
