1//! GDPR crypto-shred foundation (Issue #3359, slice PR-1a).
2//!
3//! Crypto-shred implements GDPR "right to erasure" over an append-only
4//! bi-temporal store by encrypting the erasable payload under a **random
5//! per-subject key** and erasing by **destroying that key** — the ciphertext may
6//! remain across every tier but becomes permanently undecryptable, while the
7//! record structure, temporal coordinates, and hash chain stay intact and
8//! verifiable.
9//!
10//! ## What this slice (PR-1a) contains
11//!
12//! The cryptographic core **in isolation** — no live data-path integration:
13//! - [`subject::SubjectId`] / [`subject::SubjectKey`] — subject identity + key.
14//! - [`envelope`] — the self-describing sealed-envelope codec.
15//! - [`keyring`] — durable, fail-closed per-subject key registry + breadcrumb.
16//! - [`designation`] — which entities / property keys a subject seals.
17//! - [`attestation::ErasureAttestation`] — signed proof of key destruction.
18//! - [`CryptoShredState`] + the [`crate::db::AletheiaDB`] `designate_subject` /
19//!   `erase_subject` / `subject_key` API (see [`api`]).
20//!
21//! ## Deferred to PR-1b and later slices
22//!
23//! Live seal/unseal at the write/read choke points, the erasure-tombstone write
24//! transaction, chain-leaf ciphertext binding (AC4), HNSW exclusion, and the
25//! CLI/MCP surfaces. This slice proves key destruction → envelope unrecoverable
26//! at the unit level.
27//!
28//! ## Why a RANDOM DEK, not `HKDF(MEK, subject_id)`
29//!
30//! A key derived from the MEK stays re-derivable while the MEK lives — that is
31//! access-revocation, not erasure, and fails AC1. The per-subject DEK is a
32//! random independent 32-byte key; destroying its only durable (wrapped) copy is
33//! irreversible even for a holder of the MEK.

67/// Crypto-shred status of a single materialized property value (PR-1b).
68///
69/// Surfaced as a **side-channel** from the db read boundary alongside the
70/// (possibly-unsealed) [`PropertyMap`] — it is deliberately NOT a
71/// [`PropertyValue`] variant and NOT a wire-format change. A property is
72/// [`ShredStatus::Plaintext`] when it carried no sealed envelope, or when its
73/// subject is still active and the value was unsealed in place. It is
74/// [`ShredStatus::Erased`] when the value is a sealed envelope whose subject key
75/// has been destroyed — the plaintext is permanently unrecoverable and the read
76/// hook leaves the opaque ciphertext in place rather than fabricating a value.
77#[derive(Debug, Clone, PartialEq, Eq)]
78pub enum ShredStatus {
79    /// Not sealed, or unsealed in place (subject active).
80    Plaintext,
81    /// Sealed under a subject whose key was destroyed (unrecoverable).
82    Erased {
83        /// The erasure subject the value belonged to.
84        subject_id: String,
85    },
86}
87
88/// Per-property crypto-shred status keyed by property key (PR-1b).
89///
90/// Only carries entries that are **not** plain plaintext (i.e. erased sealed
91/// values); an absent key means [`ShredStatus::Plaintext`]. Empty when nothing
92/// on the read was sealed/erased, so a non-designated read allocates nothing.
93pub type ShredStatusMap = std::collections::HashMap<String, ShredStatus>;

156/// AAD domain-separation prefix binding a wrapped subject DEK to its subject id
157/// and key version.
158const WRAP_AAD_DOMAIN: &[u8] = b"aletheiadb-subject-wrap-dek-v1";
159
160/// Build the AEAD AAD used when wrapping / unwrapping a subject DEK.
161///
162/// Binds the subject id **and** the key version, so a future key-version
163/// downgrade (presenting a wrap blob under the wrong version) is caught by the
164/// AEAD auth check once rotation lands. Both the wrap (in `designate`) and the
165/// unwrap (in `subject_key`) sides must build this identically.
166fn wrap_aad(subject_id: &str, key_version: u32) -> Vec<u8> {
167    let subj = subject_id.as_bytes();
168    let mut aad = Vec::with_capacity(WRAP_AAD_DOMAIN.len() + 4 + subj.len());
169    aad.extend_from_slice(WRAP_AAD_DOMAIN);
170    aad.extend_from_slice(&key_version.to_le_bytes());
171    aad.extend_from_slice(subj);
172    aad
173}

175/// Build the sealed-envelope **write-site context** binding a value to its
176/// entity id and property key (PR-1b).
177///
178/// Folded into the envelope AEAD AAD (alongside subject id + key version) so a
179/// sealed value cannot be relocated to a different `(entity, key)` slot within
180/// the same subject without a loud AEAD auth failure. The seal (write) and
181/// unseal (read) sites MUST build this identically: `entity_id` little-endian
182/// followed by the raw property-key bytes.
183fn seal_context(entity_id: u64, key: &str) -> Vec<u8> {
184    let kb = key.as_bytes();
185    let mut ctx = Vec::with_capacity(8 + kb.len());
186    ctx.extend_from_slice(&entity_id.to_le_bytes());
187    ctx.extend_from_slice(kb);
188    ctx
189}

191/// Per-database crypto-shred state: the in-memory keyring, its durable paths,
192/// and the attestation signing key.
193///
194/// For an ephemeral database (no index persistence) `keyring_path` is `None` and
195/// the keyring is in-memory only; nothing is ever written to disk and no path is
196/// unwrapped, so the ephemeral construction never panics.
197pub struct CryptoShredState {
198    /// The in-memory per-subject key registry.
199    keyring: Mutex<SubjectKeyring>,
200    /// Durable keyring path, or `None` for an ephemeral (in-memory) database.
201    keyring_path: Option<PathBuf>,
202    /// In-flight-erase breadcrumb path (co-located with the keyring).
203    breadcrumb_path: Option<PathBuf>,
204    /// Ed25519 key used to sign erasure attestations. Sourced from the audit
205    /// signing-key env var when set, else freshly generated for this process.
206    signing_key: AuditSigningKey,
207    /// Lock-free fast-path gate (PR-1b): `true` iff at least one subject is
208    /// `Active`. The write hook consults this **without** locking so a database
209    /// with no active designations pays zero per-transaction cost.
210    has_active: AtomicBool,
211    /// Lock-free fast-path gate (PR-1b): `true` iff at least one subject exists
212    /// (active OR erased). The read hook consults this without locking so a
213    /// database that never used crypto-shred pays zero per-read cost.
214    has_any: AtomicBool,
215    /// Memoized subject-wrapping cipher (PR-1b). Built lazily on the first
216    /// seal/unseal from the database's encryption config and reused for the
217    /// process lifetime. Safe to cache because the subject-wrapping DEK is a
218    /// deterministic `HKDF(MEK, "subject-wrap")`. On a full-MEK rotation the
219    /// derived cipher changes, so [`CryptoShredState::rewrap_keyring`] (Slice 5)
220    /// **refreshes this cache** to the NEW subject-wrapping cipher after
221    /// re-wrapping the keyring — so a live uncached unseal, and a live
222    /// `designate` in the post-rotation window, immediately use the new-MEK
223    /// cipher without waiting for a reopen.
224    wrap_cipher: Mutex<Option<Arc<dyn Cipher>>>,
225}

236impl CryptoShredState {
237    /// Open the crypto-shred state, loading the durable keyring (fail-closed) and
238    /// running breadcrumb crash-recovery.
239    ///
240    /// `keyring_path` is `Some(<data_root>/subject_keyring.dat)` when index
241    /// persistence is enabled, else `None` (in-memory only).
242    ///
243    /// # Breadcrumb crash-recovery (fail-closed)
244    /// If an in-flight-erase breadcrumb is present, the named subject is forced
245    /// to `wrapped_key = None` + `Erased` (creating a tombstone entry if absent),
246    /// the keyring is rewritten durably, and the breadcrumb is cleared. A
247    /// breadcrumbed subject is treated as erased no matter the keyring state, so
248    /// an interrupted erase never leaves the key half-present.
249    ///
250    /// # Errors
251    /// [`CryptoShredError::KeyringCorrupt`] if the keyring file is present but
252    /// corrupt, or [`CryptoShredError::Io`] on durable-write failure during
253    /// recovery.
254    pub fn open(keyring_path: Option<PathBuf>) -> Result<Self, CryptoShredError> {
255        // Source the attestation signing key: prefer the audit env var (stable
256        // across restarts), else generate a fresh per-process key. Attestations
257        // embed their signer's public key, so verification never depends on
258        // re-sourcing the same key.
259        //
260        // OPERATIONAL NOTE: for durable deployments a stable
261        // `crate::audit::SIGNING_KEY_ENV` seed SHOULD be set. Without it each
262        // process generates a fresh signing key, so attestations minted before
263        // and after a restart are signed by *different* keys; cross-restart
264        // verification then relies solely on each attestation's embedded public
265        // key (self-consistent, but with no single stable signer identity to
266        // pin out of band). A stable seed gives one durable signer identity an
267        // auditor can trust across restarts.
268        let signing_key = AuditSigningKey::from_env(crate::audit::SIGNING_KEY_ENV)
269            .unwrap_or_else(|_| AuditSigningKey::generate());
270
271        let breadcrumb_path = keyring_path
272            .as_ref()
273            .and_then(|p| p.parent())
274            .map(|dir| dir.join(BREADCRUMB_FILENAME));
275
276        let keyring = match keyring_path.as_ref() {
277            Some(path) => keyring::load_keyring(path)?,
278            None => SubjectKeyring::new(),
279        };
280
281        let has_active = keyring.any_active();
282        let has_any = !keyring.is_empty();
283        let state = Self {
284            keyring: Mutex::new(keyring),
285            keyring_path,
286            breadcrumb_path,
287            signing_key,
288            has_active: AtomicBool::new(has_active),
289            has_any: AtomicBool::new(has_any),
290            wrap_cipher: Mutex::new(None),
291        };
292
293        state.recover_from_breadcrumb()?;
294        Ok(state)
295    }
296
297    /// Resume an interrupted erase from a breadcrumb, if present.
298    fn recover_from_breadcrumb(&self) -> Result<(), CryptoShredError> {
299        let (Some(keyring_path), Some(breadcrumb_path)) =
300            (self.keyring_path.as_ref(), self.breadcrumb_path.as_ref())
301        else {
302            return Ok(());
303        };
304        let Some(subject_id) = keyring::read_breadcrumb(breadcrumb_path)? else {
305            return Ok(());
306        };
307
308        let now = crate::core::temporal::time::now().wallclock();
309        {
310            let mut keyring = self.lock_keyring();
311            keyring.force_erased(&subject_id, now);
312            keyring::save_keyring(keyring_path, &keyring)?;
313            self.refresh_gates(&keyring);
314        }
315        keyring::clear_breadcrumb(breadcrumb_path);
316        Ok(())
317    }

331    /// Lock the in-memory keyring. Poisoning is treated as a fatal fail — a
332    /// panic while holding the keyring lock leaves erasure state ambiguous, so we
333    /// prefer to surface it rather than proceed on possibly-stale key state.
334    fn lock_keyring(&self) -> std::sync::MutexGuard<'_, SubjectKeyring> {
335        self.keyring
336            .lock()
337            .unwrap_or_else(std::sync::PoisonError::into_inner)
338    }

340    /// Recompute the lock-free fast-path gates from a locked keyring guard.
341    /// Called after any mutation that can change active/any state.
342    fn refresh_gates(&self, guard: &SubjectKeyring) {
343        self.has_active.store(guard.any_active(), Ordering::Relaxed);
344        self.has_any.store(!guard.is_empty(), Ordering::Relaxed);
345    }

364    /// Export the durable keyring/designation-registry as CRC-wrapped sidecar
365    /// bytes for folding into an `.albk` backup (Issue #3665).

367    /// Returns the exact [`keyring::encode_sidecar_with_crc`] wire form — the
368    /// same bytes the durable `subject_keyring.dat` holds — so the restore path

381        }
382        keyring::encode_sidecar_with_crc(&guard.to_sidecar())
383    }

457    /// Designate `subject_id` over `targets`, generating and wrapping a random
458    /// DEK (for a new subject) or merging targets into an existing active
459    /// subject. `wrap_cipher` is the subject-wrapping cipher derived from the
460    /// MEK; the DEK is wrapped with the subject id bound as AEAD AAD.
461    ///
462    /// # Errors
463    /// - [`CryptoShredError::SubjectErased`] if the subject was already erased.
464    /// - [`CryptoShredError::Crypto`] on wrap failure.
465    /// - [`CryptoShredError::Io`] on durable-write failure.
466    pub(crate) fn designate(
467        &self,
468        subject_id: &SubjectId,
469        mut targets: Vec<DesignationTarget>,
470        wrap_cipher: &dyn crate::encryption::Cipher,
471    ) -> Result<(), CryptoShredError> {
472        use keyring::{INITIAL_SUBJECT_KEY_VERSION, SubjectEntry, SubjectState, WrappedKey};
473
474        let now = crate::core::temporal::time::now().wallclock();
475        let mut guard = self.lock_keyring();
476
477        if let Some(existing) = guard.get(subject_id.as_str()).cloned() {
478            if existing.state == SubjectState::Erased {
479                return Err(CryptoShredError::SubjectErased(subject_id.to_string()));
480            }
481            // Merge new targets into the existing active subject (dedup).
482            let mut merged = existing.designation.clone();
483            for t in targets.drain(..) {
484                if !merged.contains(&t) {
485                    merged.push(t);
486                }
487            }
488            let mut entry = existing.clone();
489            entry.designation = merged;
490            guard.upsert(entry);
491        } else {
492            // New subject: random DEK, wrapped under the subject-wrapping cipher.
493            let dek = SubjectKey::generate();
494            let wrap_aad = wrap_aad(subject_id.as_str(), INITIAL_SUBJECT_KEY_VERSION);
495            let wrapped = wrap_cipher
496                .encrypt(dek.expose_bytes(), &wrap_aad)
497                .map_err(|e| CryptoShredError::Crypto(e.to_string()))?;
498            let mut deduped: Vec<DesignationTarget> = Vec::with_capacity(targets.len());
499            for t in targets.drain(..) {
500                if !deduped.contains(&t) {
501                    deduped.push(t);
502                }
503            }
504            let entry = SubjectEntry {
505                subject_id: subject_id.as_str().to_string(),
506                wrapped_key: Some(WrappedKey {
507                    key_version: INITIAL_SUBJECT_KEY_VERSION,
508                    wrapped,
509                }),
510                designation: deduped,
511                state: SubjectState::Active,
512                created_at_micros: now,
513                erased_at_micros: None,
514                attestation: None,
515            };
516            guard.upsert(entry);
517            guard.cache_key(subject_id.as_str(), dek);
518        }
519
520        if let Some(path) = self.keyring_path() {
521            keyring::save_keyring(path, &guard)?;
522        }
523        self.refresh_gates(&guard);
524        Ok(())
525    }

542    /// Fetch a subject's unwrapped DEK from an **already-locked** keyring guard.
543    ///
544    /// Checks the unwrapped-DEK cache first, else unwraps the stored wrapped blob
545    /// with `wrap_cipher` and caches the result. Kept `&mut guard`-based (not
546    /// `&self`) so the seal/unseal hot paths, which already hold the keyring lock
547    /// to consult the reverse index, never re-lock (`std::sync::Mutex` is not
548    /// reentrant — re-locking would deadlock).
549    ///
550    /// # Errors
551    /// - [`CryptoShredError::NotDesignated`] if the subject is unknown.
552    /// - [`CryptoShredError::SubjectErased`] if the subject was erased (key gone).
553    /// - [`CryptoShredError::Crypto`] on unwrap / AEAD failure.
554    fn dek_from_guard(
555        guard: &mut SubjectKeyring,
556        subject_id: &str,
557        wrap_cipher: &dyn crate::encryption::Cipher,
558    ) -> Result<SubjectKey, CryptoShredError> {
559        use keyring::SubjectState;
560
561        if let Some(cached) = guard.cached_key(subject_id) {
562            return Ok(cached.clone());
563        }
564        let entry = guard
565            .get(subject_id)
566            .ok_or_else(|| CryptoShredError::NotDesignated(subject_id.to_string()))?;
567        if entry.state == SubjectState::Erased {
568            return Err(CryptoShredError::SubjectErased(subject_id.to_string()));
569        }
570        let wrapped = entry
571            .wrapped_key
572            .as_ref()
573            .ok_or_else(|| CryptoShredError::SubjectErased(subject_id.to_string()))?;
574        // Zeroize every transit copy of the unwrapped DEK: the decrypt output
575        // lands in a Zeroizing<Vec<u8>>, and the fixed-size array we hand to
576        // SubjectKey is built inside a Zeroizing<[u8;N]> — neither is left in a
577        // plain buffer that could survive on the stack/heap after use.
578        let key_version = wrapped.key_version;
579        let aad = wrap_aad(subject_id, key_version);
580        let raw = Zeroizing::new(
581            wrap_cipher
582                .decrypt(&wrapped.wrapped, &aad)
583                .map_err(|e| CryptoShredError::Crypto(e.to_string()))?,
584        );
585        if raw.len() != subject::SUBJECT_KEY_LEN {
586            return Err(CryptoShredError::Crypto(
587                "unwrapped subject key has wrong length".to_string(),
588            ));
589        }
590        let mut bytes = Zeroizing::new([0u8; subject::SUBJECT_KEY_LEN]);
591        bytes.copy_from_slice(&raw);
592        let key = SubjectKey::from_bytes(bytes);
593        guard.cache_key(subject_id, key.clone());
594        Ok(key)
595    }

597    /// Erase a subject: destroy its key (physically remove the wrapped blob),
598    /// zeroize any cached copy, and return a signed attestation. Multi-step and
599    /// crash-consistent via a breadcrumb.
600    ///
601    /// Re-erase is an idempotent no-op returning the recorded attestation.
602    ///
603    /// # Errors
604    /// - [`CryptoShredError::NotDesignated`] if the subject is unknown (AC6).
605    /// - [`CryptoShredError::Io`] on durable-write failure.
606    pub(crate) fn erase(
607        &self,
608        subject_id: &SubjectId,
609    ) -> Result<ErasureAttestation, CryptoShredError> {
610        use keyring::SubjectState;
611
612        let mut guard = self.lock_keyring();
613        // Clone the entry up front so the guard is free to be mutated below.
614        let existing = guard
615            .get(subject_id.as_str())
616            .cloned()
617            .ok_or_else(|| CryptoShredError::NotDesignated(subject_id.to_string()))?;
618
619        // Idempotent re-erase: return the recorded attestation if present.
620        if existing.state == SubjectState::Erased {
621            if let Some(record) = existing.attestation.as_ref() {
622                return ErasureAttestation::from_record(record);
623            }
624            // Crash-recovered tombstone with no attestation: mint one now over
625            // the recorded facts so the caller still gets a signed proof.
626            let entity_count = existing.designation.len() as u32;
627            let ts = existing
628                .erased_at_micros
629                .unwrap_or_else(|| crate::core::temporal::time::now().wallclock());
630            let attestation =
631                ErasureAttestation::sign(self.signing_key(), subject_id.as_str(), entity_count, ts);
632            let mut updated = existing.clone();
633            updated.attestation = Some(attestation.to_record());
634            guard.upsert(updated);
635            if let Some(path) = self.keyring_path() {
636                keyring::save_keyring(path, &guard)?;
637            }
638            return Ok(attestation);
639        }
640
641        let entity_count = existing.designation.len() as u32;
642        let now = crate::core::temporal::time::now().wallclock();
643
644        // (a) Durable breadcrumb BEFORE mutating the keyring, so an interrupted
645        // erase resumes deterministically (fail-closed: a breadcrumbed subject
646        // is treated as erased on restart no matter the keyring state).
647        if let Some(bc) = self.breadcrumb_path() {
648            keyring::write_breadcrumb(bc, subject_id.as_str())?;
649        }
650
651        // (b/c/d) Build+sign the attestation, then physically remove the wrapped
652        // DEK, flip to Erased, and zeroize the cached key.
653        let attestation =
654            ErasureAttestation::sign(self.signing_key(), subject_id.as_str(), entity_count, now);
655        guard.erase_in_memory(subject_id.as_str(), now, attestation.to_record());
656        self.refresh_gates(&guard);
657
658        // Rewrite the keyring durably so the wrapped blob is physically gone.
659        if let Some(path) = self.keyring_path() {
660            keyring::save_keyring(path, &guard)?;
661        }
662
663        // NOTE (PR-1b): the durable keyring record above IS the erasure record —
664        // an erased subject's key is physically gone and its state is `Erased`,
665        // fail-closed across restart. A WAL-transaction erasure *tombstone*
666        // (subject id + entity/version counts + timestamp, riding the
667        // `is_tombstone` machinery) plus changefeed observability is deliberately
668        // DEFERRED to a #3413-aligned follow-up: emitting it needs a WAL-format
669        // extension to survive crash recovery, and doing it here would force
670        // `db.write(...)` — acquiring `historical`/`wal` — while this method holds
671        // the keyring `Mutex`, inverting the lock order (the keyring lock must be
672        // *released before* the write-path primitives are taken; see the
673        // lock-order note in CLAUDE.md / seams §6). So erase does NOT call
674        // `db.write` in this slice — no lock-order change is required.
675
676        // (e) Clear the breadcrumb — erase is complete.
677        if let Some(bc) = self.breadcrumb_path() {
678            keyring::clear_breadcrumb(bc);
679        }
680
681        Ok(attestation)
682    }

701    /// on first use (see the `wrap_cipher` field doc for the caching rationale).
702    ///
703    /// # Errors
704    /// Propagates `build`'s error (e.g. encryption not configured).
705    pub(crate) fn cached_wrap_cipher(
706        &self,
707        build: impl FnOnce() -> Result<Box<dyn Cipher>, CryptoShredError>,
708    ) -> Result<Arc<dyn Cipher>, CryptoShredError> {
709        let mut guard = self
710            .wrap_cipher
711            .lock()
712            .unwrap_or_else(std::sync::PoisonError::into_inner);

721    /// Whether any **active** subject holds a wrapped DEK stamped at a key
722    /// version older than `new_version` (subject-keyring re-wrap scoping,
723    /// Slice 5).
724    ///
725    /// Used by the rotation driver to decide whether the subject-keyring layer
726    /// is in scope for a full-MEK rotation: a database with no subjects, or one
727    /// whose every subject is already at (or beyond) `new_version`, needs no
728    /// re-wrap pass and records the layer `Skipped`. Erased subjects (whose
729    /// wrapped key is physically gone) are never stale — there is nothing to
730    /// re-wrap.
731    #[must_use]
732    pub(crate) fn keyring_has_stale_wrapped_entries(&self, new_version: u32) -> bool {
733        use keyring::SubjectState;
734        let guard = self.lock_keyring();
735        guard.iter_entries().any(|e| {
736            e.state == SubjectState::Active
737                && e.wrapped_key
738                    .as_ref()
739                    .is_some_and(|w| w.key_version < new_version)
740        })
741    }

743    /// Re-wrap every stale-generation subject DEK from `old_wrap_cipher` to
744    /// `new_wrap_cipher`, stamping `new_version` (subject-keyring re-wrap on a
745    /// full-MEK rotation, Slice 5).
746    ///
747    /// A full-MEK rotation re-derives the subject-wrapping cipher
748    /// (`HKDF(MEK, "subject-wrap")`), so every per-subject wrapped DEK sealed
749    /// under the OLD MEK must be re-wrapped under the NEW MEK or it becomes
750    /// undecryptable once the operator drops the old key. This unwraps each DEK
751    /// under `old_wrap_cipher` (never leaving `Zeroizing`) and re-wraps it under
752    /// `new_wrap_cipher`, advancing only the wrapping — the random per-subject
753    /// DEK, the designation, lifecycle state, attestation, and timestamps are
754    /// untouched.
755    ///
756    /// # Erased invariant & idempotency
757    /// - An **erased** subject (or any entry with `wrapped_key == None`) is
758    ///   SKIPPED — its key is destroyed and must never be re-introduced.
759    /// - An entry already at `key_version >= new_version` is SKIPPED, so a
760    ///   crash-resumed or double-invoked pass re-wraps zero entries and is a
761    ///   safe no-op.
762    ///
763    /// After the pass the keyring is persisted **once** (atomic, durable) and
764    /// the memoized `wrap_cipher` is refreshed to `new_wrap_cipher` so a live
765    /// uncached unseal (and a live `designate`) in the post-rotation window
766    /// wraps/unwraps under the new MEK immediately, without waiting for a
767    /// reopen. Returns the number of entries re-wrapped.
768    ///
769    /// # Transactional (all-or-nothing)
770    /// The re-wrap is planned and computed **before** the keyring is mutated:
771    /// every stale DEK is unwrapped under `old_wrap_cipher` and re-wrapped under
772    /// `new_wrap_cipher` into a local `updates` buffer first. Only if EVERY
773    /// entry succeeds are the new blobs applied, persisted once, and the cache
774    /// refreshed. If ANY unwrap / re-wrap / length-check fails, the function
775    /// returns `Err` having mutated NOTHING — not the in-memory keyring, not the
776    /// on-disk file, and not the memoized cipher — so a mid-pass failure never
777    /// leaves a partially re-wrapped keyring or a stale/invalid cache.
778    ///
779    /// # Durability (v1 save-once)
780    /// The keyring is saved a single time after all entries are re-wrapped, not
781    /// per entry. A crash mid-pass therefore leaves the on-disk keyring at the
782    /// pre-pass state; the rotation ledger records the layer `Pending`, and the
783    /// idempotent resume re-runs the whole pass under the old MEK. A per-entry
784    /// durable cursor (bounding the redo window for very large subject counts)
785    /// is a named follow-up.
786    ///
787    /// # Errors
788    /// - [`CryptoShredError::Crypto`] if a DEK fails to unwrap under
789    ///   `old_wrap_cipher` (e.g. a wrong/new key presented mid-pass — a loud
790    ///   AEAD authentication failure, never silent wrong data) or fails to
791    ///   re-wrap, or unwraps to the wrong length.
792    /// - [`CryptoShredError::Io`] on durable-write failure.
793    pub(crate) fn rewrap_keyring(
794        &self,
795        old_wrap_cipher: &dyn Cipher,
796        new_wrap_cipher: &Arc<dyn Cipher>,
797        new_version: u32,
798    ) -> Result<usize, CryptoShredError> {
799        use keyring::{SubjectState, WrappedKey};
800
801        let mut guard = self.lock_keyring();
802
803        // Plan: the (subject_id, old_key_version, wrapped_blob) of every entry
804        // that needs re-wrapping. Skip erased / keyless entries (erased
805        // invariant) and entries already at/beyond the target (idempotent).
806        struct RewrapItem {
807            subject_id: String,
808            old_key_version: u32,
809            wrapped: Vec<u8>,
810        }
811        let mut plan: Vec<RewrapItem> = Vec::new();
812        for entry in guard.iter_entries() {
813            if entry.state == SubjectState::Erased {
814                continue;
815            }
816            let Some(w) = entry.wrapped_key.as_ref() else {
817                continue;
818            };
819            if w.key_version >= new_version {
820                continue;
821            }
822            plan.push(RewrapItem {
823                subject_id: entry.subject_id.clone(),
824                old_key_version: w.key_version,
825                wrapped: w.wrapped.clone(),
826            });
827        }
828
829        // Phase 1: decrypt every stale DEK under the OLD cipher and re-encrypt
830        // it under the NEW cipher into a local `updates` buffer, WITHOUT
831        // mutating the keyring. If ANY entry fails to unwrap / re-wrap / pass
832        // the length check, return `Err` with ZERO mutation and ZERO cache
833        // change — the re-wrap is all-or-nothing.
834        let mut updates: Vec<(String, WrappedKey)> = Vec::with_capacity(plan.len());
835        for item in &plan {
836            // Unwrap the DEK under the OLD wrapping cipher, binding the entry's
837            // own current key version as AAD (mirrors `dek_from_guard`). The
838            // plaintext DEK never leaves `Zeroizing`.
839            let old_aad = wrap_aad(&item.subject_id, item.old_key_version);
840            let raw = Zeroizing::new(
841                old_wrap_cipher
842                    .decrypt(&item.wrapped, &old_aad)
843                    .map_err(|e| CryptoShredError::Crypto(e.to_string()))?,
844            );
845            if raw.len() != subject::SUBJECT_KEY_LEN {
846                return Err(CryptoShredError::Crypto(
847                    "unwrapped subject key has wrong length".to_string(),
848                ));
849            }
850            // Re-wrap under the NEW cipher, binding the NEW version as AAD.
851            let new_aad = wrap_aad(&item.subject_id, new_version);
852            let new_blob = new_wrap_cipher
853                .encrypt(&raw, &new_aad)
854                .map_err(|e| CryptoShredError::Crypto(e.to_string()))?;
855            updates.push((
856                item.subject_id.clone(),
857                WrappedKey {
858                    key_version: new_version,
859                    wrapped: new_blob,
860                },
861            ));
862        }
863
864        // Phase 2 (every entry succeeded): apply the re-wraps, then persist
865        // once. A partial-failure path can never reach here.
866        let rewrapped = updates.len();
867        for (subject_id, wrapped_key) in updates {
868            guard.set_wrapped_key(&subject_id, wrapped_key);
869        }
870
871        // v1 save-once: one atomic, durable write after re-wrapping all entries.
872        if let Some(path) = self.keyring_path().filter(|_| rewrapped > 0) {
873            keyring::save_keyring(path, &guard)?;
874        }
875        drop(guard);
876
877        // Refresh the memoized subject-wrapping cipher to the NEW cipher so a
878        // live uncached seal/unseal — and a live `designate` in the
879        // post-rotation window — uses the new-MEK cipher immediately, no reopen
880        // required (the cached one was derived from the old MEK). Done
881        // unconditionally: even a zero-re-wrap pass (idempotent resume) reaches
882        // here only after Phase 1 proved the new cipher is the live one, so
883        // caching it is always correct.
884        {
885            let mut wc = self
886                .wrap_cipher
887                .lock()
888                .unwrap_or_else(std::sync::PoisonError::into_inner);
889            *wc = Some(Arc::clone(new_wrap_cipher));
890        }
891
892        Ok(rewrapped)
893    }

895    /// Seal the designated property values of an entity into `SUBJ` envelopes
896    /// (PR-1b write hook).
897    ///
898    /// For every property key that an **active** subject designates (via the
899    /// reverse index), the value is replaced with a sealed-envelope
900    /// [`PropertyValue::Bytes`]. Engine-reserved keys (`__aletheia_*` /
901    /// `__shred_*`) are never sealed (short-circuited by
902    /// [`designation::should_seal_key`]). The returned map must be routed
903    /// **byte-identically** to both the current and historical tiers (seal
904    /// exactly once — the envelope carries a random per-encryption nonce).
905    ///
906    /// # Fail-closed erase-vs-seal race
907    /// If a property key is designated but **every** covering subject has been
908    /// **erased** (no active subject can seal it — its DEK is destroyed), the
909    /// write is **aborted** with [`CryptoShredError::WriteAfterErasure`] rather
910    /// than persisting the erasable value as plaintext. This closes the race in
911    /// which an in-flight write buffered before an erasure would otherwise commit
912    /// afterward: sealing is impossible (the key is gone), so abort is the only
913    /// fail-closed outcome. A key designated by *some* active subject (even if
914    /// another covering subject is erased) still seals under the first active one.
915    ///
916    /// # Errors
917    /// - [`CryptoShredError::WriteAfterErasure`] when a designated key is covered
918    ///   only by erased subjects (fail-closed abort — see above).
919    /// - [`CryptoShredError::Crypto`] on DEK unwrap / AEAD seal failure. Fail-closed:
920    ///   a seal error aborts the enclosing write rather than persisting plaintext.
921    pub(crate) fn seal_map(
922        &self,
923        is_node: bool,
924        entity_id: u64,
925        props: PropertyMap,
926        wrap_cipher: &dyn Cipher,
927        algorithm: Algorithm,
928    ) -> Result<PropertyMap, CryptoShredError> {
929        use crate::core::interning::GLOBAL_INTERNER;
930        use crate::core::property::PropertyKey;
931
932        let mut guard = self.lock_keyring();
933        let subjects: Vec<String> = guard.subjects_for_entity(is_node, entity_id).to_vec();
934        if subjects.is_empty() {
935            return Ok(props);
936        }
937
938        // Plan which keys to seal (resolve interned keys to owned strings first,
939        // so re-interning via the builder never nests inside a resolve guard).
940        struct SealItem {
941            key: PropertyKey,
942            key_str: String,
943            value: PropertyValue,
944            subject_id: String,
945            key_version: u32,
946        }
947        let mut plan: Vec<SealItem> = Vec::new();
948        for (key, value) in props.iter() {
949            // Resolve the interned key to an owned String via `resolve_with` (no
950            // owned-Arc deprecation, and — critically — we never re-enter the
951            // interner while a resolve guard is held: `insert_by_key` below takes
952            // the already-interned key, it does not re-intern a string).
953            let Some(key_str) = GLOBAL_INTERNER.resolve_with(*key, |s| s.to_string()) else {
954                continue;
955            };
956            if designation::is_reserved_key(&key_str) {
957                continue;
958            }
959            // Fail-closed erase-vs-seal race: choose the first ACTIVE subject that
960            // designates this key. If the key is designated only by ERASED
961            // subject(s) (their DEK is destroyed), we cannot seal it — abort the
962            // write rather than persist erasable plaintext.
963            let mut active: Option<(String, u32)> = None;
964            let mut designated_by_erased: Option<String> = None;
965            for subject_id in &subjects {
966                let Some(entry) = guard.get(subject_id) else {
967                    continue;
968                };
969                if !designation::any_should_seal(&entry.designation, is_node, entity_id, &key_str) {
970                    continue;
971                }
972                match entry.state {
973                    keyring::SubjectState::Active => {
974                        let key_version = entry
975                            .wrapped_key
976                            .as_ref()
977                            .map_or(keyring::INITIAL_SUBJECT_KEY_VERSION, |w| w.key_version);
978                        active = Some((subject_id.clone(), key_version));
979                        break;
980                    }
981                    keyring::SubjectState::Erased => {
982                        designated_by_erased.get_or_insert_with(|| subject_id.clone());
983                    }
984                }
985            }
986            if let Some((subject_id, key_version)) = active {
987                plan.push(SealItem {
988                    key: *key,
989                    key_str,
990                    value: value.clone(),
991                    subject_id,
992                    key_version,
993                });
994            } else if let Some(subject_id) = designated_by_erased {
995                // Designated, but only by erased subject(s): sealing is
996                // impossible (key gone). Fail closed — abort the commit.
997                return Err(CryptoShredError::WriteAfterErasure(subject_id));
998            }
999        }
1000
1001        if plan.is_empty() {
1002            return Ok(props);
1003        }
1004
1005        let mut builder = PropertyMapBuilder::from_map(props);
1006        for item in plan {
1007            let dek = Self::dek_from_guard(&mut guard, &item.subject_id, wrap_cipher)?;
1008            let cipher = create_cipher(algorithm, &Zeroizing::new(*dek.expose_bytes()));
1009            let context = seal_context(entity_id, &item.key_str);
1010            let sealed = envelope::seal_property_value(
1011                &item.value,
1012                &item.subject_id,
1013                item.key_version,
1014                &context,
1015                cipher.as_ref(),
1016            )?;
1017            builder =
1018                builder.insert_by_key(item.key, PropertyValue::Bytes(std::sync::Arc::from(sealed)));
1019        }
1020        Ok(builder.build())
1021    }

1023    /// Materialize sealed property values on read (PR-1b read hook).
1024    ///
1025    /// For each `SUBJ` envelope whose subject the keyring knows: if the subject
1026    /// is **active**, unseal to plaintext in place; if it is **erased**, leave the
1027    /// opaque ciphertext untouched and record [`ShredStatus::Erased`] in the
1028    /// returned side-channel — never fabricated, never silently dropped. A
1029    /// `Bytes` value that merely *looks* like an envelope but whose subject is
1030    /// unknown, or that fails to unseal under an active subject (tamper /
1031    /// corruption), is left byte-untouched (opaque ciphertext, never plaintext).
1032    ///
1033    /// Returns the (possibly-transformed) map plus a [`ShredStatusMap`] carrying
1034    /// only the erased keys (empty allocation-free when nothing was sealed).
1035    pub(crate) fn unseal_map(
1036        &self,
1037        entity_id: u64,
1038        props: PropertyMap,
1039        wrap_cipher: &dyn Cipher,
1040        algorithm_hint: Algorithm,
1041    ) -> (PropertyMap, ShredStatusMap) {
1042        use crate::core::interning::GLOBAL_INTERNER;
1043        use crate::core::property::PropertyKey;
1044
1045        let mut statuses = ShredStatusMap::new();
1046        let mut guard = self.lock_keyring();
1047
1048        // Collect envelope-bearing keys whose subject the keyring recognizes.
1049        struct SealedItem {
1050            key: PropertyKey,
1051            key_str: String,
1052            header: envelope::EnvelopeHeader,
1053            bytes: std::sync::Arc<[u8]>,
1054        }
1055        let mut sealed: Vec<SealedItem> = Vec::new();
1056        for (key, value) in props.iter() {
1057            let PropertyValue::Bytes(b) = value else {
1058                continue;
1059            };
1060            if !envelope::is_envelope(b) {
1061                continue;
1062            }
1063            let Ok(header) = envelope::parse_header(b) else {
1064                continue;
1065            };
1066            if guard.get(&header.subject_id).is_none() {
1067                // Looks like an envelope but no such subject — treat as opaque
1068                // user bytes, never as ours.
1069                continue;
1070            }
1071            let Some(key_str) = GLOBAL_INTERNER.resolve_with(*key, |s| s.to_string()) else {
1072                continue;
1073            };
1074            sealed.push(SealedItem {
1075                key: *key,
1076                key_str,
1077                header,
1078                bytes: std::sync::Arc::clone(b),
1079            });
1080        }
1081
1082        if sealed.is_empty() {
1083            return (props, statuses);
1084        }
1085
1086        let mut builder = PropertyMapBuilder::from_map(props);
1087        for item in sealed {
1088            let is_erased = guard
1089                .get(&item.header.subject_id)
1090                .is_some_and(|e| e.state == keyring::SubjectState::Erased);
1091            if is_erased {
1092                // Key destroyed: leave opaque ciphertext, report erased.
1093                statuses.insert(
1094                    item.key_str,
1095                    ShredStatus::Erased {
1096                        subject_id: item.header.subject_id,
1097                    },
1098                );
1099                continue;
1100            }
1101            // Active subject: unseal to plaintext. On any failure leave the
1102            // opaque ciphertext (never fabricate a plaintext value).
1103            let Ok(dek) = Self::dek_from_guard(&mut guard, &item.header.subject_id, wrap_cipher)
1104            else {
1105                continue;
1106            };
1107            let algorithm = algorithm_from_id(item.header.algorithm_id).unwrap_or(algorithm_hint);
1108            let cipher = create_cipher(algorithm, &Zeroizing::new(*dek.expose_bytes()));
1109            let context = seal_context(entity_id, &item.key_str);
1110            if let Ok(value) =
1111                envelope::unseal_property_value(&item.bytes, &context, cipher.as_ref())
1112            {
1113                builder = builder.insert_by_key(item.key, value);
1114            }
1115        }
1116        (builder.build(), statuses)
1117    }
