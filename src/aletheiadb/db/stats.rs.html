1//! Database-wide statistics snapshot (Issue #3222).
2//!
3//! Aggregates the already-maintained internal counters — [`CurrentStats`],
4//! [`HistoricalStats`], [`TieredStorageMetrics`]/[`ColdStorageStats`], and the
5//! WAL system's atomic counters — into a single serializable
6//! [`DatabaseStats`] value so callers (including the MCP `database_stats`
7//! tool) can orient themselves in one call: how much graph is here, how much
8//! history can be time-traveled through, where that history lives
9//! (hot / warm-cache / cold), and what durability guarantees are active.
10//!
11//! # Performance: O(1) by construction — no version scans
12//!
13//! Every numeric field is read from a counter that the storage engines
14//! already maintain incrementally:
15//!
16//! - Current node/edge counts: index length reads
17//!   (`CurrentStorage::node_count`/`edge_count`).
18//! - Historical version/anchor/delta/unique counts:
19//!   `HistoricalStorage::stats()`, which per Issue #212 "returns cached
20//!   counters in O(1) time instead of iterating through all versions"
21//!   (see `src/storage/historical/mod.rs`).
22//! - Cold-tier counters and hot/warm/cold access distribution: atomic
23//!   counter snapshots from `TieredStorage::metrics()` and
24//!   `RedbColdStorage::stats()`. The cold version counts are seeded from the
25//!   persisted tables when the database is opened (an O(1) B-tree metadata
26//!   read), so they survive restarts; byte and access counters are
27//!   process-lifetime.
28//! - WAL state: a field copy (`durability_mode`) plus atomic loads
29//!   (`current_lsn`, `total_appends`, `consecutive_flush_errors`).
30//!
31//! Calling [`AletheiaDB::stats`] therefore never triggers a full version
32//! scan, regardless of database size.
33//!
34//! # Disabled subsystems are tagged, never zero-reported
35//!
36//! A disabled tier is reported as `{"enabled": false}` with **no** count
37//! fields, so a consumer (human or LLM) can never mistake "cold storage is
38//! not configured" for "cold storage holds 0 versions".
39//!
40//! [`CurrentStats`]: crate::storage::current::CurrentStats
41//! [`HistoricalStats`]: crate::storage::historical::HistoricalStats
42//! [`TieredStorageMetrics`]: crate::storage::tiered_storage::TieredStorageMetrics
43//! [`ColdStorageStats`]: crate::storage::redb_cold_storage::ColdStorageStats
44

48/// Point-in-time snapshot of database size, temporal depth, storage-tier
49/// distribution, and WAL state.
50///
51/// Produced by [`AletheiaDB::stats`]. All counters are O(1)/cached reads
52/// (see the module docs); this is a consistent-enough operational snapshot,
53/// not a transactionally frozen view — concurrent writers may advance
54/// individual counters between reads.
55#[derive(Debug, Clone, PartialEq)]
56#[cfg_attr(feature = "serde", derive(serde::Serialize))]
57#[non_exhaustive]
58pub struct DatabaseStats {
59    /// Current-state graph size (the hot, in-RAM tier queried by
60    /// current-state operations).
61    pub current: CurrentStateStats,
62    /// Bi-temporal depth of the in-RAM historical store, including
63    /// anchor/delta compression effectiveness.
64    pub historical: HistoricalDepthStats,
65    /// Cold-tier (disk) presence and, when enabled, counters plus the
66    /// hot/warm/cold read distribution. `enabled: false` means the tier is
67    /// not configured — it does NOT mean zero cold versions.
68    pub cold_storage: ColdStorageTierStats,
69    /// Write-ahead-log durability state.
70    pub wal: WalStateStats,
71    /// Tamper-evident provenance hash chain status (Issue #3351). `enabled:
72    /// false` with all-`None` fields when the chain is not configured.
73    pub chain: ProvenanceChainStats,
74    /// Per-namespace current node/edge counts (Issue #3349). One entry per
75    /// registered or populated namespace, sorted by name; O(1) membership-index
76    /// reads. Empty on a database that only uses the implicit `default`
77    /// namespace with no explicit registrations (the `default` entry is still
78    /// present when it holds entities).
79    ///
80    /// Serialized on the MCP/HTTP `database_stats` JSON since PR3a (the
81    /// coordinated surface slice); each entry is `{name, node_count,
82    /// edge_count}`.
83    pub namespaces: Vec<crate::db::namespace_query::NamespaceCount>,
84    /// Push-changefeed subscription state, including the per-principal quota
85    /// breakdown (Issue #3678). This is the authenticated surface for the
86    /// per-principal detail deliberately kept OFF the bounded `/metrics`
87    /// exposition (which stays bounded-label-only).
88    pub changefeed: ChangefeedStats,
89}

91/// Push-changefeed subscription state for the `database_stats` snapshot
92/// (Issue #3678). Reads the broadcaster registry under a brief read lock —
93/// O(live subscriptions), not a version scan.
94#[derive(Debug, Clone, PartialEq, Eq)]
95#[cfg_attr(feature = "serde", derive(serde::Serialize))]
96#[non_exhaustive]
97pub struct ChangefeedStats {
98    /// Total currently-live changefeed subscriptions across all principals and
99    /// the bare embedded path.
100    pub active_subscriptions: usize,
101    /// Per-principal live-subscription breakdown, sorted by principal id. Empty
102    /// when no principal-scoped subscriptions are live. This is the per-principal
103    /// quota visibility AC(e) surfaces on the authenticated JSON path rather than
104    /// as an unbounded `/metrics` label.
105    pub per_principal: Vec<PrincipalSubscriptionStat>,
106}

163/// Temporal depth and anchor/delta compression effectiveness of the in-RAM
164/// historical store.
165///
166/// All counts come from `HistoricalStorage::stats()`, whose counters are
167/// maintained incrementally (Issue #212) — reading them is O(1) and never
168/// iterates versions.
169#[derive(Debug, Clone, Copy, PartialEq)]
170#[cfg_attr(feature = "serde", derive(serde::Serialize))]
171#[non_exhaustive]
172pub struct HistoricalDepthStats {
173    /// Total node versions retained in the hot historical store.
174    pub total_node_versions: usize,
175    /// Total edge versions retained in the hot historical store.
176    pub total_edge_versions: usize,
177    /// Number of distinct nodes that have version history.
178    pub unique_nodes: usize,
179    /// Number of distinct edges that have version history.
180    pub unique_edges: usize,
181    /// Total anchor versions (node + edge). Anchors store full property
182    /// snapshots; a lower anchor share means better delta compression.
183    /// Always: `anchor_count + delta_count ==
184    /// total_node_versions + total_edge_versions`.
185    pub anchor_count: usize,
186    /// Total delta versions (node + edge). Deltas store only changes
187    /// relative to the previous version.
188    pub delta_count: usize,
189    /// Anchor versions among node versions only.
190    pub node_anchor_count: usize,
191    /// Delta versions among node versions only.
192    pub node_delta_count: usize,
193    /// Anchor versions among edge versions only.
194    pub edge_anchor_count: usize,
195    /// Delta versions among edge versions only.
196    pub edge_delta_count: usize,
197    /// Anchors as a fraction of total versions (0.0–1.0; 1.0 when empty).
198    /// Lower is better compression. Computed from the cached counters above.
199    pub compression_ratio: f64,
200}

202/// Cold-tier (disk) storage state.
203///
204/// When the tier is disabled this serializes as exactly `{"enabled": false}`
205/// — count fields are structurally absent so they can never be misread as
206/// "zero cold versions".
207#[derive(Debug, Clone, Copy, PartialEq)]
208#[cfg_attr(feature = "serde", derive(serde::Serialize))]
209#[non_exhaustive]
210pub struct ColdStorageTierStats {
211    /// Whether a cold-storage tier (tiered storage + Redb backend) is
212    /// configured for this database.
213    pub enabled: bool,
214    /// Counters for the enabled tier; absent when `enabled` is false.
215    #[cfg_attr(feature = "serde", serde(flatten))]
216    pub details: Option<ColdStorageDetails>,
217}

219/// Counters for an enabled cold-storage tier. All values are atomic counter
220/// snapshots — O(1) reads, no disk access.
221#[derive(Debug, Clone, Copy, PartialEq)]
222#[cfg_attr(feature = "serde", derive(serde::Serialize))]
223#[non_exhaustive]
224pub struct ColdStorageDetails {
225    /// Node versions in the cold (disk) tier. Seeded from the persisted
226    /// tables when the database is opened (an O(1) metadata read), then
227    /// incremented per migration — a restarted process reports its full
228    /// on-disk cold history, not zero.
229    pub node_versions_stored: u64,
230    /// Edge versions in the cold (disk) tier. Seeded from the persisted
231    /// tables at open, like [`node_versions_stored`](Self::node_versions_stored).
232    pub edge_versions_stored: u64,
233    /// Cold-tier write compression ratio (raw bytes / compressed bytes).
234    /// Higher is better. The underlying byte counters are **not** persisted,
235    /// so this covers writes since the current process opened the database
236    /// (1.0 when this process has not written anything yet).
237    pub compression_ratio: f64,
238    /// Distribution of historical-version reads across storage tiers,
239    /// counted since the current process opened the database (access
240    /// counters are not persisted).
241    pub tier_access: TierAccessStats,
242}

244/// Where historical-version reads were served from. Together these reveal
245/// the hot / warm-cache / cold distribution of temporal query traffic.
246#[derive(Debug, Clone, Copy, PartialEq, Eq)]
247#[cfg_attr(feature = "serde", derive(serde::Serialize))]
248#[non_exhaustive]
249pub struct TierAccessStats {
250    /// Reads served from the hot (in-RAM) tier.
251    pub hot_hits: u64,
252    /// Reads served from the warm (LRU cache) tier.
253    pub warm_hits: u64,
254    /// Reads served from the cold (disk) tier.
255    pub cold_hits: u64,
256    /// Reads that found the version in no tier.
257    pub misses: u64,
258}

260/// Write-ahead-log durability state.
261///
262/// AletheiaDB currently always runs with a WAL (there is no no-WAL
263/// construction path), so `enabled` is always `true` today. The flag is
264/// still emitted explicitly so the schema contract covers any future no-WAL
265/// build and a consumer never has to infer WAL presence from other fields.
266#[derive(Debug, Clone, PartialEq, Eq)]
267#[cfg_attr(feature = "serde", derive(serde::Serialize))]
268#[non_exhaustive]
269pub struct WalStateStats {
270    /// Whether a WAL is active. Always `true` in current builds (see type
271    /// docs); a hypothetical no-WAL build would report `false` here instead
272    /// of omitting the object.
273    pub enabled: bool,
274    /// Active durability mode as a stable token: `"synchronous"`,
275    /// `"async"`, `"group_commit"`, or `"async_batched"`.
276    pub durability_mode: String,
277    /// The **next** log sequence number to be allocated (one past the most
278    /// recently assigned LSN; starts at 1 on an empty log). Atomic read.
279    pub current_lsn: u64,
280    /// Total WAL entries appended since startup. Atomic read.
281    pub total_appends: u64,
282    /// `true` when the last WAL flush succeeded (no outstanding
283    /// consecutive flush errors). Atomic read.
284    pub healthy: bool,
285}
286
287/// Stable, machine-readable token for a durability mode.
288///
289/// Deliberately NOT the `Debug` representation, so the MCP/API contract
290/// cannot drift if variant fields change.
291fn durability_mode_token(mode: DurabilityMode) -> &'static str {
292    match mode {
293        DurabilityMode::Synchronous => "synchronous",
294        DurabilityMode::Async { .. } => "async",
295        DurabilityMode::GroupCommit { .. } => "group_commit",
296        DurabilityMode::AsyncBatched { .. } => "async_batched",
297    }
298}

300impl AletheiaDB {
301    /// Get a holistic, serializable snapshot of database statistics:
302    /// current-state size, bi-temporal depth (with anchor/delta compression
303    /// effectiveness), storage-tier presence/distribution, and WAL state.
304    ///
305    /// This is the aggregator behind the MCP `database_stats` tool. It adds
306    /// no storage logic of its own — every field is read from an existing
307    /// O(1)/cached counter (see the [module docs](self) for per-field
308    /// provenance), so this call never scans versions and completes in
309    /// microseconds regardless of database size.
310    ///
311    /// # Lock behavior
312    ///
313    /// Follows the write-path lock acquisition order (`wal` before
314    /// `historical`): WAL fields are lock-free atomic reads, then the
315    /// `historical` read lock is taken briefly for the cached counters and
316    /// released *before* the tiered/cold-storage counters are read (via the
317    /// cloned tiered-storage handle). No adjacency or ID-generator locks are
318    /// touched.
319    ///
320    /// # Example
321    ///
322    /// ```rust
323    /// # use aletheiadb::AletheiaDB;
324    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
325    /// let db = AletheiaDB::new()?;
326    /// let stats = db.stats();
327    /// assert_eq!(stats.current.node_count, 0);
328    /// assert!(!stats.cold_storage.enabled); // disabled, not "0 versions"
329    /// assert!(stats.wal.enabled);
330    /// # Ok(())
331    /// # }
332    /// ```
333    #[must_use = "the statistics snapshot should be used"]
334    pub fn stats(&self) -> DatabaseStats {
335        // Current-state counts: lock-free index length reads (not part of
336        // the ordered write-path primitives).
337        let current_stats = self.current.stats();
338
339        // WAL (lock order #2): field copy + atomic loads, acquired before
340        // `historical` per the lock-order contract.
341        let wal = WalStateStats {
342            enabled: true,
343            durability_mode: durability_mode_token(self.wal.durability_mode()).to_string(),
344            current_lsn: self.wal.current_lsn().0,
345            total_appends: self.wal.total_appends(),
346            healthy: self.wal.is_healthy(),
347        };
348
349        // Historical (lock order #3): take the read guard only long enough
350        // to snapshot the cached counters and clone the tiered-storage
351        // handle; tiered/cold counters are read after the guard is dropped.
352        let (hist, tiered) = {
353            let historical = self.historical.read();
354            (historical.stats(), historical.tiered_storage_arc())
355        };
356
357        let historical = HistoricalDepthStats {
358            total_node_versions: hist.total_node_versions,
359            total_edge_versions: hist.total_edge_versions,
360            unique_nodes: hist.unique_nodes,
361            unique_edges: hist.unique_edges,
362            anchor_count: hist.node_anchor_count + hist.edge_anchor_count,
363            delta_count: hist.node_delta_count + hist.edge_delta_count,
364            node_anchor_count: hist.node_anchor_count,
365            node_delta_count: hist.node_delta_count,
366            edge_anchor_count: hist.edge_anchor_count,
367            edge_delta_count: hist.edge_delta_count,
368            compression_ratio: hist.compression_ratio(),
369        };
370
371        let cold_storage = match tiered {
372            Some(tiered) => {
373                let metrics = tiered.metrics();
374                let cold = tiered.cold_stats();
375                ColdStorageTierStats {
376                    enabled: true,
377                    details: Some(ColdStorageDetails {
378                        node_versions_stored: cold.node_versions_stored,
379                        edge_versions_stored: cold.edge_versions_stored,
380                        compression_ratio: cold.compression_ratio(),
381                        tier_access: TierAccessStats {
382                            hot_hits: metrics.hot_hits,
383                            warm_hits: metrics.warm_hits,
384                            cold_hits: metrics.cold_hits,
385                            misses: metrics.misses,
386                        },
387                    }),
388                }
389            }
390            None => ColdStorageTierStats {
391                enabled: false,
392                details: None,
393            },
394        };
395
396        // Provenance hash chain (Issue #3351): O(1) reads of the in-memory head.
397        let chain = match self.chain.as_ref() {
398            Some(chain) => {
399                let head = chain.head();
400                ProvenanceChainStats {
401                    enabled: true,
402                    head_seq: Some(head.seq),
403                    head_digest: Some(crate::provenance_chain::to_hex(&head.digest)),
404                    genesis_digest: Some(crate::provenance_chain::to_hex(
405                        &chain.genesis().genesis_digest,
406                    )),
407                    last_verified: chain.last_verified().map(|lv| LastVerifiedStats {
408                        passed: lv.passed,
409                        at_micros: lv.at_micros,
410                    }),
411                }
412            }
413            None => ProvenanceChainStats {
414                enabled: false,
415                head_seq: None,
416                head_digest: None,
417                genesis_digest: None,
418                last_verified: None,
419            },
420        };
421
422        // Per-namespace counts (Issue #3349): O(1) membership-index reads.
423        let namespaces = self.namespace_counts();
424
425        // Changefeed subscription state (Issue #3678): a brief registry read,
426        // O(live subscriptions). The per-principal breakdown is the authenticated
427        // surface for the fairness quota — never a `/metrics` label, and its
428        // identity roster is admin-gated at the MCP/HTTP `database_stats`
429        // handlers. Both the aggregate total and the per-principal rows are read
430        // under a SINGLE registry read lock so the snapshot cannot straddle two
431        // instants under concurrent subscribe/deregister.
432        let (active_subscriptions, per_principal_counts) =
433            self.changefeed.subscription_count_and_per_principal();
434        let changefeed = ChangefeedStats {
435            active_subscriptions,
436            per_principal: per_principal_counts
437                .into_iter()
438                .map(
439                    |(principal, active_subscriptions)| PrincipalSubscriptionStat {
440                        principal,
441                        active_subscriptions,
442                    },
443                )
444                .collect(),
445        };
446
447        DatabaseStats {
448            current: CurrentStateStats {
449                node_count: current_stats.node_count,
450                edge_count: current_stats.edge_count,
451            },
452            historical,
453            cold_storage,
454            wal,
455            chain,
456            namespaces,
457            changefeed,
458        }
459    }
460}

466    #[test]
467    fn durability_mode_tokens_are_stable() {
468        assert_eq!(
469            durability_mode_token(DurabilityMode::Synchronous),
470            "synchronous"
471        );
472        assert_eq!(
473            durability_mode_token(DurabilityMode::Async {
474                flush_interval_ms: 10
475            }),
476            "async"
477        );
478        assert_eq!(
479            durability_mode_token(DurabilityMode::GroupCommit {
480                max_delay_ms: 10,
481                max_batch_size: 200
482            }),
483            "group_commit"
484        );
485        assert_eq!(
486            durability_mode_token(DurabilityMode::AsyncBatched {
487                max_delay_ms: 10,
488                max_batch_size: 200
489            }),
490            "async_batched"
491        );
492    }

494    /// WAL counters are `u64`; extreme values must survive JSON
495    /// serialization without precision loss or sign flips.
496    #[cfg(feature = "serde")]
497    #[test]
498    fn wal_stats_u64_max_survives_json_round_trip() {
499        let stats = WalStateStats {
500            enabled: true,
501            durability_mode: "synchronous".to_string(),
502            current_lsn: u64::MAX,
503            total_appends: u64::MAX,
504            healthy: true,
505        };
506        let value = serde_json::to_value(&stats).unwrap();
507        assert_eq!(value["current_lsn"].as_u64(), Some(u64::MAX));
508        assert_eq!(value["total_appends"].as_u64(), Some(u64::MAX));
509
510        // Through text and back: serde_json must keep full u64 precision
511        // (arbitrary-precision integer handling, not f64).
512        let text = serde_json::to_string(&stats).unwrap();
513        let parsed: serde_json::Value = serde_json::from_str(&text).unwrap();
514        assert_eq!(parsed["current_lsn"].as_u64(), Some(u64::MAX));
515        assert_eq!(parsed["total_appends"].as_u64(), Some(u64::MAX));
516    }
