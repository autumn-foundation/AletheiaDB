16    /// Create a portable backup artifact at `path`.
17    ///
18    /// The artifact is a single self-contained file containing the complete
19    /// bi-temporal state: all current nodes/edges, all version history (including
20    /// cold-tier versions), and the string interner table.
21    ///
22    /// The file is written atomically (temp → rename) so an interrupted backup
23    /// never leaves a partial artifact at `path`.
24    ///
25    /// # Consistency
26    ///
27    /// Takes an Arc-COW snapshot at the current WAL LSN so that the backup
28    /// represents a single consistent point in time — no concurrent write
29    /// can appear partially in the artifact.
30    ///
31    /// # Errors
32    ///
33    /// Returns `Error::Backup` on any serialization or I/O failure.
34    pub fn backup(&self, path: &Path) -> Result<BackupSummary> {
35        let source_lsn = self.wal.current_lsn().0;
36
37        // Take consistent point-in-time snapshots.
38        //
39        // Issue #3425: the current-storage snapshot is taken INSIDE the
40        // `historical.read()` guard (and under `snapshot_lock`, mirroring
41        // `create_checkpoint`'s locking contract) rather than before/outside it.
42        // A committing writer holds `historical.write()` across its
43        // `commit_timestamp` finalization, so acquiring `historical.read()` here
44        // excludes that window: without this, the backup could clone a committed
45        // node/edge whose `commit_timestamp` is still `None` (an in-memory
46        // finalize-window artifact). Lock order is preserved
47        // (`historical` -> `snapshot_lock` -> DashMap shards); `create_snapshot`
48        // takes no `historical`/`wal`/`current_timestamp` lock, so no inversion.
49        let (current_snapshot, historical_snapshot, tiered_arc) = {
50            let hist = self.historical.read();
51            let current_snapshot = {
52                let _snap_lock = self.current.snapshot_lock.write();
53                self.current
54                    .create_snapshot(crate::storage::wal::LSN(source_lsn))
55            };
56            let historical_snapshot = hist.create_snapshot(crate::storage::wal::LSN(source_lsn));
57            let tiered = hist.tiered_storage_arc();
58            (current_snapshot, historical_snapshot, tiered)
59        };
60
61        // Test-only seam (Issue #3425): lets the backup-path regression test inspect
62        // the exact current-storage snapshot this backup captured, so it can prove
63        // the snapshot never contains a committed node with `commit_timestamp: None`
64        // even when a concurrent writer is parked in the finalize window. Production
65        // builds compile this away.
66        #[cfg(test)]
67        backup_test_hooks::run_post_current_snapshot_hook(&current_snapshot);
68
69        // Scan cold-tier versions outside the historical lock (disk I/O).
70        let (cold_node_versions, cold_edge_versions) = if let Some(tiered) = tiered_arc {
71            let cold_nodes = tiered.scan_node_versions_cold().map_err(|e| {
72                Error::Backup(BackupError::Io(format!("cold node scan failed: {e}")))
73            })?;
74            let cold_edges = tiered.scan_edge_versions_cold().map_err(|e| {
75                Error::Backup(BackupError::Io(format!("cold edge scan failed: {e}")))
76            })?;
77            (cold_nodes, cold_edges)
78        } else {
79            (Vec::new(), Vec::new())
80        };
81
82        // Timestamp in microseconds — use system clock for metadata only.
83        let created_at_micros = std::time::SystemTime::now()
84            .duration_since(std::time::UNIX_EPOCH)
85            .unwrap_or_default()
86            .as_micros() as i64;
87
88        // Capture declared schema constraints (Issue #3378) and uniqueness
89        // constraints (Issue #3218) so both survive the backup→restore
90        // round-trip. Uniqueness constraints are otherwise only WAL-persisted,
91        // so a fresh-WAL restore would silently drop them; they are captured
92        // here and re-declared on restore (see `reapply_unique_constraints`),
93        // which rebuilds the reservation index and writes a durable WAL record.
94        let schema_constraints = self.constraint_registry.export_schema_constraints();
95        let unique_constraints = self.constraint_registry.export_unique_constraints();
96
97        // Capture the crypto-shred subject keyring / designation registry
98        // (Issue #3665) so designations, erased-state, `erased_at`, and
99        // attestations travel inside the archive. The exported bytes are the
100        // exact CRC-wrapped sidecar wire form (empty when crypto-shred is
101        // unused). An erased subject's wrapped DEK is already absent from these
102        // bytes, so a post-erasure backup stays erased. When the `audit-export`
103        // feature is off there is no crypto-shred state, so the field is empty.
104        #[cfg(feature = "audit-export")]
105        let keyring_sidecar = self.crypto_shred.export_sidecar_bytes();
106        #[cfg(not(feature = "audit-export"))]
107        let keyring_sidecar = Vec::new();
108
109        let payload = build_payload(
110            current_snapshot,
111            historical_snapshot,
112            cold_node_versions,
113            cold_edge_versions,
114            source_lsn,
115            created_at_micros,
116            schema_constraints,
117            unique_constraints,
118            keyring_sidecar,
119        )
120        .map_err(Error::Backup)?;
121
122        let summary = BackupSummary {
123            node_versions: payload.node_version_count,
124            edge_versions: payload.edge_version_count,
125            current_node_count: payload.current_node_count,
126            current_edge_count: payload.current_edge_count,
127            bytes_written: 0, // filled in after write
128            source_lsn,
129        };
130
131        let bytes_written = write_artifact(&payload, path).map_err(Error::Backup)?;
132
133        Ok(BackupSummary {
134            bytes_written,
135            ..summary
136        })
137    }

139    /// Restore a backup artifact into a fresh **ephemeral** (in-memory) database.
140    ///
141    /// The restored database behaves exactly like one created with `AletheiaDB::new()`.
142    /// All bi-temporal history is reconstructed from the artifact.
143    ///
144    /// # Errors
145    ///
146    /// - `Error::Backup(BackupError::BadMagic)` — file is not a valid backup artifact.
147    /// - `Error::Backup(BackupError::IncompatibleVersion { .. })` — artifact format too new.
148    /// - `Error::Backup(BackupError::Corrupt(_))` — corrupted or truncated data.
149    pub fn restore(path: &Path) -> Result<AletheiaDB> {
150        let payload = read_artifact(path).map_err(Error::Backup)?;
151
152        // Materialise into a fresh temp dir, then open via the normal startup path.
153        let tmp =
154            tempfile::TempDir::new().map_err(|e| Error::Backup(BackupError::Io(e.to_string())))?;
155
156        materialize_to_dir(&payload, tmp.path()).map_err(Error::Backup)?;
157
158        // Restore schema constraints (Issue #3378) by writing the sidecar into
159        // the data dir (parent of the reopen WAL dir = tmp root), so the
160        // reopen below loads them through the normal startup sidecar path.
161        crate::db::schema_constraint::persist_descriptors_to_dir(
162            tmp.path(),
163            &payload.schema_constraints,
164        )?;
165
166        // Restore the crypto-shred subject keyring / designation registry
167        // (Issue #3665) into the SAME data dir the schema-constraint sidecar
168        // went to (the parent of the reopen WAL dir = tmp root), so the reopen
169        // below loads it via the fail-closed `CryptoShredState::open` path.
170        #[cfg(feature = "audit-export")]
171        restore_keyring_sidecar(tmp.path(), &payload)?;
172
173        // Use an isolated WAL dir inside the temp dir to avoid cross-test contamination
174        // from the default "aletheiadb/wal" path.
175        let config = build_restore_config(tmp.path(), tmp.path().join("wal"));
176        let mut db = AletheiaDB::with_unified_config(config)?;
177
178        // Re-declare uniqueness constraints (Issue #3218) after the data is
179        // loaded: unlike the schema-constraint sidecar (loaded during startup
180        // above), uniqueness constraints are WAL-persisted, so they must be
181        // re-declared against the reopened DB — which rebuilds the reservation
182        // index from the restored nodes and writes a durable WAL record.
183        reapply_unique_constraints(&db, &payload.unique_constraints)?;
184
185        // Keep the temp dir alive for the lifetime of the ephemeral DB.
186        db._tempdir = Some(tmp);
187        Ok(db)
188    }

190    /// Restore a backup artifact into a **durable** database at `data_dir`.
191    ///
192    /// The target directory must be empty (no index manifest present) to
193    /// prevent overwriting existing data.
194    ///
195    /// After restoration, the DB is persisted under `data_dir` in the canonical
196    /// durable layout and can be reopened with either
197    /// [`AletheiaDB::open`]`(data_dir)` /
198    /// [`AletheiaDB::open_from_env`] or the equivalent
199    /// `AletheiaDB::with_unified_config(durable_config_for_data_dir(data_dir))`.
200    ///
201    /// The restored index artifacts are materialised at the same on-disk depth
202    /// that [`crate::config::durable_config_for_data_dir`] reads them from: the
203    /// index-persistence root is `data_dir/indexes` (the value that config
204    /// assigns to `PersistenceConfig::data_dir`), and the
205    /// `IndexPersistenceManager` appends its own `indexes/` beneath that base.
206    /// Writing at any other depth (e.g. using `data_dir` directly as the base)
207    /// leaves the manifest where `open`/`open_from_env` cannot find it, so a
208    /// reopen silently comes up empty.
209    ///
210    /// # Errors
211    ///
212    /// - `Error::Backup(BackupError::TargetNotEmpty)` — target already has data.
213    /// - `Error::Backup(BackupError::BadMagic)` — invalid artifact.
214    /// - `Error::Backup(BackupError::IncompatibleVersion { .. })` — format too new.
215    pub fn restore_to_data_dir(path: &Path, data_dir: &Path) -> Result<AletheiaDB> {
216        // Canonical index-persistence root: the same base that
217        // `durable_config_for_data_dir` assigns to `PersistenceConfig::data_dir`
218        // (the `IndexPersistenceManager` appends its own `indexes/` under this).
219        let index_root = index_persistence_root(data_dir);
220
221        check_target_empty(&index_root).map_err(Error::Backup)?;
222
223        let payload = read_artifact(path).map_err(Error::Backup)?;
224        materialize_to_dir(&payload, &index_root).map_err(Error::Backup)?;
225
226        // Restore schema constraints (Issue #3378): the durable reopen derives
227        // its data dir as the parent of its WAL dir (`data_dir/wal` → `data_dir`),
228        // so the sidecar must land at `data_dir`, not the index root.
229        crate::db::schema_constraint::persist_descriptors_to_dir(
230            data_dir,
231            &payload.schema_constraints,
232        )?;
233
234        // Restore the crypto-shred subject keyring / designation registry
235        // (Issue #3665) into `data_dir` (the parent of the durable reopen WAL
236        // dir, where `CryptoShredState::open` looks), NOT the index root.
237        #[cfg(feature = "audit-export")]
238        restore_keyring_sidecar(data_dir, &payload)?;
239
240        // Reopen through the canonical durable config so the layout written above
241        // is exactly the layout `open`/`open_from_env` will read on restart.
242        let config = crate::config::durable_config_for_data_dir(data_dir);
243        let db = AletheiaDB::with_unified_config(config)?;
244
245        // Re-declare uniqueness constraints (Issue #3218). Because
246        // `enable_unique_constraint` appends a `DeclareUniqueConstraint` WAL
247        // record, the constraint is durable in the restored database and is
248        // recovered by normal WAL replay on a later reopen.
249        reapply_unique_constraints(&db, &payload.unique_constraints)?;
250        Ok(db)
251    }
252}

254/// Re-declare each restored uniqueness constraint (Issue #3218) against a
255/// freshly-reopened database.
256///
257/// `enable_unique_constraint` runs the duplicate pre-flight scan (a safety net
258/// — a consistent backup enforced uniqueness, so no duplicates should exist),
259/// rebuilds the reservation index from the restored nodes, and appends a
260/// durable `DeclareUniqueConstraint` WAL record. A dangling/duplicate
261/// declaration surfaces as a normal error rather than silently dropping the
262/// constraint.
263fn reapply_unique_constraints(
264    db: &AletheiaDB,
265    descriptors: &[crate::core::constraint::UniqueConstraintDescriptor],
266) -> Result<()> {
267    for d in descriptors {
268        db.enable_unique_constraint(&d.label, &d.property)?;
269    }
270    Ok(())
271}

273/// Write the crypto-shred subject keyring / designation registry (Issue #3665)
274/// from a restored payload into `data_dir` as `subject_keyring.dat`.
275///
276/// The payload's `keyring_sidecar` is the exact CRC-wrapped sidecar wire form,
277/// so it is written verbatim; the subsequent reopen loads it via the
278/// fail-closed `CryptoShredState::open`. No-op on an empty sidecar.
279///
280/// **Backward-compat (v5/v6 archives):** an archive taken before the keyring
281/// fold carries no keyring, yet may hold designated (SUBJ-sealed) property
282/// bytes. Those restore **sealed-unreadable** — without the keyring the
283/// per-subject DEKs are gone. We emit a single loud warning in that case so an
284/// operator is never silently surprised by unreadable designated properties.
285///
286/// **Write-side size cap (Issue #3665 hardening):** the payload's keyring
287/// sidecar is rejected here if it exceeds [`keyring::MAX_KEYRING_BYTES`], the
288/// same 64 MiB bound the fail-closed loader enforces. Without this symmetry a
289/// malicious v7 archive could force a multi-GiB `subject_keyring.dat` write
290/// before the loader ever rejects it (disk-exhaustion DoS).
291///
292/// **Stale-keyring removal (Issue #3665 hardening):** when the payload carries
293/// no keyring (a plain / keyless restore) but the target `data_dir` already
294/// holds a leftover `subject_keyring.dat`, that file is removed so the reopen
295/// adopts the restored state rather than the stale keyring.
296#[cfg(feature = "audit-export")]
297fn restore_keyring_sidecar(
298    data_dir: &Path,
299    payload: &crate::storage::backup::BackupPayload,
300) -> Result<()> {
301    use crate::db::crypto_shred::keyring;
302    if !payload.keyring_sidecar.is_empty() {
303        // Reject an over-cap sidecar BEFORE writing anything to disk — symmetry
304        // with the 64 MiB load cap (`keyring::MAX_KEYRING_BYTES`).
305        if payload.keyring_sidecar.len() as u64 > keyring::MAX_KEYRING_BYTES {
306            return Err(Error::Backup(BackupError::Corrupt(format!(
307                "keyring sidecar in backup exceeds size limit ({} bytes > {} byte cap)",
308                payload.keyring_sidecar.len(),
309                keyring::MAX_KEYRING_BYTES
310            ))));
311        }
312        let path = data_dir.join(keyring::KEYRING_FILENAME);
313        keyring::save_sidecar_bytes(&path, &payload.keyring_sidecar)
314            .map_err(|e| Error::Backup(BackupError::Io(e.to_string())))?;
315    } else {
316        // Keyless restore: drop any stale keyring left in the target so the
317        // restored (empty-crypto-shred) state wins over pre-existing state.
318        // Attempt the delete directly and ignore only NotFound — this avoids the
319        // redundant `exists()` check and the TOCTOU window it opens.
320        let stale = data_dir.join(keyring::KEYRING_FILENAME);
321        match std::fs::remove_file(&stale) {
322            Ok(()) => {}
323            Err(e) if e.kind() == std::io::ErrorKind::NotFound => {}
324            Err(e) => {
325                return Err(Error::Backup(BackupError::Io(format!(
326                    "failed to remove stale keyring sidecar: {e}"
327                ))));
328            }
329        }
330        if payload_has_sealed_properties(payload) {
331            let message = "crypto-shred: restoring a pre-v7 backup that contains \
332                           designated (sealed) properties but no subject keyring — \
333                           those properties will be sealed-unreadable (the archive \
334                           predates the Issue #3665 keyring fold). Restore from a v7 \
335                           (or newer) backup to recover them.";
336            #[cfg(feature = "observability")]
337            tracing::warn!("{message}");
338            #[cfg(not(feature = "observability"))]
339            eprintln!("WARNING: {message}");
340        }
341    }
342    Ok(())
343}

345/// Whether a restored payload's current-state graph holds any SUBJ-sealed
346/// (crypto-shred envelope) property value. Feature-independent scan by the
347/// public `SUBJ` envelope magic, used only to decide whether to warn on a
348/// keyring-less restore.
349#[cfg(feature = "audit-export")]
350fn payload_has_sealed_properties(payload: &crate::storage::backup::BackupPayload) -> bool {
351    use crate::storage::index_persistence::formats::PersistedPropertyValue;
352    let is_sealed = |props: &crate::storage::index_persistence::formats::PersistedPropertyMap| {
353        props.entries.iter().any(|(_, v)| {
354            matches!(v, PersistedPropertyValue::Bytes(b)
355                if crate::db::crypto_shred::envelope::is_envelope(b))
356        })
357    };
358    payload.graph.nodes.iter().any(|n| is_sealed(&n.properties))
359        || payload.graph.edges.iter().any(|e| is_sealed(&e.properties))
360}

362/// The index-persistence base directory under a durable data root.
363///
364/// This mirrors [`crate::config::durable_config_for_data_dir`], which sets
365/// `PersistenceConfig::data_dir = data_dir/indexes`. Keeping the join in one
366/// place ensures restore materialises artifacts at the exact depth the durable
367/// reopen path reads them from.
368fn index_persistence_root(data_dir: &Path) -> std::path::PathBuf {
369    data_dir.join("indexes")
370}

372/// Build an `AletheiaDBConfig` that loads from an existing persistence directory.
373///
374/// * `persistence_dir` — base dir for `IndexPersistenceManager` (contains `indexes/`).
375/// * `wal_dir` — WAL directory; if `None` a fresh in-memory-only WAL is used.
376fn build_restore_config(persistence_dir: &Path, wal_dir: std::path::PathBuf) -> AletheiaDBConfig {
377    use crate::config::WalConfigBuilder;
378
379    let wal_builder =
380        WalConfigBuilder::new()
381            .wal_dir(wal_dir)
382            .durability_mode(DurabilityMode::GroupCommit {
383                max_delay_ms: 10,
384                max_batch_size: 200,
385            });
386
387    AletheiaDBConfig::builder()
388        .wal(wal_builder.build())
389        .persistence(PersistenceConfig {
390            enabled: true,
391            data_dir: persistence_dir.to_path_buf(),
392            load_on_startup: true,
393            ..Default::default()
394        })
395        .build()
396}

398/// Test-only interleaving seam for the Issue #3425 backup-path regression test.
399///
400/// Lets a test observe the exact current-storage snapshot that a real `backup()`
401/// call captured. The hook fires immediately after the snapshot is taken (inside
402/// the `historical.read()` guard block), so a test can drive the real backup path
403/// concurrently with a writer parked in the finalize window and assert the
404/// captured snapshot never contains a committed node with `commit_timestamp: None`.
405/// Compiled away in non-test builds.
406#[cfg(test)]
407pub(crate) mod backup_test_hooks {
408    use crate::storage::snapshot::CurrentStorageSnapshot;
409    use std::sync::{Arc, Mutex};
410
411    type Hook = Arc<dyn Fn(&CurrentStorageSnapshot) + Send + Sync>;
412
413    static POST_CURRENT_SNAPSHOT_HOOK: Mutex<Option<Hook>> = Mutex::new(None);
414
415    /// Install a hook fired just after `backup()` captures its current snapshot.
416    pub(crate) fn set_post_current_snapshot_hook(hook: Hook) {
417        *POST_CURRENT_SNAPSHOT_HOOK
418            .lock()
419            .expect("backup post-snapshot hook mutex poisoned") = Some(hook);
420    }
421
422    /// Remove any installed post-snapshot hook.
423    pub(crate) fn clear_post_current_snapshot_hook() {
424        *POST_CURRENT_SNAPSHOT_HOOK
425            .lock()
426            .expect("backup post-snapshot hook mutex poisoned") = None;
427    }
428
429    /// Invoke the installed post-snapshot hook, if any.
430    ///
431    /// The `Arc` is cloned out from under the mutex so the hook body never runs
432    /// while holding the registry lock.
433    pub(crate) fn run_post_current_snapshot_hook(snapshot: &CurrentStorageSnapshot) {
434        let hook = POST_CURRENT_SNAPSHOT_HOOK
435            .lock()
436            .expect("backup post-snapshot hook mutex poisoned")
437            .clone();
438        if let Some(hook) = hook {
439            hook(snapshot);
440        }
441    }
442}

512    #[test]
513    fn payload_has_sealed_properties_true_for_subj_envelope() {
514        // A `SUBJ`-prefixed Bytes value is a crypto-shred sealed envelope.
515        let mut sealed = b"SUBJ".to_vec();
516        sealed.extend_from_slice(&[0x01, 0x02, 0x03, 0x04]);
517        let payload = payload_with_node_property(PersistedPropertyValue::Bytes(sealed));
518        assert!(
519            payload_has_sealed_properties(&payload),
520            "a SUBJ-enveloped Bytes property must be detected as sealed"

522    }
