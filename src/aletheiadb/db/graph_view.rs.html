1//! Implementation of the `GraphView` trait for `AletheiaDB`.
2//!
3//! This module adapts the main database struct to the `GraphView` trait,
4//! allowing it to be used in query execution and other components that
5//! require a generic view of the graph.
6//!
7//! The `GraphView` trait provides a unified interface for accessing graph elements
8//! (nodes and edges) both in the current state and at specific points in time.
9
10use crate::core::error::Result;
11use crate::core::graph::{Edge, Node};
12use crate::core::id::{EdgeId, NodeId};
13use crate::core::temporal::Timestamp;
14use crate::db::AletheiaDB;
15use crate::query::traits::GraphView;
16
17/// Implementation of `GraphView` for `AletheiaDB`.
18///
19/// This implementation delegates all calls to the underlying `AletheiaDB` methods,
20/// exposing the database's full capabilities through the `GraphView` interface.
21///
22/// # Examples
23///
24/// ```rust
25/// use aletheiadb::AletheiaDB;
26/// use aletheiadb::query::traits::GraphView;
27/// use aletheiadb::core::id::NodeId;
28///
29/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
30/// let db = AletheiaDB::new()?;
31///
32/// // Use the database as a GraphView
33/// let view: &dyn GraphView = &db;
34///
35/// // Access graph elements through the trait
36/// // let node = view.get_node(NodeId::new(1)?)?;
37/// # Ok(())
38/// # }
39/// ```
40impl GraphView for AletheiaDB {
41    fn get_node(&self, id: NodeId) -> Result<Node> {
42        self.get_node(id)
43    }
44
45    fn get_edge(&self, id: EdgeId) -> Result<Edge> {
46        self.get_edge(id)
47    }
48
49    fn get_edge_target(&self, edge_id: EdgeId) -> Result<NodeId> {
50        self.get_edge_target(edge_id)
51    }
52
53    fn get_edge_source(&self, edge_id: EdgeId) -> Result<NodeId> {
54        self.get_edge_source(edge_id)
55    }
56
57    fn get_outgoing_edges(&self, node_id: NodeId) -> Vec<EdgeId> {
58        self.get_outgoing_edges(node_id)
59    }
60
61    fn get_incoming_edges(&self, node_id: NodeId) -> Vec<EdgeId> {
62        self.get_incoming_edges(node_id)
63    }
64
65    fn get_outgoing_edges_with_label(&self, node_id: NodeId, label: &str) -> Vec<EdgeId> {
66        self.get_outgoing_edges_with_label(node_id, label)
67    }
68
69    fn get_node_at_time(
70        &self,
71        node_id: NodeId,
72        valid_time: Timestamp,
73        transaction_time: Timestamp,
74    ) -> Result<Node> {
75        self.get_node_at_time(node_id, valid_time, transaction_time)
76    }
77
78    fn get_edge_at_time(
79        &self,
80        edge_id: EdgeId,
81        valid_time: Timestamp,
82        transaction_time: Timestamp,
83    ) -> Result<Edge> {
84        self.get_edge_at_time(edge_id, valid_time, transaction_time)
85    }
86
87    fn get_outgoing_edges_at_time(
88        &self,
89        source: NodeId,
90        valid_time: Timestamp,
91        tx_time: Timestamp,
92    ) -> Vec<EdgeId> {
93        self.get_outgoing_edges_at_time(source, valid_time, tx_time)
94    }
95
96    fn get_incoming_edges_at_time(
97        &self,
98        target: NodeId,
99        valid_time: Timestamp,
100        tx_time: Timestamp,
101    ) -> Vec<EdgeId> {
102        self.get_incoming_edges_at_time(target, valid_time, tx_time)
103    }
104
105    fn find_similar_as_of(
106        &self,
107        embedding: &[f32],
108        k: usize,
109        timestamp: Timestamp,
110    ) -> Result<Vec<(NodeId, f32)>> {
111        self.similarity_search(
112            crate::SimilarityQuery::from_embedding(embedding)
113                .k(k)
114                .at_time(timestamp),
115        )
116    }
117}
