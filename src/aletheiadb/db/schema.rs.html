1//! Graph schema discovery: node labels, edge types, and property keys.
2//!
3//! Provides a read-only summary of the graph's shape -- the labels and edge
4//! types present, how many entities have each, and which property keys are
5//! observed -- so callers (notably LLM/MCP integrators) can discover what is
6//! queryable without already knowing label/property names. See
7//! [`AletheiaDB::schema`] for the current-state summary and
8//! [`AletheiaDB::schema_as_of`] for the bi-temporal variant.
9

19/// Resolve an interned label/key to an owned `String`, falling back to empty
20/// if the interner entry is somehow missing (cannot happen in practice,
21/// since labels and property keys are interned before being stored on a
22/// node/edge).
23fn resolve_label(label: InternedString) -> String {
24    GLOBAL_INTERNER.resolve_or_else(label, String::new)
25}

27/// A point in bi-temporal space that a [`GraphSchema`] was computed as of.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub struct SchemaInstant {
30    /// The valid-time instant the schema reflects.
31    pub valid_time: Timestamp,
32    /// The transaction-time instant the schema reflects.
33    pub transaction_time: Timestamp,
34}

36/// Schema summary for a single node label.
37#[derive(Debug, Clone, PartialEq, Eq)]
38pub struct LabelSchema {
39    /// The node label.
40    pub label: String,
41    /// Number of nodes with this label.
42    pub count: usize,
43    /// Union of property keys **observed** on nodes with this label, sorted.
44    /// This is a discovered set — distinct from `declared_constraints`, which
45    /// reports keys with a *declared* schema constraint (Issue #3378). A key
46    /// can be declared but not yet observed, or observed but not declared.
47    pub property_keys: Vec<String>,
48    /// Schema constraints **declared** on this label (Issue #3378), empty when
49    /// the label is schemaless. Lets a caller distinguish declared keys from
50    /// merely-observed ones.
51    pub declared_constraints: Vec<PropertyConstraintDescriptor>,
52}

54/// Schema summary for a single edge/relationship type.
55#[derive(Debug, Clone, PartialEq, Eq)]
56pub struct EdgeTypeSchema {
57    /// The edge/relationship type.
58    pub edge_type: String,
59    /// Number of edges with this type.
60    pub count: usize,
61    /// Union of property keys **observed** on edges of this type, sorted.
62    pub property_keys: Vec<String>,
63    /// Schema constraints **declared** on this edge type (Issue #3378), empty
64    /// when schemaless.
65    pub declared_constraints: Vec<PropertyConstraintDescriptor>,
66}

68/// A structured summary of the graph's shape: distinct node labels and
69/// edge/relationship types, their counts, and the property keys observed on
70/// each.
71///
72/// Returned by [`AletheiaDB::schema`] (current state) and
73/// [`AletheiaDB::schema_as_of`] (bi-temporal). Property-key enumeration
74/// within a summarized label/type is always exhaustive; `sampled` instead
75/// reports whether the *set of entities summarized* was truncated (see
76/// [`AletheiaDB::schema_as_of`]'s entity cap) -- any such truncation is
77/// always disclosed via this flag, never silent.
78#[derive(Debug, Clone, PartialEq, Eq)]
79pub struct GraphSchema {
80    /// Distinct node labels present, sorted by label.
81    pub node_labels: Vec<LabelSchema>,
82    /// Distinct edge/relationship types present, sorted by type.
83    pub edge_types: Vec<EdgeTypeSchema>,
84    /// Total number of nodes summarized.
85    pub total_nodes: usize,
86    /// Total number of edges summarized.
87    pub total_edges: usize,
88    /// Whether the summarized entity set was truncated (see
89    /// [`AletheiaDB::schema_as_of`]'s entity cap). Always `false` for
90    /// [`AletheiaDB::schema`], which is always exhaustive.
91    pub sampled: bool,
92    /// The bi-temporal instant this schema reflects, or `None` for current state.
93    pub as_of: Option<SchemaInstant>,
94    /// Per-namespace current node/edge counts (Issue #3349). Populated for the
95    /// current-state [`AletheiaDB::schema`]; **empty** for
96    /// [`AletheiaDB::schema_as_of`] (the namespace membership index is a
97    /// current-state acceleration structure, so a point-in-time namespace
98    /// breakdown is a deliberate follow-up).
99    pub namespaces: Vec<crate::db::namespace_query::NamespaceCount>,
100}

102/// Accumulates per-label/per-type counts and the union of property keys
103/// seen, keyed by `InternedString` rather than resolving to `String` per
104/// entity -- resolution happens once per *distinct* label/key in
105/// [`build_schema`] instead of once per node/edge, avoiding an interner
106/// lookup + allocation for every property of every entity.
107struct Accumulator {
108    entries: BTreeMap<InternedString, (usize, BTreeSet<InternedString>)>,
109}
110
111impl Accumulator {
112    fn new() -> Self {
113        Self {
114            entries: BTreeMap::new(),
115        }
116    }
117
118    fn record(&mut self, label: InternedString, properties: &PropertyMap) {
119        let entry = self
120            .entries
121            .entry(label)
122            .or_insert_with(|| (0, BTreeSet::new()));
123        entry.0 += 1;
124        entry.1.extend(properties.keys().copied());
125    }
126}

128impl AletheiaDB {
129    /// Discover the graph's current schema: distinct node labels and edge
130    /// types, their counts, and the property keys observed on each.
131    ///
132    /// Returns an empty-but-well-formed [`GraphSchema`] on an empty database,
133    /// never an error, and is always exhaustive (`sampled` is always
134    /// `false`). Counts reflect a best-effort read over live storage: as
135    /// with every other full-scan method on this storage layer (e.g.
136    /// `label_counts`), there is no cross-call snapshot isolation, so under
137    /// concurrent writes the totals may not exactly match a `node_count`/
138    /// `edge_count` call made at a slightly different instant.
139    ///
140    /// # Example
141    ///
142    /// ```rust,no_run
143    /// # use aletheiadb::AletheiaDB;
144    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
145    /// let db = AletheiaDB::new()?;
146    /// let schema = db.schema()?;
147    /// for label in &schema.node_labels {
148    ///     println!("{}: {} nodes", label.label, label.count);
149    /// }
150    /// # Ok(())
151    /// # }
152    /// ```
153    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
154    pub fn schema(&self) -> Result<GraphSchema> {
155        let mut node_acc = Accumulator::new();
156        self.current
157            .visit_nodes(|node| node_acc.record(node.label, &node.properties));
158
159        let mut edge_acc = Accumulator::new();
160        self.current
161            .visit_edges(|edge| edge_acc.record(edge.label, &edge.properties));
162
163        Ok(build_schema(
164            node_acc,
165            edge_acc,
166            false,
167            None,
168            &self.constraint_registry,
169            self.namespace_counts(),
170        ))
171    }
172
173    /// Discover the graph's schema as it existed at a specific bi-temporal
174    /// instant: the node labels and edge types visible at
175    /// `(valid_time, transaction_time)`, with counts and property keys
176    /// reflecting only the entities visible at that instant.
177    ///
178    /// A label/type that had not yet been written as of the given instant
179    /// (in either temporal dimension) is simply absent from the result --
180    /// there is no error case for "nothing existed yet".
181    ///
182    /// The set of candidate entities (every node/edge that has ever had a
183    /// version recorded) is capped per entity kind (see
184    /// [`crate::config::HistoricalConfigBuilder::max_schema_as_of_entities`],
185    /// default 50,000) to keep this call bounded on databases with
186    /// substantial bi-temporal history; when the cap is hit,
187    /// [`GraphSchema::sampled`] is set to `true` to disclose the truncation.
188    ///
189    /// # Example
190    ///
191    /// ```rust,no_run
192    /// # use aletheiadb::AletheiaDB;
193    /// # use aletheiadb::core::temporal::time;
194    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
195    /// let db = AletheiaDB::new()?;
196    /// let now = time::now();
197    /// let schema = db.schema_as_of(now, now)?;
198    /// # Ok(())
199    /// # }
200    /// ```
201    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
202    pub fn schema_as_of(
203        &self,
204        valid_time: Timestamp,
205        transaction_time: Timestamp,
206    ) -> Result<GraphSchema> {
207        let (node_ids, edge_ids, sampled) = {
208            let historical = self.historical.read();
209            let mut node_ids = historical.versioned_node_ids();
210            let mut edge_ids = historical.versioned_edge_ids();
211            let cap = historical.max_schema_as_of_entities();
212            drop(historical);
213
214            let node_sampled = cap_ids(&mut node_ids, cap);
215            let edge_sampled = cap_ids(&mut edge_ids, cap);
216            (node_ids, edge_ids, node_sampled || edge_sampled)
217        };
218
219        let mut node_acc = Accumulator::new();
220        for (_, node) in self.get_nodes_at_time(&node_ids, valid_time, transaction_time)? {
221            if let Some(node) = node {
222                node_acc.record(node.label, &node.properties);
223            }
224        }
225
226        let mut edge_acc = Accumulator::new();
227        for (_, edge) in self.get_edges_at_time(&edge_ids, valid_time, transaction_time)? {
228            if let Some(edge) = edge {
229                edge_acc.record(edge.label, &edge.properties);
230            }
231        }
232
233        Ok(build_schema(
234            node_acc,
235            edge_acc,
236            sampled,
237            Some(SchemaInstant {
238                valid_time,
239                transaction_time,
240            }),
241            &self.constraint_registry,
242            // Namespace breakdown is current-state only; empty for AS OF.
243            Vec::new(),
244        ))
245    }
246}

248/// Resolve the declared schema constraints for `(kind, label)` into public
249/// descriptors, or an empty vec when the label is schemaless (Issue #3378).
250fn declared_for(
251    registry: &ConstraintRegistry,
252    kind: EntityKind,
253    label: InternedString,
254) -> Vec<PropertyConstraintDescriptor> {
255    match registry.schema_constraints_for(kind, label) {
256        None => Vec::new(),
257        Some(constraints) => constraints
258            .iter()
259            .filter_map(|c| {
260                let property = GLOBAL_INTERNER.resolve_with(c.property, |s| s.to_string())?;
261                Some(PropertyConstraintDescriptor {
262                    property,
263                    declared_type: c.declared_type,
264                    required: c.required,
265                    nullable: c.nullable,
266                })
267            })
268            .collect(),
269    }
270}

272/// Truncate `ids` down to the `cap` smallest elements (a deterministic set,
273/// though not necessarily in sorted order -- the order among the kept
274/// elements doesn't matter, since they're only used as a scan input, not
275/// exposed to callers). Returns `true` if truncation actually happened, so
276/// the caller can disclose it via [`GraphSchema::sampled`] (or
277/// [`crate::db::NodesAtTime::sampled`] on the AS OF node-find path, which
278/// shares this cap).
279///
280/// Uses a partial selection (`select_nth_unstable`, O(n) average) rather
281/// than a full sort (O(n log n)), since only membership in the smallest-`cap`
282/// set matters here, not a total order.
283pub(crate) fn cap_ids<T: Ord>(ids: &mut Vec<T>, cap: usize) -> bool {
284    if ids.len() > cap {
285        ids.select_nth_unstable(cap);
286        ids.truncate(cap);
287        true
288    } else {
289        false
290    }
291}

293fn build_schema(
294    node_acc: Accumulator,
295    edge_acc: Accumulator,
296    sampled: bool,
297    as_of: Option<SchemaInstant>,
298    registry: &ConstraintRegistry,
299    namespaces: Vec<crate::db::namespace_query::NamespaceCount>,
300) -> GraphSchema {
301    let mut total_nodes = 0;
302    let mut node_labels: Vec<LabelSchema> = node_acc
303        .entries
304        .into_iter()
305        .map(|(label, (count, keys))| {
306            total_nodes += count;
307            // Elide engine-reserved ride-along keys (the namespace marker,
308            // #3349, and crypto-shred markers): they are not user property keys
309            // and must never surface in the schema.
310            let mut property_keys: Vec<String> = keys
311                .into_iter()
312                .map(resolve_label)
313                .filter(|k| !crate::core::namespace::is_reserved_property_key(k))
314                .collect();
315            property_keys.sort_unstable();
316            LabelSchema {
317                label: resolve_label(label),
318                count,
319                property_keys,
320                declared_constraints: declared_for(registry, EntityKind::Node, label),
321            }
322        })
323        .collect();
324    node_labels.sort_unstable_by(|a, b| a.label.cmp(&b.label));
325
326    let mut total_edges = 0;
327    let mut edge_types: Vec<EdgeTypeSchema> = edge_acc
328        .entries
329        .into_iter()
330        .map(|(edge_type, (count, keys))| {
331            total_edges += count;
332            // Elide engine-reserved ride-along keys (see the node branch).
333            let mut property_keys: Vec<String> = keys
334                .into_iter()
335                .map(resolve_label)
336                .filter(|k| !crate::core::namespace::is_reserved_property_key(k))
337                .collect();
338            property_keys.sort_unstable();
339            EdgeTypeSchema {
340                edge_type: resolve_label(edge_type),
341                count,
342                property_keys,
343                declared_constraints: declared_for(registry, EntityKind::Edge, edge_type),
344            }
345        })
346        .collect();
347    edge_types.sort_unstable_by(|a, b| a.edge_type.cmp(&b.edge_type));
348
349    GraphSchema {
350        node_labels,
351        edge_types,
352        total_nodes,
353        total_edges,
354        sampled,
355        as_of,
356        namespaces,
357    }
358}

360#[cfg(test)]
361mod cap_ids_tests {
362    use super::cap_ids;
363
364    #[test]
365    fn under_cap_is_untouched_and_not_sampled() {
366        let mut ids = vec![5, 1, 3];
367        let sampled = cap_ids(&mut ids, 10);
368        assert!(!sampled);
369        assert_eq!(ids, vec![5, 1, 3], "no truncation, order preserved");
370    }
371
372    #[test]
373    fn exactly_at_cap_is_untouched_and_not_sampled() {
374        let mut ids = vec![5, 1, 3];
375        let sampled = cap_ids(&mut ids, 3);
376        assert!(!sampled);
377        assert_eq!(ids, vec![5, 1, 3]);
378    }
379
380    #[test]
381    fn over_cap_truncates_to_the_cap_smallest_elements_and_is_sampled() {
382        let mut ids = vec![9, 1, 5, 3, 7, 2];
383        let sampled = cap_ids(&mut ids, 3);
384        assert!(sampled);
385        assert_eq!(ids.len(), 3);
386        // select_nth_unstable only guarantees a correct partition, not a
387        // sorted order among the kept elements -- compare as a set.
388        let mut kept = ids.clone();
389        kept.sort_unstable();
390        assert_eq!(kept, vec![1, 2, 3], "kept elements are the 3 smallest");
391    }
392
393    #[test]
394    fn cap_of_zero_truncates_everything() {
395        let mut ids = vec![1, 2, 3];
396        let sampled = cap_ids(&mut ids, 0);
397        assert!(sampled);
398        assert!(ids.is_empty());
399    }
