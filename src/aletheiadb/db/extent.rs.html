1//! Queryable bi-temporal extent of the dataset (Issue #3238).
2//!
3//! AletheiaDB lets a caller ask `AS OF <t>`, but nothing tells the caller what
4//! time range the data actually covers: an `AS OF` before the earliest
5//! recorded valid time returns an empty result that is indistinguishable from
6//! "nothing existed then". [`AletheiaDB::temporal_extent`] closes that gap by
7//! reporting the earliest and latest valid-time and transaction-time
8//! coordinates observed across recorded history — including
9//! expired/superseded versions and delete tombstones — so a caller (notably
10//! an LLM over MCP) can calibrate `AS OF` queries to land inside real data.
11//!
12//! # Semantics
13//!
14//! - **Coverage**: bounds are computed over every version recorded during
15//!   the current process lifetime, plus the hot-tier history restored at
16//!   startup — not just the current state. A fact written for 2019 and
17//!   later corrected still counts toward `earliest`. This is a calendar
18//!   *range*, not a current-state count.
19//! - **Cold storage (Issue #3389)**: on databases with cold-storage migration
20//!   enabled, versions migrated to the cold tier are also reflected across
21//!   restarts. The cold store persists min/max extent bounds per dimension in
22//!   its metadata (maintained incrementally as versions migrate), and those
23//!   bounds are merged back into the extent aggregate at startup — so a fact
24//!   migrated to cold before a restart still counts toward `earliest`.
25//! - **`earliest`**: the minimum interval start observed in that dimension.
26//! - **`latest`**: the maximum *finite* coordinate observed in that
27//!   dimension — the max over every interval start and every **closed**
28//!   interval end. Open-ended intervals (still-valid facts / still-current
29//!   records, `end == TIMESTAMP_MAX`) contribute only their start, so
30//!   `latest` reports the newest recorded event coordinate and never the
31//!   open-interval sentinel (+infinity).
32//! - **Empty database**: all bounds are `None` — never `0`/epoch — so "no
33//!   data" cannot be misread as "data starting at 1970".
34

41/// Earliest/latest bounds observed in one temporal dimension.
42///
43/// Both fields are `None` when no versions have ever been recorded (empty
44/// database); otherwise both are `Some`. See the [module docs](self) for the
45/// exact `earliest`/`latest` convention.
46#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
47pub struct TimeBounds {
48    /// Minimum interval start observed, or `None` if no data was recorded.
49    pub earliest: Option<Timestamp>,
50    /// Maximum finite coordinate observed (max of interval starts and closed
51    /// interval ends; open intervals contribute only their start), or `None`
52    /// if no data was recorded.
53    pub latest: Option<Timestamp>,
54}

56/// Bi-temporal bounds for a single node label or edge type.
57#[derive(Debug, Clone, PartialEq, Eq)]
58pub struct LabelExtent {
59    /// The node label or edge/relationship type.
60    pub label: String,
61    /// Valid-time bounds across all recorded versions with this label/type.
62    pub valid_time: TimeBounds,
63    /// Transaction-time bounds across all recorded versions with this
64    /// label/type.
65    pub transaction_time: TimeBounds,
66}

68/// The dataset's queryable bi-temporal extent.
69///
70/// Returned by [`AletheiaDB::temporal_extent`] (overall bounds only) and
71/// [`AletheiaDB::temporal_extent_by_label`] (overall bounds plus a per-label
72/// / per-edge-type breakdown). See the [module docs](self) for field
73/// semantics.
74#[derive(Debug, Clone, PartialEq, Eq)]
75pub struct TemporalExtent {
76    /// Valid-time bounds across all recorded history.
77    pub valid_time: TimeBounds,
78    /// Transaction-time bounds across all recorded history.
79    pub transaction_time: TimeBounds,
80    /// Per-node-label bounds, sorted by label. `None` unless requested via
81    /// [`AletheiaDB::temporal_extent_by_label`].
82    pub node_labels: Option<Vec<LabelExtent>>,
83    /// Per-edge-type bounds, sorted by type. `None` unless requested via
84    /// [`AletheiaDB::temporal_extent_by_label`].
85    pub edge_types: Option<Vec<LabelExtent>>,
86}

88/// Accumulates min/max bounds for one temporal dimension.
89#[derive(Debug, Clone, Copy, Default)]
90struct DimAccumulator {
91    earliest: Option<Timestamp>,
92    latest: Option<Timestamp>,
93}
94
95impl DimAccumulator {
96    /// Fold one interval into the accumulator, applying the documented
97    /// convention: `earliest` tracks the min start; `latest` tracks the max
98    /// of starts and closed ends, never the `TIMESTAMP_MAX` open sentinel.
99    fn observe(&mut self, range: TimeRange) {
100        let start = range.start();
101        self.earliest = Some(self.earliest.map_or(start, |e| e.min(start)));
102
103        // TimeRange guarantees end >= start, so a closed end is itself the
104        // latest candidate; no max(start, end) needed.
105        let end = range.end();
106        let candidate = if end == TIMESTAMP_MAX { start } else { end };
107        self.latest = Some(self.latest.map_or(candidate, |l| l.max(candidate)));
108    }
109
110    fn into_bounds(self) -> TimeBounds {
111        TimeBounds {
112            earliest: self.earliest,
113            latest: self.latest,
114        }
115    }
116}

118/// Accumulates both temporal dimensions for one label/type (or overall).
119#[derive(Debug, Clone, Copy, Default)]
120struct BoundsAccumulator {
121    valid: DimAccumulator,
122    tx: DimAccumulator,
123}
124
125impl BoundsAccumulator {
126    fn observe(&mut self, temporal: &crate::core::temporal::BiTemporalInterval) {
127        self.valid.observe(temporal.valid_time());
128        self.tx.observe(temporal.transaction_time());
129    }
130}

132/// Convert a per-label accumulator map into a sorted `Vec<LabelExtent>`.
133fn into_label_extents(acc: HashMap<InternedString, BoundsAccumulator>) -> Vec<LabelExtent> {
134    let mut extents: Vec<LabelExtent> = acc
135        .into_iter()
136        .map(|(label, bounds)| LabelExtent {
137            label: GLOBAL_INTERNER.resolve_or_else(label, String::new),
138            valid_time: bounds.valid.into_bounds(),
139            transaction_time: bounds.tx.into_bounds(),
140        })
141        .collect();
142    extents.sort_unstable_by(|a, b| a.label.cmp(&b.label));
143    extents
144}

146impl AletheiaDB {
147    /// Report the dataset's queryable bi-temporal extent: the earliest and
148    /// latest valid-time and transaction-time coordinates across recorded
149    /// history, including expired/superseded versions and delete tombstones.
150    ///
151    /// Use this to calibrate `AS OF` queries: an `AS OF` before
152    /// `valid_time.earliest` (or after reconstructing a transaction-time
153    /// coordinate outside `transaction_time`) is guaranteed to land outside
154    /// recorded data, so its empty result means "out of recorded range", not
155    /// "the fact never existed".
156    ///
157    /// # Conventions (see [module docs](self) for details)
158    ///
159    /// - `earliest` = min interval start per dimension.
160    /// - `latest` = max of interval starts and *closed* interval ends;
161    ///   open-ended (still-valid / still-current) intervals contribute only
162    ///   their start, so the open-interval sentinel never leaks out.
163    /// - An empty database yields `None` for every bound — never epoch 0.
164    ///
165    /// Bounds are read O(1) from a write-time maintained aggregate in the
166    /// temporal indexes; no index traversal, no storage I/O. Bounds only
167    /// ever widen while the process runs, so callers may cache the result
168    /// for the duration of a session.
169    ///
170    /// # Coverage across restarts
171    ///
172    /// Bounds cover all history recorded during the current process lifetime
173    /// plus the hot-tier history restored at startup. On databases with
174    /// cold-storage migration enabled, history migrated to the cold tier is
175    /// covered across restarts too (Issue #3389): the cold store persists its
176    /// extent bounds and they are merged into the aggregate at startup, so a
177    /// fact migrated to cold before a restart still bounds the extent. See the
178    /// [module docs](self) for details.
179    ///
180    /// # Example
181    ///
182    /// ```rust,no_run
183    /// # use aletheiadb::AletheiaDB;
184    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
185    /// let db = AletheiaDB::new()?;
186    /// let extent = db.temporal_extent()?;
187    /// if extent.valid_time.earliest.is_none() {
188    ///     println!("no data recorded yet");
189    /// }
190    /// # Ok(())
191    /// # }
192    /// ```
193    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
194    pub fn temporal_extent(&self) -> Result<TemporalExtent> {
195        let (valid_time, transaction_time) = self.overall_extent_bounds();
196        Ok(TemporalExtent {
197            valid_time,
198            transaction_time,
199            node_labels: None,
200            edge_types: None,
201        })
202    }
203
204    /// Like [`temporal_extent`](Self::temporal_extent), but additionally
205    /// breaks the bounds down per node label and per edge/relationship type,
206    /// so a caller can scope `AS OF` calibration to exactly the labels it
207    /// queries.
208    ///
209    /// The overall bounds are identical to [`temporal_extent`](Self::temporal_extent)
210    /// and, as of Issue #3389, reflect cold-migrated history across restarts.
211    /// The **per-label breakdown**, however, is computed from the historical
212    /// version store, which retains only versions **still resident in the hot
213    /// tier**: on databases with cold-storage migration enabled, versions
214    /// whose payload has been migrated to the cold tier are not attributed to
215    /// a label (the persisted cold bounds are aggregate-only, not per-label).
216    /// So after a restart a label's per-label bound can be narrower than the
217    /// overall bound if that label's older history lives only in cold. On a
218    /// hot-only database every per-label bound lies within the overall bounds.
219    #[must_use = "this Result must be used; ignoring errors can lead to silent failures"]
220    pub fn temporal_extent_by_label(&self) -> Result<TemporalExtent> {
221        // Take the historical read lock FIRST, then read the overall bounds
222        // while holding it: commits mutate the temporal indexes while
223        // holding `historical.write()`, so holding the read lock across
224        // both phases yields one consistent snapshot (a write can no longer
225        // slip between "read overall bounds" and "scan per-label versions"
226        // and make a per-label bound exceed the overall bounds). Lock order
227        // is respected: `historical` (3) is acquired before the temporal
228        // indexes' internal extent aggregate (4), and `extent()` acquires
229        // nothing ordered before `historical`.
230        let historical = self.historical.read();
231
232        let (valid_time, transaction_time) = self.overall_extent_bounds();
233
234        let mut node_acc: HashMap<InternedString, BoundsAccumulator> = HashMap::new();
235        let mut edge_acc: HashMap<InternedString, BoundsAccumulator> = HashMap::new();
236        historical.visit_node_versions(|version| {
237            node_acc
238                .entry(version.label)
239                .or_default()
240                .observe(&version.temporal);
241        });
242        historical.visit_edge_versions(|version| {
243            edge_acc
244                .entry(version.label)
245                .or_default()
246                .observe(&version.temporal);
247        });
248        drop(historical);
249
250        Ok(TemporalExtent {
251            valid_time,
252            transaction_time,
253            node_labels: Some(into_label_extents(node_acc)),
254            edge_types: Some(into_label_extents(edge_acc)),
255        })
256    }
257
258    /// Read the temporal indexes' O(1) extent aggregate as overall
259    /// per-dimension bounds.
260    ///
261    /// The temporal indexes observe every committed write (and interval
262    /// close) at write time and never remove entries while the process
263    /// runs, making them the authoritative in-process source for recorded
264    /// history. An empty database yields all-`None` bounds.
265    fn overall_extent_bounds(&self) -> (TimeBounds, TimeBounds) {
266        match self.temporal_indexes.extent() {
267            Some(extent) => (
268                TimeBounds {
269                    earliest: Some(extent.valid_earliest),
270                    latest: Some(extent.valid_latest),
271                },
272                TimeBounds {
273                    earliest: Some(extent.tx_earliest),
274                    latest: Some(extent.tx_latest),
275                },
276            ),
277            None => (TimeBounds::default(), TimeBounds::default()),
278        }
279    }
280}

318    fn populated_db_bounds_match_min_max_across_dimensions() {
319        let db = AletheiaDB::new().expect("db init");
320        let before = time::now();
321
322        // Backdated node: valid time well before "now".
323        let t1 = ts(1_000_000_000_000_000); // ~2001-09-09
324        let t2 = ts(1_200_000_000_000_000); // ~2008-01-10
325        db.create_node_with_valid_time("Person", props("name", "Alice"), Some(t1))
326            .expect("create node");
327        db.create_node_with_valid_time("Company", props("name", "Acme"), Some(t2))
328            .expect("create node");
329
330        let after = time::now();
331        let extent = db.temporal_extent().expect("extent");
332
333        // Valid time: earliest is the backdated t1; latest is the max valid
334        // event (t2 -- both intervals are open-ended so only starts count).
335        assert_eq!(extent.valid_time.earliest, Some(t1));
336        assert_eq!(extent.valid_time.latest, Some(t2));
337
338        // Transaction time: system-assigned, bracketed by the test run;
339        // must never be the open-interval sentinel or epoch. The upper
340        // bound compares wallclocks with a small slack: a commit HLC and
341        // the `after` capture can share a microsecond, in which case the
342        // commit's logical counter makes it compare greater than `after`.
343        let tx_earliest = extent.transaction_time.earliest.expect("tx earliest");
344        let tx_latest = extent.transaction_time.latest.expect("tx latest");
345        assert!(tx_earliest >= before, "tx earliest within test window");
346        assert!(
347            tx_latest.wallclock() <= after.wallclock() + 10,
348            "tx latest within test window (with HLC same-microsecond slack)"
349        );
350        assert!(tx_earliest <= tx_latest);
351        assert!(tx_latest < TIMESTAMP_MAX, "sentinel must not leak");
352    }

426    /// Assert every per-label bound lies within the overall bounds, and
427    /// that the overall bounds equal the fold of the per-label breakdown.
428    /// Both properties hold on hot-only databases (no cold-tier migration),
429    /// which is what `AletheiaDB::new()` provides.
430    fn assert_breakdown_consistent_with_overall(extent: &TemporalExtent) {
431        let mut folded_valid = DimAccumulator::default();
432        let mut folded_tx = DimAccumulator::default();
433
434        let all_labels = extent
435            .node_labels
436            .as_deref()
437            .expect("node labels")
438            .iter()
439            .chain(extent.edge_types.as_deref().expect("edge types"));
440
441        for label_extent in all_labels {
442            for (bounds, overall, folded) in [
443                (
444                    &label_extent.valid_time,
445                    &extent.valid_time,
446                    &mut folded_valid,
447                ),
448                (
449                    &label_extent.transaction_time,
450                    &extent.transaction_time,
451                    &mut folded_tx,
452                ),
453            ] {
454                let earliest = bounds.earliest.expect("label earliest");
455                let latest = bounds.latest.expect("label latest");
456                assert!(
457                    earliest >= overall.earliest.expect("overall earliest"),
458                    "label '{}' earliest must be within overall bounds",
459                    label_extent.label
460                );
461                assert!(
462                    latest <= overall.latest.expect("overall latest"),
463                    "label '{}' latest must be within overall bounds",
464                    label_extent.label
465                );
466                folded.earliest = Some(folded.earliest.map_or(earliest, |e| e.min(earliest)));
467                folded.latest = Some(folded.latest.map_or(latest, |l| l.max(latest)));
468            }
469        }
470
471        // On a hot-only DB the breakdown covers every version the overall
472        // bounds cover, so folding it back must reproduce them exactly —
473        // guarding against the write-time aggregate and the per-label fold
474        // ever drifting apart in convention.
475        assert_eq!(
476            folded_valid.into_bounds(),
477            extent.valid_time,
478            "overall valid bounds must equal fold of the by_label breakdown"
479        );
480        assert_eq!(
481            folded_tx.into_bounds(),
482            extent.transaction_time,
483            "overall tx bounds must equal fold of the by_label breakdown"
484        );
485    }

535    fn by_label_multi_version_label_bounds_span_backdated_start_and_closed_end() {
536        // Kills two DimAccumulator mutants that a single-version-per-label
537        // dataset cannot distinguish:
538        //   (b) earliest computed with max(start) instead of min(start)
539        //       would report t2 instead of the backdated t1;
540        //   and exercises the superseded (closed) first version's interval
541        //   alongside the open second one.
542        let db = AletheiaDB::new().expect("db init");
543
544        let t1 = ts(1_000_000_000_000_000); // backdated original
545        let t2 = ts(1_400_000_000_000_000); // later correction
546        let node_id = db
547            .create_node_with_valid_time("Person", props("role", "engineer"), Some(t1))
548            .expect("create node");
549        db.update_node_with_valid_time(node_id, props("role", "manager"), Some(t2))
550            .expect("update node");
551
552        let extent = db.temporal_extent_by_label().expect("extent");
553        let node_labels = extent.node_labels.as_deref().expect("node labels");
554        assert_eq!(node_labels.len(), 1);
555        let person = &node_labels[0];
556        assert_eq!(person.label, "Person");
557
558        // The superseded version's backdated start still bounds earliest
559        // (min of starts, not max).
560        assert_eq!(
561            person.valid_time.earliest,
562            Some(t1),
563            "per-label earliest must be the min start (backdated t1), not the newest"
564        );
565        // #3504: the superseded version's valid interval stays OPEN
566        // (append-only), so `latest` reaches t2 via the UPDATE version's valid

821    /// Core widen-only invariant at the DB level (Issue #3389): a cold-migrated
822    /// OLD fact and a newer hot fact must together bound the extent — earliest
823    /// WIDENS back to cold, latest is NOT narrowed away from the hot fact. This
824    /// is the merge's central contract, previously untested at DB level because
825    /// every other DB test has an empty/absent hot tier; it simultaneously
826    /// covers "migrated old version + newer hot version → extent spans both".
827    #[test]
828    fn cold_old_and_hot_recent_extent_spans_both_widen_only() {
829        let temp = tempfile::tempdir().expect("tempdir");
830        let cold_path = temp.path().join("cold.redb");
831
832        // Cold tier: an OLD fact, earlier than anything the hot tier holds.
833        let cold_valid = ts(1_000_000_000_000_000); // ~2001
834        let cold_tx = ts(1_100_000_000_000_000); // ~2004
835        seed_cold_node(
836            &cold_path,
837            TimeRange::from(cold_valid),
838            TimeRange::from(cold_tx),
839        );
840
841        // Reopen against the same cold file, then write a RECENT fact through
842        // the public API so it lands in the hot tier. Its valid start is later
843        // than cold's (NARROWER earliest) and its coordinates are newer than
844        // cold's (WIDER latest) in BOTH dimensions.
845        let db = open_db_with_cold(temp.path(), &cold_path);
846        let hot_valid = ts(1_800_000_000_000_000); // ~2027, well after cold
847        let before_write = time::now();
848        db.create_node_with_valid_time("Person", props("name", "Bob"), Some(hot_valid))
849            .expect("create hot node");
850
851        let extent = db.temporal_extent().expect("extent");
852
853        // (i) earliest WIDENS back to the cold fact in both dimensions — the
854        //     startup merge restored cold's older bounds the hot tier lacked.
855        assert_eq!(
856            extent.valid_time.earliest,
857            Some(cold_valid),
858            "valid earliest must widen to the cold fact's older start"
859        );
860        assert_eq!(
861            extent.transaction_time.earliest,
862            Some(cold_tx),
863            "tx earliest must widen to the cold fact's older start"
864        );
865
866        // (ii) latest is NOT narrowed to cold's older coordinates — it retains
867        //      the newer hot fact. This is the widen-only proof: the merge
868        //      never pulls `latest` back to the smaller cold value.
869        assert_eq!(
870            extent.valid_time.latest,
871            Some(hot_valid),
872            "valid latest must retain the newer hot fact, not narrow to cold"
873        );
874        let tx_latest = extent.transaction_time.latest.expect("tx latest");
875        assert!(
876            tx_latest >= before_write,
877            "tx latest must retain the newer hot commit, not narrow to cold"
878        );
879        assert!(
880            tx_latest > cold_tx,
881            "tx latest must exceed the older cold coordinate"
882        );
883        assert!(
884            extent.valid_time.latest.expect("latest") < TIMESTAMP_MAX,
885            "open-interval sentinel must never leak into latest"
886        );
887    }
