1//! Push-changefeed subscription API on [`AletheiaDB`] (Issue #3375).
2//!
3//! The [`AletheiaDB::list_changes`](crate::AletheiaDB::list_changes) *pull* feed answers
4//! "what changed between T1 and T2?" on demand. This module adds the complementary *push*
5//! feed: [`subscribe_changes`](AletheiaDB::subscribe_changes) hands out a
6//! [`Subscription`] whose buffer fills with matching [`ChangeRecord`](crate::ChangeRecord)s
7//! as transactions commit — no polling.
8//!
9//! # No-loss contract
10//!
11//! Delivery is **best-effort at-least-once**; the *durable* ground truth is `list_changes`.
12//! A consumer that lags (buffer overflow → disconnected), reconnects, or survives a crash
13//! resumes with **zero loss** by pulling `list_changes` from its last
14//! [`resume_token`](Subscription::resume_token) — see
15//! [`docs/guides/reacting-to-change.md`](https://example/) and
16//! [`crate::core::changefeed_subscription`] for the full contract (including
17//! duplicate-on-resume, deduped by the stable `ChangeRecord` cursor).
18//!
19//! The HTTP SSE surface and the MCP `await_changes` tool that wrap the blocking
20//! [`Subscription::recv_timeout`] long-poll are a coordinated follow-up (Lane 1).
21
22use crate::core::changefeed_subscription::{ChangeFilter, ChangefeedConfig, Subscription};
23use crate::core::error::Result;
24use crate::db::AletheiaDB;
25
26impl AletheiaDB {
27    /// Subscribe to the push changefeed, receiving every committed change matching `filter`.
28    ///
29    /// The returned [`Subscription`] buffers matching changes as transactions commit; drain
30    /// it with [`Subscription::poll`] (non-blocking) or [`Subscription::recv_timeout`]
31    /// (blocking long-poll). Dropping the subscription deregisters it.
32    ///
33    /// # Errors
34    ///
35    /// Returns [`StorageError::CapacityExceeded`](crate::StorageError::CapacityExceeded) when
36    /// the configured maximum number of concurrent subscriptions is already reached.
37    ///
38    /// # Example
39    ///
40    /// ```rust
41    /// use aletheiadb::{AletheiaDB, ChangeFilter};
42    /// use aletheiadb::api::WriteOps;
43    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
44    /// let db = AletheiaDB::new()?;
45    /// let sub = db.subscribe_changes(ChangeFilter::all())?;
46    ///
47    /// db.write(|tx| { tx.create_node("Person", aletheiadb::PropertyMap::new())?; Ok::<_, aletheiadb::Error>(()) })?;
48    ///
49    /// let events = sub.poll();
50    /// assert_eq!(events.len(), 1);
51    /// # Ok(())
52    /// # }
53    /// ```
54    #[must_use = "the returned Subscription must be held to keep receiving changes"]
55    pub fn subscribe_changes(&self, filter: ChangeFilter) -> Result<Subscription> {
56        // Capture the committed frontier as the subscription's initial resume anchor
57        // (Issue #3375 review F4). Reading `current_timestamp` (lock order class 1, taken
58        // alone here) snapshots the last commit's HLC; any future commit gets a strictly
59        // greater timestamp, so `baseline_after(frontier)` is a gap-free "resume from where I
60        // subscribed" cursor even before the consumer drains its first event. Poison-recover
61        // the guard so a panicked writer elsewhere can never make subscribe unwind.
62        let frontier = *self
63            .current_timestamp
64            .lock()
65            .unwrap_or_else(|e| e.into_inner());
66        let baseline = crate::core::changefeed::ChangeCursor::baseline_after(frontier);
67        self.changefeed
68            .subscribe_with_baseline(filter, Some(baseline))
69    }
70
71    /// Subscribe to the push changefeed on behalf of an authenticated principal,
72    /// enforcing the per-principal subscription quota (Issue #3678).
73    ///
74    /// Identical to [`subscribe_changes`](Self::subscribe_changes) but accounts
75    /// the new subscription against `principal_key`'s quota bucket. The changefeed
76    /// surfaces (MCP `await_changes`, HTTP `/changes/await` and `/changes/stream`)
77    /// call this with the authenticated principal id, or the shared `"anonymous"`
78    /// bucket in anonymous mode, so no principal can exhaust the global cap and
79    /// starve others. Passing `None` reproduces the unquota'd
80    /// [`subscribe_changes`](Self::subscribe_changes) behavior.
81    ///
82    /// # Errors
83    ///
84    /// Returns [`StorageError::CapacityExceeded`](crate::StorageError::CapacityExceeded)
85    /// at the global cap, or
86    /// [`StorageError::PrincipalQuotaExceeded`](crate::StorageError::PrincipalQuotaExceeded)
87    /// when `principal_key` already holds its maximum concurrent subscriptions
88    /// (mapping to the `RESOURCE_EXHAUSTED`, `retriable: true` envelope).
89    #[must_use = "the returned Subscription must be held to keep receiving changes"]
90    pub fn subscribe_changes_for_principal(
91        &self,
92        principal_key: Option<&str>,
93        filter: ChangeFilter,
94    ) -> Result<Subscription> {
95        let frontier = *self
96            .current_timestamp
97            .lock()
98            .unwrap_or_else(|e| e.into_inner());
99        let baseline = crate::core::changefeed::ChangeCursor::baseline_after(frontier);
100        self.changefeed.subscribe_with_baseline_and_principal(
101            filter,
102            Some(baseline),
103            principal_key.map(str::to_string),
104        )
105    }
106
107    /// Live changefeed subscription counts per principal bucket (Issue #3678),
108    /// sorted by principal id. Powers the authenticated `database_stats`
109    /// per-principal breakdown; never exposed on the bounded `/metrics` surface.
110    pub fn changefeed_principal_counts(&self) -> Vec<(String, usize)> {
111        self.changefeed.per_principal_counts()
112    }
113
114    /// Reconfigure the changefeed caps (max concurrent subscriptions and per-subscription
115    /// buffer capacity). The subscription cap takes effect immediately; the buffer capacity
116    /// applies to subscriptions created afterward (existing ones keep their capacity).
117    pub fn set_changefeed_config(&self, config: ChangefeedConfig) {
118        self.changefeed.set_config(config);
119    }
120
121    /// The number of currently-live changefeed subscriptions (primarily for tests/metrics).
122    pub fn changefeed_subscription_count(&self) -> usize {
123        self.changefeed.subscription_count()
124    }
