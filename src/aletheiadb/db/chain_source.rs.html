1//! A [`VersionSource`] over the live historical store (Issue #3351).
2//!
3//! The provenance-chain sealer and verifier both resolve authoritative version
4//! content through this source, so a sealed leaf and a recomputed leaf hash
5//! identical bytes unless the stored version was tampered with. Reading the
6//! *stored* version (including superseded/tombstone versions) is exactly what
7//! makes on-disk tamper detectable: a byte changed in a persisted version
8//! changes the recomputed leaf and breaks the fold at that transaction.
9//!
10//! # Hashing the immutable logical version, not its storage encoding
11//!
12//! Several facets of a stored version are mutated by *later* writes and
13//! therefore cannot be bound into an append-only leaf:
14//! - its transaction-time END, its **valid-time END**, and its `is_current`
15//!   flag are all closed/flipped when a newer version supersedes it (an update
16//!   closes the prior version's open intervals at the successor's start — see
17//!   `close_previous_version_intervals`), and
18//! - its anchor/delta **encoding** changes: AletheiaDB re-encodes a prior
19//!   anchor as a reverse delta when a new version arrives, so the raw
20//!   `VersionData` for a fixed version id is not stable over time.
21//!
22//! So this source hashes the version's **reconstructed full property state**
23//! (stable regardless of anchor/delta re-encoding) plus its immutable identity
24//! and creation coordinates (`valid_from`/`transaction_from`, `prev_version`,
25//! provenance), with the interval **ends** and `is_current` normalized out. The
26//! same normalization runs on the seal and verify paths (both route through
27//! this source), so their leaves match by construction — while a tampered
28//! property value, valid-from/transaction-from, provenance, or identity still
29//! diverges and is caught.
30

40/// A [`VersionSource`] backed by the database's historical storage.
41pub(crate) struct DbVersionSource {
42    historical: Arc<RwLock<HistoricalStorage>>,
43}
44
45impl DbVersionSource {
46    /// Wrap a shared historical store as a version source.
47    pub(crate) fn new(historical: Arc<RwLock<HistoricalStorage>>) -> Self {
48        DbVersionSource { historical }
49    }
50}
51
52impl VersionSource for DbVersionSource {
53    fn fetch(&self, kind: EntityKind, _id: u64, version_id: u64) -> Option<VersionHashInput> {
54        let historical = self.historical.read();
55        fetch_locked(&historical, kind, version_id)
56    }
57
58    /// Hold the historical **read lock once** for the whole verification pass
59    /// (Issue #3351 task #2). `verify_full`/`verify_entity` route every leaf
60    /// re-fetch through the borrowed [`LockedDbVersionSource`], so the pass
61    /// acquires the lock a single time instead of once per version — turning an
62    /// O(versions) lock churn into O(1).
63    fn scoped(&self, f: &mut dyn FnMut(&dyn VersionSource)) {
64        let historical = self.historical.read();
65        let locked = LockedDbVersionSource {
66            historical: &historical,
67        };
68        f(&locked);
69    }
70}

72/// A [`VersionSource`] view over an **already-locked** historical store,
73/// borrowed for the duration of one `scoped` verification pass. Its `fetch`
74/// never re-locks (Issue #3351 task #2).
75struct LockedDbVersionSource<'a> {
76    historical: &'a HistoricalStorage,
77}
78
79impl VersionSource for LockedDbVersionSource<'_> {
80    fn fetch(&self, kind: EntityKind, _id: u64, version_id: u64) -> Option<VersionHashInput> {
81        fetch_locked(self.historical, kind, version_id)
82    }
83    // `scoped` default (call `f(self)`): the read lock is already held.
84}

86/// Reconstruct the authoritative, normalized [`VersionHashInput`] for a version
87/// from an already-locked historical store. Shared by the per-call
88/// [`DbVersionSource::fetch`] and the single-lock
89/// [`LockedDbVersionSource::fetch`] so both produce byte-identical leaves.
90///
91/// # GDPR crypto-shred invariant — DO NOT reroute through the unsealing boundary
92///
93/// Properties MUST be reconstructed via the **raw** `reconstruct_node_properties`
94/// / `reconstruct_edge_properties` storage path, which is crypto-shred-**unaware**
95/// and returns the bytes exactly as stored — for a designated property that is the
96/// sealed `PropertyValue::Bytes(SUBJ…)` envelope (ciphertext), never the plaintext.
97/// This is what makes the provenance-chain leaf **erasure-stable** (AC4, Issue
98/// #3359): the leaf binds the ciphertext, and `erase_subject` destroys only the
99/// subject key — never the stored envelope — so the recomputed leaf is unchanged
100/// and `verify_chain` still passes after a crypto-shred, while a tamper of the
101/// envelope bytes is still caught.
102///
103/// This function MUST NOT route through the db-API unsealing hooks
104/// (`materialize_shred` / `unseal_node_view` / `unseal_edge_view`,
105/// `src/db/crypto_shred/api.rs`). Doing so would make the leaf bind **plaintext**,
106/// which (1) breaks AC4 — a leaf over destroyed plaintext can no longer be
107/// recomputed after erasure, so `verify_chain` would spuriously fail — and (2)
108/// would defeat on-disk tamper detection for sealed values. Non-crypto-shred data
109/// is unaffected: with no designation, historical holds plaintext and the leaf
110/// binds it exactly as before.
111fn fetch_locked(
112    historical: &HistoricalStorage,
113    kind: EntityKind,
114    version_id: u64,
115) -> Option<VersionHashInput> {
116    let vid = VersionId::new(version_id).ok()?;
117    // Cold-tier aware (Issue #3351 finding 5): read the version metadata via
118    // the tiered getter so verify does not FALSE-FAIL ("version not found")
119    // once a sealed version has migrated to the Redb cold tier. Property
120    // reconstruction (`reconstruct_*_properties`) is already tier-aware.
121    let mut input = match kind {
122        EntityKind::Node => {
123            let v = historical.get_node_version_tiered(vid).ok().flatten()?;
124            let mut input = VersionHashInput::from(v.as_ref());
125            let full = historical.reconstruct_node_properties(vid).ok()?;
126            input.properties = resolve_properties(&full);
127            input
128        }
129        EntityKind::Edge => {
130            let v = historical.get_edge_version_tiered(vid).ok().flatten()?;
131            let mut input = VersionHashInput::from(v.as_ref());
132            let full = historical.reconstruct_edge_properties(vid).ok()?;
133            input.properties = resolve_properties(&full);
134            input
135        }
136    };
137    normalize_immutable(&mut input);
138    Some(input)
139}

141/// Resolve an interned-keyed property map into `(String, value)` pairs. Key
142/// ordering is irrelevant — the canonical encoder sorts by key.
143fn resolve_properties(
144    map: &crate::core::property::PropertyMap,
145) -> Vec<(String, crate::core::property::PropertyValue)> {
146    map.iter()
147        .filter_map(|(key, value)| {
148            GLOBAL_INTERNER
149                .resolve_with(*key, |s| s.to_string())
150                .map(|name| (name, value.clone()))
151        })
152        .collect()
153}

155/// Normalize the fields a later supersession mutates to their as-written state,
156/// so a version's leaf is stable after sealing.
157///
158/// `transaction_to` and `is_current` are ALWAYS normalized out: a successor
159/// version closes the prior version's open transaction interval and flips
160/// `is_current`, so neither can be bound into an append-only leaf.
161///
162/// The valid-time **end** (`valid_to`) is handled per finding 2c:
163/// - For a *live* version it starts open and is closed by a later supersession
164///   (`close_previous_version_intervals` only closes **open** valid intervals),
165///   so it is mutable and MUST be normalized out — otherwise a legitimately
166///   superseded version would fail verification.
167/// - For a **born-closed terminal** version (a delete tombstone, or a
168///   retraction) the valid end is immutable: it is closed at creation and never
169///   re-closed (an already-closed valid interval fails the
170///   `is_currently_valid()` guard in supersession). Binding it directly makes a
171///   re-validation tamper (extending a retraction's `valid_to`, or re-opening a
172///   tombstone) change the leaf and break verification.
173///
174/// We detect "born-closed / historical closure" as `valid_to <= transaction_from`
175/// (the fact stopped being true at or before it was recorded — a tombstone has
176/// `valid_to == valid_from <= tx_from`, a default retraction has
177/// `valid_to == now == tx_from`) plus the explicit tombstone flag. This is a
178/// stable seal==verify predicate for the common case.
179///
180/// # Known limitation (documented v1)
181///
182/// A *backdated* supersession — two heavily backdated writes to the same entity
183/// where the successor's `valid_start` lands at or before the prior version's
184/// own `transaction_from` — closes the prior (live) version's `valid_to` to a
185/// point `<= tx_from`, which this predicate would then treat as born-closed and
186/// bind, while it was open at seal time. That is a rare false-positive class, not
187/// a missed tamper; the timeline-consistency check does not add false negatives.
188fn normalize_immutable(input: &mut VersionHashInput) {
189    let born_closed = input.is_tombstone
190        || matches!(
191            (input.valid_to, input.transaction_from),
192            (Some(vt), Some(tf)) if vt <= tf
193        );
194    if !born_closed {
195        input.valid_to = None;
196    }
197    input.transaction_to = None;
198    input.is_current = true;
199}
