1//! Backend-agnostic observability contract for AletheiaDB.
2//!
3//! AletheiaDB emits OpenTelemetry-compatible spans through [`tracing`] and
4//! reports bounded-cardinality metrics through [`MetricsRecorder`]. The crate
5//! deliberately does not own exporters, collectors, HTTP scrape endpoints, or
6//! vendor clients. Applications install their preferred `tracing` subscriber
7//! and, when desired, enable the `metrics-rs` feature to forward metric samples
8//! into the [`metrics`](https://docs.rs/metrics) facade.
9

105static METRICS_RECORDER: LazyLock<ArcSwap<SharedMetricsRecorder>> =
106    LazyLock::new(|| ArcSwap::from_pointee(SharedMetricsRecorder::new(Arc::new(NoOpMetrics))));
107
108struct SharedMetricsRecorder {
109    inner: Arc<dyn MetricsRecorder>,
110}
111
112impl SharedMetricsRecorder {
113    fn new(inner: Arc<dyn MetricsRecorder>) -> Self {
114        Self { inner }
115    }
116}
117
118impl MetricsRecorder for SharedMetricsRecorder {
119    fn record_error(&self, category: ErrorCategory) {
120        self.inner.record_error(category);
121    }
122
123    fn record_write_conflict(&self) {
124        self.inner.record_write_conflict();
125    }
126
127    fn record_critical_event(&self, event: CriticalEvent) {
128        self.inner.record_critical_event(event);
129    }
130
131    fn record_transaction_commit(
132        &self,
133        duration_secs: f64,
134        operations_count: u64,
135        durability_mode: &str,
136        status: &str,
137    ) {
138        self.inner.record_transaction_commit(
139            duration_secs,
140            operations_count,
141            durability_mode,
142            status,
143        );
144    }
145}

147/// Bundle of telemetry dependencies supplied by the embedding application.
148///
149/// `TelemetryConfig::default()` keeps metrics disabled. Spans are emitted via
150/// `tracing` call sites and are controlled by whichever subscriber the
151/// application installs.
152#[derive(Clone)]
153pub struct TelemetryConfig {
154    /// Service name applications can stamp onto their subscriber/resource.
155    pub service_name: Arc<str>,
156
157    /// Metrics sink for contract-defined samples.
158    pub metrics: Arc<dyn MetricsRecorder>,
159}
160
161impl std::fmt::Debug for TelemetryConfig {
162    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
163        f.debug_struct("TelemetryConfig")
164            .field("service_name", &self.service_name)
165            .field("metrics", &std::any::type_name_of_val(&*self.metrics))
166            .finish()
167    }
168}
169
170impl Default for TelemetryConfig {
171    fn default() -> Self {
172        Self {
173            service_name: Arc::from(DB_SYSTEM_NAME),
174            metrics: Arc::new(NoOpMetrics),
175        }
176    }
177}
178
179impl TelemetryConfig {
180    /// Start a builder with safe no-op defaults.
181    #[must_use]
182    pub fn builder() -> TelemetryConfigBuilder {
183        TelemetryConfigBuilder::default()
184    }
185}
186
187/// Fluent builder for [`TelemetryConfig`].
188#[derive(Default)]
189pub struct TelemetryConfigBuilder {
190    service_name: Option<Arc<str>>,
191    metrics: Option<Arc<dyn MetricsRecorder>>,
192}
193
194impl TelemetryConfigBuilder {
195    /// Override the service name.
196    #[must_use]
197    pub fn service_name(mut self, service_name: impl Into<Arc<str>>) -> Self {
198        self.service_name = Some(service_name.into());
199        self
200    }
201
202    /// Register a metrics recorder.
203    #[must_use]
204    pub fn metrics(mut self, metrics: Arc<dyn MetricsRecorder>) -> Self {
205        self.metrics = Some(metrics);
206        self
207    }
208
209    /// Finalize the configuration.
210    #[must_use]
211    pub fn build(self) -> TelemetryConfig {
212        TelemetryConfig {
213            service_name: self
214                .service_name
215                .unwrap_or_else(|| Arc::from(DB_SYSTEM_NAME)),
216            metrics: self.metrics.unwrap_or_else(|| Arc::new(NoOpMetrics)),
217        }
218    }
219}

221/// Install process-wide telemetry hooks for AletheiaDB metrics.
222///
223/// This intentionally does not install a `tracing` subscriber. Subscriber and
224/// exporter ownership remains with the application.
225pub fn install(config: TelemetryConfig) {
226    METRICS_RECORDER.store(Arc::new(SharedMetricsRecorder::new(config.metrics)));
227}

253/// Record a transaction commit.
254pub fn record_transaction_commit(
255    duration_secs: f64,
256    operations_count: u64,
257    durability_mode: &str,
258    status: &str,
259) {
260    METRICS.record_transaction_commit(duration_secs, operations_count, durability_mode, status);
261    with_installed_recorder(|recorder| {
262        recorder.record_transaction_commit(
263            duration_secs,
264            operations_count,
265            durability_mode,
266            status,
267        );
268    });
269}
270
271fn with_installed_recorder(f: impl FnOnce(&dyn MetricsRecorder)) {
272    let recorder = METRICS_RECORDER.load();
273    f(&**recorder);
274}

354/// Create the root HTTP request span (Issue #3376).
355///
356/// The span carries the request method and route path up front. The
357/// operation name, result count, temporal scope, statement text, and error
358/// code are recorded later via [`tracing::Span::record`] once known — they are
359/// declared here as empty fields so the recording sites are cheap and the
360/// attribute set is stable. Credentials are deliberately never captured.
361#[must_use]
362pub fn http_request_span(method: &str, path: &str) -> tracing::Span {
363    // `parent: None` makes this a true root span, independent of any ambient
364    // tracing span (e.g. a tower `TraceLayer` request span). An incoming W3C
365    // context is attached afterward via `otel::attach_parent`, so the span is
366    // either a fresh root (absent traceparent) or nests under the caller's
367    // remote span — never accidentally under framework middleware.
368    tracing::info_span!(
369        parent: None,
370        SPAN_HTTP_REQUEST,
371        "db.system.name" = DB_SYSTEM_NAME,
372        "http.request.method" = method,
373        "url.path" = path,
374        "aletheiadb.operation" = tracing::field::Empty,
375        "aletheiadb.result.count" = tracing::field::Empty,
376        "aletheiadb.temporal.scope" = tracing::field::Empty,
377        "aletheiadb.error.code" = tracing::field::Empty,
378        "db.query.text" = tracing::field::Empty,
379        "otel.status_code" = tracing::field::Empty,
380    )
381}
