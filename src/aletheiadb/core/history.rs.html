12/// Information about a specific version of an entity (node or edge).
13///
14/// Represents a snapshot of an entity at a particular point in its history.
15///
16/// # Temporal Access
17///
18/// Note that `VersionInfo` does not expose `tx_time` (transaction time) directly as a field.
19/// Instead, it is embedded within the `temporal` field (a [`BiTemporalInterval`]).
20///
21/// To access the timestamp when this version was recorded:
22/// ```rust
23/// use aletheiadb::core::history::VersionInfo;
24/// use aletheiadb::core::temporal::{BiTemporalInterval, time};
25/// use aletheiadb::core::{VersionId, PropertyMap};
26///
27/// // Create a dummy version info
28/// let now = time::now();
29/// let version = VersionInfo {
30///     version_number: 1,
31///     version_id: VersionId::new(1).unwrap(),
32///     temporal: BiTemporalInterval::current(now),
33///     properties: PropertyMap::new(),
34///     label: "Person".to_string(),
35///     provenance: None,
36/// };
37///
38/// let recorded_at = version.temporal.transaction_time().start();
39/// let valid_from = version.temporal.valid_time().start();
40///
41/// assert_eq!(recorded_at, now);
42/// assert_eq!(valid_from, now);
43/// ```
44#[derive(Debug, Clone, PartialEq)]
45pub struct VersionInfo {
46    /// Sequential version number (1, 2, 3, ...)
47    pub version_number: u64,
48    /// Internal version ID
49    pub version_id: VersionId,
50    /// Bi-temporal validity interval
51    pub temporal: BiTemporalInterval,
52    /// Properties at this version
53    pub properties: PropertyMap,
54    /// Entity label
55    pub label: String,
56    /// Write-time attributive provenance (source, confidence, note, correlation_id).
57    ///
58    /// `None` if this version's write did not supply a bundle -- never a
59    /// fabricated default (Issue #3224).
60    pub provenance: Option<Provenance>,
61}

63/// Complete history of an entity (node or edge).
64///
65/// Represents the full timeline of an entity, containing all versions sorted chronologically
66/// by version number (oldest first).
67///
68/// # Example
69///
70/// ```rust
71/// use aletheiadb::core::history::{EntityHistory, VersionInfo};
72/// use aletheiadb::core::temporal::{BiTemporalInterval, time};
73/// use aletheiadb::core::{VersionId, PropertyMap};
74///
75/// // Construct a mock history
76/// let now = time::now();
77/// let v1 = VersionInfo {
78///     version_number: 1,
79///     version_id: VersionId::new(1).unwrap(),
80///     temporal: BiTemporalInterval::current(now),
81///     properties: PropertyMap::new(),
82///     label: "Person".to_string(),
83///     provenance: None,
84/// };
85///
86/// let history = EntityHistory { versions: vec![v1] };
87///
88/// println!("Entity has {} versions", history.version_count());
89///
90/// for version in &history.versions {
91///     println!("Version {}: Valid from {}",
92///         version.version_number,
93///         version.temporal.valid_time().start()
94///     );
95/// }
96/// ```
97#[derive(Debug, Clone)]
98pub struct EntityHistory {
99    /// All versions ordered by version_number (oldest first)
100    pub versions: Vec<VersionInfo>,
101}

103impl EntityHistory {
104    /// Returns the total number of versions in this entity's history.
105    ///
106    /// This is useful for understanding the update frequency or lifespan of a node or edge.
107    /// A high version count might indicate a highly volatile fact, while a count of 1
108    /// indicates a fact that has never been modified since its creation.
109    ///
110    /// # Examples
111    ///
112    /// ```rust
113    /// use aletheiadb::core::history::{EntityHistory, VersionInfo};
114    /// use aletheiadb::core::temporal::{BiTemporalInterval, time};
115    /// use aletheiadb::core::{VersionId, PropertyMap};
116    ///
117    /// let now = time::now();
118    /// let v1 = VersionInfo {
119    ///     version_number: 1,
120    ///     version_id: VersionId::new(1).unwrap(),
121    ///     temporal: BiTemporalInterval::current(now),
122    ///     properties: PropertyMap::new(),
123    ///     label: "Person".to_string(),
124    ///     provenance: None,
125    /// };
126    /// let history = EntityHistory { versions: vec![v1] };
127    /// assert_eq!(history.version_count(), 1);
128    /// ```
129    #[must_use]
130    pub fn version_count(&self) -> usize {
131        self.versions.len()
132    }
133
134    /// Returns the most recent version of this entity.
135    ///
136    /// Retrieves the latest state of the entity (the version with the highest `version_number`).
137    /// This is typically the active version, unless the entity has been logically deleted.
138    /// Returns `None` if the history is completely empty.
139    ///
140    /// # Examples
141    ///
142    /// ```rust
143    /// use aletheiadb::core::history::{EntityHistory, VersionInfo};
144    /// use aletheiadb::core::temporal::{BiTemporalInterval, time};

146    ///
147    /// let now = time::now();
148    /// let v1 = VersionInfo {
149    ///     version_number: 1,
150    ///     version_id: VersionId::new(1).unwrap(),
151    ///     temporal: BiTemporalInterval::current(now),
152    ///     properties: PropertyMap::new(),
153    ///     label: "Person".to_string(),
154    ///     provenance: None,
155    /// };
156    /// let history = EntityHistory { versions: vec![v1] };
157    /// assert_eq!(history.current_version().unwrap().version_number, 1);
158    /// ```
159    #[must_use]
160    pub fn current_version(&self) -> Option<&VersionInfo> {
161        self.versions.last()
162    }

165    ///
166    /// Retrieves the first state of the entity. This is useful for provenance tracking
167    /// to see how and when an entity originated.
168    /// Returns `None` if the history is empty.
169    ///
170    /// # Examples
171    ///
172    /// ```rust
173    /// use aletheiadb::core::history::{EntityHistory, VersionInfo};
174    /// use aletheiadb::core::temporal::{BiTemporalInterval, time};

182    ///     properties: PropertyMap::new(),
183    ///     label: "Person".to_string(),

187    /// assert_eq!(history.first_version().unwrap().version_number, 1);

191        self.versions.first()
192    }

195/// Difference between two versions of an entity.
196///
197/// Computes the semantic difference (delta) between the property maps of two versions.
198/// Useful for auditing changes, generating changelogs, or debugging.
199///
200/// # Change Categories
201///
202/// - **Added**: Key exists in `to` but not in `from`.
203/// - **Removed**: Key exists in `from` but not in `to`.
204/// - **Modified**: Key exists in both, but values are semantically different (using `semantically_equal`).
205///   - Note: Floating point `NaN`s are considered equal to each other.
206///
207/// # Example
208///
209/// ```rust
210/// use aletheiadb::core::history::VersionDiff;
211/// use aletheiadb::core::{PropertyMapBuilder, VersionId};
212///
213/// // Define properties for two versions
214/// let v1_props = PropertyMapBuilder::new()
215///     .insert("name", "Alice")
216///     .build();
217///
218/// let v2_props = PropertyMapBuilder::new()
219///     .insert("name", "Alice")
220///     .insert("age", 30) // Added property
221///     .build();
222///
223/// // Compute diff
224/// let diff = VersionDiff::compute(
225///     &v1_props,
226///     &v2_props,
227///     VersionId::new(1).unwrap(),
228///     VersionId::new(2).unwrap()
229/// );
230///
231/// if diff.has_changes() {
232///     println!("Changes detected:");
233///     println!("  Added: {} properties", diff.added.len());
234///     println!("  Removed: {} properties", diff.removed.len());
235///     println!("  Modified: {} properties", diff.modified.len());
236///     assert_eq!(diff.added.len(), 1);
237/// }
238/// ```
239#[derive(Debug, Clone, PartialEq)]
240pub struct VersionDiff {
241    /// The older version ID (baseline)
242    pub from_version: VersionId,
243    /// The newer version ID (comparison target)
244    pub to_version: VersionId,
245    /// Properties that were added in the newer version.
246    pub added: PropertyMap,
247    /// Properties that were removed in the newer version.
248    pub removed: PropertyMap,
249    /// Properties that were modified (Key, OldValue, NewValue).
250    pub modified: Vec<(InternedString, PropertyValue, PropertyValue)>,
251}
252
253impl VersionDiff {
254    /// Compute the difference between two property maps.
255    ///
256    /// # Arguments
257    ///
258    /// * `from` - Properties of the older version
259    /// * `to` - Properties of the newer version
260    /// * `from_id` - Version ID of the older version
261    /// * `to_id` - Version ID of the newer version
262    #[must_use]
263    pub fn compute(
264        from: &PropertyMap,
265        to: &PropertyMap,
266        from_id: VersionId,
267        to_id: VersionId,
268    ) -> Self {
269        use crate::core::property::PropertyMapBuilder;
270
271        let mut added_builder = PropertyMapBuilder::new();
272        let mut removed_builder = PropertyMapBuilder::new();
273        let mut modified = Vec::new();
274
275        // Find added and modified properties
276        for (key, to_value) in to.iter() {
277            match from.get_by_interned_key(key) {
278                None => {
279                    // Property was added
280                    added_builder = added_builder.insert_by_key(*key, to_value.clone());
281                }
282                Some(from_value) if !from_value.semantically_equal(to_value) => {
283                    // Property was modified
284                    modified.push((*key, from_value.clone(), to_value.clone()));
285                }
286                _ => {
287                    // Property unchanged
288                }
289            }
290        }
291
292        // Find removed properties
293        for (key, from_value) in from.iter() {
294            if !to.contains_interned_key(key) {
295                removed_builder = removed_builder.insert_by_key(*key, from_value.clone());
296            }
297        }
298
299        VersionDiff {
300            from_version: from_id,
301            to_version: to_id,
302            added: added_builder.build(),
303            removed: removed_builder.build(),
304            modified,
305        }
306    }
307
308    /// Check if there are any changes between the versions
309    #[must_use]
310    pub fn has_changes(&self) -> bool {
311        !self.added.is_empty() || !self.removed.is_empty() || !self.modified.is_empty()
312    }
313
314    /// Calculates the total number of property modifications between the two versions.
315    ///
316    /// This aggregates the counts of added, removed, and modified properties.
317    /// It is useful for quickly assessing the "size" or impact of a version update
318    /// without needing to inspect the specific properties that changed.
319    ///
320    /// # Examples
321    ///
322    /// ```rust
323    /// use aletheiadb::core::history::VersionDiff;
324    /// use aletheiadb::core::{PropertyMapBuilder, VersionId};
325    ///
326    /// let v1_props = PropertyMapBuilder::new().insert("name", "Alice").build();
327    /// let v2_props = PropertyMapBuilder::new().insert("name", "Alice").insert("age", 30).build();
328    ///
329    /// let diff = VersionDiff::compute(&v1_props, &v2_props, VersionId::new(1).unwrap(), VersionId::new(2).unwrap());
330    /// assert_eq!(diff.change_count(), 1); // 1 property added
331    /// ```
332    #[must_use]
333    pub fn change_count(&self) -> usize {
334        self.added.len() + self.removed.len() + self.modified.len()
335    }
336}

338/// Summary of version changes for display purposes.
339#[derive(Debug, Clone, PartialEq)]
340pub struct VersionSummary {
341    /// Version ID
342    pub version_id: VersionId,
343    /// Version number (1, 2, 3, ...)
344    pub version_number: u64,
345    /// When the version became valid
346    pub valid_from: Timestamp,
347    /// When the version was recorded
348    pub transaction_time: Timestamp,
349    /// Count of properties added in this version
350    pub properties_added: usize,
351    /// Count of properties removed in this version
352    pub properties_removed: usize,
353    /// Count of properties modified in this version
354    pub properties_modified: usize,
355}
356
357impl VersionSummary {
358    /// Check if this version has any property changes
359    #[must_use]
360    pub fn has_changes(&self) -> bool {
361        self.properties_added > 0 || self.properties_removed > 0 || self.properties_modified > 0
362    }
363
364    /// Calculates the total number of property modifications recorded in this summary.
365    ///
366    /// This aggregates the counts of added, removed, and modified properties.
367    /// It is useful for quickly assessing the impact of a version update
368    /// when you only have summary information (e.g., from an index or metadata scan)
369    /// without the full `VersionDiff`.
370    ///
371    /// # Examples
372    ///
373    /// ```rust
374    /// use aletheiadb::core::history::VersionSummary;
375    /// use aletheiadb::core::temporal::Timestamp;
376    /// use aletheiadb::core::VersionId;
377    ///
378    /// let summary = VersionSummary {
379    ///     version_id: VersionId::new(1).unwrap(),
380    ///     version_number: 1,
381    ///     valid_from: Timestamp::new(100, 0).unwrap(),
382    ///     transaction_time: Timestamp::new(200, 0).unwrap(),
383    ///     properties_added: 2,
384    ///     properties_removed: 0,
385    ///     properties_modified: 1,
386    /// };
387    /// assert_eq!(summary.change_count(), 3);
388    /// ```
389    #[must_use]
390    pub fn change_count(&self) -> usize {
391        self.properties_added + self.properties_removed + self.properties_modified
392    }
393}

746                1,
747            ),

754                    .build(),

782                expected_has_changes,
