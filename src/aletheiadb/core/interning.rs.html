1//! String interning for memory efficiency.
2//!
3//! This module provides a thread-safe string interner that maps strings to small
4//! integer IDs. This is particularly useful for labels and property keys that are
5//! repeated many times throughout the database.
6//!
7//! Benefits:
8//! - Reduces memory usage (4 bytes instead of 24 for each string reference)
9//! - Enables O(1) string equality checks (compare u32 instead of string contents)
10//! - Thread-safe without locking (uses DashMap)
11

13use dashmap::DashMap;
14use std::fmt;
15use std::hash::BuildHasherDefault;
16use std::sync::Arc;

19/// Default maximum number of interned strings (DoS protection).
20///
21/// Raised from 100_000 to 10_000_000 (Issue: configurable interner cap): a
22/// legitimately large dataset (e.g. a code graph with high-cardinality file
23/// paths / symbol names as property values) can intern well past 100K unique
24/// strings. This is a **COUNT** cap, not a memory cap: at ~100 bytes of
25/// map/pointer overhead per entry plus the string's own bytes, 10M *short*
26/// entries sit around ~1–1.6 GB (typical case), but the true worst case is
27/// `count × (overhead + string bytes)`, with string bytes bounded only by the
28/// per-string [`crate::storage::index_persistence::MAX_STRING_LENGTH`] (10 MB) —
29/// so it is paired with that per-string cap rather than being a hard memory
30/// ceiling. It remains near-impossible to hit for realistic workloads. The cap
31/// is configurable at runtime via
32/// [`StringInterner::set_max_capacity`] (driven from
33/// `persistence.max_interned_strings`) and via the
34/// [`MAX_INTERNED_STRINGS_ENV`] environment variable.
35pub const DEFAULT_MAX_INTERNED_STRINGS: usize = 10_000_000;
36
37/// Maximum spin iterations in get_all_strings() before assuming deadlock.
38const MAX_SPIN_WAIT_ITERATIONS: usize = 100_000;

58/// A small, copyable handle to an interned string.
59///
60/// This is just a u32 ID that can be used to look up the original string
61/// in the interner. It's Copy, so passing it around is very cheap.
62#[derive(Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
63pub struct InternedString(u32);

67    ///
68    /// # Safety
69    /// This is safe but the caller must ensure the ID is valid in the interner.

72        InternedString(id)
73    }

76    #[inline]
77    pub const fn as_u32(self) -> u32 {
78        self.0

82impl fmt::Display for InternedString {
83    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
84        // Try to resolve using the global interner first
85        // This makes debugging much easier by showing the actual string
86        // We use resolve_with to avoid allocating a new String just for display
87        let result = GLOBAL_INTERNER.resolve_with(*self, |s| write!(f, "{}", s));
88
89        // If resolution failed (e.g. ID from a local interner), fallback to showing ID
90        match result {
91            Some(res) => res,
92            None => write!(f, "Interned({})", self.0),
93        }
94    }
95}

110/// Thread-safe string interner.
111///
112/// This interner maintains a bidirectional mapping between strings and IDs:
113/// - String → ID: For interning new strings
114/// - ID → String: For resolving interned strings
115///
116/// The interner is designed to be used as a singleton (via lazy_static or similar).
117pub struct StringInterner {
118    /// Maps strings to their IDs.
119    string_to_id: DashMap<Arc<str>, InternedString>,
120    /// Maps IDs back to strings.
121    /// Uses IdentityHasher for O(1) lookups without hashing overhead.
122    id_to_string: DashMap<InternedString, Arc<str>, BuildHasherDefault<IdentityHasher>>,
123    /// Next ID to assign.
124    next_id: AtomicU32,
125    /// Maximum number of strings to intern (DoS protection).
126    ///
127    /// Atomic so it can be reconfigured at database `open()` time via
128    /// [`Self::set_max_capacity`] (driven from `persistence.max_interned_strings`)
129    /// without rebuilding the process-global interner. Read only on the slow
130    /// (new-id) intern path, so the atomic load stays off the already-interned
131    /// fast path.
132    max_capacity: AtomicUsize,
133}
134
135impl StringInterner {
136    /// Create a new empty string interner with default capacity limit.
137    pub fn new() -> Self {
138        Self::with_max_capacity(DEFAULT_MAX_INTERNED_STRINGS)
139    }
140
141    /// Create a new string interner with a custom maximum capacity.
142    pub fn with_max_capacity(max_capacity: usize) -> Self {
143        StringInterner {
144            string_to_id: DashMap::new(),
145            id_to_string: DashMap::with_hasher(BuildHasherDefault::default()),
146            next_id: AtomicU32::new(0),
147            max_capacity: AtomicUsize::new(max_capacity),
148        }
149    }
150
151    /// Reconfigure the maximum number of strings this interner will hold.
152    ///
153    /// Called at database `open()` time to drive the process-global
154    /// [`GLOBAL_INTERNER`] from `persistence.max_interned_strings`. This only
155    /// changes the threshold for interning *new* strings; it never evicts or
156    /// renumbers already-interned strings, so raising the cap is always safe and
157    /// lowering it merely refuses new interns past the lower bound.
158    pub(crate) fn set_max_capacity(&self, max_capacity: usize) {
159        self.max_capacity.store(max_capacity, Ordering::Relaxed);
160    }
161
162    /// The current maximum number of strings this interner will hold.
163    pub fn max_capacity(&self) -> usize {
164        self.max_capacity.load(Ordering::Relaxed)
165    }
166
167    /// Intern a string, returning its ID.
168    ///
169    /// If the string was already interned, returns the existing ID.
170    /// Otherwise, assigns a new ID and stores the string.
171    ///
172    /// This method is thread-safe and lock-free.
173    ///
174    /// # Errors
175    /// Returns `Error::Storage(StorageError::CapacityExceeded)` if the maximum
176    /// capacity is exceeded (DoS protection). This prevents unbounded memory growth.
177    pub fn intern<S: AsRef<str>>(
178        &self,
179        string: S,
180    ) -> std::result::Result<InternedString, crate::core::error::Error> {
181        let string = string.as_ref();
182
183        // Fast path: check if already interned (avoids Arc allocation)
184        if let Some(id) = self.get_id(string) {
185            return Ok(id);
186        }
187
188        // Slow path: need to intern the string
189        // Use entry API for atomic check-and-insert
190        // This prevents race conditions where two threads could assign different IDs
191        // to the same string
192        let arc_str: Arc<str> = Arc::from(string);
193
194        self.string_to_id
195            .entry(arc_str.clone())
196            .or_try_insert_with(|| {
197                // Atomically reserve an ID first to prevent capacity check race
198                let id_value = self.next_id.fetch_add(1, Ordering::Relaxed);
199
200                // Load the (possibly reconfigured) capacity on the slow path only.
201                let max_capacity = self.max_capacity.load(Ordering::Relaxed);
202
203                // Check if we exceeded capacity AFTER reserving ID. `id_value` is
204                // a `u32` (the id space is `AtomicU32`), so any configured cap at
205                // or above `u32::MAX` is unreachable by construction — the id
206                // generator saturates first. Callers clamp the configured cap to
207                // `u32::MAX` at `open()` so the effective cap is honest. Compare in
208                // usize so no truncation occurs at the boundary.
209                if id_value as usize >= max_capacity {
210                    // Best effort: undo the reservation
211                    self.next_id.fetch_sub(1, Ordering::Relaxed);
212
213                    return Err(crate::core::error::Error::Storage(
214                        crate::core::error::StorageError::CapacityExceeded {
215                            resource: "string interner".to_string(),
216                            current: id_value as usize,
217                            limit: max_capacity,
218                        },
219                    ));
220                }
221
222                let id = InternedString(id_value);
223
224                // Store the reverse mapping
225                self.id_to_string.insert(id, arc_str.clone());
226
227                Ok(id)
228            })
229            .map(|r| *r)
230    }

232    /// Intern a string without capacity checks.
233    ///
234    /// # Safety
235    /// This method bypasses capacity limits. It should only be used in trusted
236    /// internal contexts where the string is known to be valid and necessary,
237    /// such as WAL recovery or deserialization of known-good data.
238    ///
239    /// Using this method with untrusted input could lead to unbounded memory growth.
240    #[inline]
241    #[allow(dead_code)] // Available for internal use (WAL recovery, etc.)
242    pub(crate) fn intern_unchecked<S: AsRef<str>>(&self, string: S) -> InternedString {
243        let string = string.as_ref();
244
245        // Fast path: check if already interned
246        if let Some(id) = self.get_id(string) {
247            return id;
248        }
249
250        // Slow path: intern without capacity check
251        let arc_str: Arc<str> = Arc::from(string);
252
253        *self.string_to_id.entry(arc_str.clone()).or_insert_with(|| {
254            let id_value = self.next_id.fetch_add(1, Ordering::Relaxed);
255            let id = InternedString(id_value);
256            self.id_to_string.insert(id, arc_str.clone());
257            id
258        })
259    }

261    /// Resolve an interned string ID back to the original string.
262    ///
263    /// Returns None if the ID is not valid (was never interned).
264    ///
265    /// # Performance Note
266    ///
267    /// This method clones the underlying `Arc<str>`, which involves atomic reference
268    /// counting operations.
269    ///
270    /// - **For read-only access**: Use [`resolve_with`](Self::resolve_with) to avoid Arc cloning.
271    /// - **When an owned Arc is needed**: Use this method (`resolve`).
272    ///
273    /// Resolves an interned string to an owned `Arc<str>`.
274    ///
275    /// # Why?
276    /// If you absolutely need an owned copy of the string (e.g., to send across threads),
277    /// this function will retrieve it. However, prefer `resolve_with` to avoid cloning
278    /// the Arc if you only need temporary read access.
279    #[deprecated(
280        since = "0.1.0",
281        note = "Use resolve_with() for read-only access; use resolve() only when an owned Arc<str> is strictly required"
282    )]
283    pub fn resolve(&self, id: InternedString) -> Option<Arc<str>> {
284        self.id_to_string
285            .get(&id)
286            .map(|entry| Arc::clone(entry.value()))
287    }
288
289    /// Get the string as a &str without cloning the Arc.
290    ///
291    /// This is useful when you just need to read the string temporarily.
292    ///
293    /// **Note:** For read-only access, prefer [`resolve_with`](Self::resolve_with) to avoid Arc cloning overhead.
294    /// This method still clones the Arc internally.
295    pub fn get(&self, id: InternedString) -> Option<impl AsRef<str> + '_> {
296        self.id_to_string.get(&id).map(|entry| {
297            let arc: Arc<str> = Arc::clone(entry.value());
298            arc
299        })
300    }

302    /// Access the interned string via a callback without cloning the Arc.
303    ///
304    /// This is more efficient than `resolve()` or `get()` when you only need
305    /// temporary read access to the string, as it avoids atomic reference
306    /// counting operations.
307    ///
308    /// This is particularly useful for:
309    /// - Display and logging operations
310    /// - Serialization
311    /// - String comparisons
312    /// - Any read-only operation that doesn't need to own the string
313    ///
314    /// # Performance Example
315    ///
316    /// When serializing property keys, this avoids Arc clones:
317    /// ```ignore
318    /// // Instead of:
319    /// let key_str = interner.resolve(key_id)?;
320    /// serializer.serialize_str(key_str.as_ref())?; // Arc clone overhead
321    ///
322    /// // Use:
323    /// interner.resolve_with(key_id, |s| serializer.serialize_str(s))?;
324    /// ```
325    ///
326    /// # Examples
327    ///
328    /// ```
329    /// use aletheiadb::core::interning::StringInterner;
330    ///
331    /// let interner = StringInterner::new();
332    /// let id = interner.intern("hello").unwrap();
333    ///
334    /// // Efficient: no Arc clone
335    /// let len = interner.resolve_with(id, |s| s.len()).unwrap();
336    /// assert_eq!(len, 5);
337    ///
338    /// // Can return any type from the callback
339    /// let uppercase = interner.resolve_with(id, |s| s.to_uppercase()).unwrap();
340    /// assert_eq!(uppercase, "HELLO");
341    /// ```
342    pub fn resolve_with<F, R>(&self, id: InternedString, f: F) -> Option<R>
343    where
344        F: FnOnce(&str) -> R,
345    {
346        self.id_to_string
347            .get(&id)
348            .map(|entry| f(entry.value().as_ref()))
349    }
350
351    /// Access the interned string via a callback (deprecated - use [`resolve_with`](Self::resolve_with)).
352    #[deprecated(since = "0.1.0", note = "Use resolve_with() instead")]
353    pub fn with_str<F, R>(&self, id: InternedString, f: F) -> Option<R>
354    where
355        F: FnOnce(&str) -> R,
356    {
357        self.resolve_with(id, f)

360    /// Resolve an interned string to an owned `String`, or compute a fallback
361    /// if the ID is somehow not (or no longer) present.
362    ///
363    /// A resolve miss should not happen for an ID obtained from a live
364    /// `Node`/`Edge`/property key, but callers across the codebase each need
365    /// their own fallback text (an empty string, a diagnostic placeholder,
366    /// etc.) for the case where it does. This centralizes the resolve step
367    /// so each call site only has to supply its fallback.
368    pub fn resolve_or_else<F: FnOnce() -> String>(
369        &self,
370        id: InternedString,
371        fallback: F,
372    ) -> String {
373        self.resolve_with(id, |s| s.to_string())
374            .unwrap_or_else(fallback)
375    }

408    /// Get all interned strings in ID order.
409    ///
410    /// Returns a vector of strings sorted by their InternedString IDs.
411    /// This is useful for persistence where we need to save and restore
412    /// the interner state.
413    ///
414    /// This method is thread-safe and consistent. It waits for any concurrent
415    /// interning operations to complete to ensure there are no "holes" in the
416    /// returned sequence.
417    pub fn get_all_strings(&self) -> Vec<String> {
418        // Use next_id to estimate the required size.
419        let max_id = self.next_id.load(Ordering::Relaxed) as usize;
420
421        // Use Option<String> to track which IDs we've actually found.
422        // Initialize with None to distinguish "not found" from "empty string".
423        let mut strings: Vec<Option<String>> = vec![None; max_id];
424
425        // First pass: collect all (id, string) pairs currently in the map
426        for entry in self.id_to_string.iter() {
427            let id = entry.key().as_u32() as usize;
428
429            if id < strings.len() {
430                strings[id] = Some(entry.value().to_string());
431            } else {
432                // Handle race condition where an ID was added after we read next_id
433                strings.resize(id + 1, None);
434                strings[id] = Some(entry.value().to_string());
435            }
436        }
437
438        // Second pass: Check for holes and resolve them.
439        // Holes occur because intern() increments next_id BEFORE inserting into the map.
440        // If we see a hole at index i < strings.len(), it means an insert is in progress.
441        // We must wait for it to complete to return a consistent snapshot.
442        for (id, val) in strings.iter_mut().enumerate() {
443            if val.is_none() {
444                // Spin-wait for the missing ID to appear.
445                // In practice this is very short (just the time for a DashMap insert).
446                let mut spins = 0;
447                loop {
448                    // Check if ID is still within valid range. If next_id has been rolled back
449                    // (e.g. due to capacity exceeded), we should stop waiting.
450                    // Note: We use relaxed ordering as we are in a spin loop.
451                    let current_max_id = self.next_id.load(Ordering::Relaxed) as usize;
452                    if id >= current_max_id {
453                        // The ID was rolled back or invalid, stop waiting.
454                        break;
455                    }
456
457                    if let Some(entry) = self.id_to_string.get(&InternedString::from_raw(id as u32))
458                    {
459                        *val = Some(entry.value().to_string());
460                        break;
461                    }
462
463                    // Yield to allow the writer thread to proceed
464                    std::thread::yield_now();
465                    spins += 1;
466
467                    // Safety valve: don't spin forever if something is truly broken
468                    if spins > MAX_SPIN_WAIT_ITERATIONS {
469                        panic!(
470                            "Deadlock detected in get_all_strings: ID {} never appeared after {} spins. \
471                             This indicates a bug in the interner logic (e.g. rolled back ID not properly handled).",
472                            id, MAX_SPIN_WAIT_ITERATIONS
473                        );
474                    }
475                }
476            }
477        }
478
479        // Return only the strings that were successfully resolved.
480        // This implicitly handles rolled-back IDs by filtering out the Nones.
481        strings.into_iter().flatten().collect()
482    }

484    /// Pre-intern common strings at startup to avoid initial allocation overhead.
485    ///
486    /// This method interns frequently used property keys and labels that are
487    /// commonly accessed throughout the database's operational lifetime. By
488    /// pre-interning these strings, we eliminate the "allocation + hash + insert"
489    /// penalty on first access, providing predictable startup performance.
490    ///
491    /// The list of common strings is defined in [`COMMON_STRINGS`] and includes:
492    /// - Property keys: "name", "id", "type", "label"
493    /// - Timestamps: "created_at", "updated_at"
494    /// - Temporal: "valid_from", "valid_to", "tx_from", "tx_to"
495    ///
496    /// This method is idempotent - calling it multiple times will not duplicate
497    /// strings or change their IDs.
498    ///
499    /// # Performance Impact
500    ///
501    /// The performance impact is minimal and only affects the initial access cost
502    /// for each string. Subsequent accesses remain efficient with O(1) lookups.
503    ///
504    /// # Example
505    ///
506    /// ```
507    /// use aletheiadb::core::interning::StringInterner;
508    ///
509    /// let interner = StringInterner::new();
510    /// interner.warm_common_strings();
511    ///
512    /// // These strings are now pre-interned (no allocation on first access)
513    /// let id = interner.intern("name").unwrap();
514    /// ```
515    pub fn warm_common_strings(&self) {
516        // Intern each common string using intern_unchecked since these are
517        // known-good strings from a trusted internal source
518        for s in COMMON_STRINGS {
519            let _id = self.intern_unchecked(s);
520        }
521    }
522}

530/// Global string interner instance.
531///
532/// This can be used throughout the application for interning labels,
533/// property keys, and other frequently repeated strings.
534///
535/// # Example
536///
537/// ```ignore
538/// use aletheiadb::core::interning::GLOBAL_INTERNER;
539///
540/// let id1 = GLOBAL_INTERNER.intern("Person").unwrap();
541/// let id2 = GLOBAL_INTERNER.intern("Person").unwrap();
542/// assert_eq!(id1, id2); // Same string gets same ID
543///
544/// let string = GLOBAL_INTERNER.resolve(id1).unwrap();
545/// assert_eq!(string.as_ref(), "Person");
546/// ```
547use std::sync::LazyLock;

549/// Environment variable to configure the maximum number of interned strings.
550///
551/// Set `ALETHEIADB_MAX_INTERNED_STRINGS` to override the default limit of
552/// 10,000,000. This seeds the process-global interner for embedded users who
553/// never open a database; when a database is opened, `open()` reconfigures the
554/// cap from `persistence.max_interned_strings` (which wins).
555/// This is useful for large knowledge graphs with many unique labels or property keys.
556///
557/// Example: `ALETHEIADB_MAX_INTERNED_STRINGS=1000000`
558pub const MAX_INTERNED_STRINGS_ENV: &str = "ALETHEIADB_MAX_INTERNED_STRINGS";

560/// Global string interner for sharing common strings across the database.
561///
562/// This static provides a single, thread-safe string interner that can be used
563/// throughout the application to deduplicate common strings like labels and property keys.
564///
565/// The global interner is automatically warmed at initialization with common strings
566/// such as "name", "id", "type", "label", "created_at", "updated_at", "valid_from",
567/// "valid_to", "tx_from", and "tx_to" to provide predictable startup performance.
568///
569/// ## Configuration
570///
571/// The maximum capacity can be configured via the `ALETHEIADB_MAX_INTERNED_STRINGS`
572/// environment variable. If not set, defaults to 10,000,000. Opening a database
573/// further reconfigures it from `persistence.max_interned_strings`.
574pub static GLOBAL_INTERNER: LazyLock<StringInterner> = LazyLock::new(|| {
575    let max_capacity = std::env::var(MAX_INTERNED_STRINGS_ENV)
576        .ok()
577        .and_then(|s| s.parse::<usize>().ok())
578        .unwrap_or(DEFAULT_MAX_INTERNED_STRINGS);
579
580    let interner = StringInterner::with_max_capacity(max_capacity);
581    interner.warm_common_strings();
582    interner
583});

983

985        for thread_id in 0..10 {

987            let barrier = Arc::clone(&barrier);
988
989            let handle = thread::spawn(move || {

993                for i in 0..20 {
994                    let string = format!("thread_{}_string_{}", thread_id, i);
995                    if interner.intern(&string).is_ok() {
996                        success_count += 1;
997                    }
998                }
999                success_count
1000            });
1001            handles.push(handle);
1002        }
1003
1004        // Collect results
1005        let total_successes: usize = handles
1006            .into_iter()
1007            .map(|h: std::thread::JoinHandle<usize>| h.join().unwrap())
1008            .sum();
1009
1010        // Exactly 100 successful interns is guaranteed by atomic operations:
1011        // - fetch_add(1, Ordering::Relaxed) atomically reserves an ID

1013        // - This ensures exactly `max_capacity` successful interns, no more, no less
