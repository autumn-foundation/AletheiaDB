1//! Derivation lineage between facts (Issue #3371).
2//!
3//! Write-time *attributive* provenance ([`crate::core::provenance`]) answers
4//! "where did this fact come from?" for a fact written directly from an
5//! external source. It cannot answer the question that dominates real
6//! knowledge pipelines: **"which facts was this fact computed from?"** When an
7//! ETL job merges three records into one, when an agent summarizes ten
8//! documents into a claim, or when entity resolution collapses duplicates, the
9//! output fact's true provenance is *other facts in this database*, not a
10//! source string.
11//!
12//! This module records that derivation structure at write time and makes it
13//! queryable as a first-class, bi-temporally precise dimension. A
14//! [`LineageRecord`] declares that a *derived* fact version was computed from a
15//! set of *source* fact versions. Because every reference pins a specific
16//! [`VersionId`] (not just an entity id), lineage refers to exactly the fact
17//! that was read — later updates to an input never silently rewrite what a
18//! derivation was based on (the failure mode of modelling lineage as ordinary
19//! graph edges pointing at *current* nodes).
20//!
21//! # Version-space is a DAG
22//!
23//! A declared source must already exist when the derived version is written,
24//! so its [`VersionId`] was always allocated by an *earlier* write than the
25//! derived version's. Lineage edges therefore always point from a
26//! higher-numbered version to lower-numbered versions, which makes the lineage
27//! graph acyclic by construction. The cycle and self-derivation guards in
28//! [`LineageStore::record`] are defence-in-depth (and directly unit-tested);
29//! they cannot fire through the normal create/update write path.
30//!
31//! # Durability (v1 limitation)
32//!
33//! To avoid changing the WAL entry format (which is being restructured under
34//! Issue #3413), the v1 [`LineageStore`] is an in-memory index. Lineage
35//! records are immutable once written and survive supersession/retraction of
36//! the facts they reference, but they do **not** yet survive a process
37//! restart. Rehydrating lineage from a persisted log is a tracked follow-up;
38//! the record shape is deliberately serialisation-friendly for that work.
39

47/// A version-pinned reference to a single fact version — the unit of lineage.
48///
49/// Pins both *which* entity ([`EntityId`], a node or an edge) and *which
50/// version* of it ([`VersionId`]) was read, so lineage stays precise across
51/// later updates to that entity.
52#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
53pub struct LineageRef {
54    /// The entity (node or edge) the referenced version belongs to.
55    pub entity: EntityId,
56    /// The specific version that was read/derived.
57    pub version: VersionId,
58}
59
60impl LineageRef {
61    /// Construct a reference to a specific version of an entity.
62    #[inline]
63    pub fn new(entity: impl Into<EntityId>, version: VersionId) -> Self {
64        Self {
65            entity: entity.into(),
66            version,
67        }
68    }
69}

71/// An immutable record declaring that one fact version was derived from a set
72/// of source fact versions, recorded at a transaction time.
73#[derive(Debug, Clone, PartialEq, Eq)]
74pub struct LineageRecord {
75    /// The fact version that was produced.
76    pub derived: LineageRef,
77    /// The fact versions it was computed from (order preserved as declared;
78    /// deduplicated).
79    pub sources: Vec<LineageRef>,
80    /// The transaction time at which this derivation was recorded. Used to
81    /// answer `AS OF` lineage queries: a record is invisible to a query whose
82    /// `as_of` transaction time precedes `recorded_at`.
83    pub recorded_at: Timestamp,
84}
85
86/// Errors that can occur when recording a [`LineageRecord`].
87///
88/// Source-existence validation lives at the database layer (which owns the
89/// historical store); the errors here are the structural guards the store
90/// itself enforces.
91#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
92pub enum LineageError {
93    /// A declared source reference does not resolve to an existing fact
94    /// version (unknown version, or a version whose entity does not match the
95    /// reference). The write is rejected before any commit — no silent
96    /// dangling lineage.
97    #[error(
98        "derivation source {entity} @ version {version} does not exist",
99        entity = .reference.entity,
100        version = .reference.version.as_u64()
101    )]
102    SourceNotFound {
103        /// The unresolved source reference.
104        reference: LineageRef,
105    },
106
107    /// A version was declared as derived from itself.
108    #[error("fact {0} cannot be derived from itself")]
109    SelfDerivation(VersionId),
110
111    /// Recording the derivation would close a cycle: the derived version is
112    /// already reachable by walking upstream from one of the declared
113    /// sources. The path names the cycle, from the offending source up to the
114    /// derived version.
115    #[error("derivation would create a cycle: {}", format_cycle(.path))]
116    CycleDetected {
117        /// The upstream path from a declared source to the derived version
118        /// that closes the cycle (source first, derived version last).
119        path: Vec<VersionId>,
120    },
121
122    /// A derivation was already recorded for this derived version. Lineage is
123    /// write-once per version (versions are immutable); re-declaring is
124    /// rejected rather than silently overwriting recorded history.
125    #[error("lineage already recorded for fact {0}")]
126    AlreadyRecorded(VersionId),
127}
128
129fn format_cycle(path: &[VersionId]) -> String {
130    path.iter()
131        .map(|v| v.as_u64().to_string())
132        .collect::<Vec<_>>()
133        .join(" -> ")
134}

136/// Options bounding a lineage closure query.
137#[derive(Debug, Clone, Copy)]
138pub struct LineageQueryOptions {
139    /// Maximum transitive depth to expand (hops from the root). Depth `0`
140    /// returns an empty closure; depth `1` returns only direct
141    /// parents/children.
142    pub max_depth: usize,
143    /// Maximum number of entries to return. When the traversal would exceed
144    /// this, it stops and the closure's `has_more` flag is set.
145    pub limit: usize,
146    /// Optional `AS OF` transaction-time bound: only lineage records with
147    /// `recorded_at <= as_of` are followed, so the closure reflects lineage as
148    /// it was recorded by that transaction time. `None` uses all records.
149    pub as_of: Option<Timestamp>,
150}
151
152impl LineageQueryOptions {
153    /// Default depth cap used when a caller does not specify one.
154    pub const DEFAULT_MAX_DEPTH: usize = 32;
155    /// Default entry cap used when a caller does not specify one.
156    pub const DEFAULT_LIMIT: usize = 1000;
157
158    /// Options with the default depth and limit and no `AS OF` bound.
159    pub fn new() -> Self {
160        Self {
161            max_depth: Self::DEFAULT_MAX_DEPTH,
162            limit: Self::DEFAULT_LIMIT,
163            as_of: None,
164        }
165    }
166
167    /// Set the maximum transitive depth.
168    #[must_use]
169    pub fn with_max_depth(mut self, max_depth: usize) -> Self {
170        self.max_depth = max_depth;
171        self
172    }
173
174    /// Set the maximum number of returned entries.
175    #[must_use]
176    pub fn with_limit(mut self, limit: usize) -> Self {
177        self.limit = limit;
178        self
179    }
180
181    /// Set the `AS OF` transaction-time bound.
182    #[must_use]
183    pub fn with_as_of(mut self, as_of: Timestamp) -> Self {
184        self.as_of = Some(as_of);
185        self
186    }
187}

195/// One node in a lineage closure: a referenced fact version and the minimum
196/// number of hops at which it was reached from the query root.
197#[derive(Debug, Clone, Copy, PartialEq, Eq)]
198pub struct LineageEntry {
199    /// The reached fact version.
200    pub reference: LineageRef,
201    /// Minimum hop distance from the query root (`1` = direct
202    /// parent/child).
203    pub depth: usize,
204}
205
206/// The result of an upstream or downstream lineage closure query.
207///
208/// Entries are ordered by increasing depth (breadth-first), then by the order
209/// they were discovered, and never include the query root itself.
210#[derive(Debug, Clone, PartialEq, Eq)]
211pub struct LineageClosure {
212    /// The fact the closure was computed from.
213    pub root: LineageRef,
214    /// The reachable fact versions with their minimum depth.
215    pub entries: Vec<LineageEntry>,
216    /// `true` if traversal stopped because it hit the entry `limit` or the
217    /// `max_depth` bound while more of the graph remained unexplored — mirrors
218    /// the `has_more` pagination convention (Issue #3226).
219    pub has_more: bool,
220}
221
222impl LineageClosure {
223    /// Number of distinct fact versions in the closure (excludes the root).
224    #[inline]
225    pub fn count(&self) -> usize {
226        self.entries.len()
227    }
228
229    /// The greatest depth present in the closure (`0` for an empty closure).
230    pub fn max_depth(&self) -> usize {
231        self.entries.iter().map(|e| e.depth).max().unwrap_or(0)
232    }
233}

235/// An immutable, in-memory index of fact-to-fact derivation lineage.
236///
237/// Maintains two adjacency maps keyed by the globally-unique [`VersionId`]:
238/// `upstream` (a derived version -> the record naming its sources) and
239/// `downstream` (a source version -> the versions derived from it). Closures
240/// in either direction are breadth-first walks over these maps.
241#[derive(Debug, Default)]
242pub struct LineageStore {
243    /// derived version -> its lineage record (one per version; write-once).
244    upstream: DashMap<VersionId, LineageRecord>,
245    /// source version -> derived versions that declared it as a source.
246    downstream: DashMap<VersionId, Vec<VersionId>>,
247    /// Serializes the check-then-act critical section of [`Self::record`] so
248    /// the `contains_key` write-once check, the cycle scan, and the paired
249    /// `upstream`/`downstream` inserts are one atomic step. Without it, two
250    /// concurrent records of the *same* derived version could both pass the
251    /// `contains_key` check and the second would silently overwrite the first
252    /// (a lost update violating write-once). Reads (closure queries) never
253    /// take this lock, so they stay fully concurrent — only the low-frequency
254    /// write path is serialized.
255    write_lock: parking_lot::Mutex<()>,
256}
257
258impl LineageStore {
259    /// Create an empty lineage store.
260    pub fn new() -> Self {
261        Self::default()
262    }
263
264    /// Number of derivation records held.
265    pub fn len(&self) -> usize {
266        self.upstream.len()
267    }
268
269    /// Whether any derivation has been recorded.
270    pub fn is_empty(&self) -> bool {
271        self.upstream.is_empty()
272    }
273
274    /// Record that `derived` was computed from `sources` at `recorded_at`.
275    ///
276    /// Sources are deduplicated preserving first-seen order. An empty (or
277    /// entirely self-referential) source set is a no-op that records nothing
278    /// and returns `Ok(())` — callers normalise "no lineage" to not calling
279    /// this at all, but an all-duplicates-removed empty set must not fabricate
280    /// a record.
281    ///
282    /// # Errors
283    ///
284    /// - [`LineageError::SelfDerivation`] if a source names the derived
285    ///   version.
286    /// - [`LineageError::CycleDetected`] if the derived version is already
287    ///   reachable upstream from a source (structurally impossible via the
288    ///   normal write path; see the module docs).
289    /// - [`LineageError::AlreadyRecorded`] if a derivation was already
290    ///   recorded for `derived.version` (lineage is write-once per version).
291    pub fn record(
292        &self,
293        derived: LineageRef,
294        sources: Vec<LineageRef>,
295        recorded_at: Timestamp,
296    ) -> Result<(), LineageError> {
297        // Deduplicate sources, preserving declaration order, and reject
298        // self-derivation. This is pure (touches no shared state) so it runs
299        // outside the write lock.
300        let mut seen = HashSet::new();
301        let mut deduped: Vec<LineageRef> = Vec::with_capacity(sources.len());
302        for source in sources {
303            if source.version == derived.version {
304                return Err(LineageError::SelfDerivation(derived.version));
305            }
306            if seen.insert(source.version) {
307                deduped.push(source);
308            }
309        }
310
311        if deduped.is_empty() {
312            return Ok(());
313        }
314
315        // Everything below mutates or reads-for-mutation the shared adjacency
316        // maps and must be a single atomic step: two concurrent records of the
317        // same derived version must not both pass the write-once check, and the
318        // cycle scan must see a consistent graph. Hold the write lock across the
319        // whole check-then-act region. (Reads never take this lock.)
320        let _guard = self.write_lock.lock();
321
322        if self.upstream.contains_key(&derived.version) {
323            return Err(LineageError::AlreadyRecorded(derived.version));
324        }
325
326        // Cycle guard: for each source, ensure the derived version is not
327        // already reachable by walking upstream from that source.
328        for source in &deduped {
329            if let Some(path) = self.upstream_path_to(source.version, derived.version) {
330                return Err(LineageError::CycleDetected { path });
331            }
332        }
333
334        for source in &deduped {
335            self.downstream
336                .entry(source.version)
337                .or_default()
338                .push(derived.version);
339        }
340        self.upstream.insert(
341            derived.version,
342            LineageRecord {
343                derived,
344                sources: deduped,
345                recorded_at,
346            },
347        );
348        Ok(())
349    }
350
351    /// The recorded derivation for `version`, if any.
352    pub fn record_for(&self, version: VersionId) -> Option<LineageRecord> {
353        self.upstream.get(&version).map(|r| r.value().clone())
354    }
355
356    /// Direct sources (parents) of `version`.
357    pub fn direct_upstream(&self, version: VersionId) -> Vec<LineageRef> {
358        self.upstream
359            .get(&version)
360            .map(|r| r.value().sources.clone())
361            .unwrap_or_default()
362    }
363
364    /// If `derived` is reachable by walking upstream from `start`, return the
365    /// path `[start, ..., derived]`; otherwise `None`. Used for cycle
366    /// detection.
367    fn upstream_path_to(&self, start: VersionId, target: VersionId) -> Option<Vec<VersionId>> {
368        // BFS with parent tracking so we can reconstruct the path.
369        let mut visited = HashSet::new();
370        let mut parents: std::collections::HashMap<VersionId, VersionId> =
371            std::collections::HashMap::new();
372        let mut queue = VecDeque::new();
373        visited.insert(start);
374        queue.push_back(start);
375        while let Some(current) = queue.pop_front() {
376            if current == target {
377                // Reconstruct path from start to target.
378                let mut path = vec![target];
379                let mut node = target;
380                while let Some(&parent) = parents.get(&node) {
381                    path.push(parent);
382                    node = parent;
383                }
384                path.reverse();
385                return Some(path);
386            }
387            if let Some(record) = self.upstream.get(&current) {
388                for source in &record.value().sources {
389                    if visited.insert(source.version) {
390                        parents.insert(source.version, current);
391                        queue.push_back(source.version);
392                    }
393                }
394            }
395        }
396        None
397    }

399    /// Compute the transitive upstream closure ("what was this derived
400    /// from?") of `root`, bounded by `options`.
401    pub fn upstream_closure(
402        &self,
403        root: LineageRef,
404        options: LineageQueryOptions,
405    ) -> LineageClosure {
406        self.closure(root, options, Direction::Upstream)
407    }
408
409    /// Compute the transitive downstream closure ("what has been derived from
410    /// this?") of `root`, bounded by `options` — the retraction blast-radius
411    /// report.
412    pub fn downstream_closure(
413        &self,
414        root: LineageRef,
415        options: LineageQueryOptions,
416    ) -> LineageClosure {
417        self.closure(root, options, Direction::Downstream)
418    }
419
420    fn closure(
421        &self,
422        root: LineageRef,
423        options: LineageQueryOptions,
424        direction: Direction,
425    ) -> LineageClosure {
426        let mut entries = Vec::new();
427        let mut visited = HashSet::new();
428        visited.insert(root.version);
429        let mut has_more = false;
430
431        // BFS frontier of (reference, depth).
432        let mut frontier: Vec<LineageRef> = vec![root];
433        let mut depth = 0usize;
434
435        'outer: while !frontier.is_empty() && depth < options.max_depth {
436            depth += 1;
437            let mut next: Vec<LineageRef> = Vec::new();
438            for node in &frontier {
439                let neighbours = match direction {
440                    Direction::Upstream => self.neighbours_upstream(node.version, options.as_of),
441                    Direction::Downstream => {
442                        self.neighbours_downstream(node.version, options.as_of)
443                    }
444                };
445                for neighbour in neighbours {
446                    if !visited.insert(neighbour.version) {
447                        continue;
448                    }
449                    if entries.len() >= options.limit {
450                        has_more = true;
451                        break 'outer;
452                    }
453                    entries.push(LineageEntry {
454                        reference: neighbour,
455                        depth,
456                    });
457                    next.push(neighbour);
458                }
459            }
460            // If we stopped expanding because of the depth cap, flag has_more
461            // only when the unexplored frontier actually has an unvisited
462            // neighbour beyond the cap — a node whose one more hop would reveal
463            // a new entry. A frontier of pure leaves (nothing further upstream/
464            // downstream) means the closure is complete, so has_more stays
465            // false (returning exactly the chain at its own length is not
466            // "truncated").
467            if depth == options.max_depth
468                && self.frontier_has_unvisited_neighbour(&next, direction, options.as_of, &visited)
469            {
470                has_more = true;
471            }
472            frontier = next;
473        }
474
475        LineageClosure {
476            root,
477            entries,
478            has_more,
479        }
480    }
481
482    /// Whether any node in the (depth-capped) frontier has at least one
483    /// neighbour, in `direction`, that has not already been visited — i.e.
484    /// expanding one more hop would yield a genuinely new closure entry. Used
485    /// to decide `has_more` precisely at the depth cap so a fully-returned
486    /// chain is not falsely marked truncated.
487    fn frontier_has_unvisited_neighbour(
488        &self,
489        frontier: &[LineageRef],
490        direction: Direction,
491        as_of: Option<Timestamp>,
492        visited: &HashSet<VersionId>,
493    ) -> bool {
494        frontier.iter().any(|node| {
495            let neighbours = match direction {
496                Direction::Upstream => self.neighbours_upstream(node.version, as_of),
497                Direction::Downstream => self.neighbours_downstream(node.version, as_of),
498            };
499            neighbours.iter().any(|nb| !visited.contains(&nb.version))
500        })
501    }
502
503    /// Upstream neighbours of `version` (its declared sources), filtered by the
504    /// optional `as_of` transaction-time bound.
505    fn neighbours_upstream(&self, version: VersionId, as_of: Option<Timestamp>) -> Vec<LineageRef> {
506        match self.upstream.get(&version) {
507            Some(record) => {
508                let record = record.value();
509                if let Some(as_of) = as_of
510                    && record.recorded_at > as_of
511                {
512                    return Vec::new();
513                }
514                record.sources.clone()
515            }
516            None => Vec::new(),
517        }
518    }
519
520    /// Downstream neighbours of `version` (versions derived from it), filtered
521    /// by the optional `as_of` transaction-time bound.
522    fn neighbours_downstream(
523        &self,
524        version: VersionId,
525        as_of: Option<Timestamp>,
526    ) -> Vec<LineageRef> {
527        let derived_versions = match self.downstream.get(&version) {
528            Some(list) => list.value().clone(),
529            None => return Vec::new(),
530        };
531        let mut out = Vec::with_capacity(derived_versions.len());
532        for dv in derived_versions {
533            if let Some(record) = self.upstream.get(&dv) {
534                let record = record.value();
535                if let Some(as_of) = as_of
536                    && record.recorded_at > as_of
537                {
538                    continue;
539                }
540                out.push(record.derived);
541            }
542        }
543        out
544    }
545}

834    #[test]
835    fn concurrent_records_from_shared_source_no_lost_updates() {
836        // N threads each record a DISTINCT derived version from one shared
837        // source. The check-then-act critical section must be atomic, so every
838        // record survives (write-once is not violated by a lost update) and the
839        // downstream closure of the shared source counts exactly N.
840        use std::sync::Arc;
841        use std::thread;
842
843        const N: u64 = 64;
844        let store = Arc::new(LineageStore::new());
845        let source = node_ref(1, 1);
846
847        let mut handles = Vec::new();
848        for i in 0..N {
849            let store = Arc::clone(&store);
850            handles.push(thread::spawn(move || {
851                // Distinct derived version ids (>= 100) per thread.
852                let derived = node_ref(100 + i, 100 + i);
853                store
854                    .record(derived, vec![source], ts(1))
855                    .expect("distinct derived version records cleanly");
856            }));
857        }
858        for h in handles {
859            h.join().expect("thread panicked");
860        }
861
862        assert_eq!(store.len(), N as usize);
863        let closure = store.downstream_closure(
864            source,
865            LineageQueryOptions::new().with_limit(N as usize + 10),
866        );
867        assert_eq!(closure.count(), N as usize);
868        assert!(!closure.has_more);
869    }
