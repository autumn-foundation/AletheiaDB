1//! Graph-wide temporal changefeed types (Issue #3216).
2//!
3//! These types back [`crate::db::AletheiaDB::list_changes`], a read-only API that
4//! enumerates the entities (nodes **and** edges) whose versions were committed within a
5//! transaction-time window `[t1, t2)`. This is the *discovery* half of the bi-temporal
6//! story: callers can ask "what changed between T1 and T2?" without already knowing any
7//! entity IDs, then drill into each result with the existing per-entity history/diff APIs.
8//!
9//! The feed is **bounded** (a `limit` plus a stable, opaque continuation cursor) and
10//! **deterministically ordered** (transaction-time ascending, then entity kind, id, and
11//! version) so that paginated reads are stable and replayable.
12
13use crate::core::error::{Error, QueryError};
14use crate::core::interning::{GLOBAL_INTERNER, InternedString};
15use crate::core::namespace::{Namespace, NamespaceId, resolve_namespace_id};
16use crate::core::temporal::{BiTemporalInterval, TimeRange, Timestamp};

18/// Whether a change record refers to a node or an edge.
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
20pub enum EntityKind {
21    /// The change refers to a node.
22    Node,
23    /// The change refers to an edge.
24    Edge,
25}
26
27impl EntityKind {
28    /// Stable ordinal used for deterministic ordering and cursor encoding.
29    ///
30    /// Defined explicitly (not via enum discriminant) so ordering is unaffected by
31    /// future reordering of the enum variants.
32    #[inline]
33    pub const fn ord(self) -> u8 {
34        match self {
35            EntityKind::Node => 0,
36            EntityKind::Edge => 1,
37        }
38    }
39
40    /// Inverse of [`EntityKind::ord`] for values produced by this crate (0 = node, else edge).
41    #[inline]
42    pub(crate) const fn from_ord(ord: u8) -> EntityKind {
43        if ord == 0 {
44            EntityKind::Node
45        } else {
46            EntityKind::Edge
47        }
48    }
49
50    /// Lowercase string form used in MCP/JSON responses.
51    #[inline]
52    pub const fn as_str(self) -> &'static str {
53        match self {
54            EntityKind::Node => "node",
55            EntityKind::Edge => "edge",
56        }
57    }
58}

60/// Classification of a single committed version relative to the entity's history.
61#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
62pub enum ChangeType {
63    /// The first version of the entity (no previous version).
64    Created,
65    /// A subsequent, non-deleting version of an existing entity.
66    Modified,
67    /// A tombstone version: the entity was deleted (valid-time closed to an empty range).
68    Deleted,
69}
70
71impl ChangeType {
72    /// Lowercase string form used in MCP/JSON responses.
73    #[inline]
74    pub const fn as_str(self) -> &'static str {
75        match self {
76            ChangeType::Created => "created",
77            ChangeType::Modified => "modified",
78            ChangeType::Deleted => "deleted",
79        }
80    }
81}

83/// One row of the changefeed: a single entity version committed within the query window.
84#[derive(Debug, Clone, PartialEq, Eq)]
85pub struct ChangeRecord {
86    /// Entity id (`NodeId`/`EdgeId` as `u64`).
87    pub entity_id: u64,
88    /// Version id of the specific version this row describes (use with the per-entity
89    /// history/diff APIs to drill in).
90    pub version_id: u64,
91    /// Whether this is a node or an edge.
92    pub kind: EntityKind,
93    /// How the entity changed at this version.
94    pub change_type: ChangeType,
95    /// Resolved node label / edge type.
96    pub label: String,
97    /// Full transaction-time range of the version.
98    pub transaction_time_range: TimeRange,
99    /// Full valid-time range of the version (an empty range denotes a deletion).
100    pub valid_time_range: TimeRange,
101    /// The namespace the affected entity belongs to (Issue #3349, PR3c).
102    ///
103    /// Derived from the entity's immutable ride-along namespace
104    /// ([`crate::core::namespace::NAMESPACE_KEY`]); a legacy / default entity
105    /// resolves to [`Namespace::default`]. Surfaced as a first-class field so the
106    /// reserved ride-along key never leaks into a changefeed payload, and used by
107    /// [`crate::core::changefeed_subscription::ChangeFilter`] to scope a
108    /// subscription to a namespace.
109    pub namespace: Namespace,
110}
111
112impl ChangeRecord {
113    /// Commit timestamp of this version (== `transaction_time_range.start()`).
114    ///
115    /// Derived rather than stored so it can never disagree with the range it comes from.
116    #[inline]
117    pub fn transaction_time(&self) -> Timestamp {
118        self.transaction_time_range.start()
119    }
120
121    /// The stable total-order [`ChangeCursor`] identifying this record.
122    ///
123    /// This is the dedup / resume key for the push changefeed (Issue #3375): two
124    /// records are the same committed fact iff their cursors are equal, and a
125    /// consumer resumes a pull with `list_changes(cursor = last_delivered.cursor())`.
126    /// Derived from the record's own fields so it always agrees with what the
127    /// #3216 scan would have produced for the same version.
128    #[inline]
129    pub(crate) fn cursor(&self) -> ChangeCursor {
130        let start = self.transaction_time_range.start();
131        ChangeCursor {
132            tx_wallclock: start.wallclock(),
133            tx_logical: start.logical(),
134            kind_ord: self.kind.ord(),
135            entity_id: self.entity_id,
136            version_id: self.version_id,
137        }
138    }
139}
140
141/// Total-order sort key used for deterministic ordering and cursor pagination.
142///
143/// Ordering is by `(tx_wallclock, tx_logical, kind_ord, entity_id, version_id)` ascending.
144/// `version_id` is included as the final component so the key is **unique per emitted version**
145/// even when one transaction commits two versions of the same entity (which share a commit
146/// timestamp): without it, two such rows would collide and one would be silently dropped at a
147/// page boundary by the strict `> cursor` resume rule.
148#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
149pub(crate) struct ChangeCursor {
150    pub tx_wallclock: i64,
151    pub tx_logical: u32,
152    pub kind_ord: u8,
153    pub entity_id: u64,
154    pub version_id: u64,
155}
156
157impl ChangeCursor {
158    /// Encode this key as an opaque, URL-safe continuation token.
159    ///
160    /// The underlying form is a delimited string hex-encoded so clients treat it as
161    /// opaque and never attempt to parse or construct it by hand.
162    pub(crate) fn encode(&self) -> String {
163        let raw = format!(
164            "{}:{}:{}:{}:{}",
165            self.tx_wallclock, self.tx_logical, self.kind_ord, self.entity_id, self.version_id
166        );
167        crate::core::hex::encode(raw.as_bytes())
168    }
169
170    /// Encode a **baseline resume anchor** positioned strictly after every version committed
171    /// at or before `frontier` (Issue #3375 review F4).
172    ///
173    /// Uses `frontier`'s `(wallclock, logical)` with maximal kind/entity/version tiebreakers,
174    /// so a `list_changes` resume from this token (strict `> cursor`) returns exactly the
175    /// changes committed *after* `frontier` — i.e. everything a subscriber that captured
176    /// `frontier` at subscribe time could have missed, and nothing it should not re-see.
177    /// Because every future commit gets a strictly greater HLC timestamp, no future event is
178    /// ever excluded by this anchor.
179    pub(crate) fn baseline_after(frontier: Timestamp) -> String {
180        ChangeCursor {
181            tx_wallclock: frontier.wallclock(),
182            tx_logical: frontier.logical(),
183            kind_ord: u8::MAX,
184            entity_id: u64::MAX,
185            version_id: u64::MAX,
186        }
187        .encode()
188    }
189
190    /// The minimal [`ChangeCursor`] positioned at a transaction time `tx` (Issue #3677).
191    ///
192    /// All the cursor tie-breakers (`kind_ord`, `entity_id`, `version_id`) are zero, so this
193    /// sorts at or before every real version committed at `tx` and strictly after every version
194    /// committed before `tx`. Used by the cold-tier directory to bound a `[start, end)`
195    /// transaction-time window: `min_at(start)` (inclusive) and `min_at(end)` (exclusive) exactly
196    /// select the versions whose commit timestamp lies in the half-open window, mirroring
197    /// [`TimeRange::contains`](crate::core::temporal::TimeRange::contains).
198    pub(crate) fn min_at(tx: Timestamp) -> Self {
199        ChangeCursor {
200            tx_wallclock: tx.wallclock(),
201            tx_logical: tx.logical(),
202            kind_ord: 0,
203            entity_id: 0,
204            version_id: 0,
205        }
206    }
207
208    /// The [`ChangeCursor`] identifying a specific committed version (Issue #3677).
209    ///
210    /// Mirrors the cursor [`build_raw_change`] computes for the same version, so a directory entry
211    /// built from a migrated version is byte-identical to what the changefeed scan would produce.
212    /// `tx_start` is the version's transaction-time interval start.
213    pub(crate) fn for_version(
214        tx_start: Timestamp,
215        kind: EntityKind,
216        entity_id: u64,
217        version_id: u64,
218    ) -> Self {
219        ChangeCursor {
220            tx_wallclock: tx_start.wallclock(),
221            tx_logical: tx_start.logical(),
222            kind_ord: kind.ord(),
223            entity_id,
224            version_id,
225        }
226    }
227
228    /// Decode an opaque continuation token produced by [`ChangeCursor::encode`].
229    ///
230    /// Returns a `QueryError::InvalidParameter` (never panics) when the token is malformed.
231    pub(crate) fn decode(token: &str) -> Result<Self, Error> {
232        let bytes =
233            crate::core::hex::decode(token).ok_or_else(|| invalid_cursor("not valid hex"))?;
234        let raw = std::str::from_utf8(&bytes).map_err(|_| invalid_cursor("not valid utf-8"))?;
235        let mut parts = raw.split(':');
236        let tx_wallclock = parse_field(parts.next())?;
237        let tx_logical = parse_field(parts.next())?;
238        let kind_ord = parse_field(parts.next())?;
239        let entity_id = parse_field(parts.next())?;
240        let version_id = parse_field(parts.next())?;
241        if parts.next().is_some() {
242            return Err(invalid_cursor("too many fields"));
243        }
244        Ok(ChangeCursor {
245            tx_wallclock,
246            tx_logical,
247            kind_ord,
248            entity_id,
249            version_id,
250        })
251    }
252}
253
254fn parse_field<T: std::str::FromStr>(field: Option<&str>) -> Result<T, Error> {
255    field
256        .ok_or_else(|| invalid_cursor("missing field"))?
257        .parse::<T>()
258        .map_err(|_| invalid_cursor("unparseable field"))
259}
260
261fn invalid_cursor(reason: &str) -> Error {
262    Error::Query(QueryError::InvalidParameter {
263        parameter: "cursor".to_string(),
264        reason: format!("malformed continuation token: {reason}"),
265    })
266}

268/// A changefeed row in its cheap, pre-materialization form.
269///
270/// This is what the storage scan produces: it carries the precomputed sort [`ChangeCursor`] and
271/// the label as an [`InternedString`] (no owned `String`). The expensive label resolution is
272/// deferred to [`RawChange::into_record`], which the query layer calls only for the rows that
273/// actually survive cursor filtering and the page `limit` — so a wide window over a large store
274/// never allocates a label per matching version just to discard it.
275#[derive(Debug, Clone)]
276pub(crate) struct RawChange {
277    pub cursor: ChangeCursor,
278    pub change_type: ChangeType,
279    pub label_id: InternedString,
280    /// Interned namespace id of the affected entity (Issue #3349, PR3c). Kept as a
281    /// `Copy` [`NamespaceId`] (not an owned [`Namespace`]) so, like `label_id`, the
282    /// name resolution is deferred to [`RawChange::into_record`] and paid only for
283    /// rows that survive cursor filtering and the page `limit`.
284    pub namespace_id: NamespaceId,
285    pub transaction_time_range: TimeRange,
286    pub valid_time_range: TimeRange,
287}
288
289impl RawChange {
290    /// Resolve the label and namespace and materialize a public [`ChangeRecord`].
291    pub(crate) fn into_record(self) -> ChangeRecord {
292        let label = GLOBAL_INTERNER
293            .resolve_with(self.label_id, |s| s.to_string())
294            .unwrap_or_default();
295        // A namespace id is interned at derivation time, so it always resolves; the
296        // `unwrap_or_default` is defensive (a `default`-namespace entity resolves to
297        // the default namespace either way).
298        let namespace = resolve_namespace_id(self.namespace_id).unwrap_or_default();
299        ChangeRecord {
300            entity_id: self.cursor.entity_id,
301            version_id: self.cursor.version_id,
302            kind: EntityKind::from_ord(self.cursor.kind_ord),
303            change_type: self.change_type,
304            label,
305            transaction_time_range: self.transaction_time_range,
306            valid_time_range: self.valid_time_range,
307            namespace,
308        }
309    }
310}

312/// Build a [`RawChange`] for a single committed version if it passes all filters.
313///
314/// Shared by the node and edge scans (hot and cold tiers) — they differ only in id accessor
315/// and [`EntityKind`]. Returns `None` when the version falls outside the transaction-time
316/// window, the optional valid-time window, or the optional label filter.
317///
318/// Valid-time filtering treats a version's *valid extent* and the window as sets and includes
319/// the version iff they intersect. A live version has a non-empty extent, so this is a range
320/// overlap. A deletion (tombstone) has an empty valid range `[v, v)` — a degenerate point at
321/// the deletion instant `v` — so it intersects the half-open window iff the window contains
322/// that single instant. The two predicates are the same intersection rule applied to a range
323/// vs. a point.
324///
325/// `namespace_fn` derives the affected entity's interned [`NamespaceId`] (Issue #3349,
326/// PR3c). It is invoked **lazily** — only after every cheap filter (transaction-time
327/// window, valid-time window, label) has passed — so a candidate that will be discarded
328/// never pays the namespace derivation (which, for a delta version, may reconstruct
329/// properties). It is not part of the sort [`ChangeCursor`], so record ordering is
330/// unchanged.
331#[allow(clippy::too_many_arguments)]
332pub(crate) fn build_raw_change(
333    version_id: u64,
334    entity_id: u64,
335    kind: EntityKind,
336    temporal: &BiTemporalInterval,
337    label_id: InternedString,
338    prev_is_none: bool,
339    tx_window: &TimeRange,
340    valid_window: Option<&TimeRange>,
341    label_filter: Option<&str>,
342    namespace_fn: impl FnOnce() -> NamespaceId,
343) -> Option<RawChange> {
344    let tx_range = temporal.transaction_time();
345    // Transaction-time window is half-open [t1, t2).
346    if !tx_window.contains(tx_range.start()) {
347        return None;
348    }
349
350    let valid_range = temporal.valid_time();
351    let is_deletion = valid_range.is_empty();
352
353    if let Some(vw) = valid_window {
354        // Intersection of the version's valid extent with the window (tombstone = point).
355        let intersects = if is_deletion {
356            vw.contains(valid_range.start())
357        } else {
358            valid_range.overlaps(vw)
359        };
360        if !intersects {
361            return None;
362        }
363    }
364
365    if let Some(filter) = label_filter
366        && GLOBAL_INTERNER.resolve_with(label_id, |s| s == filter) != Some(true)
367    {
368        return None;
369    }
370
371    let change_type = if is_deletion {
372        ChangeType::Deleted
373    } else if prev_is_none {
374        ChangeType::Created
375    } else {
376        ChangeType::Modified
377    };
378
379    Some(RawChange {
380        cursor: ChangeCursor {
381            tx_wallclock: tx_range.start().wallclock(),
382            tx_logical: tx_range.start().logical(),
383            kind_ord: kind.ord(),
384            entity_id,
385            version_id,
386        },
387        change_type,
388        label_id,
389        namespace_id: namespace_fn(),
390        transaction_time_range: tx_range,
391        valid_time_range: valid_range,
392    })
393}

395/// A streaming, bounded-by-cursor accumulator of [`RawChange`]s (Issue #3216, PR 2).
396///
397/// Retains only the `bound`-smallest changes offered to it (by [`ChangeCursor`] order), so a
398/// windowed changefeed scan over a large tier holds `O(bound)` rows in memory instead of every
399/// match. This is the **limit pushdown** for `list_changes`: the query layer passes
400/// `page_limit + 1` as the bound (the `+1` lets it detect `has_more` without materializing the
401/// rest), and because the retained set always contains the true `bound`-smallest survivors, the
402/// page the query layer then selects is byte-identical to one produced by an unbounded scan.
403///
404/// Applying the bound **per tier** is sound because the page is the `page_limit`-smallest of the
405/// tiers' union: any survivor beyond a tier's `bound`-smallest has at least `bound` smaller
406/// survivors within that same tier — hence at least `page_limit` smaller survivors in the union —
407/// and can never reach the page.
408pub(crate) struct BoundedChanges {
409    bound: usize,
410    inner: BoundedInner,
411}
412
413/// Upper limit on the heap capacity we pre-reserve, so a large `bound` (e.g. an MCP
414/// `limit = 10_000` catch-up) does not attempt a giant up-front allocation while still avoiding
415/// the handful of reallocations a `bound`-sized scan would otherwise incur.
416const HEAP_PRESIZE_CAP: usize = 4096;
417
418/// Storage strategy for [`BoundedChanges`], chosen once at construction from `bound`.
419enum BoundedInner {
420    /// Unbounded fast path (`bound == usize::MAX`): accumulate every offered change into a plain
421    /// `Vec` in `O(M)`. When nothing can ever be evicted (the common "drain everything" /
422    /// large-limit case) the max-heap's `O(M log M)` maintenance buys no memory benefit — it just
423    /// re-heapifies a set the caller sorts anyway — so we skip it entirely.
424    Unbounded(Vec<RawChange>),
425    /// Bounded path: max-heap keyed by cursor, so the current largest-cursor survivor sits at the
426    /// top and is the one evicted once the heap exceeds `bound`.
427    Bounded(std::collections::BinaryHeap<ByCursor>),
428}
429
430/// [`RawChange`] wrapper ordered solely by its [`ChangeCursor`] (which is unique per emitted
431/// version), so a [`std::collections::BinaryHeap`] evicts the largest-cursor survivor.
432struct ByCursor(RawChange);
433
434impl PartialEq for ByCursor {
435    fn eq(&self, other: &Self) -> bool {
436        self.0.cursor == other.0.cursor
437    }
438}
439impl Eq for ByCursor {}
440impl PartialOrd for ByCursor {
441    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
442        Some(self.cmp(other))
443    }
444}
445impl Ord for ByCursor {
446    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
447        self.0.cursor.cmp(&other.0.cursor)
448    }
449}
450
451impl BoundedChanges {
452    /// Create an accumulator that keeps at most `bound` (min 1) smallest-cursor changes.
453    ///
454    /// A `bound` of `usize::MAX` selects the unbounded fast path (plain `Vec`, no heap); any
455    /// finite `bound` selects the bounded max-heap, pre-sized to `bound` (capped by
456    /// [`HEAP_PRESIZE_CAP`]).
457    pub(crate) fn new(bound: usize) -> Self {
458        let bound = bound.max(1);
459        let inner = if bound == usize::MAX {
460            BoundedInner::Unbounded(Vec::new())
461        } else {
462            BoundedInner::Bounded(std::collections::BinaryHeap::with_capacity(
463                bound.min(HEAP_PRESIZE_CAP),
464            ))
465        };
466        BoundedChanges { bound, inner }
467    }
468
469    /// Offer a change; the accumulator keeps only the `bound`-smallest by cursor seen so far.
470    pub(crate) fn consider(&mut self, change: RawChange) {
471        match &mut self.inner {
472            // Unbounded: keep everything (no eviction is ever possible).
473            BoundedInner::Unbounded(v) => v.push(change),
474            BoundedInner::Bounded(heap) => {
475                heap.push(ByCursor(change));
476                if heap.len() > self.bound {
477                    // Evict the current largest cursor — it cannot belong on any page of size < bound.
478                    heap.pop();
479                }
480            }
481        }
482    }
483
484    /// Number of changes currently retained.
485    ///
486    /// Used by the cold-tier directory pushdown (Issue #3677) to early-stop a bounded,
487    /// ascending-cursor point-read walk: once `len() >= bound` no later (larger-cursor) candidate
488    /// can displace a retained survivor, so the walk may stop.
489    pub(crate) fn len(&self) -> usize {
490        match &self.inner {
491            BoundedInner::Unbounded(v) => v.len(),
492            BoundedInner::Bounded(heap) => heap.len(),
493        }
494    }
495
496    /// Consume the accumulator, returning the retained changes (unordered; the caller sorts).
497    pub(crate) fn into_vec(self) -> Vec<RawChange> {
498        match self.inner {
499            BoundedInner::Unbounded(v) => v,
500            BoundedInner::Bounded(heap) => heap.into_iter().map(|w| w.0).collect(),
501        }
502    }
503}

505/// Build a [`RawChange`] for one version and, if it passes every filter **and** sorts strictly
506/// after `resume_after`, offer it to `acc` (Issue #3216, PR 2).
507///
508/// Shared verbatim by the hot ([`crate::storage::historical`]) and cold
509/// ([`crate::storage::redb_cold_storage`]) changefeed scans so their filtering, cursor-resume,
510/// and bounding are identical by construction — the guarantee that makes the pushed-down page
511/// byte-identical to the legacy unbounded merge.
512#[allow(clippy::too_many_arguments)]
513pub(crate) fn consider_version(
514    acc: &mut BoundedChanges,
515    resume_after: Option<ChangeCursor>,
516    version_id: u64,
517    entity_id: u64,
518    kind: EntityKind,
519    temporal: &BiTemporalInterval,
520    label_id: InternedString,
521    prev_is_none: bool,
522    tx_window: &TimeRange,
523    valid_window: Option<&TimeRange>,
524    label_filter: Option<&str>,
525    namespace_fn: impl FnOnce() -> NamespaceId,
526) {
527    if let Some(rec) = build_raw_change(
528        version_id,
529        entity_id,
530        kind,
531        temporal,
532        label_id,
533        prev_is_none,
534        tx_window,
535        valid_window,
536        label_filter,
537        namespace_fn,
538    )
539    .filter(|rec| resume_after.is_none_or(|c| rec.cursor > c))
540    {
541        acc.consider(rec);
542    }
543}

545/// Query options for [`crate::db::AletheiaDB::list_changes`].
546///
547/// The transaction-time window is required; everything else is optional. An empty window
548/// (`tx_from == tx_to`) is valid and yields an empty page (it is not an error). A window with
549/// `tx_from > tx_to` is rejected with `TemporalError::InvalidTimeRange`.
550#[derive(Debug, Clone)]
551pub struct ChangeFeedQuery {
552    /// Inclusive lower bound of the transaction-time window.
553    pub tx_from: Timestamp,
554    /// Exclusive upper bound of the transaction-time window.
555    pub tx_to: Timestamp,
556    /// Optional inclusive lower bound of a valid-time constraint. Must be paired with
557    /// [`ChangeFeedQuery::valid_to`].
558    pub valid_from: Option<Timestamp>,
559    /// Optional exclusive upper bound of a valid-time constraint. Must be paired with
560    /// [`ChangeFeedQuery::valid_from`].
561    pub valid_to: Option<Timestamp>,
562    /// Optional node-label / edge-type filter (exact string match).
563    pub label: Option<String>,
564    /// Maximum number of rows to return in this page. A value of `0` is treated as `1`
565    /// (a page must be able to carry a continuation cursor, so it cannot be empty when more
566    /// rows exist).
567    pub limit: usize,
568    /// Opaque continuation token from a previous page's `next_cursor` (`None` = first page).
569    pub cursor: Option<String>,
570}
571
572impl ChangeFeedQuery {
573    /// Construct a query over a transaction-time window with default options
574    /// (no valid-time or label filter, given `limit`, first page).
575    pub fn new(tx_from: Timestamp, tx_to: Timestamp, limit: usize) -> Self {
576        ChangeFeedQuery {
577            tx_from,
578            tx_to,
579            valid_from: None,
580            valid_to: None,
581            label: None,
582            limit,
583            cursor: None,
584        }
585    }
586}

643    #[test]
644    fn cursor_ordering_is_total_and_correct() {
645        let a = cur(10, 0, 0, 5, 1);
646        // Later tx-time dominates everything else.
647        let b = cur(11, 0, 0, 1, 1);
648        // Same tx-time, edge (kind_ord 1) after node (kind_ord 0).
649        let c = cur(10, 0, 1, 1, 1);
650        // Same tx-time + kind, higher id is later.
651        let d = cur(10, 0, 0, 6, 1);
652        // Same tx-time + kind + entity, higher version_id is later (the collision tiebreak).
653        let e = cur(10, 0, 0, 5, 2);
654        assert!(a < b);
655        assert!(a < c);
656        assert!(a < d);
657        assert!(d < c);
658        assert!(a < e);
659        assert!(e < d);
660    }

663    fn decode_rejects_malformed_tokens() {
