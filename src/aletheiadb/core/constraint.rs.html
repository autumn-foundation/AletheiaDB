20/// A declared property type for a schema constraint (Issue #3378).
21///
22/// Maps each declarable type to the concrete [`PropertyValue`] variant it
23/// accepts. Note there is no dedicated temporal `PropertyValue`: AletheiaDB
24/// stores temporal/timestamp property values as microseconds-since-epoch
25/// [`PropertyValue::Int`] (see [`crate::core::temporal`]), so
26/// [`DeclaredType::Temporal`] matches `Int`. This is documented here and in
27/// `docs/guides/schema-constraints.md`.
28#[derive(Debug, Clone, Copy, PartialEq, Eq, Encode, Decode)]
29#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
30pub enum DeclaredType {
31    /// UTF-8 string values ([`PropertyValue::String`]).
32    String,
33    /// 64-bit signed integer values ([`PropertyValue::Int`]).
34    Integer,
35    /// 64-bit floating point values ([`PropertyValue::Float`]).
36    Float,
37    /// Boolean values ([`PropertyValue::Bool`]).
38    Boolean,
39    /// Temporal values, stored as microseconds-since-epoch integers
40    /// ([`PropertyValue::Int`]).
41    Temporal,
42    /// Byte-array values ([`PropertyValue::Bytes`]).
43    Bytes,
44    /// Dense embedding vectors ([`PropertyValue::Vector`]). When `dim` is
45    /// `Some(d)`, the vector's length must equal exactly `d`.
46    Vector {
47        /// Required vector dimension, or `None` to accept any dimension.
48        dim: Option<usize>,
49    },
50}
51
52impl DeclaredType {
53    /// Stable token for this declared type, aligned with
54    /// [`PropertyValue::type_name`] where a 1:1 mapping exists.
55    pub const fn type_name(&self) -> &'static str {
56        match self {
57            DeclaredType::String => "string",
58            DeclaredType::Integer => "int",
59            DeclaredType::Float => "float",
60            DeclaredType::Boolean => "bool",
61            DeclaredType::Temporal => "temporal",
62            DeclaredType::Bytes => "bytes",
63            DeclaredType::Vector { .. } => "vector",
64        }
65    }
66
67    /// Returns `true` if `value` conforms to this declared type.
68    ///
69    /// `Null` never matches any declared type here — null/absence handling is
70    /// done by the required/nullable logic in
71    /// [`ConstraintRegistry::check_entity`], not by the type matcher.
72    pub fn matches(&self, value: &PropertyValue) -> bool {
73        match self {
74            DeclaredType::String => matches!(value, PropertyValue::String(_)),
75            DeclaredType::Integer => matches!(value, PropertyValue::Int(_)),
76            DeclaredType::Float => matches!(value, PropertyValue::Float(_)),
77            DeclaredType::Boolean => matches!(value, PropertyValue::Bool(_)),
78            // Temporal values are stored as micros-since-epoch integers.
79            DeclaredType::Temporal => matches!(value, PropertyValue::Int(_)),
80            DeclaredType::Bytes => matches!(value, PropertyValue::Bytes(_)),
81            DeclaredType::Vector { dim } => match value {
82                PropertyValue::Vector(v) => dim.is_none_or(|d| v.len() == d),
83                _ => false,
84            },
85        }
86    }
87}

184/// Canonical hashable + equatable key derived from a `PropertyValue`.
185///
186/// Only scalar types that can serve as unique keys are supported.
187/// Null, Vector, SparseVector, and Array are rejected.
188#[derive(Debug, Clone, PartialEq, Eq, Hash)]
189pub enum ValueKey {
190    /// Boolean value.
191    Bool(bool),
192    /// Integer value.
193    Int(i64),
194    /// Float value (stored as bits for `Hash`+`Eq`).
195    Float(u64),
196    /// String value.
197    String(Arc<str>),
198    /// Bytes value.
199    Bytes(Arc<[u8]>),
200}
201
202impl ValueKey {
203    /// Try to create a `ValueKey` from a `PropertyValue`.
204    /// Returns `None` for unsupported types (Null, Vector, SparseVector, Array).
205    pub fn from_property_value(v: &PropertyValue) -> Option<Self> {
206        match v {
207            PropertyValue::Bool(b) => Some(ValueKey::Bool(*b)),
208            PropertyValue::Int(i) => Some(ValueKey::Int(*i)),
209            PropertyValue::Float(f) => {
210                // Normalize all NaN representations to one canonical bit pattern so two
211                // NaN floats are treated as the same unique key (instead of different
212                // bit-patterns bypassing the constraint).
213                let bits = if f.is_nan() {
214                    f64::NAN.to_bits()
215                } else {
216                    f.to_bits()
217                };
218                Some(ValueKey::Float(bits))
219            }
220            PropertyValue::String(s) => Some(ValueKey::String(Arc::clone(s))),
221            PropertyValue::Bytes(b) => Some(ValueKey::Bytes(Arc::clone(b))),
222            PropertyValue::Null
223            | PropertyValue::Vector(_)
224            | PropertyValue::SparseVector(_)
225            | PropertyValue::Array(_) => None,
226        }
227    }
228
229    /// Human-readable display for error messages.
230    pub fn display_string(&self) -> String {
231        match self {
232            ValueKey::Bool(b) => b.to_string(),
233            ValueKey::Int(i) => i.to_string(),
234            ValueKey::Float(bits) => f64::from_bits(*bits).to_string(),
235            ValueKey::String(s) => s.as_ref().to_string(),
236            ValueKey::Bytes(b) => format!("<bytes:{}>", b.len()),
237        }
238    }
239}

249    pub value: ValueKey,

252/// RAII guard holding in-flight constraint reservations for one transaction.
253///
254/// - **Rollback path** (`Drop`): releases all added reservations so the index
255///   has no orphan entries.
256/// - **Success path** (`commit()`): keeps added reservations and removes the
257///   old-value entries that were displaced by update/delete operations.
258pub struct ReservationGuard {
259    registry: Arc<ConstraintRegistry>,
260    /// Keys added during this transaction (released on rollback).
261    added: Vec<ReservationKey>,
262    /// Old-value keys to remove after commit (freed on the success path).
263    /// HashSet for O(1) dedup in the removed-entries loop.
264    to_remove: std::collections::HashSet<ReservationKey>,
265    committed: bool,
266}
267
268impl ReservationGuard {
269    fn new(registry: Arc<ConstraintRegistry>) -> Self {
270        Self {
271            registry,
272            added: Vec::new(),
273            to_remove: std::collections::HashSet::new(),
274            committed: false,
275        }
276    }
277
278    /// Call on the success path to lock in added reservations and free old ones.
279    pub fn commit(mut self) {
280        self.committed = true;
281        for key in &self.to_remove {
282            self.registry.reservation_index.remove(key);
283        }
284    }
285}
286
287impl Drop for ReservationGuard {
288    fn drop(&mut self) {
289        if !self.committed {
290            for key in &self.added {
291                self.registry.reservation_index.remove(key);
292            }
293        }
294    }
295}

297/// Outcome of checking one property value against one [`PropertyConstraint`].
298///
299/// Shared by the write-path validator ([`ConstraintRegistry::check_entity`])
300/// and the enable-time scan ([`ConstraintRegistry::evaluate_conformance`]) so
301/// both compute an identical category for every value (Issue #3378 FIX-5).
302enum PropertyCheck {
303    /// The value conforms to this constraint.
304    Ok,
305    /// The value is missing/absent for a required key, or an explicit `Null`
306    /// on a non-nullable key. `reason` is the canonical human-readable string
307    /// used in conformance reports; both sub-cases are the Missing category
308    /// and both surface as `MissingRequiredKey` on the write path.
309    Missing {
310        /// Canonical reason string (`"missing required key"` or
311        /// `"null value not permitted"`).
312        reason: &'static str,
313    },
314    /// A present, non-null value whose type does not match the declared type.
315    TypeMismatch {
316        /// The declared type's token.
317        expected: &'static str,
318        /// The actual value's type token.
319        actual: &'static str,
320    },
321}
322
323/// The single per-constraint predicate shared by both validators (Issue #3378
324/// FIX-5). Classifying every `(constraint, value)` here removes the risk of the
325/// write path and the enable-time scan drifting on null/required/type
326/// semantics.
327fn check_property(c: &PropertyConstraint, value: Option<&PropertyValue>) -> PropertyCheck {
328    let is_null = matches!(value, Some(PropertyValue::Null));
329    let present_nonnull = value.is_some() && !is_null;
330
331    if c.required && !present_nonnull {
332        return PropertyCheck::Missing {
333            reason: "missing required key",
334        };
335    }
336    if is_null {
337        if !c.nullable {
338            return PropertyCheck::Missing {
339                reason: "null value not permitted",
340            };
341        }
342        return PropertyCheck::Ok;
343    }
344    // Present, non-null value: enforce the declared type if any.
345    if let (Some(dt), Some(v)) = (c.declared_type, value)
346        && !dt.matches(v)
347    {
348        return PropertyCheck::TypeMismatch {
349            expected: dt.type_name(),
350            actual: v.type_name(),
351        };
352    }
353    PropertyCheck::Ok
354}

356/// Central registry for uniqueness constraints.
357///
358/// Thread-safe via DashMap; designed for concurrent read/write access.
359///
360/// # Lock Order
361///
362/// The `ConstraintRegistry` is a **leaf** in AletheiaDB's lock hierarchy.
363/// It is acquired (via DashMap shard locks) before `current_timestamp` and
364/// never held across any other synchronization primitive.
365pub struct ConstraintRegistry {
366    /// Active constraint declarations: (label_id, property_id) → present = enabled.
367    declarations: DashMap<(InternedString, InternedString), ()>,
368    /// Currently-valid reservation index: ReservationKey → owning NodeId.
369    pub(crate) reservation_index: DashMap<ReservationKey, NodeId>,
370    /// Property type / required-key schema constraints (Issue #3378), keyed by
371    /// `(entity_kind, label_id)`. Opt-in: a label with no entry is fully
372    /// schemaless. `Arc<Vec<..>>` so the hot write-path check clones a cheap
373    /// pointer rather than the constraint vector.
374    schema_constraints: DashMap<(EntityKind, InternedString), Arc<Vec<PropertyConstraint>>>,
375    /// Serializes schema-constraint DDL (Issue #3378): the whole
376    /// export→persist-sidecar→declare/drop read-modify-write in
377    /// `enable_schema_constraints` / `drop_schema_constraint` runs under this
378    /// lock so two concurrent DDL calls on different labels cannot race the
379    /// sidecar file (last-write-wins silently dropping the other label from
380    /// disk). A **leaf** lock: acquired only around that critical section and
381    /// never held across any other AletheiaDB synchronization primitive, so it
382    /// introduces no lock-order risk. `()` payload — it guards the external
383    /// side effect (the sidecar file + the in-memory map staying consistent),
384    /// not shared data of its own.
385    schema_ddl_lock: parking_lot::Mutex<()>,
386}
387
388impl ConstraintRegistry {
389    /// Create an empty registry.
390    pub fn new() -> Self {
391        Self {
392            declarations: DashMap::new(),
393            reservation_index: DashMap::new(),
394            schema_constraints: DashMap::new(),
395            schema_ddl_lock: parking_lot::Mutex::new(()),
396        }
397    }
398
399    /// Acquire the schema-constraint DDL lock (Issue #3378). Callers hold the
400    /// returned guard across the entire export→persist→declare/drop critical
401    /// section in `enable`/`drop` so the on-disk sidecar and the in-memory
402    /// registry stay consistent under concurrent DDL. Leaf lock — see the
403    /// field docs.
404    #[must_use = "hold the guard across the whole DDL critical section"]
405    pub fn lock_schema_ddl(&self) -> parking_lot::MutexGuard<'_, ()> {
406        self.schema_ddl_lock.lock()
407    }

409    /// Returns `true` if there is an active constraint for `(label, property)`.
410    #[inline]
411    pub fn is_constrained(&self, label: InternedString, property: InternedString) -> bool {
412        self.declarations.contains_key(&(label, property))
413    }
414
415    /// Record a constraint declaration (called by `enable()` and WAL replay).
416    pub fn declare(&self, label: InternedString, property: InternedString) {
417        self.declarations.insert((label, property), ());
418    }
419
420    /// Remove a constraint declaration (called by `disable()` and WAL replay).
421    pub fn undeclare(&self, label: InternedString, property: InternedString) {
422        self.declarations.remove(&(label, property));
423        self.reservation_index
424            .retain(|k, _| !(k.label == label && k.property == property));
425    }

427    /// Export all declared uniqueness constraints as serializable, string-keyed
428    /// descriptors (for the `.albk` backup payload, Issue #3218). Sorted
429    /// deterministically by `(label, property)` so backups are reproducible.
430    pub fn export_unique_constraints(&self) -> Vec<UniqueConstraintDescriptor> {
431        let mut out: Vec<UniqueConstraintDescriptor> = self
432            .list()
433            .into_iter()
434            .map(|(label, property)| UniqueConstraintDescriptor { label, property })
435            .collect();
436        out.sort_by(|a, b| {
437            a.label
438                .cmp(&b.label)
439                .then_with(|| a.property.cmp(&b.property))
440        });
441        out
442    }
443
444    /// List all declared constraints as `(label_string, property_string)` pairs.
445    pub fn list(&self) -> Vec<(String, String)> {
446        self.declarations
447            .iter()
448            .filter_map(|entry| {
449                let (label_id, prop_id) = entry.key();
450                let label = GLOBAL_INTERNER.resolve_with(*label_id, |s| s.to_string())?;
451                let property = GLOBAL_INTERNER.resolve_with(*prop_id, |s| s.to_string())?;
452                Some((label, property))
453            })
454            .collect()
455    }

457    /// Atomically attempt to reserve `(label, property, value) → node_id`.
458    ///
459    /// - If the slot is vacant, inserts and returns `Ok(())`.
460    /// - If the slot is already owned by `node_id`, returns `Ok(())` (idempotent).
461    /// - If the slot is owned by a different node, returns `Err(UniqueViolation)`.
462    pub fn try_reserve(
463        &self,
464        key: ReservationKey,
465        node_id: NodeId,
466        label_str: &str,
467        property_str: &str,
468    ) -> std::result::Result<bool, ConstraintError> {
469        use dashmap::mapref::entry::Entry;
470        match self.reservation_index.entry(key.clone()) {
471            Entry::Vacant(e) => {
472                e.insert(node_id);
473                Ok(true) // newly inserted
474            }
475            Entry::Occupied(e) => {
476                let existing = *e.get();
477                if existing == node_id {
478                    Ok(false) // same-node idempotent, not a new insertion
479                } else {
480                    Err(ConstraintError::UniqueViolation {
481                        label: label_str.to_string(),
482                        property: property_str.to_string(),
483                        value: key.value.display_string(),
484                        existing_node_id: existing,
485                    })
486                }
487            }
488        }
489    }

491    /// Pre-flight scan: check that enabling `(label, property)` on `nodes` would
492    /// not reveal existing duplicates. Fails with `DuplicateOnEnable` if any exist.
493    pub fn check_no_duplicates(
494        nodes: &[crate::core::graph::Node],
495        label: InternedString,
496        property: InternedString,
497        label_str: &str,
498        property_str: &str,
499    ) -> Result<()> {
500        let mut seen: HashMap<ValueKey, Vec<NodeId>> = HashMap::new();
501
502        for node in nodes {
503            if node.label != label {
504                continue;
505            }
506            if let Some(pv) = node.properties.get_by_interned_key(&property) {
507                // Null means "property not set" — treat it as absent, not as a
508                // constraint violation or an unsupported type.
509                if matches!(pv, PropertyValue::Null) {
510                    continue;
511                }
512                match ValueKey::from_property_value(pv) {
513                    None => {
514                        return Err(ConstraintError::UnsupportedKeyType {
515                            label: label_str.to_string(),
516                            property: property_str.to_string(),
517                            type_name: pv.type_name().to_string(),
518                        }
519                        .into());
520                    }
521                    Some(vk) => {
522                        seen.entry(vk).or_default().push(node.id);
523                    }
524                }
525            }
526        }
527
528        for (vk, ids) in &seen {
529            if ids.len() >= 2 {
530                return Err(ConstraintError::DuplicateOnEnable {
531                    label: label_str.to_string(),
532                    property: property_str.to_string(),
533                    value: vk.display_string(),
534                    node_ids: ids.clone(),
535                }
536                .into());
537            }
538        }
539
540        Ok(())
541    }

543    /// Rebuild the reservation index from a slice of currently-valid nodes for one
544    /// `(label, property)` constraint pair. Called after WAL replay.
545    pub fn rebuild_from_nodes(
546        &self,
547        nodes: &[crate::core::graph::Node],
548        label: InternedString,
549        property: InternedString,
550    ) {
551        for node in nodes {
552            if node.label != label {
553                continue;
554            }
555            if let Some(value) = node.properties.get_by_interned_key(&property)
556                && let Some(vk) = ValueKey::from_property_value(value)
557            {
558                let key = ReservationKey {
559                    label,
560                    property,
561                    value: vk,
562                };
563                self.reservation_index.insert(key, node.id);
564            }
565        }
566    }

568    /// Build a `ReservationGuard` by atomically reserving all constraint keys
569    /// that the transaction's operations would create/update/delete.
570    ///
571    /// `added_entries`: post-tx (label, properties, node_id) for created/updated nodes.
572    /// `removed_entries`: pre-tx (label, properties, node_id) for updated/deleted nodes
573    ///   (these keys are freed on commit).
574    pub fn reserve_for_transaction(
575        self: &Arc<Self>,
576        added_entries: &[(InternedString, &PropertyMap, NodeId)],
577        removed_entries: &[(InternedString, &PropertyMap, NodeId)],
578    ) -> std::result::Result<ReservationGuard, ConstraintError> {
579        let mut guard = ReservationGuard::new(Arc::clone(self));
580
581        // Track all keys "claimed" by this tx post-commit (newly reserved OR
582        // idempotently re-confirmed on same node).  Keys in `claimed` must NOT
583        // be freed on commit even if they also appear in the pre-tx state.
584        let mut claimed: std::collections::HashSet<ReservationKey> =
585            std::collections::HashSet::new();
586
587        for (label_id, props, node_id) in added_entries {
588            for (prop_id, value) in props.iter() {
589                let prop_id = *prop_id;
590                if !self.is_constrained(*label_id, prop_id) {
591                    continue;
592                }
593                let vk = match ValueKey::from_property_value(value) {
594                    Some(k) => k,
595                    None => continue,
596                };
597                let key = ReservationKey {
598                    label: *label_id,
599                    property: prop_id,
600                    value: vk,
601                };
602                let label_str = GLOBAL_INTERNER
603                    .resolve_with(*label_id, |s| s.to_string())
604                    .unwrap_or_default();
605                let prop_str = GLOBAL_INTERNER
606                    .resolve_with(prop_id, |s| s.to_string())
607                    .unwrap_or_default();
608
609                match self.try_reserve(key.clone(), *node_id, &label_str, &prop_str) {
610                    Ok(true) => {
611                        // Newly reserved — must roll back on tx failure.
612                        guard.added.push(key.clone());
613                        claimed.insert(key);
614                    }
615                    Ok(false) => {
616                        // Same node idempotent — already in the index; not a new
617                        // insertion, but still "claimed" so we don't free it.
618                        claimed.insert(key);
619                    }
620                    Err(e) => {
621                        return Err(e);
622                    }
623                }
624            }
625        }
626
627        for (label_id, props, _node_id) in removed_entries {
628            for (prop_id, value) in props.iter() {
629                let prop_id = *prop_id;
630                if !self.is_constrained(*label_id, prop_id) {
631                    continue;
632                }
633                if let Some(vk) = ValueKey::from_property_value(value) {
634                    let key = ReservationKey {
635                        label: *label_id,
636                        property: prop_id,
637                        value: vk,
638                    };
639                    // Only free the old key if it is not still claimed by this tx.
640                    // (Handles the case where a node updates a property to its own
641                    // current value — the slot must not be freed.)
642                    // HashSet insert is idempotent, so no explicit contains check needed.
643                    if !claimed.contains(&key) {
644                        guard.to_remove.insert(key);
645                    }
646                }
647            }
648        }
649
650        Ok(guard)
651    }

692    /// Validate one entity's effective property map against the declared
693    /// schema constraints for `(kind, label)`. Returns `Ok(())` when no
694    /// constraints are declared (schemaless) or the map conforms.
695    ///
696    /// Semantics (see `docs/guides/schema-constraints.md`):
697    /// - `required`: the key must be present with a non-null value; a missing
698    ///   or `Null` value yields `MissingRequiredKey`.
699    /// - `nullable == false`: an explicitly-`Null` value yields
700    ///   `MissingRequiredKey` (null is not permitted).
701    /// - `declared_type`: a present, non-null value whose type does not match
702    ///   yields `TypeViolation`. `Null`/absent values skip the type check.
703    ///
704    /// The write path ([`Self::check_entity`]) and the enable-time scan
705    /// ([`Self::evaluate_conformance`]) share one per-constraint predicate
706    /// ([`check_property`]) so a value's category (missing / null-not-permitted
707    /// / type-mismatch) is guaranteed identical in both paths (Issue #3378
708    /// FIX-5) — no risk of the two validators drifting.
709    pub fn check_entity(
710        &self,
711        kind: EntityKind,
712        label: InternedString,
713        properties: &PropertyMap,
714    ) -> std::result::Result<(), ConstraintError> {
715        let Some(constraints) = self.schema_constraints_for(kind, label) else {
716            return Ok(());
717        };
718        let label_str = || GLOBAL_INTERNER.resolve_or_else(label, String::new);
719        let prop_str = |p: InternedString| GLOBAL_INTERNER.resolve_or_else(p, String::new);
720
721        let mut missing_keys: Vec<String> = Vec::new();
722        for c in constraints.iter() {
723            let value = properties.get_by_interned_key(&c.property);
724            match check_property(c, value) {
725                // Both "missing required key" and "null value not permitted"
726                // are the Missing category and surface as MissingRequiredKey on
727                // the write path (unchanged behavior); the shared predicate
728                // guarantees the category matches the enable-time report.
729                PropertyCheck::Missing { .. } => missing_keys.push(prop_str(c.property)),
730                PropertyCheck::TypeMismatch { expected, actual } => {
731                    return Err(ConstraintError::TypeViolation {
732                        entity_kind: kind.as_str().to_string(),
733                        label: label_str(),
734                        property: prop_str(c.property),
735                        expected_type: expected,
736                        actual_type: actual,
737                    });
738                }
739                PropertyCheck::Ok => {}
740            }
741        }
742
743        if !missing_keys.is_empty() {
744            return Err(ConstraintError::MissingRequiredKey {
745                entity_kind: kind.as_str().to_string(),
746                label: label_str(),
747                missing_keys,
748            });
749        }
750        Ok(())
751    }
752
753    /// Evaluate one entity's property map against `constraints`, returning a
754    /// list of `(property, reason)` violations (used to build a
755    /// [`ConformanceReport`] on enable). Unlike [`Self::check_entity`], this
756    /// collects *all* violations rather than short-circuiting. Shares the
757    /// per-constraint predicate [`check_property`] with the write path so the
758    /// two never disagree on a value's category (Issue #3378 FIX-5).
759    pub fn evaluate_conformance(
760        constraints: &[PropertyConstraint],
761        properties: &PropertyMap,
762    ) -> Vec<(Option<String>, String)> {
763        let mut out = Vec::new();
764        for c in constraints {
765            let value = properties.get_by_interned_key(&c.property);
766            let reason = match check_property(c, value) {
767                PropertyCheck::Ok => continue,
768                PropertyCheck::Missing { reason } => reason.to_string(),
769                PropertyCheck::TypeMismatch { expected, actual } => {
770                    format!("expected type {expected}, got {actual}")
771                }
772            };
773            let prop = GLOBAL_INTERNER.resolve_or_else(c.property, String::new);
774            out.push((Some(prop), reason));
775        }
776        out
777    }

779    /// Export all declared schema constraints as serializable, string-keyed
780    /// descriptors (for the sidecar, the `.albk` backup, and the public list
781    /// API). Deterministically sorted by (entity_kind, label).
782    pub fn export_schema_constraints(&self) -> Vec<SchemaConstraintDescriptor> {
783        let mut out: Vec<SchemaConstraintDescriptor> = self
784            .schema_constraints
785            .iter()
786            .filter_map(|entry| {
787                let (kind, label_id) = *entry.key();
788                let label = GLOBAL_INTERNER.resolve_with(label_id, |s| s.to_string())?;
789                let properties = entry
790                    .value()
791                    .iter()
792                    .filter_map(|c| {
793                        let property =
794                            GLOBAL_INTERNER.resolve_with(c.property, |s| s.to_string())?;
795                        Some(PropertyConstraintDescriptor {
796                            property,
797                            declared_type: c.declared_type,
798                            required: c.required,
799                            nullable: c.nullable,
800                        })
801                    })
802                    .collect();
803                Some(SchemaConstraintDescriptor {
804                    entity_kind: kind.as_str().to_string(),
805                    label,
806                    properties,
807                })
808            })
809            .collect();
810        out.sort_by(|a, b| {
811            a.entity_kind
812                .cmp(&b.entity_kind)
813                .then_with(|| a.label.cmp(&b.label))
814        });
815        out
816    }

818    /// Import schema constraints from serializable descriptors (re-interning
819    /// labels/properties), replacing any existing schema-constraint state.
820    /// Used by the sidecar / backup load path.
821    pub fn import_schema_constraints(&self, descriptors: &[SchemaConstraintDescriptor]) {
822        self.schema_constraints.clear();
823        for d in descriptors {
824            let kind = if d.entity_kind == "edge" {
825                EntityKind::Edge
826            } else {
827                EntityKind::Node
828            };
829            let Ok(label_id) = GLOBAL_INTERNER.intern(&d.label) else {
830                continue;
831            };
832            let mut constraints = Vec::with_capacity(d.properties.len());
833            for p in &d.properties {
834                if let Ok(prop_id) = GLOBAL_INTERNER.intern(&p.property) {
835                    constraints.push(PropertyConstraint {
836                        property: prop_id,
837                        declared_type: p.declared_type,
838                        required: p.required,
839                        nullable: p.nullable,
840                    });
841                }
842            }
843            self.schema_constraints
844                .insert((kind, label_id), Arc::new(constraints));
845        }
846    }
