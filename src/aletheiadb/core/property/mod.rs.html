1//! Property system with Arc-based deduplication.
2//!
3//! This module provides a copy-on-write property system where properties are
4//! stored in immutable, reference-counted containers. This enables:
5//! - Cheap cloning of property maps (just increment reference count)
6//! - Deduplication of unchanged properties across versions
7//! - Zero-copy sharing of immutable data
8
9use crate::core::interning::{GLOBAL_INTERNER, InternedString};
10
11// ============================================================================
12// Serialization Type Tags
13// ============================================================================
14// Binary format uses little-endian byte order (consistent with WAL format).
15// Each PropertyValue variant has a unique 1-byte type tag.
16
17/// Type tag for Null value.
18pub const TAG_NULL: u8 = 0;
19/// Type tag for Bool value.
20pub const TAG_BOOL: u8 = 1;
21/// Type tag for Int (i64) value.
22pub const TAG_INT: u8 = 2;
23/// Type tag for Float (f64) value.
24pub const TAG_FLOAT: u8 = 3;
25/// Type tag for String value.
26pub const TAG_STRING: u8 = 4;
27/// Type tag for Bytes value.
28pub const TAG_BYTES: u8 = 5;
29/// Type tag for Array value.
30pub const TAG_ARRAY: u8 = 6;
31// Re-export vector constants from core::vector

33pub use crate::core::vector::serialization::{

39// ============================================================================
40// Serialization Limits
41// ============================================================================
42// These limits prevent DoS attacks via memory exhaustion from malicious input.
43
44/// Maximum number of elements allowed in a deserialized array.
45/// Increased from 1M to 10M to support business scenarios:
46/// - Time series data: 115 days at 1kHz, multiple years at hourly resolution
47/// - IoT telemetry: High-frequency sensor data
48/// - Batch processing: Large bulk imports
49///
50///   Still provides DoS protection (max 40MB for f32 array).
51pub const MAX_ARRAY_ELEMENTS: usize = 10_000_000;
52
53/// Maximum capacity allowed for a deserialized property map.
54/// Increased from 10K to 100K to support business scenarios:
55/// - E-commerce: Products with extensive attributes and variations
56/// - Scientific data: Rich metadata and measurements
57/// - Dynamic schemas: User profiles with custom fields
58///
59///   Still provides DoS protection (~1MB per node maximum).
60pub const MAX_PROPERTY_MAP_CAPACITY: usize = 100_000;
61
62/// Maximum recursion depth for nested properties (e.g., arrays of arrays).
63/// Set to 100 to prevent stack overflow from malicious input.
64pub const MAX_RECURSION_DEPTH: usize = 100;
65
66/// Property key type.
67///
68/// Uses interned strings for memory efficiency and O(1) equality comparisons.
69/// Common keys like "name", "age", and "id" are deduplicated in memory.
70pub type PropertyKey = InternedString;
