1//! Hybrid Logical Clock (HLC) implementation for distributed temporal consistency.
2//!
3//! HLCs combine physical wallclock time with logical counters to provide:
4//! - Monotonic ordering despite clock skew
5//! - Causality preservation across distributed nodes
6//! - Human-readable wallclock semantics for temporal queries

14/// Hybrid Logical Clock timestamp combining wallclock and logical components.
15///
16/// # Structure
17/// - `wallclock`: Physical time in microseconds since Unix epoch
18/// - `logical`: Counter incremented when wallclock doesn't advance
19///
20/// # Ordering
21/// HLCs are ordered lexicographically: first by wallclock, then by logical counter.
22#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
23pub struct HybridTimestamp {
24    /// Physical wallclock time in microseconds since Unix epoch.
25    wallclock: i64,
26    /// Logical counter for ordering events with identical wallclock.
27    logical: u32,
28}
29
30/// Maximum allowed backward clock drift in microseconds (5 minutes).
31///
32/// Backward drift indicates local wallclock moved behind the last committed HLC wallclock.
33/// Beyond this threshold we either reject the write or self-heal by clamping to the frontier.
34pub const MAX_BACKWARD_DRIFT_US: i64 = 5 * 60 * 1_000_000;
35
36/// Maximum allowed forward clock jump in microseconds (1 hour).
37///
38/// Callers may choose to disable this check (for example when idle time is expected and
39/// forward drift is measured against the last commit timestamp).
40pub const MAX_FORWARD_JUMP_US: i64 = 60 * 60 * 1_000_000;

75/// Whether clock-skew self-healing is enabled via environment variable.
76///
77/// Enabled values: `1`, `true`, `on`, `yes`, `enabled` (case-insensitive).
78pub fn is_clock_skew_self_heal_enabled() -> bool {
79    #[cfg(test)]
80    if let Some(forced) = CLOCK_SKEW_AUTO_HEAL_TEST_OVERRIDE.with(|cell| cell.get()) {
81        return forced;
82    }
83
84    static CLOCK_SKEW_AUTO_HEAL: OnceLock<Option<bool>> = OnceLock::new();
85    CLOCK_SKEW_AUTO_HEAL
86        .get_or_init(|| {
87            std::env::var("ALETHEIADB_AUTO_HEAL_CLOCK_SKEW")
88                .ok()
89                .map(|value| {
90                    matches!(
91                        value.to_ascii_lowercase().as_str(),
92                        "1" | "true" | "on" | "yes" | "enabled"
93                    )
94                })
95        })
96        .unwrap_or(false)

99/// Direction of wallclock drift relative to the HLC frontier.
100#[derive(Debug, Clone, Copy, PartialEq, Eq)]
101pub enum ClockSkewDirection {
102    /// Current wallclock is behind frontier by more than allowed.
103    Backward,
104    /// Current wallclock is ahead of frontier by more than allowed.
105    Forward,
106}

108impl ClockSkewDirection {

110    pub const fn as_str(self) -> &'static str {
111        match self {
112            Self::Backward => "backward",
113            Self::Forward => "forward",
114        }
115    }

118/// Result of evaluating wallclock drift against configured skew policy.
119#[derive(Debug, Clone, Copy, PartialEq, Eq)]
120pub struct ClockSkewDecision {
121    /// Signed drift in microseconds (`current_wallclock - frontier_wallclock`).
122    pub drift_us: i64,
123    /// Wallclock to pass into `HybridTimestamp::send`.
124    pub effective_wallclock: i64,
125    /// Drift direction that was self-healed (if any).
126    pub healed_direction: Option<ClockSkewDirection>,
127}
128
129/// Drift exceeded configured threshold and self-healing was disabled.
130#[derive(Debug, Clone, Copy, PartialEq, Eq)]
131pub struct ClockSkewViolation {
132    /// Signed drift in microseconds (`current_wallclock - frontier_wallclock`).
133    pub drift_us: i64,
134    /// Threshold used for this direction (signed to preserve caller semantics).
135    pub max_allowed: i64,
136    /// Direction of the violation.
137    pub direction: ClockSkewDirection,
138}
139
140/// Evaluate clock skew and derive the effective wallclock for `send()`.
141///
142/// When self-healing is enabled and drift exceeds threshold, this clamps to the frontier
143/// wallclock so HLC remains monotonic without aborting.
144pub fn evaluate_clock_skew(
145    current_wallclock: i64,
146    frontier_wallclock: i64,
147    max_forward_jump_us: Option<i64>,
148    self_heal_clock_skew: bool,
149) -> Result<ClockSkewDecision, ClockSkewViolation> {
150    // Use saturating subtraction to prevent overflow/wrapping on extreme jumps.
151    // e.g. i64::MAX - i64::MIN would wrap to -1 without this, bypassing checks.
152    let drift = current_wallclock.saturating_sub(frontier_wallclock);
153    let mut effective_wallclock = current_wallclock;
154    let mut healed_direction = None;
155
156    if drift < -MAX_BACKWARD_DRIFT_US {
157        if !self_heal_clock_skew {
158            return Err(ClockSkewViolation {
159                drift_us: drift,
160                max_allowed: -MAX_BACKWARD_DRIFT_US,
161                direction: ClockSkewDirection::Backward,
162            });
163        }
164
165        effective_wallclock = frontier_wallclock;
166        healed_direction = Some(ClockSkewDirection::Backward);
167    }
168
169    if let Some(max_forward_jump_us) = max_forward_jump_us
170        && drift > max_forward_jump_us
171    {
172        if !self_heal_clock_skew {
173            return Err(ClockSkewViolation {
174                drift_us: drift,
175                max_allowed: max_forward_jump_us,
176                direction: ClockSkewDirection::Forward,
177            });
178        }
179
180        effective_wallclock = frontier_wallclock;
181        healed_direction = Some(ClockSkewDirection::Forward);
182    }
183
184    Ok(ClockSkewDecision {
185        drift_us: drift,
186        effective_wallclock,
187        healed_direction,
188    })
189}

191/// Error stage from `send_with_overflow_self_heal`.
192#[derive(Debug, Clone, PartialEq, Eq)]
193pub enum SendWithSelfHealError {
194    /// Initial call to `send()` failed.
195    InitialSend(TemporalError),
196    /// Logical overflow was detected but `wallclock + 1` overflowed i64.
197    FallbackWallclockOverflow {
198        /// Wallclock from the original overflow error.
199        wallclock: i64,
200        /// Logical counter from the original overflow error.
201        current_logical: u32,
202    },
203    /// Fallback call to `send(wallclock + 1)` failed.
204    FallbackSend(TemporalError),
205}
206
207/// Call `send()` with optional logical-overflow self-healing.
208///
209/// If self-heal is enabled and logical overflow occurs, this retries at `wallclock + 1`
210/// to preserve monotonicity under repeated clamping.
211pub fn send_with_overflow_self_heal<E>(
212    frontier: &HybridTimestamp,
213    effective_wallclock: i64,
214    self_heal_clock_skew: bool,
215    map_error: impl Fn(SendWithSelfHealError) -> E,
216) -> Result<HybridTimestamp, E> {
217    match frontier.send(effective_wallclock) {
218        Ok(next) => Ok(next),
219        Err(TemporalError::LogicalCounterOverflow {
220            wallclock,
221            current_logical,
222        }) if self_heal_clock_skew => {
223            let fallback_wallclock = wallclock.checked_add(1).ok_or_else(|| {
224                map_error(SendWithSelfHealError::FallbackWallclockOverflow {
225                    wallclock,
226                    current_logical,
227                })
228            })?;
229            frontier
230                .send(fallback_wallclock)
231                .map_err(|error| map_error(SendWithSelfHealError::FallbackSend(error)))
232        }
233        Err(error) => Err(map_error(SendWithSelfHealError::InitialSend(error))),
234    }
235}

238    /// Create a new HybridTimestamp with validation.
239    ///
240    /// # Errors
241    /// Returns `TemporalError::InvalidTimestamp` if wallclock exceeds `MAX_VALID_TIMESTAMP`.
242    /// This prevents DoS attacks from extreme timestamp values.
243    #[inline]
244    pub fn new(wallclock: i64, logical: u32) -> Result<Self, TemporalError> {
245        if wallclock > MAX_VALID_TIMESTAMP {
246            // Construct invalid timestamp for error reporting
247            let invalid_ts = HybridTimestamp { wallclock, logical };
248            return Err(TemporalError::InvalidTimestamp {
249                timestamp: invalid_ts,
250                reason: format!(
251                    "Wallclock {} exceeds MAX_VALID_TIMESTAMP ({})",
252                    wallclock, MAX_VALID_TIMESTAMP
253                ),
254            });
255        }
256        Ok(HybridTimestamp { wallclock, logical })
257    }

296    /// Helper: Increment logical counter with overflow check.
297    ///
298    /// This helper reduces duplication in `send()` and `receive()` methods.
299    #[inline]
300    fn increment_logical(logical: u32, wallclock: i64) -> Result<u32, TemporalError> {
301        logical
302            .checked_add(1)
303            .ok_or(TemporalError::LogicalCounterOverflow {
304                wallclock,
305                current_logical: logical,
306            })
307    }

309    /// Generate a new timestamp for a send event.
310    ///
311    /// # HLC Algorithm
312    /// - If `new_wallclock` > `self.wallclock`: Use new wallclock, reset logical to 0
313    /// - Otherwise: Keep max(wallclock, new_wallclock), increment logical
314    ///
315    /// This ensures monotonicity while preserving wallclock semantics.
316    ///
317    /// # Errors
318    /// - Returns `TemporalError::InvalidTimestamp` if `new_wallclock` exceeds `MAX_VALID_TIMESTAMP`.
319    /// - Returns `TemporalError::LogicalCounterOverflow` if the logical counter would exceed u32::MAX.
320    ///   This theoretically requires 4+ billion events at the same microsecond, indicating severe
321    ///   clock drift or pathological workload.
322    ///
323    /// # Examples
324    /// ```rust
325    /// # use aletheiadb::core::hlc::HybridTimestamp;
326    /// let hlc = HybridTimestamp::new(1000, 0).unwrap();
327    ///
328    /// // Wallclock advances: logical counter resets
329    /// let next = hlc.send(1001).unwrap();
330    /// assert_eq!(next.wallclock(), 1001);
331    /// assert_eq!(next.logical(), 0);
332    ///
333    /// // Wallclock lags: logical counter increments
334    /// let stalled = next.send(1000).unwrap();
335    /// assert_eq!(stalled.wallclock(), 1001);
336    /// assert_eq!(stalled.logical(), 1);
337    /// ```
338    #[inline]
339    pub fn send(&self, new_wallclock: i64) -> Result<Self, TemporalError> {
340        // Validate new_wallclock to prevent invalid timestamps
341        if new_wallclock > MAX_VALID_TIMESTAMP {
342            return Err(TemporalError::InvalidTimestamp {
343                timestamp: Self {
344                    wallclock: new_wallclock,
345                    logical: 0,
346                },
347                reason: format!(
348                    "Send wallclock {} exceeds MAX_VALID_TIMESTAMP ({})",
349                    new_wallclock, MAX_VALID_TIMESTAMP
350                ),
351            });
352        }
353
354        if new_wallclock > self.wallclock {
355            // Wallclock advanced - reset logical counter
356            Ok(HybridTimestamp {
357                wallclock: new_wallclock,
358                logical: 0,
359            })
360        } else {
361            // Wallclock didn't advance - increment logical counter
362            let logical = Self::increment_logical(self.logical, self.wallclock)?;
363            Ok(HybridTimestamp {
364                wallclock: self.wallclock,
365                logical,
366            })
367        }
368    }

370    /// Generate a new timestamp for a receive event (distributed message reception).
371    ///
372    /// # HLC Algorithm for Message Reception
373    /// When receiving a message with timestamp `msg` from a remote node:
374    /// 1. `new_wallclock = max(self.wallclock, msg.wallclock, physical_wallclock)`
375    /// 2. If wallclock advances beyond both: reset logical to 0
376    /// 3. If wallclock matches both: logical = max(self.logical, msg.logical) + 1
377    /// 4. If wallclock matches only one: increment that timestamp's logical counter
378    ///
379    /// This preserves causality: if message A → message B, then timestamp(A) < timestamp(B).
380    ///
381    /// # Arguments
382    /// - `msg`: The timestamp from the received message
383    /// - `physical_wallclock`: Current physical clock reading
384    ///
385    /// # Errors
386    /// - Returns `TemporalError::InvalidTimestamp` if the resulting wallclock exceeds `MAX_VALID_TIMESTAMP`.
387    /// - Returns `TemporalError::LogicalCounterOverflow` if the logical counter would exceed u32::MAX.
388    ///
389    /// # Examples
390    ///
391    /// ```
392    /// use aletheiadb::core::hlc::HybridTimestamp;
393    ///
394    /// // Receiving a message from a remote node
395    /// let local_time = HybridTimestamp::new(1000, 5).unwrap();
396    /// let message_time = HybridTimestamp::new(2000, 10).unwrap();
397    /// let physical_clock = 1500;
398    ///
399    /// let updated = local_time.receive(message_time, physical_clock).unwrap();
400    /// assert!(updated > local_time);
401    /// assert!(updated > message_time);
402    /// ```
403    ///
404    /// Receiving multiple messages maintains causality:
405    ///
406    /// ```
407    /// use aletheiadb::core::hlc::HybridTimestamp;
408    ///
409    /// let mut local = HybridTimestamp::new(1000, 0).unwrap();
410    /// let msg1 = HybridTimestamp::new(1100, 0).unwrap();
411    /// let msg2 = HybridTimestamp::new(1050, 0).unwrap();
412    ///
413    /// // Receive msg1, update local time
414    /// local = local.receive(msg1, 1000).unwrap();
415    /// assert!(local > msg1); // Causality: local happened-after msg1
416    ///
417    /// // Receive msg2 (older wallclock but causally after local)
418    /// local = local.receive(msg2, 1000).unwrap();
419    /// assert!(local > msg2); // Causality preserved despite clock skew
420    /// ```
421    #[inline]
422    pub fn receive(
423        &self,
424        msg: HybridTimestamp,
425        physical_wallclock: i64,
426    ) -> Result<Self, TemporalError> {
427        // Compute maximum wallclock across all three values
428        let new_wallclock = self.wallclock.max(msg.wallclock).max(physical_wallclock);
429
430        // Validate resulting wallclock
431        if new_wallclock > MAX_VALID_TIMESTAMP {
432            return Err(TemporalError::InvalidTimestamp {
433                timestamp: Self {
434                    wallclock: new_wallclock,
435                    logical: 0,
436                },
437                reason: format!(
438                    "Receive wallclock {} exceeds MAX_VALID_TIMESTAMP ({})",
439                    new_wallclock, MAX_VALID_TIMESTAMP
440                ),
441            });
442        }
443
444        // Determine logical counter based on which wallclock(s) were chosen
445        let logical = if new_wallclock > self.wallclock && new_wallclock > msg.wallclock {
446            // Physical clock advanced beyond both - reset to 0
447            0
448        } else if new_wallclock == self.wallclock && new_wallclock == msg.wallclock {
449            // Both local and message have same wallclock - use max logical + 1
450            Self::increment_logical(self.logical.max(msg.logical), new_wallclock)?
451        } else if new_wallclock == self.wallclock {
452            // Local wallclock was chosen - increment local logical
453            Self::increment_logical(self.logical, new_wallclock)?
454        } else {
455            // Message wallclock was chosen - increment message logical
456            Self::increment_logical(msg.logical, new_wallclock)?
457        };
458
459        Ok(HybridTimestamp {
460            wallclock: new_wallclock,
461            logical,
462        })
463    }

465    /// Serialize this HybridTimestamp to bytes.
466    ///
467    /// # Binary Format
468    /// ```text
469    /// [wallclock:8][logical:4]
470    /// ```
471    /// Total: 12 bytes, little-endian
472    ///
473    /// # Performance Note
474    /// This allocates a new `Vec<u8>`. For better performance when serializing
475    /// multiple timestamps, consider using `serialize_into()` with a reused buffer.
476    pub fn serialize(&self) -> Vec<u8> {
477        let mut buffer = Vec::with_capacity(12);
478        self.serialize_into(&mut buffer);
479        buffer
480    }
481
482    /// Serialize into an existing buffer.
483    pub fn serialize_into(&self, buffer: &mut Vec<u8>) {
484        buffer.extend_from_slice(&self.wallclock.to_le_bytes());
485        buffer.extend_from_slice(&self.logical.to_le_bytes());
486    }
487
488    /// Deserialize a HybridTimestamp from bytes with validation.
489    ///
490    /// Returns the HybridTimestamp and number of bytes consumed (always 12).
491    ///
492    /// # Errors
493    /// - Returns `StorageError::CorruptedData` if buffer is too short
494    /// - Returns `StorageError::CorruptedData` if wallclock exceeds `MAX_VALID_TIMESTAMP`
495    pub fn deserialize(bytes: &[u8]) -> Result<(Self, usize), StorageError> {
496        if bytes.len() < 12 {
497            return Err(StorageError::CorruptedData(format!(
498                "Buffer too short for HybridTimestamp: {} bytes (need 12)",
499                bytes.len()
500            )));
501        }
502
503        // Use split_at and try_into for cleaner, safer byte array conversion
504        let (wallclock_bytes, rest) = bytes.split_at(std::mem::size_of::<i64>());
505        let wallclock = i64::from_le_bytes(wallclock_bytes.try_into().unwrap());
506
507        let (logical_bytes, _) = rest.split_at(std::mem::size_of::<u32>());
508        let logical = u32::from_le_bytes(logical_bytes.try_into().unwrap());
509
510        // Validate wallclock to prevent corrupted data from injecting invalid timestamps
511        // Allow i64::MAX as a special sentinel value for TIMESTAMP_MAX (represents infinity/"still current")
512        if wallclock > MAX_VALID_TIMESTAMP && wallclock != i64::MAX {
513            return Err(StorageError::CorruptedData(format!(
514                "Deserialized wallclock {} exceeds MAX_VALID_TIMESTAMP ({})",
515                wallclock, MAX_VALID_TIMESTAMP
516            )));
517        }
518
519        // Validate that TIMESTAMP_MAX sentinel (i64::MAX) must have logical counter 0
520        if wallclock == i64::MAX && logical != 0 {
521            return Err(StorageError::CorruptedData(format!(
522                "Deserialized timestamp has wallclock i64::MAX but non-zero logical counter: {}",
523                logical
524            )));
525        }
526
527        Ok((HybridTimestamp { wallclock, logical }, 12))
528    }

538/// Convert an i64 wallclock timestamp to HybridTimestamp with logical counter = 0.
539///
540/// This enables seamless migration from Phase 1 (i64 timestamps) to Phase 2 (HybridTimestamp).
541/// The conversion is primarily used in:
542/// - Test code using integer literals
543/// - Legacy APIs that accept i64 timestamps
544/// - WAL/storage deserialization where logical counter is unknown
545///
546/// # Examples
547/// ```
548/// # use aletheiadb::core::hlc::HybridTimestamp;
549/// let ts: HybridTimestamp = 1000_i64.into();
550/// assert_eq!(ts.wallclock(), 1000);
551/// assert_eq!(ts.logical(), 0);
552/// ```
553impl From<i64> for HybridTimestamp {
554    fn from(wallclock: i64) -> Self {
555        // Use new_unchecked for performance - caller responsible for validation
556        // In test code and deserialization contexts, values are trusted
557        HybridTimestamp::new_unchecked(wallclock, 0)
558    }
559}
