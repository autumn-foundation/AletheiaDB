1//! Observable pattern for storage events.
2//!
3//! This module provides an event-driven architecture for components to react to
4//! storage operations without tight coupling. Components implement the `StorageObserver`
5//! trait to receive notifications about anchors, deletes, checkpoints, and other events.
6//!
7//! # Architecture
8//!
9//! ```text
10//! ┌─────────────────────┐
11//! │ HistoricalStorage   │
12//! │  (Event Publisher)  │
13//! └──────────┬──────────┘
14//!            │ notify()
15//!            │
16//!     ┌──────┴───────┬──────────────┐
17//!     │              │              │
18//! ┌───▼────┐   ┌────▼─────┐   ┌───▼──────┐
19//! │ Vector │   │ Metrics  │   │  WAL     │
20//! │ Index  │   │ System   │   │ Logger   │
21//! └────────┘   └──────────┘   └──────────┘
22//!   (Observer)   (Observer)     (Observer)
23//! ```
24//!
25//! # Example
26//!
27//! ```no_run
28//! use aletheiadb::core::observer::{StorageObserver, StorageEvent};
29//! use aletheiadb::storage::historical::HistoricalStorage;
30//! use std::sync::Arc;
31//!
32//! struct MetricsCollector;
33//!
34//! impl StorageObserver for MetricsCollector {
35//!     fn on_event(&self, event: &StorageEvent) -> aletheiadb::core::error::Result<()> {
36//!         match event {
37//!             StorageEvent::NodeAnchorCreated { version_id, timestamp, .. } => {
38//!                 println!("Anchor created: {} at {}", version_id, timestamp);
39//!             }
40//!             _ => {}
41//!         }
42//!         Ok(())
43//!     }
44//! }
45//!
46//! let mut storage = HistoricalStorage::new();
47//! storage.add_observer(Arc::new(MetricsCollector));
48//! ```
49//!
50//! # Hooks vs Observers: When to Use Each
51//!
52//! AletheiaDB uses **both** pre-anchor hooks and post-commit observers as complementary
53//! patterns for storage event handling. Understanding when to use each is critical for
54//! correct integration.
55//!
56//! ## Post-Commit Observers (This Module)
57//!
58//! **When**: AFTER the storage operation completes
59//! **Purpose**: Notifications for metrics, logging, monitoring, auditing
60//! **Returns**: `Result<()>` - cannot return data to caller
61//! **Characteristics**:
62//! - ✅ Extensible: Multiple observers can subscribe to same events
63//! - ✅ Non-blocking: Observer errors don't fail the storage operation
64//! - ✅ Decoupled: Observers don't affect storage logic
65//! - ❌ Cannot return data: Can't influence what gets stored
66//! - ❌ Post-commit: Data already persisted when observer fires
67//!
68//! **Use cases**:
69//! - Logging anchor creation events
70//! - Collecting metrics (anchor count, delta count, etc.)
71//! - Triggering background jobs (compaction, backup, etc.)
72//! - Notifying monitoring systems
73//! - Future: Triggering alerts, webhooks, etc.
74//!
75//! ## Pre-Anchor Hooks (See `historical::PreAnchorHook`)
76//!
77//! **When**: BEFORE the storage operation (specifically before anchor creation)
78//! **Purpose**: Synchronize related state that must be atomically linked to anchors
79//! **Returns**: `Result<Option<T>>` - can return data (e.g., snapshot IDs)
80//! **Characteristics**:
81//! - ✅ Can return data: Returned values stored with the anchor
82//! - ✅ Strong consistency: No consistency window between hook and storage
83//! - ✅ Graceful degradation: Hook errors don't block anchor creation
84//! - ❌ Single hook per entity type: Not extensible like observers
85//! - ❌ Must be fast: Blocks anchor creation until hook completes
86//!
87//! **Use cases**:
88//! - Creating vector snapshots synchronized with anchors (VS-047)
89//! - Generating snapshot IDs for provenance tracking
90//! - Creating auxiliary indexes that must be 1:1 with anchors
91//! - Future: Checkpoint coordination, distributed transaction coordination
92//!
93//! ## Architecture: Hybrid Pattern (VS-047)
94//!
95//! For temporal vector integration, we use **both** patterns together:
96//!
97//! ```text
98//! ┌─────────────────────────────────────────────────────┐
99//! │        HistoricalStorage.add_node_version()         │
100//! └─────────────────────────────────────────────────────┘
101//!                         │
102//!     ┌───────────────────┴───────────────────┐
103//!     │ Is Anchor?                            │
104//!     └───────────────────┬───────────────────┘
105//!                         │
106//!          ┌──────────────▼───────────────┐
107//!          │ BEFORE Storage                │
108//!          │ 1. Call pre_anchor_hook()     │
109//!          │ 2. Get snapshot_id            │
110//!          │ 3. Set in anchor data         │
111//!          └──────────────┬───────────────┘
112//!                         │
113//!          ┌──────────────▼───────────────┐
114//!          │ Storage                       │
115//!          │ Store anchor WITH snapshot_id │
116//!          └──────────────┬───────────────┘
117//!                         │
118//!          ┌──────────────▼───────────────┐
119//!          │ AFTER Storage                 │
120//!          │ 1. notify_observers()         │
121//!          │ 2. Observers react (metrics)  │
122//!          └───────────────────────────────┘
123//! ```
124//!
125//! ## Decision Tree: Which Pattern to Use?
126//!
127//! ```text
128//! Does your component need to:
129//! ├─ Return data to be stored with the anchor?
130//! │  └─ YES → Use Pre-Anchor Hook
131//! │     Example: Vector snapshot IDs, checkpoint IDs
132//! │
133//! └─ Just be notified after storage completes?
134//!    └─ YES → Use Post-Commit Observer
135//!       Example: Metrics, logging, monitoring, audit trails
136//!
137//! Need both? → Register both (hybrid pattern)
138//!   - Hook for data return (snapshot ID)
139//!   - Observer for notifications (metrics)
140//! ```
141//!
142//! ## Example: Temporal Vector Integration (VS-047)
143//!
144//! ```rust
145//! # use aletheiadb::storage::historical::{HistoricalStorage, PreAnchorHook};
146//! # use aletheiadb::core::observer::{StorageObserver, StorageEvent};
147//! # use aletheiadb::index::vector::temporal::{TemporalVectorIndex, TemporalVectorConfig};
148//! # use aletheiadb::index::vector::{HnswConfig, DistanceMetric};
149//! # use std::sync::Arc;
150//! #
151//! # let hnsw_config = HnswConfig::new(4, DistanceMetric::Cosine);
152//! # let config = TemporalVectorConfig::default_with_hnsw(hnsw_config);
153//!
154//! // Pre-Anchor Hook: Create snapshot BEFORE anchor storage, return ID
155//! let index = Arc::new(TemporalVectorIndex::new(config)?);
156//! let node_hook: PreAnchorHook = Arc::new({
157//!     let index = Arc::clone(&index);
158//!     move |_entity_type, _entity_id, timestamp, _properties| {
159//!         // Returns Option<snapshot_id> to be stored in anchor
160//!         index.create_snapshot_for_anchor(timestamp)
161//!     }
162//! });
163//!
164//! let mut storage = HistoricalStorage::new();
165//! storage.register_pre_node_anchor_hook(node_hook);
166//!
167//! // Post-Commit Observer: Collect metrics AFTER anchor storage
168//! struct VectorMetricsObserver;
169//! impl StorageObserver for VectorMetricsObserver {
170//!     fn on_event(&self, event: &StorageEvent) -> aletheiadb::core::error::Result<()> {
171//!         if let StorageEvent::NodeAnchorCreated { .. } = event {
172//!             // Log metrics, update counters, etc.
173//!         }
174//!         Ok(())
175//!     }
176//! }
177//!
178//! storage.add_observer(Arc::new(VectorMetricsObserver));
179//!
180//! // Result: Anchor stored with snapshot_id + observers notified
181//! # Ok::<(), aletheiadb::core::error::Error>(())
182//! ```
183//!
184//! ## Key Takeaways
185//!
186//! 1. **Observers** = Post-commit notifications (this module)
187//! 2. **Hooks** = Pre-storage data return (see `historical::PreAnchorHook`)
188//! 3. **Both** can coexist for complementary functionality
189//! 4. **Observers** are for extensibility, **hooks** are for consistency
190//! 5. See **ADR-0018** for complete architecture and design rationale

200/// Events emitted by the storage layer.
201///
202/// These events allow components to react to storage operations without tight coupling.
203/// New event types can be added as needed for observability, indexing, or coordination.
204#[derive(Debug, Clone, PartialEq, Eq)]
205pub enum StorageEvent {
206    /// A node anchor (full snapshot) was created.
207    ///
208    /// Anchors are created every N versions (default: 10) and represent complete
209    /// property snapshots. This is the key event for temporal vector index synchronization.
210    NodeAnchorCreated {
211        /// ID of the anchor version
212        version_id: VersionId,
213        /// ID of the node this anchor belongs to
214        node_id: NodeId,
215        /// Transaction time when the anchor was created
216        timestamp: Timestamp,
217    },
218
219    /// An edge anchor (full snapshot) was created.
220    EdgeAnchorCreated {
221        /// ID of the anchor version
222        version_id: VersionId,
223        /// ID of the edge this anchor belongs to
224        edge_id: EdgeId,
225        /// Transaction time when the anchor was created
226        timestamp: Timestamp,
227    },
228
229    /// A node version (anchor or delta) was created.
230    ///
231    /// This is a more general event than AnchorCreated, fired for all version types.
232    /// Useful for metrics, logging, or audit trails.
233    NodeVersionCreated {
234        /// ID of the version that was created
235        version_id: VersionId,
236        /// ID of the node this version belongs to
237        node_id: NodeId,
238        /// Transaction time when the version was created
239        timestamp: Timestamp,
240        /// Whether this version is an anchor (true) or delta (false)
241        is_anchor: bool,
242    },
243
244    /// An edge version (anchor or delta) was created.
245    EdgeVersionCreated {
246        /// ID of the version that was created
247        version_id: VersionId,
248        /// ID of the edge this version belongs to
249        edge_id: EdgeId,
250        /// Transaction time when the version was created
251        timestamp: Timestamp,
252        /// Whether this version is an anchor (true) or delta (false)
253        is_anchor: bool,
254    },
255    // Future events (examples):
256    // NodeDeleted { node_id, timestamp },
257    // EdgeDeleted { edge_id, timestamp },
258    // CheckpointCreated { timestamp },
259    // VersionPruned { version_id, timestamp },
260}
261
262impl StorageEvent {
263    /// Get the timestamp of when this event occurred.
264    pub fn timestamp(&self) -> Timestamp {
265        match self {
266            StorageEvent::NodeAnchorCreated { timestamp, .. }
267            | StorageEvent::EdgeAnchorCreated { timestamp, .. }
268            | StorageEvent::NodeVersionCreated { timestamp, .. }
269            | StorageEvent::EdgeVersionCreated { timestamp, .. } => *timestamp,
270        }
271    }
272
273    /// Returns true if this is an anchor creation event.
274    pub fn is_anchor_event(&self) -> bool {
275        matches!(
276            self,
277            StorageEvent::NodeAnchorCreated { .. } | StorageEvent::EdgeAnchorCreated { .. }
278        )
279    }
280}

282/// Observer trait for receiving storage events.
283///
284/// Components implement this trait to react to storage operations.
285/// The observer pattern enables:
286/// - **Loose coupling**: Observers don't depend on HistoricalStorage implementation
287/// - **Multiple subscribers**: Many observers can react to the same event
288/// - **Extensibility**: New event types don't break existing observers
289///
290/// # Error Handling
291///
292/// Observers should handle errors gracefully. If an observer returns an error:
293/// - The error is logged but doesn't fail the storage operation
294/// - Other observers are still notified
295/// - This ensures storage ACID guarantees are maintained
296///
297/// # Thread Safety
298///
299/// Observers must be `Send + Sync` as they may be called from any thread.
300/// Use appropriate synchronization primitives if maintaining mutable state.
301pub trait StorageObserver: Send + Sync {
302    /// Called when a storage event occurs.
303    ///
304    /// # Arguments
305    /// - `event`: The storage event that occurred
306    ///
307    /// # Returns
308    /// - `Ok(())`: Event processed successfully
309    /// - `Err(...)`: Processing failed (logged, doesn't block storage operation)
310    ///
311    /// # Example
312    /// ```no_run
313    /// # use aletheiadb::core::observer::{StorageObserver, StorageEvent};
314    /// struct VectorIndexObserver;
315    ///
316    /// impl StorageObserver for VectorIndexObserver {
317    ///     fn on_event(&self, event: &StorageEvent) -> aletheiadb::core::error::Result<()> {
318    ///         match event {
319    ///             StorageEvent::NodeAnchorCreated { version_id, timestamp, .. } => {
320    ///                 // Create vector snapshot aligned with this anchor
321    ///                 println!("Creating vector snapshot for anchor {}", version_id);
322    ///                 Ok(())
323    ///             }
324    ///             _ => Ok(()), // Ignore other events
325    ///         }
326    ///     }
327    /// }
328    /// ```
329    fn on_event(&self, event: &StorageEvent) -> Result<()>;
330
331    /// Optional: Filter events this observer cares about.
332    ///
333    /// By default, observers receive all events. Override this method to receive
334    /// only specific event types, improving performance by avoiding unnecessary calls.
335    ///
336    /// # Returns
337    /// - `true`: Call `on_event()` for this event
338    /// - `false`: Skip this observer for this event type
339    ///
340    /// # Example
341    /// ```no_run
342    /// # use aletheiadb::core::observer::{StorageObserver, StorageEvent};
343    /// struct AnchorOnlyObserver;
344    ///
345    /// impl StorageObserver for AnchorOnlyObserver {
346    ///     fn on_event(&self, event: &StorageEvent) -> aletheiadb::core::error::Result<()> {
347    ///         // Only called for anchor events due to filter
348    ///         Ok(())
349    ///     }
350    ///
351    ///     fn interested_in(&self, event: &StorageEvent) -> bool {
352    ///         event.is_anchor_event()
353    ///     }
354    /// }
355    /// ```
356    fn interested_in(&self, event: &StorageEvent) -> bool {
357        // By default, interested in all events
358        let _ = event;
359        true
360    }
361}

363/// Type alias for thread-safe observer references.
364pub type Observer = Arc<dyn StorageObserver>;
365
366/// Helper for notifying multiple observers of an event.
367///
368/// This function handles error logging and ensures all observers are notified
369/// even if some fail. Observer errors do not propagate to the caller.
370///
371/// # Arguments
372/// - `observers`: List of observers to notify
373/// - `event`: The event to broadcast
374///
375/// # Design Note
376/// This is a standalone function rather than a method to keep HistoricalStorage
377/// focused on storage logic, not observer management.
378pub fn notify_observers(observers: &[Observer], event: &StorageEvent) {
379    for observer in observers {
380        // Skip if observer not interested in this event type
381        if !observer.interested_in(event) {
382            continue;
383        }
384
385        // Notify observer, log errors but don't fail
386        if let Err(e) = observer.on_event(event) {
387            #[cfg(feature = "observability")]
388            {
389                use crate::core::error::Error;
390                match &e {
391                    Error::Vector(ve) => {
392                        tracing::warn!("Observer error for event {:?}: {}", event, ve);
393                    }
394                    _ => {
395                        tracing::warn!("Observer error for event {:?}: {:?}", event, e);
396                    }
397                }
398            }
399
400            // Without observability, silently continue (errors don't block storage)
401            #[cfg(not(feature = "observability"))]
402            {
403                let _ = e; // Suppress unused variable warning
404            }
405        }
406    }
407}

601    struct TrackingObserver {
602        interested: bool,
603        should_fail: bool,
604        call_count: AtomicUsize,

607    impl TrackingObserver {
608        fn new(interested: bool, should_fail: bool) -> Self {
609            Self {
610                interested,
611                should_fail,
612                call_count: AtomicUsize::new(0),
613            }
614        }
615
616        fn count(&self) -> usize {
617            self.call_count.load(Ordering::SeqCst)
618        }
619    }
620
621    impl StorageObserver for TrackingObserver {
622        fn on_event(&self, _event: &StorageEvent) -> Result<()> {
623            self.call_count.fetch_add(1, Ordering::SeqCst);
624            if self.should_fail {
625                Err(crate::core::error::Error::Other(
626                    "Intentional failure".into(),
627                ))
628            } else {
629                Ok(())
630            }
631        }
632
633        fn interested_in(&self, _event: &StorageEvent) -> bool {
634            self.interested
635        }
636    }
