9/// Default tolerance for floating-point comparisons in normalization operations.
10///
11/// This tolerance (1e-6) is appropriate for most f32 operations where accumulated
12/// floating-point errors are expected to be small. It's used as the default for
13/// functions like [`crate::core::vector::is_normalized`] when checking if a vector has unit magnitude.
14///
15/// For stricter or looser comparisons, functions accept an explicit tolerance parameter.
16pub const NORMALIZATION_TOLERANCE: f32 = 1e-6;
17
18/// Squared magnitude threshold for detecting near-zero vectors.
19///
20/// Vectors with squared magnitude below this threshold are treated as zero vectors
21/// in normalization operations. This prevents numerical instability from denormal
22/// numbers and avoids division by very small values that could cause overflow.
23///
24/// Value: 1e-30 corresponds to magnitude ≈ 1e-15, allowing normalization of
25/// small but valid vectors (e.g., 1e-15 components). This is well above the
26/// smallest normal f32 value (1.17e-38) and provides sufficient headroom
27/// before denormal range.
28pub(crate) const SQUARED_MAGNITUDE_THRESHOLD: f32 = 1e-30;
29
30/// Maximum allowed vector dimension.
31///
32/// This limit (100,000) far exceeds typical embedding sizes and exists to prevent
33/// DoS attacks via memory exhaustion during deserialization.
34pub const MAX_VECTOR_DIMENSIONS: usize = 100_000;
35
36/// Type tag for Vector (dense f32 array) value.
37pub const TAG_VECTOR: u8 = 7;
38
39/// Type tag for SparseVector value.
40pub const TAG_SPARSE_VECTOR: u8 = 8;
41
42/// Maximum allowed vector dimension (typed).
43///
44/// Re-exported from [`MAX_VECTOR_DIMENSIONS`] for convenience.
45pub const MAX_DIMENSION: VectorDimension = VectorDimension::new(MAX_VECTOR_DIMENSIONS);
