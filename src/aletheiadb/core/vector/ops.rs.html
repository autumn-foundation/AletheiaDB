1//! General arithmetic operations and transformations for vectors.
2//!
3//! # SIMD backend (Issue #426)
4//!
5//! The reduction kernels behind these functions are provided by the
6//! [`simsimd`](https://docs.rs/simsimd) crate when the `simsimd` feature is
7//! enabled (the default), giving runtime-dispatched x86 (SSE/AVX2/AVX-512) and
8//! ARM (NEON/SVE) acceleration with no `unsafe` code in this crate. Building
9//! with `--no-default-features` selects the portable scalar fallback. The
10//! public signatures and every documented edge case (empty / zero-vector
11//! handling, clamping, NaN/Inf propagation) are identical across both paths.
12//!
13//! ## Additional metrics available from simsimd (not yet wired in)
14//!
15//! `DistanceMetric` currently exposes cosine, Euclidean, and dot product.
16//! simsimd offers further SIMD-accelerated kernels that could be surfaced as
17//! new `DistanceMetric` variants in a follow-up:
18//!
19//! - **Jensen–Shannon (`js`) and Kullback–Leibler (`kl`) divergence** via
20//!   `simsimd::ProbabilitySimilarity` — for probability-distribution vectors.
21//! - **Hamming and Jaccard/Tanimoto** via `simsimd::BinarySimilarity` — for
22//!   packed *bit* vectors (a distinct storage type from the dense `f32`
23//!   vectors handled here).
24//!
25//! Wiring these into [`super::metric::DistanceMetric`] is a deliberately
26//! deferred follow-up: each needs input-domain validation (probability
27//! simplex / bit-packed layout) and a public storage/enum surface. Note that
28//! **Haversine (great-circle) distance is *not* provided by simsimd**, so it is
29//! not part of this list.
30

43/// Computes the cosine similarity between two vectors.
44///
45/// Cosine similarity measures the cosine of the angle between two vectors,
46/// returning a value in the range `[-1.0, 1.0]`:
47/// - `1.0`: Vectors point in the same direction (identical orientation)
48/// - `0.0`: Vectors are orthogonal (perpendicular)
49/// - `-1.0`: Vectors point in opposite directions
50///
51/// # Formula
52///
53/// ```text
54/// cosine_similarity(a, b) = (a · b) / (||a|| × ||b||)
55/// ```
56///
57/// where `a · b` is the dot product and `||a||` is the L2 norm (magnitude).
58///
59/// # Arguments
60///
61/// * `a` - First vector
62/// * `b` - Second vector (must have the same length as `a`)
63///
64/// # Returns
65///
66/// * `Ok(f32)` - The cosine similarity in range `[-1.0, 1.0]`
67/// * `Err` - If vectors have different dimensions
68///
69/// # Special Cases
70///
71/// - If either vector is a zero vector (all zeros), returns `0.0`
72/// - NaN values in input will propagate to the output (result will be NaN)
73/// - Inf values in input will propagate (may result in NaN depending on combination)
74///
75/// # Example
76///
77/// ```rust
78/// use aletheiadb::core::vector::cosine_similarity;
79///
80/// // Identical vectors have similarity 1.0
81/// let a = vec![1.0, 0.0, 0.0];
82/// let b = vec![1.0, 0.0, 0.0];
83/// let sim = cosine_similarity(&a, &b).unwrap();
84/// assert!((sim - 1.0).abs() < 1e-6);
85///
86/// // Orthogonal vectors have similarity 0.0
87/// let a = vec![1.0, 0.0];
88/// let b = vec![0.0, 1.0];
89/// let sim = cosine_similarity(&a, &b).unwrap();
90/// assert!(sim.abs() < 1e-6);
91///
92/// // Opposite vectors have similarity -1.0
93/// let a = vec![1.0, 0.0];
94/// let b = vec![-1.0, 0.0];
95/// let sim = cosine_similarity(&a, &b).unwrap();
96/// assert!((sim + 1.0).abs() < 1e-6);
97/// ```
98///
99/// # Performance
100///
101/// This implementation uses SIMD acceleration via the `simsimd` crate when the
102/// `simsimd` feature is enabled (Issue #426):
103/// - **x86**: SSE / AVX2 / AVX-512, selected at runtime
104/// - **ARM**: NEON / SVE, selected at runtime
105/// - **Scalar**: portable fallback when the feature is disabled
106///
107/// The dot product and both squared magnitudes are computed with accelerated
108/// reductions and combined here; the zero-vector guard, overflow-safe magnitude
109/// product, and clamping below are unchanged.
110///
111/// # Numerical Precision
112///
113/// This implementation uses standard f32 accumulation, which provides sufficient
114/// precision for typical embedding use cases (dimensions up to ~10,000 with values
115/// in the range [-100, 100]). For these cases, relative error is typically < 1e-5.
116///
117/// **Precision characteristics:**
118/// - Typical embeddings (normalized, dim ≤ 4096): Excellent precision (< 1e-6 error)
119/// - Large vectors (dim > 10,000): May accumulate ~1e-4 relative error
120/// - Extreme magnitudes (values > 1e6): Consider normalizing inputs first
121///
122/// For applications requiring higher precision with extreme values, consider:
123/// 1. Normalizing vectors to unit length before comparison
124/// 2. Using f64 vectors with a custom implementation
125/// 3. Implementing Kahan summation (not included due to performance overhead)
126///
127/// The result is always clamped to `[-1.0, 1.0]` to handle minor floating-point
128/// inaccuracies that could produce values slightly outside this range.
129pub fn cosine_similarity(a: &[f32], b: &[f32]) -> Result<f32> {
130    check_dimensions_match(a, b)?;
131
132    // Handle empty vectors
133    if a.is_empty() {
134        return Ok(0.0);
135    }
136
137    // Use SIMD-accelerated computation when available
138    let (dot, mag_a_sq, mag_b_sq) = dot_and_magnitudes(a, b);
139
140    // Handle effectively zero vectors early to avoid numerical instability
141    // and match behavior of normalize()
142    if mag_a_sq < SQUARED_MAGNITUDE_THRESHOLD || mag_b_sq < SQUARED_MAGNITUDE_THRESHOLD {
143        return Ok(0.0);
144    }
145
146    // Compute magnitude as product of individual square roots to avoid intermediate overflow.
147    // If we did (mag_a_sq * mag_b_sq).sqrt(), the product could overflow f32::MAX
148    // even if individual squared magnitudes are within range.
149    let magnitude = mag_a_sq.sqrt() * mag_b_sq.sqrt();
150
151    // Handle zero vectors
152    if magnitude == 0.0 {
153        return Ok(0.0);
154    }
155
156    // Compute the raw result before clamping
157    let result = dot / magnitude;
158
159    // Debug assertion to detect if clamping is hiding a significant numerical issue.
160    // For correctly computed cosine similarity, values should only exceed [-1, 1]
161    // by at most machine epsilon (~1e-7 for f32). However, with subnormal numbers
162    // or extreme scales, precision loss can be larger.
163    // We use a looser threshold (0.2) to accomodate these edge cases found by fuzzing.
164    debug_assert!(
165        result.is_nan() || result.abs() <= 1.0 + 0.2,
166        "Cosine similarity {} out of valid range before clamping. \
167         This may indicate numerical issues with the input vectors.",
168        result
169    );
170
171    // Clamp to handle minor floating-point inaccuracies that could produce
172    // values slightly outside [-1.0, 1.0]
173    Ok(result.clamp(-1.0, 1.0))
174}

176/// Computes cosine similarity between pre-normalized (unit) vectors.
177///
178/// This is an optimized version of [`cosine_similarity`] for vectors that have
179/// already been L2-normalized (i.e., `||a|| = ||b|| = 1.0`). Since the magnitudes
180/// are known to be 1.0, this function skips the magnitude computation entirely
181/// and simply computes the dot product.
182///
183/// # Performance
184///
185/// This function provides approximately **2x speedup** compared to the general
186/// [`cosine_similarity`] function because it:
187/// - Skips computing `||a||²` and `||b||²`
188/// - Skips the `sqrt()` call for the magnitude product
189/// - Skips the division (or rather, divides by 1.0 implicitly)
190///
191/// # Arguments
192///
193/// * `a` - First unit vector (must be L2-normalized: `||a|| = 1.0`)
194/// * `b` - Second unit vector (must be L2-normalized: `||b|| = 1.0`)
195///
196/// # Returns
197///
198/// * `Ok(f32)` - The cosine similarity (equivalent to dot product for unit vectors)
199/// * `Err` - If vectors have different dimensions
200///
201/// # Panics (Debug Mode)
202///
203/// In debug builds, this function asserts that both vectors are approximately
204/// unit length. If a vector's magnitude differs from 1.0 by more than 1e-4,
205/// the assertion will fail.
206///
207/// # Safety Contract
208///
209/// The caller **must ensure** that both vectors are L2-normalized. If this
210/// precondition is violated:
211/// - Results will be mathematically incorrect
212/// - The debug assertion will catch this in debug builds
213/// - Release builds will silently produce wrong results
214///
215/// # Example
216///
217/// ```rust
218/// use aletheiadb::core::vector::cosine_similarity_normalized;
219///
220/// // Pre-normalize vectors
221/// let a = vec![1.0, 0.0, 0.0];  // Already unit length
222/// let b_raw = vec![1.0, 1.0, 0.0];
223/// let b_mag = (b_raw.iter().map(|x| x * x).sum::<f32>()).sqrt();
224/// let b: Vec<f32> = b_raw.iter().map(|x| x / b_mag).collect();
225///
226/// let sim = cosine_similarity_normalized(&a, &b).unwrap();
227/// // sim ≈ cos(45°) ≈ 0.707
228/// assert!((sim - 0.707).abs() < 0.01);
229/// ```
230///
231/// # When to Use
232///
233/// Use this function when:
234/// - You pre-normalize vectors at ingestion time (common practice)
235/// - You're doing many similarity comparisons against the same query vector
236/// - Performance is critical and you can guarantee unit vectors
237///
238/// Use [`cosine_similarity`] instead when:
239/// - Vectors may not be normalized
240/// - You're unsure about vector magnitudes
241/// - Correctness is more important than performance
242#[inline]
243pub fn cosine_similarity_normalized(a: &[f32], b: &[f32]) -> Result<f32> {
244    check_dimensions_match(a, b)?;
245
246    // Handle empty vectors
247    if a.is_empty() {
248        return Ok(0.0);
249    }
250
251    // Debug assertions to verify the precondition that vectors are normalized
252    #[cfg(debug_assertions)]
253    {
254        let mag_a_sq: f32 = a.iter().map(|x| x * x).sum();
255        let mag_b_sq: f32 = b.iter().map(|x| x * x).sum();
256
257        // Allow unit vectors (mag ≈ 1.0) OR zero vectors (mag ≈ 0.0) produced by normalize()
258        let a_valid = (mag_a_sq - 1.0).abs() < 1e-4 || mag_a_sq < SQUARED_MAGNITUDE_THRESHOLD;
259        debug_assert!(
260            a_valid,
261            "First vector is not unit length: ||a||² = {} (expected 1.0). \
262             Use cosine_similarity() for non-normalized vectors.",
263            mag_a_sq
264        );
265
266        let b_valid = (mag_b_sq - 1.0).abs() < 1e-4 || mag_b_sq < SQUARED_MAGNITUDE_THRESHOLD;
267        debug_assert!(
268            b_valid,
269            "Second vector is not unit length: ||b||² = {} (expected 1.0). \
270             Use cosine_similarity() for non-normalized vectors.",
271            mag_b_sq
272        );
273    }
274
275    // For unit vectors, cosine similarity = dot product
276    // We reuse the SIMD infrastructure but only need the dot product
277    let dot = dot_product_sum(a, b);
278
279    // Clamp to handle floating-point inaccuracies
280    Ok(dot.clamp(-1.0, 1.0))
281}

287/// Computes the squared Euclidean distance between two vectors.
288///
289/// The squared Euclidean distance is the sum of squared differences between
290/// corresponding elements. This is often preferred over [`euclidean_distance`]
291/// for comparisons because it avoids the expensive square root operation while
292/// preserving ordering (if `d²(a,b) < d²(a,c)` then `d(a,b) < d(a,c)`).
293///
294/// # Formula
295///
296/// ```text
297/// squared_euclidean_distance(a, b) = Σ(aᵢ - bᵢ)²
298/// ```
299///
300/// # Arguments
301///
302/// * `a` - First vector
303/// * `b` - Second vector (must have the same length as `a`)
304///
305/// # Returns
306///
307/// * `Ok(f32)` - The squared Euclidean distance (always non-negative)
308/// * `Err` - If vectors have different dimensions
309///
310/// # When to Use
311///
312/// Use squared distance instead of regular distance when:
313/// - Comparing distances (finding nearest neighbors)
314/// - Performance is critical
315/// - The actual distance value is not needed
316///
317/// Use [`euclidean_distance`] when:
318/// - You need the actual distance value
319/// - Combining with other non-squared metrics
320///
321/// # Example
322///
323/// ```rust
324/// use aletheiadb::core::vector::squared_euclidean_distance;
325///
326/// // Distance from origin
327/// let a = vec![3.0, 4.0];
328/// let b = vec![0.0, 0.0];
329/// let dist_sq = squared_euclidean_distance(&a, &b).unwrap();
330/// assert!((dist_sq - 25.0).abs() < 1e-6); // 3² + 4² = 25
331///
332/// // Use for comparison without sqrt overhead
333/// let c = vec![1.0, 1.0];
334/// let dist_bc_sq = squared_euclidean_distance(&b, &c).unwrap();
335/// // dist_bc < dist_ab because dist_bc_sq < dist_ab_sq
336/// assert!(dist_bc_sq < dist_sq);
337/// ```
338///
339/// # Performance
340///
341/// This implementation uses SIMD acceleration via the `simsimd` crate when the
342/// `simsimd` feature is enabled (Issue #426):
343/// - **x86**: SSE / AVX2 / AVX-512, selected at runtime
344/// - **ARM**: NEON / SVE, selected at runtime
345/// - **Scalar**: portable fallback when the feature is disabled
346#[inline]
347pub fn squared_euclidean_distance(a: &[f32], b: &[f32]) -> Result<f32> {
348    check_dimensions_match(a, b)?;
349
350    // Handle empty vectors
351    if a.is_empty() {
352        return Ok(0.0);
353    }
354
355    // Use SIMD-accelerated computation when available
356    Ok(squared_diff_sum(a, b))
357}

359/// Computes the Euclidean distance between two vectors.
360///
361/// The Euclidean distance (also known as L2 distance) measures the "straight-line"
362/// distance between two points in Euclidean space. It is the most common distance
363/// metric used in machine learning and data science.
364///
365/// # Formula
366///
367/// ```text
368/// euclidean_distance(a, b) = √(Σ(aᵢ - bᵢ)²)
369/// ```
370///
371/// # Arguments
372///
373/// * `a` - First vector
374/// * `b` - Second vector (must have the same length as `a`)
375///
376/// # Returns
377///
378/// * `Ok(f32)` - The Euclidean distance (always non-negative)
379/// * `Err` - If vectors have different dimensions
380///
381/// # Performance Note
382///
383/// If you only need to compare distances (e.g., finding the k nearest neighbors),
384/// consider using [`squared_euclidean_distance`] instead, as it avoids the square
385/// root operation while preserving distance ordering.
386///
387/// # Example
388///
389/// ```rust
390/// use aletheiadb::core::vector::euclidean_distance;
391///
392/// // Classic 3-4-5 right triangle
393/// let a = vec![0.0, 0.0];
394/// let b = vec![3.0, 4.0];
395/// let dist = euclidean_distance(&a, &b).unwrap();
396/// assert!((dist - 5.0).abs() < 1e-6);
397///
398/// // Same point has distance 0
399/// let c = vec![1.0, 2.0, 3.0];
400/// let dist_same = euclidean_distance(&c, &c).unwrap();
401/// assert!(dist_same.abs() < 1e-6);
402/// ```
403///
404/// # Performance
405///
406/// This implementation uses SIMD acceleration via the `simsimd` crate when the
407/// `simsimd` feature is enabled (Issue #426):
408/// - **x86**: SSE / AVX2 / AVX-512, selected at runtime
409/// - **ARM**: NEON / SVE, selected at runtime
410/// - **Scalar**: portable fallback when the feature is disabled
411///
412/// The square root is computed after the SIMD-accelerated sum.
413#[inline]
414pub fn euclidean_distance(a: &[f32], b: &[f32]) -> Result<f32> {
415    squared_euclidean_distance(a, b).map(|sq| sq.sqrt())
416}

422/// Computes the dot product (inner product) of two vectors.
423///
424/// The dot product is the sum of element-wise products: `Σ(aᵢ × bᵢ)`.
425/// It is a fundamental operation used in:
426/// - Cosine similarity (when vectors are normalized)
427/// - Linear algebra operations
428/// - Neural network computations
429/// - Projection calculations
430///
431/// # Formula
432///
433/// ```text
434/// dot_product(a, b) = Σ(aᵢ × bᵢ) = a₀×b₀ + a₁×b₁ + ... + aₙ×bₙ
435/// ```
436///
437/// # Arguments
438///
439/// * `a` - First vector
440/// * `b` - Second vector (must have the same length as `a`)
441///
442/// # Returns
443///
444/// * `Ok(f32)` - The dot product (can be positive, negative, or zero)
445/// * `Err` - If vectors have different dimensions
446///
447/// # Properties
448///
449/// - **Commutativity**: `dot(a, b) = dot(b, a)`
450/// - **Self dot product**: `dot(a, a) = ||a||²` (squared magnitude)
451/// - **Orthogonal vectors**: `dot(a, b) = 0` when vectors are perpendicular
452/// - **Parallel vectors**: `dot(a, b) = ||a|| × ||b||` (same direction)
453///   or `-||a|| × ||b||` (opposite direction)
454///
455/// # Example
456///
457/// ```rust
458/// use aletheiadb::core::vector::dot_product;
459///
460/// // Basic dot product
461/// let a = vec![1.0, 2.0, 3.0];
462/// let b = vec![4.0, 5.0, 6.0];
463/// let result = dot_product(&a, &b).unwrap();
464/// // 1×4 + 2×5 + 3×6 = 4 + 10 + 18 = 32
465/// assert!((result - 32.0).abs() < 1e-6);
466///
467/// // Self dot product equals squared magnitude
468/// let v = vec![3.0, 4.0];
469/// let self_dot = dot_product(&v, &v).unwrap();
470/// assert!((self_dot - 25.0).abs() < 1e-6); // 3² + 4² = 25
471///
472/// // Orthogonal vectors
473/// let x = vec![1.0, 0.0];
474/// let y = vec![0.0, 1.0];
475/// let ortho = dot_product(&x, &y).unwrap();
476/// assert!(ortho.abs() < 1e-6);
477/// ```
478///
479/// # Performance
480///
481/// This implementation uses SIMD acceleration via the `simsimd` crate when the
482/// `simsimd` feature is enabled (Issue #426):
483/// - **x86**: SSE / AVX2 / AVX-512, selected at runtime
484/// - **ARM**: NEON / SVE, selected at runtime
485/// - **Scalar**: portable fallback when the feature is disabled
486///
487/// This dedicated dot product function is more efficient than
488/// [`cosine_similarity`] when you only need the dot product and not the
489/// magnitudes (e.g., when working with pre-normalized vectors).
490#[inline]
491pub fn dot_product(a: &[f32], b: &[f32]) -> Result<f32> {
492    check_dimensions_match(a, b)?;
493
494    // Handle empty vectors
495    if a.is_empty() {
496        return Ok(0.0);
497    }
498
499    // Use SIMD-accelerated computation when available
500    Ok(dot_product_sum(a, b))
501}

507/// Computes the magnitude (L2 norm) of a vector.
508///
509/// The magnitude is the square root of the sum of squared elements:
510/// `||v|| = sqrt(v₀² + v₁² + ... + vₙ²)`
511///
512/// # Arguments
513///
514/// * `v` - The vector to compute the magnitude of
515///
516/// # Returns
517///
518/// The magnitude as a non-negative f32. Returns 0.0 for empty vectors.
519///
520/// # Example
521///
522/// ```rust
523/// use aletheiadb::core::vector::magnitude;
524///
525/// // Classic 3-4-5 right triangle
526/// let v = vec![3.0, 4.0];
527/// let mag = magnitude(&v);
528/// assert!((mag - 5.0).abs() < 1e-6);
529///
530/// // Unit vector has magnitude 1
531/// let unit = vec![1.0, 0.0, 0.0];
532/// assert!((magnitude(&unit) - 1.0).abs() < 1e-6);
533/// ```
534///
535/// # Performance
536///
537/// This function uses SIMD-accelerated dot product internally:
538/// `magnitude(v) = sqrt(dot_product(v, v))`
539#[inline(always)]
540pub fn magnitude(v: &[f32]) -> f32 {
541    if v.is_empty() {
542        return 0.0;
543    }
544    squared_magnitude_simd(v).sqrt()
545}

547/// Computes the squared magnitude of a vector.
548///
549/// This is equivalent to `magnitude(v).powi(2)` but avoids the square root,
550/// making it faster for comparisons where the actual magnitude isn't needed.
551///
552/// # Arguments
553///
554/// * `v` - The vector to compute the squared magnitude of
555///
556/// # Returns
557///
558/// The squared magnitude as a non-negative f32. Returns 0.0 for empty vectors.
559///
560/// # Example
561///
562/// ```rust
563/// use aletheiadb::core::vector::squared_magnitude;
564///
565/// let v = vec![3.0, 4.0];
566/// let sq_mag = squared_magnitude(&v);
567/// assert!((sq_mag - 25.0).abs() < 1e-6); // 3² + 4² = 25
568/// ```
569#[inline(always)]
570pub fn squared_magnitude(v: &[f32]) -> f32 {
571    if v.is_empty() {
572        return 0.0;
573    }
574    squared_magnitude_simd(v)
575}

577/// Normalizes a vector to unit length (L2 normalization).
578///
579/// Creates a new vector with the same direction but magnitude 1.0.
580/// For zero vectors (magnitude = 0), returns a zero vector of the same length.
581///
582/// # Arguments
583///
584/// * `v` - The vector to normalize
585///
586/// # Returns
587///
588/// A new `Vec<f32>` with unit length, or a zero vector if the input is zero.
589///
590/// # Example
591///
592/// ```rust
593/// use aletheiadb::core::vector::{normalize, magnitude};
594///
595/// let v = vec![3.0, 4.0];
596/// let unit = normalize(&v);
597///
598/// // Normalized vector has magnitude 1
599/// assert!((magnitude(&unit) - 1.0).abs() < 1e-6);
600///
601/// // Direction is preserved: [3, 4] -> [0.6, 0.8]
602/// assert!((unit[0] - 0.6).abs() < 1e-6);
603/// assert!((unit[1] - 0.8).abs() < 1e-6);
604///
605/// // Zero vector stays zero
606/// let zero = vec![0.0, 0.0];
607/// let normalized_zero = normalize(&zero);
608/// assert_eq!(normalized_zero, vec![0.0, 0.0]);
609/// ```
610///
611/// # Performance
612///
613/// This function allocates a new vector. For in-place normalization without
614/// allocation, use [`normalize_in_place`].
615/// The element-wise scale is a plain iterator map that the compiler
616/// auto-vectorizes; there is no `unsafe` code on this path (Issue #426).
617///
618/// # Note on Dimension Validation
619///
620/// Unlike two-vector functions like [`cosine_similarity`], normalization functions
621/// do not validate against `MAX_VECTOR_DIMENSIONS`. This is intentional because:
622/// - Single-vector operations don't have dimension mismatch issues
623/// - Dimension limits are enforced at storage time (see [`crate::core::PropertyValue::vector`])
624/// - Additional checks would add overhead without safety benefit
625#[inline]
626pub fn normalize(v: &[f32]) -> Vec<f32> {
627    let sq_mag = squared_magnitude(v);
628    // Use squared magnitude threshold to avoid denormal number issues.
629    // See SQUARED_MAGNITUDE_THRESHOLD for details.
630    if sq_mag < SQUARED_MAGNITUDE_THRESHOLD {
631        // Return zero vector of same length
632        return vec![0.0; v.len()];
633    }
634    // Compute 1/sqrt(sq_mag) directly to avoid intermediate variable.
635    let inv_mag = 1.0 / sq_mag.sqrt();
636    // Safe element-wise scale; the compiler auto-vectorizes this map.
637    v.iter().map(|x| x * inv_mag).collect()
638}

640/// Normalizes a vector to unit length in place.
641///
642/// Modifies the vector in place to have magnitude 1.0.
643/// For zero vectors (magnitude = 0), leaves the vector unchanged.
644///
645/// # Arguments
646///
647/// * `v` - The vector to normalize (modified in place)
648///
649/// # Example
650///
651/// ```rust
652/// use aletheiadb::core::vector::{normalize_in_place, magnitude};
653///
654/// let mut v = vec![3.0, 4.0];
655/// normalize_in_place(&mut v);
656///
657/// // Now has magnitude 1
658/// assert!((magnitude(&v) - 1.0).abs() < 1e-6);
659///
660/// // Direction is preserved: [3, 4] -> [0.6, 0.8]
661/// assert!((v[0] - 0.6).abs() < 1e-6);
662/// assert!((v[1] - 0.8).abs() < 1e-6);
663/// ```
664///
665/// # Performance
666///
667/// This function modifies the vector in place without allocation, making it
668/// more efficient than [`normalize`] when a new vector isn't needed.
669/// The element-wise scale is a plain scalar loop that the compiler
670/// auto-vectorizes; there is no `unsafe` code on this path (Issue #426).
671#[inline]
672pub fn normalize_in_place(v: &mut [f32]) {
673    let sq_mag = squared_magnitude(v);
674    // Use squared magnitude threshold to avoid denormal number issues.
675    // See SQUARED_MAGNITUDE_THRESHOLD for details.
676    if sq_mag < SQUARED_MAGNITUDE_THRESHOLD {
677        // Zero out the vector to match `normalize` behavior
678        v.fill(0.0);
679        return;
680    }
681    // Compute 1/sqrt(sq_mag) directly to avoid intermediate variable
682    let inv_mag = 1.0 / sq_mag.sqrt();
683    scale_in_place(v, inv_mag);
684}

686/// Checks if a vector is normalized (has magnitude approximately 1.0).
687///
688/// This is useful for validating that vectors are properly normalized before
689/// using optimized functions like [`cosine_similarity_normalized`].
690///
691/// # Arguments
692///
693/// * `v` - The vector to check
694/// * `tolerance` - Maximum allowed deviation from 1.0 (e.g., 1e-6)
695///
696/// # Returns
697///
698/// `true` if the magnitude is within `tolerance` of 1.0, `false` otherwise.
699///
700/// # Example
701///
702/// ```rust
703/// use aletheiadb::core::vector::{is_normalized, normalize};
704///
705/// let v = vec![3.0, 4.0];
706/// assert!(!is_normalized(&v, 1e-6));
707///
708/// let unit = normalize(&v);
709/// assert!(is_normalized(&unit, 1e-6));
710/// ```
711#[inline]
712pub fn is_normalized(v: &[f32], tolerance: f32) -> bool {
713    debug_assert!(
714        (0.0..1.0).contains(&tolerance),
715        "tolerance must be in range [0.0, 1.0), got {}",
716        tolerance
717    );
718    // Use squared_magnitude to avoid sqrt for better numerical stability
719    // |magnitude - 1.0| <= tolerance  ⟺  (1-tolerance)² <= ||v||² <= (1+tolerance)²
720    let sq_mag = squared_magnitude(v);
721    let lower = (1.0 - tolerance).max(0.0).powi(2);
722    let upper = (1.0 + tolerance).powi(2);
723    sq_mag >= lower && sq_mag <= upper
724}

726/// Checks if a vector is normalized using the default tolerance.
727///
728/// This is a convenience wrapper around [`is_normalized`] that uses
729/// [`crate::core::vector::NORMALIZATION_TOLERANCE`] (1e-6) as the tolerance value.
730///
731/// # Example
732///
733/// ```rust
734/// use aletheiadb::core::vector::{is_normalized_default, normalize};
735///
736/// let v = vec![3.0, 4.0];
737/// assert!(!is_normalized_default(&v));
738///
739/// let unit = normalize(&v);
740/// assert!(is_normalized_default(&unit));
741/// ```
742#[inline]
743pub fn is_normalized_default(v: &[f32]) -> bool {
744    is_normalized(v, super::constants::NORMALIZATION_TOLERANCE)
