9/// Validates that a vector contains no NaN or Infinity values.
10///
11/// This function scans all elements of the vector and returns an error if any
12/// invalid floating-point values are found. NaN and Infinity values would cause
13/// incorrect results in distance/similarity calculations and should be caught early.
14///
15/// # Arguments
16///
17/// * `v` - The vector slice to validate
18///
19/// # Returns
20///
21/// * `Ok(())` if all elements are valid finite numbers
22/// * `Err(VectorError::ContainsNaN)` if any NaN values are found
23/// * `Err(VectorError::ContainsInfinity)` if any Infinity values are found (checked after NaN)
24///
25/// # Note
26///
27/// NaN is checked first, so if a vector contains both NaN and Infinity values,
28/// the NaN error will be returned. This is because NaN values are generally more
29/// problematic (NaN != NaN, propagates through calculations).
30///
31/// # Example
32///
33/// ```rust
34/// use aletheiadb::core::vector::validate_vector;
35///
36/// // Valid vector
37/// let v = vec![1.0, 2.0, 3.0];
38/// assert!(validate_vector(&v).is_ok());
39///
40/// // Vector with NaN
41/// let v_nan = vec![1.0, f32::NAN, 3.0];
42/// assert!(validate_vector(&v_nan).is_err());
43///
44/// // Vector with Infinity
45/// let v_inf = vec![1.0, f32::INFINITY, 3.0];
46/// assert!(validate_vector(&v_inf).is_err());
47///
48/// // Empty vector is valid
49/// let empty: Vec<f32> = vec![];
50/// assert!(validate_vector(&empty).is_ok());
51/// ```
52#[inline]
53pub fn validate_vector(v: &[f32]) -> Result<()> {
54    // Use a single pass to count both NaN and Infinity values for efficiency.
55    // This is more efficient than iterating twice, especially for large vectors.
56    let (nan_count, inf_count) = v.iter().fold((0usize, 0usize), |(nan, inf), &val| {
57        if val.is_nan() {
58            (nan + 1, inf)
59        } else if val.is_infinite() {
60            (nan, inf + 1)
61        } else {
62            (nan, inf)
63        }
64    });
65
66    // Per the function's contract, NaN is checked first.
67    if nan_count > 0 {
68        return Err(Error::Vector(VectorError::ContainsNaN { count: nan_count }));
69    }
70
71    if inf_count > 0 {
72        return Err(Error::Vector(VectorError::ContainsInfinity {
73            count: inf_count,
74        }));
75    }
76
77    Ok(())
78}

80/// Checks that two vectors have matching dimensions.
81///
82/// Many vector operations require vectors of equal length. This function provides
83/// a convenient way to validate dimension compatibility before performing operations.
84///
85/// # Arguments
86///
87/// * `a` - The first vector (its length is considered the "expected" dimension)
88/// * `b` - The second vector (its length is compared against `a`)
89///
90/// # Returns
91///
92/// * `Ok(())` if both vectors have the same length
93/// * `Err(VectorError::DimensionMismatch)` if lengths differ
94///
95/// # Example
96///
97/// ```rust
98/// use aletheiadb::core::vector::check_dimensions_match;
99///
100/// let v1 = vec![1.0, 2.0, 3.0];
101/// let v2 = vec![4.0, 5.0, 6.0];
102/// let v3 = vec![1.0, 2.0];
103///
104/// // Same dimensions - OK
105/// assert!(check_dimensions_match(&v1, &v2).is_ok());
106///
107/// // Different dimensions - Error
108/// assert!(check_dimensions_match(&v1, &v3).is_err());
109///
110/// // Empty vectors match
111/// let empty1: Vec<f32> = vec![];
112/// let empty2: Vec<f32> = vec![];
113/// assert!(check_dimensions_match(&empty1, &empty2).is_ok());
114/// ```
115#[inline]
116pub fn check_dimensions_match(a: &[f32], b: &[f32]) -> Result<()> {
117    if a.len() != b.len() {
118        return Err(Error::Vector(VectorError::DimensionMismatch {
119            expected: a.len(),
120            actual: b.len(),
121        }));
122    }
123    Ok(())
124}

126/// Validates a vector and checks that its dimension is within bounds.
127///
128/// This is a convenience function that combines validation (NaN/Infinity checking)
129/// with dimension bounds checking. Useful when processing user-provided vectors
130/// that need both validation and size constraints.
131///
132/// # Arguments
133///
134/// * `v` - The vector slice to validate
135/// * `max_dimension` - The maximum allowed dimension (length)
136///
137/// # Returns
138///
139/// * `Ok(())` if the vector is valid and within dimension bounds
140/// * `Err(VectorError::ContainsNaN)` if any NaN values are found
141/// * `Err(VectorError::ContainsInfinity)` if any Infinity values are found
142/// * `Err(VectorError::DimensionTooLarge)` if the vector exceeds max_dimension
143///
144/// # Example
145///
146/// ```rust
147/// use aletheiadb::core::vector::validate_vector_with_bounds;
148///
149/// // Valid vector within bounds
150/// let v = vec![1.0, 2.0, 3.0];
151/// assert!(validate_vector_with_bounds(&v, 10).is_ok());
152///
153/// // Vector too large
154/// assert!(validate_vector_with_bounds(&v, 2).is_err());
155///
156/// // Vector with invalid values
157/// let v_nan = vec![1.0, f32::NAN];
158/// assert!(validate_vector_with_bounds(&v_nan, 10).is_err());
159/// ```
160#[inline]
161pub fn validate_vector_with_bounds(v: &[f32], max_dimension: usize) -> Result<()> {
162    // Check dimension first (fast check)
163    if v.len() > max_dimension {
164        return Err(Error::Vector(VectorError::DimensionTooLarge {
165            dimension: v.len(),
166            max_allowed: max_dimension,
167        }));
168    }
169
170    // Then validate contents
171    validate_vector(v)
172}
