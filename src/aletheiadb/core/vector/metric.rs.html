11/// Specifies which distance or similarity metric to use for vector operations.
12///
13/// This enum provides a unified interface for computing distances and similarities
14/// between vectors, dispatching to the appropriate underlying function based on the
15/// selected metric.
16///
17/// # Choosing a Metric
18///
19/// | Metric | Best For | Range | Notes |
20/// |--------|----------|-------|-------|
21/// | [`Cosine`](DistanceMetric::Cosine) | Semantic similarity, text embeddings | [-1, 1] similarity, [0, 2] distance | Scale-invariant, most common for embeddings |
22/// | [`Euclidean`](DistanceMetric::Euclidean) | Spatial data, image features | [0, ∞) distance | Sensitive to vector magnitude |
23/// | [`DotProduct`](DistanceMetric::DotProduct) | Pre-normalized vectors, MaxIP search | (-∞, ∞) | Fastest; requires normalized vectors for cosine-like behavior |
24///
25/// # Example
26///
27/// ```rust
28/// use aletheiadb::core::vector::DistanceMetric;
29///
30/// let a = vec![1.0, 0.0, 0.0];
31/// let b = vec![0.0, 1.0, 0.0];
32///
33/// // Using cosine similarity (orthogonal vectors = 0 similarity)
34/// let similarity = DistanceMetric::Cosine.compute_similarity(&a, &b).unwrap();
35/// assert!((similarity - 0.0).abs() < 1e-6);
36///
37/// // Using euclidean distance
38/// let distance = DistanceMetric::Euclidean.compute_distance(&a, &b).unwrap();
39/// assert!((distance - std::f32::consts::SQRT_2).abs() < 1e-6);
40/// ```
41///
42/// # Performance
43///
44/// All metrics use SIMD acceleration (AVX2/SSE2) when available. For maximum
45/// performance with large-scale similarity search:
46///
47/// 1. Pre-normalize vectors with [`crate::core::vector::normalize`] or [`crate::core::vector::normalize_in_place`]
48/// 2. Use [`DotProduct`](DistanceMetric::DotProduct) metric (single SIMD operation)
49/// 3. Store normalized vectors to avoid repeated normalization
50///
51/// # Future Enhancements
52///
53/// - **Serialization**: When serde is added as a dependency, this enum will
54///   support `#[serde(rename_all = "snake_case")]` for JSON/config serialization
55/// - **Batch operations**: `compute_distances_batch()` for SIMD-optimized
56///   multi-vector distance computation
57#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
58pub enum DistanceMetric {
59    /// Cosine similarity/distance.
60    ///
61    /// Measures the cosine of the angle between two vectors, making it
62    /// **scale-invariant** (only direction matters, not magnitude).
63    ///
64    /// - **Similarity**: Range [-1, 1] where 1 = identical direction,
65    ///   0 = orthogonal, -1 = opposite direction
66    /// - **Distance**: Computed as `1 - similarity`, range [0, 2]
67    ///
68    /// # When to Use
69    ///
70    /// - Text embeddings (word2vec, BERT, OpenAI embeddings)
71    /// - Semantic similarity where magnitude doesn't matter
72    /// - When vectors may have different scales
73    ///
74    /// # Implementation Note
75    ///
76    /// Uses [`cosine_similarity`] internally, which handles zero vectors
77    /// by returning 0.0 similarity.
78    #[default]
79    Cosine,
80
81    /// Euclidean (L2) distance.
82    ///
83    /// Measures the straight-line distance between two points in vector space.
84    /// Also known as the L2 norm of the difference vector.
85    ///
86    /// - **Distance**: Range [0, ∞) where 0 = identical vectors
87    /// - **Similarity**: Computed as `1 / (1 + distance)`, range (0, 1]
88    ///
89    /// # When to Use
90    ///
91    /// - Spatial data (coordinates, positions)
92    /// - Image feature vectors
93    /// - When absolute magnitude differences matter
94    /// - K-means clustering (uses squared Euclidean internally)
95    ///
96    /// # Implementation Note
97    ///
98    /// Uses [`euclidean_distance`] internally, which uses SIMD-accelerated
99    /// squared difference computation.
100    Euclidean,
101
102    /// Inner (dot) product.
103    ///
104    /// Computes the sum of element-wise products. For normalized vectors,
105    /// this equals cosine similarity but is faster (single SIMD operation).
106    ///
107    /// - **Raw value**: Range (-∞, ∞)
108    /// - **Similarity**: Raw dot product value (higher = more similar)
109    /// - **Distance**: Computed as `1 - dot_product`, which is meaningful
110    ///   only for normalized vectors
111    ///
112    /// # When to Use
113    ///
114    /// - **Maximum Inner Product Search (MIPS)**: When you want the highest
115    ///   dot product, not necessarily the closest vector
116    /// - **Pre-normalized vectors**: Equivalent to cosine but faster
117    /// - **Learned embeddings**: Some models are trained with dot product loss
118    ///
119    /// # Important
120    ///
121    /// For non-normalized vectors, dot product is **not** a proper distance
122    /// metric (doesn't satisfy triangle inequality). Use [`Cosine`](DistanceMetric::Cosine)
123    /// for general similarity or ensure vectors are normalized first.
124    DotProduct,
125}
126
127impl DistanceMetric {
128    /// Computes the distance between two vectors using this metric.
129    ///
130    /// Lower values indicate more similar vectors.
131    ///
132    /// # Returns
133    ///
134    /// - [`Cosine`](DistanceMetric::Cosine): `1 - cosine_similarity`, range [0, 2]
135    /// - [`Euclidean`](DistanceMetric::Euclidean): L2 distance, range [0, ∞)
136    /// - [`DotProduct`](DistanceMetric::DotProduct): `1 - dot_product` (meaningful only for normalized vectors)
137    ///
138    /// # Errors
139    ///
140    /// Returns an error if the vectors have different lengths.
141    ///
142    /// # Example
143    ///
144    /// ```rust
145    /// use aletheiadb::core::vector::DistanceMetric;
146    ///
147    /// let a = vec![1.0, 0.0];
148    /// let b = vec![1.0, 0.0];
149    ///
150    /// // Identical vectors have zero distance
151    /// assert!((DistanceMetric::Cosine.compute_distance(&a, &b).unwrap() - 0.0).abs() < 1e-6);
152    /// assert!((DistanceMetric::Euclidean.compute_distance(&a, &b).unwrap() - 0.0).abs() < 1e-6);
153    /// ```
154    #[inline]
155    pub fn compute_distance(&self, a: &[f32], b: &[f32]) -> Result<f32> {
156        match self {
157            DistanceMetric::Cosine => cosine_similarity(a, b).map(|sim| 1.0 - sim),
158            DistanceMetric::Euclidean => euclidean_distance(a, b),
159            DistanceMetric::DotProduct => dot_product(a, b).map(|dp| 1.0 - dp),
160        }
161    }
162
163    /// Computes the similarity between two vectors using this metric.
164    ///
165    /// Higher values indicate more similar vectors.
166    ///
167    /// # Returns
168    ///
169    /// - [`Cosine`](DistanceMetric::Cosine): Cosine similarity, range [-1, 1]
170    /// - [`Euclidean`](DistanceMetric::Euclidean): `1 / (1 + distance)`, range (0, 1]
171    /// - [`DotProduct`](DistanceMetric::DotProduct): Raw dot product, range (-∞, ∞)
172    ///
173    /// # Errors
174    ///
175    /// Returns an error if the vectors have different lengths.
176    ///
177    /// # Example
178    ///
179    /// ```rust
180    /// use aletheiadb::core::vector::DistanceMetric;
181    ///
182    /// let a = vec![1.0, 0.0];
183    /// let b = vec![1.0, 0.0];
184    ///
185    /// // Identical vectors have maximum similarity
186    /// assert!((DistanceMetric::Cosine.compute_similarity(&a, &b).unwrap() - 1.0).abs() < 1e-6);
187    /// assert!((DistanceMetric::Euclidean.compute_similarity(&a, &b).unwrap() - 1.0).abs() < 1e-6);
188    /// ```
189    #[inline]
190    pub fn compute_similarity(&self, a: &[f32], b: &[f32]) -> Result<f32> {
191        match self {
192            DistanceMetric::Cosine => cosine_similarity(a, b),
193            DistanceMetric::Euclidean => euclidean_distance(a, b).map(|dist| 1.0 / (1.0 + dist)),
194            DistanceMetric::DotProduct => dot_product(a, b),
195        }
196    }
197
198    /// Returns a human-readable name for this metric.
199    ///
200    /// # Example
201    ///
202    /// ```rust
203    /// use aletheiadb::core::vector::DistanceMetric;
204    ///
205    /// assert_eq!(DistanceMetric::Cosine.name(), "cosine");
206    /// assert_eq!(DistanceMetric::Euclidean.name(), "euclidean");
207    /// assert_eq!(DistanceMetric::DotProduct.name(), "dot_product");
208    /// ```
209    #[inline]
210    pub const fn name(&self) -> &'static str {
211        match self {
212            DistanceMetric::Cosine => "cosine",
213            DistanceMetric::Euclidean => "euclidean",
214            DistanceMetric::DotProduct => "dot_product",
215        }
216    }
217
218    /// Returns whether this metric requires normalized vectors for optimal results.
219    ///
220    /// - [`Cosine`](DistanceMetric::Cosine): No (handles normalization internally)
221    /// - [`Euclidean`](DistanceMetric::Euclidean): No (works with any vectors)
222    /// - [`DotProduct`](DistanceMetric::DotProduct): Yes (otherwise not a proper similarity)
223    ///
224    /// # Example
225    ///
226    /// ```rust
227    /// use aletheiadb::core::vector::DistanceMetric;
228    ///
229    /// assert!(!DistanceMetric::Cosine.requires_normalized_vectors());
230    /// assert!(!DistanceMetric::Euclidean.requires_normalized_vectors());
231    /// assert!(DistanceMetric::DotProduct.requires_normalized_vectors());
232    /// ```
233    #[inline]
234    pub const fn requires_normalized_vectors(&self) -> bool {
235        matches!(self, DistanceMetric::DotProduct)
236    }
237}

241        write!(f, "{}", self.name())
242    }
