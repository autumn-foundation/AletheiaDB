11/// A sparse vector representation optimized for vectors with many zero values.
12///
13/// Sparse vectors store only non-zero values along with their indices, making them
14/// memory-efficient for high-dimensional vectors where most values are zero.
15/// This is particularly useful for algorithms like BM25 and SPLADE.
16///
17/// # Format
18///
19/// - `indices`: Sorted array of non-zero element positions
20/// - `values`: Corresponding non-zero values
21/// - `dimension`: Total vector dimension (including zeros)
22///
23/// # Invariants
24///
25/// The struct maintains these invariants:
26/// 1. `indices.len() == values.len()` (each index has a corresponding value)
27/// 2. `indices` are sorted in ascending order
28/// 3. All indices are `< dimension`
29/// 4. No duplicate indices
30/// 5. All values are non-zero (no stored zeros)
31///
32/// # Example
33///
34/// ```rust
35/// use aletheiadb::core::vector::SparseVec;
36///
37/// // Sparse vector: [0.0, 1.5, 0.0, 0.0, 2.3, 0.0, 0.0, 0.8]
38/// let sparse = SparseVec::new(
39///     vec![1, 4, 7],        // indices of non-zero values
40///     vec![1.5, 2.3, 0.8],  // corresponding values
41///     8                      // total dimension
42/// ).unwrap();
43///
44/// assert_eq!(sparse.nnz(), 3);  // 3 non-zero elements
45/// assert_eq!(sparse.dimension(), 8);
46/// ```
47///
48/// # Use Cases
49///
50/// - **BM25**: Text retrieval using term frequency vectors
51/// - **SPLADE**: Sparse learned embeddings for information retrieval
52/// - **One-hot encodings**: Categorical features with single non-zero value
53/// - **TF-IDF**: Document vectors with few non-zero terms
54///
55/// # Performance
56///
57/// - Space complexity: O(nnz) where nnz = number of non-zero elements
58/// - Dense equivalent would be: O(dimension)
59/// - Memory savings can be 10-1000x for sparse data
60///
61/// # Equality and Comparison
62///
63/// While this type implements `PartialEq` for compatibility with `PropertyValue`,
64/// direct equality comparison using `==` is **not recommended** for floating-point
65/// values. NaN != NaN (IEEE 754) and floating-point precision issues can cause
66/// semantically equal vectors to compare unequal. For robust equality checks,
67/// use [`approx_eq`](Self::approx_eq) with an appropriate epsilon value instead.
68#[derive(Debug, Clone, PartialEq)]
69pub struct SparseVec {
70    /// Indices of non-zero elements (sorted, unique, all < dimension).
71    indices: Vec<u32>,
72    /// Non-zero values corresponding to indices.
73    values: Vec<f32>,
74    /// Total dimension of the vector (including zeros).
75    dimension: u32,
76}

78impl SparseVec {
79    /// Creates a new sparse vector.
80    ///
81    /// # Arguments
82    ///
83    /// * `indices` - Positions of non-zero elements (will be sorted if not already)
84    /// * `values` - Non-zero values corresponding to indices
85    /// * `dimension` - Total vector dimension (must be > max(indices))
86    ///
87    /// # Returns
88    ///
89    /// - `Ok(SparseVec)` if the input is valid
90    /// - `Err` if validation fails (see Errors section)
91    ///
92    /// # Errors
93    ///
94    /// Returns `VectorError::DimensionMismatch` if:
95    /// - `indices.len() != values.len()`
96    /// - Any index >= dimension
97    ///
98    /// Returns `VectorError::InvalidSparseVector` if:
99    /// - Duplicate indices found
100    /// - Zero values found (sparse vectors should only store non-zero values)
101    ///
102    /// Returns `VectorError::DimensionTooLarge` if dimension exceeds MAX_VECTOR_DIMENSIONS.
103    ///
104    /// Returns `VectorError::ContainsNaN` if any value is NaN.
105    ///
106    /// Returns `VectorError::ContainsInfinity` if any value is infinite.
107    ///
108    /// # Example
109    ///
110    /// ```rust
111    /// use aletheiadb::core::vector::SparseVec;
112    ///
113    /// // Valid sparse vector
114    /// let sparse = SparseVec::new(vec![0, 2, 5], vec![1.0, 2.0, 3.0], 10).unwrap();
115    /// assert_eq!(sparse.nnz(), 3);
116    ///
117    /// // Error: mismatched lengths
118    /// let result = SparseVec::new(vec![0, 1], vec![1.0], 10);
119    /// assert!(result.is_err());
120    ///
121    /// // Error: index out of bounds
122    /// let result = SparseVec::new(vec![0, 10], vec![1.0, 2.0], 10);
123    /// assert!(result.is_err());
124    /// ```
125    pub fn new(mut indices: Vec<u32>, mut values: Vec<f32>, dimension: u32) -> Result<Self> {
126        // Validate dimension
127        if dimension as usize > MAX_VECTOR_DIMENSIONS {
128            return Err(VectorError::DimensionTooLarge {
129                dimension: dimension as usize,
130                max_allowed: MAX_VECTOR_DIMENSIONS,
131            }
132            .into());
133        }
134
135        // Validate lengths match
136        if indices.len() != values.len() {
137            return Err(VectorError::DimensionMismatch {
138                expected: indices.len(),
139                actual: values.len(),
140            }
141            .into());
142        }
143
144        // Validate and normalize data
145        if !indices.is_empty() {
146            // OPTIMIZATION: Try to validate in-place first to avoid allocation/sorting
147            // if the input is already sorted (common case).
148            // This fast path is O(N) and zero-allocation.
149            let mut is_sorted = true;
150
151            // Check first element
152            if indices[0] >= dimension {
153                return Err(Error::Vector(VectorError::InvalidSparseVector {
154                    reason: format!(
155                        "Index {} is out of bounds for dimension {}",
156                        indices[0], dimension
157                    ),
158                }));
159            }
160            Self::validate_value(values[0])?;
161
162            for i in 1..indices.len() {
163                let prev = indices[i - 1];
164                let curr = indices[i];
165
166                if curr <= prev {
167                    if curr == prev {
168                        return Err(Error::Vector(VectorError::InvalidSparseVector {
169                            reason: format!("Duplicate index {} found", curr),
170                        }));
171                    }
172                    // Unsorted
173                    is_sorted = false;
174                    break;
175                }
176
177                if curr >= dimension {
178                    return Err(Error::Vector(VectorError::InvalidSparseVector {
179                        reason: format!(
180                            "Index {} is out of bounds for dimension {}",
181                            curr, dimension
182                        ),
183                    }));
184                }
185
186                Self::validate_value(values[i])?;
187            }
188
189            if is_sorted {
190                return Ok(Self {
191                    indices,
192                    values,
193                    dimension,
194                });
195            }
196
197            // Fallback to slow path: sort and re-validate
198            // Note: We re-validate values, which is redundant but safe.
199            // Since we moved `indices` and `values` by reference in the loop above,
200            // we can still consume them here.
201
202            // Sort by indices
203            let mut index_value_pairs: Vec<(u32, f32)> = indices.into_iter().zip(values).collect();
204            index_value_pairs.sort_by_key(|(idx, _)| *idx);
205
206            // Check for duplicates and out-of-bounds indices
207            let mut prev_idx = None;
208            for (idx, val) in &index_value_pairs {
209                Self::validate_value(*val)?;
210
211                // Check index bounds
212                if *idx >= dimension {
213                    return Err(Error::Vector(VectorError::InvalidSparseVector {
214                        reason: format!(
215                            "Index {} is out of bounds for dimension {}",
216                            idx, dimension
217                        ),
218                    }));
219                }
220                // Check for duplicates
221                if let Some(prev) = prev_idx
222                    && prev == *idx
223                {
224                    return Err(Error::Vector(VectorError::InvalidSparseVector {
225                        reason: format!("Duplicate index {} found", idx),
226                    }));
227                }
228                prev_idx = Some(*idx);
229            }
230
231            // Unzip back to separate vectors
232            let (sorted_indices, sorted_values): (Vec<u32>, Vec<f32>) =
233                index_value_pairs.into_iter().unzip();
234            indices = sorted_indices;
235            values = sorted_values;
236        }
237
238        Ok(Self {
239            indices,
240            values,
241            dimension,
242        })
243    }
244
245    #[inline(always)]
246    fn validate_value(val: f32) -> Result<()> {
247        if val.is_nan() {
248            return Err(VectorError::ContainsNaN { count: 1 }.into());
249        }
250        if val.is_infinite() {
251            return Err(VectorError::ContainsInfinity { count: 1 }.into());
252        }
253        if val == 0.0 {
254            return Err(Error::Vector(VectorError::InvalidSparseVector {
255                reason: "Sparse vector contains zero value".to_string(),
256            }));
257        }
258        Ok(())
259    }

329    /// Converts this sparse vector to a dense vector representation.
330    ///
331    /// Creates a new vector of length `dimension()` with zeros everywhere
332    /// except at the specified indices.
333    ///
334    /// # Example
335    ///
336    /// ```rust
337    /// use aletheiadb::core::vector::SparseVec;
338    ///
339    /// let sparse = SparseVec::new(vec![1, 3], vec![1.5, 2.5], 5).unwrap();
340    /// let dense = sparse.to_dense();
341    /// assert_eq!(dense, vec![0.0, 1.5, 0.0, 2.5, 0.0]);
342    /// ```
343    pub fn to_dense(&self) -> Vec<f32> {
344        let mut dense = vec![0.0; self.dimension as usize];
345        for (&idx, &val) in self.indices.iter().zip(self.values.iter()) {
346            dense[idx as usize] = val;
347        }
348        dense
349    }

351    /// Computes the squared magnitude (L2 norm squared) of this sparse vector.
352    ///
353    /// This is more efficient than computing the full magnitude when you only
354    /// need to compare magnitudes (since sqrt is monotonic for positive values).
355    ///
356    /// # Example
357    ///
358    /// ```rust
359    /// use aletheiadb::core::vector::SparseVec;
360    ///
361    /// let sparse = SparseVec::new(vec![0, 1, 2], vec![1.0, 2.0, 2.0], 5).unwrap();
362    /// // magnitude² = 1² + 2² + 2² = 1 + 4 + 4 = 9
363    /// assert_eq!(sparse.squared_magnitude(), 9.0);
364    /// ```
365    pub fn squared_magnitude(&self) -> f32 {
366        self.values.iter().map(|v| v * v).sum()
367    }

369    /// Computes the magnitude (L2 norm) of this sparse vector.
370    ///
371    /// # Example
372    ///
373    /// ```rust
374    /// use aletheiadb::core::vector::SparseVec;
375    ///
376    /// let sparse = SparseVec::new(vec![0, 1], vec![3.0, 4.0], 5).unwrap();
377    /// // magnitude = sqrt(3² + 4²) = sqrt(9 + 16) = 5.0
378    /// assert_eq!(sparse.magnitude(), 5.0);
379    /// ```

381    pub fn magnitude(&self) -> f32 {
382        self.squared_magnitude().sqrt()

385    /// Checks if this sparse vector is approximately equal to another within a tolerance.
386    ///
387    /// This method provides an epsilon-based comparison that handles floating-point
388    /// precision issues. Two sparse vectors are considered approximately equal if:
389    /// - They have the same dimension
390    /// - They have the same indices
391    /// - All corresponding values differ by less than epsilon
392    ///
393    /// # Arguments
394    ///
395    /// * `other` - The sparse vector to compare with
396    /// * `epsilon` - Maximum allowed difference for each value (typically 1e-6 for f32)
397    ///
398    /// # Example
399    ///
400    /// ```rust
401    /// use aletheiadb::core::vector::SparseVec;
402    ///
403    /// let a = SparseVec::new(vec![0, 2], vec![1.0, 2.0], 5).unwrap();
404    /// let b = SparseVec::new(vec![0, 2], vec![1.0000001, 2.0000001], 5).unwrap();
405    ///
406    /// // Small floating-point differences are tolerated
407    /// assert!(a.approx_eq(&b, 1e-5));
408    /// assert!(!a.approx_eq(&b, 1e-10));
409    /// ```
410    ///
411    /// # Note
412    ///
413    /// This is the **recommended way to compare sparse vectors**. While `SparseVec`
414    /// implements `PartialEq` for compatibility with `PropertyValue`, direct equality
415    /// via `==` is not recommended due to floating-point comparison concerns.
416    /// See the "Equality and Comparison" section in the type documentation for details.
417    pub fn approx_eq(&self, other: &SparseVec, epsilon: f32) -> bool {
418        self.dimension == other.dimension
419            && self.indices == other.indices
420            && self.values.len() == other.values.len()
421            && self
422                .values
423                .iter()
424                .zip(other.values.iter())
425                .all(|(a, b)| (a - b).abs() < epsilon)
426    }
427}

433/// Computes the dot product between two sparse vectors.
434///
435/// The dot product is computed efficiently by iterating only over non-zero
436/// elements. Vectors with different dimensions are handled gracefully by
437/// only considering indices that exist in both vectors.
438///
439/// # Arguments
440///
441/// * `a` - First sparse vector
442/// * `b` - Second sparse vector
443///
444/// # Returns
445///
446/// * `Ok(f32)` - The dot product
447/// * `Err` - Never returns an error (kept for API consistency with dense vectors)
448///
449/// # Algorithm
450///
451/// Uses a merge-like algorithm since indices are sorted:
452/// 1. Maintain pointers to both index arrays
453/// 2. When indices match, multiply values and add to sum
454/// 3. Advance pointer with smaller index
455/// 4. Time complexity: O(nnz_a + nnz_b)
456///
457/// # Example
458///
459/// ```rust
460/// use aletheiadb::core::vector::{SparseVec, sparse_dot_product};
461///
462/// // Sparse vectors: [1, 0, 2, 0, 0] and [0, 0, 3, 0, 4]
463/// let a = SparseVec::new(vec![0, 2], vec![1.0, 2.0], 5).unwrap();
464/// let b = SparseVec::new(vec![2, 4], vec![3.0, 4.0], 5).unwrap();
465///
466/// // Only index 2 overlaps: 2.0 * 3.0 = 6.0
467/// let dot = sparse_dot_product(&a, &b).unwrap();
468/// assert_eq!(dot, 6.0);
469/// ```
470///
471/// # Performance
472///
473/// For vectors with nnz_a and nnz_b non-zero elements:
474/// - Time: O(nnz_a + nnz_b) - linear in sparsity
475/// - Space: O(1) - no allocations
476/// - Much faster than dense dot product when vectors are sparse
477pub fn sparse_dot_product(a: &SparseVec, b: &SparseVec) -> Result<f32> {
478    if a.dimension() != b.dimension() {
479        return Err(VectorError::DimensionMismatch {
480            expected: a.dimension(),
481            actual: b.dimension(),
482        }
483        .into());
484    }
485
486    let mut sum = 0.0f32;
487    let mut i = 0;
488    let mut j = 0;
489
490    let a_indices = a.indices();
491    let a_values = a.values();
492    let b_indices = b.indices();
493    let b_values = b.values();
494
495    // Merge-like algorithm since indices are sorted
496    while i < a_indices.len() && j < b_indices.len() {
497        if a_indices[i] == b_indices[j] {
498            // Indices match - multiply and add
499            sum += a_values[i] * b_values[j];
500            i += 1;
501            j += 1;
502        } else if a_indices[i] < b_indices[j] {
503            // a's index is smaller, advance a
504            i += 1;
505        } else {
506            // b's index is smaller, advance b
507            j += 1;
508        }
509    }
510
511    Ok(sum)
512}

514/// Computes cosine similarity between two sparse vectors.
515///
516/// Cosine similarity measures the angle between vectors, ranging from -1 (opposite)
517/// to 1 (identical direction). For sparse vectors, this is computed efficiently
518/// by only considering non-zero elements.
519///
520/// # Arguments
521///
522/// * `a` - First sparse vector
523/// * `b` - Second sparse vector
524///
525/// # Returns
526///
527/// * `Ok(f32)` - Cosine similarity in range [-1, 1]
528/// * `Err` - If either vector has zero magnitude
529///
530/// # Formula
531///
532/// ```text
533/// cosine_similarity = dot(a, b) / (||a|| * ||b||)
534/// ```
535///
536/// # Example
537///
538/// ```rust
539/// use aletheiadb::core::vector::{SparseVec, sparse_cosine_similarity};
540///
541/// let a = SparseVec::new(vec![0, 2], vec![1.0, 1.0], 5).unwrap();
542/// let b = SparseVec::new(vec![0, 2], vec![1.0, 1.0], 5).unwrap();
543///
544/// // Identical vectors have cosine similarity = 1.0
545/// let sim = sparse_cosine_similarity(&a, &b).unwrap();
546/// assert!((sim - 1.0).abs() < 1e-6);
547/// ```
548///
549/// # Performance
550///
551/// - Time: O(nnz_a + nnz_b) for dot product + O(nnz_a + nnz_b) for magnitudes
552/// - Space: O(1)
553/// - Much faster than dense cosine for sparse data
554pub fn sparse_cosine_similarity(a: &SparseVec, b: &SparseVec) -> Result<f32> {
555    let dot = sparse_dot_product(a, b)?;
556    let sq_mag_a = a.squared_magnitude();
557    let sq_mag_b = b.squared_magnitude();
558
559    // Handle zero magnitude vectors
560    if sq_mag_a < SQUARED_MAGNITUDE_THRESHOLD || sq_mag_b < SQUARED_MAGNITUDE_THRESHOLD {
561        return Ok(0.0);
562    }
563
564    let similarity = dot / (sq_mag_a.sqrt() * sq_mag_b.sqrt());
565
566    // Clamp to [-1, 1] to handle floating-point errors
567    Ok(similarity.clamp(-1.0, 1.0))
568}

570/// Computes squared Euclidean distance between two sparse vectors.
571///
572/// The squared distance is more efficient than computing the full Euclidean
573/// distance since it avoids the square root operation. For comparing distances,
574/// squared distance preserves ordering.
575///
576/// # Arguments
577///
578/// * `a` - First sparse vector
579/// * `b` - Second sparse vector
580///
581/// # Returns
582///
583/// * `Ok(f32)` - Squared Euclidean distance
584/// * `Err(VectorError::DimensionMismatch)` - If vectors have different dimensions
585///
586/// # Formula
587///
588/// ```text
589/// squared_distance = ||a - b||² = Σ(a_i - b_i)²
590/// ```
591///
592/// For sparse vectors, we only need to compute:
593/// - ||a||² + ||b||² - 2*dot(a,b)
594///
595/// # Example
596///
597/// ```rust
598/// use aletheiadb::core::vector::{SparseVec, sparse_squared_euclidean_distance};
599///
600/// let a = SparseVec::new(vec![0], vec![3.0], 5).unwrap();
601/// let b = SparseVec::new(vec![], vec![], 5).unwrap(); // Zero vector
602///
603/// // Distance from [3,0,0,0,0] to [0,0,0,0,0] = 9
604/// let dist_sq = sparse_squared_euclidean_distance(&a, &b).unwrap();
605/// assert_eq!(dist_sq, 9.0);
606/// ```
607///
608/// # Performance
609///
610/// - Time: O(nnz_a + nnz_b)
611/// - Space: O(1)
612pub fn sparse_squared_euclidean_distance(a: &SparseVec, b: &SparseVec) -> Result<f32> {
613    // Check dimensions match
614    if a.dimension() != b.dimension() {
615        return Err(VectorError::DimensionMismatch {
616            expected: a.dimension(),
617            actual: b.dimension(),
618        }
619        .into());
620    }
621
622    // Stable algorithm: sum((a_i - b_i)^2)
623    // Avoids catastrophic cancellation from ||a||^2 + ||b||^2 - 2<a,b>
624    // when a and b are very close.
625    // Also faster (O(N) single pass vs O(N) triple pass).
626
627    let mut sum_sq_diff = 0.0f32;
628    let mut i = 0;
629    let mut j = 0;
630
631    let a_indices = a.indices();
632    let a_values = a.values();
633    let b_indices = b.indices();
634    let b_values = b.values();
635
636    while i < a_indices.len() && j < b_indices.len() {
637        if a_indices[i] == b_indices[j] {
638            let diff = a_values[i] - b_values[j];
639            sum_sq_diff += diff * diff;
640            i += 1;
641            j += 1;
642        } else if a_indices[i] < b_indices[j] {
643            sum_sq_diff += a_values[i] * a_values[i];
644            i += 1;
645        } else {
646            sum_sq_diff += b_values[j] * b_values[j];
647            j += 1;
648        }
649    }
650
651    // Process remaining elements
652    while i < a_indices.len() {
653        sum_sq_diff += a_values[i] * a_values[i];
654        i += 1;
655    }
656
657    while j < b_indices.len() {
658        sum_sq_diff += b_values[j] * b_values[j];
659        j += 1;
660    }
661
662    Ok(sum_sq_diff)
663}

665/// Computes Euclidean distance between two sparse vectors.
666///
667/// This is the L2 distance - the straight-line distance between two points.
668///
669/// # Arguments
670///
671/// * `a` - First sparse vector
672/// * `b` - Second sparse vector
673///
674/// # Returns
675///
676/// * `Ok(f32)` - Euclidean distance
677/// * `Err(VectorError::DimensionMismatch)` - If vectors have different dimensions
678///
679/// # Example
680///
681/// ```rust
682/// use aletheiadb::core::vector::{SparseVec, sparse_euclidean_distance};
683///
684/// let a = SparseVec::new(vec![0], vec![3.0], 5).unwrap();
685/// let b = SparseVec::new(vec![], vec![], 5).unwrap(); // Zero vector
686///
687/// // Distance from [3,0,0,0,0] to [0,0,0,0,0] = 3.0
688/// let dist = sparse_euclidean_distance(&a, &b).unwrap();
689/// assert_eq!(dist, 3.0);
690/// ```
691#[inline]
692pub fn sparse_euclidean_distance(a: &SparseVec, b: &SparseVec) -> Result<f32> {
693    sparse_squared_euclidean_distance(a, b).map(|sq| sq.sqrt())
694}
