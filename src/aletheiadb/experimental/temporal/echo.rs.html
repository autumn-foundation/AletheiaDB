1//! Temporal Resonance Engine ("Echo").
2//!
3//! # The Spark
4//! The "Echo" engine generates a `TemporalFingerprint` for a node based on its
5//! mutation history (when it was updated) and compares these fingerprints
6//! to find "resonant" nodes.
7//!
8//! # The Details
9//! This module implements "Temporal Resonance", a way to find nodes that share
10//! similar historical activity patterns.
11//!
12//! # The Hook
13//! "Find me every sensor that failed with the same stuttering pattern as Sensor X."

21/// A normalized representation of a node's temporal activity.
22///
23/// # The Details
24/// Temporal fingerprints are used to compute the similarity of update histories between nodes.
25///
26/// # Examples
27///
28/// ```
29/// # #[cfg(feature = "semantic-temporal")]
30/// # fn main() {
31/// use aletheiadb::experimental::echo::TemporalFingerprint;
32///
33/// let fp1 = TemporalFingerprint { bins: vec![1.0, 0.0], resolution_us: 1000 };
34/// let fp2 = TemporalFingerprint { bins: vec![0.0, 1.0], resolution_us: 1000 };
35/// let fp3 = TemporalFingerprint { bins: vec![1.0, 0.0], resolution_us: 1000 };
36///
37/// // fp1 and fp2 are orthogonal
38/// assert!(fp1.similarity(&fp2) < 0.01);
39///
40/// // fp1 and fp3 are identical
41/// assert!(fp1.similarity(&fp3) > 0.99);
42/// # }
43/// # #[cfg(not(feature = "semantic-temporal"))]
44/// # fn main() {}
45/// ```
46#[derive(Debug, Clone)]
47pub struct TemporalFingerprint {
48    /// Normalized activity bins (sum of squares = 1.0 for cosine similarity).
49    pub bins: Vec<f32>,
50    /// The resolution of each bin in microseconds.
51    pub resolution_us: i64,
52}
53
54impl TemporalFingerprint {
55    /// Compute the Cosine Similarity between this fingerprint and another.
56    /// Returns a score between 0.0 (orthogonal) and 1.0 (identical).
57    ///
58    /// # Panics
59    /// Panics if fingerprints have different lengths.
60    pub fn similarity(&self, other: &Self) -> f32 {
61        if self.bins.len() != other.bins.len() {
62            // In a real system we might handle this gracefully, but for Nova we expect
63            // the Resonator to produce consistent fingerprints.
64            panic!(
65                "Fingerprint length mismatch: {} vs {}",
66                self.bins.len(),
67                other.bins.len()
68            );
69        }
70
71        let dot_product: f32 = self
72            .bins
73            .iter()
74            .zip(other.bins.iter())
75            .map(|(a, b)| a * b)
76            .sum();
77
78        // Since bins are pre-normalized (unit vectors), cosine similarity is just the dot product.
79        // We clamp to [0.0, 1.0] to handle floating point errors.
80        dot_product.clamp(0.0, 1.0)
81    }
82}

84/// Trait for generating temporal fingerprints from entity history.
85pub trait Resonator {
86    /// Generate a fingerprint from the given history.
87    fn resonate(&self, history: &EntityHistory) -> TemporalFingerprint;
88}

90/// A resonator that measures activity density over a fixed time window.
91///
92/// # The Details
93/// It looks at the "Last N" microseconds of history and bins updates into
94/// a histogram.
95///
96/// # Examples
97///
98/// ```
99/// # #[cfg(feature = "semantic-temporal")]
100/// # fn main() {
101/// use aletheiadb::experimental::echo::{ActivityDensityResonator, Resonator};
102///
103/// let resonator = ActivityDensityResonator {
104///     window_size_us: 3600 * 1_000_000, // 1 hour
105///     num_bins: 60,                     // 1 minute resolution
106/// };
107/// # }
108/// # #[cfg(not(feature = "semantic-temporal"))]
109/// # fn main() {}
110/// ```
111pub struct ActivityDensityResonator {
112    /// Total time window to consider (in microseconds).
113    pub window_size_us: i64,
114    /// Number of bins to divide the window into.
115    pub num_bins: usize,
116}
117
118impl Default for ActivityDensityResonator {
119    fn default() -> Self {
120        Self {
121            window_size_us: 3600 * 1_000_000, // 1 hour
122            num_bins: 60,                     // 1 minute resolution
123        }
124    }
125}
126
127impl Resonator for ActivityDensityResonator {
128    fn resonate(&self, history: &EntityHistory) -> TemporalFingerprint {
129        let mut bins = vec![0.0; self.num_bins];
130        let bin_size_us = self.window_size_us / self.num_bins as i64;
131
132        // We anchor the window to "now" (or the latest possible time).
133        // Actually, to be deterministic for historical data, let's anchor to the *latest version*
134        // in the history, OR use `time::now()`.
135        // Using `time::now()` makes it sensitive to *when* you run the query (decaying echo).
136        // Using `latest version` makes it relative to the entity's own timeline.
137        //
138        // "Echo" implies matching patterns in absolute time usually (e.g. "did everything crash at 2pm?").
139        // So let's anchor to `time::now()`.
140        let now = time::now().wallclock();
141        let start_time = now - self.window_size_us;
142
143        for version in &history.versions {
144            let ts = version.temporal.valid_time().start().wallclock();
145
146            if ts >= start_time && ts <= now {
147                let offset = ts - start_time;
148                let bin_idx = (offset / bin_size_us) as usize;
149                if bin_idx < self.num_bins {
150                    bins[bin_idx] += 1.0;
151                }
152            }
153        }
154
155        // Normalize (L2 Norm)
156        let magnitude: f32 = bins.iter().map(|x| x * x).sum::<f32>().sqrt();
157        if magnitude > 0.0 {
158            for x in &mut bins {
159                *x /= magnitude;
160            }
161        }
162
163        TemporalFingerprint {
164            bins,
165            resolution_us: bin_size_us,
166        }
167    }
168}

238/// A simulator for testing semantic drift and reinforcement loops.
239///
240/// # Why?
241/// In recommender systems, repeatedly suggesting content similar to a user's
242/// previous interactions creates an echo chamber. This struct simulates that
243/// process over time to measure the degradation of content diversity.
244pub struct EchoChamber<'a> {
245    _marker: std::marker::PhantomData<&'a AletheiaDB>,
246}
247
248#[cfg(feature = "semantic-temporal")]
249impl<'a> EchoChamber<'a> {
250    /// Create a new EchoChamber with default settings.
251    pub fn new(db: &'a AletheiaDB) -> Self {
252        Self {
253            db,
254            resonator: Box::new(ActivityDensityResonator::default()),
255        }
256    }
257
258    /// Configure the EchoChamber with a custom resonator.
259    pub fn with_resonator<R: Resonator + 'static>(mut self, resonator: R) -> Self {
260        self.resonator = Box::new(resonator);
261        self
262    }
263
264    /// Find nodes that resonate with the target node.
265    pub fn find_echoes(&self, target: NodeId, candidates: &[NodeId]) -> Result<Vec<(NodeId, f32)>> {
266        // 1. Get Target History & Fingerprint
267        let target_history = self.db.get_node_history(target)?;
268        let target_fp = self.resonator.resonate(&target_history);
269
270        let mut results = Vec::with_capacity(candidates.len());
271
272        // 2. Scan Candidates
273        // Note: In a real implementation, we would want an index for this.
274        // For Nova, a linear scan over provided candidates is acceptable.
275        for &candidate_id in candidates {
276            if candidate_id == target {
277                continue;
278            }
279
280            // We accept that history might be missing for some candidates
281            if let Ok(history) = self.db.get_node_history(candidate_id) {
282                let fp = self.resonator.resonate(&history);
283                let score = target_fp.similarity(&fp);
284                if score > 0.0 {
285                    results.push((candidate_id, score));
286                }
287            }
288        }
289
290        // 3. Sort by Score (Descending)
291        // unwrap_or is safe because we clamped scores to 0.0-1.0 (no NaNs ideally, but safety first)
292        results.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
293
294        Ok(results)
295    }
296}

349    #[test]
350    fn test_temporal_fingerprint_similarity() {
351        let fp1 = TemporalFingerprint {
352            bins: vec![1.0, 0.0], // Unit vector
353            resolution_us: 1000,
354        };
355        let fp2 = TemporalFingerprint {
356            bins: vec![0.0, 1.0], // Orthogonal
357            resolution_us: 1000,
358        };
359        let fp3 = TemporalFingerprint {
360            bins: vec![
361                std::f32::consts::FRAC_1_SQRT_2,
362                std::f32::consts::FRAC_1_SQRT_2,
363            ], // ~45 degrees
364            resolution_us: 1000,
365        };
366
367        assert!(fp1.similarity(&fp1) > 0.99); // Identity
368        assert!(fp1.similarity(&fp2) < 0.01); // Orthogonal
369        assert!(fp1.similarity(&fp3) > 0.6 && fp1.similarity(&fp3) < 0.8); // 45 deg
370    }

372    #[test]
373    fn test_echo_chamber_integration() {
374        let db = AletheiaDB::new().unwrap();
375
376        // Define a "Now" for our test universe
377        let now_wallclock = time::now().wallclock();
378        // Window: Last 100 seconds
379        let window = 100 * 1_000_000;
380
381        // Create 3 nodes with specific histories
382
383        // Node A (Target): Updates at T-80s, T-50s
384        // Node B (Resonant): Updates at T-80s, T-50s (Identical pattern)
385        // Node C (Noise): Updates at T-20s (Different pattern)
386
387        let t_minus_80 = HybridTimestamp::new(now_wallclock - 80 * 1_000_000, 0).unwrap();
388        let t_minus_50 = HybridTimestamp::new(now_wallclock - 50 * 1_000_000, 0).unwrap();
389        let t_minus_20 = HybridTimestamp::new(now_wallclock - 20 * 1_000_000, 0).unwrap();
390
391        let props = PropertyMapBuilder::new().insert("val", 0).build();
392
393        // Helper to populate history
394        let create_node_with_history = |timestamps: Vec<Timestamp>| -> NodeId {
395            assert!(
396                !timestamps.is_empty(),
397                "history requires at least one timestamp"
398            );
399
400            // Create and commit the base version first.
401            let id = {
402                let mut tx = db.write_transaction().unwrap();
403                let id = tx
404                    .create_node_with_valid_time("Node", props.clone(), Some(timestamps[0]))
405                    .unwrap();
406                tx.commit().unwrap();
407                id
408            };
409
410            // Apply subsequent versions as committed updates.
411            for &ts in &timestamps[1..] {
412                let mut tx = db.write_transaction().unwrap();
413                tx.update_node_with_valid_time(id, props.clone(), Some(ts))
414                    .unwrap();
415                tx.commit().unwrap();
416            }
417
418            id
419        };
420
421        let node_a = create_node_with_history(vec![t_minus_80, t_minus_50]);
422        let node_b = create_node_with_history(vec![t_minus_80, t_minus_50]);
423        let node_c = create_node_with_history(vec![t_minus_20]);
424
425        // Configure Resonator: 100s window, 10 bins (10s per bin)
426        let resonator = ActivityDensityResonator {
427            window_size_us: window,
428            num_bins: 10,
429        };
430
431        let chamber = EchoChamber::new(&db).with_resonator(resonator);
432
433        // Find echoes for Node A among B and C
434        let echoes = chamber.find_echoes(node_a, &[node_b, node_c]).unwrap();
435
436        // Expectation:
437        // Node B should be ~1.0 (Identical)
438        // Node C should be ~0.0 (Different bins)
439
440        assert_eq!(echoes.len(), 1, "Should only return non-zero similarity"); // C might be 0.0
441
442        let (id, score) = echoes[0];
443        assert_eq!(id, node_b);
444        assert!(
445            score > 0.9,
446            "Node B should resonate strongly (score: {})",
447            score
448        );
449
450        // Verify C is not in results or very low
451        let c_score = echoes.iter().find(|(id, _)| *id == node_c);
452        if let Some((_, score)) = c_score {
453            assert!(
454                *score < 0.1,
455                "Node C should not resonate (score: {})",
456                score
457            );
458        }
459    }
