1//! Omen: Semantic Collision Detection.
2//!
3//! "I have a bad feeling about this..."
4//!
5//! Omen predicts the future interaction of two concepts by analyzing their
6//! semantic trajectories. It answers questions like:
7//! - "Will 'Artificial Intelligence' and 'Ethics' collide soon?"
8//! - "Is 'Project X' moving away from its 'Original Goal'?"
9//!
10//! # Concepts
11//! - **Trajectory**: The path a node is taking through semantic space, modeled as a velocity vector.
12//! - **Encounter**: The point in time (future or past) where two trajectories are closest.
13//!
14//! # Logic
15//! Given two nodes A and B with positions $P_a, P_b$ and velocities $V_a, V_b$:
16//! 1. Relative Position: $P = P_b - P_a$
17//! 2. Relative Velocity: $V = V_b - V_a$
18//! 3. Distance Squared at time $t$: $D^2(t) = ||P + V*t||^2$
19//! 4. Minimized when derivative is 0: $t = -(P \cdot V) / ||V||^2$
20//!
21//! # Examples
22//!
23//! ```rust
24//! # #[cfg(feature = "semantic-reasoning")]
25//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
26//! use aletheiadb::AletheiaDB;
27//! use aletheiadb::experimental::omen::Omen;
28//! use aletheiadb::core::temporal::{TimeRange, time};
29//!
30//! let db = AletheiaDB::new()?;
31//! # let node_a = db.create_node("A", Default::default())?;
32//! # let node_b = db.create_node("B", Default::default())?;
33//! let omen = Omen::new(&db);
34//!
35//! // Analyze last hour
36//! let start = time::from_secs(time::now().wallclock() / 1_000_000 - 3600);
37//! let window = TimeRange::new(start, time::now())?;
38//!
39//! if let Some(encounter) = omen.predict_encounter(node_a, node_b, window, "embedding")? {
40//!     println!("Predicted encounter in {:.2} seconds.", encounter.time_to_encounter.as_secs_f32());
41//!     println!("Closest distance: {:.4}", encounter.predicted_distance);
42//! }
43//! # Ok(())
44//! # }
45//! # #[cfg(not(feature = "semantic-reasoning"))]
46//! # fn main() {}
47//! ```

56/// The predicted outcome of two trajectories.
57///
58/// # Examples
59///
60/// ```rust
61/// use aletheiadb::experimental::omen::Encounter;
62/// use std::time::Duration;
63///
64/// let encounter = Encounter {
65///     time_to_encounter: Duration::from_secs(30),
66///     is_past: false, // It will happen in the future
67///     predicted_distance: 0.1,
68/// };
69///
70/// if !encounter.is_past {
71///     println!("Collision incoming!");
72/// }
73/// ```
74#[derive(Debug, Clone)]
75pub struct Encounter {
76    /// Time until the encounter (from window end).
77    /// Negative means the encounter happened in the past (diverging).
78    pub time_to_encounter: Duration,
79    /// Is the encounter in the past? (Helper boolean since Duration is unsigned in std, but we model offset conceptually).
80    /// Note: std::time::Duration is unsigned. We store magnitude in `time_to_encounter` and sign here.
81    pub is_past: bool,
82    /// The estimated semantic distance at the point of closest approach.
83    pub predicted_distance: f32,
84}

97    /// Predict the encounter between two nodes based on their trajectories in the given window.
98    pub fn predict_encounter(
99        &self,
100        node_a: NodeId,
101        node_b: NodeId,
102        window: TimeRange,
103        vector_property: &str,
104    ) -> Result<Option<Encounter>> {
105        // Fix transaction time to "now" for consistent view
106        let tx = self.db.read_transaction()?;
107        let tx_time = tx.metadata().start_timestamp;
108
109        // 1. Calculate Trajectories
110        // We need Position (at window end) and Velocity (change over window)
111        let traj_a = self.calculate_trajectory(node_a, window, vector_property, tx_time)?;
112        let traj_b = self.calculate_trajectory(node_b, window, vector_property, tx_time)?;
113
114        // If either has no vector, we can't predict
115        let (pos_a, vel_a) = match traj_a {
116            Some(t) => t,
117            None => return Ok(None),
118        };
119        let (pos_b, vel_b) = match traj_b {
120            Some(t) => t,
121            None => return Ok(None),
122        };
123
124        // Keep the read transaction alive for the duration of the historical scan
125        let _ = tx;
126
127        // 2. Physics Math
128        // Relative Position P = Pb - Pa (at t=0, which is window.end)
129        let rel_pos: Vec<f32> = pos_b.iter().zip(pos_a.iter()).map(|(b, a)| b - a).collect();
130
131        // Relative Velocity V = Vb - Va
132        let rel_vel: Vec<f32> = vel_b.iter().zip(vel_a.iter()).map(|(b, a)| b - a).collect();
133
134        // Calculate dot products
135        let p_dot_v: f32 = rel_pos.iter().zip(rel_vel.iter()).map(|(p, v)| p * v).sum();
136
137        let v_dot_v: f32 = rel_vel.iter().map(|v| v * v).sum();
138
139        // If relative velocity is effectively zero, paths are parallel.
140        // Closest distance is current distance. t = 0.
141        if v_dot_v < 1e-9 {
142            let current_dist = rel_pos.iter().map(|x| x * x).sum::<f32>().sqrt();
143            return Ok(Some(Encounter {
144                time_to_encounter: Duration::from_secs(0),
145                is_past: false,
146                predicted_distance: current_dist,
147            }));
148        }
149
150        // t = -(P . V) / ||V||^2
151        let t_secs = -p_dot_v / v_dot_v;
152
153        // Calculate closest distance at t
154        // Pos(t) = P + V*t
155        let mut pos_at_t = Vec::with_capacity(rel_pos.len());
156        for (p, v) in rel_pos.iter().zip(rel_vel.iter()) {
157            pos_at_t.push(p + v * t_secs);
158        }
159        let min_dist = pos_at_t.iter().map(|x| x * x).sum::<f32>().sqrt();
160
161        let is_past = t_secs < 0.0;
162        let abs_secs = t_secs.abs();
163        let time_to_encounter = Duration::from_secs_f32(abs_secs);
164
165        Ok(Some(Encounter {
166            time_to_encounter,
167            is_past,
168            predicted_distance: min_dist,
169        }))
170    }

172    /// Calculate (Position, Velocity) for a node.
173    /// Position is vector at window.end.
174    /// Velocity is (Vector_End - Vector_Start) / Duration.
175    fn calculate_trajectory(
176        &self,
177        node_id: NodeId,
178        window: TimeRange,
179        property: &str,
180        tx_time: crate::core::temporal::Timestamp,
181    ) -> Result<Option<(Vec<f32>, Vec<f32>)>> {
182        // Fetch full history directly to bypass potential temporal index issues in tests.
183        // In real usage, `get_node_at_time` is preferred, but for unit tests with simulated time,
184        // history scan is more robust.
185        //
186        // Note: We use `window.start()` and `window.end()` to find the bracketing versions.
187
188        let history = self.db.get_node_history(node_id)?;
189
190        // Find vector at or immediately before window.start
191        let start_vec = self.find_vector_at(&history, window.start(), property, tx_time);
192        // Find vector at or immediately before window.end
193        let end_vec = self.find_vector_at(&history, window.end(), property, tx_time);
194
195        match (start_vec, end_vec) {
196            (Some(start), Some(end)) => {
197                if start.len() != end.len() {
198                    return Ok(None); // Dim mismatch
199                }
200
201                let duration_micros = window.duration_micros().unwrap_or(0);
202                if duration_micros == 0 {
203                    // Instantaneous window, velocity is zero? Or undefined?
204                    // Treat as zero velocity.
205                    let zero_vel = vec![0.0; start.len()];
206                    return Ok(Some((end, zero_vel)));
207                }
208
209                let duration_secs = duration_micros as f32 / 1_000_000.0;
210
211                let velocity: Vec<f32> = end
212                    .iter()
213                    .zip(start.iter())
214                    .map(|(e, s)| (e - s) / duration_secs)
215                    .collect();
216
217                Ok(Some((end, velocity)))
218            }
219            _ => Ok(None), // Missing data
220        }
221    }
222
223    fn find_vector_at(
224        &self,
225        history: &crate::core::history::EntityHistory,
226        time: crate::core::temporal::Timestamp,
227        property: &str,
228        tx_time: crate::core::temporal::Timestamp,
229    ) -> Option<Vec<f32>> {
230        // Iterate history to find the latest version valid at `time`.
231        // History is typically sorted by version ID. We want the latest version where valid_time.start <= time.
232        // Assuming versions are sorted by valid_time or we scan all.
233        let mut best_vec = None;
234        let mut best_time = i64::MIN;
235
236        for v in &history.versions {
237            let vt_start = v.temporal.valid_time().start().wallclock();
238            let tt_start = v.temporal.transaction_time().start().wallclock();
239            if tt_start <= tx_time.wallclock()
240                && vt_start <= time.wallclock()
241                && vt_start >= best_time
242                && let Some(val) = v.properties.get(property)
243                && let Some(vec) = val.as_vector()
244            {
245                best_vec = Some(vec.to_vec());
246                best_time = vt_start;
247            }
248        }
249        best_vec
250    }

261    #[test]
262    fn test_omen_head_on_collision() {
263        let db = AletheiaDB::new().unwrap();
264        // Enable index
265        db.enable_vector_index("vec", HnswConfig::new(2, DistanceMetric::Euclidean))
266            .unwrap();
267
268        // Node A: Starts at [0, 0], moving Right (+X)
269        let props_a = PropertyMapBuilder::new()
270            .insert_vector("vec", &[0.0, 0.0])
271            .build();
272        let a = db.create_node("A", props_a).unwrap();
273
274        // Node B: Starts at [10, 0], moving Left (-X)
275        let props_b = PropertyMapBuilder::new()
276            .insert_vector("vec", &[10.0, 0.0])
277            .build();
278        let b = db.create_node("B", props_b).unwrap();
279
280        // Wait to establish initial state time
281        std::thread::sleep(Duration::from_millis(10));
282        let t_start = time::now();
283
284        // Wait 100ms
285        std::thread::sleep(Duration::from_millis(100));
286
287        // Update A to [1, 0] (Vel = 1 unit/100ms = 10 units/s)
288        db.write(|tx| {
289            tx.update_node(
290                a,
291                PropertyMapBuilder::new()
292                    .insert_vector("vec", &[1.0, 0.0])
293                    .build(),
294            )
295        })
296        .unwrap();
297
298        // Update B to [9, 0] (Vel = -1 unit/100ms = -10 units/s)
299        db.write(|tx| {
300            tx.update_node(
301                b,
302                PropertyMapBuilder::new()
303                    .insert_vector("vec", &[9.0, 0.0])
304                    .build(),
305            )
306        })
307        .unwrap();
308
309        let t_end = time::now();
310
311        let omen = Omen::new(&db);
312        let window = TimeRange::new(t_start, t_end).unwrap();
313
314        // At T_end:
315        // Pos A = [1, 0]
316        // Pos B = [9, 0]
317        // Vel A = +X
318        // Vel B = -X
319        // Rel Pos (B-A) = [8, 0]
320        // Rel Vel (B-A) = (-X) - (+X) = -2X
321        // They are closing distance.
322        // Distance is 8. Closing speed is 2X (relative).
323        // Time to impact = 8 / relative_speed.
324        // Wait, velocity magnitude depends on exact duration.
325        // Let Omen calculate it.
326
327        let encounter = omen
328            .predict_encounter(a, b, window, "vec")
329            .unwrap()
330            .unwrap();
331
332        println!("Encounter: {:?}", encounter);
333
334        assert!(!encounter.is_past, "Collision should be in future");
335        assert!(
336            encounter.predicted_distance < 0.01,
337            "Should be a direct hit"
338        );
339        // Time should be positive
340        assert!(encounter.time_to_encounter.as_secs_f32() > 0.0);
341    }

344    fn test_omen_diverging() {
345        let db = AletheiaDB::new().unwrap();
346        db.enable_vector_index("vec", HnswConfig::new(2, DistanceMetric::Euclidean))
347            .unwrap();
348
349        // Start together at [0,0]
350        let pa = PropertyMapBuilder::new()
351            .insert_vector("vec", &[0.0, 0.0])
352            .build();
353        let a = db.create_node("A", pa.clone()).unwrap();
354        let b = db.create_node("B", pa.clone()).unwrap();
355
356        std::thread::sleep(Duration::from_millis(10));
357        let t_start = time::now();
358        std::thread::sleep(Duration::from_millis(50));
359
360        // Move apart
361        // A -> [-1, 0]
362        db.write(|tx| {
363            tx.update_node(
364                a,
365                PropertyMapBuilder::new()
366                    .insert_vector("vec", &[-1.0, 0.0])
367                    .build(),
368            )
369        })
370        .unwrap();
371
372        // B -> [1, 0]
373        db.write(|tx| {
374            tx.update_node(
375                b,
376                PropertyMapBuilder::new()
377                    .insert_vector("vec", &[1.0, 0.0])
378                    .build(),
379            )
380        })
381        .unwrap();
382
383        let t_end = time::now();
384        let omen = Omen::new(&db);
385        let window = TimeRange::new(t_start, t_end).unwrap();
386
387        // At T_end, they are at [-1,0] and [1,0]. Moving apart.
388        // Closest approach was in the past (at start).
389
390        let encounter = omen
391            .predict_encounter(a, b, window, "vec")
392            .unwrap()
393            .unwrap();
394
395        assert!(encounter.is_past, "Encounter should be in the past");
396        // Closest distance should be ~0 (when they were together)
397        assert!(encounter.predicted_distance < 0.1);
398    }

476            tx.update_node(
477                a,
478                PropertyMapBuilder::new()
479                    .insert_vector("vec", &[1.0, 0.0])

481            )
482        })

486            tx.update_node(
487                b,
