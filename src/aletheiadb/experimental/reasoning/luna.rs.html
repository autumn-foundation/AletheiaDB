1//! Luna: Semantic Subgraph Synthesis.
2//!
3//! "What holds these ideas together?"
4//!
5//! Luna explores a set of connected or related concepts (seeds), computes their
6//! semantic "center of gravity" (centroid), and dynamically instantiates a new
7//! node representing that core concept. It then automatically connects this new
8//! core node back to the seeds, thereby structurally materializing a previously
9//! implicit relationship.
10//!
11//! # Concepts
12//! - **Synthesis**: The creation of a new "Core" node from the semantic average of seeds.
13//! - **Materialization**: Automatically adding structural edges (`CORE_OF`) from the
14//!   new node to the original seed nodes.
15
16#![allow(clippy::collapsible_if)]
17
18use crate::AletheiaDB;
19use crate::api::transaction::WriteOps;
20use crate::core::error::{Error, Result, VectorError};
21use crate::core::id::NodeId;
22use crate::core::property::PropertyMapBuilder;
23
24/// The Luna Engine for Semantic Synthesis.
25pub struct Luna<'a> {
26    db: &'a AletheiaDB,
27}
28
29#[cfg(feature = "semantic-reasoning")]
30impl<'a> Luna<'a> {
31    /// Create a new Luna instance.
32    pub fn new(db: &'a AletheiaDB) -> Self {
33        Self { db }
34    }
35
36    /// Synthesize a core concept from a set of seed nodes.
37    ///
38    /// # Arguments
39    /// * `seeds` - The source nodes to synthesize.
40    /// * `property` - The vector property to use for calculation and assignment.
41    /// * `label` - The label for the newly created core node.
42    ///
43    /// # Returns
44    /// The `NodeId` of the newly synthesized core node.
45    pub fn synthesize_core(&self, seeds: &[NodeId], property: &str, label: &str) -> Result<NodeId> {
46        if seeds.is_empty() {
47            return Err(Error::other(
48                "Cannot synthesize from an empty set of seeds.",
49            ));
50        }
51
52        // 1. Fetch Vectors from seeds
53        let mut vectors = Vec::with_capacity(seeds.len());
54        for &node_id in seeds {
55            let node = self.db.get_node(node_id)?;
56            if let Some(val) = node.properties.get(property) {
57                if let Some(vec) = val.as_vector() {
58                    vectors.push(vec.to_vec());
59                }
60            }
61        }
62
63        if vectors.is_empty() {
64            return Err(Error::Vector(VectorError::IndexError(format!(
65                "No vectors found in property '{}' for the given seeds.",
66                property
67            ))));
68        }
69
70        // 2. Calculate Centroid
71        let dim = vectors[0].len();
72        let mut centroid = vec![0.0; dim];
73        for vec in &vectors {
74            if vec.len() != dim {
75                return Err(Error::Vector(VectorError::DimensionMismatch {
76                    expected: dim,
77                    actual: vec.len(),
78                }));
79            }
80            for (i, val) in vec.iter().enumerate() {
81                centroid[i] += val;
82            }
83        }
84
85        // Normalize the centroid to unit length for directional semantics (Cosine)
86        let centroid = crate::core::vector::ops::normalize(&centroid);
87
88        // 3. Materialize: Create new node and edges
89        let new_node_id = self.db.write(|tx| {
90            // Build properties for the new core node
91            let props = PropertyMapBuilder::new()
92                .insert_vector(property, &centroid)
93                .build();
94
95            // Create the Core Node
96            let core_node_id = tx.create_node(label, props)?;
97
98            // Create edges from the Core Node to the Seeds
99            let edge_props = PropertyMapBuilder::new().build();
100            for &seed_id in seeds {
101                // Ensure we don't try to link to a seed that failed to provide a vector?
102                // Actually, linking to all requested seeds makes structural sense even if they lacked the vector.
103                tx.create_edge(core_node_id, seed_id, "CORE_OF", edge_props.clone())?;
104            }
105
106            Ok::<NodeId, Error>(core_node_id)
107        })?;
108
109        Ok(new_node_id)
110    }
111}

113#[cfg(all(test, feature = "semantic-reasoning"))]
114mod tests {
115    use super::*;
116    use crate::index::vector::{DistanceMetric, HnswConfig};
117
118    #[test]
119    fn test_luna_synthesize_core() {
120        let db = AletheiaDB::new().unwrap();
121        let config = HnswConfig::new(2, DistanceMetric::Cosine);
122        db.enable_vector_index("vec", config).unwrap();
123
124        // Node A: [1, 0]
125        let props_a = PropertyMapBuilder::new()
126            .insert_vector("vec", &[1.0, 0.0])
127            .build();
128        let a = db.create_node("Concept", props_a).unwrap();
129
130        // Node B: [0, 1]
131        let props_b = PropertyMapBuilder::new()
132            .insert_vector("vec", &[0.0, 1.0])
133            .build();
134        let b = db.create_node("Concept", props_b).unwrap();
135
136        let luna = Luna::new(&db);
137        let core_id = luna.synthesize_core(&[a, b], "vec", "CoreConcept").unwrap();
138
139        // Check that the new node was created and has the correct label
140        let core_node = db.get_node(core_id).unwrap();
141        let label_str = crate::core::GLOBAL_INTERNER
142            .resolve_with(core_node.label, |s| s.to_string())
143            .unwrap();
144        assert_eq!(label_str, "CoreConcept");
145
146        // Check the synthesized vector (Normalized [0.5, 0.5] -> [0.707, 0.707])
147        let vec_val = core_node.properties.get("vec").unwrap();
148        let vec = vec_val.as_vector().unwrap();
149        assert!((vec[0] - std::f32::consts::FRAC_1_SQRT_2).abs() < 0.01);
150        assert!((vec[1] - std::f32::consts::FRAC_1_SQRT_2).abs() < 0.01);
151
152        // Verify the structural materialization: Edges from Core -> Seeds
153        let out_edges = db.get_outgoing_edges(core_id);
154        assert_eq!(out_edges.len(), 2, "Core node should have 2 outgoing edges");
155
156        let mut targets = std::collections::HashSet::new();
157        for edge_id in out_edges {
158            let edge = db.get_edge(edge_id).unwrap();
159            let edge_label_str = crate::core::GLOBAL_INTERNER
160                .resolve_with(edge.label, |s| s.to_string())
161                .unwrap();
162            assert_eq!(edge_label_str, "CORE_OF");
163            targets.insert(edge.target);
164        }
165
166        assert!(targets.contains(&a));
167        assert!(targets.contains(&b));
168    }
169}
