1//! Oracle: Probabilistic Graph Reasoning.
2//!
3//! "Stop guessing. Ask the Oracle."
4//!
5//! The Oracle uses Monte Carlo simulations to answer probabilistic questions about the graph.
6//! - "Who is the most relevant node to X?" (Personalized PageRank)
7//! - "What is the probability of reaching Y from X?" (Reachability)
8//!
9//! # Use Cases
10//! - **Influence Analysis**: Identify key influencers in a local context.
11//! - **Risk Assessment**: "If this server fails, what is the probability it cascades to the database?"
12//! - **Recommendation**: "Users who viewed X are 80% likely to view Y."
13
14use crate::AletheiaDB;
15use crate::core::error::{Result, StorageError};
16use crate::core::hasher::IdentityHasher;

18use rand::prelude::*;
19use std::collections::HashMap;

22/// The Oracle Engine.
23pub struct Oracle<'a> {
24    db: &'a AletheiaDB,
25}
26
27impl<'a> Oracle<'a> {
28    /// Create a new Oracle instance.
29    pub fn new(db: &'a AletheiaDB) -> Self {
30        Self { db }
31    }
32
33    /// Calculate Personalized PageRank (PPR) starting from a seed node.
34    ///
35    /// This simulates random walks starting from the seed node.
36    /// At each step, the walk either:
37    /// - Terminates with probability `alpha` (restart probability).
38    /// - Moves to a random neighbor.
39    ///
40    /// # Arguments
41    /// * `seed`: The starting node.
42    /// * `alpha`: Teleport probability (probability of restart/termination). Typical value 0.15.
43    /// * `num_walks`: Number of random walks to perform.
44    /// * `max_steps`: Maximum steps per walk (to prevent infinite loops in cycles).
45    ///
46    /// # Returns
47    /// A map of NodeId -> Visit Count (normalized probability).
48    pub fn personalized_page_rank(
49        &self,
50        seed: NodeId,
51        alpha: f32,
52        num_walks: usize,
53        max_steps: usize,
54    ) -> Result<HashMap<NodeId, f32>> {
55        // PERFORMANCE: Use IdentityHasher instead of the default SipHash for the `visits` map.
56        // `NodeId` is a wrapper around a unique `u64`. In a Monte Carlo simulation (random walks),
57        // we perform millions of map insertions and lookups. Bypassing SipHash eliminates
58        // unnecessary hashing overhead on already-unique integer keys, significantly improving throughput.
59        let mut visits: HashMap<NodeId, usize, BuildHasherDefault<IdentityHasher>> =
60            HashMap::default();
61        let mut rng = rand::thread_rng();
62
63        // Ensure seed exists
64        if self.db.get_node(seed).is_err() {
65            return Err(crate::core::error::Error::Storage(
66                StorageError::NodeNotFound(seed),
67            ));
68        }
69
70        for _ in 0..num_walks {
71            let mut current = seed;
72            // Count start node
73            *visits.entry(current).or_default() += 1;
74
75            for _step in 0..max_steps {
76                // Check for restart/termination
77                if rng.r#gen::<f32>() < alpha {
78                    break;
79                }
80
81                // Get outgoing edges
82                let edges = self.db.get_outgoing_edges(current);
83                if edges.is_empty() {
84                    break; // Stuck at sink
85                }
86
87                // Pick random neighbor
88                let edge_idx = rng.gen_range(0..edges.len());
89                let edge_id = edges[edge_idx];
90
91                // Move
92                // If edge target is invalid (shouldn't happen in consistent DB), we stop.
93                match self.db.get_edge_target(edge_id) {
94                    Ok(target) => {
95                        current = target;
96                        *visits.entry(current).or_default() += 1;
97                    }
98                    Err(_) => break,
99                }
100            }
101        }
102
103        // Normalize
104        let total_visits: usize = visits.values().sum();
105        let mut scores = HashMap::new();
106
107        if total_visits > 0 {
108            for (node, count) in visits {
109                scores.insert(node, count as f32 / total_visits as f32);
110            }
111        } else {
112            // Should not happen if num_walks > 0, as start node is always visited
113            scores.insert(seed, 1.0);
114        }
115
116        Ok(scores)
117    }
118
119    /// Estimate the probability of reaching `target` from `start` within `max_steps`.
120    ///
121    /// # Arguments
122    /// * `start`: Starting node.
123    /// * `target`: Target node.
124    /// * `max_steps`: Maximum depth to search.
125    /// * `num_simulations`: Number of Monte Carlo simulations.
126    pub fn reachability_probability(
127        &self,
128        start: NodeId,
129        target: NodeId,
130        max_steps: usize,
131        num_simulations: usize,
132    ) -> Result<f32> {
133        // Basic checks
134        if self.db.get_node(start).is_err() {
135            return Err(crate::core::error::Error::Storage(
136                StorageError::NodeNotFound(start),
137            ));
138        }
139        if self.db.get_node(target).is_err() {
140            return Err(crate::core::error::Error::Storage(
141                StorageError::NodeNotFound(target),
142            ));
143        }
144
145        if start == target {
146            return Ok(1.0);
147        }
148
149        let mut hits = 0;
150        let mut rng = rand::thread_rng();
151
152        for _ in 0..num_simulations {
153            let mut current = start;
154            let mut found = false;
155
156            for _ in 0..max_steps {
157                let edges = self.db.get_outgoing_edges(current);
158                if edges.is_empty() {
159                    break;
160                }
161
162                let edge_idx = rng.gen_range(0..edges.len());
163                let edge_id = edges[edge_idx];
164
165                match self.db.get_edge_target(edge_id) {
166                    Ok(next) => {
167                        if next == target {
168                            found = true;
169                            break;
170                        }
171                        current = next;
172                    }
173                    Err(_) => break,
174                }
175            }
176
177            if found {
178                hits += 1;
179            }
180        }
181
182        Ok(hits as f32 / num_simulations as f32)
183    }
184}

191    #[test]
192    fn test_oracle_ppr_sink() {
193        let db = AletheiaDB::new().unwrap();
194        let props = PropertyMapBuilder::new().build();
195
196        // A -> B (Sink)
197        let a = db.create_node("Node", props.clone()).unwrap();
198        let b = db.create_node("Node", props.clone()).unwrap();
199        db.create_edge(a, b, "NEXT", props.clone()).unwrap();
200
201        let oracle = Oracle::new(&db);
202
203        // Run PPR from A
204        // Alpha = 0.5 (50% chance to stop at each step)
205        // Expected: A is visited (start), then 50% chance to visit B.
206        // Normalized: A should be higher than B because A is always visited at start.
207        // Visits: A (N times), B (N * 0.5 times).
208        // Total: 1.5 N.
209        // P(A) = 1 / 1.5 = 0.66
210        // P(B) = 0.5 / 1.5 = 0.33
211
212        let ppr = oracle.personalized_page_rank(a, 0.5, 1000, 10).unwrap();
213
214        let score_a = *ppr.get(&a).unwrap();
215        let score_b = *ppr.get(&b).unwrap();
216
217        assert!(
218            score_a > score_b,
219            "Start node should have higher probability in sink graph"
220        );
221        assert!(
222            score_a > 0.6 && score_a < 0.75,
223            "P(A) should be around 0.66 (got {})",
224            score_a
225        );
226    }

228    #[test]
229    fn test_oracle_reachability() {
230        let db = AletheiaDB::new().unwrap();
231        let props = PropertyMapBuilder::new().build();
232
233        // A -> B -> C
234        let a = db.create_node("Node", props.clone()).unwrap();
235        let b = db.create_node("Node", props.clone()).unwrap();
236        let c = db.create_node("Node", props.clone()).unwrap();
237
238        db.create_edge(a, b, "NEXT", props.clone()).unwrap();
239        db.create_edge(b, c, "NEXT", props.clone()).unwrap();
240
241        let oracle = Oracle::new(&db);
242
243        // Reach C from A in 2 steps
244        // Deterministic path, so probability should be 1.0 (approximated)
245        let prob = oracle.reachability_probability(a, c, 5, 100).unwrap();
246        assert!(prob > 0.95, "Should reach C with high probability");
247
248        // Reach C from A in 1 step (Impossible)
249        let prob_short = oracle.reachability_probability(a, c, 1, 100).unwrap();
250        assert!(prob_short < 0.01, "Should not reach C in 1 step");
251    }

253    #[test]
254    fn test_oracle_cycle() {
255        let db = AletheiaDB::new().unwrap();
256        let props = PropertyMapBuilder::new().build();
257
258        // A <-> B
259        let a = db.create_node("Node", props.clone()).unwrap();
260        let b = db.create_node("Node", props.clone()).unwrap();
261
262        db.create_edge(a, b, "NEXT", props.clone()).unwrap();
263        db.create_edge(b, a, "NEXT", props.clone()).unwrap();
264
265        let oracle = Oracle::new(&db);
266
267        // PPR from A with low alpha (long walks)
268        // Should converge to 0.5 / 0.5 distribution
269        let ppr = oracle.personalized_page_rank(a, 0.1, 2000, 100).unwrap();
270
271        let score_a = *ppr.get(&a).unwrap();
272        let score_b = *ppr.get(&b).unwrap();
273
274        assert!(
275            (score_a - score_b).abs() < 0.1,
276            "Scores should be roughly equal in symmetric cycle (A: {}, B: {})",
277            score_a,
278            score_b
279        );
280    }
