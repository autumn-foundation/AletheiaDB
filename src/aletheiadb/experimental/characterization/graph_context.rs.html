1//! Graph Context Exporter ("The Scribe")
2//!
3//! This module provides tools to generate rich, human-readable context descriptions
4//! for nodes in the graph. It is designed specifically for **Retrieval-Augmented Generation (RAG)**
5//! scenarios where an LLM needs to understand a node's state, history, and relationships.
6//!
7//! # Purpose
8//!
9//! When feeding graph data to an LLM, raw JSON or triples can be token-inefficient and hard
10//! for models to reason about. The `GraphContextBuilder` converts a node's "ego network"
11//! into a structured Markdown format that highlights:
12//!
13//! 1.  **Identity**: Who/what is this node? (Label, ID)
14//! 2.  **State**: What are its properties?
15//! 3.  **Evolution**: How has it changed over time? (Narrative history)
16//! 4.  **Context**: Who is it connected to? (Immediate neighborhood)
17//!
18//! # Usage
19//!
20//! ```rust
21//! use aletheiadb::AletheiaDB;
22//! use aletheiadb::experimental::graph_context::GraphContextBuilder;
23//! use aletheiadb::core::id::NodeId;
24//!
25//! # fn example(db: &AletheiaDB, node_id: NodeId) -> aletheiadb::core::error::Result<()> {
26//! // Create a builder for a specific node
27//! let context = GraphContextBuilder::new(db, node_id)
28//!     .with_history_limit(3)   // Limit history to last 3 events
29//!     .with_neighbor_limit(5)  // Limit neighbors to 5
30//!     .build()?;
31//!
32//! println!("{}", context);
33//! # Ok(())
34//! # }
35//! ```
36//!
37//! # Output Format
38//!
39//! The output is a Markdown string structured as follows:
40//!
41//! ```markdown
42//! # Node Context: 42 (Person)
43//!
44//! ## Properties
45//! - name: "Alice"
46//! - role: "Engineer"
47//!
48//! ## Evolution
49//! - 2023-10-27T10:00:00Z (v2): updated properties
50//!   - Modified property 'role' from '"Intern"' to '"Engineer"'
51//!
52//! ## Neighborhood
53//! - WORKS_AT -> Company (101)
54//!   - Properties: { since: 2020 }
55//! ```

57use crate::AletheiaDB;
58use crate::core::error::Result;
59use crate::core::id::NodeId;
60use crate::core::interning::{GLOBAL_INTERNER, InternedString};
61use crate::experimental::temporal_narrative::NarrativeGenerator;
62use std::fmt::Write;
63
64/// Builder for generating a rich context description of a node.
65///
66/// This generates a Markdown-formatted string containing the node's current state,
67/// recent history (evolution), and immediate neighborhood. It is designed for
68/// injecting context into LLM prompts.
69///
70/// # Example
71///
72/// ```rust
73/// # use aletheiadb::AletheiaDB;
74/// # use aletheiadb::experimental::graph_context::GraphContextBuilder;
75/// # fn example(db: &AletheiaDB, node_id: aletheiadb::core::id::NodeId) -> aletheiadb::core::error::Result<()> {
76/// let context = GraphContextBuilder::new(db, node_id)
77///     .with_history_limit(5)
78///     .with_neighbor_limit(10)
79///     .build()?;
80/// # Ok(())
81/// # }
82/// ```
83pub struct GraphContextBuilder<'a> {
84    db: &'a AletheiaDB,
85    center_node: NodeId,
86    history_limit: usize,
87    neighbor_limit: usize,
88}
89
90impl<'a> GraphContextBuilder<'a> {
91    /// Create a new builder for the given node.
92    ///
93    /// # Arguments
94    ///
95    /// * `db` - Reference to the database instance.
96    /// * `center_node` - The ID of the node to generate context for.
97    pub fn new(db: &'a AletheiaDB, center_node: NodeId) -> Self {
98        Self {
99            db,
100            center_node,
101            history_limit: 5,
102            neighbor_limit: 10,
103        }
104    }
105
106    /// Set the maximum number of history events to include.
107    ///
108    /// # Trade-offs
109    ///
110    /// *   **Higher limit**: Provides more context on how the node evolved, useful for understanding causality.
111    /// *   **Lower limit**: Saves tokens in the LLM prompt window.
112    ///
113    /// Defaults to 5.
114    pub fn with_history_limit(mut self, limit: usize) -> Self {
115        self.history_limit = limit;
116        self
117    }
118
119    /// Set the maximum number of neighbors to include.
120    ///
121    /// # Trade-offs
122    ///
123    /// *   **Higher limit**: Provides a broader view of the node's connections.
124    /// *   **Lower limit**: Saves tokens and prevents "context pollution" from irrelevant neighbors.
125    ///
126    /// Defaults to 10.
127    pub fn with_neighbor_limit(mut self, limit: usize) -> Self {
128        self.neighbor_limit = limit;
129        self
130    }
131
132    fn resolve(s: InternedString) -> String {
133        GLOBAL_INTERNER.resolve_or_else(s, || format!("<interned:{}>", s.as_u32()))
134    }
135
136    /// Build the context string (Markdown).
137    ///
138    /// # Returns
139    ///
140    /// A formatted Markdown string containing the node's context.
141    ///
142    /// # Errors
143    ///
144    /// Returns an error if the `center_node` does not exist in the database.
145    pub fn build(&self) -> Result<String> {
146        let mut output = String::new();
147        let node = self.db.get_node(self.center_node)?;
148        let label = Self::resolve(node.label);
149
150        // 1. Header
151        writeln!(
152            &mut output,
153            "# Node Context: {} ({})",
154            self.center_node.as_u64(),
155            label
156        )
157        .unwrap();
158
159        // 2. Properties
160        writeln!(&mut output, "\n## Properties").unwrap();
161        if node.properties.is_empty() {
162            writeln!(&mut output, "- (No properties)").unwrap();
163        } else {
164            // Sort keys for deterministic output
165            let mut props: Vec<_> = node.properties.iter().collect();
166            props.sort_by_key(|(k, _)| *k);
167
168            for (key_id, val) in props {
169                let key = Self::resolve(*key_id);
170                writeln!(&mut output, "- {}: {}", key, val).unwrap();
171            }
172        }
173
174        // 3. Evolution (History)
175        writeln!(&mut output, "\n## Evolution").unwrap();
176        let generator = NarrativeGenerator::new(self.db);
177        match generator.generate_node_narrative(self.center_node) {
178            Ok(events) => {
179                if events.is_empty() {
180                    writeln!(&mut output, "- No history available.").unwrap();
181                } else {
182                    for event in events.iter().rev().take(self.history_limit) {
183                        writeln!(
184                            &mut output,
185                            "- {} (v{}): {}",
186                            event.timestamp, event.version_number, event.description
187                        )
188                        .unwrap();
189                        for change in &event.changes {
190                            writeln!(&mut output, "  - {}", change).unwrap();
191                        }
192                    }
193                    if events.len() > self.history_limit {
194                        writeln!(
195                            &mut output,
196                            "- ... ({} more versions)",
197                            events.len() - self.history_limit
198                        )
199                        .unwrap();
200                    }
201                }
202            }
203            Err(e) => {
204                writeln!(&mut output, "- Error retrieving history: {}", e).unwrap();
205            }
206        }
207
208        // 4. Neighborhood
209        writeln!(&mut output, "\n## Neighborhood").unwrap();
210        let edges = self.db.get_outgoing_edges(self.center_node);
211        if edges.is_empty() {
212            writeln!(&mut output, "- (No outgoing edges)").unwrap();
213        } else {
214            writeln!(
215                &mut output,
216                "{} outgoing edges (showing max {}):",
217                edges.len(),
218                self.neighbor_limit
219            )
220            .unwrap();
221            for edge_id in edges.iter().take(self.neighbor_limit) {
222                if let Ok(edge) = self.db.get_edge(*edge_id) {
223                    let edge_label = Self::resolve(edge.label);
224                    // Try to get target node label if possible, otherwise just ID
225                    let target_desc = if let Ok(target_node) = self.db.get_node(edge.target) {
226                        format!(
227                            "{} ({})",
228                            Self::resolve(target_node.label),
229                            edge.target.as_u64()
230                        )
231                    } else {
232                        format!("Node {}", edge.target.as_u64())
233                    };
234
235                    writeln!(&mut output, "- {} -> {}", edge_label, target_desc).unwrap();
236
237                    // Edge properties (compact)
238                    if !edge.properties.is_empty() {
239                        let mut props_str: Vec<String> = edge
240                            .properties
241                            .iter()
242                            .map(|(k, v)| format!("{}: {}", Self::resolve(*k), v))
243                            .collect();
244                        // Sort for deterministic output
245                        props_str.sort();
246                        writeln!(
247                            &mut output,
248                            "  - Properties: {{ {} }}",
249                            props_str.join(", ")
250                        )
251                        .unwrap();
252                    }
253                }
254            }
255        }
256
257        Ok(output)
258    }
259}

261#[cfg(test)]
262mod tests {
263    use super::*;
264    use crate::api::transaction::WriteOps;
265    use crate::core::property::PropertyMapBuilder;
266
267    #[test]
268    fn test_graph_context_generation() {
269        let db = AletheiaDB::new().unwrap();
270
271        // 1. Create Node A (Center)
272        let props1 = PropertyMapBuilder::new()
273            .insert("name", "Alice")
274            .insert("role", "Engineer")
275            .build();
276        let node_a = db.create_node("Person", props1).unwrap();
277
278        // 2. Update Node A (History)
279        db.write(|tx| {
280            let props2 = PropertyMapBuilder::new()
281                .insert("name", "Alice")
282                .insert("role", "Senior Engineer")
283                .build();
284            tx.update_node(node_a, props2)
285        })
286        .unwrap();
287
288        // 3. Create Node B (Neighbor)
289        let props_b = PropertyMapBuilder::new()
290            .insert("name", "Gallifrey Inc")
291            .build();
292        let node_b = db.create_node("Company", props_b).unwrap();
293
294        // 4. Create Edge A -> B
295        let props_edge = PropertyMapBuilder::new().insert("since", 2020i64).build();
296        db.create_edge(node_a, node_b, "WORKS_AT", props_edge)
297            .unwrap();
298
299        // 5. Build Context
300        let context = GraphContextBuilder::new(&db, node_a)
301            .with_history_limit(5)
302            .build()
303            .unwrap();
304
305        println!("{}", context);
306
307        // 6. Assertions
308        assert!(context.contains("# Node Context:"));
309        assert!(context.contains("Person"));
310
311        // Check Properties
312        assert!(context.contains("name: \"Alice\""));
313        assert!(context.contains("role: \"Senior Engineer\""));
314
315        // Check Evolution
316        assert!(context.contains("## Evolution"));
317        assert!(context.contains("updated properties")); // Description
318        assert!(
319            context
320                .contains("Modified property 'role' from '\"Engineer\"' to '\"Senior Engineer\"'")
321        );
322
323        // Check Neighborhood
324        assert!(context.contains("## Neighborhood"));
325        assert!(context.contains("WORKS_AT -> Company"));
326        assert!(context.contains("since: 2020"));
327    }
