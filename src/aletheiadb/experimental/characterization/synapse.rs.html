1//! Synapse: Adaptive Graph Hebbian Learning.
2//!
3//! "Cells that fire together, wire together."
4//!
5//! Synapse implements a mechanism for the graph to "learn" from traversal patterns.
6//! It tracks usage statistics (traversals) for edges and adjusts their effective "weight"
7//! dynamically. Frequently used paths become "cheaper" to traverse, while unused paths
8//! decay over time.
9//!
10//! This enables **Adaptive Semantic Pathfinding**: finding paths that are both
11//! semantically relevant AND popular/proven.
12//!
13//! # Concepts
14//! - **Synaptic Weight**: A derived value from usage history. High usage = Low Cost.
15//! - **Hebbian Learning**: `observe(edge)` strengthens the connection.
16//! - **Forgetting**: `decay()` weakens all connections, simulating memory fading.
17//!
18//! # Usage
19//!
20//! ```rust
21//! // Requires features = ["nova"]
22//! use aletheiadb::AletheiaDB;
23//! use aletheiadb::experimental::synapse::{Synapse, SynapseContext};
24//!
25//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
26//! let db = AletheiaDB::new()?;
27//! // Create a persistent context to hold weights
28//! let context = SynapseContext::new();
29//! let synapse = Synapse::new(&db, &context);
30//!
31//! // ... perform traversals ...
32//! // synapse.observe(edge_id);
33//!
34//! // Find an adaptive path
35//! // let path = synapse.adaptive_semantic_path(start, end, "embedding")?;
36//! # Ok(())
37//! # }
38//! ```

51/// Shared state for Synapse weights.
52/// This structure holds the "memory" of the graph's usage.
53/// It is designed to be shared across multiple Synapse instances (e.g., in different threads).
54pub struct SynapseContext {
55    /// Maps EdgeId -> Usage Count.
56    /// Usage count is monotonic increasing (until manual reset/decay handling).
57    weights: DashMap<EdgeId, AtomicU64, BuildHasherDefault<IdentityHasher>>,
58}
59
60impl Default for SynapseContext {
61    fn default() -> Self {
62        Self::new()
63    }
64}
65
66impl SynapseContext {
67    /// Create a new empty Synapse context.
68    pub fn new() -> Self {
69        Self {
70            weights: DashMap::with_hasher(BuildHasherDefault::default()),
71        }
72    }
73
74    /// Get the raw usage count for an edge.
75    pub fn get_usage(&self, edge_id: EdgeId) -> u64 {
76        match self.weights.get(&edge_id) {
77            Some(val) => val.load(AtomicOrdering::Relaxed),
78            None => 0,
79        }
80    }
81
82    /// Increment usage for an edge.
83    pub fn observe(&self, edge_id: EdgeId) {
84        // Optimistic update
85        // If entry exists, fetch_add. If not, insert 1.
86        // DashMap's entry API is the cleanest way.
87        self.weights
88            .entry(edge_id)
89            .and_modify(|val| {
90                val.fetch_add(1, AtomicOrdering::Relaxed);
91            })
92            .or_insert(AtomicU64::new(1));
93    }
94
95    /// Apply decay to all weights.
96    ///
97    /// This multiplies all usage counts by `factor` (0.0 to 1.0).
98    /// Since we store integers, this is `(usage * factor) as u64`.
99    pub fn decay(&self, factor: f32) {
100        if factor >= 1.0 {
101            return;
102        }
103        // Iterate and update.
104        // DashMap iter() gives read access, but we access Atomics via &AtomicU64.
105        // Atomics can be mutated via shared reference.
106        for entry in self.weights.iter() {
107            let current = entry.value().load(AtomicOrdering::Relaxed);
108            let new_val = (current as f32 * factor) as u64;
109            entry.value().store(new_val, AtomicOrdering::Relaxed);
110        }
111    }
112
113    /// Clear all memory.
114    pub fn clear(&self) {
115        self.weights.clear();
116    }
117}

119/// The Synapse Engine.
120pub struct Synapse<'a> {
121    db: &'a AletheiaDB,
122    context: &'a SynapseContext,
123}
124
125#[derive(Clone, Copy, PartialEq)]
126struct State {
127    cost: f32,
128    node: NodeId,
129}
130
131impl Eq for State {}
132
133impl Ord for State {
134    fn cmp(&self, other: &Self) -> Ordering {
135        other
136            .cost
137            .partial_cmp(&self.cost)
138            .unwrap_or(Ordering::Equal)
139    }
140}
141
142impl PartialOrd for State {
143    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
144        Some(self.cmp(other))
145    }
146}
147
148impl<'a> Synapse<'a> {
149    /// Create a new Synapse engine linked to a DB and a Context.
150    pub fn new(db: &'a AletheiaDB, context: &'a SynapseContext) -> Self {
151        Self { db, context }
152    }
153
154    /// Record a traversal event.
155    pub fn observe(&self, edge_id: EdgeId) {
156        self.context.observe(edge_id);
157    }
158
159    /// Find an adaptive path that balances Semantic Similarity and Popularity.
160    ///
161    /// Cost Function:
162    /// `Cost = (1.0 - Similarity) * WeightFactor`
163    ///
164    /// Where `WeightFactor = 1.0 / (1.0 + log2(1.0 + Usage))`
165    ///
166    /// - If Usage = 0, WeightFactor = 1.0 (Standard Semantic Cost).
167    /// - If Usage = 100, WeightFactor ≈ 1/7. Cost is 7x cheaper.
168    ///
169    /// This encourages the algorithm to choose paths that are semantically "good enough"
170    /// but highly popular, over paths that are semantically perfect but unknown.
171    pub fn adaptive_semantic_path(
172        &self,
173        start: NodeId,
174        end: NodeId,
175        vector_prop: &str,
176    ) -> Result<Vec<NodeId>> {
177        // Validate Start/End and get Goal Vector for heuristic
178        let start_node = self.db.get_node(start)?;
179        let end_node = self.db.get_node(end)?;
180
181        let _start_vec = start_node
182            .properties
183            .get(vector_prop)
184            .and_then(|v| v.as_arc_vector())
185            .ok_or_else(|| {
186                Error::other(format!(
187                    "Start node {} missing vector property '{}'",
188                    start, vector_prop
189                ))
190            })?;
191
192        // Used for validation but not strictly needed for pathfinding (Dijkstra)
193        // Kept for consistency and potential heuristic addition
194        let _end_vec = end_node
195            .properties
196            .get(vector_prop)
197            .and_then(|v| v.as_arc_vector())
198            .ok_or_else(|| {
199                Error::other(format!(
200                    "End node {} missing vector property '{}'",
201                    end, vector_prop
202                ))
203            })?;
204
205        // Dijkstra / A*
206        let mut open_set = BinaryHeap::new();
207        open_set.push(State {
208            cost: 0.0,
209            node: start,
210        });
211
212        let mut came_from: HashMap<NodeId, NodeId, BuildHasherDefault<IdentityHasher>> =
213            HashMap::with_hasher(BuildHasherDefault::default());
214        let mut g_score: HashMap<NodeId, f32, BuildHasherDefault<IdentityHasher>> =
215            HashMap::with_hasher(BuildHasherDefault::default());
216        g_score.insert(start, 0.0);
217
218        while let Some(State {
219            cost: _current_f,
220            node: current,
221        }) = open_set.pop()
222        {
223            if current == end {
224                return Ok(self.reconstruct_path(came_from, current));
225            }
226
227            let current_node = if current == start {
228                start_node.clone()
229            } else {
230                self.db.get_node(current)?
231            };
232
233            let current_vec = current_node
234                .properties
235                .get(vector_prop)
236                .and_then(|v| v.as_arc_vector());
237
238            for edge_id in self.db.get_outgoing_edges(current) {
239                let neighbor = self.db.get_edge_target(edge_id)?;
240
241                let neighbor_node = self.db.get_node(neighbor)?;
242                let neighbor_vec = neighbor_node
243                    .properties
244                    .get(vector_prop)
245                    .and_then(|v| v.as_arc_vector());
246
247                // 1. Calculate Semantic Cost (Base Cost)
248                let semantic_cost = match (&current_vec, &neighbor_vec) {
249                    (Some(a), Some(b)) => (1.0 - cosine_similarity(a, b)?).max(0.001), // Ensure > 0
250                    _ => 1.0, // Penalize missing vectors
251                };
252
253                // 2. Calculate Synaptic Modulator (Popularity Bonus)
254                let usage = self.context.get_usage(edge_id);
255                // Formula: 1.0 / (1.0 + log2(1 + usage))
256                // Usage 0 -> 1.0
257                // Usage 1 -> 1.0 / (1 + 1) = 0.5
258                // Usage 3 -> 1.0 / (1 + 2) = 0.33
259                let weight_factor = 1.0 / (1.0 + ((1 + usage) as f32).log2());
260
261                let edge_cost = semantic_cost * weight_factor;
262
263                let tentative_g = g_score.get(&current).unwrap_or(&f32::INFINITY) + edge_cost;
264
265                if tentative_g < *g_score.get(&neighbor).unwrap_or(&f32::INFINITY) {
266                    came_from.insert(neighbor, current);
267                    g_score.insert(neighbor, tentative_g);
268
269                    // H = 0 (Dijkstra)
270                    let h_score = 0.0;
271
272                    let f = tentative_g + h_score;
273                    open_set.push(State {
274                        cost: f,
275                        node: neighbor,
276                    });
277                }
278            }
279        }
280
281        Err(Error::other("No path found"))
282    }
283
284    fn reconstruct_path(
285        &self,
286        came_from: HashMap<NodeId, NodeId, BuildHasherDefault<IdentityHasher>>,
287        mut current: NodeId,
288    ) -> Vec<NodeId> {
289        let mut total_path = vec![current];
290        while let Some(&prev) = came_from.get(&current) {
291            current = prev;
292            total_path.push(current);
293        }
294        total_path.reverse();
295        total_path
296    }
297}

304    #[test]
305    fn test_synapse_hebbian_learning() {
306        let db = AletheiaDB::new().unwrap();
307        // A -> B -> D (Path 1)
308        // A -> C -> D (Path 2)
309        // All vectors identical (Semantic cost 0).
310        // Initially, pathfinding should be ambiguous or pick based on ID order.
311        // Then we strengthen Path 1.
312
313        let props = PropertyMapBuilder::new()
314            .insert_vector("vec", &[1.0, 0.0])
315            .build();
316        let a = db.create_node("Node", props.clone()).unwrap();
317        let b = db.create_node("Node", props.clone()).unwrap();
318        let c = db.create_node("Node", props.clone()).unwrap();
319        let d = db.create_node("Node", props.clone()).unwrap();
320
321        let e_ab = db
322            .create_edge(a, b, "NEXT", PropertyMapBuilder::new().build())
323            .unwrap();
324        let e_bd = db
325            .create_edge(b, d, "NEXT", PropertyMapBuilder::new().build())
326            .unwrap();
327
328        let _e_ac = db
329            .create_edge(a, c, "NEXT", PropertyMapBuilder::new().build())
330            .unwrap();
331        let _e_cd = db
332            .create_edge(c, d, "NEXT", PropertyMapBuilder::new().build())
333            .unwrap();
334
335        let context = SynapseContext::new();
336        let synapse = Synapse::new(&db, &context);
337
338        // Train Path 1 (A->B->D)
339        for _ in 0..10 {
340            synapse.observe(e_ab);
341            synapse.observe(e_bd);
342        }
343
344        // Find path. Should prefer A->B->D because of low cost.
345        let path = synapse.adaptive_semantic_path(a, d, "vec").unwrap();
346
347        assert_eq!(path, vec![a, b, d]);
348    }

350    #[test]
351    fn test_synapse_semantic_vs_popularity_tradeoff() {
352        let db = AletheiaDB::new().unwrap();
353        // A -> B -> Target (Short, but semantically bad)
354        // A -> C -> Target (Longer/Equal length, but semantically perfect)
355
356        // A: [1, 0]
357        // B: [0, 1] (Opposite/Orthogonal) -> Sim 0.0 -> Cost 1.0
358        // C: [0.8, 0.6] -> Sim 0.8 -> Cost 0.2
359        // Target: [1, 0]
360
361        let props_a = PropertyMapBuilder::new()
362            .insert_vector("vec", &[1.0, 0.0])
363            .build();
364        let a = db.create_node("Node", props_a).unwrap();
365
366        let props_b = PropertyMapBuilder::new()
367            .insert_vector("vec", &[0.0, 1.0])
368            .build();
369        let b = db.create_node("Node", props_b).unwrap();
370
371        let props_c = PropertyMapBuilder::new()
372            .insert_vector("vec", &[0.8, 0.6]) // Sim 0.8 -> Cost 0.2
373            .build();
374        let c = db.create_node("Node", props_c).unwrap();
375
376        let props_target = PropertyMapBuilder::new()
377            .insert_vector("vec", &[1.0, 0.0])
378            .build();
379        let target = db.create_node("Node", props_target).unwrap();
380
381        let e_ab = db
382            .create_edge(a, b, "NEXT", PropertyMapBuilder::new().build())
383            .unwrap();
384        let e_bt = db
385            .create_edge(b, target, "NEXT", PropertyMapBuilder::new().build())
386            .unwrap();
387
388        let _e_ac = db
389            .create_edge(a, c, "NEXT", PropertyMapBuilder::new().build())
390            .unwrap();
391        let _e_ct = db
392            .create_edge(c, target, "NEXT", PropertyMapBuilder::new().build())
393            .unwrap();
394
395        let context = SynapseContext::new();
396        let synapse = Synapse::new(&db, &context);
397
398        // Initially, A->C->Target should be preferred (better semantics).
399        let initial_path = synapse.adaptive_semantic_path(a, target, "vec").unwrap();
400        assert_eq!(
401            initial_path,
402            vec![a, c, target],
403            "Should prefer semantic match initially"
404        );
405
406        // Boost Path B massively
407        // Usage 10,000 ensures boost factor ~0.07.
408        // Cost(B) = 1.0 * 0.07 = 0.07.
409        // Cost(C) = 0.2.
410        // B wins comfortably.
411
412        for _ in 0..10_000 {
413            synapse.observe(e_ab);
414            synapse.observe(e_bt);
415        }
416
417        let adapted_path = synapse.adaptive_semantic_path(a, target, "vec").unwrap();
418        assert_eq!(
419            adapted_path,
420            vec![a, b, target],
421            "Should prefer popular path after training"
422        );
423    }

425    #[test]
426    fn test_synapse_decay() {
427        let context = SynapseContext::new();
428        let edge = EdgeId::new(1).unwrap();
429
430        context.observe(edge);
431        context.observe(edge);
432        // Usage = 2
433
434        assert_eq!(context.get_usage(edge), 2);
435
436        context.decay(0.5);
437        // 2 * 0.5 = 1
438        assert_eq!(context.get_usage(edge), 1);
439
440        context.decay(0.5);
441        // 1 * 0.5 = 0.5 -> 0 (integer truncation)
442        assert_eq!(context.get_usage(edge), 0);
443    }
