1//! Gravity: Semantic Mass and Orbit Analysis.
2//!
3//! "Who are the real influencers?"
4//!
5//! The Gravity engine analyzes the semantic influence of nodes over time.
6//! It measures how much a node's neighbors are "pulled" towards it in vector space.
7//!
8//! # Concepts
9//! - **Mass**: The degree centrality of a node (number of connections).
10//! - **Orbit**: The trajectory of a neighbor relative to the center node.
11//! - **Velocity**: The rate at which a neighbor is moving towards (attraction) or away from (repulsion) the center.
12//!
13//! # Use Cases
14//! - **Influence Mapping**: Identifying nodes that drive semantic change in their community.
15//! - **Community Dynamics**: Detecting if a community is tightening (converging) or dispersing (diverging).
16//! - **Trend Analysis**: Measuring the velocity of adoption for new concepts.
17//! - **Simulation**: Predicting how vectors should move if influenced by "Gravity".

25/// Metrics describing the orbital dynamics of a neighbor relative to a center node.
26#[derive(Debug, Clone, PartialEq)]
27pub struct OrbitMetrics {
28    /// The neighbor node being analyzed.
29    pub neighbor_id: NodeId,
30    /// Semantic distance at the start of the window.
31    pub start_distance: Option<f32>,
32    /// Semantic distance at the end of the window.
33    pub end_distance: Option<f32>,
34    /// Velocity of semantic movement (change in distance per second).
35    /// Negative = Attraction (moving closer).
36    /// Positive = Repulsion (moving away).
37    pub velocity: Option<f32>,
38}

40/// The Gravity engine for analyzing semantic influence.
41pub struct GravityWell<'a> {
42    db: &'a AletheiaDB,
43}
44
45impl<'a> GravityWell<'a> {
46    /// Create a new GravityWell.
47    pub fn new(db: &'a AletheiaDB) -> Self {
48        Self { db }
49    }
50
51    /// Analyze the orbit of neighbors around a center node.
52    ///
53    /// # Arguments
54    ///
55    /// * `center_id` - The central node (the "Sun").
56    /// * `property` - The vector property to analyze (e.g., "embedding").
57    /// * `window` - The time range for analysis.
58    ///
59    /// # Returns
60    ///
61    /// A list of `OrbitMetrics` for each neighbor connected at `window.end`.
62    pub fn analyze_orbit(
63        &self,
64        center_id: NodeId,
65        property: &str,
66        window: TimeRange,
67    ) -> Result<Vec<OrbitMetrics>> {
68        // Fix transaction time to "now" for consistent view
69        let tx_time = time::now();
70
71        // 1. Get neighbors at the end of the window
72        // We only care about nodes that are currently connected (or at window end)
73        let edge_ids = self
74            .db
75            .get_outgoing_edges_at_time(center_id, window.end(), tx_time);
76
77        if edge_ids.is_empty() {
78            return Ok(Vec::new());
79        }
80
81        let edges = self
82            .db
83            .get_edges_at_time(&edge_ids, window.end(), tx_time)?;
84
85        // Extract valid target IDs
86        let target_ids: Vec<NodeId> = edges
87            .into_iter()
88            .filter_map(|(_, edge_opt)| edge_opt.map(|e| e.target))
89            .collect();
90
91        // 2. Get Center Node vectors at start and end
92        let center_start_vec = self.get_vector_at(center_id, property, window.start(), tx_time)?;
93        let center_end_vec = self.get_vector_at(center_id, property, window.end(), tx_time)?;
94
95        let duration_secs = window.duration_micros().unwrap_or(0) as f32 / 1_000_000.0;
96        let mut metrics = Vec::with_capacity(target_ids.len());
97
98        for target_id in target_ids {
99            let target_start_vec =
100                self.get_vector_at(target_id, property, window.start(), tx_time)?;
101            let target_end_vec = self.get_vector_at(target_id, property, window.end(), tx_time)?;
102
103            let start_dist = if let (Some(c), Some(t)) = (&center_start_vec, &target_start_vec) {
104                Some(cosine_distance(c, t))
105            } else {
106                None
107            };
108
109            let end_dist = if let (Some(c), Some(t)) = (&center_end_vec, &target_end_vec) {
110                Some(cosine_distance(c, t))
111            } else {
112                None
113            };
114
115            let velocity = if let (Some(s), Some(e)) = (start_dist, end_dist) {
116                if duration_secs > 0.0 {
117                    Some((e - s) / duration_secs)
118                } else {
119                    Some(0.0)
120                }
121            } else {
122                None
123            };
124
125            metrics.push(OrbitMetrics {
126                neighbor_id: target_id,
127                start_distance: start_dist,
128                end_distance: end_dist,
129                velocity,
130            });
131        }
132
133        Ok(metrics)
134    }
135
136    /// Helper to get vector property at a specific time.
137    fn get_vector_at(
138        &self,
139        node_id: NodeId,
140        property: &str,
141        valid_time: Timestamp,
142        tx_time: Timestamp,
143    ) -> Result<Option<Vec<f32>>> {
144        // 1. Try full history scan (most robust for experimental analysis)
145        // This ensures we find the exact version even if temporal indices are lagging.
146        // We use is_valid_at (ignoring transaction time) to find *any* version that covers the valid time.
147        // We iterate through all versions and keep the last match, effectively preferring the
148        // most recent version (highest ID) that claims validity for that time.
149        // This works around issues where past versions might be transactionally closed (corrections)
150        // but still represent the best knowledge of that time period.
151        if let Ok(history) = self.db.get_node_history(node_id) {
152            let mut best_match: Option<Vec<f32>> = None;
153            for version in history.versions {
154                if version.temporal.is_valid_at(valid_time)
155                    && let Some(val) = version.properties.get(property)
156                    && let Some(vec) = val.as_vector()
157                {
158                    best_match = Some(vec.to_vec());
159                }
160            }
161            if let Some(vec) = best_match {
162                return Ok(Some(vec));
163            }
164        }
165
166        // 2. Try standard temporal lookup (fast path)
167        if let Ok(node) = self.db.get_node_at_time(node_id, valid_time, tx_time) {
168            return Self::extract_vector(&node, property);
169        }
170
171        // 3. Fallback: Current state (latest)
172        if let Ok(node) = self.db.get_node(node_id) {
173            return Self::extract_vector(&node, property);
174        }
175
176        Ok(None)
177    }
178
179    fn extract_vector(node: &crate::core::graph::Node, property: &str) -> Result<Option<Vec<f32>>> {
180        if let Some(val) = node.properties.get(property)
181            && let Some(vec) = val.as_vector()
182        {
183            return Ok(Some(vec.to_vec()));
184        }
185        Ok(None)
186    }
187}

189/// The Gravity Simulator applies semantic forces to propose new vector positions.
190pub struct GravitySimulator<'a> {
191    db: &'a AletheiaDB,
192}
193
194impl<'a> GravitySimulator<'a> {
195    /// Create a new Gravity Simulator.
196    pub fn new(db: &'a AletheiaDB) -> Self {
197        Self { db }
198    }
199
200    /// Simulate gravitational pull from a center node on its neighbors.
201    ///
202    /// # Arguments
203    ///
204    /// * `center_id` - The node exerting the force.
205    /// * `property` - The vector property to simulate.
206    /// * `mass` - The "mass" of the center node (strength of pull).
207    /// * `step_size` - Simulation step size (0.0 to 1.0).
208    ///
209    /// # Returns
210    ///
211    /// A list of `(NodeId, Vec<f32>)` pairs representing the proposed new vector positions for neighbors.
212    pub fn simulate_pull(
213        &self,
214        center_id: NodeId,
215        property: &str,
216        mass: f32,
217        step_size: f32,
218    ) -> Result<Vec<(NodeId, Vec<f32>)>> {
219        // Get Center Vector (Current State)
220        let center_node = self.db.get_node(center_id)?;
221        let center_vec = match center_node
222            .properties
223            .get(property)
224            .and_then(|v| v.as_vector())
225        {
226            Some(v) => v,
227            None => return Ok(Vec::new()), // No gravity without a vector
228        };
229
230        // Get Neighbors
231        let edge_ids = self.db.get_outgoing_edges(center_id);
232        if edge_ids.is_empty() {
233            return Ok(Vec::new());
234        }
235
236        let mut proposed_updates = Vec::new();
237
238        for edge_id in edge_ids {
239            let target_id = match self.db.get_edge_target(edge_id) {
240                Ok(t) => t,
241                Err(_) => continue,
242            };
243
244            let target_node = match self.db.get_node(target_id) {
245                Ok(n) => n,
246                Err(_) => continue,
247            };
248
249            let target_vec = match target_node
250                .properties
251                .get(property)
252                .and_then(|v| v.as_vector())
253            {
254                Some(v) => v,
255                None => continue,
256            };
257
258            // Physics Calculation
259            // 1. Direction Vector (Center - Neighbor)
260            // Note: Vectors might have different lengths if database is messy, assuming same dim.
261            if center_vec.len() != target_vec.len() {
262                continue;
263            }
264
265            let mut diff = Vec::with_capacity(center_vec.len());
266            for (c, t) in center_vec.iter().zip(target_vec.iter()) {
267                diff.push(c - t);
268            }
269
270            let dist = magnitude(&diff);
271
272            // Avoid division by zero or extreme forces at close range
273            let effective_dist = dist.max(0.0001);
274
275            // Force Magnitude: F = G * M / r^2 (Assume G=1, neighbor mass=1)
276            // Damping factor to prevent explosion
277            let force = mass / (effective_dist * effective_dist + 0.1);
278
279            // Apply force
280            // New = Old + (Direction * Force * Step)
281            // Direction = Diff / Dist (Unit vector)
282            // So: New = Old + (Diff / Dist) * Force * Step
283            //     New = Old + Diff * (Force * Step / Dist)
284
285            let scale = (force * step_size) / effective_dist;
286
287            let mut new_vec = Vec::with_capacity(target_vec.len());
288            for (t, d) in target_vec.iter().zip(diff.iter()) {
289                new_vec.push(t + d * scale);
290            }
291
292            // Normalize result to keep it on the hypersphere
293            let normalized_vec = normalize(&new_vec);
294
295            proposed_updates.push((target_id, normalized_vec));
296        }
297
298        Ok(proposed_updates)
299    }
300}

302/// Calculate Cosine Distance (1.0 - Cosine Similarity).
303/// Range: [0.0, 2.0]. 0.0 = identical, 1.0 = orthogonal, 2.0 = opposite.
304fn cosine_distance(a: &[f32], b: &[f32]) -> f32 {
305    crate::core::vector::cosine_similarity(a, b)
306        .map(|sim| 1.0 - sim)
307        .unwrap_or(1.0)
308}

317    fn test_gravity_attraction() {
318        let db = AletheiaDB::new().unwrap();
319
320        // 1. Setup Center Node (The Sun) at [1.0, 0.0]
321        let center_props = PropertyMapBuilder::new()
322            .insert_vector("vec", &[1.0, 0.0])
323            .build();
324        let center = db.create_node("Sun", center_props).unwrap();
325
326        // 2. Setup Neighbor (The Planet) at [0.0, 1.0] (Orthogonal, dist = 1.0)
327        let neighbor_props = PropertyMapBuilder::new()
328            .insert_vector("vec", &[0.0, 1.0])
329            .build();
330        let neighbor = db.create_node("Planet", neighbor_props).unwrap();
331
332        // Connect them
333        db.create_edge(
334            center,
335            neighbor,
336            "ORBITS",
337            PropertyMapBuilder::new().build(),
338        )
339        .unwrap();
340
341        // Ensure start time is strictly after creation
342        std::thread::sleep(std::time::Duration::from_millis(10));
343        let t_start = time::now();
344
345        // Wait a bit to establish duration
346        std::thread::sleep(std::time::Duration::from_millis(50));
347
348        // 3. Move Neighbor closer to [1.0, 0.0] -> [0.7, 0.7] (approx 45 deg)
349        let update_props = PropertyMapBuilder::new()
350            .insert_vector("vec", &[0.707, 0.707])
351            .build();
352
353        db.write(|tx| tx.update_node(neighbor, update_props))
354            .unwrap();
355
356        std::thread::sleep(std::time::Duration::from_millis(50));
357        let t_end = time::now();
358
359        // 4. Analyze Orbit
360        let gravity = GravityWell::new(&db);
361        let window = TimeRange::new(t_start, t_end).unwrap();
362
363        let metrics = gravity.analyze_orbit(center, "vec", window).unwrap();
364
365        assert_eq!(metrics.len(), 1);
366        let m = &metrics[0];
367
368        assert!(m.start_distance.is_some(), "Start distance is None");
369        assert!(m.end_distance.is_some(), "End distance is None");
370
371        // Start dist (orthogonal) ~= 1.0
372        // End dist (45 deg) ~= 1.0 - 0.707 = 0.293
373        assert!(
374            m.start_distance.unwrap() > 0.9,
375            "Start distance should be high"
376        );
377        assert!(
378            m.end_distance.unwrap() < 0.4,
379            "End distance should be lower"
380        );
381
382        // Velocity should be negative (attraction)
383        assert!(
384            m.velocity.unwrap() < 0.0,
385            "Velocity should be negative (attraction)"
386        );
387    }

389    #[test]
390    fn test_gravity_simulator() {
391        let db = AletheiaDB::new().unwrap();
392
393        // Center: [1.0, 0.0]
394        let center = db
395            .create_node(
396                "Sun",
397                PropertyMapBuilder::new()
398                    .insert_vector("vec", &[1.0, 0.0])
399                    .build(),
400            )
401            .unwrap();
402
403        // Neighbor: [0.0, 1.0] (Orthogonal)
404        let neighbor = db
405            .create_node(
406                "Planet",
407                PropertyMapBuilder::new()
408                    .insert_vector("vec", &[0.0, 1.0])
409                    .build(),
410            )
411            .unwrap();
412
413        // Connect
414        db.create_edge(
415            center,
416            neighbor,
417            "ORBITS",
418            PropertyMapBuilder::new().build(),
419        )
420        .unwrap();
421
422        let sim = GravitySimulator::new(&db);
423
424        // Simulate
425        // Mass = 10.0, Step = 0.1
426        // Force will be significant.
427        // Diff = [1, -1]
428        // Dist = sqrt(2) = 1.414
429        // Force = 10 / (2 + 0.1) = 4.76
430        // Scale = 4.76 * 0.1 / 1.414 = 0.33
431        // NewVec = [0, 1] + [1, -1] * 0.33 = [0.33, 0.67]
432        // Normalize([0.33, 0.67]) -> [0.44, 0.89] approx
433        // It moved towards [1, 0] (x increased from 0)
434
435        let updates = sim.simulate_pull(center, "vec", 10.0, 0.1).unwrap();
436
437        assert_eq!(updates.len(), 1);
438        let (id, new_vec) = &updates[0];
439        assert_eq!(*id, neighbor);
440
441        // Check X component increased (moved towards 1.0)
442        assert!(
443            new_vec[0] > 0.1,
444            "X component should increase (got {})",
445            new_vec[0]
446        );
447        // Check Y component decreased (moved away from 1.0)
448        assert!(
449            new_vec[1] < 0.95,
450            "Y component should decrease (got {})",
451            new_vec[1]
452        );
453
454        // Verify normalization
455        let mag = new_vec.iter().map(|x| x * x).sum::<f32>().sqrt();
456        assert!((mag - 1.0).abs() < 1e-5, "Vector should be normalized");
457    }
