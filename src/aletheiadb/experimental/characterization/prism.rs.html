1//! Prism: Semantic Spectroscopy for Vectors.
2//!
3//! "Unbox the Black Box."
4//!
5//! The Prism decomposes a vector embedding into named semantic components (facets).
6//! It allows you to understand *why* a vector is positioned where it is by projecting
7//! it onto a set of user-defined axes (concepts).
8//!
9//! # Use Cases
10//! - **Explainability**: Why did the search retrieve this? "It's 80% 'Sci-Fi' and 20% 'Romance'."
11//! - **Debugging**: Detect if a vector has drifted into an unwanted semantic region (e.g., "NSFW").
12//! - **Filtering**: "Show me nodes that are high in 'Innovation' but low in 'Risk'."
13//!
14//! # Example
15//! ```rust,no_run
16//! use aletheiadb::AletheiaDB;
17//! use aletheiadb::experimental::prism::Prism;
18//! use aletheiadb::core::property::PropertyMapBuilder;
19//!
20//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
21//! let db = AletheiaDB::new()?;
22//!
23//! // Create dummy nodes for the example
24//! let props = PropertyMapBuilder::new().insert_vector("embedding", &[1.0, 0.0]).build();
25//! let tech_node_id = db.create_node("Concept", props)?;
26//!
27//! let props = PropertyMapBuilder::new().insert_vector("embedding", &[0.0, 1.0]).build();
28//! let magic_node_id = db.create_node("Concept", props)?;
29//!
30//! let props = PropertyMapBuilder::new().insert_vector("embedding", &[0.5, 0.5]).build();
31//! let target_node_id = db.create_node("Target", props)?;
32//!
33//! let mut prism = Prism::new(&db).with_vector_property("embedding");
34//!
35//! // Define the "lens"
36//! prism.add_axis_from_node("Tech", tech_node_id)?;
37//! prism.add_axis_from_node("Magic", magic_node_id)?;
38//!
39//! // Analyze a target
40//! let spectrum = prism.analyze_node(target_node_id)?;
41//!
42//! println!("Tech: {:.2}", spectrum.get("Tech").unwrap_or(&0.0));
43//! println!("Magic: {:.2}", spectrum.get("Magic").unwrap_or(&0.0));
44//! # Ok(())
45//! # }
46//! ```

55/// A point in the semantic evolution of a node.
56#[derive(Debug, Clone)]
57pub struct EvolutionPoint {
58    /// The timestamp when this state became valid.
59    pub timestamp: Timestamp,
60    /// The semantic scores at this time.
61    pub scores: HashMap<String, f32>,
62}
63
64/// A named axis in the semantic space.
65#[derive(Debug, Clone)]
66struct Axis {
67    name: String,
68    vector: Vec<f32>,
69}
70
71/// The Prism engine for vector decomposition.
72pub struct Prism<'a> {
73    db: &'a AletheiaDB,
74    axes: Vec<Axis>,
75    vector_property: Option<String>,
76}

78impl<'a> Prism<'a> {
79    /// Create a new Prism.
80    pub fn new(db: &'a AletheiaDB) -> Self {
81        Self {
82            db,
83            axes: Vec::new(),
84            vector_property: None,
85        }
86    }

88    /// Configure the vector property Prism should use for node-based analysis.
89    ///
90    /// When set, `add_axis_from_node` and `analyze_node` always read this property.
91    /// This avoids nondeterministic behavior on nodes with multiple vector properties.
92    pub fn with_vector_property(mut self, property: &str) -> Self {
93        self.vector_property = Some(property.to_string());
94        self
95    }

97    fn resolve_node_vector<'n>(
98        &self,
99        node_id: NodeId,
100        properties: &'n crate::core::property::PropertyMap,
101        explicit_property: Option<&str>,
102        context: &str,
103    ) -> Result<&'n [f32]> {
104        let selected_property = explicit_property.or(self.vector_property.as_deref());
105        if let Some(property) = selected_property {
106            let value = properties.get(property).ok_or_else(|| {
107                Error::Vector(VectorError::IndexError(format!(
108                    "Node {} is missing vector property '{}'",
109                    node_id, property
110                )))
111            })?;
112
113            return value.as_vector().ok_or_else(|| {
114                Error::Vector(VectorError::IndexError(format!(
115                    "Node {} property '{}' is not a dense vector",
116                    node_id, property
117                )))
118            });
119        }
120
121        let mut first_vector: Option<&[f32]> = None;
122        let mut vector_keys = Vec::new();
123
124        for (key, value) in properties.iter() {
125            if let Some(vector) = value.as_vector() {
126                if first_vector.is_none() {
127                    first_vector = Some(vector);
128                }
129                vector_keys.push(key.to_string());
130            }
131        }
132
133        match vector_keys.len() {
134            0 => Err(Error::Vector(VectorError::IndexError(format!(
135                "Node {} has no vector properties to {}",
136                node_id, context
137            )))),
138            1 => match first_vector {
139                Some(vector) => Ok(vector),
140                None => unreachable!("vector_keys length and first_vector state diverged"),
141            },
142            _ => {
143                vector_keys.sort();
144                Err(Error::Vector(VectorError::IndexError(format!(
145                    "Node {} has multiple vector properties [{}]. Configure Prism with \
146                     with_vector_property(...) or use *_with_property APIs for deterministic selection.",
147                    node_id,
148                    vector_keys.join(", ")
149                ))))
150            }
151        }
152    }

154    /// Add an axis defined by a raw vector.
155    /// The vector will be normalized to unit length.
156    pub fn add_axis(&mut self, name: &str, vector: Vec<f32>) {
157        // We normalize axes to ensure consistent projection values (cosine similarity-like)
158        let normalized = normalize(&vector);
159        self.axes.push(Axis {
160            name: name.to_string(),
161            vector: normalized,
162        });
163    }
164
165    /// Add an axis defined by a node's vector.
166    ///
167    /// Selection rules:
168    /// - Uses the configured property from `with_vector_property`, if set.
169    /// - Otherwise requires exactly one dense vector property on the node.
170    /// - Returns an error if the node has multiple vector properties.
171    pub fn add_axis_from_node(&mut self, name: &str, node_id: NodeId) -> Result<()> {
172        let node = self.db.get_node(node_id)?;
173        let vector = self.resolve_node_vector(node_id, &node.properties, None, "use as an axis")?;
174
175        self.add_axis(name, vector.to_vec());
176        Ok(())
177    }
178
179    /// Add an axis defined by a specific vector property on a node.
180    pub fn add_axis_from_node_with_property(
181        &mut self,
182        name: &str,
183        node_id: NodeId,
184        property: &str,
185    ) -> Result<()> {
186        let node = self.db.get_node(node_id)?;
187        let vector =
188            self.resolve_node_vector(node_id, &node.properties, Some(property), "use as an axis")?;
189        self.add_axis(name, vector.to_vec());
190        Ok(())
191    }

193    /// Orthogonalize the current axes using the Gram-Schmidt process.
194    ///
195    /// This ensures that the axes are mutually perpendicular, which makes
196    /// the decomposition strictly additive (Component A + Component B = Target).
197    ///
198    /// Priority is given to axes added earlier (they keep their direction).
199    /// Later axes have the components of earlier axes removed from them.
200    pub fn orthogonalize(&mut self) {
201        let mut ortho_axes: Vec<Axis> = Vec::with_capacity(self.axes.len());
202
203        for axis in &self.axes {
204            let mut v = axis.vector.clone();
205
206            // Subtract projections onto existing orthogonal axes
207            for u_axis in &ortho_axes {
208                let u = &u_axis.vector;
209                // proj_u(v) = (v . u) / (u . u) * u
210                // Since u is normalized, u . u = 1.0
211                // So proj_u(v) = (v . u) * u
212                let dot = dot_product(&v, u).unwrap_or(0.0);
213
214                // v = v - (dot * u)
215                for (vi, ui) in v.iter_mut().zip(u.iter()) {
216                    *vi -= dot * ui;
217                }
218            }
219
220            // If the remaining vector is significant, add it
221            if magnitude(&v) > 1e-6 {
222                let normalized = normalize(&v);
223                ortho_axes.push(Axis {
224                    name: axis.name.clone(),
225                    vector: normalized,
226                });
227            } else {
228                // The axis was redundant (linear combination of previous axes)
229                // We keep it but it will be effectively zero or we drop it?
230                // Let's drop it to maintain a clean basis, but this might confuse users who added it.
231                // Better to keep it? If it's zero, we can't normalize.
232                // Let's skip it.
233            }
234        }
235
236        self.axes = ortho_axes;
237    }

239    /// Analyze a raw vector against the axes.
240    ///
241    /// Returns a map of `Axis Name -> Score`.
242    /// The score is the scalar projection of the target onto the axis.
243    /// If axes are normalized, this is the dot product.
244    pub fn analyze(&self, target: &[f32]) -> Result<HashMap<String, f32>> {
245        let mut spectrum = HashMap::new();
246
247        for axis in &self.axes {
248            if axis.vector.len() != target.len() {
249                // Skip mismatching dimensions or return error?
250                // Returning error is safer.
251                return Err(Error::Vector(VectorError::DimensionMismatch {
252                    expected: axis.vector.len(),
253                    actual: target.len(),
254                }));
255            }
256
257            let score = dot_product(target, &axis.vector)?;
258            spectrum.insert(axis.name.clone(), score);
259        }
260
261        Ok(spectrum)
262    }

264    /// Analyze a node's vector.
265    ///
266    /// Selection rules mirror `add_axis_from_node`.
267    pub fn analyze_node(&self, node_id: NodeId) -> Result<HashMap<String, f32>> {
268        let node = self.db.get_node(node_id)?;
269        let vector = self.resolve_node_vector(node_id, &node.properties, None, "analyze")?;
270        self.analyze(vector)
271    }

273    /// Analyze a node using a specific vector property.
274    pub fn analyze_node_with_property(
275        &self,
276        node_id: NodeId,
277        property: &str,
278    ) -> Result<HashMap<String, f32>> {
279        let node = self.db.get_node(node_id)?;
280        let vector =
281            self.resolve_node_vector(node_id, &node.properties, Some(property), "analyze")?;
282        self.analyze(vector)
283    }

285    /// Analyze the semantic evolution of a node over a time range.
286    ///
287    /// This method retrieves the node's history and projects the vector at each version
288    /// onto the defined axes.
289    pub fn analyze_evolution(
290        &self,
291        node_id: NodeId,
292        time_range: TimeRange,
293    ) -> Result<Vec<EvolutionPoint>> {
294        let history = self.db.get_node_history(node_id)?;
295        let mut points = Vec::new();
296
297        for version in history.versions {
298            // Check if the version is relevant to the time range
299            if version.temporal.valid_time().overlaps(&time_range) {
300                // Try to extract the vector
301                // We ignore errors here (e.g. missing vector property in old versions)
302                // because we want to return points for versions that DO have vectors.
303                if let Ok(vector) = self.resolve_node_vector(
304                    node_id,
305                    &version.properties,
306                    None,
307                    "analyze evolution",
308                ) {
309                    let scores = self.analyze(vector)?;
310                    points.push(EvolutionPoint {
311                        timestamp: version.temporal.valid_time().start(),
312                        scores,
313                    });
314                }
315            }
316        }
317
318        Ok(points)
319    }

321    /// Calculate the "Residual" energy.
322    ///
323    /// Returns the magnitude of the part of the vector *not* explained by the current axes.
324    /// `Residual = || Vector - Sum(Projection_i) ||`
325    ///
326    /// NOTE: This is only mathematically rigorous if the axes are orthogonal.
327    pub fn residual(&self, target: &[f32]) -> Result<f32> {
328        // Reconstruct the vector from components
329        let mut reconstruction = vec![0.0; target.len()];
330
331        for axis in &self.axes {
332            let score = dot_product(target, &axis.vector)?;
333
334            // Add (score * axis_vector) to reconstruction
335            for (r, a) in reconstruction.iter_mut().zip(axis.vector.iter()) {
336                *r += score * a;
337            }
338        }
339
340        // Calculate difference
341        let diff: Vec<f32> = target
342            .iter()
343            .zip(reconstruction.iter())
344            .map(|(t, r)| t - r)
345            .collect();
346
347        Ok(magnitude(&diff))
348    }

359        let db = AletheiaDB::new().unwrap();
360        // Enable vector index to ensure vector logic works (though Prism works on raw props too)
361        let config = HnswConfig::new(2, DistanceMetric::Cosine);
362        db.enable_vector_index("embedding", config).unwrap();

365

379    #[test]
380    fn test_prism_orthogonalization() {
381        let db = AletheiaDB::new().unwrap();
382        let mut prism = Prism::new(&db);
383
384        // Axis 1: [1.0, 0.0]
385        prism.add_axis("X", vec![1.0, 0.0]);
386        // Axis 2: [1.0, 1.0] (Not orthogonal to X)
387        prism.add_axis("Diag", vec![1.0, 1.0]);
388
389        // Orthogonalize
390        prism.orthogonalize();
391
392        // Check axes
393        // X should remain [1.0, 0.0]
394        let x_axis = prism.axes.iter().find(|a| a.name == "X").unwrap();
395        assert!((x_axis.vector[0] - 1.0).abs() < 1e-5);
396        assert!((x_axis.vector[1] - 0.0).abs() < 1e-5);
397
398        // Diag should become [0.0, 1.0] (Y-axis)
399        // Original Diag was [0.707, 0.707] (normalized).
400        // Wait, add_axis normalizes input!
401        // Input: [1.0, 1.0] -> Normalized: [0.707, 0.707]
402        // Proj_X(Diag) = (Diag . X) * X = 0.707 * [1, 0] = [0.707, 0.0]
403        // Remainder = [0.707, 0.707] - [0.707, 0.0] = [0.0, 0.707]
404        // Normalized Remainder = [0.0, 1.0]
405        let diag_axis = prism.axes.iter().find(|a| a.name == "Diag").unwrap();
406        assert!((diag_axis.vector[0] - 0.0).abs() < 1e-5);
407        assert!((diag_axis.vector[1] - 1.0).abs() < 1e-5);
408    }

493    #[test]
494    fn test_prism_analyze_evolution() {
495        use crate::api::transaction::WriteOps;
496        use crate::core::temporal::{TimeRange, time};
497
498        let db = AletheiaDB::new().unwrap();
499        // Enable vector index so vectors are allowed
500        let config = HnswConfig::new(2, DistanceMetric::Cosine);
501        db.enable_vector_index("vec", config).unwrap();
502
503        let t1 = time::from_millis(1000);
504        let t2 = time::from_millis(2000);
505
506        // 1. Create Node at T1: [1.0, 0.0] (Pure Logic)
507        let props1 = PropertyMapBuilder::new()
508            .insert_vector("vec", &[1.0, 0.0])
509            .build();
510        let node_id = db
511            .write(|tx| tx.create_node_with_valid_time("Concept", props1, Some(t1)))
512            .unwrap();
513
514        // 2. Update Node at T2: [0.0, 1.0] (Pure Emotion)
515        let props2 = PropertyMapBuilder::new()
516            .insert_vector("vec", &[0.0, 1.0])
517            .build();
518        db.write(|tx| tx.update_node_with_valid_time(node_id, props2, Some(t2)))
519            .unwrap();
520
521        // 3. Setup Prism
522        let mut prism = Prism::new(&db).with_vector_property("vec");
523        prism.add_axis("Logic", vec![1.0, 0.0]);
524        prism.add_axis("Emotion", vec![0.0, 1.0]);
525
526        // 4. Analyze
527        let range = TimeRange::new(time::from_millis(0), time::from_millis(3000)).unwrap();
528        let points = prism.analyze_evolution(node_id, range).unwrap();
529
530        // 5. Verify
531        assert_eq!(points.len(), 2);
532
533        // Point 1: T1, Logic=1, Emotion=0
534        let p1 = &points[0];
535        assert_eq!(p1.timestamp, t1);
536        assert!((p1.scores.get("Logic").unwrap() - 1.0).abs() < 1e-5);
537        assert!((p1.scores.get("Emotion").unwrap() - 0.0).abs() < 1e-5);
538
539        // Point 2: T2, Logic=0, Emotion=1
540        let p2 = &points[1];
541        assert_eq!(p2.timestamp, t2);
542        assert!((p2.scores.get("Logic").unwrap() - 0.0).abs() < 1e-5);
543        assert!((p2.scores.get("Emotion").unwrap() - 1.0).abs() < 1e-5);
544    }
