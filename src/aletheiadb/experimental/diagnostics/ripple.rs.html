1//! Ripple: Semantic Causality Detector.
2//!
3//! "Does a change in A cause a Ripple in B?"
4//!
5//! The Ripple Detector identifies causal relationships between nodes by analyzing
6//! the time-lagged correlation of their semantic flux (rate of vector change).
7//!
8//! # Concept
9//!
10//! If Node A moves in vector space (high flux) and Node B consistently moves
11//! in vector space shortly after (lagged flux), there may be a causal link.
12//!
13//! # Use Cases
14//! - **Causal Inference**: "Did the news article (A) cause the stock price (B) to move?"
15//! - **Leading Indicators**: Identifying nodes that predict future activity in others.
16//! - **Propagation Analysis**: Tracing how information flows through the graph.
17//!
18//! # Example
19//!
20//! ```rust,no_run
21//! use aletheiadb::AletheiaDB;
22//! use aletheiadb::experimental::ripple::{RippleDetector, RippleConfig};
23//! use aletheiadb::core::temporal::{TimeRange, time};
24//!
25//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
26//! let db = AletheiaDB::new()?;
27//! # let source = db.create_node("Source", Default::default())?;
28//! # let target = db.create_node("Target", Default::default())?;
29//!
30//! let detector = RippleDetector::new(&db);
31//!
32//! // Look for causal ripples in the last hour, with 1-minute resolution
33//! let end = time::now();
34//! let start = (end.wallclock() - 3600 * 1_000_000).into();
35//! let config = RippleConfig {
36//!     window: TimeRange::new(start, end)?,
37//!     bin_size_us: 60 * 1_000_000, // 1 minute
38//!     max_lag_bins: 10,            // Check lag up to 10 minutes
39//!     min_correlation: 0.5,        // Minimum correlation to report
40//! };
41//!
42//! if let Some(ripple) = detector.detect_causality(source, target, &config, "embedding")? {
43//!     println!("Found Ripple! Lag: {} mins, Correlation: {:.2}",
44//!         ripple.lag_bins, ripple.correlation);
45//! }
46//! # Ok(())
47//! # }
48//! ```

50use crate::AletheiaDB;
51use crate::core::error::{Error, Result, StorageError};
52use crate::core::history::EntityHistory;

57/// Maximum number of bins allowed to prevent OOM / DoS.
58const MAX_BINS: usize = 1_000_000;
59
60/// Configuration for Ripple detection.
61#[derive(Debug, Clone)]
62pub struct RippleConfig {
63    /// The time window to analyze.
64    pub window: TimeRange,
65    /// The resolution of time bins in microseconds.
66    pub bin_size_us: i64,
67    /// Maximum lag to check (in number of bins).
68    pub max_lag_bins: usize,
69    /// Minimum correlation coefficient to consider a valid ripple.
70    pub min_correlation: f32,
71}
72
73impl Default for RippleConfig {
74    fn default() -> Self {
75        // Safe default: 1 hour window from Epoch 0.
76        // Users should override `window` with a meaningful range relative to their data.
77        let hour_us = 3600 * 1_000_000;
78        Self {
79            window: TimeRange::new(0.into(), hour_us.into()).unwrap(),
80            bin_size_us: 1_000_000, // 1 second
81            max_lag_bins: 10,
82            min_correlation: 0.5,
83        }
84    }
85}

87/// A detected causal ripple.
88#[derive(Debug, Clone, PartialEq)]
89pub struct RippleEffect {
90    /// The source node.
91    pub source: NodeId,
92    /// The target node.
93    pub target: NodeId,
94    /// The lag at which maximum correlation was found (in bins).
95    /// A positive lag means Source leads Target.
96    pub lag_bins: usize,
97    /// The lag in microseconds.
98    pub lag_us: i64,
99    /// The Pearson correlation coefficient at the optimal lag (-1.0 to 1.0).
100    pub correlation: f32,
101    /// Confidence score (based on number of data points).
102    pub confidence: f32,
103}
104
105/// The Ripple Detector engine.
106pub struct RippleDetector<'a> {
107    db: &'a AletheiaDB,
108}
109
110impl<'a> RippleDetector<'a> {
111    /// Create a new RippleDetector.
112    pub fn new(db: &'a AletheiaDB) -> Self {
113        Self { db }
114    }
115
116    /// Detect if there is a causal ripple from source to target.
117    pub fn detect_causality(
118        &self,
119        source: NodeId,
120        target: NodeId,
121        config: &RippleConfig,
122        property: &str,
123    ) -> Result<Option<RippleEffect>> {
124        // 1. Get histories
125        let source_hist = self.db.get_node_history(source)?;
126        let target_hist = self.db.get_node_history(target)?;
127
128        // 2. Compute Semantic Flux Series
129        let source_flux = self.compute_flux(&source_hist, config, property)?;
130        let target_flux = self.compute_flux(&target_hist, config, property)?;
131
132        // 3. Check data sufficiency
133        // If either series is all zeros or too short, we can't detect causality.
134        if source_flux.iter().all(|&x| x == 0.0) || target_flux.iter().all(|&x| x == 0.0) {
135            return Ok(None);
136        }
137
138        // 4. Cross-Correlate
139        let (best_lag, max_corr) =
140            self.cross_correlate(&source_flux, &target_flux, config.max_lag_bins);
141
142        if max_corr >= config.min_correlation {
143            Ok(Some(RippleEffect {
144                source,
145                target,
146                lag_bins: best_lag,
147                lag_us: best_lag as i64 * config.bin_size_us,
148                correlation: max_corr,
149                confidence: 1.0, // Placeholder
150            }))
151        } else {
152            Ok(None)
153        }
154    }
155
156    /// Compute the semantic flux (magnitude of vector change) time series.
157    fn compute_flux(
158        &self,
159        history: &EntityHistory,
160        config: &RippleConfig,
161        property: &str,
162    ) -> Result<Vec<f32>> {
163        let start_time = config.window.start().wallclock();
164        let end_time = config.window.end().wallclock();
165        let duration = end_time - start_time;
166
167        if duration <= 0 {
168            return Ok(Vec::new());
169        }
170
171        let num_bins = (duration / config.bin_size_us) as usize + 1;
172
173        if num_bins > MAX_BINS {
174            return Err(Error::Storage(StorageError::CapacityExceeded {
175                resource: "Ripple detection bins".to_string(),
176                current: num_bins,
177                limit: MAX_BINS,
178            }));
179        }
180
181        let mut flux = vec![0.0; num_bins];
182
183        // We need pairs of consecutive versions to compute distance.
184        // We filter versions within the window first.
185        // Actually, we need the version BEFORE the window start to compute the first flux if an update happens right at start?
186        // Let's iterate all versions and check timestamps.
187
188        let mut prev_vec: Option<Vec<f32>> = None;
189
190        for version in &history.versions {
191            let ts = version.temporal.valid_time().start().wallclock();
192
193            // Extract vector
194            let current_vec = version
195                .properties
196                .get(property)
197                .and_then(|v| v.as_vector())
198                .map(|v| v.to_vec());
199
200            // If timestamp is within window
201            #[allow(clippy::collapsible_if)]
202            if ts >= start_time && ts < end_time {
203                if let (Some(curr), Some(prev)) = (&current_vec, &prev_vec) {
204                    let dist = euclidean_distance(curr, prev)?;
205                    let bin_idx = ((ts - start_time) / config.bin_size_us) as usize;
206                    if bin_idx < num_bins {
207                        flux[bin_idx] += dist;
208                    }
209                }
210            }
211
212            // Update previous vector for next iteration
213            // We update prev_vec regardless of window, so that if we enter the window,
214            // we have the state just before it.
215            if current_vec.is_some() {
216                prev_vec = current_vec;
217            }
218        }
219
220        Ok(flux)
221    }
222
223    /// Compute cross-correlation for lags 0..max_lag.
224    /// Returns (best_lag, max_correlation).
225    fn cross_correlate(&self, source: &[f32], target: &[f32], max_lag: usize) -> (usize, f32) {
226        let n = source.len().min(target.len());
227        if n == 0 {
228            return (0, 0.0);
229        }
230
231        let source = &source[0..n];
232        let target = &target[0..n];
233
234        // Pre-calculate means and std devs for full series?
235        // Technically, correlation should be over the overlapping window.
236        // But for simplicity in sliding window, we'll assume stationarity or use full series stats.
237        // Let's use the standard Pearson correlation on the shifted arrays.
238
239        let mut best_lag = 0;
240        let mut max_corr = -1.0;
241
242        for lag in 0..=max_lag {
243            if lag >= n {
244                break;
245            }
246
247            // Shift source forward by lag (Source leads Target)
248            // So we compare Source[0..N-lag] with Target[lag..N]
249            let s_slice = &source[0..n - lag];
250            let t_slice = &target[lag..n];
251
252            let corr = self.pearson_correlation(s_slice, t_slice);
253
254            if corr > max_corr {
255                max_corr = corr;
256                best_lag = lag;
257            }
258        }
259
260        (best_lag, max_corr)
261    }
262
263    fn pearson_correlation(&self, x: &[f32], y: &[f32]) -> f32 {
264        let n = x.len();
265        if n < 2 {
266            return 0.0;
267        }
268
269        let mean_x = x.iter().sum::<f32>() / n as f32;
270        let mean_y = y.iter().sum::<f32>() / n as f32;
271
272        let mut num = 0.0;
273        let mut den_x = 0.0;
274        let mut den_y = 0.0;
275
276        for i in 0..n {
277            let dx = x[i] - mean_x;
278            let dy = y[i] - mean_y;
279            num += dx * dy;
280            den_x += dx * dx;
281            den_y += dy * dy;
282        }
283
284        if den_x == 0.0 || den_y == 0.0 {
285            return 0.0;
286        }
287
288        num / (den_x.sqrt() * den_y.sqrt())
289    }

299    #[test]
300    fn test_ripple_exact_match_zero_lag() {
301        let db = AletheiaDB::new().unwrap();
302        let t0 = time::now();
303        let bin_size = 1_000; // 1ms
304
305        // Create Source
306        let props_a = PropertyMapBuilder::new()
307            .insert_vector("vec", &[0.0, 0.0])
308            .build();
309        let a = db.create_node("Source", props_a).unwrap();
310
311        // Create Target
312        let props_b = PropertyMapBuilder::new()
313            .insert_vector("vec", &[0.0, 0.0])
314            .build();
315        let b = db.create_node("Target", props_b).unwrap();
316
317        // Update both at T+10ms
318        std::thread::sleep(std::time::Duration::from_millis(10)); // Ensure distinct time from creation
319        let _t1 = time::now(); // Approx creation time? No, now is T1.
320
321        // Actually, creating history with precise timestamps is hard with `create_node`.
322        // We'll use `update_node_with_valid_time` if available, or just sleep.
323        // Sleeping is flaky.
324        // Let's use `update_node` and rely on system clock, but with small sleeps.
325
326        // Update A and B in the same transaction to ensure identical timestamps
327        db.write(|tx| {
328            tx.update_node(
329                a,
330                PropertyMapBuilder::new()
331                    .insert_vector("vec", &[1.0, 0.0])
332                    .build(),
333            )?;
334            tx.update_node(
335                b,
336                PropertyMapBuilder::new()
337                    .insert_vector("vec", &[1.0, 0.0])
338                    .build(),
339            )
340        })
341        .unwrap();
342
343        let t_end = time::now();
344        let detector = RippleDetector::new(&db);
345        let config = RippleConfig {
346            window: TimeRange::new(t0, t_end).unwrap(),
347            bin_size_us: bin_size, // 1s bins might be too large if updates are instant
348            // Let's make bins small
349            max_lag_bins: 5,
350            min_correlation: 0.9,
351        };
352
353        // If updates happened in same bin, lag is 0.
354        // Correlation should be 1.0 (both spiked).
355        // Wait, if flux is [0, 0, 1, 0...] for both.
356        // Correlation is 1.0.
357
358        let ripple = detector
359            .detect_causality(a, b, &config, "vec")
360            .unwrap()
361            .unwrap();
362
363        assert_eq!(ripple.lag_bins, 0);
364        assert!(ripple.correlation > 0.9);
365    }

367    #[test]
368    fn test_ripple_lagged() {
369        // We need to simulate lag.
370        // Since we can't easily force timestamps in the public API without `update_node_with_valid_time`
371        // (which is available on `WriteTransaction` but maybe not exposed conveniently in `AletheiaDB` wrapper?
372        // Let's check `AletheiaDB::write_transaction`.
373
374        let db = AletheiaDB::new().unwrap();
375        let t_base = time::now().wallclock();
376        let bin = 1_000_000; // 1s
377
378        let props = PropertyMapBuilder::new()
379            .insert_vector("vec", &[0.0])
380            .build();
381
382        // Helper to update at specific offset
383        let update_at = |id: NodeId, offset_us: i64, val: f32| {
384            let ts = crate::core::temporal::Timestamp::new(t_base + offset_us, 0).unwrap();
385            let mut tx = db.write_transaction().unwrap();
386            tx.update_node_with_valid_time(
387                id,
388                PropertyMapBuilder::new()
389                    .insert_vector("vec", &[val])
390                    .build(),
391                Some(ts),
392            )
393            .unwrap();
394            tx.commit().unwrap();
395        };
396
397        // Create nodes
398        let a = db.create_node("S", props.clone()).unwrap();
399        let b = db.create_node("T", props.clone()).unwrap();
400
401        // Pattern: Spike at T=2s for A, T=5s for B (Lag 3s)
402        update_at(a, 2 * bin, 1.0); // Flux 1.0 at bin 2
403        update_at(a, 3 * bin, 1.0); // Flux 0.0 at bin 3
404
405        update_at(b, 5 * bin, 1.0); // Flux 1.0 at bin 5 (Lag 3)
406        update_at(b, 6 * bin, 1.0); // Flux 0.0
407
408        let detector = RippleDetector::new(&db);
409        let config = RippleConfig {
410            window: TimeRange::new((t_base).into(), (t_base + 10 * bin).into()).unwrap(),
411            bin_size_us: bin,
412            max_lag_bins: 5,
413            min_correlation: 0.8,
414        };
415
416        // Flux A: [0, 0, 1, 0, 0, 0, 0...]
417        // Flux B: [0, 0, 0, 0, 0, 1, 0...]
418        // Lag should be 5 - 2 = 3.
419
420        let ripple = detector.detect_causality(a, b, &config, "vec").unwrap();
421
422        assert!(ripple.is_some(), "Should detect ripple");
423        let r = ripple.unwrap();
424        assert_eq!(r.lag_bins, 3);
425        assert!(r.correlation > 0.9);
426    }
