1//! Wormhole: Detecting Semantic-Structural Gaps.
2//!
3//! "Mind the Gap."
4//!
5//! The Wormhole Detector identifies pairs of nodes that are **semantically similar** (close in vector space)
6//! but **structurally distant** (or disconnected) in the graph.
7//! These represent "latent edges" or "missing links" in the knowledge graph.
8//!
9//! # Use Cases
10//! - **Link Prediction**: Suggesting new edges based on semantic similarity.
11//! - **Anomaly Detection**: Finding nodes that "should be connected" but aren't.
12//! - **Serendipity**: Discovering unexpected relationships.
13//!
14//! # Example
15//! ```rust,no_run
16//! use aletheiadb::AletheiaDB;
17//! use aletheiadb::experimental::wormhole::WormholeDetector;
18//!
19//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
20//! let db = AletheiaDB::new()?;
21//! # let node_id_1 = db.create_node("Node", Default::default())?;
22//! # let node_id_2 = db.create_node("Node", Default::default())?;
23//! let detector = WormholeDetector::new(&db);
24//!
25//! // Find candidates for wormholes (e.g., all Person nodes)
26//! let candidates = vec![node_id_1, node_id_2]; // Populate with actual IDs
27//!
28//! // Find wormholes: Top 10 similar nodes, max 2 hops structural distance
29//! let wormholes = detector.find_wormholes(&candidates, 10, 2)?;
30//!
31//! for w in wormholes {
32//!     println!("Found wormhole from {} to {} (similarity: {}, distance: {:?})",
33//!         w.source, w.target, w.similarity, w.structural_distance);
34//! }
35//! # Ok(())
36//! # }
37//! ```
38
39use crate::AletheiaDB;
40use crate::core::error::Result;
41use crate::core::id::NodeId;
42use std::collections::{HashSet, VecDeque};
43
44/// A detected wormhole (semantic shortcut).
45#[derive(Debug, Clone, PartialEq)]
46pub struct Wormhole {
47    /// The source node of the wormhole.
48    pub source: NodeId,
49    /// The target node (semantically close to source).
50    pub target: NodeId,
51    /// The semantic similarity score (e.g., cosine similarity).
52    pub similarity: f32,
53    /// The shortest path distance in the graph.
54    /// `None` if no path exists within `max_hops`.
55    pub structural_distance: Option<usize>,
56}
57
58/// Detector for finding semantic wormholes.
59pub struct WormholeDetector<'a> {
60    db: &'a AletheiaDB,
61}
62
63impl<'a> WormholeDetector<'a> {
64    /// Create a new WormholeDetector.
65    pub fn new(db: &'a AletheiaDB) -> Self {
66        Self { db }
67    }
68
69    /// Find wormholes for a set of candidate nodes.
70    ///
71    /// For each candidate:
72    /// 1. Finds `k` most similar nodes in vector space.
73    /// 2. Checks structural distance to each similar node using BFS up to `max_hops`.
74    /// 3. Returns pairs where distance is `None` (disconnected or > max_hops).
75    ///
76    /// # Arguments
77    ///
78    /// * `candidates` - List of source nodes to check.
79    /// * `k` - Number of nearest semantic neighbors to consider.
80    /// * `max_hops` - Maximum graph distance to search before considering it a wormhole.
81    pub fn find_wormholes(
82        &self,
83        candidates: &[NodeId],
84        k: usize,
85        max_hops: usize,
86    ) -> Result<Vec<Wormhole>> {
87        let mut wormholes = Vec::new();
88
89        for &source in candidates {
90            // Find semantic neighbors using the default vector index.
91            // Note: the similarity search excludes the query node itself.
92            let semantic_neighbors = self
93                .db
94                .similarity_search(crate::SimilarityQuery::from_node(source).k(k))?;
95
96            for (target, similarity) in semantic_neighbors {
97                // Check structural distance
98                let distance = self.bfs_distance(source, target, max_hops)?;
99
100                // If distance is None, it means no path within max_hops was found.
101                // This is a candidate for a wormhole (strong semantic link, weak/no structural link).
102                if distance.is_none() {
103                    wormholes.push(Wormhole {
104                        source,
105                        target,
106                        similarity,
107                        structural_distance: None,
108                    });
109                }
110            }
111        }
112
113        Ok(wormholes)
114    }
115
116    /// Calculate shortest path distance using BFS up to max_depth.
117    ///
118    /// Returns:
119    /// - `Ok(Some(d))` if path found with length `d <= max_depth`.
120    /// - `Ok(None)` if no path found within `max_depth`.
121    fn bfs_distance(&self, start: NodeId, end: NodeId, max_depth: usize) -> Result<Option<usize>> {
122        if start == end {
123            return Ok(Some(0));
124        }
125
126        let mut queue = VecDeque::new();
127        queue.push_back((start, 0));
128
129        let mut visited = HashSet::new();
130        visited.insert(start);
131
132        while let Some((current, depth)) = queue.pop_front() {
133            if depth >= max_depth {
134                // If we reached max depth and haven't found target,
135                // we stop this branch. If queue empties, we return None.
136                continue;
137            }
138
139            // Iterate over outgoing edges
140            let edges_iter = self.db.current.get_outgoing_edges_iter(current);
141            for edge_id in edges_iter {
142                // Resolve target node (zero-copy)
143                let target = self.db.current.get_edge_target(edge_id)?;
144
145                if target == end {
146                    return Ok(Some(depth + 1));
147                }
148
149                if visited.insert(target) {
150                    queue.push_back((target, depth + 1));
151                }
152            }
153        }
154
155        Ok(None)
156    }
157}

159#[cfg(test)]
160mod tests {
161    use super::*;
162    use crate::core::property::PropertyMapBuilder;
163    use crate::index::vector::{DistanceMetric, HnswConfig};
164
165    #[test]
166    fn test_wormhole_detection() {
167        let db = AletheiaDB::new().unwrap();
168
169        // 1. Setup Vector Index
170        let config = HnswConfig::new(2, DistanceMetric::Cosine);
171        db.enable_vector_index("vec", config).unwrap();
172
173        // 2. Create Nodes
174        // A: [1.0, 0.0]
175        // B: [0.0, 1.0] (Orthogonal to A)
176        // C: [0.9, 0.1] (Close to A)
177
178        let props_a = PropertyMapBuilder::new()
179            .insert_vector("vec", &[1.0, 0.0])
180            .build();
181        let a = db.create_node("Node", props_a).unwrap();
182
183        let props_b = PropertyMapBuilder::new()
184            .insert_vector("vec", &[0.0, 1.0])
185            .build();
186        let b = db.create_node("Node", props_b).unwrap();
187
188        let props_c = PropertyMapBuilder::new()
189            .insert_vector("vec", &[0.9, 0.1])
190            .build();
191        let c = db.create_node("Node", props_c).unwrap();
192
193        // 3. Create Edges
194        // A -> B -> C
195        // Path A -> C exists (dist 2), but no direct edge A -> C.
196        db.create_edge(a, b, "NEXT", PropertyMapBuilder::new().build())
197            .unwrap();
198        db.create_edge(b, c, "NEXT", PropertyMapBuilder::new().build())
199            .unwrap();
200
201        let detector = WormholeDetector::new(&db);
202
203        // Case 1: Max Hops = 1
204        // BFS checks distance <= 1.
205        // A -> C distance is 2.
206        // So BFS returns None.
207        // A and C are semantically close.
208        // Expect Wormhole A -> C.
209        let wormholes = detector.find_wormholes(&[a], 10, 1).unwrap();
210
211        // Should find A -> C
212        let ac = wormholes.iter().find(|w| w.target == c);
213        assert!(ac.is_some(), "Should find wormhole A -> C with max_hops=1");
214        let w = ac.unwrap();
215        assert_eq!(w.source, a);
216        assert_eq!(w.target, c);
217        assert!(w.structural_distance.is_none());
218        assert!(w.similarity > 0.8); // High similarity
219
220        // Case 2: Max Hops = 2
221        // BFS checks distance <= 2.
222        // A -> C distance is 2.
223        // So BFS returns Some(2).
224        // Since distance found, it is NOT a wormhole (structurally connected enough).
225        let wormholes_2 = detector.find_wormholes(&[a], 10, 2).unwrap();
226        let ac_2 = wormholes_2.iter().find(|w| w.target == c);
227        assert!(
228            ac_2.is_none(),
229            "Should NOT find wormhole A -> C with max_hops=2 (path exists)"
230        );
231    }

233    #[test]
234    fn test_wormhole_disconnected() {
235        let db = AletheiaDB::new().unwrap();
236        let config = HnswConfig::new(2, DistanceMetric::Cosine);
237        db.enable_vector_index("vec", config).unwrap();
238
239        // A and D are disconnected but similar
240        let props_a = PropertyMapBuilder::new()
241            .insert_vector("vec", &[1.0, 0.0])
242            .build();
243        let a = db.create_node("Node", props_a).unwrap();
244
245        let props_d = PropertyMapBuilder::new()
246            .insert_vector("vec", &[0.95, 0.0])
247            .build();
248        let d = db.create_node("Node", props_d).unwrap();
249
250        let detector = WormholeDetector::new(&db);
251        let wormholes = detector.find_wormholes(&[a], 10, 5).unwrap();
252
253        assert!(!wormholes.is_empty());
254        assert_eq!(wormholes[0].target, d);
255        assert!(wormholes[0].structural_distance.is_none());
256    }
