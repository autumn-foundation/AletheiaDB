1//! Semantic Pathfinding Module (Experimental)
2//!
3//! This module implements pathfinding algorithms that consider semantic similarity
4//! (vector embeddings) as part of the cost function. Instead of just finding the shortest
5//! structural path between two nodes, semantic pathfinding finds the *most contextually relevant*
6//! path, allowing queries to be guided by a concept.
7//!
8//! # Features
9//! - **Semantic A***: Finds paths where intermediate nodes are semantically similar to a query concept.
10//! - **Time-Travel Pathfinding**: Finds paths that were valid at a specific historical point in time.
11//!
12//! # Examples
13//!
14//! ```no_run
15//! use aletheiadb::db::AletheiaDB;
16//! use aletheiadb::core::property::PropertyMapBuilder;
17//! use aletheiadb::query::semantic_pathfinding::SemanticPathfinder;
18//! use aletheiadb::index::vector::{HnswConfig, DistanceMetric};
19//! use aletheiadb::api::transaction::WriteOps;
20//!
21//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
22//! // 1. Setup Database with vector index
23//! let db = AletheiaDB::new()?;
24//! db.vector_index("embedding")
25//!     .hnsw(HnswConfig::new(3, DistanceMetric::Cosine))
26//!     .enable()?;
27//!
28//! // 2. Create nodes with vector embeddings
29//! let start = db.create_node("Start", PropertyMapBuilder::new().insert_vector("embedding", &[0.5, 0.5, 0.0]).build())?;
30//! let end = db.create_node("End", PropertyMapBuilder::new().insert_vector("embedding", &[0.5, 0.5, 0.0]).build())?;
31//!
32//! // Intermediate nodes
33//! let apple = db.create_node("Apple", PropertyMapBuilder::new().insert_vector("embedding", &[1.0, 0.0, 0.0]).build())?; // Fruit
34//! let laptop = db.create_node("Laptop", PropertyMapBuilder::new().insert_vector("embedding", &[0.0, 1.0, 0.0]).build())?; // Tech
35//!
36//! // Create edges: Start -> [Apple, Laptop] -> End
37//! db.create_edge(start, apple, "NEXT", PropertyMapBuilder::new().build())?;
38//! db.create_edge(apple, end, "NEXT", PropertyMapBuilder::new().build())?;
39//! db.create_edge(start, laptop, "NEXT", PropertyMapBuilder::new().build())?;
40//! db.create_edge(laptop, end, "NEXT", PropertyMapBuilder::new().build())?;
41//!
42//! // 3. Find path using a query vector (e.g., "Banana", which is closer to "Apple" than "Laptop")
43//! let query_embedding = vec![0.9, 0.1, 0.0];
44//! let pathfinder = SemanticPathfinder::new(&db, "embedding");
45//!
46//! // The pathfinder will prefer the route through 'Apple' because it's semantically closer
47//! let path = pathfinder.find_path(start, end, &query_embedding, 10, false)?.unwrap();
48//! assert_eq!(path, vec![start, apple, end]);
49//! # Ok(())
50//! # }
51//! ```
52//!
53//! # Errors
54//! Methods in this module will return a [`Result`] indicating failure if:
55//! - An underlying database read operation fails.
56//! - Property maps contain malformed vector properties.
57//!
58//! # Panics
59//! This module does not intentionally panic. If any node has a vector that differs in dimension from the `query_embedding`,
60//! it treats the cost as infinite (effectively blocking the path) instead of panicking.

72/// Per-hop structural cost to prefer shorter paths when semantic costs are equal.
73const STRUCTURAL_HOP_COST: f32 = 0.1;
74
75/// Default cost assigned to nodes without embeddings.
76const NO_EMBEDDING_COST: f32 = 1.0;
77
78/// State for priority queue (min-heap based on cost).
79#[derive(Debug, Clone, Copy, PartialEq)]
80struct State {
81    cost: f32,
82    node: NodeId,
83    depth: usize,
84}
85
86impl Eq for State {}
87
88impl Ord for State {
89    fn cmp(&self, other: &Self) -> Ordering {
90        // Reverse order for min-heap (we want smallest cost)
91        // Handle NaN by treating it as "greater than" (lowest priority)
92        if self.cost.is_nan() {
93            return Ordering::Less; // self > other (in reverse logic)
94        }
95        if other.cost.is_nan() {
96            return Ordering::Greater; // other > self
97        }
98        other
99            .cost
100            .partial_cmp(&self.cost)
101            .unwrap_or(Ordering::Equal)
102    }
103}
104
105impl PartialOrd for State {
106    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
107        Some(self.cmp(other))
108    }
109}

111/// A pathfinder that uses semantic similarity as a heuristic/cost.
112pub struct SemanticPathfinder<'a, G: GraphView + ?Sized> {
113    db: &'a G,
114    vector_property: String,
115}

117impl<'a, G: GraphView + ?Sized> SemanticPathfinder<'a, G> {
118    /// Creates a new [`SemanticPathfinder`].
119    ///
120    /// # Arguments
121    /// * `db` - Reference to the graph view (e.g., database instance).
122    /// * `vector_property` - Name of the property containing vector embeddings.
123    ///
124    /// # Examples
125    ///
126    /// ```no_run
127    /// use aletheiadb::db::AletheiaDB;
128    /// use aletheiadb::query::semantic_pathfinding::SemanticPathfinder;
129    ///
130    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
131    /// let db = AletheiaDB::new()?;
132    /// let pathfinder = SemanticPathfinder::new(&db, "embedding");
133    /// # Ok(())
134    /// # }
135    /// ```
136    pub fn new(db: &'a G, vector_property: &str) -> Self {
137        Self {
138            db,
139            vector_property: vector_property.to_string(),
140        }

143    /// Finds a path from `start` to `end` that minimizes semantic distance to the `query_embedding`.
144    ///
145    /// Uses Dijkstra's algorithm where the edge weight is calculated as:
146    /// `1.0 - cosine_similarity(target_node, query_embedding)`.
147    /// This heavily favors paths through nodes that are semantically relevant to the query concept.
148    /// A small structural cost (0.1) is added per hop to prefer shorter paths when semantics are equal.
149    ///
150    /// # Arguments
151    /// * `start` - Starting [`NodeId`].
152    /// * `end` - Target [`NodeId`].
153    /// * `query_embedding` - Vector representing the semantic concept to follow.
154    /// * `max_depth` - Maximum path length (to prevent infinite loops in large graphs).
155    /// * `bidirectional` - If `true`, traverses both outgoing and incoming edges.
156    ///
157    /// # Returns
158    /// An `Ok(Some(Vec<NodeId>))` representing the path from start to end, or `Ok(None)` if no valid path exists within the `max_depth`.
159    ///
160    /// # Examples
161    ///
162    /// ```no_run
163    /// use aletheiadb::db::AletheiaDB;
164    /// use aletheiadb::core::id::NodeId;
165    /// use aletheiadb::query::semantic_pathfinding::SemanticPathfinder;
166    ///
167    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
168    /// let db = AletheiaDB::new()?;
169    /// let pathfinder = SemanticPathfinder::new(&db, "embedding");
170    ///
171    /// let start = NodeId::new(1).unwrap();
172    /// let end = NodeId::new(2).unwrap();
173    /// let concept = vec![0.5, 0.5, 0.0];
174    ///
175    /// if let Some(path) = pathfinder.find_path(start, end, &concept, 5, true)? {
176    ///     println!("Found semantic path: {:?}", path);
177    /// }
178    /// # Ok(())
179    /// # }
180    /// ```
181    ///
182    /// # Errors
183    /// Returns an error if an underlying database operation fails.
184    pub fn find_path(
185        &self,
186        start: NodeId,
187        end: NodeId,
188        query_embedding: &[f32],
189        max_depth: usize,
190        bidirectional: bool,
191    ) -> Result<Option<Vec<NodeId>>> {
192        let mut pq = BinaryHeap::new();
193        // ⚡ Bolt: Use `IdentityHasher` instead of the default SipHash.
194        // `NodeId` is already a unique wrapper over `u64`. In pathfinding (like A*),
195        // we perform thousands of map insertions and lookups. Bypassing SipHash eliminates
196        // unnecessary hashing overhead on already-unique integer keys.
197        let mut dist: HashMap<NodeId, f32, BuildHasherDefault<IdentityHasher>> = HashMap::default();
198        let mut came_from: HashMap<NodeId, NodeId, BuildHasherDefault<IdentityHasher>> =
199            HashMap::default();
200
201        // Initialize start node
202        dist.insert(start, 0.0);
203        pq.push(State {
204            cost: 0.0,
205            node: start,
206            depth: 0,
207        });
208
209        // Reuse allocation across iterations
210        // ⚡ Bolt: Pre-allocate neighbor vector with typical branching factor (32) to avoid standard 0->4->8->16->32 heap reallocation chain.
211        let mut neighbors = Vec::with_capacity(32);
212
213        while let Some(State { cost, node, depth }) = pq.pop() {
214            if node == end {
215                return Ok(Some(self.reconstruct_path(came_from, end)));
216            }
217
218            // Optimization: Skip if we found a better path already
219            #[allow(clippy::collapsible_if)]
220            if let Some(&d) = dist.get(&node) {
221                if cost > d {
222                    continue;
223                }
224            }
225
226            // Check depth limit
227            if depth >= max_depth {
228                continue;
229            }
230
231            // Collect neighbors (outgoing, or both directions if bidirectional)
232            neighbors.clear();
233
234            // Get outgoing edges
235            for edge_id in self.db.get_outgoing_edges(node) {
236                if let Ok(target) = self.db.get_edge_target(edge_id) {
237                    neighbors.push(target);
238                }
239            }
240
241            // Also get incoming edges if bidirectional
242            if bidirectional {
243                for edge_id in self.db.get_incoming_edges(node) {
244                    if let Ok(edge) = self.db.get_edge(edge_id) {
245                        neighbors.push(edge.source);
246                    }
247                }
248            }
249
250            // Process all neighbors
251            for &target in &neighbors {
252                // Calculate semantic cost of moving to target
253                let semantic_cost = self.calculate_semantic_cost(target, query_embedding)?;
254
255                let new_cost = cost + semantic_cost + STRUCTURAL_HOP_COST;
256
257                if new_cost < *dist.get(&target).unwrap_or(&f32::INFINITY) {
258                    dist.insert(target, new_cost);
259                    came_from.insert(target, node);
260                    pq.push(State {
261                        cost: new_cost,
262                        node: target,
263                        depth: depth + 1,
264                    });
265                }
266            }
267        }
268
269        Ok(None)
270    }

272    /// Finds a path at a specific point in time.
273    ///
274    /// Similar to [`find_path`](Self::find_path), but only considers nodes and edges valid at the specified `time`.
275    /// Uses the temporal adjacency index to find edges that existed at the specified
276    /// time, even if they have since been deleted from current storage.
277    ///
278    /// # Arguments
279    /// * `start` - Starting [`NodeId`].
280    /// * `end` - Target [`NodeId`].
281    /// * `query_embedding` - Vector representing the semantic concept to follow.
282    /// * `time` - The [`Timestamp`] at which to evaluate the graph state.
283    /// * `max_depth` - Maximum path length (to prevent infinite loops in large graphs).
284    /// * `bidirectional` - If `true`, traverses both outgoing and incoming edges.
285    ///
286    /// # Returns
287    /// An `Ok(Some(Vec<NodeId>))` representing the path from start to end as it existed at `time`, or `Ok(None)` if no valid path existed within the `max_depth`.
288    ///
289    /// # Examples
290    ///
291    /// ```no_run
292    /// use aletheiadb::db::AletheiaDB;
293    /// use aletheiadb::core::id::NodeId;
294    /// use aletheiadb::core::temporal::Timestamp;
295    /// use aletheiadb::query::semantic_pathfinding::SemanticPathfinder;
296    ///
297    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
298    /// let db = AletheiaDB::new()?;
299    /// let pathfinder = SemanticPathfinder::new(&db, "embedding");
300    ///
301    /// let start = NodeId::new(1).unwrap();
302    /// let end = NodeId::new(2).unwrap();
303    /// let concept = vec![0.5, 0.5, 0.0];
304    /// let historical_time: Timestamp = 1622505600.into();
305    ///
306    /// if let Some(path) = pathfinder.find_path_at_time(start, end, &concept, historical_time, 5, false)? {
307    ///     println!("Found historical semantic path: {:?}", path);
308    /// }
309    /// # Ok(())
310    /// # }
311    /// ```
312    ///
313    /// # Errors
314    /// Returns an error if an underlying database operation fails.
315    pub fn find_path_at_time(
316        &self,
317        start: NodeId,
318        end: NodeId,
319        query_embedding: &[f32],
320        time: Timestamp,
321        max_depth: usize,
322        bidirectional: bool,
323    ) -> Result<Option<Vec<NodeId>>> {
324        let mut pq = BinaryHeap::new();
325        // ⚡ Bolt: Use `IdentityHasher` instead of the default SipHash.
326        // `NodeId` is already a unique wrapper over `u64`. In pathfinding (like A*),
327        // we perform thousands of map insertions and lookups. Bypassing SipHash eliminates
328        // unnecessary hashing overhead on already-unique integer keys.
329        let mut dist: HashMap<NodeId, f32, BuildHasherDefault<IdentityHasher>> = HashMap::default();
330        let mut came_from: HashMap<NodeId, NodeId, BuildHasherDefault<IdentityHasher>> =
331            HashMap::default();
332
333        // Verify start node existed
334        if self.db.get_node_at_time(start, time, time).is_err() {
335            return Ok(None);
336        }
337
338        dist.insert(start, 0.0);
339        pq.push(State {
340            cost: 0.0,
341            node: start,
342            depth: 0,
343        });
344
345        // Reuse allocation across iterations
346        // ⚡ Bolt: Pre-allocate neighbor vector with typical branching factor (32) to avoid standard 0->4->8->16->32 heap reallocation chain.
347        let mut neighbor_edges = Vec::with_capacity(32);
348
349        while let Some(State { cost, node, depth }) = pq.pop() {
350            if node == end {
351                return Ok(Some(self.reconstruct_path(came_from, end)));
352            }
353
354            #[allow(clippy::collapsible_if)]
355            if let Some(&d) = dist.get(&node) {
356                if cost > d {
357                    continue;
358                }
359            }
360
361            // Check depth limit to prevent infinite loops
362            if depth >= max_depth {
363                continue;
364            }
365
366            // Collect neighbors (outgoing, or both directions if bidirectional)
367            neighbor_edges.clear();
368
369            // Get outgoing edges at the specified time
370            for edge_id in self.db.get_outgoing_edges_at_time(node, time, time) {
371                neighbor_edges.push((edge_id, true)); // true = outgoing
372            }
373
374            // Also get incoming edges if bidirectional
375            if bidirectional {
376                for edge_id in self.db.get_incoming_edges_at_time(node, time, time) {
377                    neighbor_edges.push((edge_id, false)); // false = incoming
378                }
379            }
380
381            for &(edge_id, is_outgoing) in &neighbor_edges {
382                // Get edge details at the specified time
383                if let Ok(edge) = self.db.get_edge_at_time(edge_id, time, time) {
384                    // For outgoing edges, target is the neighbor; for incoming, source is the neighbor
385                    let target = if is_outgoing {
386                        edge.target
387                    } else {
388                        edge.source
389                    };
390
391                    // Check if target node existed at time T (and get embedding)
392                    if let Ok(target_node) = self.db.get_node_at_time(target, time, time) {
393                        // Calculate semantic cost using historical embedding
394                        let target_embedding = target_node
395                            .properties
396                            .get(&self.vector_property)
397                            .and_then(|v| v.as_vector());
398
399                        let semantic_cost =
400                            self.compute_semantic_cost(target_embedding, query_embedding)?;
401
402                        let new_cost = cost + semantic_cost + STRUCTURAL_HOP_COST;
403
404                        if new_cost < *dist.get(&target).unwrap_or(&f32::INFINITY) {
405                            dist.insert(target, new_cost);
406                            came_from.insert(target, node);
407                            pq.push(State {
408                                cost: new_cost,
409                                node: target,
410                                depth: depth + 1,
411                            });
412                        }
413                    }
414                }
415            }
416        }
417
418        Ok(None)
419    }

421    /// Calculate cost based on semantic similarity (1.0 - similarity).
422    /// Returns 1.0 (max cost) if node has no embedding.
423    fn calculate_semantic_cost(&self, node_id: NodeId, query: &[f32]) -> Result<f32> {
424        let node = self.db.get_node(node_id)?;
425        let embedding = node
426            .properties
427            .get(&self.vector_property)
428            .and_then(|v| v.as_vector());
429
430        self.compute_semantic_cost(embedding, query)
431    }
432
433    /// Helper to compute semantic cost from an embedding slice.
434    ///
435    /// Handles dimension mismatches gracefully by returning infinite cost.
436    fn compute_semantic_cost(&self, embedding: Option<&[f32]>, query: &[f32]) -> Result<f32> {
437        if let Some(emb) = embedding {
438            match cosine_similarity(emb, query) {
439                Ok(sim) => {
440                    // Clamp to [0, 2] (cosine sim is [-1, 1])
441                    // We want high similarity -> low cost
442                    // 1.0 - 1.0 = 0.0 (perfect match)
443                    // 1.0 - (-1.0) = 2.0 (opposite)
444                    Ok(1.0 - sim)
445                }
446                Err(crate::core::error::Error::Vector(
447                    crate::core::error::VectorError::DimensionMismatch { .. },
448                )) => {
449                    // Sentry 🛡️: Dimension mismatch implies incompatibility.
450                    // Return infinite cost to strictly avoid this node unless no other path exists.
451                    // This prevents the entire search from failing due to one malformed node.
452                    Ok(f32::INFINITY)
453                }
454                Err(e) => Err(e),
455            }
456        } else {
457            // Penalize nodes without embeddings
458            Ok(NO_EMBEDDING_COST)
459        }
460    }
461
462    fn reconstruct_path(
463        &self,
464        came_from: HashMap<NodeId, NodeId, BuildHasherDefault<IdentityHasher>>,
465        current: NodeId,
466    ) -> Vec<NodeId> {
467        let mut path = vec![current];
468        let mut curr = current;
469        while let Some(&prev) = came_from.get(&curr) {
470            path.push(prev);
471            curr = prev;
472        }
473        path.reverse();
474        path
475    }

575    #[test]
576    fn test_semantic_pathfinding_time_travel() {
577        use crate::core::temporal::time;
578
579        let db = create_test_db();
580        let _now = time::now();
581
582        // Create nodes
583        let start = db
584            .create_node("Start", PropertyMapBuilder::new().build())
585            .unwrap();
586        let middle = db
587            .create_node(
588                "Middle",
589                PropertyMapBuilder::new()
590                    .insert_vector("embedding", &[1.0, 0.0, 0.0])
591                    .build(),
592            )
593            .unwrap();
594        let end = db
595            .create_node("End", PropertyMapBuilder::new().build())
596            .unwrap();
597
598        // Create edges at t0
599        // Use write_with_timestamp to ensure t0 covers the creation
600        let (_, t_edges) = db
601            .write_with_timestamp(|tx| {
602                tx.create_edge(start, middle, "NEXT", PropertyMapBuilder::new().build())?;
603                tx.create_edge(middle, end, "NEXT", PropertyMapBuilder::new().build())?;
604                Ok::<_, Error>(())
605            })
606            .unwrap();
607
608        let t0 = t_edges;
609
610        let query = vec![1.0, 0.0, 0.0];
611        let pathfinder = SemanticPathfinder::new(&db, "embedding");
612
613        // Query at t0: Path should exist (BEFORE DELETION)
614        let path_t0 = pathfinder
615            .find_path_at_time(start, end, &query, t0, 10, false)
616            .unwrap();
617        assert!(path_t0.is_some(), "Path should exist at t0 before deletion");
618
619        // Delete "Middle" node at t1 (which should break the path)
620        // Use delete_node_cascade to ensure edges are also deleted from current storage
621        let (_, t_delete) = db
622            .write_with_timestamp(|tx| tx.delete_node_cascade(middle))
623            .unwrap();
624        let _t1 = t_delete;
625
626        // Verify time monotonicity (HLC guarantees distinct timestamps)
627        assert!(
628            t_delete > t0,
629            "Time must advance monotonically for subsequent transactions"
630        );
631
632        // Query at t0 AFTER deletion: With temporal adjacency index (enabled by default),
633        // the path SHOULD be found even though edges are deleted from current storage
634        let path_t0_after_delete = pathfinder
635            .find_path_at_time(start, end, &query, t0, 10, false)
636            .unwrap();
637        assert!(
638            path_t0_after_delete.is_some(),
639            "Temporal adjacency index (enabled by default) should find path through deleted edges"
640        );
641        let path = path_t0_after_delete.unwrap();
642        assert_eq!(path.len(), 3);
643        assert_eq!(path[0], start);
644        assert_eq!(path[1], middle);
645        assert_eq!(path[2], end);
646
647        // Test "Future Path" scenario: Path exists now but didn't in past
648
649        let new_middle = db
650            .create_node(
651                "NewMiddle",
652                PropertyMapBuilder::new()
653                    .insert_vector("embedding", &[1.0, 0.0, 0.0])
654                    .build(),
655            )
656            .unwrap();
657
658        let (_, t_new_edges) = db
659            .write_with_timestamp(|tx| {
660                tx.create_edge(start, new_middle, "NEXT", PropertyMapBuilder::new().build())?;
661                tx.create_edge(new_middle, end, "NEXT", PropertyMapBuilder::new().build())?;
662                Ok::<_, Error>(())
663            })
664            .unwrap();
665
666        let t2 = t_new_edges;
667
668        // Path exists at t2
669        let path_t2 = pathfinder
670            .find_path_at_time(start, end, &query, t2, 10, false)
671            .unwrap();
672        assert!(path_t2.is_some(), "Path should exist at t2");
673
674        // Query at t0 again: Should find the ORIGINAL path through middle
675        // (not the new path through new_middle which was created at t2)
676        // With temporal adjacency index, deleted edges are still accessible
677        // when querying at times before they were deleted.
678        let path_t0_check = pathfinder
679            .find_path_at_time(start, end, &query, t0, 10, false)
680            .unwrap();
681        assert!(
682            path_t0_check.is_some(),
683            "Should find original path at t0 (through middle, not new_middle)"
684        );
685        // Verify it's the original middle node, not new_middle
686        assert_eq!(path_t0_check.as_ref().unwrap()[1], middle);

834        #[test]
835        fn test_pathfinding_skips_incompatible_dimensions() {
836            // 🛡️ Sentry Test: Mixed dimensions should not crash pathfinding.
837            // Setup:
838            // Start (3D) -> Broken (4D) -> End (3D)
839            //            -> Valid (3D)  -> End (3D)
840            //
841            // Pathfinding should navigate around Broken and use Valid.
842
843            let db = create_test_db();
844
845            // Nodes
846            let start = db
847                .create_node(
848                    "Start",
849                    PropertyMapBuilder::new()
850                        .insert_vector("embedding", &[1.0, 0.0, 0.0])
851                        .build(),
852                )
853                .unwrap();
854
855            let end = db
856                .create_node(
857                    "End",
858                    PropertyMapBuilder::new()
859                        .insert_vector("embedding", &[0.0, 0.0, 1.0])
860                        .build(),
861                )
862                .unwrap();
863
864            // Create nodes with different property name for "broken" 4D vector
865            // to bypass potential index validation during creation.
866            // We will tell pathfinder to use "embedding_mixed".
867
868            let broken = db
869                .create_node(
870                    "Broken",
871                    PropertyMapBuilder::new()
872                        .insert_vector("embedding_mixed", &[0.5, 0.5, 0.5, 0.5])
873                        .build(),
874                )
875                .unwrap();
876
877            let valid = db
878                .create_node(
879                    "Valid",
880                    PropertyMapBuilder::new()
881                        .insert_vector("embedding_mixed", &[0.5, 0.5, 0.0])
882                        .build(),
883                )
884                .unwrap();
885
886            // Update Start and End to also use "embedding_mixed" using explicit transaction
887            db.write(|tx| {
888                tx.update_node(
889                    start,
890                    PropertyMapBuilder::new()
891                        .insert_vector("embedding_mixed", &[1.0, 0.0, 0.0])
892                        .build(),
893                )
894            })
895            .unwrap();
896            db.write(|tx| {
897                tx.update_node(
898                    end,
899                    PropertyMapBuilder::new()
900                        .insert_vector("embedding_mixed", &[0.0, 0.0, 1.0])
901                        .build(),
902                )
903            })
904            .unwrap();
905
906            // Connect
907            db.create_edge(start, broken, "NEXT", PropertyMapBuilder::new().build())
908                .unwrap();
909            db.create_edge(broken, end, "NEXT", PropertyMapBuilder::new().build())
910                .unwrap();
911
912            db.create_edge(start, valid, "NEXT", PropertyMapBuilder::new().build())
913                .unwrap();
914            db.create_edge(valid, end, "NEXT", PropertyMapBuilder::new().build())
915                .unwrap();
916
917            // Pathfinding
918            let query = vec![1.0, 0.0, 0.0];
919            let pathfinder = SemanticPathfinder::new(&db, "embedding_mixed");
920
921            let result = pathfinder.find_path(start, end, &query, 10, false);
922
923            match result {
924                Ok(Some(p)) => {
925                    // If it succeeds, verify it took the valid path
926                    assert_eq!(p, vec![start, valid, end], "Should take valid path");
927                }
928                Ok(None) => panic!("Should find a path (returned None)"),
929                Err(e) => {
930                    panic!(
931                        "Regression: Dimension mismatch error was not suppressed. Expected successful pathfinding skipping invalid node. Error: {}",
932                        e
933                    );
934                }
935            }
936        }
