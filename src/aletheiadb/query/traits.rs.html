1//! Defines core traits used throughout the query execution engine.
2
3use crate::core::error::Result;
4use crate::core::graph::{Edge, Node};
5use crate::core::id::{EdgeId, NodeId};
6use crate::core::temporal::Timestamp;
7
8/// A trait representing a read-only view of the graph.
9///
10/// This trait abstracts the underlying database implementation, allowing
11/// query algorithms to operate on any source that provides graph access.
12/// It effectively decouples the query engine from the storage engine.
13pub trait GraphView {
14    /// Get a node by ID.
15    fn get_node(&self, id: NodeId) -> Result<Node>;
16
17    /// Get an edge by ID.
18    fn get_edge(&self, id: EdgeId) -> Result<Edge>;
19
20    /// Get the target node ID of an edge.
21    fn get_edge_target(&self, edge_id: EdgeId) -> Result<NodeId>;
22
23    /// Get the source node ID of an edge.
24    fn get_edge_source(&self, edge_id: EdgeId) -> Result<NodeId>;
25
26    /// Get outgoing edges from a node.
27    fn get_outgoing_edges(&self, node_id: NodeId) -> Vec<EdgeId>;
28
29    /// Get incoming edges to a node.
30    fn get_incoming_edges(&self, node_id: NodeId) -> Vec<EdgeId>;
31
32    /// Get outgoing edges with a specific label.
33    fn get_outgoing_edges_with_label(&self, node_id: NodeId, label: &str) -> Vec<EdgeId>;
34
35    // Temporal methods
36
37    /// Get a node as it existed at a specific time.
38    fn get_node_at_time(
39        &self,
40        node_id: NodeId,
41        valid_time: Timestamp,
42        transaction_time: Timestamp,
43    ) -> Result<Node>;
44
45    /// Get an edge as it existed at a specific time.
46    fn get_edge_at_time(
47        &self,
48        edge_id: EdgeId,
49        valid_time: Timestamp,
50        transaction_time: Timestamp,
51    ) -> Result<Edge>;
52
53    /// Get outgoing edges valid at a specific time.
54    fn get_outgoing_edges_at_time(
55        &self,
56        source: NodeId,
57        valid_time: Timestamp,
58        tx_time: Timestamp,
59    ) -> Vec<EdgeId>;
60
61    /// Get incoming edges valid at a specific time.
62    fn get_incoming_edges_at_time(
63        &self,
64        target: NodeId,
65        valid_time: Timestamp,
66        tx_time: Timestamp,
67    ) -> Vec<EdgeId>;
68
69    // Vector methods
70
71    /// Find k most similar nodes at a specific point in time.
72    ///
73    /// With multiple temporal vector indexes enabled and no property specified,
74    /// implementations query the alphabetically first indexed property; prefer
75    /// `AletheiaDB::find_similar_as_of_in` with an explicit property when more
76    /// than one temporal index exists.
77    fn find_similar_as_of(
78        &self,
79        embedding: &[f32],
80        k: usize,
81        timestamp: Timestamp,
82    ) -> Result<Vec<(NodeId, f32)>>;
83}
