1//! Logical Query Plan
2//!
3//! Represents the structure of a query as a tree of logical operations.
4//! The logical plan is optimized and then converted to a physical plan
5//! for execution.

16/// A logical query plan represented as a tree of operations.
17#[derive(Debug, Clone, PartialEq)]
18pub struct LogicalPlan {
19    /// Root operation of the plan tree
20    pub root: LogicalOp,
21    /// Temporal context for the entire query (if any)
22    pub temporal_context: Option<TemporalContext>,
23    /// Hints for query optimization
24    pub hints: QueryHints,
25}
26
27impl LogicalPlan {
28    /// Create a new logical plan with the given root operation
29    #[must_use]
30    pub fn new(root: LogicalOp) -> Self {
31        LogicalPlan {
32            root,
33            temporal_context: None,
34            hints: QueryHints::default(),
35        }
36    }
37
38    /// Add temporal context to the plan
39    #[must_use]
40    pub fn with_temporal_context(mut self, context: TemporalContext) -> Self {
41        self.temporal_context = Some(context);
42        self
43    }
44
45    /// Add query hints
46    #[must_use]
47    pub fn with_hints(mut self, hints: QueryHints) -> Self {
48        self.hints = hints;
49        self
50    }
51
52    /// Check if this query requires temporal storage access
53    #[must_use]
54    pub fn is_temporal(&self) -> bool {
55        self.temporal_context.is_some()
56    }
57
58    /// Check if this query involves vector operations
59    #[must_use]
60    pub fn has_vector_ops(&self) -> bool {
61        self.root.has_vector_ops()
62    }
63
64    /// Check if this query involves graph traversal
65    #[must_use]
66    pub fn has_traversal(&self) -> bool {
67        self.root.has_traversal()
68    }
69}

71/// Logical operation nodes in the query plan tree.
72#[derive(Debug, Clone, PartialEq)]
73pub enum LogicalOp {
74    /// Leaf operation: data source (scan, lookup, search)
75    Scan(ScanOp),
76
77    /// Unary operation: transforms a single input
78    Unary {
79        /// The operation to apply
80        op: UnaryOp,
81        /// Input to this operation
82        input: Box<LogicalOp>,
83    },
84
85    /// Binary operation: combines two inputs
86    Binary {
87        /// The operation to apply
88        op: BinaryOp,
89        /// Left input
90        left: Box<LogicalOp>,
91        /// Right input
92        right: Box<LogicalOp>,
93    },
94
95    /// Empty result (used for optimization)
96    Empty,
97}
98
99impl LogicalOp {
100    /// Create a unary operation
101    #[must_use]
102    pub fn unary(op: UnaryOp, input: LogicalOp) -> Self {
103        LogicalOp::Unary {
104            op,
105            input: Box::new(input),
106        }
107    }
108
109    /// Create a binary operation
110    #[must_use]
111    pub fn binary(op: BinaryOp, left: LogicalOp, right: LogicalOp) -> Self {
112        LogicalOp::Binary {
113            op,
114            left: Box::new(left),
115            right: Box::new(right),
116        }
117    }
118
119    /// Check if this operation or its children involve vector operations
120    #[must_use]
121    pub fn has_vector_ops(&self) -> bool {
122        match self {
123            LogicalOp::Scan(scan) => matches!(scan, ScanOp::VectorSearch { .. }),
124            LogicalOp::Unary { op, input } => {
125                matches!(op, UnaryOp::VectorRank { .. }) || input.has_vector_ops()
126            }
127            LogicalOp::Binary { left, right, .. } => {
128                left.has_vector_ops() || right.has_vector_ops()
129            }
130            LogicalOp::Empty => false,
131        }
132    }
133
134    /// Check if this operation or its children involve graph traversal
135    #[must_use]
136    pub fn has_traversal(&self) -> bool {
137        match self {
138            LogicalOp::Scan(_) => false,
139            LogicalOp::Unary { op, input } => {
140                matches!(op, UnaryOp::Traverse { .. }) || input.has_traversal()
141            }
142            LogicalOp::Binary { left, right, .. } => left.has_traversal() || right.has_traversal(),
143            LogicalOp::Empty => false,
144        }
145    }
146}

148/// Scan operations (leaf nodes in the plan tree).
149///
150/// These represent data sources - the starting points for query execution.
151#[derive(Debug, Clone, PartialEq)]
152pub enum ScanOp {
153    /// Direct node lookup by ID(s) - O(1) per node
154    NodeLookup(Vec<NodeId>),
155
156    /// Full node scan with optional label filter
157    NodeScan {
158        /// Optional label to filter by
159        label: Option<String>,
160        /// Estimated number of rows (for cost estimation)
161        estimated_rows: Option<usize>,
162    },
163
164    /// Vector k-NN search using HNSW index
165    VectorSearch {
166        /// Query embedding
167        embedding: Arc<[f32]>,
168        /// Number of results to return
169        k: usize,
170        /// Optional label filter
171        label_filter: Option<String>,
172        /// Distance metric
173        metric: DistanceMetric,
174        /// Property key containing the embedding (None = "embedding")
175        property_key: Option<String>,
176    },
177
178    /// Temporal point-in-time node lookup
179    TemporalNodeLookup {
180        /// Node IDs to look up
181        node_ids: Vec<NodeId>,
182        /// Valid time for the query
183        valid_time: Timestamp,
184        /// Transaction time for the query
185        transaction_time: Timestamp,
186    },
187
188    /// Temporal vector search using historical snapshots
189    TemporalVectorSearch {
190        /// Query embedding
191        embedding: Arc<[f32]>,
192        /// Number of results to return
193        k: usize,
194        /// Timestamp for the historical query
195        timestamp: Timestamp,
196        /// Property key containing the embedding vector
197        /// If None, uses the default/first indexed property
198        property_key: Option<String>,
199    },
200    /// Find nodes similar to a specific node by extracting its embedding
201    SimilarToNode {
202        /// Source node whose embedding to use
203        source_node: NodeId,
204        /// Property key containing the embedding vector
205        property_key: String,
206        /// Number of results to return
207        k: usize,
208        /// Optional label filter for results
209        label_filter: Option<String>,
210    },
211
212    /// Full edge scan with optional edge type filter
213    EdgeScan {
214        /// Optional edge type to filter by (e.g., "KNOWS", "FOLLOWS")
215        edge_type: Option<String>,
216        /// Estimated number of rows (for cost estimation)
217        estimated_rows: Option<usize>,
218    },
219
220    /// Property-based node scan: nodes with a given label where property == value.
221    ///
222    /// This is the fused form of `NodeScan { label } + Filter(Eq { key, value })`,
223    /// produced by the `FilterScanFusion` optimization rule. It delegates to
224    /// `CurrentStorage::find_nodes_by_property` which avoids per-node predicate
225    /// evaluation on non-matching nodes.
226    PropertyScan {
227        /// Label to filter by
228        label: String,
229        /// Property key to match
230        key: String,
231        /// Expected property value
232        value: super::ir::PredicateValue,
233    },
234}

236/// Unary operations that transform a single input.
237#[derive(Debug, Clone, PartialEq)]
238pub enum UnaryOp {
239    /// Filter rows by predicate
240    Filter(Predicate),
241
242    /// Project specific properties
243    Project(Vec<String>),
244
245    /// Project provenance accessors as output columns (Issue #3354). Mirrors
246    /// [`super::ir::QueryOp::ProjectProvenance`].
247    ProjectProvenance(super::ir::ProvenanceProjection),
248
249    /// Limit number of results
250    Limit(usize),
251
252    /// Skip a number of results
253    Skip(usize),
254
255    /// Graph traversal
256    Traverse {
257        /// Traversal direction
258        direction: Direction,
259        /// Optional edge label filter
260        label: Option<String>,
261        /// Depth specification
262        depth: TraversalDepth,
263    },
264
265    /// Rank by vector similarity
266    VectorRank {
267        /// Target embedding for comparison
268        embedding: Arc<[f32]>,
269        /// Optional limit on ranked results
270        top_k: Option<usize>,
271        /// Property key containing the embedding (None = "embedding")
272        property_key: Option<String>,
273        /// Distance/similarity metric selected by the calling surface.
274        metric: VectorMetric,
275        /// Optional similarity-score threshold (keeps only passing rows).
276        threshold: Option<ScoreThreshold>,
277        /// Optional output-column alias for the materialized similarity score.
278        score_alias: Option<String>,
279    },
280
281    /// Sort by property or score
282    Sort {
283        /// Sort key
284        key: SortKey,
285        /// Sort order (true = descending)
286        descending: bool,
287    },
288
289    /// Track temporal changes
290    TemporalTrack {
291        /// Time range to track within
292        time_range: TimeRange,
293    },
294
295    /// Distinct/deduplicate results
296    Distinct,
297
298    /// Count results (aggregate)
299    Count,
300
301    /// Grouped aggregation (openCypher implicit grouping): partition by
302    /// `group_keys` and compute `aggregates` per group. Mirrors
303    /// [`super::ir::QueryOp::Aggregate`].
304    Aggregate {
305        /// Grouping keys (empty = single global group).
306        group_keys: Vec<super::ir::AggregateGroupKey>,
307        /// Aggregate expressions computed per group.
308        aggregates: Vec<super::ir::AggregateSpec>,
309    },
310
311    /// Temporal aggregation window (Issue #3363). Mirrors
312    /// [`super::ir::QueryOp::TemporalWindowAggregate`]: buckets each upstream
313    /// entity's valid-time history into tumbling windows and emits per-window
314    /// aggregate rows.
315    TemporalWindowAggregate(super::ir::TemporalWindowSpec),
316
317    /// Temporal join / align (Issue #3379). Mirrors
318    /// [`super::ir::QueryOp::TemporalAlign`]: aligns the upstream matched
319    /// participants at matching valid-time coordinates and emits per-coordinate
320    /// computed-column rows.
321    TemporalAlign(super::ir::TemporalAlignSpec),
322
323    /// Left-outer application of an optional sub-pattern (`OPTIONAL MATCH`).
324    ///
325    /// For each input row, the `steps` sub-pipeline runs seeded from that row;
326    /// if it produces nothing, a single null-bound row is emitted instead, so
327    /// the input row is never lost. When the first step is
328    /// [`OptionalStep::Scan`] the operator is *standalone* (a leading
329    /// `OPTIONAL MATCH`): the sub-pipeline runs once from its own scan and
330    /// the input (typically [`LogicalOp::Empty`]) is ignored.
331    ///
332    /// The steps are deliberately opaque to optimization rules: filters
333    /// inside the optional segment participate in the matched/unmatched
334    /// decision and must not be reordered across this boundary.
335    OptionalApply {
336        /// The optional sub-pipeline, applied per input row.
337        steps: Vec<OptionalStep>,
338    },
339}
340
341/// A single step inside an [`UnaryOp::OptionalApply`] sub-pipeline.
342#[derive(Debug, Clone, PartialEq)]
343pub enum OptionalStep {
344    /// Scan source for a standalone (first-clause) `OPTIONAL MATCH`.
345    /// Only valid as the first step.
346    Scan {
347        /// Optional node label filter.
348        label: Option<String>,
349    },
350
351    /// A traversal hop inside the optional pattern.
352    Traverse {
353        /// Traversal direction.
354        direction: Direction,
355        /// Optional edge label filter.
356        label: Option<String>,
357        /// Depth specification.
358        depth: TraversalDepth,
359    },
360
361    /// A predicate filter inside the optional pattern (inline pattern
362    /// properties or the clause's `WHERE`).
363    Filter(Predicate),
364}

390/// Temporal context for a query.
391///
392/// Specifies when to evaluate the query in time.
393/// Temporal query context supporting independent valid_time and transaction_time dimensions.
394///
395/// Enables true bi-temporal queries where each dimension can be queried independently:
396/// - "What did we know at time T?" → `as_of_transaction_time(T)`
397/// - "What was true at time T?" → `as_of_valid_time(T)`
398/// - "What did we know at time T about what was true at time V?" → `as_of(V, T)`
399#[derive(Debug, Clone, Default, PartialEq)]
400pub struct TemporalContext {
401    /// Point-in-time query for valid_time dimension
402    pub valid_time_as_of: Option<Timestamp>,
403    /// Point-in-time query for transaction_time dimension
404    pub transaction_time_as_of: Option<Timestamp>,
405    /// Time range query for valid_time dimension
406    pub valid_time_between: Option<TimeRange>,
407    /// Time range query for transaction_time dimension
408    pub transaction_time_between: Option<TimeRange>,
409    /// Include historical versions in results (for history queries)
410    pub include_history: bool,
411}
412
413impl TemporalContext {
414    /// Create a point-in-time context for both dimensions (traditional bi-temporal query).
415    ///
416    /// # Example
417    /// ```ignore
418    /// // "What did we know at tx_time about what was valid at valid_time?"
419    /// let ctx = TemporalContext::as_of(valid_time, tx_time);
420    /// ```
421    #[must_use]
422    pub fn as_of(valid_time: Timestamp, transaction_time: Timestamp) -> Self {
423        TemporalContext {
424            valid_time_as_of: Some(valid_time),
425            transaction_time_as_of: Some(transaction_time),
426            valid_time_between: None,
427            transaction_time_between: None,
428            include_history: false,
429        }
430    }
431
432    /// Create a context querying only the valid_time dimension.
433    ///
434    /// Transaction time defaults to "now" (most recent recorded state).
435    ///
436    /// # Example
437    /// ```ignore
438    /// // "What was valid/true at this time?" (using current database state)
439    /// let ctx = TemporalContext::as_of_valid_time(timestamp);
440    /// ```
441    #[must_use]
442    pub fn as_of_valid_time(timestamp: Timestamp) -> Self {
443        TemporalContext {
444            valid_time_as_of: Some(timestamp),
445            transaction_time_as_of: None,
446            valid_time_between: None,
447            transaction_time_between: None,
448            include_history: false,
449        }
450    }
451
452    /// Create a context querying only the transaction_time dimension.
453    ///
454    /// Valid time defaults to "now" (current validity).
455    ///
456    /// # Example
457    /// ```ignore
458    /// // "What did we know/record at this time?" (database state as of tx_time)
459    /// let ctx = TemporalContext::as_of_transaction_time(timestamp);
460    /// ```
461    #[must_use]
462    pub fn as_of_transaction_time(timestamp: Timestamp) -> Self {
463        TemporalContext {
464            valid_time_as_of: None,
465            transaction_time_as_of: Some(timestamp),
466            valid_time_between: None,
467            transaction_time_between: None,
468            include_history: false,
469        }
470    }
471
472    /// Create a context for querying a valid_time range.
473    ///
474    /// # Example
475    /// ```ignore
476    /// // "What was valid between start and end?"
477    /// let ctx = TemporalContext::valid_time_between(range);
478    /// ```
479    #[must_use]
480    pub fn valid_time_between(range: TimeRange) -> Self {
481        TemporalContext {
482            valid_time_as_of: None,
483            transaction_time_as_of: None,
484            valid_time_between: Some(range),
485            transaction_time_between: None,
486            include_history: false,
487        }
488    }
489
490    /// Create a context for querying a transaction_time range.
491    ///
492    /// # Example
493    /// ```ignore
494    /// // "What did we record between start and end?"
495    /// let ctx = TemporalContext::transaction_time_between(range);
496    /// ```
497    #[must_use]
498    pub fn transaction_time_between(range: TimeRange) -> Self {
499        TemporalContext {
500            valid_time_as_of: None,
501            transaction_time_as_of: None,
502            valid_time_between: None,
503            transaction_time_between: Some(range),
504            include_history: false,
505        }
506    }
507
508    /// Create a time range context (backward compatibility - uses valid_time dimension).
509    ///
510    /// **Deprecated**: Use `valid_time_between()` or `transaction_time_between()` for clarity.
511    /// Sets the temporal context to query data valid between the specified `TimeRange`.
512    /// Creates a query step that filters elements valid between two timestamps.
513    ///
514    /// # Why?
515    /// This was the original API for basic temporal filtering. It has been superseded
516    /// by more explicit methods that distinguish between valid time and transaction time.
517    #[must_use]
518    #[deprecated(
519        since = "0.1.0",
520        note = "Use valid_time_between() or transaction_time_between() instead"
521    )]
522    pub fn between(range: TimeRange) -> Self {
523        Self::valid_time_between(range)
524    }
525
526    /// Set whether to include historical versions in results.
527    ///
528    /// When enabled, queries return all versions within the temporal context
529    /// rather than just the current state.
530    #[must_use]
531    pub fn with_history(mut self, include: bool) -> Self {
532        self.include_history = include;
533        self
534    }
535
536    /// Resolve temporal dimensions by filling missing dimensions with current time.
537    ///
538    /// Returns `(valid_time, transaction_time)` where:
539    /// - Missing `valid_time_as_of` → `time::now()`
540    /// - Missing `transaction_time_as_of` → `time::now()`
541    ///
542    /// # Example
543    /// ```ignore
544    /// let ctx = TemporalContext::as_of_valid_time(ts);
545    /// let (vt, tt) = ctx.resolve_now();
546    /// // vt == ts, tt == now()
547    /// ```
548    #[must_use]
549    pub fn resolve_now(&self) -> (Timestamp, Timestamp) {
550        use crate::core::temporal::time;
551
552        let now = time::now();
553        let valid_time = self.valid_time_as_of.unwrap_or(now);
554        let transaction_time = self.transaction_time_as_of.unwrap_or(now);
555
556        (valid_time, transaction_time)
557    }
558
559    // =========================================================================
560    // Backward Compatibility Helpers
561    // =========================================================================
562
563    /// Get point-in-time timestamps as a tuple (backward compatibility).
564    ///
565    /// Returns `Some((valid_time, transaction_time))` only if both dimensions
566    /// are set as point-in-time queries.
567    #[must_use]
568    pub fn as_of_tuple(&self) -> Option<(Timestamp, Timestamp)> {
569        match (self.valid_time_as_of, self.transaction_time_as_of) {
570            (Some(vt), Some(tt)) => Some((vt, tt)),
571            _ => None,
572        }
573    }
574
575    /// Check if this context has any temporal constraints.
576    #[must_use]
577    pub fn is_temporal(&self) -> bool {
578        self.valid_time_as_of.is_some()
579            || self.transaction_time_as_of.is_some()
580            || self.valid_time_between.is_some()
581            || self.transaction_time_between.is_some()
582    }
583}

585/// Hints for query optimization.
586#[derive(Debug, Clone, Default, PartialEq)]
587pub struct QueryHints {
588    /// User-provided cardinality estimate
589    pub estimated_cardinality: Option<usize>,
590    /// Force specific index usage
591    pub force_index: Option<IndexHint>,
592    /// Disabled optimizations
593    pub disabled_optimizations: Vec<String>,
594    /// Enable parallel execution
595    pub parallel: bool,
596    /// Include provenance metadata (timestamps, paths, version info) in results
597    pub include_provenance: bool,
598}

600/// Index hint for forcing specific index usage.
601#[derive(Debug, Clone, PartialEq, Eq)]
602pub enum IndexHint {
603    /// Use the HNSW vector index
604    UseVectorIndex,
605    /// Use the temporal index
606    UseTemporalIndex,
607    /// Use current storage (fast path)
608    UseCurrentStorage,
609    /// Use historical storage
610    UseHistoricalStorage,
611    /// Force brute-force vector search (skip HNSW)
612    UseBruteForce,
613}
