1//! Hybrid Query Functions (VS-063)
2//!
3//! This module provides direct hybrid query functions that combine graph traversal
4//! with vector similarity ranking. These are lower-level functions compared to the
5//! query builder API, offering more direct access to hybrid operations.
6//!
7//! # Example Usage
8//!
9//! ```rust,ignore
10//! use aletheiadb::query::hybrid::traverse_and_rank;
11//! use aletheiadb::AletheiaDB;
12//!
13//! let db = AletheiaDB::new();
14//! // ... populate database with nodes and embeddings ...
15//!
16//! // Find neighbors similar to a target embedding
17//! let results = traverse_and_rank(
18//!     &db,
19//!     alice_id,
20//!     "KNOWS",
21//!     &target_embedding,
22//!     10
23//! )?;
24//!
25//! for (node_id, similarity) in results {
26//!     println!("Node {:?}: similarity = {}", node_id, similarity);
27//! }
28//! ```

37/// A candidate node with its similarity score, ordered by similarity (min-heap).
38///
39/// This is used internally for efficient top-k tracking with a BinaryHeap.
40/// The `Ord` implementation is reversed to create a min-heap (lowest similarity at the top).
41#[derive(Debug, Clone, PartialEq)]
42struct ScoredCandidate {
43    node_id: NodeId,
44    similarity: f32,
45}
46
47impl ScoredCandidate {
48    fn new(node_id: NodeId, similarity: f32) -> Self {
49        Self {
50            node_id,
51            similarity,
52        }
53    }
54}
55
56impl Eq for ScoredCandidate {}
57
58impl PartialOrd for ScoredCandidate {
59    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
60        Some(self.cmp(other))
61    }
62}
63
64impl Ord for ScoredCandidate {
65    fn cmp(&self, other: &Self) -> Ordering {
66        // Reverse ordering for min-heap (lower similarity = higher priority to pop)
67        other
68            .similarity
69            .partial_cmp(&self.similarity)
70            .unwrap_or(Ordering::Equal)
71    }
72}

74/// Traverse graph from a starting node and rank results by vector similarity.
75///
76/// This function performs a hybrid graph+vector query:
77/// 1. Traverse outgoing edges from the start node matching the edge label
78/// 2. For each neighbor, compute similarity to the target embedding
79/// 3. Return top-k neighbors ranked by similarity (highest first)
80///
81/// # Arguments
82///
83/// * `db` - Database instance
84/// * `start` - Starting node ID for traversal
85/// * `edge_label` - Edge label to traverse (e.g., "KNOWS", "LINKS_TO")
86/// * `target_embedding` - Target embedding vector to rank by similarity
87/// * `k` - Maximum number of results to return
88///
89/// # Returns
90///
91/// A vector of (NodeId, similarity_score) tuples, sorted by similarity in descending order.
92/// The similarity score is cosine similarity in the range [-1, 1], where higher is more similar.
93///
94/// # Errors
95///
96/// - `Error::Storage(StorageError::NodeNotFound)` if the start node doesn't exist
97/// - `Error::Vector(VectorError::*)` if the target embedding is invalid
98/// - `Error::Storage(*)` if database access fails
99///
100/// # Behavior Notes
101///
102/// - Nodes without the "embedding" property are silently skipped (no error)
103/// - Nodes with embeddings of mismatched dimensions are skipped with a warning
104/// - Cycles in the graph are handled by visiting each node only once
105/// - If fewer than k neighbors exist, returns all available neighbors
106/// - Self-loops are traversed normally (start node can be in results if it has self-edge)
107///
108/// # Examples
109///
110/// ```rust,ignore
111/// // Find Alice's friends most similar to Bob
112/// let bob_embedding = db.get_node(bob_id)?.get_property("embedding")?.as_vector()?;
113/// let similar_friends = traverse_and_rank(&db, alice_id, "KNOWS", bob_embedding, 5)?;
114/// ```
115pub fn traverse_and_rank<G: GraphView + ?Sized>(
116    db: &G,
117    start: NodeId,
118    edge_label: &str,
119    target_embedding: &[f32],
120    k: usize,
121) -> Result<Vec<(NodeId, f32)>> {
122    // Validate target embedding
123    validate_vector(target_embedding)?;
124
125    // Check that start node exists
126    let _start_node = db.get_node(start)?;
127
128    // Get all outgoing edges from start node with matching label
129    let edge_ids = db.get_outgoing_edges_with_label(start, edge_label);
130
131    // Use a min-heap to track top-k candidates efficiently (O(N log k) instead of O(N log N))
132    let mut top_k_heap = BinaryHeap::with_capacity(k);
133
134    // Pre-allocate visited set for cycle detection
135    let mut visited = HashSet::with_capacity(edge_ids.len().min(k * 2));
136
137    for edge_id in edge_ids {
138        // Zero-copy: only get target NodeId, not full Edge (Issue #190)
139        let target_id = db.get_edge_target(edge_id)?;
140
141        // Handle cycles: skip if already visited
142        if visited.contains(&target_id) {
143            continue;
144        }
145        visited.insert(target_id);
146
147        // Get target node
148        let target_node = match db.get_node(target_id) {
149            Ok(node) => node,
150            Err(_) => continue, // Skip if node doesn't exist (shouldn't happen)
151        };
152
153        // Get embedding from node (skip nodes without embeddings)
154        let embedding = match target_node.get_property("embedding") {
155            Some(prop) => match prop.as_vector() {
156                Some(vec) => vec,
157                None => continue, // Property exists but isn't a vector
158            },
159            None => continue, // No embedding property
160        };
161
162        // Compute cosine similarity
163        match cosine_similarity(target_embedding, embedding) {
164            Ok(similarity) => {
165                let candidate = ScoredCandidate::new(target_id, similarity);
166
167                if top_k_heap.len() < k {
168                    // Heap not full yet, add candidate
169                    top_k_heap.push(candidate);
170                } else if let Some(min_candidate) = top_k_heap.peek() {
171                    // Heap is full, only add if better than current minimum
172                    if similarity > min_candidate.similarity {
173                        top_k_heap.pop(); // Remove minimum
174                        top_k_heap.push(candidate); // Add new candidate
175                    }
176                }
177            }
178            Err(_e) => {
179                // Skip nodes with dimension mismatch or invalid embeddings (with warning)
180                #[cfg(feature = "observability")]
181                tracing::warn!(
182                    target_id = %target_id,
183                    error = %_e,
184                    "Skipping node in traverse_and_rank due to incompatible embedding"
185                );
186                continue;
187            }
188        }
189    }
190
191    // Convert heap to sorted vector (highest similarity first)
192    let mut results: Vec<(NodeId, f32)> = top_k_heap
193        .into_iter()
194        .map(|c| (c.node_id, c.similarity))
195        .collect();
196
197    // Sort in descending order (highest similarity first)
198    results.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(Ordering::Equal));
199
200    Ok(results)
201}

203/// Find k most similar nodes at a specific point in time.
204///
205/// This function performs a temporal vector search, finding nodes with embeddings
206/// most similar to the query embedding as they existed at the specified timestamp.
207///
208/// # Arguments
209///
210/// * `db` - Database instance
211/// * `embedding` - Query embedding vector to search for
212/// * `k` - Maximum number of results to return
213/// * `timestamp` - Point in time to query (in microseconds since epoch)
214///
215/// # Returns
216///
217/// A vector of (NodeId, similarity_score) tuples, sorted by similarity in descending order.
218/// The similarity score is cosine similarity in the range [-1, 1], where higher is more similar.
219///
220/// # Errors
221///
222/// - `Error::Vector(VectorError::IndexError)` if temporal vector index is not enabled
223/// - `Error::Vector(VectorError::*)` if the query embedding is invalid
224/// - `Error::Temporal(*)` if the timestamp is invalid or no snapshot exists
225///
226/// # Behavior Notes
227///
228/// - Requires temporal vector indexing to be enabled via `enable_temporal_vector_index()`
229/// - With multiple temporal vector indexes enabled and no property specified,
230///   the alphabetically first indexed property is queried; prefer
231///   `AletheiaDB::find_similar_as_of_in` with an explicit property to
232///   disambiguate
233/// - Returns results from the nearest snapshot at or before the given timestamp
234/// - If no snapshots exist at the given timestamp, returns an error
235/// - Empty results indicate no vectors existed at that timestamp (not an error)
236///
237/// # Examples
238///
239/// ```rust,ignore
240/// use aletheiadb::query::hybrid::find_similar_as_of;
241/// use aletheiadb::core::temporal::time;
242///
243/// // Find documents similar to a query embedding at a specific timestamp
244/// let query_embedding = vec![0.1f32; 384];
245/// let timestamp_2023 = 1672531200000000; // 2023-01-01 in microseconds
246/// let similar_docs = find_similar_as_of(&db, &query_embedding, 10, timestamp_2023)?;
247///
248/// for (node_id, similarity) in similar_docs {
249///     println!("Document {:?} was similar at that time: {:.3}", node_id, similarity);
250/// }
251/// ```
252pub fn find_similar_as_of<G: GraphView + ?Sized>(
253    db: &G,
254    embedding: &[f32],
255    k: usize,
256    timestamp: crate::core::temporal::Timestamp,
257) -> Result<Vec<(NodeId, f32)>> {
258    // Validate embedding
259    validate_vector(embedding)?;
260
261    // Delegate to the GraphView trait method
262    db.find_similar_as_of(embedding, k, timestamp)
263}

283    /// Alice -> Bob (embedding [0.9, 0.1, 0.0, 0.0] - similar to Alice)
284    /// Alice -> Carol (embedding [0.0, 1.0, 0.0, 0.0] - dissimilar to Alice)

297

305            )
306            .expect("Failed to create Bob");
307
308        let carol = db
309            .create_node(
