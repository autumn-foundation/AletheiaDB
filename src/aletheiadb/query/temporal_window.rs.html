1//! Pure interval math for temporal aggregation windows (Issue #3363).
2//!
3//! This module holds the *storage-free* logic behind AQL temporal aggregation
4//! windows: parsing a window boundary timestamp, formatting a boundary back to
5//! RFC 3339, and generating the sequence of tumbling `[start, end)` windows for
6//! a given granularity. Keeping it separate from the executor iterator lets the
7//! subtle interval-edge and calendar arithmetic be unit-tested without a
8//! database.
9//!
10//! # Window semantics (falsifiable contract)
11//!
12//! Windows are half-open `[b_k, b_{k+1})` with `b_0 = start` and
13//! `b_{k+1} = step(b_k)`, generated until `b_k >= end`. The final window's end
14//! is **clamped to `end`**, so the union of all windows equals exactly
15//! `[start, end)` with no overlap and no gap. A version whose `valid_from`
16//! equals a boundary belongs to the window it *starts*
17//! (`b_k <= valid_from < b_{k+1}`).
18//!
19//! Fixed-duration units (minute/hour/day/week) advance by a constant
20//! microsecond delta. Calendar units (month/quarter/year) advance by
21//! chrono UTC calendar arithmetic, so month lengths (28/29/30/31) and leap
22//! years are honored exactly.
23
24use chrono::{DateTime, Months, TimeZone, Utc};
25
26use crate::core::temporal::Timestamp;

28/// Microseconds in one minute.
29const MICROS_PER_MINUTE: i64 = 60 * 1_000_000;
30/// Microseconds in one hour.
31const MICROS_PER_HOUR: i64 = 60 * MICROS_PER_MINUTE;
32/// Microseconds in one day.
33const MICROS_PER_DAY: i64 = 24 * MICROS_PER_HOUR;
34/// Microseconds in one week.
35const MICROS_PER_WEEK: i64 = 7 * MICROS_PER_DAY;
36
37/// Hard cap on the number of windows a single query may generate, guarding
38/// against a caller pairing a huge range with a tiny granularity (which would
39/// otherwise allocate unbounded rows and reconstruct history per window).
40pub const MAX_WINDOWS: usize = 100_000;
41
42/// A window granularity: a positive multiplier and a time unit.
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub struct WindowGranularity {
45    /// The positive multiplier (e.g. `3` in `3 months`).
46    pub count: u32,
47    /// The time unit.
48    pub unit: WindowUnit,
49}
50
51/// A temporal window unit. Fixed-duration units advance by a constant
52/// microsecond delta; calendar units advance via chrono month arithmetic.
53#[derive(Debug, Clone, Copy, PartialEq, Eq)]
54pub enum WindowUnit {
55    /// A minute (fixed 60s).
56    Minute,
57    /// An hour (fixed 3600s).
58    Hour,
59    /// A day (fixed 86400s).
60    Day,
61    /// A week (fixed 7 days).
62    Week,
63    /// A calendar month.
64    Month,
65    /// A calendar quarter (3 calendar months).
66    Quarter,
67    /// A calendar year (12 calendar months).
68    Year,
69}
70
71impl WindowUnit {
72    /// Parse a unit word (case-insensitive, singular or plural). Returns `None`
73    /// for an unknown unit so the caller can raise a structured error.
74    pub fn parse(word: &str) -> Option<Self> {
75        match word.to_ascii_lowercase().as_str() {
76            "minute" | "minutes" => Some(WindowUnit::Minute),
77            "hour" | "hours" => Some(WindowUnit::Hour),
78            "day" | "days" => Some(WindowUnit::Day),
79            "week" | "weeks" => Some(WindowUnit::Week),
80            "month" | "months" => Some(WindowUnit::Month),
81            "quarter" | "quarters" => Some(WindowUnit::Quarter),
82            "year" | "years" => Some(WindowUnit::Year),
83            _ => None,
84        }
85    }
86
87    /// The fixed microsecond delta for one unit, or `None` for calendar units.
88    fn fixed_micros(self) -> Option<i64> {
89        match self {
90            WindowUnit::Minute => Some(MICROS_PER_MINUTE),
91            WindowUnit::Hour => Some(MICROS_PER_HOUR),
92            WindowUnit::Day => Some(MICROS_PER_DAY),
93            WindowUnit::Week => Some(MICROS_PER_WEEK),
94            _ => None,
95        }
96    }
97
98    /// The number of calendar months per unit, or `None` for fixed units.
99    fn calendar_months(self) -> Option<u32> {
100        match self {
101            WindowUnit::Month => Some(1),
102            WindowUnit::Quarter => Some(3),
103            WindowUnit::Year => Some(12),
104            _ => None,
105        }
106    }
107}

109/// A single tumbling window: half-open `[start, end)` in microseconds.
110#[derive(Debug, Clone, Copy, PartialEq, Eq)]
111pub struct Window {
112    /// Inclusive start (microseconds since epoch).
113    pub start_micros: i64,
114    /// Exclusive end (microseconds since epoch), clamped to the range end.
115    pub end_micros: i64,
116}
117
118/// Errors from window construction. Kept as a small enum so the converter can
119/// map each to the right structured [`crate::core::error::QueryError`] variant.
120#[derive(Debug, Clone, PartialEq, Eq)]
121pub enum WindowError {
122    /// The multiplier was zero.
123    ZeroCount,
124    /// `end <= start`.
125    EmptyRange,
126    /// The granularity/range pair would generate more than [`MAX_WINDOWS`].
127    TooManyWindows(usize),
128    /// A boundary timestamp could not be parsed.
129    BadTimestamp(String),
130    /// Calendar arithmetic overflowed the representable timestamp range.
131    CalendarOverflow,
132}
133
134/// Parse a window boundary timestamp: RFC 3339 / ISO-8601 string, or an integer
135/// (or digit string) of microseconds since the Unix epoch. This is a superset
136/// of the existing AQL timestamp handling (which accepted only microseconds),
137/// so previously valid queries are unaffected.
138pub fn parse_boundary_micros(raw: &str) -> Result<i64, WindowError> {
139    let trimmed = raw.trim();
140    // Integer microseconds since epoch (matches existing AQL convention).
141    if let Ok(micros) = trimmed.parse::<i64>() {
142        return Ok(micros);
143    }
144    // RFC 3339 / ISO-8601 with an explicit offset (e.g. `...Z`, `...+00:00`).
145    if let Ok(dt) = DateTime::parse_from_rfc3339(trimmed) {
146        return Ok(dt.with_timezone(&Utc).timestamp_micros());
147    }
148    // Bare date `YYYY-MM-DD` interpreted as UTC midnight.
149    if let Ok(date) = chrono::NaiveDate::parse_from_str(trimmed, "%Y-%m-%d")
150        && let Some(dt) = date.and_hms_opt(0, 0, 0)
151    {
152        return Ok(Utc.from_utc_datetime(&dt).timestamp_micros());
153    }
154    // Naive date-time `YYYY-MM-DDTHH:MM:SS` interpreted as UTC.
155    for fmt in ["%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S"] {
156        if let Ok(dt) = chrono::NaiveDateTime::parse_from_str(trimmed, fmt) {
157            return Ok(Utc.from_utc_datetime(&dt).timestamp_micros());
158        }
159    }
160    Err(WindowError::BadTimestamp(raw.to_string()))
161}

163/// Format a microsecond timestamp as an RFC 3339 UTC string (`Z` suffix,
164/// microsecond precision), mirroring the MCP surface's convention. Falls back to
165/// a `<micros>us` token if the value is outside chrono's representable range.
166pub fn format_rfc3339(micros: i64) -> String {
167    DateTime::<Utc>::from_timestamp_micros(micros)
168        .map(|dt| dt.to_rfc3339_opts(chrono::SecondsFormat::Micros, true))
169        .unwrap_or_else(|| format!("{micros}us"))
170}
171
172/// Advance a boundary by one granularity step. Fixed units add a constant
173/// microsecond delta; calendar units add whole months via chrono UTC
174/// arithmetic (clamping to the last valid day of the target month, chrono's
175/// documented behavior).
176fn step(boundary_micros: i64, gran: WindowGranularity) -> Result<i64, WindowError> {
177    if let Some(delta) = gran.unit.fixed_micros() {
178        return boundary_micros
179            .checked_add(
180                delta
181                    .checked_mul(i64::from(gran.count))
182                    .ok_or(WindowError::CalendarOverflow)?,
183            )
184            .ok_or(WindowError::CalendarOverflow);
185    }
186    let months_per = gran
187        .unit
188        .calendar_months()
189        .expect("non-fixed unit has months");
190    let total_months = months_per
191        .checked_mul(gran.count)
192        .ok_or(WindowError::CalendarOverflow)?;
193    let dt = DateTime::<Utc>::from_timestamp_micros(boundary_micros)
194        .ok_or(WindowError::CalendarOverflow)?;
195    // Preserve the sub-second/day-of-month position; chrono clamps day-of-month
196    // to the target month's length (e.g. Jan 31 + 1 month -> Feb 28/29).
197    let advanced = dt
198        .checked_add_months(Months::new(total_months))
199        .ok_or(WindowError::CalendarOverflow)?;
200    Ok(advanced.timestamp_micros())
201}

203/// Generate the tumbling windows tiling `[start_micros, end_micros)` at the
204/// given granularity. See the module docs for the half-open boundary contract.
205pub fn generate_windows(
206    start_micros: i64,
207    end_micros: i64,
208    gran: WindowGranularity,
209) -> Result<Vec<Window>, WindowError> {
210    if gran.count == 0 {
211        return Err(WindowError::ZeroCount);
212    }
213    if end_micros <= start_micros {
214        return Err(WindowError::EmptyRange);
215    }
216    // Sanity pre-check for fixed units: cheap upper-bound on window count so a
217    // pathological range/granularity fails fast with a structured error rather
218    // than looping to allocate millions of rows.
219    if let Some(delta) = gran.unit.fixed_micros() {
220        let span = (end_micros - start_micros) as i128;
221        let step_span = delta as i128 * i128::from(gran.count);
222        let approx = (span / step_span) + 1;
223        if approx > MAX_WINDOWS as i128 {
224            return Err(WindowError::TooManyWindows(approx as usize));
225        }
226    }
227
228    let mut windows = Vec::new();
229    let mut b = start_micros;
230    while b < end_micros {
231        let next = step(b, gran)?;
232        // Guard against a non-advancing step (should be impossible given
233        // count>0, but defend against any degenerate calendar edge).
234        if next <= b {
235            return Err(WindowError::CalendarOverflow);
236        }
237        let clamped_end = next.min(end_micros);
238        windows.push(Window {
239            start_micros: b,
240            end_micros: clamped_end,
241        });
242        if windows.len() > MAX_WINDOWS {
243            return Err(WindowError::TooManyWindows(windows.len()));
244        }
245        b = next;
246    }
247    Ok(windows)
248}

250/// Convenience: window start as a [`Timestamp`] for point-in-time reads.
251pub fn window_start_timestamp(w: &Window) -> Timestamp {
252    Timestamp::from(w.start_micros)
253}

265        // 2024-01-01T00:00:00Z == 1704067200 seconds == 1704067200_000_000 us.
266        let a = parse_boundary_micros("2024-01-01T00:00:00Z").unwrap();
267        let b = parse_boundary_micros("1704067200000000").unwrap();
268        assert_eq!(a, b);
